1//! Semantic terminal themes and capability-aware color quantization.
2//!
3//! The default theme is a warm, light "cookie" palette: one white surface
4//! (black in the dark table) across the whole frame, espresso-brown text,
5//! and caramel/cinnamon/sage accents. Every TrueColor foreground is chosen for WCAG-AA-or-better
6//! contrast against the white surface; ANSI-256 and ANSI-16 are hue-faithful
7//! degradations. State is never conveyed by color alone (bold/italic/underline
8//! and text markers accompany every semantic color). `Mono` drops all color,
9//! `HighContrast` keeps bright ANSI colors on the terminal's own background.
10//! The dark table re-lights every role; names denote hue, not value.
11
12use ratatui::style::{Color, Modifier, Style};

209    pub fn from_env() -> Self {
210        let requested = std::env::var("COOKIE_THEME").unwrap_or_default();
211        let no_color = std::env::var_os("NO_COLOR").is_some();
212        let term = std::env::var("TERM").unwrap_or_default();
213        Self::from_environment(&requested, no_color, &term, &env_colorterm())
214    }

216    /// Apply a theme kind chosen by the TUI config file while still honoring
217    /// `NO_COLOR`/`TERM=dumb` and detected terminal color capability.
218    pub fn with_kind_from_env(kind: ThemeKind) -> Self {
219        let no_color = std::env::var_os("NO_COLOR").is_some();
220        let term = std::env::var("TERM").unwrap_or_default();
221        Self::from_kind_environment(kind, no_color, &term, &env_colorterm())
222    }
223
224    pub fn from_environment(requested: &str, no_color: bool, term: &str, colorterm: &str) -> Self {
225        let requested_kind = match requested.to_ascii_lowercase().as_str() {
226            "auto" => ThemeKind::Default,
227            "dark" | "dark-roast" | "darkroast" => ThemeKind::Dark,
228            "mono" | "monochrome" => ThemeKind::Mono,
229            "high-contrast" | "high_contrast" | "contrast" => ThemeKind::HighContrast,
230            _ => ThemeKind::Default,
231        };
232        Self::from_kind_environment(requested_kind, no_color, term, colorterm)
233    }
234
235    pub fn from_kind_environment(
236        requested_kind: ThemeKind,
237        no_color: bool,
238        term: &str,
239        colorterm: &str,
240    ) -> Self {
241        let colors =
242            if requested_kind == ThemeKind::Mono || no_color || term.eq_ignore_ascii_case("dumb") {
243                ColorLevel::None
244            } else if requested_kind == ThemeKind::HighContrast {
245                ColorLevel::Ansi16
246            } else if colorterm.eq_ignore_ascii_case("truecolor")
247                || colorterm.eq_ignore_ascii_case("24bit")
248            {
249                ColorLevel::TrueColor
250            } else if term.contains("256color") {
251                ColorLevel::Ansi256
252            } else {
253                ColorLevel::Ansi16
254            };
255        let kind = if colors == ColorLevel::None {
256            ThemeKind::Mono
257        } else {
258            requested_kind
259        };
260        Self::new(kind, colors)
261    }
262
263    pub const fn new(kind: ThemeKind, colors: ColorLevel) -> Self {
264        let colors = match (kind, colors) {
265            (ThemeKind::Mono, _) | (ThemeKind::HighContrast, ColorLevel::None) => ColorLevel::None,
266            (ThemeKind::HighContrast, _) => ColorLevel::Ansi16,
267            (ThemeKind::Default | ThemeKind::Dark, colors) => colors,
268        };
269        Self {
270            key: ThemeKey { kind, colors },
271        }
272    }

285    const fn is_bakery_palette(&self) -> bool {

289    /// Base surface painted beneath the bakery-palette UI. Mono and
290    /// high-contrast themes keep the terminal background.
291    pub fn surface(&self) -> Style {
292        if !self.is_bakery_palette() {
293            return Style::default();
294        }
295        let palette = self.palette();
296        match self.key.colors {
297            ColorLevel::None => Style::default(),
298            ColorLevel::Ansi16 => Style::default()
299                .fg(palette.espresso.ansi16)
300                .bg(palette.cream.ansi16),
301            ColorLevel::Ansi256 => Style::default()
302                .fg(Color::Indexed(palette.espresso.ansi256))
303                .bg(Color::Indexed(palette.cream.ansi256)),
304            ColorLevel::TrueColor => Style::default()
305                .fg(Color::Rgb(
306                    palette.espresso.rgb.0,
307                    palette.espresso.rgb.1,
308                    palette.espresso.rgb.2,
309                ))
310                .bg(Color::Rgb(
311                    palette.cream.rgb.0,
312                    palette.cream.rgb.1,
313                    palette.cream.rgb.2,
314                )),
315        }
316    }

333    /// Keyboard-selected row in pickers, palettes, and the session tree: a
334    /// toasted background with bold espresso text. Reverse video carries the
335    /// selection where no color is available.
336    pub fn selected(&self) -> Style {
337        let style = Style::default().add_modifier(Modifier::BOLD);
338        match self.key.colors {
339            ColorLevel::None => Style::default().add_modifier(Modifier::REVERSED),
340            _ if self.key.kind == ThemeKind::HighContrast => {
341                style.fg(Color::Black).bg(Color::LightYellow)
342            }
343            ColorLevel::Ansi16 => style.fg(Color::Black).bg(self.palette().toasted.ansi16),
344            ColorLevel::Ansi256 | ColorLevel::TrueColor => {
345                let palette = self.palette();
346                let mut style = style;
347                if let Some(foreground) = self.swatch_color(palette.espresso) {
348                    style = style.fg(foreground);
349                }
350                if let Some(background) = self.swatch_color(palette.toasted) {
351                    style = style.bg(background);
352                }
353                style
354            }
355        }
356    }

358    /// Selection surface without foreground or emphasis, for rows whose spans
359    /// retain their own visual hierarchy.
360    pub fn selected_overlay(&self) -> Style {
361        let selected = self.selected();
362        let mut overlay = selected
363            .bg
364            .map_or_else(Style::default, |background| Style::default().bg(background));
365        if selected.add_modifier.contains(Modifier::REVERSED) {
366            overlay = overlay.add_modifier(Modifier::REVERSED);
367        }
368        overlay
369    }

371    /// Mouse-drag text selection: a deeper crust wash than the keyboard
372    /// `selected` row and never bold, so the two selection kinds never read
373    /// alike. Where subtle color exists (ANSI-256, true color) only the
374    /// background is set — patched cells keep their foregrounds, so
375    /// selected code keeps its syntax colors. ANSI-16 and high-contrast
376    /// targets are the one documented exception: their text is always a
377    /// bright color, which a light-cyan wash would swallow, so the pair is
378    /// pinned to black-on-light-cyan. Cruder targets add bold on top of
379    /// reverse video to stay distinct.
380    pub fn text_selection(&self) -> Style {
381        match self.key.colors {
382            ColorLevel::None => Style::default().add_modifier(Modifier::REVERSED | Modifier::BOLD),
383            _ if self.key.kind == ThemeKind::HighContrast => {
384                Style::default().fg(Color::Black).bg(Color::LightCyan)
385            }
386            ColorLevel::Ansi16 => Style::default()
387                .fg(Color::Black)
388                .bg(self.palette().crust.ansi16),
389            ColorLevel::Ansi256 | ColorLevel::TrueColor => self
390                .swatch_color(self.palette().crust)
391                .map_or_else(Style::default, |background| Style::default().bg(background)),
392        }
393    }

395    /// Hover affordance patched over interactive text cells: a quiet glaze
396    /// background — one calm step deeper than the cream surface, never a
397    /// reddish tint. Cruder targets (mono, ANSI-16, high contrast) underline
398    /// as well, so the affordance never relies on a subtle background shift
399    /// alone. Existing foreground colors are preserved by the patch.
400    pub fn hover(&self) -> Style {
401        let quiet_background_only = matches!(
402            (self.key.kind, self.key.colors),
403            (
404                ThemeKind::Default | ThemeKind::Dark,
405                ColorLevel::Ansi256 | ColorLevel::TrueColor
406            )
407        );
408        let style = if quiet_background_only {
409            Style::default()
410        } else {
411            Style::default().add_modifier(Modifier::UNDERLINED)
412        };
413        self.background_color(self.palette().glaze, Some(Color::DarkGray))
414            .map_or(style, |background| style.bg(background))
415    }

463    /// One approval decision button. `active` (hover) fills the button with
464    /// a decision-tinted background; the border/label color and the button
465    /// glyphs carry the meaning, never color alone.
466    pub fn decision(&self, tone: DecisionTone, active: bool) -> Style {
467        if self.key.colors == ColorLevel::None {
468            let mut style = Style::default().add_modifier(Modifier::BOLD);
469            if active {
470                style = style.add_modifier(Modifier::REVERSED);
471            }
472            return style;
473        }
474        if self.key.kind == ThemeKind::HighContrast {
475            let foreground = match tone {
476                DecisionTone::Allow => Color::LightGreen,
477                DecisionTone::Deny => Color::LightRed,
478                DecisionTone::Neutral => Color::White,
479            };
480            let mut style = Style::default().fg(foreground).add_modifier(Modifier::BOLD);
481            if active {
482                style = style.add_modifier(Modifier::REVERSED);
483            }
484            return style;
485        }
486        if self.key.colors == ColorLevel::Ansi16 {
487            let palette = self.palette();
488            if active {
489                let background = match tone {
490                    DecisionTone::Allow => palette.allow_tint.ansi16,
491                    DecisionTone::Deny => palette.deny_tint.ansi16,
492                    DecisionTone::Neutral => palette.neutral_tint.ansi16,
493                };
494                return Style::default()
495                    .fg(Color::Black)
496                    .bg(background)
497                    .add_modifier(Modifier::BOLD);
498            }
499            let foreground = match tone {
500                DecisionTone::Allow => palette.basil.ansi16,
501                DecisionTone::Deny => palette.cranberry.ansi16,
502                DecisionTone::Neutral => palette.cocoa.ansi16,
503            };
504            return Style::default().fg(foreground).add_modifier(Modifier::BOLD);
505        }
506        let palette = self.palette();
507        let (foreground, tint) = match tone {
508            DecisionTone::Allow => (palette.basil, palette.allow_tint),
509            DecisionTone::Deny => (palette.cranberry, palette.deny_tint),
510            DecisionTone::Neutral => (palette.cocoa, palette.neutral_tint),
511        };
512        let mut style = Style::default().add_modifier(Modifier::BOLD);
513        let foreground = if active && self.key.kind == ThemeKind::Dark {
514            palette.cream
515        } else {
516            foreground
517        };
518        if let Some(foreground) = self.swatch_color(foreground) {
519            style = style.fg(foreground);
520        }
521        if active && let Some(background) = self.swatch_color(tint) {
522            style = style.bg(background);
523        }
524        style
525    }

618    /// Inline code in assistant Markdown: a distinct warm terracotta
619    /// foreground, never a background in the default theme — the source
620    /// backticks stay visible, and the bold modifier carries the distinction
621    /// in mono terminals. High contrast keeps its inverse-video chip.
622    /// Inline code. Bakery palettes set it in a muted body-text colour on the
623    /// code-block parchment, in regular weight, which is marker enough that
624    /// the renderer drops the backticks; other themes keep bold text and
625    /// the backticks.
626    /// Inline code in the bakery palettes is text, not a chip: a close
627    /// cousin of the body colour (cocoa beside espresso) with no tint, so it
628    /// never breaks a line into padded blocks. Sixteen colours have no such
629    /// near shade and keep the body colour on the code tint; colourless and
630    /// high-contrast themes keep a weight or chip cue.
631    pub fn inline_code(&self) -> Style {
632        if self.inline_code_is_coloured() {
633            return self.semantic(self.palette().cocoa, Color::Black, Modifier::empty());
634        }
635        if let Some(background) = self.code_background() {
636            return self
637                .semantic(self.palette().espresso, Color::Black, Modifier::empty())
638                .bg(background);
639        }
640        let foreground = self.semantic(self.palette().terracotta, Color::Black, Modifier::BOLD);
641        let background = match self.key.colors {
642            ColorLevel::None => None,
643            _ if self.key.kind == ThemeKind::HighContrast => Some(Color::LightYellow),
644            ColorLevel::Ansi16 | ColorLevel::Ansi256 | ColorLevel::TrueColor => None,
645        };
646        background.map_or(foreground, |background| foreground.bg(background))
647    }

650    /// chip nor backticks.

710    pub fn quantize_rgb(&self, red: u8, green: u8, blue: u8) -> Option<Color> {
711        match self.key.colors {
712            ColorLevel::None => None,
713            _ if self.key.kind == ThemeKind::HighContrast => {
714                Some(nearest_high_contrast(red, green, blue))
715            }
716            ColorLevel::TrueColor => Some(Color::Rgb(red, green, blue)),
717            ColorLevel::Ansi256 => Some(Color::Indexed(
718                self.palette()
719                    .swatch_for_rgb((red, green, blue))
720                    .map_or_else(|| rgb_to_ansi256(red, green, blue), |swatch| swatch.ansi256),
721            )),
722            ColorLevel::Ansi16 => Some(
723                self.palette()
724                    .swatch_for_rgb((red, green, blue))
725                    .map_or_else(|| nearest_ansi16(red, green, blue), |swatch| swatch.ansi16),
726            ),
727        }
728    }

730    fn swatch_color(&self, swatch: Swatch) -> Option<Color> {
731        match self.key.colors {
732            ColorLevel::None => None,
733            _ if self.key.kind == ThemeKind::HighContrast => Some(nearest_high_contrast(
734                swatch.rgb.0,

738            ColorLevel::Ansi16 => Some(swatch.ansi16),
739            ColorLevel::Ansi256 => Some(Color::Indexed(swatch.ansi256)),
740            ColorLevel::TrueColor => Some(Color::Rgb(swatch.rgb.0, swatch.rgb.1, swatch.rgb.2)),
741        }
742    }

744    /// A background color honoring kind/level fallbacks from its palette
745    /// swatch. High contrast paints only explicitly requested backgrounds.
746    fn background_color(&self, swatch: Swatch, high_contrast: Option<Color>) -> Option<Color> {
747        match self.key.colors {
748            ColorLevel::None => None,
749            _ if self.key.kind == ThemeKind::HighContrast => high_contrast,
750            ColorLevel::Ansi16 => Some(swatch.ansi16),
751            ColorLevel::Ansi256 => Some(Color::Indexed(swatch.ansi256)),
752            ColorLevel::TrueColor => Some(Color::Rgb(swatch.rgb.0, swatch.rgb.1, swatch.rgb.2)),

756    fn semantic(&self, swatch: Swatch, high_contrast: Color, modifier: Modifier) -> Style {
757        let mut modifier = match self.key.kind {
758            ThemeKind::HighContrast => modifier | Modifier::BOLD,
759            ThemeKind::Default | ThemeKind::Dark | ThemeKind::Mono => modifier,
760        };
761        if self.is_bakery_palette() {
762            modifier.remove(Modifier::DIM);
763        }
764        let color = match self.key.colors {
765            ColorLevel::None => None,
766            _ if self.key.kind == ThemeKind::HighContrast => Some(high_contrast),
767            ColorLevel::Ansi16 => Some(swatch.ansi16),
768            ColorLevel::Ansi256 => Some(Color::Indexed(swatch.ansi256)),
769            ColorLevel::TrueColor => Some(Color::Rgb(swatch.rgb.0, swatch.rgb.1, swatch.rgb.2)),
770        };
771        color.map_or_else(
772            || Style::default().add_modifier(modifier),
773            |color| Style::default().fg(color).add_modifier(modifier),
774        )
775    }

778/// `COLORTERM`, or `truecolor` for a terminal known to render 24-bit colour
779/// that does not advertise it there: Windows Terminal (`WT_SESSION`, which it
780/// also shares into WSL) and a few `TERM_PROGRAM` identities.
781fn env_colorterm() -> String {
782    let colorterm = std::env::var("COLORTERM").unwrap_or_default();
783    let term_program = std::env::var("TERM_PROGRAM").unwrap_or_default();
784    colorterm_with_hints(
785        colorterm,
786        std::env::var_os("WT_SESSION").is_some(),
787        &term_program,
788    )
789}
790
791fn colorterm_with_hints(colorterm: String, windows_terminal: bool, term_program: &str) -> String {
792    let known_truecolor =
793        windows_terminal || matches!(term_program, "vscode" | "iTerm.app" | "WezTerm" | "ghostty");
794    if colorterm.is_empty() && known_truecolor {
795        "truecolor".to_owned()
796    } else {
797        colorterm
798    }
799}

801fn nearest_high_contrast(red: u8, green: u8, blue: u8) -> Color {
802    const COLORS: [(Color, (u8, u8, u8)); 7] = [
803        (Color::LightRed, (255, 0, 0)),
804        (Color::LightGreen, (0, 255, 0)),
805        (Color::LightYellow, (255, 255, 0)),
806        (Color::LightBlue, (0, 0, 255)),
807        (Color::LightMagenta, (255, 0, 255)),
808        (Color::LightCyan, (0, 255, 255)),
809        (Color::White, (255, 255, 255)),
810    ];
811    nearest_color(red, green, blue, &COLORS)
812}
813
814fn rgb_to_ansi256(red: u8, green: u8, blue: u8) -> u8 {
815    let component = |value: u8| ((u16::from(value) * 5 + 127) / 255) as u8;
816    16 + 36 * component(red) + 6 * component(green) + component(blue)
817}
818
819fn nearest_ansi16(red: u8, green: u8, blue: u8) -> Color {
820    const COLORS: [(Color, (u8, u8, u8)); 16] = [
821        (Color::Black, (0, 0, 0)),
822        (Color::Red, (128, 0, 0)),
823        (Color::Green, (0, 128, 0)),
824        (Color::Yellow, (128, 128, 0)),
825        (Color::Blue, (0, 0, 128)),
826        (Color::Magenta, (128, 0, 128)),
827        (Color::Cyan, (0, 128, 128)),
828        (Color::Gray, (192, 192, 192)),
829        (Color::DarkGray, (128, 128, 128)),
830        (Color::LightRed, (255, 0, 0)),
831        (Color::LightGreen, (0, 255, 0)),
832        (Color::LightYellow, (255, 255, 0)),
833        (Color::LightBlue, (0, 0, 255)),
834        (Color::LightMagenta, (255, 0, 255)),
835        (Color::LightCyan, (0, 255, 255)),
836        (Color::White, (255, 255, 255)),
837    ];
838    nearest_color(red, green, blue, &COLORS)
839}
840
841fn nearest_color(red: u8, green: u8, blue: u8, colors: &[(Color, (u8, u8, u8))]) -> Color {
842    colors
843        .iter()
844        .min_by_key(|(_, (candidate_red, candidate_green, candidate_blue))| {
845            let red = i32::from(red) - i32::from(*candidate_red);
846            let green = i32::from(green) - i32::from(*candidate_green);
847            let blue = i32::from(blue) - i32::from(*candidate_blue);
848            red * red + green * green + blue * blue
849        })
850        .map_or(Color::White, |(color, _)| *color)
851}
