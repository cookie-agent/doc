10/// Preserve LF/tab, remove terminal/format controls, and truncate at UTF-8 boundaries.
11pub fn sanitize(value: &str, maximum: usize) -> String {
12    let mut text = if let Ok(json) = serde_json::from_str::<Value>(value) {
13        serde_json::to_string_pretty(&json).expect("JSON value serializes")
14    } else {
15        value.to_owned()
16    };
17    static CONTROLS: LazyLock<Regex> =
18        LazyLock::new(|| Regex::new(r"[\p{Cc}\p{Cf}&&[^\n\t]]").unwrap());
19    text = CONTROLS.replace_all(&text, " ").into_owned();
20    if text.trim().is_empty() {
21        text = "No error detail was supplied".into();
22    }
23    const MARKER: &str = "\n[diagnostic truncated]";
24    if text.len() > maximum {
25        let mut end = maximum.saturating_sub(MARKER.len());
26        while !text.is_char_boundary(end) {
27            end -= 1;
28        }
29        text.truncate(end);
30        text.push_str(&MARKER[..MARKER.len().min(maximum)]);
31    }
32    text
33}

35pub fn detail(value: &str) -> DiagnosticText {
36    DiagnosticText::new(sanitize(value, DiagnosticText::MAX_BYTES)).expect("sanitized diagnostic")
37}

39pub fn headline(value: &str) -> SafeErrorMessage {
40    SafeErrorMessage::new(sanitize(value, SafeErrorMessage::MAX_BYTES).replace(['\n', '\t'], " "))
41        .expect("sanitized headline")
42}

44pub fn error_chain(error: &(dyn std::error::Error + 'static)) -> String {
45    let mut parts = vec![error.to_string()];
46    let mut source = error.source();
47    for _ in 0..16 {
48        let Some(cause) = source else {
49            break;
50        };
51        let text = cause.to_string();
52        if !parts.last().is_some_and(|last| last.contains(&text)) {
53            parts.push(text);
54        }
55        source = cause.source();
56    }
57    detail(&parts.join(": ")).to_string()
58}

60pub fn model(error: &ModelErrorSummary) -> String {
61    let mut text = error.message.to_string();
62    if let Some(status) = error.http_status {
63        text.push_str(&format!(" · HTTP {status}"));
64    }
65    if let Some(code) = &error.vendor_code {
66        text.push_str(&format!(" · code {code}"));
67    }
68    if let Some(id) = &error.request_id {
69        text.push_str(&format!(" · request {id}"));
70    }
71    text.push_str(&format!(
72        " · stage {:?} · retryable {} · {} bytes received",
73        error.stage, error.retryable, error.bytes_received
74    ));
75    if let Some(delay) = error.retry_after_ms {
76        text.push_str(&format!(" · retry after {delay}ms"));
77    }
78    if let Some(body) = &error.response_body {
79        text.push_str(&format!("\nResponse body:\n{}", detail(body.as_str())));
80    }
81    text
82}

84pub fn run_error(
85    error: &SafeErrorMessage,
86    model_error: Option<&ModelErrorSummary>,
87    model: Option<&crate::ResolvedModelRef>,
88) -> String {
89    let mut text = error.to_string();
90    if let Some(model) = model {
91        text.push_str(&format!(
92            "\nModel: {} (provider {}, adapter {:?})",
93            model.selection.model, model.provider_id, model.adapter_id
94        ));
95    }
96    if let Some(model_error) = model_error {
97        text.push_str(&format!("\n{}", self::model(model_error)));
98    }
99    text
100}

105        None => failure.message.to_string(),

127        backend(to),

132pub fn tool(termination: &crate::ToolCallTermination) -> String {
133    if let Some(error) = &termination.error {
134        // Preserve both status/context and final cause when adding the prefix.
135        return excerpt(
136            &format!("{}: {}", error.code, error.message),
137            DiagnosticText::MAX_BYTES,
138        );
139    }
140    if let Some(result) = &termination.result {
141        return tool_result(result);
142    }
143    format!(
144        "tool ended with {:?}; no diagnostic supplied",
145        termination.outcome
146    )
147}

149pub fn tool_result(result: &crate::PersistedToolResult) -> String {
150    let mut parts = vec![excerpt(result.title.as_str(), 256)];
151    // Give status/source context and each output field separate budgets.
152    for key in [
153        "status",
154        "exit_code",
155        "exitCode",
156        "signal",
157        "source",
158        "path",
159        "filePath",
160        "error",
161        "message",
162    ] {
163        if let Some(value) = result.metadata.get(key).filter(|value| !value.is_null()) {
164            let value = value
165                .as_str()
166                .map_or_else(|| value.to_string(), str::to_owned);
167            parts.push(format!("{key}: {}", excerpt(&value, 160)));
168        }
169    }
170    let mut sections = tool_sections(result);
171    // Named stream headings come from the runtime in declaration order.
172    sections.sort_by_key(|(name, _)| match *name {
173        "stderr" | "diagnostics" | "errors" | "error" => 0,
174        "stdout" => 2,
175        _ => 1,
176    });
177    let used = parts.iter().map(|part| part.len() + 1).sum::<usize>();
178    let budget = DiagnosticText::MAX_BYTES.saturating_sub(used);
179    if sections.is_empty() {
180        parts.push(excerpt(
181            result
182                .display
183                .as_deref()
184                .filter(|value| !value.trim().is_empty())
185                .unwrap_or("Tool reported failure without a reason"),
186            budget.saturating_sub(1),
187        ));
188    } else {
189        let per_section = budget / sections.len();
190        for (name, value) in sections {
191            let heading = format!("[{name}]\n");
192            parts.push(format!(
193                "{heading}{}",
194                excerpt(value, per_section.saturating_sub(heading.len() + 1))
195            ));
196        }
197    }
198    detail(&parts.join("\n")).to_string()
199}

201fn tool_sections(result: &crate::PersistedToolResult) -> Vec<(&str, &str)> {
202    let mut sections = Vec::new();
203    if let Some(retained) = &result.retained_output {
204        let mut end = result.output.len();
205        for (index, stream) in retained.streams.iter().enumerate().rev() {
206            let Some(name) = stream.name.as_deref() else {
207                break;
208            };
209            let marker = if index == 0 {
210                format!("[{name}]\n")
211            } else {
212                format!("\n\n[{name}]\n")
213            };
214            let Some(start) = result.output[..end].rfind(&marker) else {
215                sections.clear();
216                break;
217            };
218            let value = &result.output[start + marker.len()..end];
219            if !value.trim().is_empty() {
220                sections.push((name, value));
221            }
222            end = start;
223        }
224    }
225    if sections.is_empty() && !result.output.trim().is_empty() {
226        sections.push(("output", &result.output));
227    }
228    sections
229}

231// Keep the head and tail: final causes often follow a noisy stream.
232fn excerpt(value: &str, maximum: usize) -> String {
233    let clean = sanitize(value, usize::MAX);
234    if clean.len() <= maximum {
235        return clean;
236    }
237    const MARKER: &str = "\n[diagnostic truncated]\n";
238    if maximum < MARKER.len() {
239        return sanitize(&clean, maximum);
240    }
241    let mut head = (maximum - MARKER.len()) / 2;
242    let mut tail = clean.len() - (maximum - MARKER.len() - head);
243    while !clean.is_char_boundary(head) {
244        head -= 1;
245    }
246    while !clean.is_char_boundary(tail) {
247        tail += 1;
248    }
249    format!("{}{MARKER}{}", &clean[..head], &clean[tail..])
250}

252/// Only diagnostic fields are selected from RPC data. Debug remains redacted.
253pub fn rpc(error: &crate::JsonRpcError) -> String {
254    let mut text = format!("{} (RPC {})", error.message, error.code);
255    if let Some(Value::Object(data)) = &error.data {
256        for key in [
257            "message",
258            "reason",
259            "detail",
260            "cause",
261            "code",
262            "path",
263            "line",
264            "column",
265            "request_id",
266            "expected_revision",
267            "found_revision",
268        ] {
269            if let Some(value) = data.get(key).filter(|value| !value.is_null()) {
270                let value = value.as_str().map_or_else(
271                    || detail(&value.to_string()).to_string(),
272                    |value| detail(value).to_string(),
273                );
274                text.push_str(&format!("\n{key}: {value}"));
275            }
276        }
277    }
278    detail(&text).to_string()
279}
