105            .skills_runtime
106            .grants

108            .unwrap_or_else(|poisoned| poisoned.into_inner())

112                    .values()
113                    .flat_map(|grant| {
114                        debug_assert!(grant.source.starts_with("skill:"));

122    pub fn list_skills(&self, session: SessionId) -> Result<SkillsListResult, EngineError> {
123        let projection = self.inner.store.get(session)?;
124        let policy = permissions::governing_agent_for_skills(&projection);
125        let grants = self.skill_grants_for_session(session);
126        let mut skills = self
127            .inner
128            .skills
129            .discoveries()
130            .iter()
131            .filter_map(|discovery| {
132                let skill = self.inner.skills.get(&discovery.name)?;
133                let (effect, visible) = model_skill_visibility(
134                    &policy,
135                    &projection.permission_overlay,
136                    grants.as_ref(),
137                    self.inner.store.cwd(),
138                    &discovery.name,
139                    skill,
140                );
141                let mut descriptor =
142                    skill_descriptor(skill, discovery.precedence_winner, effect, visible);
143                descriptor.location = discovery.path.to_string_lossy().into_owned();
144                descriptor.source = discovery.source;
145                Some(descriptor)
146            })
147            .collect::<Vec<_>>();
148        skills.sort_by(|left, right| {
149            left.name
150                .cmp(&right.name)
151                .then_with(|| left.location.cmp(&right.location))
152        });
153        Ok(SkillsListResult { skills })
154    }

250    pub async fn invoke_skill(
251        &self,
252        session: SessionId,
253        run: Option<cookie_agent_protocol::RunId>,
254        name: &str,
255        args: &str,
256        permission_preapproved: bool,
257    ) -> Result<SkillInvocation, EngineError> {
258        let projection = self.inner.store.get(session)?;
259        let policy = permissions::governing_agent_for_skills(&projection);
260        let skill = self.skill(name)?.clone();
261        if !permission_preapproved && !skill.frontmatter.user_invocable {
262            return Err(EngineError::Permission(format!(
263                "skill `{name}` is not user-invocable"
264            )));
265        }
266        let effect = permissions::effective_permission_with_overlay(
267            &policy,
268            Some(&projection.permission_overlay),
269            PermissionAction::Skill,
270            name,
271            self.inner.store.cwd(),
272        )
273        .0;
274        if !permission_preapproved && effect != PermissionEffect::Allow {
275            return Err(EngineError::Permission(format!(
276                "skill `{name}` requires {}",
277                if effect == PermissionEffect::Deny {
278                    "permission that is denied"
279                } else {
280                    "approval"
281                }
282            )));
283        }
284        let plan = prepared_skill_invocation(&skill, args);
285        self.install_prepared_skill(session, run, &plan).await
286    }

288    pub(super) async fn install_prepared_skill(
289        &self,
290        session: SessionId,
291        run: Option<RunId>,
292        plan: &PreparedSkillInvocation,
293    ) -> Result<SkillInvocation, EngineError> {
294        let projection = self.inner.store.get(session)?;
295        let name = &plan.name;
296        let duplicate = projection.log.event_snapshot().iter().any(|event| {
297            matches!(
298                &event.payload,
299                EventPayload::SkillLoaded { name: prior_name, rendered_body, .. }
300                    if prior_name == name && rendered_body.as_bytes() == plan.rendered.as_bytes()
301            )
302        });
303        let event = if duplicate {
304            EventPayload::SkillInvocationNoted { name: name.clone() }
305        } else {
306            EventPayload::SkillLoaded {
307                name: name.clone(),
308                rendered_body: plan.rendered.clone(),
309                source_path: plan.source_path.clone(),
310                args: plan.args.clone(),
311                base_dir: plan.base_dir.clone(),
312                supporting_files: plan.supporting_files.clone(),
313            }
314        };
315        self.append(session, run, super::event_origin("engine:skills"), event)
316            .await?;
317        let mut overlays = self
318            .inner
319            .skills_runtime
320            .grants
321            .lock()
322            .unwrap_or_else(|poisoned| poisoned.into_inner());
323        let per_skill = overlays.entry(session).or_default();
324        if plan.grants.is_empty() {
325            per_skill.remove(name);
326        } else {
327            per_skill.insert(
328                name.clone(),
329                SkillGrantOverlay {
330                    source: format!("skill:{name}"),
331                    overlay: SessionPermissionOverlay {
332                        rules: plan.grants.clone(),
333                    },
334                },
335            );
336        }
337        if per_skill.is_empty() {
338            overlays.remove(&session);
339        }
340        drop(overlays);
341        if let Some(model) = &plan.model {
342            self.inner
343                .skills_runtime
344                .models
345                .lock()
346                .unwrap_or_else(|poisoned| poisoned.into_inner())
347                .insert(session, model.clone());
348        }
349        Ok(SkillInvocation {
350            name: name.clone(),
351            rendered: plan.rendered.clone(),
352            context: plan.context,
353        })
354    }

386    pub(super) async fn execute_direct_skill(
387        &self,
388        active: Arc<ActiveRun>,
389        run: RunId,
390        name: String,
391        args: String,
392    ) -> Result<(), EngineError> {
393        self.get_user_skill(active.session, &name, &args)?;
394        let call_id = ToolCallId::new_v7();
395        self.inner
396            .skills_runtime
397            .direct_calls
398            .lock()
399            .unwrap_or_else(|poisoned| poisoned.into_inner())
400            .insert(call_id);
401        let result = self
402            .execute_prepared_call(
403                active,
404                run,
405                ToolCall {
406                    id: call_id,
407                    name: "skill".into(),
408                    arguments: serde_json::json!({"name": name, "args": args}),
409                },
410            )
411            .await;
412        self.inner
413            .skills_runtime
414            .direct_calls
415            .lock()
416            .unwrap_or_else(|poisoned| poisoned.into_inner())
417            .remove(&call_id);
418        result.map(|_| ()).map_err(|error| {
419            if error.to_string().contains("tool_denied") {
420                EngineError::Permission(error.to_string())
421            } else {
422                error
423            }
424        })
425    }

427    pub async fn execute_skill_fork(
428        &self,
429        session: SessionId,
430        run: RunId,
431        call_id: ToolCallId,
432        name: &str,
433        args: &str,
434    ) -> Result<cookie_agent_protocol::PersistedToolResult, EngineError> {
435        let plan = self.prepare_skill_invocation(name, args)?;
436        let agent_type = self
437            .delegate_targets(session)?
438            .into_iter()
439            .next()
440            .ok_or_else(|| {
441                EngineError::ToolFailed("fork skill requires an available delegation target".into())
442            })?;
443        self.inner
444            .skills_runtime
445            .pending_forks
446            .lock()
447            .unwrap_or_else(|poisoned| poisoned.into_inner())
448            .insert(call_id, plan.clone());
449        let active = self
450            .inner
451            .sessions
452            .active
453            .lock()
454            .unwrap_or_else(|poisoned| poisoned.into_inner())
455            .get(&run)
456            .cloned()
457            .filter(|active| active.session == session)
458            .ok_or(EngineError::MissingRun(run))?;
459        let result = self
460            .execute_prepared_call(
461                active,
462                run,
463                ToolCall {
464                    id: call_id,
465                    name: "delegate_subagent".into(),
466                    arguments: serde_json::json!({
467                        "description": format!("Skill {name}"),
468                        "prompt": format!("Apply the staged skill `{name}`."),
469                        "agent_type": agent_type,
470                        "background": false,
471                        "resume_session_id": null,
472                        "inherit_context": false,
473                    }),
474                },
475            )
476            .await;
477        self.inner
478            .skills_runtime
479            .pending_forks
480            .lock()
481            .unwrap_or_else(|poisoned| poisoned.into_inner())
482            .remove(&call_id);
483        result
484    }

486    async fn execute_prepared_call(
487        &self,
488        active: Arc<ActiveRun>,
489        run: RunId,
490        call: ToolCall,
491    ) -> Result<cookie_agent_protocol::PersistedToolResult, EngineError> {
492        let binding = active
493            .policy
494            .selected_suffix
495            .first()
496            .ok_or(EngineError::NoRunnableModel)?;
497        let turn_context = Arc::new(TurnAgentContext {
498            agent: active.policy.agent.agent.clone(),
499            model: binding.selection.model.clone(),
500            adapter: crate::policy::wire_adapter(binding.descriptor.adapter_id.as_str()),
501            adapter_family: crate::policy::adapter_family(binding.descriptor.adapter_id.as_str()),
502            capabilities: active
503                .policy
504                .model_capabilities(binding)
505                .ok_or(EngineError::NoRunnableModel)?,
506        });
507        let prepared = self
508            .prepare_tool_call(
509                active.session,
510                run,
511                call,
512                &active.policy,
513                turn_context.clone(),
514            )
515            .await;
516        self.execute_tool(active, run, prepared, turn_context)
517            .await
518            .map_err(|failure| EngineError::ToolFailed(failure.message))
519    }

635    pub(crate) fn compose_skill_listing(
636        &self,
637        session: SessionId,
638        policy: &mut FrozenRunPolicy,
639        prospective_grants: Option<&SessionPermissionOverlay>,
640    ) -> Result<(), EngineError> {
641        let projection = self.inner.store.get(session)?;
642        let visible = self
643            .inner
644            .skills
645            .skills()
646            .filter_map(|(name, skill)| {
647                model_skill_visibility(
648                    &policy.agent,
649                    &projection.permission_overlay,
650                    prospective_grants,
651                    self.inner.store.cwd(),
652                    name,
653                    skill,
654                )
655                .1
656                .then_some(skill)
657            })
658            .collect::<Vec<_>>();
659        if visible.is_empty() {
660            return Ok(());
661        }
662        let context_tokens = policy.selected_suffix[0]
663            .descriptor
664            .capabilities
665            .limits
666            .context
667            .unwrap_or(8_192);
668        let listing = cookie_agent_config::render_available_skills(visible, context_tokens)
669            .map_err(|error| EngineError::Config(Box::new(error)))?;
670        push_prompt_block(&mut policy.agent.composed_prompt, &listing);
671        policy.agent.prompt_fingerprint =
672            Sha256Digest::of_bytes(policy.agent.composed_prompt.as_bytes());
673        let mut document_fingerprint = policy
674            .agent
675            .document_fingerprint
676            .as_str()
677            .as_bytes()
678            .to_vec();
679        document_fingerprint.extend_from_slice(listing.as_bytes());
680        policy.agent.document_fingerprint = Sha256Digest::of_bytes(&document_fingerprint);
681        Ok(())
682    }

705fn model_skill_visibility(
706    policy: &cookie_agent_protocol::AgentSnapshot,
707    overlay: &SessionPermissionOverlay,
708    grants: Option<&SessionPermissionOverlay>,
709    workspace: &Path,
710    name: &str,
711    skill: &SkillDocument,
712) -> (PermissionEffect, bool) {
713    let effect = permissions::effective_permission_with_overlay(
714        policy,
715        Some(overlay),
716        PermissionAction::Skill,
717        name,
718        workspace,
719    )
720    .0;
721    let action_visible = permissions::PermissionPipeline::tool_visible_with_grants(
722        policy,
723        Some(overlay),
724        grants,
725        "skill",
726        workspace,
727    );
728    (
729        effect,
730        action_visible
731            && effect != PermissionEffect::Deny
732            && !skill.frontmatter.disable_model_invocation,
733    )
734}

736fn prepared_skill_invocation(skill: &SkillDocument, args: &str) -> PreparedSkillInvocation {
737    PreparedSkillInvocation {
738        name: skill.frontmatter.name.clone(),
739        args: args.to_owned(),
740        rendered: render_skill(skill, args),
741        source_path: skill.path.to_string_lossy().into_owned(),
742        base_dir: skill.base_dir.to_string_lossy().into_owned(),
743        supporting_files: skill
744            .supporting_files
745            .iter()
746            .map(|path| path.to_string_lossy().into_owned())
747            .collect(),
748        grants: skill
749            .frontmatter
750            .allowed_tools
751            .iter()
752            .map(|grant| PermissionRule {
753                action: grant.action,
754                resource: grant.pattern.clone(),
755                effect: PermissionEffect::Allow,
756            })
757            .collect(),
758        model: skill.frontmatter.model.clone(),
759        context: skill.frontmatter.context,
760    }
761}

779fn render_skill(skill: &SkillDocument, args: &str) -> String {
780    let positional = args.split_whitespace().collect::<Vec<_>>();
781    let mut variables = BTreeMap::new();
782    variables.insert("ARGUMENTS".to_owned(), args.to_owned());
783    variables.insert(
784        "{COOKIE_SKILL_DIR}".to_owned(),
785        skill.base_dir.to_string_lossy().into_owned(),
786    );
787    for (index, value) in positional.into_iter().enumerate() {
788        variables.insert((index + 1).to_string(), value.to_owned());
789    }
790    let mut output = String::new();
791    let bytes = skill.body.as_bytes();
792    let mut index = 0;
793    while index < bytes.len() {
794        if bytes[index] == b'$' {
795            let rest = &skill.body[index + 1..];
796            let key = if rest.starts_with("{COOKIE_SKILL_DIR}") {
797                Some("{COOKIE_SKILL_DIR}")
798            } else if rest.starts_with("ARGUMENTS") {
799                Some("ARGUMENTS")
800            } else {
801                let digits = rest.bytes().take_while(u8::is_ascii_digit).count();
802                (digits > 0).then_some(&rest[..digits])
803            };
804            if let Some(key) = key {
805                output.push_str(variables.get(key).map_or("", String::as_str));
806                index += 1 + key.len();
807                continue;
808            }
809        }
810        let character = skill.body[index..]
811            .chars()
812            .next()
813            .expect("character boundary");
814        output.push(character);
815        index += character.len_utf8();
816    }
817    output
818}
