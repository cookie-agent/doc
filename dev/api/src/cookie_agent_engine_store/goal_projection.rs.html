63/// What a log's producer fold keeps of a message that settled, once it lets go of
64/// the message (`docs/site/specs/forever-sessions.md` section 12): a message
65/// consumed and recorded, or discarded, whose run ended, with no claim on it.
66/// Nothing names such a message again but a producer's retry of its key, so this
67/// keeps that, and whether a goal's reminder was consumed. Both are kept for as
68/// long as the log, each 8 bytes and an ID.
69#[derive(Clone, Debug, Default, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
70pub struct SettledMessages {
71    /// By the digest of each message's owner and idempotency key, its ID.
72    #[serde(with = "crate::events::sorted::map")]
73    keys: HashMap<String, ProducerMessageId>,
74    /// The goals a consumed reminder of continued.
75    #[serde(with = "crate::events::sorted::set")]
76    continued_goals: HashSet<GoalId>,
77}
78
79/// The digest [`SettledMessages`] keeps a message's key as.
80fn key_digest(owner: &ProducerOwner, key: &ProducerIdempotencyKey) -> String {
81    use sha2::Digest as _;
82    let mut digest = sha2::Sha256::new();
83    digest.update(serde_json::to_vec(owner).unwrap_or_default());
84    digest.update([0]);
85    digest.update(key.as_str().as_bytes());
86    digest.finalize()[..8]
87        .iter()
88        .map(|byte| format!("{byte:02x}"))
89        .collect()
90}

94    #[must_use]
95    pub fn goal_is_open(&self) -> bool {
96        self.goal
97            .as_ref()
98            .is_some_and(|goal| matches!(goal.status, GoalStatus::Active | GoalStatus::Paused))

101    pub fn next_reminder_kind(&self, goal_id: GoalId) -> GoalReminderKind {
102        // Acceptance and claims are retractable; only committed input coverage introduces a goal.
103        if self.settled.continued_goals.contains(&goal_id)
104            || self.messages.iter().any(|message| {
105                message.consumed
106                    && message
107                        .reminder
108                        .is_some_and(|reminder| reminder.goal_id == goal_id)
109            })
110        {
111            GoalReminderKind::Continuation
112        } else {
113            GoalReminderKind::Started
114        }
115    }

117    /// The message accepted from `owner` under `key`, whether this holds it or let
118    /// go of it as settled: a producer's retry of the key is answered with it.
119    #[must_use]
120    pub fn accepted(
121        &self,
122        owner: &ProducerOwner,
123        key: &ProducerIdempotencyKey,
124    ) -> Option<ProducerMessageId> {
125        self.messages
126            .iter()
127            .find(|message| &message.producer_owner == owner && &message.idempotency_key == key)
128            .map(|message| message.message_id)
129            .or_else(|| self.settled.keys.get(&key_digest(owner, key)).copied())
130    }
131
132    /// Hop depth a `send_message` from `run` must carry: one more than the
133    /// deepest Agent-owned mail this session's log shows `run` has seen, or `0`
134    /// when the run has seen no agent mail.
135    ///
136    /// The basis is durable observation — admission, a live claim, or committed
137    /// consumption — not payload heuristics, so a chain cannot be reset by
138    /// releasing a claim or by restarting the process. A message with no hop
139    /// metadata is a pre-guard send and counts as depth `0`.
140    ///
141    /// This over-approximates on purpose: any send from a run that consumed
142    /// hop-`N` mail inherits `N + 1`, even when the model treats the send as an
143    /// unrelated topic. The guard's bound, not per-message linkage, is the
144    /// control operators tune.
145    #[must_use]
146    pub fn inherited_agent_hop(&self, run: RunId) -> u32 {
147        self.messages
148            .iter()
149            .filter(|message| {
150                matches!(message.producer_owner, ProducerOwner::Agent { .. })
151                    && (message.consumed_run == Some(run)
152                        || message
153                            .admission
154                            .is_some_and(|(admitted_run, _)| admitted_run == run)
155                        || self.claims.values().any(|claim| {
156                            claim.run_id == run && claim.message_ids.contains(&message.message_id)
157                        }))
158            })
159            .filter_map(|message| message.agent_hop)
160            .max()
161            .map_or(0, |deepest| deepest.saturating_add(1))
162    }

164    #[must_use]
165    pub fn from_events<E: Borrow<StoredEvent>>(events: &[E]) -> Self {
166        let mut fold = GoalProducerFold::default();
167        for event in events {
168            fold.observe(event.borrow());
169        }
170        Arc::unwrap_or_clone(fold.projection)
171    }

173    /// Test-only: whether this, a log's own fold, which let go of messages that
174    /// settled, agrees with `whole`, the same log folded keeping every message:
175    /// what it holds is `whole`'s, in order; what it let go of had settled, and
176    /// its key answers with its ID; and the rest is the same.
177    #[cfg(any(test, feature = "test-support"))]
178    #[must_use]
179    pub fn agrees_with_whole(&self, whole: &Self) -> bool {
180        let held = self
181            .messages
182            .iter()
183            .map(|message| message.message_id)
184            .collect::<HashSet<_>>();
185        let let_go_settled = whole
186            .messages
187            .iter()
188            .filter(|message| !held.contains(&message.message_id))
189            .all(|message| {
190                message.claims.is_empty()
191                    && ((message.consumed && message.consumption_recorded) || message.discarded)
192                    && self.accepted(&message.producer_owner, &message.idempotency_key)
193                        == Some(message.message_id)
194            });
195        let continued = whole
196            .messages
197            .iter()
198            .filter(|message| message.consumed)
199            .filter_map(|message| message.reminder.map(|reminder| reminder.goal_id))
200            .collect::<HashSet<_>>();
201        self.goal == whole.goal
202            && self.selection == whole.selection
203            && self.claims == whole.claims
204            && self.invalid == whole.invalid
205            && whole
206                .messages
207                .iter()
208                .filter(|message| held.contains(&message.message_id))
209                .eq(self.messages.iter())
210            && let_go_settled
211            && self.settled.continued_goals == continued
212    }

214    fn reject(&mut self, seq: u64, reason: &str) {
215        self.invalid.push((seq, reason.to_owned()));
216    }

219/// [`GoalProducerProjection`] folded one event at a time, so a log that
220/// grew is folded only through what it gained.
221#[derive(Clone, Debug, Default, serde::Deserialize, serde::Serialize)]
222pub struct GoalProducerFold {
223    /// Shared with the readers handed it; copied only when an event changes
224    /// it while one still holds it.
225    projection: Arc<GoalProducerProjection>,
226    goals: GoalFold,
227    #[serde(with = "crate::events::sorted::map")]
228    message_indexes: HashMap<ProducerMessageId, usize>,
229    #[serde(with = "crate::events::sorted::map")]
230    dedup_indexes: HashMap<(ProducerOwner, ProducerIdempotencyKey), usize>,
231    #[serde(with = "crate::events::sorted::set")]
232    terminal_runs: HashSet<RunId>,
233    /// The messages accepted and neither admitted nor discarded yet.
234    #[serde(with = "crate::events::sorted::set")]
235    unadmitted: HashSet<usize>,
236    /// The runs that ended, oldest first, of those `terminal_runs` keeps: the
237    /// newest [`SETTLED_KEPT`], and any a message it holds names.
238    terminal_order: VecDeque<RunId>,
239}

241/// How many settled messages, and ended runs, a log's own fold keeps beyond what
242/// it must: what settles is read back for a moment, as a cron job reads where its
243/// message landed once that run ended.
244const SETTLED_KEPT: usize = 16;

277    /// Folds the next event of the log.
278    pub fn observe(&mut self, event: &StoredEvent) {
279        match &event.payload {
280            EventPayload::GoalActivated { selection, .. } => match self.goals.observe(event) {
281                Ok(()) => {
282                    let goal = self.goals.goal.clone();
283                    let projection = self.projection_mut();
284                    projection.selection = selection.clone();
285                    projection.goal = goal;
286                }
287                Err(reason) => self.projection_mut().reject(event.seq, reason),
288            },
289            EventPayload::GoalChecklistRevised { .. } => match self.goals.observe(event) {
290                Ok(()) => self.projection_mut().goal = self.goals.goal.clone(),
291                Err(reason) => self.projection_mut().reject(event.seq, reason),
292            },
293            EventPayload::GoalLifecycleChanged { selection, .. } => {
294                match self.goals.observe(event) {
295                    Ok(()) => {
296                        let goal = self.goals.goal.clone();
297                        let projection = self.projection_mut();
298                        if let Some(selection) = selection {
299                            projection.selection = Some(selection.clone());
300                        }
301                        projection.goal = goal;
302                    }
303                    Err(reason) => self.projection_mut().reject(event.seq, reason),
304                }
305            }
306            EventPayload::ProducerMessageAccepted {
307                message_id,
308                producer_owner,
309                mode,
310                idempotency_key,
311                description,
312                body,
313                reminder,
314                agent_hop,
315                ..
316            } => {
317                if !valid_reminder_owner(producer_owner, reminder.as_ref()) {
318                    self.projection_mut()
319                        .reject(event.seq, "invalid producer reminder ownership");
320                    return;
321                }
322                if !valid_hop_owner(producer_owner, *agent_hop) {
323                    self.projection_mut()
324                        .reject(event.seq, "invalid producer hop ownership");
325                    return;
326                }
327                // A retry of a key whose message settled and went: as an exact
328                // retry of one held, it changes nothing.
329                if !self.projection.settled.keys.is_empty()
330                    && self
331                        .projection
332                        .settled
333                        .keys
334                        .contains_key(&key_digest(producer_owner, idempotency_key))
335                {
336                    return;
337                }
338                let dedup_key = (producer_owner.clone(), idempotency_key.clone());
339                if let Some(index) = self.dedup_indexes.get(&dedup_key).copied() {
340                    let prior = &self.projection.messages[index];
341                    // `agent_hop` is guard metadata, not payload: a replayed
342                    // acceptance for the same key stays idempotent even when
343                    // the stamped depth differs from the stored record.
344                    let exact = prior.message_id == *message_id
345                        && prior.mode == *mode
346                        && prior.description == *description
347                        && prior.body == *body
348                        && prior.reminder == *reminder;
349                    if exact {
350                        return;
351                    }
352                    self.projection_mut()
353                        .reject(event.seq, "conflicting producer idempotency key");
354                    return;
355                }
356                if self.message_indexes.contains_key(message_id) {
357                    self.projection_mut()
358                        .reject(event.seq, "conflicting producer message id");
359                    return;
360                }
361                let index = self.projection.messages.len();
362                self.projection_mut().messages.push(ProducerMessageRecord {
363                    message_id: *message_id,
364                    producer_owner: producer_owner.clone(),
365                    mode: *mode,
366                    idempotency_key: idempotency_key.clone(),
367                    description: description.clone(),
368                    body: body.clone(),
369                    reminder: *reminder,
370                    agent_hop: *agent_hop,
371                    accepted_seq: event.seq,
372                    admission: None,
373                    claims: HashSet::new(),
374                    consumed: false,
375                    discarded: false,
376                    discarded_seq: None,
377                    consumption_recorded: false,
378                    consumed_run: None,
379                });
380                self.message_indexes.insert(*message_id, index);
381                self.dedup_indexes.insert(dedup_key, index);
382                self.unadmitted.insert(index);
383            }
384            EventPayload::ProducerMessageAdmitted { message_id } => {
385                let Some(run_id) = event.run_id else {
386                    self.projection_mut()
387                        .reject(event.seq, "producer admission is missing run id");
388                    return;
389                };
390                if self.terminal_runs.contains(&run_id) {
391                    self.projection_mut()
392                        .reject(event.seq, "producer admission targets a terminal run");
393                    return;
394                }
395                let Some(index) = self.message_indexes.get(message_id).copied() else {
396                    self.projection_mut()
397                        .reject(event.seq, "producer admission has no accepted message");
398                    return;
399                };
400                let message = &self.projection.messages[index];
401                let replaceable = message.admission.is_some_and(|(prior_run, _)| {
402                    prior_run != run_id && self.terminal_runs.contains(&prior_run)
403                });
404                if message.consumed
405                    || message.discarded_seq.is_some()
406                    || (message.admission.is_some() && !replaceable)
407                {
408                    self.projection_mut()
409                        .reject(event.seq, "invalid producer admission");
410                    return;
411                }
412                self.projection_mut().messages[index].admission = Some((run_id, event.seq));
413                self.unadmitted.remove(&index);
414            }
415            EventPayload::ProducerMessagesClaimed { message_ids } => {
416                let Some(run_id) = event.run_id else {
417                    self.projection_mut()
418                        .reject(event.seq, "producer claim is missing run id");
419                    return;
420                };
421                let unique = message_ids.iter().copied().collect::<HashSet<_>>();
422                let indexes = message_ids
423                    .iter()
424                    .filter_map(|message_id| self.message_indexes.get(message_id).copied())
425                    .collect::<Vec<_>>();
426                let valid = !message_ids.is_empty()
427                    && unique.len() == message_ids.len()
428                    && indexes.len() == message_ids.len()
429                    && !self.projection.claims.contains_key(&event.seq)
430                    && indexes.iter().all(|index| {
431                        let message = &self.projection.messages[*index];
432                        message.accepted_seq < event.seq
433                            && !message.consumed
434                            && message.discarded_seq.is_none()
435                            && message
436                                .admission
437                                .is_some_and(|(admission_run, admission_seq)| {
438                                    admission_run == run_id && admission_seq < event.seq
439                                })
440                    });
441                if !valid {
442                    self.projection_mut()
443                        .reject(event.seq, "invalid producer claim");
444                    return;
445                }
446                for index in indexes {
447                    self.projection_mut().messages[index]
448                        .claims
449                        .insert(event.seq);
450                }
451                self.projection_mut().claims.insert(
452                    event.seq,
453                    ProducerClaimRecord {
454                        run_id,
455                        message_ids: message_ids.clone(),
456                    },
457                );
458            }
459            EventPayload::ProducerMessagesReleased { claim_seq } => {
460                let Some(run_id) = event.run_id else {
461                    self.projection_mut()
462                        .reject(event.seq, "producer release is missing run id");
463                    return;
464                };
465                let Some(claim) = self.projection.claims.get(claim_seq) else {
466                    self.projection_mut()
467                        .reject(event.seq, "producer release has no claim");
468                    return;
469                };
470                if *claim_seq == 0 || claim.run_id != run_id {
471                    self.projection_mut()
472                        .reject(event.seq, "producer release ownership is invalid");
473                    return;
474                }
475                let message_ids = claim.message_ids.clone();
476                for message_id in &message_ids {
477                    if let Some(index) = self.message_indexes.get(message_id).copied() {
478                        self.projection_mut().messages[index]
479                            .claims
480                            .remove(claim_seq);
481                    }
482                }
483                self.projection_mut().claims.remove(claim_seq);
484            }
485            EventPayload::ModelTurnCommitted {
486                input_through_seq, ..
487            } => {
488                let Some(run_id) = event.run_id else { return };
489                let covered = |message: &ProducerMessageRecord| {
490                    !message.consumed
491                        && message
492                            .admission
493                            .is_some_and(|(admission_run, admission_seq)| {
494                                admission_run == run_id && admission_seq <= *input_through_seq
495                            })
496                        && message
497                            .discarded_seq
498                            .is_none_or(|discarded_seq| discarded_seq > *input_through_seq)
499                };
500                // Most turns consume nothing; they leave a shared projection
501                // unshared.
502                if !self.projection.messages.iter().any(covered) {
503                    return;
504                }
505                let projection = self.projection_mut();
506                for message in &mut projection.messages {
507                    if covered(message) {
508                        message.discarded = false;
509                        message.discarded_seq = None;
510                        message.consumed = true;
511                        message.consumed_run = Some(run_id);
512                        if let Some(reminder) = message.reminder {
513                            projection.settled.continued_goals.insert(reminder.goal_id);
514                        }
515                    }
516                }
517            }
518            EventPayload::ProducerMessageConsumed { message_id, run_id } => {
519                let Some(index) = self.message_indexes.get(message_id).copied() else {
520                    self.projection_mut()
521                        .reject(event.seq, "producer consumption has no accepted message");
522                    return;
523                };
524                let message = &self.projection.messages[index];
525                if event.run_id != Some(*run_id)
526                    || message.admission.map(|(run, _)| run) != Some(*run_id)
527                    || message.consumed_run != Some(*run_id)
528                    || message.consumption_recorded
529                {
530                    self.projection_mut()
531                        .reject(event.seq, "invalid producer consumption marker");
532                    return;
533                }
534                self.projection_mut().messages[index].consumption_recorded = true;
535            }
536            EventPayload::ProducerMessageDiscarded {
537                message_id,
538                reminder,
539                producer_owner,
540            } => {
541                let Some(index) = self.message_indexes.get(message_id).copied() else {
542                    self.projection_mut()
543                        .reject(event.seq, "producer discard has no accepted message");
544                    return;
545                };
546                let message = &self.projection.messages[index];
547                let identity_matches = producer_owner == &message.producer_owner
548                    && reminder
549                        .as_ref()
550                        .is_none_or(|identity| message.reminder.as_ref() == Some(identity));
551                if !identity_matches || message.consumed || !message.claims.is_empty() {
552                    self.projection_mut()
553                        .reject(event.seq, "invalid producer message discard");
554                    return;
555                }
556                if message.discarded_seq.is_none() {
557                    self.projection_mut().messages[index].discarded = true;
558                    self.projection_mut().messages[index].discarded_seq = Some(event.seq);
559                    self.unadmitted.remove(&index);
560                }
561            }
562            EventPayload::RunFinished { .. } => {
563                if let Some(run_id) = event.run_id
564                    && self.terminal_runs.insert(run_id)
565                {
566                    self.terminal_order.push_back(run_id);
567                }
568            }
569            _ => {}
570        }
571    }

573    /// [`Self::observe`] for a log's own fold, which lets go of what settled, at
574    /// each committed turn and run's end, so it holds what is open and not what
575    /// the log grew to (`docs/site/specs/forever-sessions.md` section 6). A fold
576    /// for a model's history ([`GoalProducerProjection::from_events`]) keeps every
577    /// message, whose bodies are the history.
578    pub fn observe_letting_go(&mut self, event: &StoredEvent) {
579        self.observe(event);
580        if matches!(
581            event.payload,
582            EventPayload::RunFinished { .. } | EventPayload::ModelTurnCommitted { .. }
583        ) {
584            self.let_go_settled();
585        }
586    }

588    /// Lets go of the messages that settled, but for the newest [`SETTLED_KEPT`]:
589    /// those consumed and recorded, or discarded, with no claim, whose run ended,
590    /// so no record can change them. Their keys stay, as a digest, and so does
591    /// whether one was a goal's consumed reminder. Then of the ended runs, those
592    /// no message held names, but for the newest [`SETTLED_KEPT`]: a message is
593    /// admitted to the run that starts, so only a held message asks of an older one.
594    fn let_go_settled(&mut self) {
595        let ended = |run: &RunId| self.terminal_runs.contains(run);
596        let mut settled = self
597            .projection
598            .messages
599            .iter()
600            .enumerate()
601            .filter(|(_, message)| {
602                message.claims.is_empty()
603                    && ((message.consumed
604                        && message.consumption_recorded
605                        && message.consumed_run.as_ref().is_none_or(ended))
606                        || (message.discarded
607                            && message.admission.as_ref().is_none_or(|(run, _)| ended(run))))
608            })
609            .map(|(index, message)| (message.accepted_seq, index))
610            .collect::<Vec<_>>();
611        if settled.len() > SETTLED_KEPT {
612            settled.sort_unstable();
613            settled.truncate(settled.len() - SETTLED_KEPT);
614            let gone = settled
615                .into_iter()
616                .map(|(_, index)| index)
617                .collect::<HashSet<_>>();
618            let projection = self.projection_mut();
619            let mut index = 0;
620            let mut keys = Vec::new();
621            projection.messages.retain(|message| {
622                let kept = !gone.contains(&index);
623                if !kept {
624                    keys.push((
625                        key_digest(&message.producer_owner, &message.idempotency_key),
626                        message.message_id,
627                    ));
628                }
629                index += 1;
630                kept
631            });
632            projection.settled.keys.extend(keys);
633            self.message_indexes.clear();
634            self.dedup_indexes.clear();
635            self.unadmitted.clear();
636            for (index, message) in self.projection.messages.iter().enumerate() {
637                self.message_indexes.insert(message.message_id, index);
638                self.dedup_indexes.insert(
639                    (
640                        message.producer_owner.clone(),
641                        message.idempotency_key.clone(),
642                    ),
643                    index,
644                );
645                if message.admission.is_none() && message.discarded_seq.is_none() {
646                    self.unadmitted.insert(index);
647                }
648            }
649        }
650        if self.terminal_order.len() > SETTLED_KEPT {
651            let named = self
652                .projection
653                .messages
654                .iter()
655                .flat_map(|message| {
656                    message
657                        .admission
658                        .map(|(run, _)| run)
659                        .into_iter()
660                        .chain(message.consumed_run)
661                })
662                .collect::<HashSet<_>>();
663            let beyond = self.terminal_order.len() - SETTLED_KEPT;
664            let mut kept = VecDeque::with_capacity(self.terminal_order.len());
665            for (position, run) in self.terminal_order.drain(..).enumerate() {
666                if position >= beyond || named.contains(&run) {
667                    kept.push_back(run);
668                } else {
669                    self.terminal_runs.remove(&run);
670                }
671            }
672            self.terminal_order = kept;
673        }
674    }
675}

677/// The goal part of a session's log: its current goal, after the
678/// activations, checklist revisions, and lifecycle changes that were valid
679/// where they stand. Shared by the producer projection and the session
680/// projection, which reports an active goal in `SessionMeta::waiting_on`.
681#[derive(Clone, Debug, Default, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
682pub struct GoalFold {
683    pub goal: Option<GoalState>,
684    /// The latest revision of every goal the log activated.
685    #[serde(with = "crate::events::sorted::map")]
686    revisions: HashMap<GoalId, u64>,
687}
688
689impl GoalFold {
690    /// Applies `event` when it is a goal event, or says why it is invalid
691    /// and leaves the fold unchanged. Any other event is ignored.
692    pub fn observe(&mut self, event: &StoredEvent) -> Result<(), &'static str> {
693        match &event.payload {
694            EventPayload::GoalActivated {
695                goal_id,
696                objective,
697                revision,
698                ..
699            } => {
700                let replaceable = self
701                    .goal
702                    .as_ref()
703                    .is_none_or(|goal| is_terminal(goal.status));
704                let distinct = self
705                    .goal
706                    .as_ref()
707                    .is_none_or(|goal| goal.goal_id != *goal_id);
708                let unseen = !self.revisions.contains_key(goal_id);
709                if objective.trim().is_empty() || !replaceable || !distinct || !unseen {
710                    return Err("invalid goal activation");
711                }
712                self.goal = Some(GoalState {
713                    goal_id: *goal_id,
714                    objective: objective.clone(),
715                    status: GoalStatus::Active,
716                    items: Vec::new(),
717                    revision: *revision,
718                });
719                self.revisions.insert(*goal_id, *revision);
720            }
721            EventPayload::GoalChecklistRevised {
722                goal_id,
723                items,
724                revision,
725            } => {
726                let valid_items = items.iter().all(|item| !item.description.trim().is_empty());
727                let valid = valid_items
728                    && self.goal.as_ref().is_some_and(|goal| {
729                        goal.goal_id == *goal_id
730                            && !is_terminal(goal.status)
731                            && *revision > goal.revision
732                    });
733                if !valid {
734                    return Err("invalid goal checklist revision");
735                }
736                let goal = self.goal.as_mut().expect("validated current goal");
737                goal.items = items.clone();
738                goal.revision = *revision;
739                self.revisions.insert(*goal_id, *revision);
740            }
741            EventPayload::GoalLifecycleChanged {
742                goal_id,
743                status,
744                revision,
745                selection,
746            } => {
747                let valid = self.goal.as_ref().is_some_and(|goal| {
748                    goal.goal_id == *goal_id
749                        && *revision > goal.revision
750                        && valid_lifecycle_change(goal, *status)
751                        && (selection.is_none() || *status == GoalStatus::Active)
752                });
753                if !valid {
754                    return Err("invalid goal lifecycle transition");
755                }
756                let goal = self.goal.as_mut().expect("validated current goal");
757                goal.status = *status;
758                goal.revision = *revision;
759                self.revisions.insert(*goal_id, *revision);
760            }
761            _ => {}
762        }
763        Ok(())
764    }
765
766    /// Whether the current goal is active, so goal mode keeps scheduling
767    /// runs. A paused, completed, or cancelled goal schedules none.
768    #[must_use]
769    pub fn is_active(&self) -> bool {
770        self.goal
771            .as_ref()
772            .is_some_and(|goal| goal.status == GoalStatus::Active)
773    }
774}
775
776fn is_terminal(status: GoalStatus) -> bool {
777    matches!(status, GoalStatus::Completed | GoalStatus::Cancelled)
778}
779
780fn valid_lifecycle_change(goal: &GoalState, status: GoalStatus) -> bool {
781    match (goal.status, status) {
782        (GoalStatus::Active, GoalStatus::Paused | GoalStatus::Cancelled)
783        | (GoalStatus::Paused, GoalStatus::Active | GoalStatus::Cancelled) => true,
784        (GoalStatus::Active | GoalStatus::Paused, GoalStatus::Completed) => {
785            !goal.items.is_empty() && goal.items.iter().all(|item| item.finished)
786        }
787        _ => false,
788    }
789}
790
791fn valid_reminder_owner(owner: &ProducerOwner, reminder: Option<&GoalReminderIdentity>) -> bool {
792    match (owner, reminder) {
793        (ProducerOwner::Goal { goal_id }, Some(reminder)) => *goal_id == reminder.goal_id,
794        (ProducerOwner::Goal { .. }, None) | (_, Some(_)) => false,
795        (_, None) => true,
796    }
797}

799/// Hop metadata belongs exclusively to agent mail. A malformed durable event
800/// that stamps it onto plugin, goal, or delegation mail is quarantined so it
801/// cannot inflate an agent chain it never joined.
802const fn valid_hop_owner(owner: &ProducerOwner, agent_hop: Option<u32>) -> bool {
803    agent_hop.is_some() == matches!(owner, ProducerOwner::Agent { .. })
804}

806/// Whether a log's producer state needs reconciliation after a restart: an active goal, an unread message, or a message consumed without its
807/// consumption record.
808pub fn producer_state_pending(projection: &GoalProducerProjection) -> bool {
809    projection.goal.is_some()
810        || projection
811            .messages
812            .iter()
813            .any(|message| pending(message) || (message.consumed && !message.consumption_recorded))
814}
815
816/// Whether a message is accepted but neither consumed nor discarded: mail
817/// addressed to the session that it still owes a wake.
818///
819/// This is the race-free "a wake is owed" signal: `ProducerMessageAccepted` is
820/// durable before a send reports success, and a message stays pending through
821/// the whole window before its consumer's `RunStarted` flips session status to
822/// `Running` (it clears only when a committed model turn has actually consumed
823/// it). Liveness checks therefore cannot miss a steered finished child in the
824/// gap between acceptance and run admission.
825pub fn pending(message: &ProducerMessageRecord) -> bool {
826    !message.consumed && !message.discarded
827}
