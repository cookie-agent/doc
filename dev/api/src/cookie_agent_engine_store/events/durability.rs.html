1//! When a session log's records reach the disk.
2//!
3//! A record is written to the file as it is appended, so readers in this
4//! process see it at once, but only a commit point is synced before it takes
5//! effect (`docs/site/specs/store-locks-and-durability.md`, section 3). The
6//! sync runs after the store's locks are released, on a duplicate of the
7//! writer's descriptor: `fdatasync` covers everything written to the file so
8//! far, whoever wrote it, so one sync serves every commit point written
9//! before it started, and a commit point written while a sync is in flight
10//! waits for it to end and then runs the next one.
11
12use std::{
13    collections::HashSet,
14    fs::File,
15    io,
16    path::{Path, PathBuf},
17    sync::{Arc, Condvar, Mutex, MutexGuard},
18};
19
20use super::{EventLogError, fsync_directory};

22/// One record written to a log, numbered from the log's first write in this
23/// process. A commit point's append returns it, and
24/// [`super::EventLog::sync_through`] waits until a sync covers it. Reverts
25/// do not reuse these numbers, unlike sequences.
26#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
27pub struct Commit(u64);
28
29/// The sync state of one log: what was written, what a finished sync
30/// covers, and the descriptor syncs run on.
31#[derive(Debug)]
32pub(super) struct Durability {
33    path: PathBuf,
34    state: Mutex<State>,
35    /// Signalled whenever a sync ends.
36    synced: Condvar,
37}
38
39#[derive(Debug, Default)]
40struct State {
41    /// Records written so far.
42    written: u64,
43    /// `written` when the last successful sync started.
44    synced: u64,
45    /// `written` when the last sync, successful or not, started: what no
46    /// longer waits on a sync.
47    settled: u64,
48    /// A sync is in flight; a commit point written meanwhile waits for it.
49    syncing: bool,
50    /// A duplicate of the open writer's descriptor.
51    file: Option<Arc<File>>,
52    /// The log's directory, synced with the next sync after the writer
53    /// created the file.
54    directory: Option<PathBuf>,
55    /// Blobs records written since the last sync name, synced before the
56    /// log, with their folder, so a synced record never names a blob a
57    /// crash could lose.
58    blobs: Vec<PathBuf>,
59    /// The folder of `blobs`.
60    blob_directory: Option<PathBuf>,
61    /// The blobs a sync of this log covered, which a later record naming
62    /// them again does not sync again.
63    synced_blobs: HashSet<PathBuf>,
64    #[cfg(any(test, feature = "test-support"))]
65    hook: Option<SyncHook>,
66    #[cfg(any(test, feature = "test-support"))]
67    syncs: usize,
68}
69
70#[cfg(any(test, feature = "test-support"))]
71#[derive(Debug)]
72pub(super) struct SyncHook {
73    pub(super) reached: std::sync::mpsc::Sender<()>,
74    pub(super) release: std::sync::mpsc::Receiver<()>,
75}
76
77impl Durability {
78    pub(super) fn new(path: &Path) -> Self {
79        Self {
80            path: path.to_owned(),
81            state: Mutex::default(),
82            synced: Condvar::new(),
83        }
84    }
85
86    fn state(&self) -> MutexGuard<'_, State> {
87        self.state
88            .lock()
89            .unwrap_or_else(|poisoned| poisoned.into_inner())
90    }
91
92    /// A writer opened `file`; later syncs run on a duplicate of its
93    /// descriptor, which stays valid if the writer closes meanwhile. When
94    /// the writer `created` the file, the next sync also syncs its directory.
95    pub(super) fn attach(&self, file: &File, created: bool) -> io::Result<()> {
96        let duplicate = Arc::new(file.try_clone()?);
97        let mut state = self.state();
98        state.file = Some(duplicate);
99        if created {
100            state.directory = self.path.parent().map(Path::to_owned);
101        }
102        Ok(())
103    }
104
105    /// Counts one record written, returning its number and whether it has to
106    /// sync whatever it is: the file is new and its directory entry is not
107    /// durable yet.
108    pub(super) fn written(&self) -> (Commit, bool) {
109        let mut state = self.state();
110        state.written += 1;
111        (Commit(state.written), state.directory.is_some())
112    }
113
114    /// A record about to be written names the blobs at `blobs`, in the
115    /// folder `directory`, each with whether this write just stored it: the
116    /// sync that covers the record syncs them first. One stored earlier
117    /// that a sync of this log covered already is not synced again.
118    pub(super) fn names_blobs(
119        &self,
120        directory: &Path,
121        blobs: impl IntoIterator<Item = (PathBuf, bool)>,
122    ) {
123        let mut state = self.state();
124        let mut any = false;
125        for (blob, stored) in blobs {
126            if stored {
127                state.synced_blobs.remove(&blob);
128            } else if state.synced_blobs.contains(&blob) {
129                continue;
130            }
131            if !state.blobs.contains(&blob) {
132                state.blobs.push(blob);
133            }
134            any = true;
135        }
136        if any {
137            state.blob_directory = Some(directory.to_owned());
138        }
139    }
140
141    /// The last record written.
142    pub(super) fn latest(&self) -> Commit {
143        Commit(self.state().written)
144    }
145
146    /// Whether `commit` waits on no sync any more: one that covers it ended,
147    /// successfully or not.
148    pub(super) fn is_settled(&self, commit: Commit) -> bool {
149        self.state().settled >= commit.0
150    }
151
152    /// Returns once a sync that covers `commit` succeeded, running it when no
153    /// other sync is in flight. A sync in flight that started before
154    /// `commit` was written is waited out and followed by the next one,
155    /// which covers every record written by then. A failed sync's error goes
156    /// to the caller that ran it; the others waiting try again.
157    pub(super) fn sync_through(&self, commit: Commit) -> Result<(), EventLogError> {
158        let mut state = self.state();
159        loop {
160            if state.synced >= commit.0 {
161                return Ok(());
162            }
163            if state.syncing {
164                state = self
165                    .synced
166                    .wait(state)
167                    .unwrap_or_else(|poisoned| poisoned.into_inner());
168                continue;
169            }
170            state.syncing = true;
171            let target = state.written;
172            let file = state.file.clone();
173            let directory = state.directory.take();
174            let blobs = Blobs::take(&mut state);
175            #[cfg(any(test, feature = "test-support"))]
176            let hook = state.hook.take();
177            drop(state);
178            #[cfg(any(test, feature = "test-support"))]
179            if let Some(hook) = hook {
180                let _ = hook.reached.send(());
181                let _ = hook.release.recv();
182            }
183            let result = self.sync(file.as_deref(), directory.as_deref(), &blobs);
184            state = self.state();
185            state.syncing = false;
186            state.settled = state.settled.max(target);
187            #[cfg(any(test, feature = "test-support"))]
188            {
189                state.syncs += usize::from(file.is_some());
190            }
191            match &result {
192                Ok(()) => {
193                    state.synced = state.synced.max(target);
194                    blobs.synced(&mut state);
195                }
196                Err(_) => {
197                    if state.directory.is_none() {
198                        state.directory = directory;
199                    }
200                    blobs.restore(&mut state);
201                }
202            }
203            self.synced.notify_all();
204            result?;
205        }
206    }
207
208    fn sync(
209        &self,
210        file: Option<&File>,
211        directory: Option<&Path>,
212        blobs: &Blobs,
213    ) -> Result<(), EventLogError> {
214        blobs.sync()?;
215        if let Some(file) = file {
216            #[cfg(any(test, feature = "test-support"))]
217            super::EVENT_LOG_SYNCS.with(|syncs| syncs.set(syncs.get() + 1));
218            crate::timing::time(
219                crate::timing::Meter::LogSync,
220                || self.path.display().to_string(),
221                || file.sync_data(),
222            )
223            .map_err(|source| EventLogError::Io {
224                path: self.path.clone(),
225                source,
226            })?;
227        }
228        if let Some(directory) = directory {
229            fsync_directory(directory)?;
230        }
231        Ok(())
232    }
233
234    /// The writer closes: syncs, on its own descriptor, whatever no sync has
235    /// covered yet, then lets go of the duplicate descriptor, which on
236    /// Windows would keep the file and its session folder from being deleted.
237    /// A failure has no caller to reach; those records are what a crash may
238    /// drop anyway, and a commit point among them is still synced by the
239    /// append waiting on it, so the duplicate stays for that sync.
240    pub(super) fn close(&self, file: &File) {
241        let mut state = self.state();
242        if state.synced >= state.written {
243            state.file = None;
244            return;
245        }
246        let target = state.written;
247        let directory = state.directory.take();
248        let blobs = Blobs::take(&mut state);
249        drop(state);
250        let result = self.sync(Some(file), directory.as_deref(), &blobs);
251        let mut state = self.state();
252        #[cfg(any(test, feature = "test-support"))]
253        {
254            state.syncs += 1;
255        }
256        match result {
257            Ok(()) => {
258                state.synced = state.synced.max(target);
259                state.settled = state.settled.max(target);
260                blobs.synced(&mut state);
261                if state.synced >= state.written {
262                    state.file = None;
263                }
264            }
265            Err(_) => {
266                if state.directory.is_none() {
267                    state.directory = directory;
268                }
269                blobs.restore(&mut state);
270            }
271        }
272        self.synced.notify_all();
273    }
274
275    /// The file was cut and synced whole (a revert): everything written so
276    /// far is as durable as the cut, once the blobs its records name are.
277    pub(super) fn cut(&self) -> Result<(), EventLogError> {
278        let blobs = Blobs::take(&mut self.state());
279        let result = blobs.sync();
280        let mut state = self.state();
281        match result {
282            Ok(()) => blobs.synced(&mut state),
283            Err(error) => {
284                blobs.restore(&mut state);
285                return Err(error);
286            }
287        }
288        state.synced = state.written;
289        state.settled = state.written;
290        self.synced.notify_all();
291        Ok(())
292    }

294    #[cfg(any(test, feature = "test-support"))]
295    pub(super) fn install_hook(&self, hook: SyncHook) {
296        self.state().hook = Some(hook);
297    }

299    /// Test-only: the blobs the next sync syncs first.

309    }

311    /// Test-only: how many times this log was synced.

315    }

318/// The blobs one sync covers, taken from the state as it starts.
319struct Blobs {
320    paths: Vec<PathBuf>,
321    directory: Option<PathBuf>,
322}
323
324impl Blobs {
325    fn take(state: &mut State) -> Self {
326        Self {
327            paths: std::mem::take(&mut state.blobs),
328            directory: state.blob_directory.take(),
329        }
330    }
331
332    /// Syncs each blob, then their folder.
333    fn sync(&self) -> Result<(), EventLogError> {
334        for path in &self.paths {
335            super::blobs::sync(path).map_err(|source| EventLogError::Io {
336                path: path.clone(),
337                source,
338            })?;
339        }
340        match &self.directory {
341            Some(directory) => fsync_directory(directory),
342            None => Ok(()),
343        }
344    }
345
346    fn synced(self, state: &mut State) {
347        state.synced_blobs.extend(self.paths);
348    }
349
350    /// A failed sync leaves them for the next one.
351    fn restore(self, state: &mut State) {
352        let mut paths = self.paths;
353        paths.append(&mut state.blobs);
354        state.blobs = paths;
355        if state.blob_directory.is_none() {
356            state.blob_directory = self.directory;
357        }
358    }
359}

361#[cfg(test)]
362mod tests {
363    use std::io::Write as _;
364
365    use super::Durability;
366
367    /// Closing the writer lets go of the duplicate descriptor once nothing is
368    /// left to sync, so a deletion right after a run can remove the session's
369    /// folder on Windows, where an open descriptor keeps it.
370    #[test]
371    fn closing_lets_go_of_the_duplicate_descriptor() {
372        let directory = tempfile::tempdir().expect("a directory");
373        let path = directory.path().join("events.jsonl");
374        let mut file = std::fs::File::create(&path).expect("a log");
375        let durability = Durability::new(&path);
376        durability.attach(&file, false).expect("a duplicate");
377        file.write_all(b"{}\n").expect("a record");
378        let (commit, _) = durability.written();
379        durability.sync_through(commit).expect("a sync");
380        assert!(
381            durability.state().file.is_some(),
382            "kept while the writer is open"
383        );
384        durability.close(&file);
385        assert!(durability.state().file.is_none(), "nothing left to sync");
386
387        // A record no sync covered yet is synced on close, then let go too.
388        durability.attach(&file, false).expect("a duplicate");
389        file.write_all(b"{}\n").expect("a record");
390        let _ = durability.written();
391        durability.close(&file);
392        assert!(durability.state().file.is_none(), "synced on close");
393        assert!(durability.is_settled(durability.latest()));
394    }
