1//! `state.json`: what a windowed log's records fold to, saved beside it, so that
2//! an open reads only what comes after (`docs/site/specs/forever-sessions.md`
3//! section 4).
4//!
5//! The file names a record, the tip of the log when it was written, and holds
6//! what folds give as of that record, and where the window starts. It is a
7//! cache: it is written without a sync, and a missing, unreadable, or mismatched
8//! one means folding the log from its start, one record at a time and keeping
9//! only the window, so what that takes in memory does not depend on the log's
10//! length. Only a log that is damaged, or of an older history format, reads the
11//! way it always has: whole, as quarantining a record needs.
12//!
13//! It stays valid across builds. A change to what a fold gives leaves the file's
14//! shape alone and its contents wrong, so it must raise [`STATE_VERSION`], and a
15//! test holding a digest of what the folds give on the golden logs fails until
16//! that happens (section 4.4).
17

28/// The file beside a session's `events.jsonl`.
29pub const STATE_FILE: &str = "state.json";

31/// What [`STATE_FILE`]'s contents mean. Raise it when what a fold gives changes,
32/// or the file's shape does: a different one means folding the log from its
33/// start. The test `the_folds_of_the_golden_logs_match_their_recorded_digest`
34/// says when.
35pub const STATE_VERSION: u32 = 10;

37/// How much of a log comes between two writes of [`STATE_FILE`] on a run's way,
38/// so a long run is covered and a turn does not pay for a write.
39const WRITE_AFTER_BYTES: u64 = 1024 * 1024;

41/// A log shorter than this has no [`STATE_FILE`]: folding it from its start takes
42/// a few milliseconds, and the file would be about as large as the log, or larger.
43pub const STATE_MIN_LOG_BYTES: u64 = 1024 * 1024;
44
45/// The shortest log a test lets have a [`STATE_FILE`]: none, so that every test's
46/// logs go through one, unless a test of the rule sets it.
47#[cfg(any(test, feature = "test-support"))]
48static TEST_MIN_LOG_BYTES: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
49
50/// Test-only: makes logs shorter than `bytes` have no [`STATE_FILE`] in this
51/// process, as [`STATE_MIN_LOG_BYTES`] does in a build without `test-support`.
52#[cfg(any(test, feature = "test-support"))]
53pub fn set_state_min_log_bytes_for_test(bytes: u64) {
54    TEST_MIN_LOG_BYTES.store(bytes, Ordering::Release);
55}

57fn state_min_log_bytes() -> u64 {
58    #[cfg(any(test, feature = "test-support"))]
59    {
60        TEST_MIN_LOG_BYTES.load(Ordering::Acquire)
61    }
62    #[cfg(not(any(test, feature = "test-support")))]
63    {
64        STATE_MIN_LOG_BYTES
65    }
66}

114/// [`RawLines`], with what the line index already says left out: every line not
115/// named here is in the file, as the record of its sequence.
116#[derive(Clone, Default, Deserialize, PartialEq, Serialize)]
117struct StoredRaw {
118    /// The lines no record reproduces, by sequence, in base64 as a line need not
119    /// be text.
120    kept: Vec<(u64, String)>,
121    /// Where lines in the file end, for those another line follows that carries
122    /// no sequence.
123    ends: Vec<(u64, u64)>,
124    /// Only the kept lines are known: [`RawLines::partial`].
125    partial: bool,
126}
127
128impl StoredRaw {
129    /// `None` when `raw` does not match `lines`.
130    fn of(raw: &RawLines, lines: &LineIndex) -> Option<Self> {
131        let mut stored = Self {
132            partial: raw.partial,
133            ..Self::default()
134        };
135        for line in &raw.lines {
136            match &line.bytes {
137                RawBytes::Kept(bytes) => stored.kept.push((line.seq, STANDARD.encode(bytes))),
138                RawBytes::InFile { start, end } => {
139                    let index = lines
140                        .starts
141                        .binary_search_by_key(&line.seq, |(seq, _)| *seq)
142                        .ok()?;
143                    if lines.starts[index].1 != *start {
144                        return None;
145                    }
146                    if *end != lines.line_end(index) {
147                        stored.ends.push((line.seq, *end));
148                    }
149                }
150            }
151        }
152        Some(stored)
153    }
154
155    fn restore(self, lines: &LineIndex, path: &Path) -> Option<RawLines> {
156        if self.partial {
157            return Some(RawLines {
158                file: Some(Arc::from(path)),
159                lines: self
160                    .kept
161                    .into_iter()
162                    .map(|(seq, bytes)| {
163                        Some(RawLine {
164                            seq,
165                            bytes: RawBytes::Kept(Arc::from(STANDARD.decode(bytes).ok()?)),
166                        })
167                    })
168                    .collect::<Option<_>>()?,
169                partial: true,
170            });
171        }
172        let mut kept = self.kept.into_iter().peekable();
173        let mut ends = self.ends.into_iter().peekable();
174        let mut in_file = false;
175        let mut raw = Vec::with_capacity(lines.starts.len());
176        for (index, (seq, start)) in lines.starts.iter().enumerate() {
177            let bytes = if let Some((_, bytes)) = kept.next_if(|(kept, _)| kept == seq) {
178                RawBytes::Kept(Arc::from(STANDARD.decode(bytes).ok()?))
179            } else {
180                in_file = true;
181                RawBytes::InFile {
182                    start: *start,
183                    end: ends
184                        .next_if(|(line, _)| line == seq)
185                        .map_or_else(|| lines.line_end(index), |(_, end)| end),
186                }
187            };
188            raw.push(RawLine { seq: *seq, bytes });
189        }
190        (kept.peek().is_none() && ends.peek().is_none()).then(|| RawLines {
191            file: in_file.then(|| Arc::from(path)),
192            lines: raw,
193            partial: false,
194        })
195    }
196}

198impl LineIndex {
199    /// Where the line at `index` of [`Self::starts`] ends, newline excluded, when
200    /// the next sequenced line, or the file's end, follows it.
201    fn line_end(&self, index: usize) -> u64 {
202        self.starts
203            .get(index + 1)
204            .map_or(self.len, |(_, start)| *start)
205            .saturating_sub(1)
206    }

224/// The lines of a file from an offset on.
225struct LineCursor {
226    reader: io::BufReader<fs::File>,
227    /// Where the next line starts.
228    position: u64,
229    line: Vec<u8>,
230}
231
232/// What a [`LineCursor`] found next.
233#[derive(Debug, Eq, PartialEq)]
234enum Next {
235    /// A line, newline included, in `line`.
236    Line,
237    /// The file ends in part of a line.
238    Torn,
239    End,
240}
241
242impl LineCursor {
243    fn open(path: &Path, offset: u64) -> Option<Self> {
244        let mut file = fs::File::open(path).ok()?;
245        file.seek(io::SeekFrom::Start(offset)).ok()?;
246        Some(Self {
247            reader: io::BufReader::with_capacity(64 * 1024, file),
248            position: offset,
249            line: Vec::new(),
250        })
251    }
252
253    fn next(&mut self) -> Option<Next> {
254        self.line.clear();
255        let read = self.reader.read_until(b'\n', &mut self.line).ok()?;
256        if read == 0 {
257            return Some(Next::End);
258        }
259        if !self.line.ends_with(b"\n") {
260            return Some(Next::Torn);
261        }
262        self.position += read as u64;
263        Some(Next::Line)
264    }
265}

267/// A log being read one line at a time, each record through the steps a live
268/// append takes: validation, index, projection, and folds, and the window moving
269/// on at each committed turn. What the window lets go of is dropped.
270struct Folding<'a> {
271    path: &'a Path,
272    session_id: SessionId,
273    storage: EventStorage,
274    validation: ValidationState,
275    projection: Option<ProjectionFolds>,
276    /// Nothing has been folded yet, so the next record starts the projection.
277    fresh: bool,
278    reader: LineReader<'a>,
279}
280
281impl Folding<'_> {
282    /// Reads a line, newline excluded, of `length` bytes with it, at `start`.
283    /// `None` when the log is not one this can read: damage, or a record that
284    /// does not validate.
285    fn read(
286        &mut self,
287        index: usize,
288        start: u64,
289        line: &[u8],
290        length: u64,
291        fetch: &mut dyn FnMut(&str) -> io::Result<Vec<u8>>,
292    ) -> Option<()> {
293        let read = self.reader.read(index, line, fetch).ok()?;
294        if read.stop
295            || self.reader.render_only.is_some()
296            || self.reader.validation_taint != ValidationTaint::default()
297        {
298            return None;
299        }
300        self.storage
301            .lines
302            .as_mut()?
303            .push(read.sequenced.map(|(seq, _)| seq), length);
304        if let Some((seq, strict)) = read.sequenced {
305            let bytes = if strict && read.record.is_some() {
306                RawBytes::InFile {
307                    start,
308                    end: start + line.len() as u64,
309                }
310            } else {
311                RawBytes::Kept(Arc::from(line))
312            };
313            // A log that let go of its lines keeps only those no record reproduces.
314            if !self.storage.raw.partial || matches!(bytes, RawBytes::Kept(_)) {
315                Arc::make_mut(&mut self.storage.raw)
316                    .lines
317                    .push(RawLine { seq, bytes });
318            }
319        }
320        let Some(record) = read.record else {
321            return Some(());
322        };
323        let record = Arc::new(record);
324        validate_record_incremental(
325            self.path,
326            self.session_id,
327            &record,
328            &mut self.validation,
329            false,
330        )
331        .ok()?;
332        self.storage.push(Arc::clone(&record));
333        if std::mem::take(&mut self.fresh) {
334            self.projection = ProjectionFolds::of_creation(&record, None);
335        }
336        if self
337            .projection
338            .as_mut()
339            .is_some_and(|projection| !projection.observe_read(&record))
340        {
341            self.projection = None;
342        }
343        self.storage.narrow_after(&record);
344        Some(())
345    }
346
347    fn observed_tip(&self) -> u64 {
348        self.reader.last_observed_seq.max(
349            self.reader
350                .diagnostics
351                .iter()
352                .filter(|diagnostic| diagnostic.skipped)
353                .map(|diagnostic| diagnostic.seq)
354                .max()
355                .unwrap_or(0),
356        )
357    }
358}

360/// A file that ended in part of a record: an owned open cuts it, as the loader does.
361fn settle_torn_tail(
362    path: &Path,
363    cursor: &LineCursor,
364    next: &Next,
365    torn_tail: TornTail,
366) -> Option<bool> {
367    let torn = *next == Next::Torn;
368    if torn && matches!(torn_tail, TornTail::Truncate) {
369        cut_event_file(path, cursor.position).ok()?;
370    }
371    Some(torn)

374/// Opens the log at `path` from its [`STATE_FILE`] and the log from the window's
375/// start: the records up to the file's join the window as they are, and each
376/// after it goes through the steps of a live append. `None` on any mismatch, for
377/// the caller to fold the log from its start. Nothing is changed by then, so a
378/// crash anywhere in this costs nothing.
379pub(super) fn open_from_state(
380    path: &Path,
381    session_id: SessionId,
382    torn_tail: TornTail,
383) -> Option<OpenedLog> {
384    let directory = path.parent()?;
385    let bytes = secure_store::read_replaced(&directory.join(STATE_FILE)).ok()?;
386    let file = serde_json::from_slice::<StateFile>(&bytes).ok()?;
387    drop(bytes);
388    if file.version != STATE_VERSION
389        || file.history_format != cookie_agent_protocol::HISTORY_FORMAT
390        || file.session_id != session_id
391        || file.folds.render_only.is_some()
392    {
393        return None;
394    }
395    let StateFile { at, folds, .. } = file;
396    let blob_dir = blobs::tree_blob_dir(path);
397    let mut fetch = |digest: &str| blobs::read(&blob_dir, digest);
398    let (creation, start, first_seq) =
399        window_start(path, session_id, folds.window, &folds.lines, &mut fetch)?;
400    let mut cursor = LineCursor::open(path, start)?;
401    // Up to the record the file names, whole, and at the offset it names. A
402    // record the file says validation refused stays out, as a pass leaves it out.
403    let clean = !folds
404        .diagnostics
405        .iter()
406        .any(|diagnostic| diagnostic.skipped);
407    let mut read = LineReader::new(path);
408    read.last_observed_seq = first_seq.saturating_sub(1);
409    let mut held = Vec::new();
410    loop {
411        let line_start = cursor.position;
412        if line_start > at.offset || cursor.next()? != Next::Line {
413            return None;
414        }
415        let line = &cursor.line[..cursor.line.len() - 1];
416        if line.is_empty() {
417            continue;
418        }
419        let index = usize::from(line_start != 0);
420        let line = read.read(index, line, &mut fetch).ok()?;
421        if line.stop || (clean && read.diagnostics.iter().any(|noted| noted.skipped)) {
422            return None;
423        }
424        if let Some((seq, _)) = line.sequenced
425            && folds
426                .lines
427                .starts
428                .binary_search(&(seq, line_start))
429                .is_err()
430        {
431            return None;
432        }
433        let at_record = line_start == at.offset;
434        let Some(record) = line.record else {
435            if at_record {
436                return None;
437            }
438            continue;
439        };
440        if at_record && (record.seq != at.seq || record.timestamp != at.timestamp) {
441            return None;
442        }
443        if folds.refused.binary_search(&record.seq).is_err() {
444            held.push(Arc::new(record));
445        }
446        if at_record {
447            break;
448        }
449    }
450    if cursor.position != folds.lines.len {
451        return None;
452    }
453    drop(read);
454    let creation = creation.or_else(|| held.first().cloned())?;
455    let covered = folds.lines.len;
456    let raw = folds.raw.restore(&folds.lines, path)?;
457    if folds.pinned.iter().any(|pinned| pinned.seq >= first_seq)
458        || !folds.pinned.is_sorted_by_key(|pinned| pinned.seq)
459        || (!folds.context && !folds.pinned.is_empty())
460    {
461        return None;
462    }
463    let mut all = folds.pinned;
464    all.extend(held);
465    let storage = EventStorage {
466        all: Arc::new(all),
467        creation: Some(creation),
468        window: folds.window,
469        context: folds.context,
470        view_start: folds.view_start,
471        refused: folds.refused,
472        index: folds.index,
473        physical_model_turns: folds.physical_model_turns,
474        lines: Some(folds.lines),
475        raw: Arc::new(raw),
476    };
477    let mut reader = LineReader::new(path);
478    reader.last_observed_seq = at.seq;
479    let mut folding = Folding {
480        path,
481        session_id,
482        storage,
483        validation: folds.validation,
484        projection: Some(folds.projection),
485        fresh: false,
486        reader,
487    };
488    let next = loop {
489        let line_start = cursor.position;
490        let next = cursor.next()?;
491        if next != Next::Line {
492            break next;
493        }
494        let length = cursor.line.len() as u64;
495        let line = &cursor.line[..cursor.line.len() - 1];
496        if line.is_empty() {
497            folding.storage.lines.as_mut()?.push(None, length);
498            continue;
499        }
500        folding.read(1, line_start, line, length, &mut fetch)?;
501        // The loader would repair this line or skip it, and report that.
502        if !folding.reader.diagnostics.is_empty() {
503            return None;
504        }
505    };
506    let torn = settle_torn_tail(path, &cursor, &next, torn_tail)?;
507    let next_seq = folds.next_seq.max(folding.reader.last_observed_seq + 1);
508    let mut projection = folding.projection;
509    if let Some(projection) = &mut projection {
510        projection.meta.last_event_seq = next_seq - 1;
511    }
512    Some(OpenedLog {
513        storage: folding.storage,
514        validation: folding.validation,
515        initial_taint: folds.initial_taint,
516        diagnostics: folds.diagnostics,
517        next_seq,
518        torn_tail: torn,
519        projection,
520        state_covers: Some(covered),
521    })
522}

524/// Where a pass of a window read from `lines` starts: the log's first record
525/// when the window does not hold it, the offset of the window's first line, and
526/// its sequence. With no window it reads the log from its start.
527#[allow(clippy::type_complexity)]
528fn window_start(
529    path: &Path,
530    session_id: SessionId,
531    window: Option<u64>,
532    lines: &LineIndex,
533    fetch: &mut dyn FnMut(&str) -> io::Result<Vec<u8>>,
534) -> Option<(Option<Arc<StoredEvent>>, u64, u64)> {
535    let Some(window) = window else {
536        return Some((None, 0, 0));
537    };
538    // A native model's context starts at a sequence, which need not be a record's.
539    let index = lines.starts.partition_point(|(seq, _)| *seq < window);
540    let (first, start) = *lines.starts.get(index)?;
541    // The index holds every line from the window on, so a sequence between the
542    // two is one the file skips. One that starts later would open a window
543    // missing its first records, which a build wrote for a while.
544    if first > window
545        && !lines
546            .gaps
547            .iter()
548            .any(|(skipped, through)| *skipped <= window && first - 1 <= *through)
549    {
550        return None;
551    }
552    let mut cursor = LineCursor::open(path, 0)?;
553    let creation = loop {
554        if cursor.next()? != Next::Line {
555            return None;
556        }
557        let line = &cursor.line[..cursor.line.len() - 1];
558        if !line.is_empty() {
559            let mut reader = LineReader::new(path);
560            break reader.read(0, line, fetch).ok()?.record?;
561        }
562    };
563    if creation.session_id != session_id {
564        return None;
565    }
566    // The byte before the window's line ends the line before it.
567    let mut before = [0_u8];
568    let mut file = fs::File::open(path).ok()?;
569    file.seek(io::SeekFrom::Start(start.checked_sub(1)?)).ok()?;
570    file.read_exact(&mut before).ok()?;
571    (before == *b"\n").then_some((Some(Arc::new(creation)), start, first))
572}

574/// Folds the log at `path` from its start, one record at a time, holding what the
575/// window holds and no more. `None` when the log is damaged or of an older format,
576/// which only the whole read deals with.
577pub(super) fn fold_file(
578    path: &Path,
579    session_id: SessionId,
580    torn_tail: TornTail,
581) -> Option<OpenedLog> {
582    let blob_dir = blobs::tree_blob_dir(path);
583    let mut fetch = |digest: &str| blobs::read(&blob_dir, digest);
584    let mut cursor = LineCursor::open(path, 0)?;
585    let mut storage = EventStorage::new(Vec::new());
586    storage.lines = Some(LineIndex::default());
587    let mut folding = Folding {
588        path,
589        session_id,
590        storage,
591        validation: ValidationState::default(),
592        projection: None,
593        fresh: true,
594        reader: LineReader::new(path),
595    };
596    let mut index = 0;
597    let next = loop {
598        let line_start = cursor.position;
599        let next = cursor.next()?;
600        if next != Next::Line {
601            break next;
602        }
603        let length = cursor.line.len() as u64;
604        let line = &cursor.line[..cursor.line.len() - 1];
605        if line.is_empty() {
606            folding.storage.lines.as_mut()?.push(None, length);
607            continue;
608        }
609        folding.read(index, line_start, line, length, &mut fetch)?;
610        index += 1;
611    };
612    folding.storage.creation.as_ref()?;
613    let torn = settle_torn_tail(path, &cursor, &next, torn_tail)?;
614    #[cfg(any(test, feature = "test-support"))]
615    EVENT_LOG_PARSES.with(|parses| parses.set(parses.get() + 1));
616    let next_seq = folding.observed_tip().saturating_add(1).max(1);
617    let diagnostics = folding.reader.diagnostics;
618    let mut projection = folding.projection;
619    if let Some(projection) = &mut projection {
620        projection.finish_read(next_seq - 1, &diagnostics);
621    }
622    let raw = Arc::make_mut(&mut folding.storage.raw);
623    if raw
624        .lines
625        .iter()
626        .any(|line| matches!(line.bytes, RawBytes::InFile { .. }))
627    {
628        raw.file = Some(Arc::from(path));
629    }
630    Some(OpenedLog {
631        storage: folding.storage,
632        validation: folding.validation,
633        initial_taint: ValidationTaint::default(),
634        diagnostics,
635        next_seq,
636        torn_tail: torn,
637        projection,
638        state_covers: None,
639    })
640}

642impl EventLog {
643    /// How much of the log came since [`STATE_FILE`] was last written, if this log
644    /// should have one: it has a window, an ACP agent's or a native model's
645    /// context (sections 4.2 and 5.6), a file this process owns, and at least
646    /// [`STATE_MIN_LOG_BYTES`] of it. A native log that holds itself whole, having
647    /// no checkpoint, has none.
648    #[must_use]
649    pub fn state_behind(&self) -> Option<u64> {
650        if self.read_only || self.converted || self.render_only.is_some() || !self.is_persisted() {
651            return None;
652        }
653        let storage = self.storage();
654        storage.window?;
655        if storage.index.window_start.is_none() && !storage.context {
656            return None;
657        }
658        let len = storage.lines.as_ref()?.len;
659        if len < state_min_log_bytes() {
660            return None;
661        }
662        Some(len.saturating_sub(self.state_written.load(Ordering::Acquire)))
663    }
664
665    /// Whether a committed turn should write [`STATE_FILE`] now.
666    #[must_use]
667    pub fn state_due_after_turn(&self) -> bool {
668        self.state_behind()
669            .is_some_and(|behind| behind > WRITE_AFTER_BYTES)
670    }
671
672    /// What [`STATE_FILE`] holds as of this log's tip, with `projection`, this
673    /// log's. `None` when this log should not have one, or its layout or tip do
674    /// not allow it: the tip is not a record it holds, or a sequence jumped
675    /// without a record of damage, which a parse would find and this log did not.
676    /// The caller holds off appends, so `projection` is of the same moment.
677    #[must_use]
678    pub fn state_file(&self, projection: &ProjectionFolds) -> Option<StateFile> {
679        self.state_behind()?;
680        let _append = self
681            .append
682            .lock()
683            .unwrap_or_else(|poisoned| poisoned.into_inner());
684        let storage = self.storage();
685        let lines = storage.lines.as_ref()?;
686        let tip = storage.all.last()?;
687        let (last, offset) = *lines.starts.last()?;
688        let next_seq = self.next_seq.load(Ordering::Acquire);
689        if last != tip.seq || next_seq != tip.seq + 1 || projection.meta.last_event_seq != tip.seq {
690            return None;
691        }
692        let initial_taint = self.initial_taint();
693        let unexplained = lines
694            .gaps
695            .iter()
696            .any(|gap| !initial_taint.broad.contains(gap));
697        if unexplained {
698            return None;
699        }
700        Some(StateFile {
701            version: STATE_VERSION,
702            history_format: cookie_agent_protocol::HISTORY_FORMAT,
703            session_id: self.session_id,
704            at: StateAt {
705                seq: tip.seq,
706                timestamp: tip.timestamp,
707                offset,
708            },
709            folds: LogFolds::of(
710                &storage,
711                self.validation
712                    .lock()
713                    .unwrap_or_else(|poisoned| poisoned.into_inner())
714                    .clone(),
715                initial_taint,
716                self.diagnostics.clone(),
717                self.render_only.clone(),
718                next_seq,
719                projection.clone(),
720            )?,
721        })
722    }
723
724    /// On an owned open: removes [`STATE_FILE`] when this log did not open from
725    /// it, as it does not match the log, or when the log is shorter than the
726    /// least that has one ([`STATE_MIN_LOG_BYTES`]), though it matches. Nothing
727    /// else would: such a file is read at every open and thrown away, and only a
728    /// log that is due one writes it again. A removal that fails leaves it for
729    /// the next owned open.
730    pub(super) fn settle_state_file(&self) {
731        let short = self
732            .storage()
733            .lines
734            .as_ref()
735            .is_none_or(|lines| lines.len < state_min_log_bytes());
736        if !short && self.state_written.load(Ordering::Acquire) > 0 {
737            return;
738        }
739        let Some(directory) = self.path.parent() else {
740            return;
741        };
742        let path = directory.join(STATE_FILE);
743        match fs::remove_file(&path) {
744            Ok(()) => {}
745            Err(error) if error.kind() == io::ErrorKind::NotFound => {}
746            Err(error) => {
747                eprintln!("{}: could not be removed: {error}", path.display());
748                return;
749            }
750        }
751        self.state_written.store(0, Ordering::Release);
752    }
753
754    /// Notes that `state`, this log's, is now in [`STATE_FILE`].
755    pub fn note_state_written(&self, state: &StateFile) {
756        self.state_written
757            .store(state.folds.lines.len, Ordering::Release);
758    }
759}
760
761impl StateFile {
762    /// The file's bytes. Two equal states give the same ones.
763    pub fn to_bytes(&self) -> Result<Vec<u8>, serde_json::Error> {
764        serde_json::to_vec(self)
765    }
766}

768impl LogFolds {
769    fn of(
770        storage: &EventStorage,
771        validation: ValidationState,
772        initial_taint: ValidationTaint,
773        diagnostics: Vec<EventLoadDiagnostic>,
774        render_only: Option<RenderOnly>,
775        next_seq: u64,
776        projection: ProjectionFolds,
777    ) -> Option<Self> {
778        let lines = storage.lines.clone()?;
779        Some(Self {
780            window: storage.window,
781            context: storage.context,
782            pinned: storage.all[..storage.window_position()].to_vec(),
783            view_start: storage.view_start.clone(),
784            refused: Arc::clone(&storage.refused),
785            index: storage.index.clone(),
786            physical_model_turns: storage.physical_model_turns,
787            raw: StoredRaw::of(&storage.raw, &lines)?,
788            lines,
789            validation,
790            initial_taint,
791            diagnostics,
792            render_only,
793            next_seq,
794            projection,
795        })
796    }
