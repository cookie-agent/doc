1//! History format 2's blobs (`docs/site/specs/session-storage-v3.md`,
2//! section 4.2): the large values of a session log's records live in the
3//! tree's `blobs/` folder, each named by the SHA-256 of its content, and the
4//! record names them by digest.
5//!
6//! What moves out of a record:
7//!
8//! - a tool result's `output` longer than [`OUTPUT_BLOB_MIN_BYTES`]: the
9//!   record keeps its first [`OUTPUT_HEAD_BYTES`] (cut at a character
10//!   boundary) as `output`, its byte length as `output_bytes`, and
11//!   `output_blob`;
12//! - a tool result's `display` and every part of its `metadata` whose JSON
13//!   is longer than [`OUTPUT_BLOB_MIN_BYTES`], each whole: an object of the
14//!   metadata only once the large members inside it moved and it is still
15//!   that long, and the metadata itself the same way. The result lists
16//!   them in `moved_values`, each `{"pointer", "value_blob"}`, `pointer` a
17//!   JSON pointer into the result (`/display`, `/metadata/acp/raw_output`),
18//!   in the order they moved;
19//! - `acp_session_updated`'s `commands` and `options`, and `run_started`'s
20//!   `agent.fallback_chain` and each internal agent's `fallbacks`, whose
21//!   JSON reaches [`VALUE_BLOB_MIN_BYTES`]: the member becomes
22//!   `<member>_blob`, and the blob holds its JSON. Repeated lists are stored
23//!   once.
24//!
25//! A reference is the string `artifact://sha256/<digest>`, so artifact
26//! liveness (forks, reverts) sees blobs as it sees every other artifact:
27//! the folder is the one the tree's artifacts live in.
28//!
29//! Records in memory are whole: the loader reads every value back as it
30//! reads a line, and the writer moves them out as it writes one. So
31//! everything that reads a record (model replay, compaction, the approval
32//! reviewer, clients, plugins) gets the whole value.
33

45/// A tree's blobs, beside its root's log. It replaced `artifacts/`.
46pub const BLOBS_DIR: &str = "blobs";
47/// Where a tree kept its artifacts before history format 2: a tree that was
48/// not converted yet (one another process holds) is read from there.
49pub const LEGACY_ARTIFACTS_DIR: &str = "artifacts";
50/// A tool result's output longer than this moves to a blob.
51pub const OUTPUT_BLOB_MIN_BYTES: usize = 8 * 1024;
52/// How much of a moved output the record keeps.
53pub const OUTPUT_HEAD_BYTES: usize = 2 * 1024;
54/// A list whose JSON is at least this long moves to a blob; a shorter one is
55/// smaller than its reference would be worth.
56pub const VALUE_BLOB_MIN_BYTES: usize = 512;
57
58const URI_PREFIX: &str = "artifact://sha256/";
59/// Only a member name can be followed by `":"` straight away: inside a
60/// string, a quote is escaped.
61const MARKER: &[u8] = b"_blob\":\"artifact://sha256/";

63/// Blobs a conversion produced in memory, by digest, until they are written
64/// or, for a read-only open, read back.
65pub type Blobs = HashMap<String, Vec<u8>>;

67/// The `blobs/` folder of the tree the log at `log` belongs to: beside the
68/// root's log, also for a child's under `subagents/`.
69#[must_use]
70pub fn tree_blob_dir(log: &Path) -> PathBuf {
71    let member = log.parent().unwrap_or_else(|| Path::new("."));
72    let root = match member.parent() {
73        Some(parent)
74            if parent.file_name() == Some(std::ffi::OsStr::new(crate::session::SUBAGENTS_DIR)) =>
75        {
76            parent.parent().unwrap_or(parent)
77        }
78        _ => member,
79    };
80    root.join(BLOBS_DIR)
81}

89/// Whether the record of `payload` may have a value that moves to a blob,
90/// so it is written through [`externalize`].
91#[must_use]
92pub fn may_move(payload: &EventPayload) -> bool {
93    match payload {
94        // Its display and metadata are sized as JSON, as it is written.
95        EventPayload::ToolCallTerminated { termination } => termination.result.is_some(),
96        EventPayload::AcpSessionUpdated { update } => {
97            update.commands.is_some() || update.options.is_some()
98        }
99        EventPayload::RunStarted { .. } => true,
100        _ => false,
101    }
102}

104/// Whether the stored line `line` names a blob, so it has to be read
105/// through [`internalize`]. A false positive (a member of that name
106/// somewhere else) only costs the slower read.
107#[must_use]
108pub fn has_references(line: &[u8]) -> bool {
109    let mut rest = line;
110    while let Some(at) = rest.iter().position(|byte| *byte == b'_') {
111        if rest[at..].starts_with(MARKER) {
112            return true;
113        }
114        rest = &rest[at + 1..];
115    }
116    false
117}

119/// Moves the large values of a record's `payload` (its JSON in the storage
120/// encoding) to blobs, handing each blob's digest and content to `keep`.
121pub fn externalize(payload: &mut Value, keep: &mut dyn FnMut(&str, Vec<u8>)) {
122    let Some(object) = payload.as_object_mut() else {
123        return;
124    };
125    match object.get("type").and_then(Value::as_str) {
126        Some("tool_call_terminated") => {
127            if let Some(result) = object.get_mut("result").and_then(Value::as_object_mut) {
128                move_output(result, keep);
129                move_result_values(result, keep);
130            }
131        }
132        Some("acp_session_updated") => {
133            move_value(object, "commands", keep);
134            move_value(object, "options", keep);
135        }
136        Some("run_started") => {
137            if let Some(agent) = object.get_mut("agent").and_then(Value::as_object_mut) {
138                move_value(agent, "fallback_chain", keep);
139            }
140            if let Some(agents) = object
141                .get_mut("internal_agents")
142                .and_then(Value::as_array_mut)
143            {
144                for agent in agents.iter_mut().filter_map(Value::as_object_mut) {
145                    move_value(agent, "fallbacks", keep);
146                }
147            }
148        }
149        _ => {}
150    }
151}

153fn move_output(result: &mut Map<String, Value>, keep: &mut dyn FnMut(&str, Vec<u8>)) {
154    if result.contains_key("output_blob") {
155        return;
156    }
157    let Some(Value::String(output)) = result.get_mut("output") else {
158        return;
159    };
160    if output.len() <= OUTPUT_BLOB_MIN_BYTES {
161        return;
162    }
163    let mut end = OUTPUT_HEAD_BYTES;
164    while !output.is_char_boundary(end) {
165        end -= 1;
166    }
167    let head = output[..end].to_owned();
168    let whole = std::mem::replace(output, head).into_bytes();
169    let length = whole.len();
170    let digest = digest(&whole);
171    result.insert("output_bytes".into(), Value::from(length));
172    result.insert("output_blob".into(), Value::String(reference(&digest)));
173    keep(&digest, whole);
174}

176/// Moves a tool result's display and every large part of its metadata, each
177/// whole, naming them in `moved_values` by JSON pointer into the result, in
178/// the order they moved: a member of the metadata whose JSON is longer than
179/// [`OUTPUT_BLOB_MIN_BYTES`] moves, an object only once the large members
180/// inside it moved and it is still that long, and so does the metadata as
181/// a whole.
182fn move_result_values(result: &mut Map<String, Value>, keep: &mut dyn FnMut(&str, Vec<u8>)) {
183    if result.contains_key(MOVED_VALUES) {
184        return;
185    }
186    let mut moved = Vec::new();
187    if let Some(Value::Object(metadata)) = result.get_mut("metadata") {
188        move_large_members(metadata, "/metadata", &mut moved, keep);
189    }
190    for member in ["metadata", "display"] {
191        if result
192            .get(member)
193            .is_some_and(|value| json_len(value) > OUTPUT_BLOB_MIN_BYTES)
194        {
195            let value = result.remove(member).expect("present");
196            moved.push(move_whole(&format!("/{member}"), &value, keep));
197        }
198    }
199    if !moved.is_empty() {
200        result.insert(MOVED_VALUES.into(), Value::Array(moved));
201    }
202}

204fn move_large_members(
205    object: &mut Map<String, Value>,
206    pointer: &str,
207    moved: &mut Vec<Value>,
208    keep: &mut dyn FnMut(&str, Vec<u8>),
209) {
210    let keys = object.keys().cloned().collect::<Vec<_>>();
211    for key in keys {
212        let path = format!("{pointer}/{}", key.replace('~', "~0").replace('/', "~1"));
213        let Some(member) = object.get_mut(&key) else {
214            continue;
215        };
216        if json_len(member) <= OUTPUT_BLOB_MIN_BYTES {
217            continue;
218        }
219        if let Value::Object(inner) = member {
220            move_large_members(inner, &path, moved, keep);
221            if json_len(member) <= OUTPUT_BLOB_MIN_BYTES {
222                continue;
223            }
224        }
225        let value = object.remove(&key).expect("present");
226        moved.push(move_whole(&path, &value, keep));
227    }
228}

230/// Moves `value`, which was at `pointer`, to a blob of its JSON, returning
231/// its `moved_values` entry.
232fn move_whole(pointer: &str, value: &Value, keep: &mut dyn FnMut(&str, Vec<u8>)) -> Value {
233    let bytes = serde_json::to_vec(value).expect("a JSON value serializes");
234    let digest = digest(&bytes);
235    keep(&digest, bytes);
236    serde_json::json!({ "pointer": pointer, "value_blob": reference(&digest) })
237}

239fn json_len(value: &Value) -> usize {
240    serde_json::to_vec(value).map_or(0, |bytes| bytes.len())
241}

243/// The member of a tool result naming the values it moved whole.
244const MOVED_VALUES: &str = "moved_values";

246fn move_value(object: &mut Map<String, Value>, member: &str, keep: &mut dyn FnMut(&str, Vec<u8>)) {
247    let key = format!("{member}_blob");
248    let Some(value) = object.get(member).filter(|value| !value.is_null()) else {
249        return;
250    };
251    if object.contains_key(&key) {
252        return;
253    }
254    let bytes = serde_json::to_vec(value).expect("a JSON value serializes");
255    if bytes.len() < VALUE_BLOB_MIN_BYTES {
256        return;
257    }
258    let digest = digest(&bytes);
259    object.remove(member);
260    object.insert(key, Value::String(reference(&digest)));
261    keep(&digest, bytes);
262}

264fn reference(digest: &str) -> String {
265    format!("{URI_PREFIX}{digest}")
266}

268/// The digest a reference names.
269fn referenced(reference: &Value) -> Result<&str, String> {
270    reference
271        .as_str()
272        .and_then(|reference| reference.strip_prefix(URI_PREFIX))
273        .filter(|digest| is_digest(digest))
274        .ok_or_else(|| format!("{reference} is not a blob reference"))
275}

277/// Whether `name` is a blob's name: 64 lowercase hex digits.
278#[must_use]
279pub fn is_digest(name: &str) -> bool {
280    name.len() == 64
281        && name
282            .bytes()
283            .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte))
284}

286/// Reads the values of a record's `payload` back from the blobs it names,
287/// each through `fetch`, which returns a blob's verified content. Fails when
288/// one does not read: the record cannot be read whole.
289pub fn internalize(
290    payload: &mut Value,
291    fetch: &mut dyn FnMut(&str) -> io::Result<Vec<u8>>,
292) -> Result<(), String> {
293    let Some(object) = payload.as_object_mut() else {
294        return Ok(());
295    };
296    match object.get("type").and_then(Value::as_str) {
297        Some("tool_call_terminated") => {
298            if let Some(result) = object.get_mut("result").and_then(Value::as_object_mut) {
299                restore_result_values(result, fetch)?;
300                restore_output(result, fetch)?;
301            }
302        }
303        Some("acp_session_updated") => {
304            restore_value(object, "commands", fetch)?;
305            restore_value(object, "options", fetch)?;
306        }
307        Some("run_started") => {
308            if let Some(agent) = object.get_mut("agent").and_then(Value::as_object_mut) {
309                restore_value(agent, "fallback_chain", fetch)?;
310            }
311            if let Some(agents) = object
312                .get_mut("internal_agents")
313                .and_then(Value::as_array_mut)
314            {
315                for agent in agents.iter_mut().filter_map(Value::as_object_mut) {
316                    restore_value(agent, "fallbacks", fetch)?;
317                }
318            }
319        }
320        _ => {}
321    }
322    Ok(())
323}

325fn restore_output(
326    result: &mut Map<String, Value>,
327    fetch: &mut dyn FnMut(&str) -> io::Result<Vec<u8>>,
328) -> Result<(), String> {
329    let Some(reference) = result.remove("output_blob") else {
330        return Ok(());
331    };
332    let digest = referenced(&reference)?;
333    let length = result
334        .remove("output_bytes")
335        .and_then(|length| length.as_u64())
336        .ok_or("the tool result's output_blob has no output_bytes")?;
337    let bytes = fetch(digest)
338        .map_err(|error| format!("blob {digest} of the tool result's output: {error}"))?;
339    if bytes.len() as u64 != length {
340        return Err(format!(
341            "blob {digest} holds {} bytes, the tool result's output {length}",
342            bytes.len()
343        ));
344    }
345    let output = String::from_utf8(bytes)
346        .map_err(|_| format!("blob {digest} of the tool result's output is not UTF-8"))?;
347    result.insert("output".into(), Value::String(output));
348    Ok(())
349}

351/// Puts back the values `moved_values` names, the last moved first, so an
352/// object that moved after members inside it is in place before they are.
353fn restore_result_values(
354    result: &mut Map<String, Value>,
355    fetch: &mut dyn FnMut(&str) -> io::Result<Vec<u8>>,
356) -> Result<(), String> {
357    let Some(moved) = result.remove(MOVED_VALUES) else {
358        return Ok(());
359    };
360    let moved = moved
361        .as_array()
362        .ok_or("the tool result's moved_values is not a list")?;
363    for entry in moved.iter().rev() {
364        let pointer = entry
365            .get("pointer")
366            .and_then(Value::as_str)
367            .ok_or("a moved value has no pointer")?;
368        let digest = referenced(entry.get("value_blob").unwrap_or(&Value::Null))?;
369        let bytes =
370            fetch(digest).map_err(|error| format!("blob {digest} of `{pointer}`: {error}"))?;
371        let value = serde_json::from_slice::<Value>(&bytes)
372            .map_err(|error| format!("blob {digest} of `{pointer}` is not JSON: {error}"))?;
373        let (parent, key) = pointer
374            .rsplit_once('/')
375            .ok_or_else(|| format!("{pointer} is not a JSON pointer"))?;
376        let key = key.replace("~1", "/").replace("~0", "~");
377        let target = if parent.is_empty() {
378            Some(&mut *result)
379        } else {
380            let mut current = None::<&mut Map<String, Value>>;
381            let mut segments = parent.split('/').skip(1);
382            let first = segments
383                .next()
384                .map(|first| first.replace("~1", "/").replace("~0", "~"));
385            if let Some(first) = first {
386                current = result.get_mut(&first).and_then(Value::as_object_mut);
387                for segment in segments {
388                    let segment = segment.replace("~1", "/").replace("~0", "~");
389                    current = current.and_then(|object| object.get_mut(&segment)?.as_object_mut());
390                }
391            }
392            current
393        };
394        target
395            .ok_or_else(|| format!("{pointer} names no place in the tool result"))?
396            .insert(key, value);
397    }
398    Ok(())
399}

401fn restore_value(
402    object: &mut Map<String, Value>,
403    member: &str,
404    fetch: &mut dyn FnMut(&str) -> io::Result<Vec<u8>>,
405) -> Result<(), String> {
406    let Some(reference) = object.remove(&format!("{member}_blob")) else {
407        return Ok(());
408    };
409    let digest = referenced(&reference)?;
410    let bytes = fetch(digest).map_err(|error| format!("blob {digest} of `{member}`: {error}"))?;
411    let value = serde_json::from_slice::<Value>(&bytes)
412        .map_err(|error| format!("blob {digest} of `{member}` is not JSON: {error}"))?;
413    object.insert(member.to_owned(), value);
414    Ok(())
415}

417/// The content of blob `digest` in `dir`, verified against its digest.
418pub fn read(dir: &Path, digest: &str) -> io::Result<Vec<u8>> {
419    if !is_digest(digest) {
420        return Err(io::Error::new(
421            io::ErrorKind::InvalidInput,
422            "not a blob digest",
423        ));
424    }
425    let bytes = fs::read(dir.join(digest))?;
426    if self::digest(&bytes) != digest {
427        return Err(io::Error::new(
428            io::ErrorKind::InvalidData,
429            "the blob's content does not match its digest",
430        ));
431    }
432    Ok(bytes)
433}

435/// Stores `bytes` as blob `digest` in `dir`, unless a blob of that length is
436/// there already, without syncing it. Returns the blob's path when this call
437/// wrote it.
438///
439/// The content goes to a private temporary first, closed before it is
440/// renamed to its address: two writers of one digest write the same bytes,
441/// and Windows renames no open file.
442pub fn write(dir: &Path, digest: &str, bytes: &[u8]) -> io::Result<Option<PathBuf>> {
443    let path = dir.join(digest);
444    if holds(&path, bytes.len()) {
445        return Ok(None);
446    }
447    create_private_dir(dir)?;
448    let temporary = dir.join(format!(".{digest}.{}.tmp", uuid::Uuid::now_v7()));
449    let result = (|| {
450        let mut file = create_private_file(&temporary)?;
451        file.write_all(bytes)?;
452        drop(file);
453        match fs::rename(&temporary, &path) {
454            Ok(()) => Ok(()),
455            // A concurrent writer of the same content took the name first.
456            Err(_) if holds(&path, bytes.len()) => fs::remove_file(&temporary),
457            Err(error) => Err(error),
458        }
459    })();
460    if result.is_err() {
461        let _ = fs::remove_file(&temporary);
462    }
463    result.map(|()| Some(path))
464}

466/// Whether `path` is a blob of `length` bytes. Its name is its content's
467/// address, so that is the content; a torn one (another length) is
468/// replaced, and readers verify the digest anyway.
469fn holds(path: &Path, length: usize) -> bool {
470    fs::symlink_metadata(path)
471        .is_ok_and(|metadata| metadata.is_file() && metadata.len() == length as u64)
472}

474/// Syncs the blob at `path`, which a writer closed. The handle that syncs
475/// has write access, which Windows needs to flush a file, and is closed
476/// again at once, so it never keeps the blob or its folder from being
477/// deleted. A blob deleted meanwhile has nothing left to sync.
478pub fn sync(path: &Path) -> io::Result<()> {
479    match OpenOptions::new().write(true).open(path) {
480        Ok(file) => file.sync_data(),
481        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(()),
482        Err(error) => Err(error),
483    }
484}
485
486/// Writes every blob of `blobs` to `dir` and syncs the ones it wrote, and
487/// the folder: a log that names them is synced after this.
488pub fn write_synced(dir: &Path, blobs: &Blobs) -> io::Result<()> {
489    let mut wrote = false;
490    for (digest, bytes) in blobs {
491        if let Some(path) = write(dir, digest, bytes)? {
492            sync(&path)?;
493            wrote = true;
494        }
495    }
496    if wrote {
497        super::fsync_directory(dir).map_err(|error| match error {
498            super::EventLogError::Io { source, .. } => source,
499            error => io::Error::other(error),
500        })?;
501    }
502    Ok(())
503}

505/// Creates the private `blobs/` folder in a tree folder that exists: a tree
506/// deleted meanwhile is not brought back.
507#[cfg(unix)]
508fn create_private_dir(dir: &Path) -> io::Result<()> {
509    use std::os::unix::fs::DirBuilderExt as _;
510    match fs::DirBuilder::new().mode(0o700).create(dir) {
511        Ok(()) => Ok(()),
512        Err(error) if error.kind() == io::ErrorKind::AlreadyExists && dir.is_dir() => Ok(()),
513        Err(error) => Err(error),
514    }
515}

712    fn blobs_are_written_once_and_read_verified() {
713        let directory = tempfile::tempdir().unwrap();
714        let dir = directory.path().join(BLOBS_DIR);
715        let content = b"blob content".to_vec();
716        let digest = digest(&content);
717        let path = write(&dir, &digest, &content).unwrap().expect("written");
718        sync(&path).unwrap();
719        assert_eq!(write(&dir, &digest, &content).unwrap(), None);
720        assert_eq!(read(&dir, &digest).unwrap(), content);
721        fs::write(&path, b"other content").unwrap();
722        assert_eq!(
723            read(&dir, &digest).unwrap_err().kind(),
724            io::ErrorKind::InvalidData
725        );
726        let leftovers = fs::read_dir(&dir)
727            .unwrap()
728            .flatten()
729            .filter(|entry| entry.file_name().to_string_lossy().ends_with(".tmp"))
730            .count();
731        assert_eq!(leftovers, 0);
732    }
