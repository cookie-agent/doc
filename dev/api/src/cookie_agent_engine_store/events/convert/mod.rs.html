1//! History format conversion (`docs/site/specs/compatibility-v1.md`
2//! section 8.7).
3//!
4//! A log written under an older history format is converted to
5//! [`HISTORY_FORMAT`] when a newer build opens it. Each step is a pure
6//! function from the old log's JSON lines to the next format's lines, in
7//! `v<N>_to_v<N+1>.rs`, registered in `CONVERTERS` and kept, so the steps
8//! chain: a format-1 log opened by a format-3 build converts 1 to 2 to 3.
9//! Converters never use old Rust types; they read and write JSON.
10//!
11//! A log without a `history_format` predates history formats and is not
12//! converted.
13//!
14//! # Adding the first real converter
15//!
16//! Format 1 is the format the protocols froze with. When a change cannot
17//! follow the additive rules of section 8.4, the history format bumps to 2
18//! and this module gains `v1_to_v2.rs`:
19//!
20//! 1. Bump [`HISTORY_FORMAT`] in `cookie_agent_protocol` to 2.
21//! 2. Write `pub(super) fn convert(lines: Vec<Value>) -> Result<Vec<Value>,
22//!    String>` in `v1_to_v2.rs`: it reads format-1 lines as JSON (envelopes
23//!    and payloads exactly as a format-1 writer stored them, the storage
24//!    encodings included, such as a tool result's `display_span`,
25//!    `text_span`, and `lines_span`) and returns format-2 lines. It names no
26//!    Rust event type, since those are format 2's; a line it does not
27//!    understand (an unknown or newer record) is copied unchanged. The chain
28//!    stamps `history_format: 2` on `session_created` afterwards.
29//! 3. Declare `mod v1_to_v2;` and register `Converter { from: 1, convert:
30//!    v1_to_v2::convert }` in `CONVERTERS`. Converters are never removed:
31//!    a format-1 log opened by a format-3 build converts 1 to 2 to 3.
32//! 4. Record new golden trees in `crates/engine/tests/fixtures/history/2/`
33//!    with the golden recorder, beside `history/1/`, which stays frozen: its
34//!    test then opens each format-1 tree through the converter and compares
35//!    the result with the snapshot it was recorded with, wherever format 2
36//!    does not change what the snapshot shows (where it does, the test
37//!    compares the converted tree with what format 2 defines instead).
38//! 5. Describe the format in `docs/site/reference/events.md` (History
39//!    formats) and the converter in the compatibility spec's notes.
40//!
41//! The test-only step from format 0 (`TEST_CONVERTERS`) shows the shape of
42//! a step and exercises the chain, the tree conversion, and the backups.
43
44use cookie_agent_protocol::HISTORY_FORMAT;
45use serde_json::Value;
46
47#[cfg(any(test, feature = "test-support"))]
48mod v0_to_v1;
49
50/// One conversion step, from format `from` to `from + 1`.
51pub struct Converter {
52    pub from: u32,
53    /// The log's lines in format `from + 1`. The chain stamps the new
54    /// format on `session_created` afterwards.
55    pub convert: fn(Vec<Value>) -> Result<Vec<Value>, String>,
56}
57
58/// Every conversion step this build has, by the format it reads.
59const CONVERTERS: &[Converter] = &[];
60
61/// Steps only tests use: format 0 exists only to exercise the machinery
62/// before the first real converter does.
63#[cfg(any(test, feature = "test-support"))]
64const TEST_CONVERTERS: &[Converter] = &[Converter {
65    from: 0,
66    convert: v0_to_v1::convert,
67}];
68#[cfg(not(any(test, feature = "test-support")))]
69const TEST_CONVERTERS: &[Converter] = &[];
70
71fn converter(from: u32) -> Option<&'static Converter> {
72    CONVERTERS
73        .iter()
74        .chain(TEST_CONVERTERS)
75        .find(|converter| converter.from == from)
76}
77
78/// Why a log could not be converted.
79#[derive(Debug, thiserror::Error)]
80pub enum ConvertError {
81    #[error("no conversion from history format {0} exists")]
82    NoPath(u32),
83    #[error("converting history format {from} to {} failed: {message}", from + 1)]
84    Failed { from: u32, message: String },
85}
86
87/// The history format `session_created`, the first line, declares.
88pub fn history_format(first: &Value) -> Option<u32> {
89    first
90        .get("payload")
91        .filter(|payload| payload.get("type").and_then(Value::as_str) == Some("session_created"))?
92        .get("history_format")
93        .and_then(Value::as_u64)
94        .map(|format| u32::try_from(format).unwrap_or(u32::MAX))
95}
96
97/// Whether a log in `format` can be converted to this build's.
98pub fn convertible(format: u32) -> bool {
99    format < HISTORY_FORMAT && (format..HISTORY_FORMAT).all(|from| converter(from).is_some())
100}
101
102/// Converts a log's lines from `from` to [`HISTORY_FORMAT`], one step at a
103/// time.
104pub fn convert(from: u32, mut lines: Vec<Value>) -> Result<Vec<Value>, ConvertError> {
105    for format in from..HISTORY_FORMAT {
106        let step = converter(format).ok_or(ConvertError::NoPath(format))?;
107        lines = (step.convert)(lines).map_err(|message| ConvertError::Failed {
108            from: format,
109            message,
110        })?;
111        let creation = lines
112            .first_mut()
113            .and_then(|line| line.get_mut("payload"))
114            .filter(|payload| {
115                payload.get("type").and_then(Value::as_str) == Some("session_created")
116            })
117            .ok_or_else(|| ConvertError::Failed {
118                from: format,
119                message: "the converted log does not start with session_created".into(),
120            })?;
121        creation["history_format"] = Value::from(format + 1);
122    }
123    Ok(lines)
124}

126/// Converts the complete lines of a log in `from` to [`HISTORY_FORMAT`] and
127/// returns the new log's bytes. A line that is not JSON is left out; the
128/// original keeps it.
129pub fn convert_bytes(from: u32, bytes: &[u8]) -> Result<Vec<u8>, ConvertError> {
130    let lines = bytes
131        .split(|byte| *byte == b'\n')
132        .filter(|line| !line.is_empty())
133        .filter_map(|line| serde_json::from_slice::<Value>(line).ok())
134        .collect::<Vec<_>>();
135    let mut converted = Vec::with_capacity(bytes.len());
136    for line in convert(from, lines)? {
137        serde_json::to_writer(&mut converted, &line).map_err(|error| ConvertError::Failed {
138            from,
139            message: error.to_string(),
140        })?;
141        converted.push(b'\n');
142    }
143    Ok(converted)
144}
