1//! History format conversion (`docs/site/specs/compatibility-v1.md`
2//! section 8.7).
3//!
4//! A log written under an older history format is converted to
5//! [`HISTORY_FORMAT`] when a newer build opens it. Each step is a function
6//! from the old log's JSON lines to the next format's lines, in
7//! `v<N>_to_v<N+1>.rs`, registered in `CONVERTERS` and kept, so the steps
8//! chain: a format-1 log opened by a format-3 build converts 1 to 2 to 3.
9//! Converters never use old Rust types; they read and write JSON. A step
10//! may move values out of the log into the tree's blobs ([`Blobs`], by
11//! digest): the caller writes them to `blobs/` before the converted log
12//! (`convert_tree`), or keeps them in memory for a read-only open.
13//!
14//! A log without a `history_format` predates history formats and is not
15//! converted.
16//!
17//! # Adding a converter
18//!
19//! When a change cannot follow the additive rules of section 8.4, the
20//! history format bumps from `N` to `N + 1` and this module gains
21//! `v<N>_to_v<N+1>.rs`, as `v1_to_v2.rs` did for format 2:
22//!
23//! 1. Bump [`HISTORY_FORMAT`] in `cookie_agent_protocol`.
24//! 2. Write `pub(super) fn convert(lines: Vec<Value>, blobs: &mut Blobs)
25//!    -> Result<Vec<Value>, String>` in `v<N>_to_v<N+1>.rs`: it reads
26//!    format-`N` lines as JSON (envelopes and payloads exactly as a
27//!    format-`N` writer stored them, the storage encodings included, such as
28//!    a tool result's `display_span`, `text_span`, and `lines_span`, and
29//!    format 2's blob references) and returns format-`N + 1` lines. It names
30//!    no Rust event type, since those are the newest format's; a line it
31//!    does not understand (an unknown or newer record) is copied unchanged.
32//!    The chain stamps the new `history_format` on `session_created`
33//!    afterwards.
34//! 3. Declare the module and register `Converter { from: N, convert }` in
35//!    `CONVERTERS`. Converters are never removed.
36//! 4. Record new golden trees in `crates/engine/tests/fixtures/history/<N +
37//!    1>/` with the golden recorder, beside the older formats' trees, which
38//!    stay frozen: their test then opens each older tree through the
39//!    converters and compares the result with the snapshot it was recorded
40//!    with, wherever the new format does not change what the snapshot shows
41//!    (where it does, the test compares the converted tree with what the new
42//!    format defines instead).
43//! 5. Describe the format in `docs/site/reference/events.md` (History
44//!    formats) and the converter in the compatibility spec's notes.
45//!
46//! The test-only step from format 0 (`TEST_CONVERTERS`) exercises the
47//! chain, the tree conversion, and the backups apart from any real format.
48

58/// One conversion step, from format `from` to `from + 1`.
59pub struct Converter {
60    pub from: u32,
61    /// The log's lines in format `from + 1`, with the values it moved to
62    /// blobs added to the blobs. The chain stamps the new format on
63    /// `session_created` afterwards.
64    pub convert: fn(Vec<Value>, &mut Blobs) -> Result<Vec<Value>, String>,
65}
66
67/// Every conversion step this build has, by the format it reads.
68const CONVERTERS: &[Converter] = &[Converter {
69    from: 1,
70    convert: v1_to_v2::convert,
71}];
72
73/// Steps only tests use: format 0 exists only to exercise the machinery
74/// before the first real converter does.
75#[cfg(any(test, feature = "test-support"))]
76const TEST_CONVERTERS: &[Converter] = &[Converter {
77    from: 0,
78    convert: v0_to_v1::convert,
79}];
80#[cfg(not(any(test, feature = "test-support")))]
81const TEST_CONVERTERS: &[Converter] = &[];
82
83fn converter(from: u32) -> Option<&'static Converter> {
84    CONVERTERS
85        .iter()
86        .chain(TEST_CONVERTERS)
87        .find(|converter| converter.from == from)
88}

90/// Why a log could not be converted.
91#[derive(Debug, thiserror::Error)]
92pub enum ConvertError {
93    #[error("no conversion from history format {0} exists")]
94    NoPath(u32),
95    #[error("converting history format {from} to {} failed: {message}", from + 1)]
96    Failed { from: u32, message: String },
97}
98
99/// The history format `session_created`, the first line, declares.
100pub fn history_format(first: &Value) -> Option<u32> {
101    first
102        .get("payload")
103        .filter(|payload| payload.get("type").and_then(Value::as_str) == Some("session_created"))?
104        .get("history_format")
105        .and_then(Value::as_u64)
106        .map(|format| u32::try_from(format).unwrap_or(u32::MAX))
107}
108
109/// Whether a log in `format` can be converted to this build's.
110pub fn convertible(format: u32) -> bool {
111    format < HISTORY_FORMAT && (format..HISTORY_FORMAT).all(|from| converter(from).is_some())
112}
113
114/// Converts a log's lines from `from` to [`HISTORY_FORMAT`], one step at a
115/// time, returning them with the blobs the steps moved values to.
116pub fn convert(from: u32, mut lines: Vec<Value>) -> Result<(Vec<Value>, Blobs), ConvertError> {
117    let mut blobs = Blobs::new();
118    for format in from..HISTORY_FORMAT {
119        let step = converter(format).ok_or(ConvertError::NoPath(format))?;
120        lines = (step.convert)(lines, &mut blobs).map_err(|message| ConvertError::Failed {
121            from: format,
122            message,
123        })?;
124        let creation = lines
125            .first_mut()
126            .and_then(|line| line.get_mut("payload"))
127            .filter(|payload| {
128                payload.get("type").and_then(Value::as_str) == Some("session_created")
129            })
130            .ok_or_else(|| ConvertError::Failed {
131                from: format,
132                message: "the converted log does not start with session_created".into(),
133            })?;
134        creation["history_format"] = Value::from(format + 1);
135    }
136    Ok((lines, blobs))
137}

139/// Converts the complete lines of a log in `from` to [`HISTORY_FORMAT`] and
140/// returns the new log's bytes, with the blobs it names that the
141/// conversion moved values to. A line that is not JSON is left out; the
142/// original keeps it.
143pub fn convert_bytes(from: u32, bytes: &[u8]) -> Result<(Vec<u8>, Blobs), ConvertError> {
144    let lines = bytes
145        .split(|byte| *byte == b'\n')
146        .filter(|line| !line.is_empty())
147        .filter_map(|line| serde_json::from_slice::<Value>(line).ok())
148        .collect::<Vec<_>>();
149    let (lines, blobs) = convert(from, lines)?;
150    let mut converted = Vec::with_capacity(bytes.len());
151    for line in lines {
152        serde_json::to_writer(&mut converted, &line).map_err(|error| ConvertError::Failed {
153            from,
154            message: error.to_string(),
155        })?;
156        converted.push(b'\n');
157    }
158    Ok((converted, blobs))
159}
