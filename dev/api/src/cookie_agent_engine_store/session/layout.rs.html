29/// A UTC calendar month, the name of a folder of roots (`2026-10`) and of
30/// an activity journal (`2026-10.jsonl`).
31#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
32pub struct Month {
33    year: i16,
34    month: i8,
35}
36
37impl Month {
38    /// The month `timestamp` falls in, in UTC.
39    #[must_use]
40    pub fn of(timestamp: Timestamp) -> Self {
41        let date = timestamp.to_zoned(TimeZone::UTC).date();
42        Self {
43            year: date.year(),
44            month: date.month(),
45        }
46    }
47
48    /// The month of a UUIDv7 id's timestamp; `None` for any other id.
49    #[must_use]
50    pub fn of_id(id: SessionId) -> Option<Self> {
51        if id.0.get_version_num() != 7 {
52            return None;
53        }
54        let (seconds, _) = id.0.get_timestamp()?.to_unix();
55        Timestamp::from_second(i64::try_from(seconds).ok()?)
56            .ok()
57            .map(Self::of)
58    }
59
60    /// Parses a folder or journal name, `yyyy-mm`.
61    #[must_use]
62    pub fn parse(name: &str) -> Option<Self> {
63        let (year, month) = name.split_once('-')?;
64        if year.len() != 4
65            || month.len() != 2
66            || !year
67                .bytes()
68                .chain(month.bytes())
69                .all(|byte| byte.is_ascii_digit())
70        {
71            return None;
72        }
73        let year = year.parse::<i16>().ok()?;
74        let month = month.parse::<i8>().ok()?;
75        ((1..=9999).contains(&year) && (1..=12).contains(&month)).then_some(Self { year, month })
76    }
77
78    /// The month before this one.
79    #[must_use]
80    pub fn previous(self) -> Self {
81        if self.month == 1 {
82            Self {
83                year: self.year - 1,
84                month: 12,
85            }
86        } else {
87            Self {
88                year: self.year,
89                month: self.month - 1,
90            }
91        }
92    }
93
94    /// The first instant of the month.
95    #[must_use]
96    pub fn start(self) -> Timestamp {
97        jiff::civil::Date::new(self.year, self.month, 1)
98            .and_then(|date| date.to_zoned(TimeZone::UTC))
99            .map_or(Timestamp::MIN, |zoned| zoned.timestamp())
100    }
101}
102
103impl std::fmt::Display for Month {
104    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
105        write!(formatter, "{:04}-{:02}", self.year, self.month)
106    }
107}

109/// The month folders under `sessions_dir`, oldest first.
110pub(super) fn month_dirs(sessions_dir: &Path) -> Vec<Month> {
111    let mut months = fs::read_dir(sessions_dir)
112        .map(|entries| {
113            entries
114                .flatten()
115                .filter(|entry| entry.path().is_dir())
116                .filter_map(|entry| Month::parse(entry.file_name().to_str()?))
117                .collect::<Vec<_>>()
118        })
119        .unwrap_or_default();
120    months.sort();
121    months
122}

129/// The session-id-named folders directly inside `directory`.
130fn session_dir_ids(directory: &Path) -> Vec<SessionId> {
131    let Ok(entries) = fs::read_dir(directory) else {
132        return Vec::new();
133    };
134    let mut ids = entries
135        .flatten()
136        .filter(|entry| entry.path().is_dir())
137        .filter_map(|entry| entry.file_name().to_str()?.parse::<SessionId>().ok())
138        .collect::<Vec<_>>();
139    ids.sort();
140    ids

143/// The folder of the root session `id` under `sessions_dir`: its id's month
144/// for a UUIDv7 id. Any other id (none this build creates) is found in the
145/// month folder that holds it, newest first; one that none holds would be
146/// created now, in this month.
147#[must_use]
148pub fn root_dir_in(sessions_dir: &Path, id: SessionId) -> PathBuf {
149    let month = Month::of_id(id)
150        .or_else(|| find_root_month(sessions_dir, id))
151        .unwrap_or_else(|| Month::of(Timestamp::now()));
152    sessions_dir.join(month.to_string()).join(id.to_string())
153}
154
155/// The month folder holding the root `id`, searched newest first.
156pub(super) fn find_root_month(sessions_dir: &Path, id: SessionId) -> Option<Month> {
157    month_dirs(sessions_dir).into_iter().rev().find(|month| {
158        sessions_dir
159            .join(month.to_string())
160            .join(id.to_string())
161            .is_dir()
162    })
163}

165/// `sessions/layout.json`: the layout and history format of the whole store.
166#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
167pub struct StoreLayout {
168    pub version: u32,
169    #[serde(default)]
170    pub history_format: u32,
171    /// The version of the cookie build that wrote the file, which an older
172    /// build refusing the store names.
173    #[serde(default, skip_serializing_if = "Option::is_none")]
174    pub build: Option<String>,
175    /// The trees the migration could not move yet, retried at every start.
176    #[serde(default)]
177    pub pending: Vec<PendingTree>,
178}

180impl StoreLayout {
181    fn current(pending: Vec<PendingTree>) -> Self {
182        Self {
183            version: LAYOUT_VERSION,
184            history_format: cookie_agent_protocol::HISTORY_FORMAT,
185            build: Some(env!("CARGO_PKG_VERSION").to_owned()),
186            pending,
187        }
188    }
189
190    fn is_current(&self) -> bool {
191        self.version == LAYOUT_VERSION
192            && self.history_format == cookie_agent_protocol::HISTORY_FORMAT
193            && self.pending.is_empty()
194    }
195}

215/// Brings the store at `sessions_dir` to this build's layout before
216/// anything opens a session: refuses a newer store, migrates an older one
217/// (or retries its pending trees), and removes what a finished migration or
218/// deletion left behind. A current store costs one read of `layout.json`
219/// and one listing of `sessions/`.
220pub(super) fn prepare(
221    sessions_dir: &Path,
222    board: &StatusBoard,
223) -> Result<PreparedLayout, SessionError> {
224    // A marker that does not read is treated as missing: migrating again
225    // moves nothing that moved already and rebuilds the journals.
226    let existing = match read_layout(sessions_dir) {
227        Err(SessionError::Json { path, source }) => {
228            eprintln!(
229                "cookie-agent: {} does not read ({source}); checking the store again",
230                path.display()
231            );
232            None
233        }
234        existing => existing?,
235    };
236    if let Some(layout) = &existing {
237        refuse_newer(sessions_dir, layout)?;
238    }
239    let layout = match existing {
240        Some(layout) if layout.is_current() => {
241            // Journals someone removed are rebuilt rather than listing
242            // nothing.
243            if !sessions_dir.join(ACTIVITY_DIR).is_dir() && !month_dirs(sessions_dir).is_empty() {
244                eprintln!(
245                    "cookie-agent: the session activity journals are missing; rebuilding them"
246                );
247                let _task = board.start("Rebuilding the session activity journals", true);
248                super::activity::rebuild_journals(sessions_dir)?;
249            }
250            layout
251        }
252        existing => migrate(sessions_dir, existing.as_ref(), board)?,
253    };
254    let deleted = sweep_top_level(sessions_dir, &layout.pending);
255    Ok(PreparedLayout { layout, deleted })
256}

258fn layout_path(sessions_dir: &Path) -> PathBuf {
259    sessions_dir.join(LAYOUT_MARKER_FILE)

262/// `sessions/layout.json`, or `None` before the first v3 start.
263pub fn read_layout(sessions_dir: &Path) -> Result<Option<StoreLayout>, SessionError> {
264    let path = layout_path(sessions_dir);
265    let bytes = match fs::read(&path) {
266        Ok(bytes) => bytes,
267        Err(source) if source.kind() == std::io::ErrorKind::NotFound => return Ok(None),
268        Err(source) => return Err(SessionError::Io { path, source }),
269    };
270    serde_json::from_slice(&bytes)
271        .map(Some)
272        .map_err(|source| SessionError::Json { path, source })
273}

275/// Refuses a store a newer build laid out, naming the build it needs.
276fn refuse_newer(sessions_dir: &Path, layout: &StoreLayout) -> Result<(), SessionError> {
277    if layout.version <= LAYOUT_VERSION
278        && layout.history_format <= cookie_agent_protocol::HISTORY_FORMAT
279    {
280        return Ok(());
281    }
282    Err(SessionError::NewerStore {
283        path: sessions_dir.to_owned(),
284        version: layout.version,
285        history_format: layout.history_format,
286        needed: layout.build.as_deref().map_or_else(
287            || "a newer cookie build".to_owned(),
288            |build| format!("cookie {build} or newer"),
289        ),
290    })
291}

293/// Writes `layout.json` durably: the file, then its directory. This is the
294/// migration's commit point.
295fn write_layout(sessions_dir: &Path, layout: &StoreLayout) -> Result<(), SessionError> {
296    let path = layout_path(sessions_dir);
297    let bytes = serde_json::to_vec_pretty(layout).map_err(|source| SessionError::Json {
298        path: path.clone(),
299        source,
300    })?;
301    replace_cache_file(&path, &bytes, CacheSync::Durable)?;
302    // The rename above syncs the directory on Unix; Windows has no
303    // directory sync to offer beyond this best effort.
304    fsync_directory(sessions_dir)?;
305    Ok(())
306}

310    root: SessionId,
311    dir: PathBuf,

314/// Every v2 work-dir folder under `sessions_dir`: a folder that is not a
315/// month, not the activity journals, and not hidden.
316fn old_work_dirs(sessions_dir: &Path) -> Vec<PathBuf> {
317    let Ok(entries) = fs::read_dir(sessions_dir) else {
318        return Vec::new();
319    };
320    let mut dirs = entries
321        .flatten()
322        .filter(|entry| entry.path().is_dir())
323        .filter(|entry| {
324            entry.file_name().to_str().is_some_and(|name| {
325                Month::parse(name).is_none() && name != ACTIVITY_DIR && !name.starts_with('.')
326            })
327        })
328        .map(|entry| entry.path())
329        .collect::<Vec<_>>();
330    dirs.sort();
331    dirs
332}

334/// Converts every tree in a month folder an older history format wrote (or,
335/// for a store of this format, the pending ones), moves every root folder
336/// of every v2 work-dir folder into its month, converting it too, each tree
337/// one step, then writes the activity journals, moves the runtime revision
338/// index up, and commits by writing `layout.json`. A step a crash cut short
339/// is repeated at the next start, which finds the trees it finished current.
340fn migrate(
341    sessions_dir: &Path,
342    existing: Option<&StoreLayout>,
343    board: &StatusBoard,
344) -> Result<StoreLayout, SessionError> {
345    let started = Instant::now();
346    // A store laid out before v3 has no journals yet; one retrying its
347    // pending trees has them and only gains entries.
348    let first = existing.is_none_or(|layout| layout.version < LAYOUT_VERSION);
349    let work_dirs = old_work_dirs(sessions_dir);
350    let trees = work_dirs
351        .iter()
352        .flat_map(|work_dir| {
353            session_dir_ids(work_dir)
354                .into_iter()
355                .map(move |root| OldTree {
356                    root,
357                    dir: work_dir.join(root.to_string()),
358                })
359        })
360        .collect::<Vec<_>>();
361    // The trees in month folders an older history format wrote are
362    // converted where they are, every one of them once; a retry converts
363    // the pending ones only.
364    let format = cookie_agent_protocol::HISTORY_FORMAT;
365    let older_format = existing.is_none_or(|layout| layout.history_format < format);
366    let in_months = if older_format {
367        month_dirs(sessions_dir)
368            .into_iter()
369            .flat_map(|month| {
370                let month_dir = sessions_dir.join(month.to_string());
371                month_root_ids(sessions_dir, month)
372                    .into_iter()
373                    .map(move |root| OldTree {
374                        root,
375                        dir: month_dir.join(root.to_string()),
376                    })
377            })
378            .collect::<Vec<_>>()
379    } else {
380        existing
381            .map(|layout| layout.pending.as_slice())
382            .unwrap_or(&[])
383            .iter()
384            .filter(|tree| {
385                Path::new(&tree.path)
386                    .components()
387                    .next()
388                    .and_then(|first| Month::parse(first.as_os_str().to_str()?))
389                    .is_some()
390            })
391            .map(|tree| OldTree {
392                root: tree.root_session_id,
393                dir: sessions_dir.join(&tree.path),
394            })
395            .filter(|tree| tree.dir.is_dir())
396            .collect()
397    };
398    // A new store has nothing to move and says nothing.
399    if !trees.is_empty() {
400        eprintln!(
401            "cookie-agent: moving {} session tree(s) to session storage layout {LAYOUT_VERSION}",
402            trees.len()
403        );
404    }
405    check_free_space(sessions_dir, trees.iter().chain(&in_months))?;
406    let mut pending = Vec::new();
407    let converting = Instant::now();
408    let mut converted = 0;
409    if !in_months.is_empty() {
410        eprintln!(
411            "cookie-agent: converting {} session tree(s) to history format {format}",
412            in_months.len()
413        );
414    }
415    let converting_task = (!in_months.is_empty()).then(|| {
416        let task = board.start(
417            format!("Converting sessions to history format {format}"),
418            true,
419        );
420        task.set_total(in_months.len() as u64, "trees");
421        task
422    });
423    for tree in &in_months {
424        if let Some(task) = &converting_task {
425            task.set_detail(tree_detail(sessions_dir, &tree.dir));
426        }
427        let converted_one = convert_in_place(sessions_dir, tree);
428        if let Some(task) = &converting_task {
429            task.advance(1);
430        }
431        match converted_one {
432            Ok(true) => converted += 1,
433            Ok(false) => {}
434            Err(error) => pending.push(pending_tree(sessions_dir, tree, error)),
435        }
436    }
437    if !in_months.is_empty() {
438        eprintln!(
439            "cookie-agent: history format {format}: converted {converted} of {} tree(s) in \
440             {:.1?}, {} pending",
441            in_months.len(),
442            converting.elapsed(),
443            pending.len()
444        );
445    }
446    drop(converting_task);
447    let mut moved = Vec::new();
448    let moving_task = (!trees.is_empty()).then(|| {
449        let task = board.start("Moving sessions to the new storage layout", true);
450        task.set_total(trees.len() as u64, "trees");
451        task
452    });
453    for tree in &trees {
454        if let Some(task) = &moving_task {
455            task.set_detail(tree_detail(sessions_dir, &tree.dir));
456        }
457        let moved_one = move_tree(sessions_dir, tree);
458        if let Some(task) = &moving_task {
459            task.advance(1);
460        }
461        match moved_one {
462            Ok(dir) => moved.push((tree.root, dir)),
463            Err(error) => pending.push(pending_tree(sessions_dir, tree, error)),
464        }
465    }
466    // A tree an earlier start moved before a crash kept it from recording
467    // that is in its month already, but has no journal entry yet.
468    for earlier in existing
469        .map(|layout| layout.pending.as_slice())
470        .unwrap_or(&[])
471    {
472        let root = earlier.root_session_id;
473        let dir = root_dir_in(sessions_dir, root);
474        if !sessions_dir.join(&earlier.path).exists()
475            && dir.is_dir()
476            && !moved.iter().any(|(id, _)| *id == root)
477        {
478            moved.push((root, dir));
479        }
480    }
481    drop(moving_task);
482    let _finishing = board.start("Recording the new session storage layout", true);
483    sync_month_dirs(sessions_dir, &moved)?;
484    for work_dir in &work_dirs {
485        fsync_directory(work_dir)?;
486    }
487    if first {
488        super::activity::rebuild_journals(sessions_dir)?;
489    } else {
490        let entries = moved
491            .iter()
492            .filter_map(|(root, dir)| root_meta(*root, dir))
493            .collect::<Vec<_>>();
494        super::activity::append_migrated(sessions_dir, &entries)?;
495    }
496    merge_runtime_revisions(sessions_dir, &work_dirs)?;
497    let layout = StoreLayout::current(pending);
498    write_layout(sessions_dir, &layout)?;
499    if !trees.is_empty() {
500        eprintln!(
501            "cookie-agent: session storage layout {LAYOUT_VERSION}: moved {} tree(s) in {:.1?}, {} pending",
502            moved.len(),
503            started.elapsed(),
504            layout.pending.len()
505        );
506    }
507    Ok(layout)
508}

510/// The folder a tree being migrated is in, relative to `sessions/`, for
511/// the task's detail.
512fn tree_detail(sessions_dir: &Path, dir: &Path) -> Option<String> {
513    Some(
514        dir.strip_prefix(sessions_dir)
515            .ok()?
516            .parent()?
517            .to_string_lossy()
518            .into_owned(),
519    )

522/// Syncs every month folder a tree moved into, and `sessions/` itself, so
523/// the moves are durable before the commit point.
524fn sync_month_dirs(
525    sessions_dir: &Path,
526    moved: &[(SessionId, PathBuf)],
527) -> Result<(), SessionError> {
528    let months = moved
529        .iter()
530        .filter_map(|(_, dir)| dir.parent().map(Path::to_path_buf))
531        .collect::<std::collections::BTreeSet<_>>();
532    for month in months {
533        fsync_directory(&month)?;
534    }
535    fsync_directory(sessions_dir)?;
536    Ok(())
537}

539/// The month a root files under: its id's, or for an id that is not a
540/// UUIDv7 the time its log was created, else its `metadata`'s last
541/// activity, else now.
542fn tree_month(root: SessionId, dir: &Path) -> Month {
543    Month::of_id(root)
544        .or_else(|| first_event_time(&dir.join(EVENTS_FILE)).map(Month::of))
545        .or_else(|| {
546            read_cache(&meta_path(dir), &dir.join(EVENTS_FILE))
547                .ok()
548                .map(|meta| Month::of(meta.last_activity))
549        })
550        .unwrap_or_else(|| Month::of(Timestamp::now()))
551}

553/// The timestamp of the first record in the log at `path`.
554fn first_event_time(path: &Path) -> Option<Timestamp> {
555    use std::io::BufRead as _;
556    let mut first = Vec::new();
557    std::io::BufReader::new(fs::File::open(path).ok()?)
558        .read_until(b'\n', &mut first)
559        .ok()?;
560    serde_json::from_slice::<serde_json::Value>(&first)
561        .ok()?
562        .get("timestamp")?
563        .as_str()?
564        .parse()
565        .ok()
566}

588/// One conversion step for a tree already in its month folder: takes the
589/// tree's lock, converts its logs to this build's history format, and backs
590/// up the originals. Returns whether anything changed; an error leaves the
591/// tree as it was.
592fn convert_in_place(sessions_dir: &Path, tree: &OldTree) -> Result<bool, String> {
593    let _lock = match try_acquire(&tree.dir) {
594        Ok(SessionOwnership::Owned(lock)) => lock,
595        Ok(SessionOwnership::Foreign) => {
596            return Err("another cookie process holds the tree".to_owned());
597        }
598        Err(error) => return Err(format!("its lock does not open: {error}")),
599    };
600    let conversion =
601        convert::convert_tree(tree.root, &tree.dir).map_err(|error| error.to_string())?;
602    back_up_converted(sessions_dir, tree);
603    Ok(!conversion.converted.is_empty() || conversion.artifacts_moved)
604}

606/// One migration step: takes the tree's lock, converts its logs to this
607/// build's history format, backs up the originals, and renames the root
608/// folder into its month. An error leaves the tree where it was.
609fn move_tree(sessions_dir: &Path, tree: &OldTree) -> Result<PathBuf, String> {
610    let month = tree_month(tree.root, &tree.dir);
611    let month_dir = sessions_dir.join(month.to_string());
612    let destination = month_dir.join(tree.root.to_string());
613    if destination.symlink_metadata().is_ok() {
614        return Err(format!("{} already exists", destination.display()));
615    }
616    let lock = match try_acquire(&tree.dir) {
617        Ok(SessionOwnership::Owned(lock)) => lock,
618        Ok(SessionOwnership::Foreign) => {
619            return Err("another cookie process holds the tree".to_owned());
620        }
621        Err(error) => return Err(format!("its lock does not open: {error}")),
622    };
623    convert::convert_tree(tree.root, &tree.dir).map_err(|error| error.to_string())?;
624    back_up_converted(sessions_dir, tree);
625    // Windows refuses to rename a folder with an open file; the tree is
626    // this process's alone until the daemon accepts connections.
627    drop(lock);
628    #[cfg(windows)]
629    {
630        let _ = fs::remove_file(crate::ownership::owner_lock_path(&tree.dir));
631    }
632    #[cfg(unix)]
633    create_unix_session_directory_all(&month_dir).map_err(|error| error.to_string())?;
634    #[cfg(windows)]
635    create_windows_session_directory(&month_dir).map_err(|error| error.to_string())?;
636    fs::rename(&tree.dir, &destination).map_err(|error| error.to_string())?;
637    Ok(destination)
638}

640/// Moves the originals of the tree's converted logs (each member's
641/// `events.jsonl.v<N>.bak`, also one a conversion a crash cut short left)
642/// into `sessions/.backup-v2/<root-id>/`, which the migration deletes once
643/// it commits. One that does not move stays beside its log.
644fn back_up_converted(sessions_dir: &Path, tree: &OldTree) {
645    let mut members = vec![PathBuf::new()];
646    members.extend(
647        session_dir_ids(&tree.dir.join(SUBAGENTS_DIR))
648            .into_iter()
649            .map(|child| Path::new(SUBAGENTS_DIR).join(child.to_string())),
650    );
651    for relative in members {
652        let Ok(entries) = fs::read_dir(tree.dir.join(&relative)) else {
653            continue;
654        };
655        for entry in entries.flatten() {
656            let name = entry.file_name();
657            if !name.to_str().is_some_and(convert::is_log_backup) {
658                continue;
659            }
660            let original = entry.path();
661            let backup = sessions_dir
662                .join(BACKUP_DIR)
663                .join(tree.root.to_string())
664                .join(&relative);
665            let moved = fs::create_dir_all(&backup)
666                .and_then(|()| fs::rename(&original, backup.join(&name)));
667            if let Err(error) = moved {
668                eprintln!(
669                    "cookie-agent: the original of {} stays beside it: {error}",
670                    original.display()
671                );
672            }
673        }
674    }
675}

677/// Refuses to migrate when the disk cannot hold the logs it converts: a
678/// converted log and the blobs it names take about what the original does,
679/// which stays until the migration commits.
680fn check_free_space<'a>(
681    sessions_dir: &Path,
682    trees: impl IntoIterator<Item = &'a OldTree>,
683) -> Result<(), SessionError> {
684    let needed = trees
685        .into_iter()
686        .flat_map(|tree| {
687            let mut logs = vec![tree.dir.join(EVENTS_FILE)];
688            logs.extend(
689                session_dir_ids(&tree.dir.join(SUBAGENTS_DIR))
690                    .into_iter()
691                    .map(|child| {
692                        tree.dir
693                            .join(SUBAGENTS_DIR)
694                            .join(child.to_string())
695                            .join(EVENTS_FILE)
696                    }),
697            );
698            logs
699        })
700        .filter(|log| {
701            matches!(
702                convert::log_format(log),
703                Ok(Some(Some(format)))
704                    if format != cookie_agent_protocol::HISTORY_FORMAT
705                        && crate::events::convert::convertible(format)
706            )
707        })
708        .filter_map(|log| fs::metadata(log).ok().map(|metadata| metadata.len()))
709        .sum::<u64>();
710    if needed == 0 {
711        return Ok(());
712    }
713    match available_space(sessions_dir) {
714        Some(available) if available < needed => Err(SessionError::StoreSpace {
715            path: sessions_dir.to_owned(),
716            needed,
717            available,
718        }),
719        _ => Ok(()),
720    }
721}

723/// Bytes an unprivileged writer can still use on the filesystem of `path`.
724#[cfg(unix)]
725fn available_space(path: &Path) -> Option<u64> {
726    let path = std::ffi::CString::new(path.as_os_str().as_bytes()).ok()?;
727    let mut stats = std::mem::MaybeUninit::<libc::statvfs>::uninit();
728    // SAFETY: `path` is a NUL-terminated string and `stats` is written by
729    // the call before it is read.
730    let stats = unsafe {
731        if libc::statvfs(path.as_ptr(), stats.as_mut_ptr()) != 0 {
732            return None;
733        }
734        stats.assume_init()
735    };
736    #[allow(clippy::useless_conversion)]
737    Some(u64::from(stats.f_bavail).saturating_mul(u64::from(stats.f_frsize)))
738}
739
740/// Bytes the calling user can still use on the volume of `path`.
741#[cfg(windows)]
742fn available_space(path: &Path) -> Option<u64> {
743    use std::os::windows::ffi::OsStrExt as _;
744
745    use windows_sys::Win32::Storage::FileSystem::GetDiskFreeSpaceExW;
746
747    let wide = path
748        .as_os_str()
749        .encode_wide()
750        .chain(std::iter::once(0))
751        .collect::<Vec<u16>>();
752    let mut available = 0_u64;
753    // SAFETY: `wide` is NUL-terminated and outlives the call, which writes
754    // only the one counter asked for; the others may be null.
755    let succeeded = unsafe {
756        GetDiskFreeSpaceExW(
757            wide.as_ptr(),
758            &mut available,
759            std::ptr::null_mut(),
760            std::ptr::null_mut(),
761        )
762    };
763    (succeeded != 0).then_some(available)
764}

766/// The list entry of the root in `dir`: its `metadata`, or a fold of its
767/// log when that does not read. `None` for a folder that is not a root
768/// session at all.
769pub(super) fn root_meta(root: SessionId, dir: &Path) -> Option<SessionMeta> {
770    let meta = match read_cache(&meta_path(dir), &dir.join(EVENTS_FILE)) {
771        Ok(meta) if meta.session_id == root => meta,
772        _ => match EventLog::open_read_only(dir.join(EVENTS_FILE), root)
773            .map_err(SessionError::from)
774            .and_then(projection)
775        {
776            Ok(projection) => projection.meta.clone(),
777            Err(error) => {
778                eprintln!("cookie-agent: session {root} is not listed: {error}");
779                return None;
780            }
781        },
782    };
783    matches!(meta.origin, SessionOrigin::Root).then_some(meta)
784}

786/// Writes `sessions/runtime-revisions.jsonl` from the index it already
787/// holds and each work-dir folder's: every record once, and of two that
788/// map one runtime revision differently, the first.
789fn merge_runtime_revisions(sessions_dir: &Path, work_dirs: &[PathBuf]) -> Result<(), SessionError> {
790    let target = sessions_dir.join(RUNTIME_REVISIONS_FILE);
791    let sources = std::iter::once(target.clone())
792        .chain(
793            work_dirs
794                .iter()
795                .map(|dir| dir.join(V2_RUNTIME_REVISIONS_FILE)),
796        )
797        .filter(|path| path.is_file())
798        .collect::<Vec<_>>();
799    if sources.iter().all(|path| *path == target) {
800        return Ok(());
801    }
802    let mut kept = Vec::<serde_json::Value>::new();
803    let mut mapped = HashMap::<String, serde_json::Value>::new();
804    for source in &sources {
805        let bytes = fs::read(source).map_err(|error| SessionError::Io {
806            path: source.clone(),
807            source: error,
808        })?;
809        for line in bytes.split(|byte| *byte == b'\n') {
810            let Ok(record) = serde_json::from_slice::<serde_json::Value>(line) else {
811                continue;
812            };
813            let Some(revision) = record.get("runtime_revision").map(ToString::to_string) else {
814                continue;
815            };
816            let model = record
817                .get("model_runtime_revision")
818                .cloned()
819                .unwrap_or_default();
820            match mapped.get(&revision) {
821                Some(known) if *known != model => continue,
822                _ if kept.contains(&record) => continue,
823                _ => {
824                    mapped.insert(revision, model);
825                    kept.push(record);
826                }
827            }
828        }
829    }
830    let mut bytes = Vec::new();
831    for record in &kept {
832        serde_json::to_writer(&mut bytes, record).map_err(|source| SessionError::Json {
833            path: target.clone(),
834            source,
835        })?;
836        bytes.push(b'\n');
837    }
838    replace_cache_file(&target, &bytes, CacheSync::Durable)
839}

841/// Removes what a finished migration or an interrupted deletion left in
842/// `sessions/`: deletion tombstones, the backups, staged files, and the
843/// v2 work-dir folders no pending tree lives in. Only `sessions/` itself
844/// and such leftovers are read. Returns the roots of the tombstones it
845/// removed.
846fn sweep_top_level(sessions_dir: &Path, pending: &[PendingTree]) -> Vec<SessionId> {
847    let deleted = delete::sweep_tombstones(sessions_dir);
848    let backup = sessions_dir.join(BACKUP_DIR);
849    remove_backed_up_obsolete_blobs(sessions_dir, &backup);
850    if backup.exists()
851        && let Err(error) = fs::remove_dir_all(&backup)
852    {
853        eprintln!(
854            "cookie-agent: {} could not be removed: {error}",
855            backup.display()
856        );
857    }
858    remove_staged_files(sessions_dir);
859    for work_dir in old_work_dirs(sessions_dir) {
860        let keeps_pending = pending
861            .iter()
862            .any(|tree| sessions_dir.join(&tree.path).starts_with(&work_dir));
863        if !keeps_pending {
864            remove_emptied_work_dir(&work_dir);
865        }
866    }
867    deleted
868}

870/// Deletes the blobs the trees in `backup` stopped naming, before their backups
871/// go, so a crash in between does this again at the next start. A tree another
872/// process holds keeps them.
873fn remove_backed_up_obsolete_blobs(sessions_dir: &Path, backup: &Path) {
874    for root in session_dir_ids(backup) {
875        let dir = root_dir_in(sessions_dir, root);
876        if !dir.is_dir() {
877            continue;
878        }
879        let removed = match try_acquire(&dir) {
880            Ok(SessionOwnership::Owned(_lock)) => convert::remove_obsolete_blobs(&dir),
881            Ok(SessionOwnership::Foreign) => continue,
882            Err(source) => Err(SessionError::Io {
883                path: dir.clone(),
884                source,
885            }),
886        };
887        if let Err(error) = removed {
888            eprintln!(
889                "cookie-agent: the old blobs of {} stay: {error}",
890                dir.display()
891            );
892        }
893    }
894}

896/// Removes the temporaries a replaced file staged in `directory` and a
897/// crash left behind.
898fn remove_staged_files(directory: &Path) {
899    let Ok(entries) = fs::read_dir(directory) else {
900        return;
901    };
902    for entry in entries.flatten() {
903        let name = entry.file_name();
904        let name = name.to_string_lossy();
905        if name.starts_with('.') && name.ends_with(".tmp") && entry.path().is_file() {
906            let _ = fs::remove_file(entry.path());
907        }
908    }
909}

911/// Removes a v2 work-dir folder whose trees all moved: the files the v2
912/// layout kept there, then the folder. Anything else keeps it, and is named.
913fn remove_emptied_work_dir(work_dir: &Path) {
914    delete::sweep_tombstones(work_dir);
915    remove_staged_files(work_dir);
916    let Ok(entries) = fs::read_dir(work_dir) else {
917        return;
918    };
919    let mut kept = Vec::new();
920    for entry in entries.flatten() {
921        let name = entry.file_name().to_string_lossy().into_owned();
922        let known = V2_WORK_DIR_FILES.contains(&name.as_str())
923            || name.ends_with(crate::ownership::OWNER_LOCK_SUFFIX);
924        if known && entry.path().is_file() {
925            let _ = fs::remove_file(entry.path());
926        } else {
927            kept.push(name);
928        }
929    }
930    if kept.is_empty() {
931        if let Err(error) = fs::remove_dir(work_dir) {
932            eprintln!(
933                "cookie-agent: {} could not be removed: {error}",
934                work_dir.display()
935            );
936        }
937    } else {
938        kept.sort();
939        eprintln!(
940            "cookie-agent: {} stays: it still holds {}",
941            work_dir.display(),
942            kept.join(", ")
943        );
944    }
