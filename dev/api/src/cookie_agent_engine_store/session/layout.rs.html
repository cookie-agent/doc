28/// A UTC calendar month, the name of a folder of roots (`2026-10`) and of
29/// an activity journal (`2026-10.jsonl`).
30#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
31pub struct Month {
32    year: i16,
33    month: i8,
34}
35
36impl Month {
37    /// The month `timestamp` falls in, in UTC.
38    #[must_use]
39    pub fn of(timestamp: Timestamp) -> Self {
40        let date = timestamp.to_zoned(TimeZone::UTC).date();
41        Self {
42            year: date.year(),
43            month: date.month(),
44        }
45    }
46
47    /// The month of a UUIDv7 id's timestamp; `None` for any other id.
48    #[must_use]
49    pub fn of_id(id: SessionId) -> Option<Self> {
50        if id.0.get_version_num() != 7 {
51            return None;
52        }
53        let (seconds, _) = id.0.get_timestamp()?.to_unix();
54        Timestamp::from_second(i64::try_from(seconds).ok()?)
55            .ok()
56            .map(Self::of)
57    }
58
59    /// Parses a folder or journal name, `yyyy-mm`.
60    #[must_use]
61    pub fn parse(name: &str) -> Option<Self> {
62        let (year, month) = name.split_once('-')?;
63        if year.len() != 4
64            || month.len() != 2
65            || !year
66                .bytes()
67                .chain(month.bytes())
68                .all(|byte| byte.is_ascii_digit())
69        {
70            return None;
71        }
72        let year = year.parse::<i16>().ok()?;
73        let month = month.parse::<i8>().ok()?;
74        ((1..=9999).contains(&year) && (1..=12).contains(&month)).then_some(Self { year, month })
75    }
76
77    /// The month before this one.
78    #[must_use]
79    pub fn previous(self) -> Self {
80        if self.month == 1 {
81            Self {
82                year: self.year - 1,
83                month: 12,
84            }
85        } else {
86            Self {
87                year: self.year,
88                month: self.month - 1,
89            }
90        }
91    }
92
93    /// The first instant of the month.
94    #[must_use]
95    pub fn start(self) -> Timestamp {
96        jiff::civil::Date::new(self.year, self.month, 1)
97            .and_then(|date| date.to_zoned(TimeZone::UTC))
98            .map_or(Timestamp::MIN, |zoned| zoned.timestamp())
99    }
100}

102impl std::fmt::Display for Month {
103    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
104        write!(formatter, "{:04}-{:02}", self.year, self.month)
105    }
106}

108/// The month folders under `sessions_dir`, oldest first.
109pub(super) fn month_dirs(sessions_dir: &Path) -> Vec<Month> {
110    let mut months = fs::read_dir(sessions_dir)
111        .map(|entries| {
112            entries
113                .flatten()
114                .filter(|entry| entry.path().is_dir())
115                .filter_map(|entry| Month::parse(entry.file_name().to_str()?))
116                .collect::<Vec<_>>()
117        })
118        .unwrap_or_default();
119    months.sort();
120    months
121}

128/// The session-id-named folders directly inside `directory`.
129fn session_dir_ids(directory: &Path) -> Vec<SessionId> {
130    let Ok(entries) = fs::read_dir(directory) else {

132    };
133    let mut ids = entries
134        .flatten()
135        .filter(|entry| entry.path().is_dir())

142/// The folder of the root session `id` under `sessions_dir`: its id's month
143/// for a UUIDv7 id. Any other id (none this build creates) is found in the
144/// month folder that holds it, newest first; one that none holds would be
145/// created now, in this month.
146#[must_use]
147pub fn root_dir_in(sessions_dir: &Path, id: SessionId) -> PathBuf {
148    let month = Month::of_id(id)
149        .or_else(|| find_root_month(sessions_dir, id))
150        .unwrap_or_else(|| Month::of(Timestamp::now()));
151    sessions_dir.join(month.to_string()).join(id.to_string())
152}
153
154/// The month folder holding the root `id`, searched newest first.
155pub(super) fn find_root_month(sessions_dir: &Path, id: SessionId) -> Option<Month> {
156    month_dirs(sessions_dir).into_iter().rev().find(|month| {
157        sessions_dir
158            .join(month.to_string())
159            .join(id.to_string())
160            .is_dir()
161    })
162}

164/// `sessions/layout.json`: the layout and history format of the whole store.
165#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
166pub struct StoreLayout {
167    pub version: u32,
168    #[serde(default)]
169    pub history_format: u32,
170    /// The version of the cookie build that wrote the file, which an older
171    /// build refusing the store names.
172    #[serde(default, skip_serializing_if = "Option::is_none")]
173    pub build: Option<String>,
174    /// The trees the migration could not move yet, retried at every start.
175    #[serde(default)]
176    pub pending: Vec<PendingTree>,
177}
178
179impl StoreLayout {
180    fn current(pending: Vec<PendingTree>) -> Self {
181        Self {
182            version: LAYOUT_VERSION,
183            history_format: cookie_agent_protocol::HISTORY_FORMAT,
184            build: Some(env!("CARGO_PKG_VERSION").to_owned()),
185            pending,
186        }
187    }
188
189    fn is_current(&self) -> bool {
190        self.version == LAYOUT_VERSION
191            && self.history_format == cookie_agent_protocol::HISTORY_FORMAT
192            && self.pending.is_empty()
193    }
194}

214/// Brings the store at `sessions_dir` to this build's layout before
215/// anything opens a session: refuses a newer store, migrates an older one
216/// (or retries its pending trees), and removes what a finished migration or
217/// deletion left behind. A current store costs one read of `layout.json`
218/// and one listing of `sessions/`.
219pub(super) fn prepare(sessions_dir: &Path) -> Result<PreparedLayout, SessionError> {
220    // A marker that does not read is treated as missing: migrating again
221    // moves nothing that moved already and rebuilds the journals.
222    let existing = match read_layout(sessions_dir) {
223        Err(SessionError::Json { path, source }) => {
224            eprintln!(
225                "cookie-agent: {} does not read ({source}); checking the store again",
226                path.display()
227            );
228            None
229        }
230        existing => existing?,
231    };
232    if let Some(layout) = &existing {
233        refuse_newer(sessions_dir, layout)?;
234    }
235    let layout = match existing {
236        Some(layout) if layout.is_current() => {
237            // Journals someone removed are rebuilt rather than listing
238            // nothing.
239            if !sessions_dir.join(ACTIVITY_DIR).is_dir() && !month_dirs(sessions_dir).is_empty() {
240                eprintln!(
241                    "cookie-agent: the session activity journals are missing; rebuilding them"
242                );
243                super::activity::rebuild_journals(sessions_dir)?;
244            }
245            layout
246        }
247        existing => migrate(sessions_dir, existing.as_ref())?,
248    };
249    let deleted = sweep_top_level(sessions_dir, &layout.pending);
250    Ok(PreparedLayout { layout, deleted })
251}

253fn layout_path(sessions_dir: &Path) -> PathBuf {
254    sessions_dir.join(LAYOUT_MARKER_FILE)
255}

257/// `sessions/layout.json`, or `None` before the first v3 start.
258pub fn read_layout(sessions_dir: &Path) -> Result<Option<StoreLayout>, SessionError> {
259    let path = layout_path(sessions_dir);
260    let bytes = match fs::read(&path) {
261        Ok(bytes) => bytes,
262        Err(source) if source.kind() == std::io::ErrorKind::NotFound => return Ok(None),
263        Err(source) => return Err(SessionError::Io { path, source }),
264    };
265    serde_json::from_slice(&bytes)
266        .map(Some)
267        .map_err(|source| SessionError::Json { path, source })
268}

270/// Refuses a store a newer build laid out, naming the build it needs.
271fn refuse_newer(sessions_dir: &Path, layout: &StoreLayout) -> Result<(), SessionError> {
272    if layout.version <= LAYOUT_VERSION
273        && layout.history_format <= cookie_agent_protocol::HISTORY_FORMAT
274    {
275        return Ok(());
276    }
277    Err(SessionError::NewerStore {
278        path: sessions_dir.to_owned(),
279        version: layout.version,
280        history_format: layout.history_format,
281        needed: layout.build.as_deref().map_or_else(
282            || "a newer cookie build".to_owned(),
283            |build| format!("cookie {build} or newer"),
284        ),
285    })
286}

288/// Writes `layout.json` durably: the file, then its directory. This is the
289/// migration's commit point.
290fn write_layout(sessions_dir: &Path, layout: &StoreLayout) -> Result<(), SessionError> {
291    let path = layout_path(sessions_dir);
292    let bytes = serde_json::to_vec_pretty(layout).map_err(|source| SessionError::Json {
293        path: path.clone(),
294        source,
295    })?;
296    replace_cache_file(&path, &bytes, CacheSync::Durable)?;
297    // The rename above syncs the directory on Unix; Windows has no
298    // directory sync to offer beyond this best effort.
299    fsync_directory(sessions_dir)?;
300    Ok(())
301}

309/// Every v2 work-dir folder under `sessions_dir`: a folder that is not a
310/// month, not the activity journals, and not hidden.
311fn old_work_dirs(sessions_dir: &Path) -> Vec<PathBuf> {
312    let Ok(entries) = fs::read_dir(sessions_dir) else {
313        return Vec::new();
314    };
315    let mut dirs = entries
316        .flatten()
317        .filter(|entry| entry.path().is_dir())
318        .filter(|entry| {
319            entry.file_name().to_str().is_some_and(|name| {
320                Month::parse(name).is_none() && name != ACTIVITY_DIR && !name.starts_with('.')
321            })
322        })
323        .map(|entry| entry.path())
324        .collect::<Vec<_>>();
325    dirs.sort();
326    dirs
327}

329/// Moves every root folder of every v2 work-dir folder into its month, as
330/// one step per tree, then writes the activity journals, moves the runtime
331/// revision index up, and commits by writing `layout.json`.
332fn migrate(
333    sessions_dir: &Path,
334    existing: Option<&StoreLayout>,
335) -> Result<StoreLayout, SessionError> {
336    let started = Instant::now();
337    // A store laid out before v3 has no journals yet; one retrying its
338    // pending trees has them and only gains entries.
339    let first = existing.is_none_or(|layout| layout.version < LAYOUT_VERSION);
340    let work_dirs = old_work_dirs(sessions_dir);
341    let trees = work_dirs
342        .iter()
343        .flat_map(|work_dir| {
344            session_dir_ids(work_dir)
345                .into_iter()
346                .map(move |root| OldTree {
347                    root,
348                    dir: work_dir.join(root.to_string()),
349                })
350        })
351        .collect::<Vec<_>>();
352    // A new store has nothing to move and says nothing.
353    if !trees.is_empty() {
354        eprintln!(
355            "cookie-agent: moving {} session tree(s) to session storage layout {LAYOUT_VERSION}",
356            trees.len()
357        );
358    }
359    check_free_space(sessions_dir, &trees)?;
360    let mut pending = Vec::new();
361    let mut moved = Vec::new();
362    for tree in &trees {
363        match move_tree(sessions_dir, tree) {
364            Ok(dir) => moved.push((tree.root, dir)),
365            Err(error) => {
366                let path = tree
367                    .dir
368                    .strip_prefix(sessions_dir)
369                    .unwrap_or(&tree.dir)
370                    .to_string_lossy()
371                    .into_owned();
372                eprintln!(
373                    "cookie-agent: session tree {} stays in {path}: {error}",
374                    tree.root
375                );
376                pending.push(PendingTree {
377                    root_session_id: tree.root,
378                    path,
379                    error,
380                });
381            }
382        }
383    }
384    // A tree an earlier start moved before a crash kept it from recording
385    // that is in its month already, but has no journal entry yet.
386    for earlier in existing
387        .map(|layout| layout.pending.as_slice())
388        .unwrap_or(&[])
389    {
390        let root = earlier.root_session_id;
391        let dir = root_dir_in(sessions_dir, root);
392        if !sessions_dir.join(&earlier.path).exists()
393            && dir.is_dir()
394            && !moved.iter().any(|(id, _)| *id == root)
395        {
396            moved.push((root, dir));
397        }
398    }
399    sync_month_dirs(sessions_dir, &moved)?;
400    for work_dir in &work_dirs {
401        fsync_directory(work_dir)?;
402    }
403    if first {
404        super::activity::rebuild_journals(sessions_dir)?;
405    } else {
406        let entries = moved
407            .iter()
408            .filter_map(|(root, dir)| root_meta(*root, dir))
409            .collect::<Vec<_>>();
410        super::activity::append_migrated(sessions_dir, &entries)?;
411    }
412    merge_runtime_revisions(sessions_dir, &work_dirs)?;
413    let layout = StoreLayout::current(pending);
414    write_layout(sessions_dir, &layout)?;
415    if !trees.is_empty() {
416        eprintln!(
417            "cookie-agent: session storage layout {LAYOUT_VERSION}: moved {} tree(s) in {:.1?}, {} pending",
418            moved.len(),
419            started.elapsed(),
420            layout.pending.len()
421        );
422    }
423    Ok(layout)
424}

426/// Syncs every month folder a tree moved into, and `sessions/` itself, so
427/// the moves are durable before the commit point.
428fn sync_month_dirs(
429    sessions_dir: &Path,
430    moved: &[(SessionId, PathBuf)],
431) -> Result<(), SessionError> {
432    let months = moved
433        .iter()
434        .filter_map(|(_, dir)| dir.parent().map(Path::to_path_buf))
435        .collect::<std::collections::BTreeSet<_>>();
436    for month in months {
437        fsync_directory(&month)?;
438    }
439    fsync_directory(sessions_dir)?;
440    Ok(())
441}

443/// The month a root files under: its id's, or for an id that is not a
444/// UUIDv7 the time its log was created, else its `metadata`'s last
445/// activity, else now.
446fn tree_month(root: SessionId, dir: &Path) -> Month {
447    Month::of_id(root)
448        .or_else(|| first_event_time(&dir.join(EVENTS_FILE)).map(Month::of))
449        .or_else(|| {
450            read_cache(&meta_path(dir), &dir.join(EVENTS_FILE))
451                .ok()
452                .map(|meta| Month::of(meta.last_activity))
453        })
454        .unwrap_or_else(|| Month::of(Timestamp::now()))
455}

457/// The timestamp of the first record in the log at `path`.
458fn first_event_time(path: &Path) -> Option<Timestamp> {
459    use std::io::BufRead as _;
460    let mut first = Vec::new();
461    std::io::BufReader::new(fs::File::open(path).ok()?)
462        .read_until(b'\n', &mut first)
463        .ok()?;
464    serde_json::from_slice::<serde_json::Value>(&first)
465        .ok()?
466        .get("timestamp")?
467        .as_str()?
468        .parse()
469        .ok()
470}

472/// One migration step: takes the tree's lock, converts its logs to this
473/// build's history format, backs up the originals, and renames the root
474/// folder into its month. An error leaves the tree where it was.
475fn move_tree(sessions_dir: &Path, tree: &OldTree) -> Result<PathBuf, String> {
476    let month = tree_month(tree.root, &tree.dir);
477    let month_dir = sessions_dir.join(month.to_string());
478    let destination = month_dir.join(tree.root.to_string());
479    if destination.symlink_metadata().is_ok() {
480        return Err(format!("{} already exists", destination.display()));
481    }
482    let lock = match try_acquire(&tree.dir) {
483        Ok(SessionOwnership::Owned(lock)) => lock,
484        Ok(SessionOwnership::Foreign) => {
485            return Err("another cookie process holds the tree".to_owned());
486        }
487        Err(error) => return Err(format!("its lock does not open: {error}")),
488    };
489    let conversion =
490        convert::convert_tree(tree.root, &tree.dir).map_err(|error| error.to_string())?;
491    back_up_converted(sessions_dir, tree, &conversion);
492    // Windows refuses to rename a folder with an open file; the tree is
493    // this process's alone until the daemon accepts connections.
494    drop(lock);
495    #[cfg(windows)]
496    {
497        let _ = fs::remove_file(crate::ownership::owner_lock_path(&tree.dir));
498    }
499    #[cfg(unix)]
500    create_unix_session_directory_all(&month_dir).map_err(|error| error.to_string())?;
501    #[cfg(windows)]
502    create_windows_session_directory(&month_dir).map_err(|error| error.to_string())?;
503    fs::rename(&tree.dir, &destination).map_err(|error| error.to_string())?;
504    Ok(destination)
505}

507/// Moves the originals of the logs `conversion` replaced into
508/// `sessions/.backup-v2/<root-id>/`, which the migration deletes once it
509/// commits. One that does not move stays beside its log.
510fn back_up_converted(sessions_dir: &Path, tree: &OldTree, conversion: &convert::TreeConversion) {
511    for (id, format) in &conversion.converted {
512        let relative = if *id == tree.root {
513            PathBuf::new()
514        } else {
515            Path::new(SUBAGENTS_DIR).join(id.to_string())
516        };
517        let name = format!("{EVENTS_FILE}.v{format}.bak");
518        let original = tree.dir.join(&relative).join(&name);
519        let backup = sessions_dir
520            .join(BACKUP_DIR)
521            .join(tree.root.to_string())
522            .join(&relative);
523        let moved =
524            fs::create_dir_all(&backup).and_then(|()| fs::rename(&original, backup.join(&name)));
525        if let Err(error) = moved {
526            eprintln!(
527                "cookie-agent: the original of {} stays beside it: {error}",
528                original.display()
529            );
530        }
531    }
532}

534/// Refuses to migrate when the disk cannot hold the logs it converts.
535fn check_free_space(sessions_dir: &Path, trees: &[OldTree]) -> Result<(), SessionError> {
536    let needed = trees
537        .iter()
538        .flat_map(|tree| {
539            let mut logs = vec![tree.dir.join(EVENTS_FILE)];
540            logs.extend(
541                session_dir_ids(&tree.dir.join(SUBAGENTS_DIR))
542                    .into_iter()
543                    .map(|child| {
544                        tree.dir
545                            .join(SUBAGENTS_DIR)
546                            .join(child.to_string())
547                            .join(EVENTS_FILE)
548                    }),
549            );
550            logs
551        })
552        .filter(|log| {
553            matches!(
554                convert::log_format(log),
555                Ok(Some(Some(format)))
556                    if format != cookie_agent_protocol::HISTORY_FORMAT
557                        && crate::events::convert::convertible(format)
558            )
559        })
560        .filter_map(|log| fs::metadata(log).ok().map(|metadata| metadata.len()))
561        .sum::<u64>();
562    if needed == 0 {
563        return Ok(());
564    }
565    match available_space(sessions_dir) {
566        Some(available) if available < needed => Err(SessionError::StoreSpace {
567            path: sessions_dir.to_owned(),
568            needed,
569            available,
570        }),
571        _ => Ok(()),
572    }
573}

575/// Bytes an unprivileged writer can still use on the filesystem of `path`.
576#[cfg(unix)]
577fn available_space(path: &Path) -> Option<u64> {
578    let path = std::ffi::CString::new(path.as_os_str().as_bytes()).ok()?;
579    let mut stats = std::mem::MaybeUninit::<libc::statvfs>::uninit();
580    // SAFETY: `path` is a NUL-terminated string and `stats` is written by
581    // the call before it is read.
582    let stats = unsafe {
583        if libc::statvfs(path.as_ptr(), stats.as_mut_ptr()) != 0 {
584            return None;
585        }
586        stats.assume_init()
587    };
588    #[allow(clippy::useless_conversion)]
589    Some(u64::from(stats.f_bavail).saturating_mul(u64::from(stats.f_frsize)))
590}

600/// The list entry of the root in `dir`: its `metadata`, or a fold of its
601/// log when that does not read. `None` for a folder that is not a root
602/// session at all.
603pub(super) fn root_meta(root: SessionId, dir: &Path) -> Option<SessionMeta> {
604    let meta = match read_cache(&meta_path(dir), &dir.join(EVENTS_FILE)) {
605        Ok(meta) if meta.session_id == root => meta,
606        _ => match EventLog::open_read_only(dir.join(EVENTS_FILE), root)
607            .map_err(SessionError::from)
608            .and_then(projection)
609        {
610            Ok(projection) => projection.meta,
611            Err(error) => {
612                eprintln!("cookie-agent: session {root} is not listed: {error}");
613                return None;
614            }
615        },
616    };
617    matches!(meta.origin, SessionOrigin::Root).then_some(meta)
618}

620/// Writes `sessions/runtime-revisions.jsonl` from the index it already
621/// holds and each work-dir folder's: every record once, and of two that
622/// map one runtime revision differently, the first.
623fn merge_runtime_revisions(sessions_dir: &Path, work_dirs: &[PathBuf]) -> Result<(), SessionError> {
624    let target = sessions_dir.join(RUNTIME_REVISIONS_FILE);
625    let sources = std::iter::once(target.clone())
626        .chain(
627            work_dirs
628                .iter()
629                .map(|dir| dir.join(V2_RUNTIME_REVISIONS_FILE)),
630        )
631        .filter(|path| path.is_file())
632        .collect::<Vec<_>>();
633    if sources.iter().all(|path| *path == target) {
634        return Ok(());
635    }
636    let mut kept = Vec::<serde_json::Value>::new();
637    let mut mapped = HashMap::<String, serde_json::Value>::new();
638    for source in &sources {
639        let bytes = fs::read(source).map_err(|error| SessionError::Io {
640            path: source.clone(),
641            source: error,
642        })?;
643        for line in bytes.split(|byte| *byte == b'\n') {
644            let Ok(record) = serde_json::from_slice::<serde_json::Value>(line) else {
645                continue;
646            };
647            let Some(revision) = record.get("runtime_revision").map(ToString::to_string) else {
648                continue;
649            };
650            let model = record
651                .get("model_runtime_revision")
652                .cloned()
653                .unwrap_or_default();
654            match mapped.get(&revision) {
655                Some(known) if *known != model => continue,
656                _ if kept.contains(&record) => continue,
657                _ => {
658                    mapped.insert(revision, model);
659                    kept.push(record);
660                }
661            }
662        }
663    }
664    let mut bytes = Vec::new();
665    for record in &kept {
666        serde_json::to_writer(&mut bytes, record).map_err(|source| SessionError::Json {
667            path: target.clone(),
668            source,
669        })?;
670        bytes.push(b'\n');
671    }
672    replace_cache_file(&target, &bytes, CacheSync::Durable)
673}

675/// Removes what a finished migration or an interrupted deletion left in
676/// `sessions/`: deletion tombstones, the backups, staged files, and the
677/// v2 work-dir folders no pending tree lives in. Only `sessions/` itself
678/// and such leftovers are read. Returns the roots of the tombstones it
679/// removed.
680fn sweep_top_level(sessions_dir: &Path, pending: &[PendingTree]) -> Vec<SessionId> {
681    let deleted = delete::sweep_tombstones(sessions_dir);
682    let backup = sessions_dir.join(BACKUP_DIR);
683    if backup.exists()
684        && let Err(error) = fs::remove_dir_all(&backup)
685    {
686        eprintln!(
687            "cookie-agent: {} could not be removed: {error}",
688            backup.display()
689        );
690    }
691    remove_staged_files(sessions_dir);
692    for work_dir in old_work_dirs(sessions_dir) {
693        let keeps_pending = pending
694            .iter()
695            .any(|tree| sessions_dir.join(&tree.path).starts_with(&work_dir));
696        if !keeps_pending {
697            remove_emptied_work_dir(&work_dir);
698        }
699    }
700    deleted
701}

703/// Removes the temporaries a replaced file staged in `directory` and a
704/// crash left behind.
705fn remove_staged_files(directory: &Path) {
706    let Ok(entries) = fs::read_dir(directory) else {
707        return;
708    };
709    for entry in entries.flatten() {
710        let name = entry.file_name();
711        let name = name.to_string_lossy();
712        if name.starts_with('.') && name.ends_with(".tmp") && entry.path().is_file() {
713            let _ = fs::remove_file(entry.path());
714        }
715    }
716}

718/// Removes a v2 work-dir folder whose trees all moved: the files the v2
719/// layout kept there, then the folder. Anything else keeps it, and is named.
720fn remove_emptied_work_dir(work_dir: &Path) {
721    delete::sweep_tombstones(work_dir);
722    remove_staged_files(work_dir);
723    let Ok(entries) = fs::read_dir(work_dir) else {
724        return;
725    };
726    let mut kept = Vec::new();
727    for entry in entries.flatten() {
728        let name = entry.file_name().to_string_lossy().into_owned();
729        let known = V2_WORK_DIR_FILES.contains(&name.as_str())
730            || name.ends_with(crate::ownership::OWNER_LOCK_SUFFIX);
731        if known && entry.path().is_file() {
732            let _ = fs::remove_file(entry.path());
733        } else {
734            kept.push(name);
735        }
736    }
737    if kept.is_empty() {
738        if let Err(error) = fs::remove_dir(work_dir) {
739            eprintln!(
740                "cookie-agent: {} could not be removed: {error}",
741                work_dir.display()
742            );
743        }
744    } else {
745        kept.sort();
746        eprintln!(
747            "cookie-agent: {} stays: it still holds {}",
748            work_dir.display(),
749            kept.join(", ")
750        );
751    }
752}
