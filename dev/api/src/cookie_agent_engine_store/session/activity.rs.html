1//! The monthly activity journals, `sessions/activity/<yyyy-mm>.jsonl`
2//! (`docs/site/specs/session-storage-v3.md` section 4.3): the list entries
3//! of root sessions, appended when one is created, changes its title,
4//! project, or status, or is deleted, and never for every event. A listing
5//! reads them newest month first instead of every session's `metadata`.

12/// One line of a journal: a root's list entry, or its deletion.
13#[derive(Clone, Debug)]
14pub(super) enum Entry {
15    Listed(Box<SessionMeta>),
16    Deleted,
17}
18
19/// A deletion's line.
20#[derive(serde::Serialize)]
21struct Tombstone {
22    session_id: SessionId,
23    deleted: bool,
24}
25
26/// One month's journal as read: each session's last line.
27#[derive(Debug, Default)]
28pub(super) struct MonthEntries(HashMap<SessionId, Entry>);
29
30impl MonthEntries {
31    fn parse(bytes: &[u8]) -> Self {
32        let mut entries = HashMap::new();
33        for line in bytes.split(|byte| *byte == b'\n') {
34            if let Some((id, entry)) = parse_line(line) {
35                entries.insert(id, entry);
36            }
37        }
38        Self(entries)
39    }
40}
41
42/// A journal line, or `None` for one that does not read (a torn final line,
43/// or an entry an incompatible build wrote).
44fn parse_line(line: &[u8]) -> Option<(SessionId, Entry)> {
45    let value = serde_json::from_slice::<serde_json::Value>(line).ok()?;
46    if value.get("deleted") == Some(&serde_json::Value::Bool(true)) {
47        let id = value.get("session_id")?.as_str()?.parse().ok()?;
48        return Some((id, Entry::Deleted));
49    }
50    let meta = serde_json::from_value::<SessionMeta>(value).ok()?;
51    Some((meta.session_id, Entry::Listed(Box::new(meta))))
52}

54/// The session id a journal line names, without decoding the rest.
55fn line_session(line: &[u8]) -> Option<SessionId> {
56    #[derive(serde::Deserialize)]
57    struct Named {
58        session_id: SessionId,
59    }
60    serde_json::from_slice::<Named>(line)
61        .ok()
62        .map(|named| named.session_id)
63}

65fn journal_name(month: Month) -> String {
66    format!("{month}.jsonl")

69/// The journals of one store, with the current and the previous month's
70/// held in memory.
71#[derive(Debug)]
72pub(super) struct ActivityJournals {
73    dir: PathBuf,
74    /// Months in memory, each with the journal's length when it was read or
75    /// last appended to here: a journal of another length was written by
76    /// another process, and is read again.
77    cache: Mutex<BTreeMap<Month, (Arc<MonthEntries>, u64)>>,
78}
79
80impl ActivityJournals {
81    /// The journals under `sessions_dir`. Reads nothing: a month is read
82    /// when a listing first needs it.
83    pub(super) fn open(sessions_dir: &Path) -> Self {
84        Self {
85            dir: sessions_dir.join(ACTIVITY_DIR),
86            cache: Mutex::new(BTreeMap::new()),
87        }
88    }
89
90    fn path(&self, month: Month) -> PathBuf {
91        self.dir.join(journal_name(month))
92    }
93
94    /// The months that have a journal, newest first.
95    pub(super) fn months(&self) -> Vec<Month> {
96        let mut months = fs::read_dir(&self.dir)
97            .map(|entries| {
98                entries
99                    .flatten()
100                    .filter_map(|entry| {
101                        Month::parse(entry.file_name().to_str()?.strip_suffix(".jsonl")?)
102                    })
103                    .collect::<Vec<_>>()
104            })
105            .unwrap_or_default();
106        months.sort_unstable_by(|left, right| right.cmp(left));
107        months
108    }
109
110    /// One month's entries: from memory for the current and the previous
111    /// month while their journals are as this process last saw them.
112    pub(super) fn month(&self, month: Month) -> Arc<MonthEntries> {
113        let path = self.path(month);
114        let oldest_held = Month::of(Timestamp::now()).previous();
115        if month < oldest_held {
116            return Arc::new(read_journal(&path).0);
117        }
118        let mut cache = self
119            .cache
120            .lock()
121            .unwrap_or_else(|poisoned| poisoned.into_inner());
122        cache.retain(|held, _| *held >= oldest_held);
123        let length = fs::metadata(&path).map_or(0, |metadata| metadata.len());
124        if let Some((entries, held)) = cache.get(&month)
125            && *held == length
126        {
127            return Arc::clone(entries);
128        }
129        let (entries, length) = read_journal(&path);
130        let entries = Arc::new(entries);
131        cache.insert(month, (Arc::clone(&entries), length));
132        entries
133    }
134
135    /// Appends `meta`, a root's list entry, to this month's journal.
136    pub(super) fn record(&self, meta: &SessionMeta) {
137        match serde_json::to_vec(meta) {
138            Ok(line) => self.append(meta.session_id, Entry::Listed(Box::new(meta.clone())), line),
139            Err(error) => eprintln!("session {} activity entry: {error}", meta.session_id),
140        }
141    }
142
143    /// Appends the deletion of the root `id` to this month's journal.
144    pub(super) fn record_deleted(&self, id: SessionId) {
145        let tombstone = Tombstone {
146            session_id: id,
147            deleted: true,
148        };
149        match serde_json::to_vec(&tombstone) {
150            Ok(line) => self.append(id, Entry::Deleted, line),
151            Err(error) => eprintln!("session {id} activity entry: {error}"),
152        }
153    }
154
155    /// Appends one line. The first write of a month compacts the previous
156    /// month's journal first. A failure is reported and loses only this
157    /// entry: the session's own log and `metadata` are untouched.
158    fn append(&self, id: SessionId, entry: Entry, mut line: Vec<u8>) {
159        let month = Month::of(Timestamp::now());
160        let path = self.path(month);
161        let mut cache = self
162            .cache
163            .lock()
164            .unwrap_or_else(|poisoned| poisoned.into_inner());
165        let result = (|| -> std::io::Result<()> {
166            let before = match fs::metadata(&path) {
167                Ok(metadata) => metadata.len(),
168                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
169                    create_journal_dir(&self.dir)?;
170                    let previous = self.path(month.previous());
171                    if previous.exists() {
172                        // Compaction only saves space: a failure keeps the
173                        // whole journal, which reads the same.
174                        if let Err(error) = compact(&previous) {
175                            eprintln!(
176                                "activity journal {} was not compacted: {error}",
177                                previous.display()
178                            );
179                        }
180                        cache.remove(&month.previous());
181                    }
182                    0
183                }
184                Err(error) => return Err(error),
185            };
186            line.push(b'\n');
187            append_line(&path, &line)?;
188            let after = before + line.len() as u64;
189            match cache.get_mut(&month) {
190                Some((entries, held)) if *held == before => {
191                    entries_mut(entries).0.insert(id, entry);
192                    *held = after;
193                }
194                Some(_) => {
195                    cache.remove(&month);
196                }
197                None if before == 0 => {
198                    let mut entries = MonthEntries::default();
199                    entries.0.insert(id, entry);
200                    cache.insert(month, (Arc::new(entries), after));
201                }
202                None => {}
203            }
204            Ok(())
205        })();
206        if let Err(error) = result {
207            eprintln!(
208                "session {id} activity entry was not written to {}: {error}",
209                path.display()
210            );
211        }
212    }
213}

215/// Copy-on-write for a month's entries a listing may still hold.
216fn entries_mut(entries: &mut Arc<MonthEntries>) -> &mut MonthEntries {
217    if Arc::get_mut(entries).is_none() {
218        *entries = Arc::new(MonthEntries(entries.0.clone()));
219    }
220    Arc::get_mut(entries).expect("a fresh copy has one owner")
221}

223/// A journal's entries and the length read, empty when there is none.
224fn read_journal(path: &Path) -> (MonthEntries, u64) {
225    match fs::read(path) {
226        Ok(bytes) => (MonthEntries::parse(&bytes), bytes.len() as u64),
227        Err(error) => {
228            if error.kind() != std::io::ErrorKind::NotFound {
229                eprintln!("activity journal {} does not read: {error}", path.display());
230            }
231            (MonthEntries::default(), 0)
232        }
233    }
234}

236fn create_journal_dir(dir: &Path) -> std::io::Result<()> {
237    if dir.is_dir() {
238        return Ok(());
239    }
240    #[cfg(unix)]
241    let created = create_unix_session_directory_all(dir);
242    #[cfg(windows)]
243    let created = create_windows_session_directory(dir);
244    created.map_err(|error| match error {
245        SessionError::Io { source, .. } => source,
246        error => std::io::Error::other(error.to_string()),
247    })
248}

250/// Appends `bytes`, whole lines, to the private file at `path` in one
251/// write. Not synced: a journal is a listing aid the sessions' own files
252/// can rebuild, and an append must not wait on the disk.
253fn append_line(path: &Path, bytes: &[u8]) -> std::io::Result<()> {
254    #[cfg(windows)]
255    if !path.exists() {
256        match cookie_agent_models::secure_store::create_windows_private_file(path) {
257            Ok(_) => {}
258            Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {}
259            Err(error) => return Err(error),
260        }
261    }
262    let mut options = fs::OpenOptions::new();
263    options.create(true).append(true);
264    #[cfg(unix)]
265    options.mode(0o600);
266    options.open(path)?.write_all(bytes)
267}

269/// Rewrites a journal keeping each session's last line, in the order of
270/// those lines.
271fn compact(path: &Path) -> std::io::Result<()> {
272    let bytes = fs::read(path)?;
273    let lines = bytes
274        .split(|byte| *byte == b'\n')
275        .filter_map(|line| Some((line_session(line)?, line)))
276        .collect::<Vec<_>>();
277    let mut last = HashMap::new();
278    for (index, (id, _)) in lines.iter().enumerate() {
279        last.insert(*id, index);
280    }
281    let mut compacted = Vec::with_capacity(bytes.len());
282    for (index, (id, line)) in lines.iter().enumerate() {
283        if last.get(id) == Some(&index) {
284            compacted.extend_from_slice(line);
285            compacted.push(b'\n');
286        }
287    }
288    replace_cache_file(path, &compacted, CacheSync::Durable).map_err(|error| match error {
289        SessionError::Io { source, .. } => source,
290        error => std::io::Error::other(error.to_string()),
291    })
292}

294fn journal_bytes(entries: &[SessionMeta]) -> Result<Vec<u8>, SessionError> {
295    let mut bytes = Vec::new();
296    for meta in entries {
297        serde_json::to_writer(&mut bytes, meta).map_err(|source| SessionError::Json {
298            path: PathBuf::from(ACTIVITY_DIR),
299            source,
300        })?;
301        bytes.push(b'\n');
302    }
303    Ok(bytes)
304}

306/// Writes every journal anew from the `metadata` of each root filed in a
307/// month folder, each entry in the journal of its last activity's month.
308/// The first migration runs it before its commit point, so a crash before
309/// that only has the next start write them again.
310pub(super) fn rebuild_journals(sessions_dir: &Path) -> Result<(), SessionError> {
311    let dir = sessions_dir.join(ACTIVITY_DIR);
312    let mut by_month = BTreeMap::<Month, Vec<SessionMeta>>::new();
313    for month in month_dirs(sessions_dir) {
314        for root in month_root_ids(sessions_dir, month) {
315            let root_dir = sessions_dir.join(month.to_string()).join(root.to_string());
316            if let Some(meta) = root_meta(root, &root_dir) {
317                by_month
318                    .entry(Month::of(meta.last_activity))
319                    .or_default()
320                    .push(meta);
321            }
322        }
323    }
324    create_journal_dir(&dir).map_err(|source| SessionError::Io {
325        path: dir.clone(),
326        source,
327    })?;
328    if let Ok(entries) = fs::read_dir(&dir) {
329        for entry in entries.flatten() {
330            let stale = entry
331                .file_name()
332                .to_str()
333                .and_then(|name| name.strip_suffix(".jsonl"))
334                .and_then(Month::parse)
335                .is_some_and(|month| !by_month.contains_key(&month));
336            if stale {
337                let _ = fs::remove_file(entry.path());
338            }
339        }
340    }
341    for (month, mut entries) in by_month {
342        entries.sort_by_key(|meta| meta.last_activity);
343        replace_cache_file(
344            &dir.join(journal_name(month)),
345            &journal_bytes(&entries)?,
346            CacheSync::Durable,
347        )?;
348    }
349    fsync_directory(&dir)?;
350    Ok(())
351}

353/// Appends the entries of roots a retried migration step moved, each to
354/// the journal of its last activity's month, and syncs them before the
355/// commit point that stops retrying them.
356pub(super) fn append_migrated(
357    sessions_dir: &Path,
358    entries: &[SessionMeta],
359) -> Result<(), SessionError> {
360    if entries.is_empty() {
361        return Ok(());
362    }
363    let dir = sessions_dir.join(ACTIVITY_DIR);
364    create_journal_dir(&dir).map_err(|source| SessionError::Io {
365        path: dir.clone(),
366        source,
367    })?;
368    let mut by_month = BTreeMap::<Month, Vec<SessionMeta>>::new();
369    for meta in entries {
370        by_month
371            .entry(Month::of(meta.last_activity))
372            .or_default()
373            .push(meta.clone());
374    }
375    for (month, entries) in by_month {
376        let path = dir.join(journal_name(month));
377        let io_error = |source| SessionError::Io {
378            path: path.clone(),
379            source,
380        };
381        append_line(&path, &journal_bytes(&entries)?).map_err(io_error)?;
382        // Opened for writing: Windows refuses to flush a read-only handle.
383        fs::OpenOptions::new()
384            .append(true)
385            .open(&path)
386            .and_then(|file| file.sync_all())
387            .map_err(io_error)?;
388    }
389    fsync_directory(&dir)?;
390    Ok(())
391}

393/// Which root sessions a `session.list` answers with (client protocol
394/// `active_since` and `before`).
395#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
396pub struct ListWindow {
397    /// Only roots last active at or after this time.
398    pub active_since: Option<Timestamp>,
399    /// Only roots last active before this time: the `older` cursor of the
400    /// previous answer. Without `active_since`, the answer reaches back to
401    /// the start of the newest month that has older entries.
402    pub before: Option<Timestamp>,
403}
404
405/// One window of a listing, before the pinned roots join it.
406#[derive(Default)]
407struct Page {
408    listed: HashMap<SessionId, SessionMeta>,
409    /// Each session's newest entry in the journals read.
410    seen: HashMap<SessionId, Entry>,
411    older: Option<Timestamp>,
412}
413
414/// A listing of root sessions.
415#[derive(Clone, Debug, Default, PartialEq)]
416pub struct RootList {
417    pub sessions: Vec<SessionMeta>,
418    /// The cursor of the next older page, or `None` when nothing is older.
419    pub older: Option<Timestamp>,
420}
421
422impl SessionStore {
423    /// The root sessions last active within `window`, and every root that is
424    /// unread, pinned, or in `busy` whatever its age (section 4.3). Entries come from
425    /// the activity journals, newest month first; a root this process holds
426    /// in memory answers from its projection; a pinned root no journal read
427    /// so far names is read from its own `metadata`. The default window
428    /// lists every root.
429    #[must_use]
430    pub fn list_roots_in(&self, window: ListWindow, busy: &HashSet<SessionId>) -> RootList {
431        crate::blocking::off_worker(|| self.list_roots_in_place(window, busy))
432    }
433
434    /// Every root session.
435    #[must_use]
436    pub fn list_roots(&self) -> Vec<SessionMeta> {
437        self.list_roots_in(ListWindow::default(), &HashSet::new())
438            .sessions
439    }
440
441    fn list_roots_in_place(&self, window: ListWindow, busy: &HashSet<SessionId>) -> RootList {
442        let months = self.activity.months();
443        // Roots created here can be in memory before their folder, and their
444        // first entry, exist.
445        let resident = self
446            .residency
447            .lock()
448            .unwrap_or_else(|poisoned| poisoned.into_inner())
449            .resident
450            .values()
451            .filter(|session| matches!(session.meta.origin, SessionOrigin::Root))
452            .map(|session| (session.meta.session_id, session.meta.clone()))
453            .collect::<HashMap<_, _>>();
454        let mut page = match (window.active_since, window.before) {
455            (None, None) => self.window_page(&months, &resident, None, None),
456            (Some(since), before) => self.window_page(&months, &resident, Some(since), before),
457            // A page of older sessions reaches back to the start of the
458            // newest month that holds any, passing over months that hold
459            // none.
460            (None, Some(mut before)) => loop {
461                let Some(month) = months.iter().find(|month| month.start() < before) else {
462                    break Page::default();
463                };
464                let page = self.window_page(&months, &resident, Some(month.start()), Some(before));
465                match page.older {
466                    Some(older) if page.listed.is_empty() => before = older,
467                    _ => break page,
468                }
469            },
470        };
471        let pinned = self
472            .unread
473            .ids()
474            .into_iter()
475            .chain(self.pinned.ids())
476            .chain(busy.iter().copied())
477            .collect::<HashSet<_>>();
478        for id in pinned {
479            if page.listed.contains_key(&id) {
480                continue;
481            }
482            let meta = match (resident.get(&id), page.seen.get(&id)) {
483                (Some(meta), _) => Some(meta.clone()),
484                (None, Some(Entry::Listed(meta))) => Some((**meta).clone()),
485                (None, Some(Entry::Deleted)) => None,
486                (None, None) => self.pinned_root_meta(id),
487            };
488            if let Some(meta) = meta {
489                page.listed.insert(id, meta);
490            }
491        }
492        let mut sessions = page.listed.into_values().collect::<Vec<_>>();
493        sessions.sort_by(|left, right| {
494            right
495                .last_activity
496                .cmp(&left.last_activity)
497                .then_with(|| left.session_id.cmp(&right.session_id))
498        });
499        RootList {
500            sessions,
501            older: page.older,
502        }
503    }
504
505    /// The roots last active in `[low, high)`, from the journals of the
506    /// months that reach `low` (newest first, so a session's newest entry is
507    /// the one kept) and the resident roots, and whether anything is older.
508    fn window_page(
509        &self,
510        months: &[Month],
511        resident: &HashMap<SessionId, SessionMeta>,
512        low: Option<Timestamp>,
513        high: Option<Timestamp>,
514    ) -> Page {
515        let floor = low.map(Month::of);
516        let mut seen = HashMap::<SessionId, Entry>::new();
517        for month in months
518            .iter()
519            .filter(|month| floor.is_none_or(|floor| **month >= floor))
520        {
521            for (id, entry) in &self.activity.month(*month).0 {
522                seen.entry(*id).or_insert_with(|| entry.clone());
523            }
524        }
525        let in_window = |meta: &SessionMeta| {
526            low.is_none_or(|low| meta.last_activity >= low)
527                && high.is_none_or(|high| meta.last_activity < high)
528        };
529        let mut listed = HashMap::<SessionId, SessionMeta>::new();
530        for (id, entry) in &seen {
531            if let Entry::Listed(meta) = entry {
532                let meta = resident
533                    .get(id)
534                    .cloned()
535                    .unwrap_or_else(|| (**meta).clone());
536                if in_window(&meta) {
537                    listed.insert(*id, meta);
538                }
539            }
540        }
541        for (id, meta) in resident {
542            if !listed.contains_key(id)
543                && !matches!(seen.get(id), Some(Entry::Deleted))
544                && in_window(meta)
545            {
546                listed.insert(*id, meta.clone());
547            }
548        }
549        let older = low.filter(|low| {
550            months.iter().any(|month| Month::of(*low) > *month)
551                || seen
552                    .values()
553                    .any(|entry| matches!(entry, Entry::Listed(meta) if meta.last_activity < *low))
554        });
555        Page {
556            listed,
557            seen,
558            older,
559        }
560    }
561
562    /// A pinned root's entry from its own `metadata`: one small read, and
563    /// none for a root that is gone.
564    /// The root `id`'s list entry as it is now: its projection's while it
565    /// is in memory, else its own `metadata`, without reading its log or
566    /// any journal. `None` for a session that is not a root, or is gone.
567    #[must_use]
568    pub fn root_meta(&self, id: SessionId) -> Option<SessionMeta> {
569        if let Some(session) = self.get_resident(id) {
570            return matches!(session.meta.origin, SessionOrigin::Root)
571                .then(|| session.meta.clone());
572        }
573        self.pinned_root_meta(id)
574    }
575
576    fn pinned_root_meta(&self, id: SessionId) -> Option<SessionMeta> {
577        let dir = self.root_path(id);
578        match self.read_meta(id, &dir) {
579            Ok(meta) if matches!(meta.origin, SessionOrigin::Root) => {
580                self.record_location(id, SessionLocation::Root);
581                Some(meta)
582            }
583            Ok(_) => None,
584            Err(SessionError::Io { source, .. })
585                if source.kind() == std::io::ErrorKind::NotFound =>
586            {
587                None
588            }
589            Err(error) => {
590                eprintln!("session {id} skipped: {error}");
591                None
592            }
593        }
594    }

596    /// Journals a root's list entry now. Called where its title, project,
597    /// status, or attachments change, when it is first published, and when
598    /// a root whose later events were not journaled leaves memory.
599    pub(super) fn journal_root(&self, meta: &SessionMeta) {
600        if !matches!(meta.origin, SessionOrigin::Root) {
601            return;
602        }
603        self.residency
604            .lock()
605            .unwrap_or_else(|poisoned| poisoned.into_inner())
606            .unjournaled
607            .remove(&meta.session_id);
608        self.activity.record(meta);
609    }
610
611    /// Journals the deletion of the root `id`.
612    pub(super) fn journal_deleted(&self, id: SessionId) {
613        self.activity.record_deleted(id);
614    }

616    /// Journals `id` if it is a persisted root whose resident projection
617    /// changed since its last entry, as before it leaves memory: the list
618    /// then shows the sequence a delete is checked against.
619    pub(super) fn journal_if_behind(&self, id: SessionId) {
620        let meta = {
621            let mut residency = self
622                .residency
623                .lock()
624                .unwrap_or_else(|poisoned| poisoned.into_inner());
625            if !residency.unjournaled.remove(&id) {
626                return;
627            }
628            match residency.resident.get(&id) {
629                Some(session) if session.log.is_persisted() => session.meta.clone(),
630                _ => return,
631            }
632        };
633        self.activity.record(&meta);
634    }
