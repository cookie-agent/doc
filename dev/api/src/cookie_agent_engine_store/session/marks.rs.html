1//! The root sessions the user marked, each kind in a file of its own
2//! under `<data_root>`: `unread.json`, those with a finished turn, an
3//! approval, or a question the user has not seen
4//! (`docs/site/specs/unread-sessions.md` section 4), and `pinned.json`,
5//! those the user pinned (`docs/site/specs/session-pins-and-filters.md`).
6//! Each is written through a flushed temporary file renamed over it, like
7//! `crons.json`, and kept across restarts. Marking a session writes
8//! nothing to its log.
9
10use std::{
11    collections::BTreeMap,
12    fs, io,
13    path::{Path, PathBuf},
14    sync::Mutex,
15};
16
17use cookie_agent_protocol::SessionId;
18use jiff::Timestamp;
19use serde::{Deserialize, Serialize};
20
21pub const UNREAD_FILE: &str = "unread.json";
22pub const PINNED_FILE: &str = "pinned.json";
23
24#[derive(Debug, Default, Deserialize, Serialize)]
25#[serde(deny_unknown_fields)]
26struct MarksFile {
27    sessions: Vec<MarkEntry>,
28}
29
30#[derive(Debug, Deserialize, Serialize)]
31#[serde(deny_unknown_fields)]
32struct MarkEntry {
33    session_id: SessionId,
34    /// When it was marked.
35    since: Timestamp,
36}
37
38/// One kind of marked root sessions, and the file that keeps them.
39#[derive(Debug)]
40pub struct MarkedSessions {
41    path: PathBuf,
42    /// What the marks are, for messages: `unread sessions`.
43    what: &'static str,
44    sessions: Mutex<BTreeMap<SessionId, Timestamp>>,
45}
46
47impl MarkedSessions {
48    /// Reads `<data_root>/<file>`; none yet is no session marked. A file
49    /// that does not read is reported and replaced by the next change:
50    /// losing the marks is better than a daemon that does not start.
51    pub fn open(data_root: &Path, file: &'static str, what: &'static str) -> Self {
52        let path = data_root.join(file);
53        let sessions = match read(&path) {
54            Ok(sessions) => sessions,
55            Err(error) => {
56                eprintln!(
57                    "{what}: {} does not read ({error}); starting with none",
58                    path.display()
59                );
60                BTreeMap::new()
61            }
62        };
63        Self {
64            path,
65            what,
66            sessions: Mutex::new(sessions),
67        }
68    }
69
70    fn lock(&self) -> std::sync::MutexGuard<'_, BTreeMap<SessionId, Timestamp>> {
71        self.sessions
72            .lock()
73            .unwrap_or_else(std::sync::PoisonError::into_inner)
74    }
75
76    pub fn contains(&self, id: SessionId) -> bool {
77        self.lock().contains_key(&id)
78    }
79
80    /// Every marked root session.
81    pub fn ids(&self) -> Vec<SessionId> {
82        self.lock().keys().copied().collect()
83    }
84
85    /// Marks `ids` (since `now`) or clears them, and returns those whose
86    /// state changed. The file is written once, when anything changed.
87    pub fn set(&self, ids: &[SessionId], marked: bool, now: Timestamp) -> Vec<SessionId> {
88        let mut sessions = self.lock();
89        let changed = ids
90            .iter()
91            .copied()
92            .filter(|id| {
93                if marked {
94                    sessions.insert(*id, now).is_none()
95                } else {
96                    sessions.remove(id).is_some()
97                }
98            })
99            .collect::<Vec<_>>();
100        if !changed.is_empty() {
101            self.save(&sessions);
102        }
103        changed
104    }
105
106    /// Forgets deleted sessions.
107    pub fn forget(&self, ids: impl IntoIterator<Item = SessionId>) {
108        let mut sessions = self.lock();
109        let before = sessions.len();
110        for id in ids {
111            sessions.remove(&id);
112        }
113        if sessions.len() != before {
114            self.save(&sessions);
115        }
116    }
117
118    /// Writes the file; a failure is reported and the state kept in memory.
119    fn save(&self, sessions: &BTreeMap<SessionId, Timestamp>) {
120        let file = MarksFile {
121            sessions: sessions
122                .iter()
123                .map(|(session_id, since)| MarkEntry {
124                    session_id: *session_id,
125                    since: *since,
126                })
127                .collect(),
128        };
129        if let Err(error) = write(&self.path, &file) {
130            eprintln!(
131                "{}: {} was not written: {error}",
132                self.what,
133                self.path.display()
134            );
135        }
136    }
137}
138
139fn read(path: &Path) -> io::Result<BTreeMap<SessionId, Timestamp>> {
140    let bytes = match fs::read(path) {
141        Ok(bytes) => bytes,
142        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(BTreeMap::new()),
143        Err(error) => return Err(error),
144    };
145    let file: MarksFile = serde_json::from_slice(&bytes).map_err(io::Error::other)?;
146    Ok(file
147        .sessions
148        .into_iter()
149        .map(|entry| (entry.session_id, entry.since))
150        .collect())
151}
152
153/// Replaces the file atomically but without a sync: a crash may lose the
154/// latest change, which marks a session read again or unread again, or
155/// pins or unpins it again (store locks and durability, section 4.5).
156fn write(path: &Path, file: &MarksFile) -> io::Result<()> {
157    let directory = path.parent().expect("the data root is a folder");
158    fs::create_dir_all(directory)?;
159    let bytes = serde_json::to_vec_pretty(file).map_err(io::Error::other)?;
160    let name = path
161        .file_name()
162        .map_or_else(String::new, |name| name.to_string_lossy().into_owned());
163    let temporary = directory.join(format!(".{name}.{}.tmp", uuid::Uuid::now_v7()));
164    let result = (|| {
165        let mut output = fs::File::create(&temporary)?;
166        io::Write::write_all(&mut output, &bytes)?;
167        drop(output);
168        fs::rename(&temporary, path)
169    })();
170    if result.is_err() {
171        let _ = fs::remove_file(&temporary);
172    }
173    result
174}

176#[cfg(test)]
177mod tests {
178    use super::*;
179
180    #[test]
181    fn unread_sessions_survive_a_reopen_and_change_only_once() {
182        let data = tempfile::tempdir().unwrap();
183        let (first, second) = (SessionId::new_v7(), SessionId::new_v7());
184        let now = Timestamp::from_second(1_800_000_000).unwrap();
185        let unread = MarkedSessions::open(data.path(), UNREAD_FILE, "unread sessions");
186        assert_eq!(unread.set(&[first, second], true, now), [first, second]);
187        assert!(unread.set(&[first], true, now).is_empty());
188        assert_eq!(unread.set(&[second], false, now), [second]);
189        assert!(unread.set(&[second], false, now).is_empty());
190        let reopened = MarkedSessions::open(data.path(), UNREAD_FILE, "unread sessions");
191        assert!(reopened.contains(first) && !reopened.contains(second));
192        reopened.forget([first]);
193        assert!(!MarkedSessions::open(data.path(), UNREAD_FILE, "unread sessions").contains(first));
194    }
195
196    #[test]
197    fn each_kind_of_mark_keeps_a_file_of_its_own() {
198        let data = tempfile::tempdir().unwrap();
199        let id = SessionId::new_v7();
200        let now = Timestamp::from_second(1_800_000_000).unwrap();
201        let pinned = MarkedSessions::open(data.path(), PINNED_FILE, "pinned sessions");
202        assert_eq!(pinned.set(&[id], true, now), [id]);
203        assert!(data.path().join(PINNED_FILE).exists());
204        assert!(!data.path().join(UNREAD_FILE).exists());
205        assert!(MarkedSessions::open(data.path(), PINNED_FILE, "pinned sessions").contains(id));
206        assert!(!MarkedSessions::open(data.path(), UNREAD_FILE, "unread sessions").contains(id));
207    }
208
209    #[test]
210    fn a_file_that_does_not_read_starts_with_none() {
211        let data = tempfile::tempdir().unwrap();
212        fs::write(
213            data.path().join(UNREAD_FILE),
214            r#"{"sessions": [], "extra": 1}"#,
215        )
216        .unwrap();
217        let unread = MarkedSessions::open(data.path(), UNREAD_FILE, "unread sessions");
218        let id = SessionId::new_v7();
219        let now = Timestamp::from_second(1_800_000_000).unwrap();
220        assert_eq!(unread.set(&[id], true, now), [id]);
221        assert!(MarkedSessions::open(data.path(), UNREAD_FILE, "unread sessions").contains(id));
222    }
