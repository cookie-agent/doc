1//! Publishing a session's records in append order, each commit point only
2//! after its sync (`docs/site/specs/store-locks-and-durability.md`,
3//! section 4.1).
4//!
5//! An append writes its record under the session's slot and queues its
6//! publication there. A record that rides along publishes at once unless an
7//! earlier commit point is still syncing; a commit point publishes when the
8//! sync covering it ends, which its append waits for after releasing the
9//! slot, and takes the records queued behind it along.
10
11use std::{
12    collections::{HashMap, VecDeque},
13    sync::{Arc, Mutex},
14};
15
16use cookie_agent_protocol::{SessionId, SessionsChangedNotification, StoredEvent};
17
18use super::{SessionMeta, SessionStore};
19use crate::events::{Commit, EventLog};
20
21/// Each session's publications not yet made, oldest first.
22#[derive(Debug, Default)]
23pub(super) struct Publications {
24    queues: Mutex<HashMap<SessionId, Arc<Mutex<VecDeque<Pending>>>>>,
25}
26
27/// One publication waiting its turn.
28#[derive(Debug)]
29pub(super) enum Pending {
30    Event {
31        event: Arc<StoredEvent>,
32        /// The sync a commit point waits for, on the log it was written to.
33        commit: Option<(Arc<EventLog>, Commit)>,
34        /// The session's listed metadata, when the record changed it.
35        upserted: Option<SessionMeta>,
36        /// The root whose pending approvals the record changed.
37        approvals_of: Option<SessionId>,
38    },
39    /// A revert cut the log after `through_seq`.
40    Rewound { through_seq: u64, meta: SessionMeta },
41}
42
43impl Pending {
44    fn ready(&self) -> bool {
45        match self {
46            Self::Event {
47                commit: Some((log, commit)),
48                ..
49            } => log.is_settled(*commit),
50            Self::Event { commit: None, .. } | Self::Rewound { .. } => true,
51        }
52    }
53}
54
55impl Publications {
56    fn queue(&self, id: SessionId) -> Arc<Mutex<VecDeque<Pending>>> {
57        Arc::clone(
58            self.queues
59                .lock()
60                .unwrap_or_else(|poisoned| poisoned.into_inner())
61                .entry(id)
62                .or_default(),
63        )
64    }
65
66    /// Drops `id`'s queue once it is empty and nobody else holds it, so the
67    /// map keeps only sessions with something pending. The caller holds
68    /// `queue`'s lock; anyone else about to use the queue holds a clone, and
69    /// clones are made only under the map's lock, taken here.
70    fn forget_if_idle(&self, id: SessionId, queue: &Arc<Mutex<VecDeque<Pending>>>) {
71        let mut queues = self
72            .queues
73            .lock()
74            .unwrap_or_else(|poisoned| poisoned.into_inner());
75        if queues
76            .get(&id)
77            .is_some_and(|held| Arc::ptr_eq(held, queue) && Arc::strong_count(queue) == 2)
78        {
79            queues.remove(&id);
80        }
81    }
82}
83
84impl SessionStore {
85    /// Queues `pending` behind `id`'s earlier publications and publishes
86    /// whatever is ready. Called under the session's slot, so a session's
87    /// publications queue in append order.
88    pub(super) fn queue_publication(&self, id: SessionId, pending: Pending) {
89        let queue = self.publications.queue(id);
90        let mut pending_queue = queue
91            .lock()
92            .unwrap_or_else(|poisoned| poisoned.into_inner());
93        pending_queue.push_back(pending);
94        self.publish_ready(id, &queue, &mut pending_queue);
95    }
96
97    /// Publishes `id`'s queued records whose syncs ended, in order, up to the
98    /// first commit point still waiting.
99    pub(super) fn publish_settled(&self, id: SessionId) {
100        let queue = self.publications.queue(id);
101        let mut pending_queue = queue
102            .lock()
103            .unwrap_or_else(|poisoned| poisoned.into_inner());
104        self.publish_ready(id, &queue, &mut pending_queue);
105    }
106
107    /// Holding the queue's lock for the whole pass keeps two passes from
108    /// interleaving a session's publications.
109    fn publish_ready(
110        &self,
111        id: SessionId,
112        queue: &Arc<Mutex<VecDeque<Pending>>>,
113        pending_queue: &mut VecDeque<Pending>,
114    ) {
115        while pending_queue.front().is_some_and(Pending::ready) {
116            match pending_queue.pop_front().expect("a ready publication") {
117                Pending::Event {
118                    event,
119                    upserted,
120                    approvals_of,
121                    ..
122                } => {
123                    self.publish_stored_event(&event);
124                    if let Some(meta) = upserted {
125                        self.notify_upserted(&meta);
126                    }
127                    if let Some(root_session_id) = approvals_of {
128                        self.notify_index_change(SessionsChangedNotification::ApprovalsChanged {
129                            root_session_id,
130                            session_id: id,
131                        });
132                    }
133                }
134                Pending::Rewound { through_seq, meta } => {
135                    self.publish_rewound(id, through_seq);
136                    self.notify_upserted(&meta);
137                }
138            }
139        }
140        if pending_queue.is_empty() {
141            self.publications.forget_if_idle(id, queue);
142        }
143    }
144}
