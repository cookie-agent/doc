1//! Onboarding (`docs/site/specs/default-models-and-onboarding.md` section
2//! 9): the notice offered while the daemon is not set up, and the
3//! `/onboard` panel with its four steps. Step 1 hands over to the provider
4//! connect panel and steps 2 to 4 to the model picker; each returns to the
5//! panel when it closes. Everything the panel does is a daemon call:
6//! `onboarding.get`, `onboarding.run`, and `provider.connect`. There is no
7//! starter kit: the daemon keeps the official agents itself.

43/// A button of the notice or of the panel.
44#[derive(Clone, Copy, Debug, Eq, PartialEq)]
45pub(in crate::ui) enum OnboardingAction {
46    /// The notice's "Set up": opens the panel.
47    SetUp,
48    /// The notice's or the panel's "Not now": hides the notice until what
49    /// the daemon misses changes.
50    NotNow,
51    /// Step 1: the provider connect panel.
52    Connect,
53    /// Steps 2 to 4: the model picker.
54    ChooseModel,
55    Back,
56    Next,
57    Finish,
58    /// After a run: a new-session draft on the default agent.
59    StartSession,
60    Close,
61}

98/// What the TUI keeps between runs.
99#[derive(Debug, Default, Deserialize, Serialize)]
100struct StoredState {
101    /// What the daemon missed when "Not now" was chosen.
102    #[serde(default, skip_serializing_if = "Option::is_none")]
103    onboarding_not_now: Option<Vec<OnboardingRequirement>>,
104}

106#[derive(Default)]
107pub(in crate::ui) struct OnboardingState {
108    /// `onboarding.get` or the last `onboarding.changed`; `None` before the
109    /// daemon answers, and from a daemon without onboarding.
110    pub(in crate::ui) status: Option<OnboardingStatus>,
111    /// The `onboarding.get` reads: an answer older than another read, or
112    /// than an `onboarding.changed`, is not installed.
113    status_reads: Latest,
114    /// Where "Not now" is remembered; `None` keeps it in memory only.
115    pub(in crate::ui) state_file: Option<PathBuf>,
116    /// What the daemon missed when "Not now" was chosen.
117    pub(in crate::ui) not_now: Option<Vec<OnboardingRequirement>>,
118    pub(in crate::ui) step: Option<OnboardingStep>,
119    pub(in crate::ui) default_choice: Option<(ModelKey, Option<VariantId>)>,
120    pub(in crate::ui) small_choice: Option<(ModelKey, Option<VariantId>)>,
121    pub(in crate::ui) utility_choice: Option<(ModelKey, Option<VariantId>)>,
122    pub(in crate::ui) running: bool,
123    pub(in crate::ui) summary: Option<Vec<String>>,
124    pub(in crate::ui) error: Option<String>,
125    /// The panel handed over to another modal and reopens when it closes.
126    pub(in crate::ui) resume: bool,
127    /// The model picker is choosing this model for the panel.
128    pub(in crate::ui) picking: Option<ModelSlot>,
129}

131impl OnboardingState {
132    /// Reads what "Not now" remembered from `path`.
133    pub(in crate::ui) fn with_state_file(path: PathBuf) -> Self {
134        let not_now = std::fs::read(&path)
135            .ok()
136            .and_then(|bytes| serde_json::from_slice::<StoredState>(&bytes).ok())
137            .and_then(|state| state.onboarding_not_now);
138        Self {
139            state_file: Some(path),
140            not_now,
141            ..Self::default()
142        }
143    }
144
145    /// The default state file, `~/.cookie-agent/tui-state.json`.
146    pub(in crate::ui) fn default_state_file() -> Option<PathBuf> {
147        cookie_agent_protocol::paths::user_data_root()
148            .ok()
149            .map(|root| root.join(STATE_FILE))
150    }
151
152    fn remember_not_now(&mut self, missing: Option<Vec<OnboardingRequirement>>) {
153        self.not_now = missing;
154        let Some(path) = &self.state_file else {
155            return;
156        };
157        let state = StoredState {
158            onboarding_not_now: self.not_now.clone(),
159        };
160        if let Ok(text) = serde_json::to_vec_pretty(&state) {
161            if let Some(parent) = path.parent() {
162                let _ = std::fs::create_dir_all(parent);
163            }
164            let _ = std::fs::write(path, text);
165        }
166    }
167
168    fn step(&self) -> OnboardingStep {
169        self.step.unwrap_or(OnboardingStep::Provider)
170    }
171
172    fn choice(&self, slot: ModelSlot) -> Option<&(ModelKey, Option<VariantId>)> {
173        match slot {
174            ModelSlot::Default => self.default_choice.as_ref(),
175            ModelSlot::Small => self.small_choice.as_ref(),
176            ModelSlot::Utility => self.utility_choice.as_ref(),
177        }
178    }
179}

190    /// Asks the daemon whether it is set up.
191    pub(in crate::ui) fn poll_onboarding(&mut self) {
192        let read = self.onboarding.status_reads.begin();
193        let client = self.client.clone();
194        let updates = self.rpc_updates_tx.clone();
195        self.spawn_rpc(async move {
196            // A daemon without onboarding leaves the status unknown.
197            let status = client.get_onboarding().await.ok();
198            let _ = updates.send(RpcUpdate::OnboardingStatus { read, status });
199        });
200    }

271    /// Why Finish is disabled, if it is.
272    pub(in crate::ui) fn onboarding_finish_refusal(&self) -> Option<String> {
273        let Some(status) = &self.onboarding.status else {
274            return Some("Waiting for the daemon.".into());
275        };
276        if self.onboarding.running {
277            return Some("Setting up…".into());
278        }
279        finish_refusal(status, self.onboarding.default_choice.is_some())
280    }
281
282    fn finish_onboarding(&mut self) {
283        if let Some(reason) = self.onboarding_finish_refusal() {
284            self.status = reason;
285            return;
286        }
287        let input = |choice: Option<&(ModelKey, Option<VariantId>)>| {
288            choice.and_then(|(model, variant)| {
289                self.model_descriptor(model)
290                    .map(|descriptor| model_input(descriptor, variant.as_ref()))
291            })
292        };
293        let params = OnboardingRunParams {
294            install_kit: false,
295            source: None,
296            default_model: input(self.onboarding.choice(ModelSlot::Default)),
297            small_model: input(self.onboarding.choice(ModelSlot::Small)),
298            utility_model: input(self.onboarding.choice(ModelSlot::Utility)),
299        };
300        self.onboarding.running = true;
301        self.onboarding.error = None;
302        // The daemon fetches the official agents first when it has none.
303        self.status = "setting up…".into();
304        let client = self.client.clone();
305        let updates = self.rpc_updates_tx.clone();
306        self.spawn_rpc(async move {
307            let result = client
308                .run_onboarding(params)
309                .await
310                .map_err(|error| error.to_string());
311            let _ = updates.send(RpcUpdate::OnboardingRun { result });
312        });
313    }

315    pub(in crate::ui) fn install_onboarding_run(
316        &mut self,
317        result: Result<OnboardingRunResult, String>,
318    ) {
319        self.onboarding.running = false;
320        match result {
321            Ok(result) => {
322                self.status = if result.status.ready {
323                    "Cookie is set up.".into()
324                } else {
325                    "Onboarding ran; the daemon still needs more.".into()
326                };
327                self.onboarding.summary = Some(run_summary(&result));
328                self.onboarding.status = Some(result.status);
329            }
330            Err(error) => {
331                self.status = "Not set up.".into();
332                self.onboarding.error = Some(error);
333            }
334        }
335    }

440    pub(in crate::ui) fn activate_onboarding_action(&mut self, action: OnboardingAction) {
441        match action {
442            OnboardingAction::SetUp => self.open_onboarding_panel(),
443            OnboardingAction::NotNow => self.not_now(),
444            OnboardingAction::Connect => self.connect_from_onboarding(),
445            OnboardingAction::ChooseModel => match self.onboarding.step() {
446                OnboardingStep::DefaultModel => {
447                    self.choose_model_from_onboarding(ModelSlot::Default)
448                }
449                OnboardingStep::SmallModel => self.choose_model_from_onboarding(ModelSlot::Small),
450                OnboardingStep::UtilityModel => {
451                    self.choose_model_from_onboarding(ModelSlot::Utility)
452                }
453                OnboardingStep::Provider => {}
454            },
455            OnboardingAction::Back => {
456                if let Some(step) = self.onboarding.step().previous() {
457                    self.go_to_step(step);
458                }
459            }
460            OnboardingAction::Next => {
461                if let Some(step) = self.onboarding.step().next() {
462                    self.go_to_step(step);
463                }
464            }
465            OnboardingAction::Finish => self.finish_onboarding(),
466            OnboardingAction::StartSession => self.start_primary_session(),
467            OnboardingAction::Close => self.modal = Modal::None,
468        }
469    }

471    /// The actions the panel offers now.
472    pub(in crate::ui) fn onboarding_actions(&self) -> Vec<OnboardingAction> {
473        if self.onboarding.summary.is_some() {
474            let mut actions = Vec::new();
475            if self.default_root_agent().is_some() {
476                actions.push(OnboardingAction::StartSession);
477            }
478            actions.push(OnboardingAction::Close);
479            return actions;
480        }
481        let step = self.onboarding.step();
482        let mut actions = match step {
483            OnboardingStep::Provider => vec![OnboardingAction::Connect],
484            OnboardingStep::DefaultModel
485            | OnboardingStep::SmallModel
486            | OnboardingStep::UtilityModel => vec![OnboardingAction::ChooseModel],
487        };
488        if step.previous().is_some() {
489            actions.push(OnboardingAction::Back);
490        }
491        if step.next().is_some() {
492            actions.push(OnboardingAction::Next);
493        }
494        if self.onboarding_finish_refusal().is_none() {
495            actions.push(OnboardingAction::Finish);
496        }
497        if self
498            .onboarding
499            .status
500            .as_ref()
501            .is_some_and(|status| !status.ready)
502        {
503            actions.push(OnboardingAction::NotNow);
504        }
505        actions.push(OnboardingAction::Close);
506        actions
507    }

509    pub(super) fn handle_onboarding_key(&mut self, key: KeyEvent) {
510        let offered = self.onboarding_actions();
511        let offers = |action| offered.contains(&action);
512        let action = match key.code {
513            KeyCode::Esc => Some(OnboardingAction::Close),
514            KeyCode::Left | KeyCode::BackTab => Some(OnboardingAction::Back),
515            KeyCode::Right | KeyCode::Tab => Some(OnboardingAction::Next),
516            KeyCode::Enter => offered.first().copied().filter(|action| {
517                matches!(
518                    action,
519                    OnboardingAction::Connect
520                        | OnboardingAction::ChooseModel
521                        | OnboardingAction::StartSession
522                )
523            }),
524            KeyCode::Char('n') if key.modifiers.is_empty() => Some(OnboardingAction::NotNow),
525            KeyCode::Char('f') if key.modifiers == KeyModifiers::NONE => {
526                if let Some(reason) = self.onboarding_finish_refusal() {
527                    self.status = reason;
528                }
529                Some(OnboardingAction::Finish)
530            }
531            _ => None,
532        };
533        if let Some(action) = action.filter(|action| offers(*action)) {
534            self.activate_onboarding_action(action);
535        }
536    }

538    /// The notice, in the bar row above the composer: what is left, and its
539    /// two buttons.
540    pub(super) fn render_onboarding_notice(&mut self, frame: &mut Frame, area: Rect) {
541        self.hit_map.onboarding.notice.clear();
542        if area.width == 0 || area.height == 0 {
543            return;
544        }
545        let missing = self
546            .onboarding
547            .status
548            .as_ref()
549            .map(|status| {
550                status
551                    .missing
552                    .iter()
553                    .map(|requirement| rules::requirement_label(*requirement))
554                    .collect::<Vec<_>>()
555                    .join(", ")
556            })
557            .unwrap_or_default();
558        let text = format!("Cookie is not set up: {missing}. ");
559        let mut spans = vec![Span::styled(text.clone(), self.theme.warning())];
560        let mut x = area.x.saturating_add(
561            u16::try_from(unicode_width::UnicodeWidthStr::width(text.as_str())).unwrap_or(u16::MAX),
562        );
563        for action in [OnboardingAction::SetUp, OnboardingAction::NotNow] {
564            let label = format!(" {} ", action.label());
565            let width = u16::try_from(label.chars().count()).unwrap_or(u16::MAX);
566            if x.saturating_add(width) <= area.right() {
567                self.hit_map
568                    .onboarding
569                    .notice
570                    .push((Rect::new(x, area.y, width, 1), action));
571            }
572            spans.push(Span::styled(label, self.theme.link()));
573            spans.push(Span::raw(" "));
574            x = x.saturating_add(width + 1);
575        }
576        frame.render_widget(
577            Paragraph::new(Line::from(spans)).style(self.theme.panel()),
578            area,
579        );
580    }

582    pub(super) fn render_onboarding_panel(&mut self, frame: &mut Frame) {
583        let area = centered(frame.area(), 86, 76);
584        paint_panel(frame, area, &self.theme);
585        let mut hits = OnboardingHits::default();
586        let block = crate::ui::panel_block()
587            .border_style(self.theme.panel_border())
588            .title(crate::ui::panel_title("Set up Cookie"));
589        let inner = block.inner(area);
590        frame.render_widget(block, area);
591        let [tabs_area, body_area, reason_area, actions_area] = Layout::default()
592            .direction(Direction::Vertical)
593            .constraints([
594                Constraint::Length(2),
595                Constraint::Min(3),
596                Constraint::Length(1),
597                Constraint::Length(1),
598            ])
599            .areas(inner);
600        // The tabs, a check on each done step.
601        let current = self.onboarding.step();
602        let mut spans = Vec::new();
603        let mut x = tabs_area.x;
604        for step in OnboardingStep::ALL {
605            let done = self
606                .onboarding
607                .status
608                .as_ref()
609                .is_some_and(|status| step.done(status))
610                || self.chosen_for(step);
611            let label = format!(" {} {} ", if done { "✓" } else { "·" }, step.label());
612            let width = u16::try_from(label.chars().count()).unwrap_or(u16::MAX);
613            if x.saturating_add(width) <= tabs_area.right() {
614                hits.tabs.push((Rect::new(x, tabs_area.y, width, 1), step));
615            }
616            let style = if step == current {
617                self.theme.selected()
618            } else {
619                self.theme.link()
620            };
621            spans.push(Span::styled(label, style));
622            spans.push(Span::raw(" "));
623            x = x.saturating_add(width + 1);
624        }
625        frame.render_widget(Paragraph::new(Line::from(spans)), tabs_area);
626        // The body.
627        let lines = self.onboarding_body(current);
628        frame.render_widget(Paragraph::new(lines).wrap(Wrap { trim: false }), body_area);
629        if let Some(reason) = self
630            .onboarding_finish_refusal()
631            .filter(|_| self.onboarding.summary.is_none())
632        {
633            frame.render_widget(
634                Paragraph::new(Line::from(Span::styled(
635                    format!("Finish: {reason}"),
636                    self.theme.muted(),
637                ))),
638                reason_area,
639            );
640        }
641        let mut spans = Vec::new();
642        let mut x = actions_area.x;
643        for action in self.onboarding_actions() {
644            let label = format!(" {} ({}) ", action.label(), action.key_label());
645            let width = u16::try_from(label.chars().count()).unwrap_or(u16::MAX);
646            if x.saturating_add(width) <= actions_area.right() {
647                hits.actions
648                    .push((Rect::new(x, actions_area.y, width, 1), action));
649            }
650            spans.push(Span::styled(label, self.theme.link()));
651            spans.push(Span::raw(" "));
652            x = x.saturating_add(width + 1);
653        }
654        frame.render_widget(Paragraph::new(Line::from(spans)), actions_area);
655        let notice = std::mem::take(&mut self.hit_map.onboarding.notice);
656        self.hit_map.onboarding = OnboardingHits { notice, ..hits };
657    }
658
659    /// Whether the panel has a choice of its own for `step`.
660    fn chosen_for(&self, step: OnboardingStep) -> bool {
661        match step {
662            OnboardingStep::DefaultModel => self.onboarding.default_choice.is_some(),
663            OnboardingStep::SmallModel => self.onboarding.small_choice.is_some(),
664            OnboardingStep::UtilityModel => self.onboarding.utility_choice.is_some(),
665            OnboardingStep::Provider => false,
666        }
667    }
668
669    fn onboarding_body(&self, step: OnboardingStep) -> Vec<Line<'static>> {
670        let body = self.theme.body();
671        let muted = self.theme.muted();
672        let mut lines = Vec::new();
673        if let Some(summary) = &self.onboarding.summary {
674            lines.extend(summary.iter().map(|line| Line::styled(line.clone(), body)));
675            return lines;
676        }
677        if let Some(error) = &self.onboarding.error {
678            lines.push(Line::styled(error.clone(), self.theme.error()));
679            lines.push(Line::raw(""));
680        }
681        let Some(status) = &self.onboarding.status else {
682            lines.push(Line::styled("Asking the daemon…", muted));
683            return lines;
684        };
685        match step {
686            OnboardingStep::Provider => {
687                if status.providers_connected.is_empty() {
688                    lines.push(Line::styled("No provider is connected yet.", body));
689                } else {
690                    lines.push(Line::styled("Connected providers:", body));
691                    for provider in &status.providers_connected {
692                        let count = self
693                            .models
694                            .iter()
695                            .filter(|model| &model.key.provider_id() == provider)
696                            .count();
697                        lines.push(Line::styled(
698                            format!("  ✓ {provider} · {count} models available"),
699                            body,
700                        ));
701                    }
702                }
703                lines.push(Line::raw(""));
704                lines.push(Line::styled(
705                    "Enter connects a provider (API key or browser login); several may be connected.",
706                    muted,
707                ));
708            }
709            OnboardingStep::DefaultModel
710            | OnboardingStep::SmallModel
711            | OnboardingStep::UtilityModel => {
712                let (slot, configured, about) = match step {
713                    OnboardingStep::DefaultModel => (
714                        ModelSlot::Default,
715                        status.default_model.as_ref(),
716                        "The model root agents reach as ${default_model}.",
717                    ),
718                    OnboardingStep::SmallModel => (
719                        ModelSlot::Small,
720                        status.small_model.as_ref(),
721                        "Optional: the model subagents that carry out work, such as the official worker and explorer, reach as ${small_model}. Without it they use the default model.",
722                    ),
723                    _ => (
724                        ModelSlot::Utility,
725                        status.utility_model.as_ref(),
726                        "Optional: a fast, cheap model for approvals and session titles, reached as ${utility_model}. Without it they use the session's own model.",
727                    ),
728                };
729                lines.push(Line::styled(about, body));
730                lines.push(Line::raw(""));
731                lines.push(Line::styled(
732                    format!(
733                        "Configured: {}",
734                        configured.map_or_else(|| "not set".into(), model_label)
735                    ),
736                    body,
737                ));
738                if let Some((model, variant)) = self.onboarding.choice(slot) {
739                    lines.push(Line::styled(
740                        format!(
741                            "Chosen: {model}{}",
742                            variant
743                                .as_ref()
744                                .map_or_else(String::new, |variant| format!(" ({variant})"))
745                        ),
746                        self.theme.tool_success(),
747                    ));
748                }
749                lines.push(Line::raw(""));
750                lines.push(Line::styled(
751                    "Enter opens the model list: type to search, then pick a variant.",
752                    muted,
753                ));
754            }
755        }
756        lines
757    }
758}
