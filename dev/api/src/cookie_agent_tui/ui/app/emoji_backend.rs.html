1//! Width-agnostic drawing of VS16 emoji (`❤️`, `☀️`, `✔️`).
2//!
3//! A text-default glyph followed by VS16 (U+FE0F) is two cells to the buffer,
4//! but terminals split on it: VTE, xterm, and Alacritty draw it in one cell
5//! and never paint the second, while macOS terminals, kitty, Ghostty, and
6//! WezTerm draw it in two. Whatever is written to the shadow cell *after* the
7//! emoji clobbers it on the wide terminals, and whatever is skipped lingers
8//! on the narrow ones. [`EmojiBackend`] writes the shadow cell *first* and the
9//! emoji over it, so the narrow terminals keep the blank and the wide ones
10//! cover it. [`bind_emoji_shadows`] gives the shadow cell the emoji's style,
11//! which makes the shadow change exactly when the emoji does: the diff resends
12//! the emoji on those frames, and the backend repaints the shadow with it.
13
14use ratatui::{
15    backend::{Backend, ClearType, WindowSize},
16    buffer::{Buffer, Cell, CellWidth},
17    layout::{Position, Size},
18};
19
20/// A wide cell whose width comes from a presentation selector, which not
21/// every terminal honours.
22fn is_selector_wide(cell: &Cell) -> bool {
23    cell.cell_width() > 1 && cell.symbol().contains('\u{FE0F}')
24}
25
26/// Rewrites the cells behind every VS16 emoji as blanks in the emoji's own
27/// style, the contract [`EmojiBackend`] relies on to paint them itself.
28pub(super) fn bind_emoji_shadows(buffer: &mut Buffer) {
29    let width = usize::from(buffer.area.width);
30    for index in 0..buffer.content.len() {
31        let cell = &buffer.content[index];
32        if !is_selector_wide(cell) {
33            continue;
34        }
35        let style = cell.style();
36        let row_end = (index / width + 1) * width;
37        let end = (index + usize::from(cell.cell_width())).min(row_end);
38        for shadow in &mut buffer.content[index + 1..end] {
39            shadow.reset();
40            shadow.set_style(style);
41        }
42    }
43}
44
45/// Reorders each frame's diff so a VS16 emoji's shadow cells are painted
46/// before the emoji, never after it, and drops the trailing writes ratatui
47/// emits behind such an emoji: those assume the cursor moved one cell per
48/// column, so on a terminal that drew the emoji wide they land one column to
49/// the right, and every contiguous write after them with it. Leaving a gap
50/// behind the emoji makes the backend reposition before the next cell.
51pub(super) struct EmojiBackend<B> {
52    inner: B,
53}
54
55impl<B> EmojiBackend<B> {
56    pub(super) const fn new(inner: B) -> Self {
57        Self { inner }
58    }
59}
60
61impl<B: Backend> Backend for EmojiBackend<B> {
62    type Error = B::Error;
63
64    fn draw<'a, I>(&mut self, content: I) -> Result<(), Self::Error>
65    where
66        I: Iterator<Item = (u16, u16, &'a Cell)>,
67    {
68        let content: Vec<_> = content.collect();
69        let blanks: Vec<Cell> = content
70            .iter()
71            .filter(|(_, _, cell)| is_selector_wide(cell))
72            .map(|(_, _, cell)| {
73                let mut blank = Cell::default();
74                blank.set_style(cell.style());
75                blank
76            })
77            .collect();
78        let mut blanks = blanks.iter();
79        let mut ordered = Vec::with_capacity(content.len() + blanks.len());
80        let mut shadowed = None;
81        for (x, y, cell) in content {
82            if let Some((row, ref columns)) = shadowed
83                && row == y
84                && std::ops::Range::contains(columns, &x)
85            {
86                continue;
87            }
88            if is_selector_wide(cell) {
89                let blank = blanks.next().expect("one blank per VS16 emoji");
90                let columns = x.saturating_add(1)..x.saturating_add(cell.cell_width());
91                ordered.extend(columns.clone().map(|shadow_x| (shadow_x, y, blank)));
92                shadowed = Some((y, columns));
93            }
94            ordered.push((x, y, cell));
95        }
96        self.inner.draw(ordered.into_iter())
97    }
98
99    fn append_lines(&mut self, n: u16) -> Result<(), Self::Error> {
100        self.inner.append_lines(n)
101    }
102

105    }

113    }

121    }

140#[cfg(test)]
141mod tests {
142    use ratatui::{
143        backend::{Backend, CrosstermBackend},
144        buffer::Buffer,
145        layout::Rect,
146        style::{Color, Style},
147    };
148
149    use super::{EmojiBackend, bind_emoji_shadows};
150
151    /// Keeps the bytes the backend wrote readable after it is done with them.
152    #[derive(Clone, Default)]
153    struct Output(std::rc::Rc<std::cell::RefCell<Vec<u8>>>);
154
155    impl std::io::Write for Output {
156        fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
157            self.0.borrow_mut().write(bytes)
158        }
159
160        fn flush(&mut self) -> std::io::Result<()> {
161            Ok(())
162        }
163    }
164
165    fn frames(previous: &str, next: &str, style: Style) -> (Buffer, Buffer) {
166        let area = Rect::new(0, 0, 6, 1);
167        let mut before = Buffer::empty(area);
168        before.set_string(0, 0, previous, Style::default());
169        let mut after = Buffer::empty(area);
170        after.set_string(0, 0, next, style);
171        bind_emoji_shadows(&mut after);
172        (before, after)
173    }
174
175    fn drawn(previous: &str, next: &str) -> String {
176        let (before, after) = frames(previous, next, Style::default());
177        let output = Output::default();
178        let mut backend = EmojiBackend::new(CrosstermBackend::new(output.clone()));
179        backend.draw(before.diff(&after).into_iter()).expect("draw");
180        String::from_utf8(output.0.take()).expect("utf-8")
181    }
182
183    #[test]
184    fn shadow_cells_take_the_emoji_style() {
185        let style = Style::default().fg(Color::Red).bg(Color::Blue);
186        let (_, after) = frames("", "a❤️b", style);
187        let shadow = &after.content[2];
188        assert_eq!(shadow.symbol(), " ");
189        assert_eq!(shadow.style(), after.content[1].style());
190        assert_eq!(after.content[3].symbol(), "b");
191    }
192
193    #[test]
194    fn shadow_is_painted_before_the_emoji_and_the_next_cell_is_repositioned() {
195        // MoveTo(x, y) is `ESC [ y+1 ; x+1 H`. The shadow at column 2 goes
196        // out first, the emoji at column 1 over it, and `b` at column 3 gets
197        // its own MoveTo instead of trusting the cursor after the emoji.
198        let output = drawn("zzzz", "a❤️b");
199        assert!(
200            output.contains("\x1b[1;1Ha\x1b[1;3H \x1b[1;2H❤️\x1b[1;4Hb"),
201            "{output:?}"
202        );
203    }
204
205    #[test]
206    fn unchanged_emoji_and_shadow_are_not_resent() {
207        assert!(!drawn("a❤️b", "a❤️b").contains('❤'));
208    }
209
210    #[test]
211    fn already_wide_emoji_draw_as_usual() {
212        let output = drawn("zzzz", "a💻b");
213        assert!(output.contains("\x1b[1;1Ha💻\x1b[1;4Hb"), "{output:?}");
214    }
