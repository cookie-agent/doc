1//! Per-session `events.subscribe` state on the current connection.
2//!
3//! Every subscribe costs a paged history replay over the wire, a staged
4//! rebuild in the view, and a replacement of the connection's live tail, so
5//! the view subscribes each session at most once while that subscription is
6//! healthy. Opening, reselecting, and watching a session (and each tree
7//! refresh) all go through [`App::subscribe_session`], which coalesces into
8//! whatever the session already has.
9//!
10//! The state machine is `cookie_agent_client_state::subscriptions`, shared
11//! with the web client; this module sends the requests it returns and feeds
12//! their outcomes back. Whether a finished subscription is live depends on
13//! ownership, which the view classifies.
14

23    /// Make sure `session_id` is subscribed on this connection, issuing an
24    /// `events.subscribe` only when it has no usable subscription yet.
25    pub(in crate::ui) fn subscribe_session(
26        &mut self,
27        session_id: SessionId,
28        intent: SubscribeIntent,
29    ) {
30        let owned = self.owned_sessions.contains(&session_id);
31        if let Some(subscribe) = self.subscriptions.subscribe(session_id, owned, intent) {
32            self.send_subscribe(subscribe);
33        }
34    }

36    /// Subscribe `session_id` with a full replay rather than from the view's
37    /// cursor. The rebuilt projection replaces the current one when the
38    /// replay ends, and output streaming now arrives whole: the view may
39    /// have missed deltas while it had no subscription.
40    pub(in crate::ui) fn resubscribe_from_scratch(&mut self, session_id: SessionId) {
41        let owned = self.owned_sessions.contains(&session_id);
42        let subscribe = self.subscriptions.subscribe_from_scratch(session_id, owned);
43        self.send_subscribe(subscribe);
44    }
45
46    fn send_subscribe(&mut self, subscribe: Subscribe) {
47        let Subscribe {
48            session_id,
49            attempt,
50            from_scratch,
51        } = subscribe;
52        let cursor = if from_scratch {
53            None
54        } else {
55            self.store
56                .sessions
57                .get(&session_id)
58                .map(|state| state.last_seq)
59        };
60        let (finished, done) = tokio::sync::watch::channel(false);
61        self.subscription_done.insert(session_id, done);
62        let background = self.store.background_mark(session_id);
63        let client = self.client.clone();
64        let updates = self.rpc_updates_tx.clone();
65        self.spawn_rpc(async move {
66            // No outer deadline: the client bounds every replay page itself,
67            // and a long history can take several pages.
68            let outcome = match client.subscribe_events(session_id, cursor).await {
69                Ok(()) => {
70                    if let Ok(listed) = client.list_background(session_id).await {
71                        let _ = updates.send(RpcUpdate::BackgroundCalls {
72                            session_id,
73                            mark: background,
74                            calls: listed.calls,
75                        });
76                    }
77                    SubscriptionOutcome::Established
78                }
79                Err(crate::ClientError::ReplayInProgress) => SubscriptionOutcome::ReplayInProgress,
80                Err(error) => SubscriptionOutcome::Failed(error.to_string()),
81            };
82            let _ = finished.send(true);
83            let _ = updates.send(RpcUpdate::SubscriptionFinished {
84                session_id,
85                attempt,
86                outcome,
87            });
88        });
89    }

91    pub(super) fn finish_subscription(
92        &mut self,
93        session_id: SessionId,
94        attempt: u64,
95        outcome: SubscriptionOutcome,
96    ) {
97        let current = matches!(
98            self.subscriptions.state(session_id),
99            Some(SubscriptionState::Subscribing { attempt: current, .. }) if current == attempt
100        );
101        if current && let SubscriptionOutcome::Failed(error) = &outcome {
102            self.session_errors.record(error);
103            if self.selected == Some(session_id) {
104                self.status = error.clone();
105            }
106        }
107        let owned = self.owned_sessions.contains(&session_id);
108        if let Some(subscribe) = self
109            .subscriptions
110            .finish(session_id, attempt, &outcome, owned)
111        {
112            self.send_subscribe(subscribe);
113        }
114    }

116    /// A replay of `session_id` ended (already applied to the store).
117    pub(super) fn note_replay_end(&mut self, session_id: SessionId) {
118        let owned = self.owned_sessions.contains(&session_id);
119        if let Some(subscribe) = self.subscriptions.note_replay_end(session_id, owned) {
120            self.send_subscribe(subscribe);
121        }

124    /// Rebuild `session_id` with a full replay: its projection can no longer
125    /// be trusted. This is the one deliberate re-subscribe of a session that
126    /// is already subscribed.
127    pub(in crate::ui) fn recover_session(&mut self, session_id: SessionId) {
128        self.client.recover_session(session_id, true);
129        let owned = self.owned_sessions.contains(&session_id);
130        self.subscriptions.recover(session_id, owned);
131    }

133    /// The client gave up recovering `session_id` (every subscription when

139    /// Ends this connection's live tail for `session_id` and drops its
140    /// projection: reopening it later replays it from the start. A session
141    /// with a tail cannot be evicted by the daemon, so a TUI that has browsed
142    /// many sessions releases the ones it no longer shows.
143    pub(super) fn release_subscription(&mut self, session_id: SessionId) {
144        if self.subscriptions.release(session_id) {
145            self.send_unsubscribe(session_id);
146        }
147    }
148
149    fn send_unsubscribe(&mut self, session_id: SessionId) {
150        // Its replay from the start lays it out afresh: a rebuilt item can
151        // repeat a live one's ID and version with other content.
152        if self.store.evict_session(session_id) {
153            self.parked_layout_caches
154                .retain(|cache| cache.key.is_none_or(|key| key.session_id != session_id));
155        }
156        let client = self.client.clone();
157        self.spawn_rpc(async move {
158            // Best effort: a failure leaves a tail the connection's end drops.
159            let _ = client.unsubscribe_events(session_id).await;
160        });
161    }

170    /// Whether nothing in `session_id` is still owed to this view: no run, and
171    /// no pending or queued input whose start or return it must see.
172    pub(super) fn is_settled(&self, session_id: SessionId) -> bool {
173        self.store
174            .sessions
175            .get(&session_id)
176            .is_none_or(crate::state::SessionState::is_settled)
177    }
178
179    /// Records `root` as the most recently viewed tree. The tree this pushes
180    /// out of the recent set has its settled sessions released; ones still
181    /// running are released when their run ends.
182    pub(super) fn note_viewed_tree(&mut self, root: SessionId) {
183        let mut subscriptions = std::mem::take(&mut self.subscriptions);
184        let released = subscriptions.note_viewed_tree(
185            root,
186            |session_id| self.permission_mode_root(session_id),
187            |session_id| self.is_settled(session_id),
188        );
189        self.subscriptions = subscriptions;
190        for session_id in released {
191            self.send_unsubscribe(session_id);
192        }
193    }

201    /// Starts redialing the daemon after the connection dropped, when this
202    /// TUI has a reconnector. Returns whether a reconnect is under way.
203    pub(in crate::ui) fn begin_reconnect(&mut self) -> bool {
204        let Some(reconnector) = self.reconnector.clone() else {
205            return false;
206        };
207        self.reconnecting = true;
208        self.reset_subscriptions();
209        self.status = "daemon disconnected — reconnecting…".into();
210        let updates = self.rpc_updates_tx.clone();
211        self.spawn_rpc(async move {
212            let mut delay = RECONNECT_FIRST_DELAY;
213            loop {
214                tokio::time::sleep(delay).await;
215                if updates.is_closed() {
216                    return;
217                }
218                match reconnector().await {
219                    Ok(client) => {
220                        let _ = updates.send(RpcUpdate::Reconnected { client });
221                        return;
222                    }
223                    Err(error) => {
224                        let _ = updates.send(RpcUpdate::Notice(format!(
225                            "daemon disconnected — reconnecting ({error})"
226                        )));
227                        delay = next_reconnect_delay(delay);
228                    }
229                }
230            }
231        });
232        true
233    }

235    /// Adopts the redialed connection and rebuilds what the old one fed:
236    /// the viewed session from a full replay (which also brings back output
237    /// streaming now), the session list, the runtime, approvals, the
238    /// permission mode, and the Agents tree's subscriptions.
239    pub(in crate::ui) async fn finish_reconnect(&mut self, client: crate::Client) {
240        self.deliveries = client.subscribe_deliveries();
241        self.client = client;
242        self.reconnecting = false;
243        self.reset_presence();
244        self.reset_restart_after_reconnect();
245        self.reset_subscriptions();
246        self.status = "reconnected to the daemon".into();
247        self.poll_onboarding();
248        self.request_runtime_resync();
249        // The daemon may be a new process, which owns none of the sessions
250        // the last one did: a session it does not own answers
251        // `events.subscribe` with a snapshot that no tail keeps current.
252        // Each is classified (taken) again, as opening it does.
253        self.owned_sessions.clear();
254        self.refresh_lists().await;
255        if let Some(selected) = self.selected {
256            self.resubscribe_from_scratch(selected);
257            // Owned again, the snapshot subscription becomes live.
258            self.classify_session_background(selected);
259            self.refresh_approvals(selected);
260            self.refresh_permission_mode_for_session(selected);
261        }
262        self.refresh_tree_background();
263    }

265    /// After the client gave up recovering a session, tries the viewed one
266    /// again with a fresh subscription once the connection had time to
267    /// settle. Other sessions resubscribe when next opened.
268    pub(super) fn schedule_subscription_retry(&mut self, session_id: Option<SessionId>) {
269        let Some(selected) = self.selected else {
270            return;
271        };
272        if session_id.is_some_and(|session_id| session_id != selected) {
273            return;
274        }
275        let updates = self.rpc_updates_tx.clone();
276        self.spawn_rpc(async move {
277            tokio::time::sleep(SUBSCRIPTION_RETRY_DELAY).await;
278            let _ = updates.send(RpcUpdate::RetrySubscription {
279                session_id: selected,
280            });
281        });
282    }

284    /// Wait (bounded) for an in-flight subscribe of `session_id` to deliver
285    /// its replay. The replay itself is never cancelled.
286    pub(super) async fn wait_for_subscription(&mut self, session_id: SessionId) {
287        if !matches!(
288            self.subscriptions.state(session_id),
289            Some(SubscriptionState::Subscribing { .. })
290        ) {
291            return;
292        }
293        let Some(mut done) = self.subscription_done.get(&session_id).cloned() else {
294            return;
295        };
296        if tokio::time::timeout(SELECT_SUBSCRIPTION_WAIT, done.wait_for(|done| *done))
297            .await
298            .is_err()
299        {
300            // The replay keeps going without us and lands through the
301            // delivery stream; only the wait is bounded.
302            self.status = "still loading session history".into();
303        }
304    }
