8    /// Resolve the element under a point. Overlay *ownership* comes from
9    /// current state, never from hit-map geometry: an overlay that opened
10    /// since the last frame still owns the pointer (its geometry may not
11    /// exist yet), and one that just closed leaves content geometry
12    /// underneath intact. Each overlay branch therefore always returns —
13    /// geometry only picks the element within the panel, never whether the
14    /// panel owns the point. Content follows in click priority order.
15    pub(in crate::ui) fn pointer_target_at(&self, column: u16, row: u16) -> Option<PointerTarget> {
16        let over = |rect: Rect| contains(rect, column, row);
17        let hits = &self.hit_map;
18        if self.command_palette_visible() {
19            return hits
20                .palette_rows
21                .iter()
22                .find(|hit| over(hit.rect))
23                .map(|hit| PointerTarget::PaletteRow(hit.index));
24        }
25        if self.modal != Modal::None {
26            return self.modal_target_at(over);
27        }
28        if self.current_approval().is_some() {
29            if let Some(hit) = hits.approval_actions.iter().find(|hit| over(hit.rect)) {
30                return Some(PointerTarget::ApprovalAction(hit.decision));
31            }
32            return hits
33                .approval_details
34                .filter(|rect| over(*rect))
35                .map(|_| PointerTarget::ApprovalDetails);
36        }
37        // A goal that went away since the last frame leaves no target
38        // behind on its stale bar geometry.
39        if self.goal_bar_visible()
40            && let Some((_, action)) = hits.goal_actions.iter().find(|(rect, _)| over(*rect))
41        {
42            return Some(PointerTarget::GoalAction(*action));
43        }
44        if hits.background_bar.is_some_and(over) {
45            return Some(PointerTarget::BackgroundBar);
46        }
47        if hits.permission_mode.is_some_and(over) {
48            return Some(PointerTarget::PermissionMode);
49        }
50        if hits.session_cost.is_some_and(over) {
51            return Some(PointerTarget::SessionCost);
52        }
53        if hits.event_level_filter.is_some_and(over) {
54            return Some(PointerTarget::EventLevelFilter);
55        }
56        if let Some(hit) = hits.title_segments.iter().find(|hit| over(hit.rect)) {
57            return Some(PointerTarget::TitleSegment(hit.segment));
58        }
59        if let Some(hit) = hits.input.filter(|hit| over(hit.rect)) {
60            // The composer's reserved scrollbar column mirrors the
61            // conversation's; every other composer cell places the caret.
62            if hit.scrollbar.is_some_and(|geometry| over(geometry.track)) {
63                return Some(PointerTarget::Scrollbar(ScrollbarTarget::Input));
64            }
65            return Some(PointerTarget::ComposerText);
66        }
67        if let Some(hit) = hits.queue_entries.iter().find(|hit| over(hit.rect)) {
68            return Some(if hit.recall_seq.is_some() {
69                PointerTarget::QueueEntry(hit.index)
70            } else {
71                PointerTarget::PassiveQueueEntry(hit.index)
72            });
73        }
74        // The scrollbar column is reserved from content and block hit
75        // regions; it is a target only while there is a thumb to move.
76        if hits.scrollbar.is_some_and(over) && self.scrollbar_geometry.is_some() {
77            return Some(PointerTarget::Scrollbar(ScrollbarTarget::Conversation));
78        }
79        if hits.conversation.is_some_and(over)
80            && let Some(target) = self.transcript_target_at(over)
81        {
82            return Some(target);
83        }
84        hits.tree_rows.iter().find(|hit| over(hit.rect)).map(|hit| {
85            match hit.expand_rect.filter(|rect| over(*rect)) {
86                Some(_) => PointerTarget::TreeExpand(hit.session_id),
87                None => PointerTarget::TreeRow(hit.session_id),
88            }
89        })
90    }

92    /// The transcript element under a point inside the conversation
93    /// viewport, which bounds every transcript target.
94    fn transcript_target_at(&self, over: impl Fn(Rect) -> bool) -> Option<PointerTarget> {
95        let hits = &self.hit_map;
96        // Past user-message rows open the copy/revert/fork menu — never
97        // assistant/tool rows, which keep their expand/collapse toggle.
98        if let Some(hit) = hits.user_messages.iter().find(|hit| over(hit.rect)) {
99            return Some(PointerTarget::UserMessage(hit.seq));
100        }
101        // A running call's background button sits on its title row, which
102        // otherwise toggles the call.
103        if let Some(call_id) = hits
104            .blocks
105            .iter()
106            .rev()
107            .find(|hit| hit.action_rect.is_some_and(&over))
108            .and_then(|hit| match hit.id {
109                BlockId::Tool(call_id) => Some(call_id),
110                _ => None,
111            })
112        {
113            return Some(PointerTarget::BackgroundButton(call_id));
114        }
115        hits.blocks
116            .iter()
117            .rev()
118            .find(|hit| hit.toggle_rect.is_some_and(&over))
119            .map(|hit| PointerTarget::TranscriptBlock(hit.id))
120    }

122    /// The element under the pointer inside the open modal, which owns
123    /// every point while it is up.
124    fn modal_target_at(&self, over: impl Fn(Rect) -> bool) -> Option<PointerTarget> {
125        let hits = &self.hit_map;
126        match self.modal {
127            Modal::GoalDetail => {
128                return hits
129                    .goal_close
130                    .filter(|rect| over(*rect))
131                    .map(|_| PointerTarget::GoalClose);
132            }
133            Modal::Permissions => {
134                let permissions = &hits.permissions;
135                if permissions.mode.is_some_and(&over) {
136                    return Some(PointerTarget::PermissionPanelMode);
137                }
138                // The rule form owns the panel: only its own effects act.
139                let effects: &[(Rect, PermissionEffect)] = if self.permission_panel.form.is_some() {
140                    &permissions.form_effects
141                } else if self.permission_panel.selected().is_some() {
142                    &permissions.effects
143                } else {
144                    // A row effect sets the selected rule's effect; with no
145                    // rule selected the click selects the row instead.
146                    &[]
147                };
148                if let Some((_, effect)) = effects.iter().find(|(rect, _)| over(*rect)) {
149                    return Some(PointerTarget::PermissionEffect(*effect));
150                }
151                if self.permission_panel.form.is_some() {
152                    return None;
153                }
154                return permissions
155                    .rows
156                    .iter()
157                    .find(|(rect, _)| over(*rect))
158                    .map(|(_, index)| PointerTarget::PermissionRow(*index));
159            }
160            Modal::Runtimes => {
161                let runtimes = &hits.runtimes;
162                if let Some((_, view)) = runtimes.tabs.iter().find(|(rect, _)| over(*rect)) {
163                    return Some(PointerTarget::RuntimeTab(*view));
164                }
165                if let Some((_, action)) = runtimes
166                    .machine_actions
167                    .iter()
168                    .find(|(rect, _)| over(*rect))
169                {
170                    return Some(PointerTarget::MachineAction(*action));
171                }
172                return runtimes
173                    .machine_rows
174                    .iter()
175                    .find(|(rect, _)| over(*rect))
176                    .map(|(_, index)| PointerTarget::MachineRow(*index));
177            }
178            Modal::Memory => {
179                let memory = &hits.memory;
180                if let Some((_, action)) = memory.actions.iter().find(|(rect, _)| over(*rect)) {
181                    return Some(PointerTarget::MemoryAction(*action));
182                }
183                return memory
184                    .rows
185                    .iter()
186                    .find(|(rect, _)| over(*rect))
187                    .map(|(_, index)| PointerTarget::MemoryRow(*index));
188            }
189            // A waiting delete owns the picker: only its answers act.
190            Modal::Sessions if self.session_delete.is_some() => {
191                return hits
192                    .confirm_buttons
193                    .iter()
194                    .find(|(rect, _)| over(*rect))
195                    .map(|(_, button)| PointerTarget::ConfirmButton(*button));
196            }
197            Modal::Sessions => {
198                if let Some((_, action)) = hits
199                    .session_picker_actions
200                    .iter()
201                    .find(|(rect, _)| over(*rect))
202                {
203                    return Some(PointerTarget::SessionPickerAction(*action));
204                }
205            }
206            Modal::Background => {
207                if let Some((_, index)) = hits.background_stops.iter().find(|(rect, _)| over(*rect))
208                {
209                    return Some(PointerTarget::BackgroundStop(*index));
210                }
211                return hits
212                    .background_rows
213                    .iter()
214                    .find(|(rect, _)| over(*rect))
215                    .map(|(_, index)| PointerTarget::BackgroundRow(*index));
216            }
217            _ => {}
218        }
219        if let Some((_, button)) = hits.confirm_buttons.iter().find(|(rect, _)| over(*rect)) {
220            return Some(PointerTarget::ConfirmButton(*button));
221        }
222        if hits.picker_input.is_some_and(|hit| over(hit.rect)) {
223            return Some(PointerTarget::PickerInput);
224        }
225        if let Some(hit) = hits.picker_rows.iter().find(|hit| over(hit.rect)) {
226            return Some(PointerTarget::PickerRow(hit.index));
227        }
228        if hits.provider_submit.is_some_and(&over) {
229            return Some(PointerTarget::ProviderSubmit);
230        }
231        if hits.provider_cancel.is_some_and(&over) {
232            return Some(PointerTarget::ProviderCancel);
233        }
234        hits.provider_fields
235            .iter()
236            .find(|hit| over(hit.rect))
237            .map(|hit| PointerTarget::ProviderField(hit.focus))
238    }

240    /// The hover target under a point: the pointer target, unless it is
241    /// one of the hover-quiet exemptions.
242    pub(in crate::ui) fn hover_target_at(&self, column: u16, row: u16) -> Option<PointerTarget> {
243        self.pointer_target_at(column, row)
244            .filter(|target| target.is_hoverable())

247    /// Recompute the hovered element from the current hit map. Returns true
248    /// when the hover target changed (the only case needing a redraw). A
249    /// captured scrollbar drag freezes hover until the press is released.
250    pub(super) fn update_hover(&mut self, column: u16, row: u16) -> bool {
251        if self.scrollbar_drag.is_some() {
252            return false;
253        }
254        let next = self.hover_target_at(column, row);
255        if next == self.hover {
256            return false;
257        }
258        self.hover = next;
259        match next {
260            Some(PointerTarget::GoalAction(action)) => {
261                self.status = format!("Goal: {}", goal::goal_action_label(action));
262            }
263            Some(PointerTarget::GoalClose) => self.status = "Close goal details".into(),
264            _ => {}
265        }
266        true
267    }

269    pub(in crate::ui) async fn handle_click(&mut self, column: u16, row: u16) {
270        let target = self.pointer_target_at(column, row);
271        if self.command_palette_visible() {
272            if let Some(target) = target {
273                self.activate_pointer_target(target, column, row).await;
274            }
275            return;
276        }
277        if self.modal != Modal::None {
278            if self.modal == Modal::Permissions
279                && self.permission_panel.form.is_none()
280                && target != Some(PointerTarget::PermissionPanelMode)
281            {
282                self.permission_panel.notice = None;
283            }
284            match target {
285                Some(target) => self.activate_pointer_target(target, column, row).await,
286                // A click beside the background panel's rows closes it.
287                None if self.modal == Modal::Background => self.modal = Modal::None,
288                None => {}
289            }
290            return;
291        }
292        // Any content click past the goal and background bars leaves the
293        // goal bar's keyboard focus.
294        if self.current_approval().is_none()
295            && !matches!(
296                target,
297                Some(PointerTarget::GoalAction(_) | PointerTarget::BackgroundBar)
298            )
299        {
300            self.goal_focus = None;
301        }
302        if let Some(target) = target {
303            self.activate_pointer_target(target, column, row).await;
304        }
305    }

307    /// Perform a resolved target's click action. `column`/`row` place a
308    /// caret or a scrollbar grab within the target.
309    async fn activate_pointer_target(&mut self, target: PointerTarget, column: u16, row: u16) {
310        match target {
311            PointerTarget::PaletteRow(index) => self.activate_palette_row(index).await,
312            PointerTarget::GoalClose => {
313                self.handle_goal_detail_key(KeyEvent::new(KeyCode::Esc, KeyModifiers::NONE));
314            }
315            PointerTarget::PermissionPanelMode | PointerTarget::PermissionMode => {
316                self.cycle_permission_mode();
317            }
318            PointerTarget::PermissionEffect(effect) => {
319                if let Some(form) = &mut self.permission_panel.form {
320                    form.effect = effect;
321                    form.focus = PermissionFormFocus::Effect;
322                } else if let Some(selected) = self.permission_panel.selected() {
323                    self.set_permission_effect(&selected, effect);
324                }
325            }
326            PointerTarget::PermissionRow(index) => {
327                self.permission_panel.selection.select(Some(index));
328                if self.permission_panel.add_row_selected() {
329                    self.open_permission_form();
330                }
331            }
332            PointerTarget::BackgroundRow(index) => self.background_panel.selected = index,
333            PointerTarget::RuntimeTab(view) => self.show_runtime_view(view),
334            PointerTarget::MachineRow(index) => {
335                self.runtime_panel.machine_selection.select(Some(index));
336            }
337            PointerTarget::MachineAction(action) => self.begin_machine_action(action),
338            PointerTarget::MemoryRow(index) => {
339                self.memory_panel.selection.select(Some(index));
340            }
341            PointerTarget::MemoryAction(action) => self.activate_memory_action(action),
342            PointerTarget::BackgroundStop(index) => self.stop_background_call(index),
343            PointerTarget::ConfirmButton(button) => {
344                if self.modal == Modal::RevertConfirm {
345                    self.answer_revert_confirm(button);
346                } else if self.modal == Modal::Sessions {
347                    self.answer_session_delete(button);
348                } else if self.modal == Modal::AttachConfirm {
349                    self.answer_attach_offer(button).await;
350                }
351            }
352            PointerTarget::PickerInput => {
353                let Some(hit) = self.hit_map.picker_input else {
354                    return;
355                };
356                let search = match self.modal {
357                    Modal::Sessions => &mut self.session_search,
358                    Modal::Agents => &mut self.agent_search,
359                    Modal::Models => &mut self.model_search,
360                    Modal::ConnectProviders => &mut self.provider_search,
361                    _ => return,
362                };
363                search.focus_input();
364                search.input_mut().set_cursor_from_display_position(
365                    row.saturating_sub(hit.text_rect.y)
366                        .min(hit.text_rect.height.saturating_sub(1)),
367                    column
368                        .saturating_sub(hit.text_rect.x)
369                        .min(hit.text_rect.width),
370                );
371            }
372            PointerTarget::PickerRow(index) => self.choose_picker_entry(index).await,
373            PointerTarget::SessionPickerAction(action) => {
374                self.activate_session_picker_action(action).await;
375            }
376            PointerTarget::ProviderSubmit => self.dispatch_provider_connect(),
377            PointerTarget::ProviderCancel => self.cancel_connect_form(),
378            PointerTarget::ProviderField(focus) => {
379                if focus == ProviderFormFocus::AuthMethod {
380                    // Clicking the selector mirrors pressing Enter on it:
381                    // cycle to the next method, wiping its stale secrets.
382                    if let Some(form) = self.provider_form.as_mut() {
383                        form.cycle_auth_method(false);
384                    }
385                } else if let Some(hit) = self
386                    .hit_map
387                    .provider_fields
388                    .iter()
389                    .find(|hit| hit.focus == focus)
390                    .copied()
391                {
392                    self.focus_provider_field(hit, column, row);
393                }
394            }
395            PointerTarget::ApprovalAction(decision) => self.answer_approval(decision).await,
396            PointerTarget::ApprovalDetails => self.toggle_approval_details(),
397            PointerTarget::GoalAction(action) => self.activate_goal_action(action),
398            PointerTarget::BackgroundBar => self.open_background_panel(),
399            PointerTarget::SessionCost => {
400                self.modal = Modal::Usage;
401                self.load_usage();
402            }
403            PointerTarget::EventLevelFilter => self.cycle_event_level_filter(),
404            // Agent and model title segments open selectors. The bracketed
405            // variant suffix cycles in place and never opens a separate panel.
406            PointerTarget::TitleSegment(segment) => match segment {
407                TitleSegment::Agent => self.open_selection_modal(Modal::Agents),
408                TitleSegment::Model => self.open_selection_modal(Modal::Models),
409                TitleSegment::Variant => self.cycle_draft_variant(),
410            },
411            PointerTarget::ComposerText => {
412                let Some(hit) = self.hit_map.input else {
413                    return;
414                };
415                self.input.set_cursor_from_display_position(
416                    row.saturating_sub(hit.text_rect.y)
417                        .min(hit.text_rect.height.saturating_sub(1)),
418                    column
419                        .saturating_sub(hit.text_rect.x)
420                        .min(hit.text_rect.width),
421                );
422            }
423            PointerTarget::QueueEntry(index) | PointerTarget::PassiveQueueEntry(index) => {
424                let Some((kind, recall_seq)) = self
425                    .hit_map
426                    .queue_entries
427                    .iter()
428                    .find(|hit| hit.index == index)
429                    .map(|hit| (hit.kind, hit.recall_seq))
430                else {
431                    return;
432                };
433                if let Some(seq) = recall_seq {
434                    self.recall_input(seq);
435                } else if matches!(kind, QueueEntryKind::User | QueueEntryKind::Queued) {
436                    self.status = "only the client that sent a message can recall it".into();
437                }
438            }
439            // Both scrollbars behave alike: a press on the thumb captures a
440            // drag, a press on the bare track pages to the matching offset —
441            // all without moving the text cursor.
442            PointerTarget::Scrollbar(target) => {
443                let geometry = match target {
444                    ScrollbarTarget::Conversation => self.scrollbar_geometry,
445                    ScrollbarTarget::Input => self.hit_map.input.and_then(|hit| hit.scrollbar),
446                };
447                let Some(geometry) = geometry else {
448                    return;
449                };
450                if contains(geometry.thumb, column, row) {
451                    self.scrollbar_drag = Some(ScrollbarDrag {
452                        grab_row: row.saturating_sub(geometry.thumb.y),
453                        target,
454                    });
455                } else {
456                    let offset = geometry.clamp_offset(geometry.offset_for_track_row(row));
457                    match target {
458                        ScrollbarTarget::Conversation => self.conversation_scroll.scroll_to(offset),
459                        ScrollbarTarget::Input => self.input.scroll_to(offset),
460                    }
461                }
462            }
463            PointerTarget::UserMessage(seq) => {
464                if let Some(hit) = self
465                    .hit_map
466                    .user_messages
467                    .iter()
468                    .find(|hit| hit.seq == seq)
469                    .copied()
470                {
471                    self.open_user_menu(hit);
472                }
473            }
474            PointerTarget::BackgroundButton(call_id) => self.request_background(call_id),
475            PointerTarget::TranscriptBlock(id) => self.toggle_block(id),
476            PointerTarget::TreeExpand(session_id) => self.toggle_tree_session(session_id),
477            PointerTarget::TreeRow(session_id) => {
478                self.tree_cursor = Some(session_id);
479                // Watching a descendant changes the conversation/highlight
480                // only; the tree root snapshot is retained.
481                self.watch_session(session_id);
482            }
483        }
484    }

486    /// Patch the hover affordance onto the resolved target's cells. Text
487    /// targets get the glaze text style; buttons get the fill-only variant
488    /// so their glyphs stay put; transcript blocks and messages get the
489    /// block fill; scrollbars take their own accent.
490    pub(super) fn apply_hover(&self, frame: &mut ratatui::Frame) {
491        let Some(hover) = self.hover else {
492            return;
493        };
494        let text_style = self.theme.hover();
495        let fill_style = self.theme.hover_fill();
496        let patch = |frame: &mut ratatui::Frame, rect: Rect, style: Style| {
497            for y in rect.y..rect.y.saturating_add(rect.height) {
498                for x in rect.x..rect.x.saturating_add(rect.width) {
499                    if frame.area().contains(Position::new(x, y)) {
500                        let cell = &mut frame.buffer_mut()[(x, y)];
501                        cell.set_style(style);
502                    }
503                }
504            }
505        };
506        let hits = &self.hit_map;
507        let one = |rect: Option<Rect>| rect.into_iter().collect::<Vec<_>>();
508        let (rects, style) = match hover {
509            // The block's own header content cells, minus a background
510            // button, are the paint target; the wider hit rect stays the
511            // click/toggle target.
512            PointerTarget::TranscriptBlock(id) => (
513                hits.blocks
514                    .iter()
515                    .find(|hit| hit.id == id)
516                    .map(|hit| hit.hover_cells(hits.conversation))
517                    .unwrap_or_default(),
518                self.theme.block_hover(),
519            ),
520            PointerTarget::BackgroundButton(call_id) => (
521                one(hits
522                    .blocks
523                    .iter()
524                    .find(|hit| hit.id == BlockId::Tool(call_id))
525                    .and_then(|hit| hit.action_rect)
526                    .map(|rect| match hits.conversation {
527                        Some(viewport) => rect.intersection(viewport),
528                        None => rect,
529                    })),
530                fill_style,
531            ),
532            PointerTarget::UserMessage(seq) => (
533                one(hits
534                    .user_messages
535                    .iter()
536                    .find(|hit| hit.seq == seq)
537                    .map(|hit| match hits.conversation {
538                        Some(viewport) => hit.hover_rect.intersection(viewport),
539                        None => hit.hover_rect,
540                    })),
541                self.theme.block_hover(),
542            ),
543            PointerTarget::GoalAction(action) => (
544                one(hits
545                    .goal_actions
546                    .iter()
547                    .find(|(_, candidate)| *candidate == action)
548                    .map(|(rect, _)| *rect)),
549                if action == GoalBarAction::Details {
550                    self.theme.block_hover()
551                } else {
552                    text_style
553                },
554            ),
555            PointerTarget::GoalClose => (one(hits.goal_close), text_style),
556            PointerTarget::PaletteRow(index) => (
557                one(hits
558                    .palette_rows
559                    .iter()
560                    .find(|hit| hit.index == index)
561                    .map(|hit| hit.rect)),
562                text_style,
563            ),
564            PointerTarget::PermissionPanelMode => (one(hits.permissions.mode), text_style),
565            PointerTarget::PermissionEffect(effect) => (
566                one(hits
567                    .permissions
568                    .form_effects
569                    .iter()
570                    .chain(&hits.permissions.effects)
571                    .find(|(_, candidate)| *candidate == effect)
572                    .map(|(rect, _)| *rect)),
573                fill_style,
574            ),
575            PointerTarget::PermissionRow(index) => {
576                (one(indexed_rect(&hits.permissions.rows, index)), text_style)
577            }
578            PointerTarget::BackgroundRow(index) => {
579                (one(indexed_rect(&hits.background_rows, index)), text_style)
580            }
581            PointerTarget::BackgroundStop(index) => {
582                (one(indexed_rect(&hits.background_stops, index)), fill_style)
583            }
584            PointerTarget::RuntimeTab(view) => (
585                one(hits
586                    .runtimes
587                    .tabs
588                    .iter()
589                    .find(|(_, candidate)| *candidate == view)
590                    .map(|(rect, _)| *rect)),
591                fill_style,
592            ),
593            PointerTarget::MachineRow(index) => (
594                one(indexed_rect(&hits.runtimes.machine_rows, index)),
595                text_style,
596            ),
597            PointerTarget::MachineAction(action) => (
598                one(hits
599                    .runtimes
600                    .machine_actions
601                    .iter()
602                    .find(|(_, candidate)| *candidate == action)
603                    .map(|(rect, _)| *rect)),
604                fill_style,
605            ),
606            PointerTarget::MemoryRow(index) => {
607                (one(indexed_rect(&hits.memory.rows, index)), text_style)
608            }
609            PointerTarget::MemoryAction(action) => (
610                one(hits
611                    .memory
612                    .actions
613                    .iter()
614                    .find(|(_, candidate)| *candidate == action)
615                    .map(|(rect, _)| *rect)),
616                fill_style,
617            ),
618            PointerTarget::PickerRow(index) => (
619                one(hits
620                    .picker_rows
621                    .iter()
622                    .find(|hit| hit.index == index)
623                    .map(|hit| hit.rect)),
624                text_style,
625            ),
626            PointerTarget::SessionPickerAction(action) => (
627                one(hits
628                    .session_picker_actions
629                    .iter()
630                    .find(|(_, candidate)| *candidate == action)
631                    .map(|(rect, _)| *rect)),
632                fill_style,
633            ),
634            PointerTarget::ProviderField(focus) => (
635                one(hits
636                    .provider_fields
637                    .iter()
638                    .find(|hit| hit.focus == focus)
639                    .map(|hit| hit.text_rect)),
640                fill_style,
641            ),
642            PointerTarget::ProviderSubmit => (one(hits.provider_submit), fill_style),
643            PointerTarget::ProviderCancel => (one(hits.provider_cancel), fill_style),
644            PointerTarget::ConfirmButton(button) => (
645                one(hits
646                    .confirm_buttons
647                    .iter()
648                    .find(|(_, candidate)| *candidate == button)
649                    .map(|(rect, _)| *rect)),
650                fill_style,
651            ),
652            PointerTarget::ApprovalDetails => (one(hits.approval_details), text_style),
653            PointerTarget::ApprovalAction(decision) => (
654                one(hits
655                    .approval_actions
656                    .iter()
657                    .find(|hit| hit.decision == decision)
658                    .map(|hit| hit.rect)),
659                fill_style,
660            ),
661            PointerTarget::BackgroundBar => (one(hits.background_bar), text_style),
662            PointerTarget::TitleSegment(segment) => (
663                one(hits
664                    .title_segments
665                    .iter()
666                    .find(|hit| hit.segment == segment)
667                    .map(|hit| hit.rect)),
668                text_style,
669            ),
670            PointerTarget::PermissionMode => (one(hits.permission_mode), text_style),
671            PointerTarget::SessionCost => (one(hits.session_cost), text_style),
672            PointerTarget::EventLevelFilter => (one(hits.event_level_filter), text_style),
673            PointerTarget::QueueEntry(index) => (
674                one(hits
675                    .queue_entries
676                    .iter()
677                    .find(|hit| hit.index == index && hit.recall_seq.is_some())
678                    .map(|hit| hit.rect)),
679                text_style,
680            ),
681            // The whole column lights up, rail and thumb alike, wherever on
682            // it the pointer rests.
683            PointerTarget::Scrollbar(target) => (
684                one(match target {
685                    ScrollbarTarget::Conversation => self.scrollbar_geometry,
686                    ScrollbarTarget::Input => hits.input.and_then(|hit| hit.scrollbar),
687                }
688                .map(|geometry| geometry.track)),
689                self.theme.scrollbar_hover(),
690            ),
691            PointerTarget::TreeExpand(session_id) => (
692                one(hits
693                    .tree_rows
694                    .iter()
695                    .find(|hit| hit.session_id == session_id)
696                    .and_then(|hit| hit.expand_rect)),
697                fill_style,
698            ),
699            PointerTarget::TreeRow(session_id) => (
700                one(hits
701                    .tree_rows
702                    .iter()
703                    .find(|hit| hit.session_id == session_id)
704                    .map(TreeRowHit::hover_rect)),
705                text_style,
706            ),
707            PointerTarget::PickerInput
708            | PointerTarget::ComposerText
709            | PointerTarget::PassiveQueueEntry(_) => return,
710        };
711        for rect in rects {
712            patch(frame, rect, style);
713        }
714    }
715}

717/// The rect of the `(rect, index)` hit for `index`, if it rendered.
718fn indexed_rect(hits: &[(Rect, usize)], index: usize) -> Option<Rect> {
719    hits.iter()
720        .find(|(_, candidate)| *candidate == index)
721        .map(|(rect, _)| *rect)
