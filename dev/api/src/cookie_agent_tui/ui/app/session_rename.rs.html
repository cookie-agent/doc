1//! Renaming sessions (`docs/site/specs/projects.md` section 8): `/rename`
2//! with a title, with `--auto` for the generated title, or alone for a
3//! field holding the current title; and Ctrl+R in the session picker, for
4//! the highlighted session. Enter on an emptied field goes back to the
5//! automatic title.
6

16/// The session picker's title field for the highlighted session, over the
17/// picker's search.
18pub(in crate::ui) struct SessionRenamePrompt {
19    pub(in crate::ui) session_id: SessionId,
20    pub(in crate::ui) input: InputState,
21}
22
23/// What a title field's key hint says.
24pub(in crate::ui) const TITLE_HINT: &str =
25    "enter: rename · empty enter: automatic title · esc: back";
26
27impl App {
28    /// The title `session_id` has now, as its title field starts.
29    fn current_session_title(&self, session_id: SessionId) -> String {
30        self.sessions
31            .iter()
32            .find(|session| session.session_id == session_id)
33            .map(session_rename::current_title)
34            .or_else(|| {
35                self.tree
36                    .as_ref()
37                    .and_then(|tree| find_session(tree, session_id))
38                    .map(session_rename::current_title)
39            })
40            .unwrap_or_default()
41    }
42
43    /// A field holding `session_id`'s title, its caret at the end.
44    fn title_input(&self, session_id: SessionId) -> InputState {
45        let mut input = InputState::default();
46        input.set_buffer(self.current_session_title(session_id));
47        input.move_buffer_end();
48        input
49    }

51    /// `/rename` chosen in the palette, with the text typed after it: a
52    /// title, `--auto`, or nothing, which asks for the title.
53    pub(in crate::ui) fn run_rename_command(&mut self, arguments: &str) {
54        if self.new_session_draft.is_some() {
55            self.status = "Send the first message before renaming the session.".into();
56            return;
57        }
58        let Some(session_id) = self.selected else {
59            self.status = "Select a session to rename.".into();
60            return;
61        };
62        match session_rename::rename_command(arguments) {
63            Ok(RenameCommand::Prompt) => {
64                let input = self.title_input(session_id);
65                if let Some(palette) = self.palette.as_mut() {
66                    palette.steps.push(PaletteStep::Text {
67                        target: TextTarget::SessionTitle { session_id },
68                        input,
69                    });
70                }
71            }
72            Ok(RenameCommand::Change(change)) => {
73                self.close_command_palette();
74                self.send_session_rename(session_id, change);
75            }
76            Err(error) => self.status = error.to_string(),
77        }
78    }
79
80    /// Enter in a title field: renames to what it holds, or goes back to
81    /// the automatic title when it is empty. Returns whether it sent the
82    /// rename; a title the daemon would refuse stays to be fixed.
83    pub(in crate::ui) fn submit_session_title(
84        &mut self,
85        session_id: SessionId,
86        text: &str,
87    ) -> bool {
88        let change = match session_rename::title_change(text) {
89            Ok(change) => change,
90            Err(TitleError::Empty) => SessionRenameChange::Reset,
91            Err(error) => {
92                self.status = error.to_string();
93                return false;
94            }
95        };
96        self.send_session_rename(session_id, change);
97        true
98    }
99
100    fn send_session_rename(&mut self, session_id: SessionId, change: SessionRenameChange) {
101        let params =
102            session_rename::rename_params(session_id, change, &uuid::Uuid::now_v7().to_string());
103        let client = self.client.clone();
104        let updates = self.rpc_updates_tx.clone();
105        self.spawn_rpc(async move {
106            let result = session_rename::rename_session(&client, params).await;
107            let _ = updates.send(RpcUpdate::SessionRenamed(Box::new(result)));
108        });
109    }
110
111    /// Shows the title a rename left in the session list at once, or why
112    /// it failed; `sessions.changed` tells every other client.
113    pub(super) fn finish_session_rename(&mut self, result: Result<SessionMeta, ClientError>) {
114        match result {
115            Ok(session) => {
116                let session = self.merge_session_meta(session);
117                if let Some(existing) = self
118                    .sessions
119                    .iter_mut()
120                    .find(|existing| existing.session_id == session.session_id)
121                {
122                    *existing = session;
123                    self.note_sessions_changed();
124                }
125            }
126            Err(error) => self.handle_rpc_update(RpcUpdate::failure(&error)),
127        }
128    }
129
130    /// Ctrl+R or the Rename button in the picker: a title field for the
131    /// highlighted session, over the picker's search.
132    pub(in crate::ui) fn begin_session_rename(&mut self) {
133        let index = self.picker_state.selected().unwrap_or(0);
134        let Some(session_id) = self.session_search_ids().get(index).copied() else {
135            self.status = "There is no session to rename.".into();
136            return;
137        };
138        let input = self.title_input(session_id);
139        self.session_rename = Some(SessionRenamePrompt { session_id, input });
140    }
141
142    /// Keys while the picker's title field is open: Enter renames, Esc
143    /// goes back to the picker, and the rest edit the title. Returns
144    /// whether the field took the key.
145    pub(super) fn handle_session_rename_key(&mut self, key: KeyEvent) -> bool {
146        let Some(prompt) = self.session_rename.as_mut() else {
147            return false;
148        };
149        match key.code {
150            KeyCode::Esc => self.session_rename = None,
151            KeyCode::Enter => {
152                let (session_id, text) = (prompt.session_id, prompt.input.as_str().to_owned());
153                if self.submit_session_title(session_id, &text) {
154                    self.session_rename = None;
155                }
156            }
157            _ => keys::edit_plain_input(&mut prompt.input, key),
158        }
159        true
160    }
161
162    /// Paste into the picker's title field as one line.
163    pub(super) fn paste_into_session_rename(&mut self, text: &str) -> bool {
164        let Some(prompt) = self.session_rename.as_mut() else {
165            return false;
166        };
167        prompt
168            .input
169            .insert_text(&text.replace("\r\n", " ").replace(['\r', '\n'], " "));
170        true
171    }
172
173    /// The picker's title field over its search, at the top of `area`.
174    pub(super) fn render_session_rename(&mut self, frame: &mut ratatui::Frame, area: Rect) {
175        let theme = self.theme.clone();
176        let Some(prompt) = self.session_rename.as_mut() else {
177            return;
178        };
179        crate::ui::slash::render_text(
180            frame,
181            area,
182            "Rename session",
183            "The session's title; empty for the automatic one",
184            TITLE_HINT,
185            &mut prompt.input,
186            &theme,
187        );
188    }
