11/// A palette row for one of the agent's commands, drawn after its "agent"
12/// badge.
13pub(super) fn agent_command_label(command: &AcpCommand) -> String {
14    let hint = command
15        .input_hint
16        .as_deref()
17        .map_or(String::new(), |hint| format!(" {hint}"));
18    let label = cookie_agent_client_state::acp::command_label(&command.name);
19    if command.description.trim().is_empty() {
20        format!("{label}{hint}")
21    } else {
22        format!("{label}{hint} — {}", command.description)
23    }
24}

37/// A row of `/options`: the option and its current value's name (`On` or
38/// `Off` for a boolean). Its category (`thought_level`) is the agent's
39/// label for cookie, not for the user.
40fn option_label(option: &AcpConfigOption) -> String {
41    let current = option
42        .values
43        .iter()
44        .find(|value| value.value == option.current_value)
45        .map_or(option.current_value.as_str(), |value| value.name.as_str());
46    format!("{}: {current}", option.name)
47}

59    /// The agent's commands matching the palette's search, listed after
60    /// cookie's own.
61    pub(in crate::ui) fn palette_agent_commands(&self) -> Vec<AcpCommand> {
62        let Some(palette) = &self.palette else {
63            return Vec::new();
64        };
65        if !palette.steps.is_empty() || !self.session_writable() {
66            return Vec::new();
67        }
68        let query = crate::ui::slash::normalized_query(palette.search.as_str());
69        let query = query.trim_start_matches('/');
70        self.viewed_acp()
71            .map(|acp| {
72                acp.commands
73                    .iter()
74                    .filter(|command| {
75                        cookie_agent_client_state::acp::command_listed(&command.name, query)
76                            && crate::ui::slash::match_rank([command.name.as_str()], query)
77                                .is_some()
78                    })
79                    .cloned()
80                    .collect()
81            })
82            .unwrap_or_default()
83    }

97    /// Runs agent command `name` with `input`: the agent's `compact` as
98    /// cookie's compaction, which waits for it, any other as a prompt.
99    pub(super) async fn run_agent_command(&mut self, name: &str, input: &str) {
100        let input = input.trim();
101        if name == "compact" {
102            self.compact_selected_session((!input.is_empty()).then(|| input.to_owned()))
103                .await;
104        } else {
105            self.submit_prompt(agent_command_prompt(name, input)).await;
106        }
107    }

115    /// The ACP agent the draft runs, when it runs one.
116    pub(super) fn draft_acp_agent(&self) -> Option<&cookie_agent_protocol::AgentDescriptor> {
117        let draft = self.new_session_draft.as_ref().or(self.draft.as_ref())?;
118        if !cookie_agent_client_state::acp::is_acp_model(&draft.model) {
119            return None;
120        }
121        self.agents
122            .iter()
123            .find(|agent| agent.id == draft.agent && agent.preset == draft.preset)
124    }

126    /// The runtime the draft's ACP agent must run on
127    /// (`docs/site/specs/acp-several-runtimes.md` section 3.1): a new
128    /// session's choice, else the viewed session's host or choice.
129    pub(in crate::ui) fn agent_host(&self) -> Option<cookie_agent_protocol::RuntimeId> {
130        if let Some(draft) = &self.new_session_draft {
131            return draft.acp_runtime.clone();
132        }
133        let draft = self.draft.as_ref()?;
134        let state = self
135            .viewed_session()
136            .and_then(|session_id| self.store.sessions.get(&session_id))?;
137        cookie_agent_client_state::acp::session_host(&state.acp, Some(draft)).cloned()
138    }

140    /// Whether the viewed session's ACP agent has started, so where it runs
141    /// is settled.
142    pub(in crate::ui) fn agent_host_settled(&self) -> bool {
143        if self.new_session_draft.is_some() {
144            return false;
145        }
146        let (Some(draft), Some(state)) = (
147            self.draft.as_ref(),
148            self.viewed_session()
149                .and_then(|session_id| self.store.sessions.get(&session_id)),
150        ) else {
151            return false;
152        };
153        state.acp.agent.as_ref() == Some(&draft.agent) && state.acp.runtime_id.is_some()
154    }

164    /// The runtime the draft's ACP agent runs on, as the composer title
165    /// names it, when more than one runtime offers it.
166    pub(in crate::ui) fn runs_on_label(&self) -> Option<cookie_agent_protocol::RuntimeId> {
167        let agent = self.draft_acp_agent()?;
168        if cookie_agent_client_state::acp::runs_on_choices(agent).len() < 2 {
169            return None;
170        }
171        cookie_agent_client_state::acp::placed_runtime(

174            self.agent_host().as_ref(),
175        )
176        .map(|runtime| runtime.runtime_id.clone())

179    /// Opens the Runs on step for the draft's ACP agent, unless where it
180    /// runs is settled.
181    pub(in crate::ui) fn open_runs_on(&mut self) {
182        if self.agent_host_settled() {
183            if let Some(runtime) = self.runs_on_label() {
184                self.status = format!(
185                    "the agent runs on {runtime} for this session; start a new session to run it \
186                     elsewhere"
187                );
188            }
189            return;
190        }
191        let rows = self.runs_on_rows();
192        if rows.len() < 2 {
193            return;
194        }
195        let current = self.runs_on_label();
196        let index = rows
197            .iter()
198            .position(|(runtime, _)| Some(runtime) == current.as_ref())
199            .unwrap_or(0);
200        self.modal = Modal::RunsOn;
201        self.picker_state.select(Some(index));
202    }

204    /// Row `index` of the Runs on step: the draft's agent runs there, which
205    /// the daemon attaches when the session does not.
206    pub(in crate::ui) fn choose_runs_on(&mut self, index: usize) {
207        let Some((runtime, available)) = self.runs_on_rows().get(index).cloned() else {
208            return;
209        };
210        if !available {
211            self.status =
212                format!("the agent cannot run on {runtime} now: it is away or disabled there");
213            return;
214        }
215        self.modal = Modal::None;
216        if let Some(draft) = &mut self.new_session_draft {
217            draft.acp_runtime = Some(runtime.clone());
218            self.status = format!("The new session's agent runs on {runtime}");
219            return;
220        }
221        if let Some(draft) = &mut self.draft {
222            draft.acp_runtime = Some(runtime.clone());
223            self.status = format!("The agent runs on {runtime}");
224            self.publish_draft_selection();
225        }
226    }

228    /// The options of the draft's ACP agent (`docs/site/specs/acp-agents.md`
229    /// section 15, input and pickers): a new session's draft offers what the
230    /// agent last started a session with on the runtime it would run on,
231    /// with the draft's choices; a session's are its agent's, or, before its
232    /// agent session started, what it was created with.
233    pub(in crate::ui) fn agent_options(&self) -> Vec<AcpConfigOption> {
234        let attached = self.agent_runtimes();
235        let agent = self.draft_acp_agent();
236        let host = self.agent_host();
237        if self.new_session_draft.is_some() {
238            return agent.map_or_else(Vec::new, |agent| {
239                cookie_agent_client_state::acp::draft_options(
240                    agent,
241                    &attached,
242                    host.as_ref(),
243                    self.draft_agent_options.for_agent(&agent.id),
244                )
245            });
246        }
247        let Some(state) = self
248            .viewed_session()
249            .and_then(|session_id| self.store.sessions.get(&session_id))
250        else {
251            return Vec::new();
252        };
253        cookie_agent_client_state::acp::session_options(&state.acp, agent, &attached, host.as_ref())
254    }

278    /// Opens the values of the agent's model option (`model`) or effort
279    /// option (`effort`) as the model or variant picker, else `/options`.
280    pub(super) fn open_agent_model_picker(&mut self, effort: bool) {
281        let options = self.agent_options();
282        let option = if effort {
283            cookie_agent_client_state::acp::effort_option(&options)
284        } else {
285            cookie_agent_client_state::acp::model_option(&options)
286        };
287        let Some(option) = option.filter(|option| !option.values.is_empty()) else {
288            self.open_agent_options();
289            return;
290        };
291        let current = option
292            .values
293            .iter()
294            .position(|value| value.value == option.current_value)
295            .unwrap_or(0);
296        self.agent_option_step = Some(option.id.clone());
297        self.picker_state.select(Some(current));
298        self.modal = Modal::AgentOptions;
299    }

301    /// The next value of the agent's effort option, as the variant segment
302    /// cycles cookie's own variants.
303    pub(super) fn cycle_agent_effort(&mut self) {
304        let options = self.agent_options();
305        let Some(option) = cookie_agent_client_state::acp::effort_option(&options) else {
306            self.status = "the agent offers no effort option".into();
307            return;
308        };
309        if option.values.len() <= 1 {
310            return;
311        }
312        let index = option
313            .values
314            .iter()
315            .position(|value| value.value == option.current_value)
316            .map_or(0, |index| (index + 1) % option.values.len());
317        let value = &option.values[index];
318        let label = format!("{}: {}", option.name, value.name);
319        self.apply_agent_option(option.id.clone(), value.value.clone(), label);
320    }

329    /// The title and rows of `/options`, `width` columns wide: the
330    /// options, or the values of the option being changed, the current one
331    /// marked and each with its ID dimmed (what an agent document's
332    /// `models` entry writes for a model or effort).
333    pub(super) fn agent_option_rows(&self, width: usize) -> (String, Vec<Line<'static>>) {
334        let agent = self
335            .draft_acp_agent()
336            .map(|agent| &agent.id)
337            .or_else(|| self.viewed_acp().and_then(|acp| acp.agent.as_ref()))
338            .map_or_else(String::new, ToString::to_string);
339        let selected = self.picker_state.selected();
340        match self.agent_option_in_step() {
341            Some(option) => (
342                format!("{} — {agent}", option.name),
343                option
344                    .values
345                    .iter()
346                    .enumerate()
347                    .map(|(index, value)| {
348                        let suffix = if value.value == option.current_value {
349                            " (current)"
350                        } else {
351                            ""
352                        };
353                        super::pickers::document_id_row(
354                            &value.name,
355                            suffix,
356                            &value.value,
357                            width,
358                            selected == Some(index),
359                            &self.theme,
360                        )
361                    })
362                    .collect(),
363            ),
364            None => (
365                format!("Agent options — {agent}"),
366                self.agent_options()
367                    .iter()
368                    .enumerate()
369                    .map(|(index, option)| {
370                        let style = if selected == Some(index) {
371                            self.theme.body().patch(self.theme.selected())
372                        } else {
373                            self.theme.body()
374                        };
375                        Line::from(Span::styled(
376                            truncate_with_ellipsis(&option_label(option), width),
377                            style,
378                        ))
379                    })
380                    .collect(),
381            ),
382        }
383    }

392    /// Row `index` of `/options`: an option opens its values, and a boolean
393    /// one toggles; a value sets it on the agent, or on the draft.
394    pub(super) fn choose_agent_option(&mut self, index: usize) {
395        if let Some(option) = self.agent_option_in_step() {
396            let Some(value) = option.values.get(index) else {
397                return;
398            };
399            let (option_id, value_id) = (option.id.clone(), value.value.clone());
400            let label = format!("{}: {}", option.name, value.name);
401            self.agent_option_step = None;
402            self.modal = Modal::None;
403            self.apply_agent_option(option_id, value_id, label);
404            return;
405        }
406        let Some(option) = self.agent_options().get(index).cloned() else {
407            return;
408        };
409        // A boolean toggles at once; the picker stays open on it.
410        if option.kind == AcpConfigOptionKind::Boolean {
411            let on = option.current_value != "true";
412            let label = format!("{}: {}", option.name, if on { "on" } else { "off" });
413            let option_id = option.id.clone();
414            self.apply_agent_option(option_id, on.to_string(), label);
415            return;
416        }
417        if option.values.is_empty() {
418            self.status = format!("the agent lists no values for {}", option.name);
419            return;
420        }
421        let current = option
422            .values
423            .iter()
424            .position(|value| value.value == option.current_value)
425            .unwrap_or(0);
426        self.agent_option_step = Some(option.id.clone());
427        self.picker_state.select(Some(current));
428    }

441    /// Chooses `value` for option `option_id`: a new session's draft keeps
442    /// it for `session.create`; a session sets it on its agent.
443    fn apply_agent_option(&mut self, option_id: String, value: String, label: String) {
444        if let Some(draft) = &self.new_session_draft {
445            let agent = draft.agent.clone();
446            self.draft_agent_options.choose(&agent, &option_id, &value);
447            self.status = format!("Draft agent option {label}");
448            return;
449        }
450        self.set_agent_option(option_id, value, label);
451    }

453    /// Sets the session's agent option `option_id` to `value`
454    /// (`session.agent_option.set`).
455    fn set_agent_option(&mut self, option_id: String, value: String, label: String) {
456        let Some(session_id) = self.viewed_session() else {
457            return;
458        };
459        let client = self.client.clone();
460        let updates = self.rpc_updates_tx.clone();
461        self.spawn_rpc(async move {
462            let result = client
463                .set_session_agent_option(session_id, option_id, value)
464                .await
465                .map(|result| result.options)
466                .map_err(|error| error.to_string());
467            let _ = updates.send(RpcUpdate::AgentOptionSet {
468                session_id,
469                label,
470                result,
471            });
472        });
473    }

475    pub(super) fn finish_agent_option_set(
476        &mut self,
477        session_id: SessionId,
478        label: String,
479        result: Result<Vec<AcpConfigOption>, String>,
480    ) {
481        match result {
482            Ok(options) => {
483                if let Some(state) = self.store.sessions.get_mut(&session_id) {
484                    state.acp.options = options;
485                    state.version = state.version.wrapping_add(1);
486                }
487                self.status = format!("Agent option {label}");
488            }
489            Err(error) => self.status = format!("setting {label} failed: {error}"),
490        }
491    }
