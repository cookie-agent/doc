11/// A palette row for one of the agent's commands, drawn after its "agent"
12/// badge.
13pub(super) fn agent_command_label(command: &AcpCommand) -> String {
14    let hint = command
15        .input_hint
16        .as_deref()
17        .map_or(String::new(), |hint| format!(" {hint}"));
18    let label = cookie_agent_client_state::acp::command_label(&command.name);
19    if command.description.trim().is_empty() {
20        format!("{label}{hint}")
21    } else {
22        format!("{label}{hint} — {}", command.description)
23    }
24}

37/// A row of `/options`: the option and its current value's name (`On` or
38/// `Off` for a boolean). Its category (`thought_level`) is the agent's
39/// label for cookie, not for the user.
40fn option_label(option: &AcpConfigOption) -> String {
41    let current = option
42        .values
43        .iter()
44        .find(|value| value.value == option.current_value)
45        .map_or(option.current_value.as_str(), |value| value.name.as_str());
46    format!("{}: {current}", option.name)
47}

59    /// The agent's commands matching the palette's search, listed after
60    /// cookie's own.
61    pub(in crate::ui) fn palette_agent_commands(&self) -> Vec<AcpCommand> {
62        let Some(palette) = &self.palette else {
63            return Vec::new();
64        };
65        if !palette.steps.is_empty() || !self.session_writable() {
66            return Vec::new();
67        }
68        let query = crate::ui::slash::normalized_query(palette.search.as_str());
69        let query = query.trim_start_matches('/');
70        self.viewed_acp()
71            .map(|acp| {
72                acp.commands
73                    .iter()
74                    .filter(|command| {
75                        cookie_agent_client_state::acp::command_listed(&command.name, query)
76                            && crate::ui::slash::match_rank([command.name.as_str()], query)
77                                .is_some()
78                    })
79                    .cloned()
80                    .collect()
81            })
82            .unwrap_or_default()
83    }

97    /// Runs agent command `name` with `input`: the agent's `compact` as
98    /// cookie's compaction, which waits for it, any other as a prompt.
99    pub(super) async fn run_agent_command(&mut self, name: &str, input: &str) {
100        let input = input.trim();
101        if name == "compact" {
102            self.compact_selected_session((!input.is_empty()).then(|| input.to_owned()))
103                .await;
104        } else {
105            self.submit_prompt(agent_command_prompt(name, input)).await;
106        }
107    }

115    /// The ACP agent the draft runs, when it runs one.
116    pub(super) fn draft_acp_agent(&self) -> Option<&cookie_agent_protocol::AgentDescriptor> {
117        let draft = self.new_session_draft.as_ref().or(self.draft.as_ref())?;
118        if !cookie_agent_client_state::acp::is_acp_model(&draft.model) {
119            return None;
120        }
121        self.agents
122            .iter()
123            .find(|agent| agent.id == draft.agent && agent.preset == draft.preset)
124    }

126    /// The options of the draft's ACP agent (`docs/site/specs/acp-agents.md`
127    /// section 15, input and pickers): a new session's draft offers what the
128    /// agent last started a session with on the runtime it would run on,
129    /// with the draft's choices; a session's are its agent's, or, before its
130    /// agent session started, what it was created with.
131    pub(in crate::ui) fn agent_options(&self) -> Vec<AcpConfigOption> {
132        let attached = self.agent_runtimes();
133        let agent = self.draft_acp_agent();
134        if self.new_session_draft.is_some() {
135            return agent.map_or_else(Vec::new, |agent| {
136                cookie_agent_client_state::acp::draft_options(
137                    agent,
138                    &attached,
139                    self.draft_agent_options.for_agent(&agent.id),
140                )
141            });
142        }
143        let Some(state) = self
144            .viewed_session()
145            .and_then(|session_id| self.store.sessions.get(&session_id))
146        else {
147            return Vec::new();
148        };
149        cookie_agent_client_state::acp::session_options(&state.acp, agent, &attached)
150    }

158    /// Opens `/options` on the session's agent options.
159    pub(super) fn open_agent_options(&mut self) {
160        if self.agent_options().is_empty() {
161            self.status = if self.viewed_acp().is_some() {
162                "the session's agent offers no options".into()
163            } else {
164                "the agent's options appear once it has started a session: send a first message"
165                    .into()
166            };
167            return;
168        }
169        self.agent_option_step = None;
170        self.picker_state.select(Some(0));
171        self.modal = Modal::AgentOptions;
172    }

174    /// Opens the values of the agent's model option (`model`) or effort
175    /// option (`effort`) as the model or variant picker, else `/options`.
176    pub(super) fn open_agent_model_picker(&mut self, effort: bool) {
177        let options = self.agent_options();
178        let option = if effort {
179            cookie_agent_client_state::acp::effort_option(&options)
180        } else {
181            cookie_agent_client_state::acp::model_option(&options)
182        };
183        let Some(option) = option.filter(|option| !option.values.is_empty()) else {
184            self.open_agent_options();
185            return;
186        };
187        let current = option
188            .values
189            .iter()
190            .position(|value| value.value == option.current_value)
191            .unwrap_or(0);
192        self.agent_option_step = Some(option.id.clone());
193        self.picker_state.select(Some(current));
194        self.modal = Modal::AgentOptions;
195    }

197    /// The next value of the agent's effort option, as the variant segment
198    /// cycles cookie's own variants.
199    pub(super) fn cycle_agent_effort(&mut self) {
200        let options = self.agent_options();
201        let Some(option) = cookie_agent_client_state::acp::effort_option(&options) else {
202            self.status = "the agent offers no effort option".into();
203            return;
204        };
205        if option.values.len() <= 1 {
206            return;
207        }
208        let index = option
209            .values
210            .iter()
211            .position(|value| value.value == option.current_value)
212            .map_or(0, |index| (index + 1) % option.values.len());
213        let value = &option.values[index];
214        let label = format!("{}: {}", option.name, value.name);
215        self.apply_agent_option(option.id.clone(), value.value.clone(), label);
216    }

218    fn agent_option_in_step(&self) -> Option<AcpConfigOption> {
219        let id = self.agent_option_step.as_deref()?;
220        self.agent_options()

225    /// The title and rows of `/options`, `width` columns wide: the
226    /// options, or the values of the option being changed, the current one
227    /// marked and each with its ID dimmed (what an agent document's
228    /// `models` entry writes for a model or effort).
229    pub(super) fn agent_option_rows(&self, width: usize) -> (String, Vec<Line<'static>>) {
230        let agent = self
231            .draft_acp_agent()
232            .map(|agent| &agent.id)
233            .or_else(|| self.viewed_acp().and_then(|acp| acp.agent.as_ref()))
234            .map_or_else(String::new, ToString::to_string);
235        let selected = self.picker_state.selected();
236        match self.agent_option_in_step() {
237            Some(option) => (
238                format!("{} — {agent}", option.name),
239                option
240                    .values
241                    .iter()
242                    .enumerate()
243                    .map(|(index, value)| {
244                        let suffix = if value.value == option.current_value {
245                            " (current)"
246                        } else {
247                            ""
248                        };
249                        super::pickers::document_id_row(
250                            &value.name,
251                            suffix,
252                            &value.value,
253                            width,
254                            selected == Some(index),
255                            &self.theme,
256                        )
257                    })
258                    .collect(),
259            ),
260            None => (
261                format!("Agent options — {agent}"),
262                self.agent_options()
263                    .iter()
264                    .enumerate()
265                    .map(|(index, option)| {
266                        let style = if selected == Some(index) {
267                            self.theme.body().patch(self.theme.selected())
268                        } else {
269                            self.theme.body()
270                        };
271                        Line::from(Span::styled(
272                            truncate_with_ellipsis(&option_label(option), width),
273                            style,
274                        ))
275                    })
276                    .collect(),
277            ),
278        }
279    }

288    /// Row `index` of `/options`: an option opens its values, and a boolean
289    /// one toggles; a value sets it on the agent, or on the draft.
290    pub(super) fn choose_agent_option(&mut self, index: usize) {
291        if let Some(option) = self.agent_option_in_step() {
292            let Some(value) = option.values.get(index) else {
293                return;
294            };
295            let (option_id, value_id) = (option.id.clone(), value.value.clone());
296            let label = format!("{}: {}", option.name, value.name);
297            self.agent_option_step = None;
298            self.modal = Modal::None;
299            self.apply_agent_option(option_id, value_id, label);
300            return;
301        }
302        let Some(option) = self.agent_options().get(index).cloned() else {
303            return;
304        };
305        // A boolean toggles at once; the picker stays open on it.
306        if option.kind == AcpConfigOptionKind::Boolean {
307            let on = option.current_value != "true";
308            let label = format!("{}: {}", option.name, if on { "on" } else { "off" });
309            let option_id = option.id.clone();
310            self.apply_agent_option(option_id, on.to_string(), label);
311            return;
312        }
313        if option.values.is_empty() {
314            self.status = format!("the agent lists no values for {}", option.name);
315            return;
316        }
317        let current = option
318            .values
319            .iter()
320            .position(|value| value.value == option.current_value)
321            .unwrap_or(0);
322        self.agent_option_step = Some(option.id.clone());
323        self.picker_state.select(Some(current));
324    }

326    /// Chooses `value` for option `option_id`: a new session's draft keeps
327    /// it for `session.create`; a session sets it on its agent.
328    fn apply_agent_option(&mut self, option_id: String, value: String, label: String) {
329        if let Some(draft) = &self.new_session_draft {
330            let agent = draft.agent.clone();
331            self.draft_agent_options.choose(&agent, &option_id, &value);
332            self.status = format!("Draft agent option {label}");
333            return;
334        }
335        self.set_agent_option(option_id, value, label);
336    }

338    /// Sets the session's agent option `option_id` to `value`
339    /// (`session.agent_option.set`).
340    fn set_agent_option(&mut self, option_id: String, value: String, label: String) {
341        let Some(session_id) = self.viewed_session() else {
342            return;
343        };
344        let client = self.client.clone();
345        let updates = self.rpc_updates_tx.clone();
346        self.spawn_rpc(async move {
347            let result = client
348                .set_session_agent_option(session_id, option_id, value)
349                .await
350                .map(|result| result.options)
351                .map_err(|error| error.to_string());
352            let _ = updates.send(RpcUpdate::AgentOptionSet {
353                session_id,
354                label,
355                result,
356            });
357        });
358    }

360    pub(super) fn finish_agent_option_set(
361        &mut self,
362        session_id: SessionId,
363        label: String,
364        result: Result<Vec<AcpConfigOption>, String>,
365    ) {
366        match result {
367            Ok(options) => {
368                if let Some(state) = self.store.sessions.get_mut(&session_id) {
369                    state.acp.options = options;
370                    state.version = state.version.wrapping_add(1);
371                }
372                self.status = format!("Agent option {label}");
373            }
374            Err(error) => self.status = format!("setting {label} failed: {error}"),
375        }
376    }
