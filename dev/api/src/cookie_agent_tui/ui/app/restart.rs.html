1//! Restarting the daemon (`docs/site/specs/engine-configuration.md`
2//! section 11): the banner while configuration waits for a restart, `/restart`
3//! and its confirmation, and the waiting state until the connection closes.
4
5use cookie_agent_client_state::restart::{
6    self as shared, RESTART_WHEN_IDLE, RestartOutcome, daemon_can_restart,
7};
8use cookie_agent_protocol::{ConfigStatus, DaemonRestartingNotification, RestartWhen};

10use super::*;

12/// A button of the restart confirmation.
13#[derive(Clone, Copy, Debug, Eq, PartialEq)]
14pub(in crate::ui) enum RestartChoice {
15    WhenIdle,
16    Now,
17    Cancel,
18}
19
20/// What the TUI knows about restarting the daemon.
21#[derive(Debug, Default)]
22pub(in crate::ui) struct RestartState {
23    /// The daemon's configuration status, for the banner.
24    pub(in crate::ui) config_status: Option<ConfigStatus>,
25    /// A restart the daemon announced, until the connection closes.
26    pub(in crate::ui) restarting: Option<DaemonRestartingNotification>,
27    /// The running turns the open confirmation counted.
28    pub(in crate::ui) running: usize,
29    /// The panel the confirmation opened over, which it returns to.
30    pub(in crate::ui) return_to: Option<Modal>,
31    status_in_flight: bool,
32}
33
34/// The banner's button.
35const RESTART_BUTTON: &str = " Restart daemon ";
36
37impl App {
38    /// Whether the daemon said in its handshake that it predates
39    /// `daemon.restart`. Before a handshake the TUI offers it anyway, and an
40    /// older daemon refuses the method.
41    fn daemon_too_old_to_restart(&self) -> bool {
42        self.client
43            .daemon_version()
44            .is_some_and(|version| !daemon_can_restart(Some(version)))
45    }
46
47    /// The banner's text: the waiting state once the daemon announced a
48    /// restart, else what the configuration waits for.
49    pub(super) fn restart_banner_text(&self) -> Option<String> {
50        if let Some(restarting) = &self.restart.restarting {
51            return Some(shared::restarting_text(restarting));
52        }
53        if self.daemon_too_old_to_restart() {
54            return None;
55        }
56        self.restart
57            .config_status
58            .as_ref()
59            .and_then(shared::restart_banner)
60    }
61
62    /// Fetches the configuration status for the banner.
63    pub(super) fn poll_config_status(&mut self) {
64        if self.restart.status_in_flight || self.daemon_too_old_to_restart() {
65            return;
66        }
67        self.restart.status_in_flight = true;
68        let client = self.client.clone();
69        let updates = self.rpc_updates_tx.clone();
70        self.spawn_rpc(async move {
71            let status = client.get_config().await.ok().map(|config| config.status);
72            let _ = updates.send(RpcUpdate::ConfigStatusFetched(status));
73        });
74    }
75
76    pub(super) fn install_config_status(&mut self, status: Option<ConfigStatus>) {
77        self.restart.status_in_flight = false;
78        if status.is_some() {
79            self.restart.config_status = status;
80        }
81    }
82
83    /// The daemon announced or updated a restart.
84    pub(super) fn note_daemon_restarting(&mut self, restarting: DaemonRestartingNotification) {
85        self.status = shared::restarting_text(&restarting);
86        self.restart.restarting = Some(restarting);
87    }
88
89    /// A new connection: the restart, if any, is over.
90    pub(super) fn reset_restart_after_reconnect(&mut self) {
91        self.restart.restarting = None;
92        self.restart.config_status = None;
93    }
94
95    /// `/restart` or the banner's button: asks how to restart.
96    pub(in crate::ui) fn open_restart_confirm(&mut self) {
97        if self.daemon_too_old_to_restart() {
98            self.status = shared::DAEMON_TOO_OLD_TO_RESTART.into();
99            return;
100        }
101        self.restart.running = shared::running_turns(&self.sessions);
102        self.restart.return_to = (self.modal != Modal::None).then_some(self.modal);
103        self.modal = Modal::RestartConfirm;
104    }
105
106    pub(super) fn handle_restart_confirm_key(&mut self, key: KeyEvent) {
107        match key.code {
108            KeyCode::Esc => self.answer_restart_confirm(RestartChoice::Cancel),
109            KeyCode::Enter | KeyCode::Char('i' | 'I') => {
110                self.answer_restart_confirm(RestartChoice::WhenIdle);
111            }
112            KeyCode::Char('n' | 'N') => self.answer_restart_confirm(RestartChoice::Now),
113            _ => {}
114        }
115    }
116
117    /// Sends the chosen restart, or backs out.
118    pub(in crate::ui) fn answer_restart_confirm(&mut self, choice: RestartChoice) {
119        self.modal = self.restart.return_to.take().unwrap_or(Modal::None);
120        let when = match choice {
121            RestartChoice::Cancel => {
122                self.status = "The daemon keeps running.".into();
123                return;
124            }
125            RestartChoice::WhenIdle => RestartWhen::Idle,
126            RestartChoice::Now => RestartWhen::Now,
127        };
128        self.status = "Asking the daemon to restart…".into();
129        let client = self.client.clone();
130        let updates = self.rpc_updates_tx.clone();
131        self.spawn_rpc(async move {
132            let outcome = RestartOutcome::from_result(when, client.restart_daemon(when).await);
133            let _ = updates.send(RpcUpdate::RestartAnswered(outcome));
134        });
135    }
136
137    pub(super) fn finish_restart_request(&mut self, outcome: RestartOutcome) {
138        self.status = outcome.status();
139    }

141    /// The banner row: its text, and the button while nothing is under way.
142    pub(super) fn render_restart_banner(&mut self, frame: &mut ratatui::Frame, area: Rect) {
143        self.hit_map.restart_banner = None;
144        if area.width == 0 || area.height == 0 {
145            return;
146        }
147        let Some(text) = self.restart_banner_text() else {
148            return;
149        };
150        let width = usize::from(area.width);
151        let button = if self.restart.restarting.is_none()
152            && width >= UnicodeWidthStr::width(RESTART_BUTTON) + 12
153        {
154            RESTART_BUTTON
155        } else {
156            ""
157        };
158        let button_width = UnicodeWidthStr::width(button);
159        let text_width = width.saturating_sub(button_width + 1);
160        let mut text = truncate_with_ellipsis(&format!("⟳ {text}"), text_width);
161        text.push_str(
162            &" ".repeat(text_width.saturating_sub(UnicodeWidthStr::width(text.as_str()))),
163        );
164        frame.render_widget(
165            Paragraph::new(Line::from(vec![
166                Span::styled(text, self.theme.assistant()),
167                Span::raw(" "),
168                Span::styled(button, self.theme.link()),
169            ]))
170            .style(self.theme.panel()),
171            area,
172        );
173        if !button.is_empty() {
174            let x = area
175                .x
176                .saturating_add(u16::try_from(text_width + 1).unwrap_or(u16::MAX));
177            let button_width = u16::try_from(button_width).unwrap_or(u16::MAX);
178            self.hit_map.restart_banner = Some(Rect::new(x, area.y, button_width, 1));
179        }
180    }

182    pub(super) fn render_restart_confirm(&mut self, frame: &mut ratatui::Frame, screen: Rect) {
183        let theme = &self.theme;
184        let running = self.restart.running;
185        let body = |width: u16| {
186            let mut lines = vec![
187                Line::styled("Restart the daemon?", theme.heading()),
188                Line::default(),
189            ];
190            lines.extend(wrapped_line(
191                Line::styled(shared::restart_explanation(running), theme.muted_text()),
192                width,
193            ));
194            if running > 0 {
195                lines.push(Line::default());
196                lines.extend(wrapped_line(
197                    Line::styled(
198                        format!("{}.", shared::restart_now_label(running)),
199                        theme.body(),
200                    ),
201                    width,
202                ));
203            }
204            lines
205        };
206        let buttons = [
207            (
208                RestartChoice::WhenIdle,
209                RESTART_WHEN_IDLE,
210                "enter/i",
211                DecisionTone::Allow,
212            ),
213            (RestartChoice::Now, "Restart now", "n", DecisionTone::Deny),
214            (
215                RestartChoice::Cancel,
216                "Cancel",
217                "esc",
218                DecisionTone::Neutral,
219            ),
220        ];
221        self.hit_map.restart_choices =
222            render_choice_panel(frame, screen, "Restart daemon", body, &buttons, theme);
223    }
