1//! The held banner (`docs/site/specs/held-sessions.md`): while the viewed
2//! session's automatic runs are held, the conversation's last row says why
3//! and what waits, with a *resume* button; Enter in an empty composer
4//! resumes too, and a message sent resumes as it goes.
5
6use super::*;
7
8const RESUME_BUTTON: &str = " resume ";
9
10impl App {
11    /// The banner's text while the viewed session is held.
12    pub(super) fn held_banner_text(&self) -> Option<String> {
13        let state = self.store.sessions.get(&self.selected?)?;
14        cookie_agent_client_state::held::composer_line(state)
15    }
16
17    /// The banner row: why the session is held, what waits, and the button.
18    pub(super) fn render_held_banner(&mut self, frame: &mut ratatui::Frame, area: Rect) {
19        self.hit_map.held_resume = None;
20        if area.width == 0 || area.height == 0 {
21            return;
22        }
23        let Some(text) = self.held_banner_text() else {
24            return;
25        };
26        let width = usize::from(area.width);
27        let button = if width >= UnicodeWidthStr::width(RESUME_BUTTON) + 12 {
28            RESUME_BUTTON
29        } else {
30            ""
31        };
32        let button_width = UnicodeWidthStr::width(button);
33        let text_width = width.saturating_sub(button_width + 1);
34        let mut text = truncate_with_ellipsis(&format!("⏸ {text}"), text_width);
35        text.push_str(
36            &" ".repeat(text_width.saturating_sub(UnicodeWidthStr::width(text.as_str()))),
37        );
38        frame.render_widget(
39            Paragraph::new(Line::from(vec![
40                Span::styled(text, self.theme.assistant()),
41                Span::raw(" "),
42                Span::styled(button, self.theme.link()),
43            ]))
44            .style(self.theme.panel()),
45            area,
46        );
47        if !button.is_empty() {
48            let x = area
49                .x
50                .saturating_add(u16::try_from(text_width + 1).unwrap_or(u16::MAX));
51            let button_width = u16::try_from(button_width).unwrap_or(u16::MAX);
52            self.hit_map.held_resume = Some(Rect::new(x, area.y, button_width, 1));
53        }
54    }
55
56    /// Clears the viewed session's hold (`session.held.resume`).
57    pub(super) fn resume_held(&mut self) {
58        let Some(session_id) = self.selected else {
59            return;
60        };
61        let client = self.client.clone();
62        let updates = self.rpc_updates_tx.clone();
63        self.spawn_rpc(async move {
64            let update = match client
65                .resume_held_session(cookie_agent_protocol::SessionHeldResumeParams { session_id })
66                .await
67            {
68                Ok(result) if result.resumed => return,
69                Ok(_) => RpcUpdate::Notice("the session was not held".into()),
70                Err(error) => RpcUpdate::failure(&error),
71            };
72            let _ = updates.send(update);
73        });
74    }
75}
