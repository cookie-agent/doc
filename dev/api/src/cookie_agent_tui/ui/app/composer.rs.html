6    /// Retire any composer-leg selection: its byte offsets address the
7    /// draft as it was when the drag happened, so a later buffer mutation
8    /// or cursor move would leave it painting and cutting the wrong slice.
9    /// Conversation selections address rendered lines and are unaffected.
10    pub(super) fn retire_composer_selection(&mut self) {
11        if matches!(self.selection, Some(TextSelection::Composer { .. })) {
12            self.selection = None;
13        }
14    }

16    /// Mutate the composer draft, retiring any composer-leg selection.
17    pub(super) fn mutate_input(&mut self, mutation: impl FnOnce(&mut InputState)) {
18        mutation(&mut self.input);
19        self.retire_composer_selection();
20    }

22    /// Move the composer cursor, retiring any composer-leg selection the
23    /// same way a mutation does.
24    pub(super) fn navigate_input(&mut self, navigation: impl FnOnce(&mut InputState)) {
25        navigation(&mut self.input);
26        self.retire_composer_selection();
27    }

29    /// Delete a non-empty composer selection, as Backspace and Delete do in
30    /// any editor. Returns whether there was one to delete.
31    fn delete_composer_selection(&mut self) -> bool {
32        let Some(selection @ TextSelection::Composer { .. }) = self.selection else {
33            return false;
34        };
35        let (start, end) = selection.byte_range();
36        if start == end {
37            return false;
38        }
39        self.mutate_input(|input| input.delete_byte_range(start, end));
40        true
41    }

43    /// Whether the composer may send to the current view: a read-only
44    /// snapshot accepts a message only as the first one of a pending new
45    /// root session.
46    pub(super) fn session_writable(&self) -> bool {
47        self.new_session_draft.is_some()
48            || self
49                .selected
50                .is_none_or(|session| !self.read_only_sessions.contains(&session))
51    }

53    pub(in crate::ui) async fn handle_input_key(&mut self, key: KeyEvent) {
54        // The palette opens from anywhere, read-only views included, and
55        // leaves the draft untouched. `/` opens it only on an empty draft so
56        // paths and other mid-text slashes still type.
57        if (key.code == KeyCode::Char('p') && key.modifiers == KeyModifiers::CONTROL)
58            || (key.code == KeyCode::Char('/')
59                && is_printable_key(key)
60                && self.input.as_str().is_empty())
61        {
62            self.open_command_palette();
63            return;
64        }
65        if !self.session_writable() {
66            self.input_focused = false;
67            self.status = "Session is owned by another cookie process; input is disabled.".into();
68            return;
69        }
70        if self.runtime.phase() == RuntimePhase::ErrorRetry
71            && key.code == KeyCode::Enter
72            && self.input.as_str().is_empty()
73        {
74            self.status = "Retrying runtime snapshot…".into();
75            self.refresh_coherent_lists().await;
76            return;
77        }
78        if self.runtime.phase() == RuntimePhase::Loading {
79            self.status = "loading runtime snapshot".into();
80            return;
81        }
82        if is_newline_key(key) {
83            self.input_focused = true;
84            self.mutate_input(|input| input.insert_newline());
85            return;
86        }
87        if !self.input_focused {
88            match key.code {
89                KeyCode::Enter => self.input_focused = true,
90                KeyCode::Char(character) if is_printable_key(key) => {
91                    self.input_focused = true;
92                    self.mutate_input(|input| input.insert(character));
93                }
94                _ => {}
95            }
96            return;
97        }
98        // A selection is deleted whole, with or without Ctrl, rather than
99        // retired in favor of a one-character or one-word delete.
100        if matches!(key.code, KeyCode::Backspace | KeyCode::Delete)
101            && self.delete_composer_selection()
102        {
103            return;
104        }
105        match key.code {
106            KeyCode::Enter => self.submit_input().await,
107            KeyCode::Backspace if key.modifiers.contains(KeyModifiers::CONTROL) => {
108                self.mutate_input(|input| input.delete_word_left());
109            }
110            KeyCode::Backspace => {
111                self.mutate_input(|input| input.backspace());
112            }
113            KeyCode::Delete if key.modifiers.contains(KeyModifiers::CONTROL) => {
114                self.mutate_input(|input| input.delete_word_right());
115            }
116            KeyCode::Delete => {
117                self.mutate_input(|input| input.delete());
118            }
119            KeyCode::Left if key.modifiers.contains(KeyModifiers::CONTROL) => {
120                self.navigate_input(|input| input.move_word_left());
121            }
122            KeyCode::Left => self.navigate_input(|input| input.move_left()),
123            KeyCode::Right if key.modifiers.contains(KeyModifiers::CONTROL) => {
124                self.navigate_input(|input| input.move_word_right());
125            }
126            KeyCode::Right => self.navigate_input(|input| input.move_right()),
127            KeyCode::Up => {
128                // Recall gesture: with an empty composer and a non-empty
129                // pending lane, Up withdraws the newest pending message back
130                // for editing instead of moving a cursor that has nothing
131                // to move through.
132                if self.input.as_str().is_empty() && self.selected_pending_inputs().is_some() {
133                    self.recall_newest_pending();
134                } else {
135                    self.navigate_input(|input| input.move_up());
136                }
137            }
138            KeyCode::Down => self.navigate_input(|input| input.move_down()),
139            KeyCode::PageUp => {
140                let page = self
141                    .hit_map
142                    .conversation
143                    .map_or(1, |rect| rect.height.max(1));
144                self.conversation_scroll.up(usize::from(page));
145            }
146            KeyCode::PageDown => {
147                let page = self
148                    .hit_map
149                    .conversation
150                    .map_or(1, |rect| rect.height.max(1));
151                self.conversation_scroll.down(usize::from(page));
152            }
153            KeyCode::Home if key.modifiers.contains(KeyModifiers::CONTROL) => {
154                self.navigate_input(|input| input.move_buffer_home());
155            }
156            KeyCode::Home => self.navigate_input(|input| input.move_home()),
157            KeyCode::End if key.modifiers.contains(KeyModifiers::CONTROL) => {
158                self.navigate_input(|input| input.move_buffer_end());
159            }
160            KeyCode::End => self.navigate_input(|input| input.move_end()),
161            KeyCode::Char('a') if key.modifiers == KeyModifiers::CONTROL => {
162                self.navigate_input(|input| input.move_home());
163            }
164            KeyCode::Char('e') if key.modifiers == KeyModifiers::CONTROL => {
165                self.navigate_input(|input| input.move_end());
166            }
167            KeyCode::Char(character) if is_printable_key(key) => {
168                self.mutate_input(|input| input.insert(character));
169            }
170            _ => {}
171        }
172    }

174    pub(in crate::ui) fn handle_paste(&mut self, text: &str) {
175        if self.command_palette_visible() {
176            self.paste_into_palette(text);
177            return;
178        }
179        if self.modal == Modal::Sessions {
180            let sanitized = text.replace(['\r', '\n'], "");
181            self.session_search.focus_input();
182            self.session_search.input_mut().insert_text(&sanitized);
183            self.session_search_changed();
184            return;
185        }
186        if self.modal == Modal::ConnectProviders {
187            let sanitized = text.replace(['\r', '\n'], "");
188            self.provider_search.focus_input();
189            self.provider_search.input_mut().insert_text(&sanitized);
190            self.provider_search_changed();
191            return;
192        }
193        if self.modal == Modal::Models {
194            let sanitized = text.replace(['\r', '\n'], "");
195            self.model_search.focus_input();
196            self.model_search.input_mut().insert_text(&sanitized);
197            self.model_search_changed();
198            return;
199        }
200        if self.modal == Modal::Agents {
201            let sanitized = text.replace(['\r', '\n'], "");
202            self.agent_search.focus_input();
203            self.agent_search.input_mut().insert_text(&sanitized);
204            self.agent_search_changed();
205            return;
206        }
207        if self.modal == Modal::ConnectSetup {
208            let mut sanitized = Zeroizing::new(text.replace(['\r', '\n'], ""));
209            if let Some(form) = &mut self.provider_form {
210                match form.focus() {
211                    ProviderFormFocus::Credential(index) => {
212                        form.error = None;
213                        form.secrets[index]
214                            .input
215                            .insert_owned(std::mem::take(&mut *sanitized));
216                    }
217                    ProviderFormFocus::Setup(index) => {
218                        form.error = None;
219                        form.setup[index]
220                            .input
221                            .insert_owned(std::mem::take(&mut *sanitized));
222                    }
223                    ProviderFormFocus::AuthMethod
224                    | ProviderFormFocus::Submit
225                    | ProviderFormFocus::Cancel => {}
226                }
227            }
228            return;
229        }
230        if self.modal == Modal::Mcp {
231            if let Some(input) = self
232                .mcp_panel
233                .form
234                .as_mut()
235                .and_then(McpForm::focused_input)
236            {
237                input.insert_text(&text.replace(['\r', '\n'], ""));
238            }
239            return;
240        }
241        if self.modal == Modal::Permissions {
242            if let Some(form) = &mut self.permission_panel.form
243                && form.focus_pattern
244            {
245                form.pattern.insert_text(&text.replace(['\r', '\n'], ""));
246            }
247            return;
248        }
249        // An open approval owns input; a paste must not land in the
250        // composer hidden behind it.
251        if self.modal != Modal::None || self.current_approval().is_some() {
252            return;
253        }
254        if !self.session_writable() {
255            self.input_focused = false;
256            self.status = "Session is owned by another cookie process; input is disabled.".into();
257            return;
258        }
259        self.input_focused = true;
260        let normalized = text.replace("\r\n", "\n").replace('\r', "\n");
261        self.mutate_input(|input| input.insert_text(&normalized));
262    }

265    /// composer: they run from the palette only, so every draft, including
266    /// one that begins with `/`, is sent verbatim.
267    pub(in crate::ui) async fn submit_input(&mut self) {
268        if !self.session_writable() {
269            self.status = "Session is owned by another cookie process; input is disabled.".into();
270            return;
271        }
272        if self.input.as_str().trim().is_empty() {
273            return;
274        }
275        let prompt = self.input.as_str().to_owned();
276        self.submit_prompt(prompt).await;
277    }

279    pub(in crate::ui) async fn submit_prompt(&mut self, input: String) {
280        if self.runtime.phase() == RuntimePhase::Loading {
281            self.status = "loading runtime snapshot".into();
282            return;
283        }
284        if self.runtime.phase() == RuntimePhase::ErrorRetry && self.runtime.snapshot().is_none() {
285            self.status = self
286                .runtime
287                .durable_explanation()
288                .unwrap_or("runtime snapshot unavailable; retry")
289                .into();
290            return;
291        }
292        if self.runtime.is_empty() {
293            self.status = EMPTY_RUNTIME_GUIDANCE.into();
294            return;
295        }
296        if let Some(selection) = self.new_session_draft.clone() {
297            // `/new` may be opened while an existing session remains
298            // selected. The draft is the authoritative signal that this
299            // first message belongs to a new root.
300            self.create_root_session(selection).await;
301            if self.new_session_draft.is_some() {
302                return;
303            }
304        }
305        let Some(session_id) = self.selected else {
306            self.status = "create or select a session first".into();
307            return;
308        };
309        let client = self.client.clone();
310        let updates = self.rpc_updates_tx.clone();
311        let active_run = self
312            .store
313            .sessions
314            .get(&session_id)
315            .and_then(|state| state.active_run);
316        let selection = if active_run.is_none() {
317            self.validated_draft_selection()
318        } else {
319            None
320        };
321        let reset_fallback = self.draft_reset_fallback;
322        if active_run.is_none() && selection.is_none() {
323            self.status = "select a draft agent/model before submitting".into();
324            return;
325        }
326        self.mutate_input(|input| {
327            input.take();
328        });
329        let submitted_id = client_run_id();
330        let draft_generation = self.draft_generation;
331        if active_run.is_none() {
332            self.track_fallback_reset(session_id, &submitted_id);
333        }
334        self.spawn_rpc(async move {
335            if let Some(run_id) = active_run {
336                // The engine admits steered inputs even across compaction
337                // reservations now and reports the pending lane through
338                // events; only a transport failure can strand the text, and
339                // that is owed back to the composer.
340                match client
341                    .steer_run(RunSteerParams {
342                        run_id,
343                        input: input.clone(),
344                    })
345                    .await
346                {
347                    Ok(result) if !result.accepted => {
348                        let error = result
349                            .handled_reason
350                            .unwrap_or_else(|| "steer request was rejected".into());
351                        let _ = updates.send(RpcUpdate::SteerFailed {
352                            session_id,
353                            input,
354                            error,
355                        });
356                    }
357                    Ok(result) if result.handled_reason.is_some() => {
358                        let _ = updates.send(RpcUpdate::Notice(
359                            result.handled_reason.expect("reason is present"),
360                        ));
361                    }
362                    Ok(_) => {}
363                    Err(error) => {
364                        let _ = updates.send(RpcUpdate::SteerFailed {
365                            session_id,
366                            input,
367                            error: error.to_string(),
368                        });
369                    }
370                }
371            } else {
372                let result = client
373                    .start_run(RunStartParams {
374                        reset_fallback,
375                        session_id,
376                        client_run_id: submitted_id.clone(),
377                        selection: selection.expect("draft selection checked"),
378                        input: input.clone(),
379                    })
380                    .await
381                    .map(|_| ())
382                    .map_err(|error| error.to_string());
383                let _ = updates.send(RpcUpdate::RunStartFinished {
384                    session_id,
385                    client_run_id: submitted_id,
386                    draft_generation,
387                    reset_fallback,
388                    input,
389                    result,
390                });
391            }
392        });
393    }

405    pub(in crate::ui) fn selected_queue_entries(&self) -> Vec<PendingQueueEntry> {
406        use crate::state::ProducerMessageStatus;
407
408        let Some(state) = self.selected.and_then(|id| self.store.sessions.get(&id)) else {
409            return Vec::new();
410        };
411        let mut entries = state
412            .pending_inputs
413            .iter()
414            .map(|input| PendingQueueEntry {
415                kind: QueueEntryKind::User,
416                seq: input.admission_seq,
417                accepted_at: input.admitted_at,
418                preview: input.text.clone(),
419            })
420            .collect::<Vec<_>>();
421        for item in &state.transcript {
422            let TranscriptItem::ProducerMessage {
423                seq,
424                accepted_at,
425                message_id,
426                producer_owner,
427                mode,
428                summary,
429                status,
430                ..
431            } = item
432            else {
433                continue;
434            };
435            match status {
436                ProducerMessageStatus::Pending | ProducerMessageStatus::Admitted => {}
437                ProducerMessageStatus::Claimed
438                | ProducerMessageStatus::Consumed
439                | ProducerMessageStatus::Discarded => continue,
440            };
441            entries.push(PendingQueueEntry {
442                kind: QueueEntryKind::Producer(*message_id),
443                seq: *seq,
444                accepted_at: *accepted_at,
445                preview: crate::ui::transcript::producer_summary(
446                    producer_owner,
447                    *mode,
448                    summary.as_deref(),
449                ),
450            });
451        }
452        entries.sort_by_key(|entry| entry.seq);
453        entries
454    }

456    /// Strip height in rows for the selected session's pending lane: zero
457    /// while empty so the layout never leaves a stray border behind.
458    pub(in crate::ui) fn queue_strip_height(&self) -> u16 {
459        let pending = self.selected_queue_entries();
460        if pending.is_empty() {
461            return 0;
462        }
463        // Visible entries plus block borders; the "+N more" folding row
464        // shares the entry budget, so the cap never grows past it.
465        (pending.len().min(MAX_VISIBLE_QUEUE_ROWS) as u16).saturating_add(2)
466    }

468    /// Render the pending-input strip between the conversation pane and the
469    /// status line. Only user rows can recall input for editing.
470    pub(in crate::ui) fn render_queue_strip(&mut self, frame: &mut ratatui::Frame, area: Rect) {
471        self.hit_map.queue_entries.clear();
472        if area.height == 0 || area.width < 3 {
473            return;
474        }
475        let pending = self.selected_queue_entries();
476        if pending.is_empty() {
477            return;
478        }
479        let oldest_age = jiff::Timestamp::now()
480            .duration_since(pending[0].accepted_at)
481            .as_secs()
482            .max(0);
483        let title = truncate_with_ellipsis(
484            &format!("Pending · oldest {}", queue_age_label(oldest_age)),
485            usize::from(
486                area.width
487                    .saturating_sub(2 + 2 * crate::ui::PANEL_TITLE_PAD),
488            ),
489        );
490        let block = crate::ui::panel_block()
491            .title(crate::ui::panel_title(Span::styled(
492                title,
493                self.theme.muted(),
494            )))
495            .border_style(self.theme.panel_border())
496            .style(self.theme.panel());
497        let inner = block.inner(area);
498        frame.render_widget(block, area);
499        let entry_rows = inner.height as usize;
500        if entry_rows == 0 {
501            return;
502        }
503        // Entries fill the budget; one row folds the remainder into
504        // "+N more" so the strip never grows past its cap.
505        let shown = if pending.len() > entry_rows {
506            entry_rows.saturating_sub(1)
507        } else {
508            pending.len()
509        };
510        let overflow = pending.len() - shown;
511        let mut lines = Vec::new();
512        for (index, entry) in pending.iter().enumerate().take(shown) {
513            let prefix =
514                truncate_with_ellipsis(&format!("⏳ {} ", index + 1), usize::from(inner.width));
515            let available =
516                usize::from(inner.width).saturating_sub(UnicodeWidthStr::width(prefix.as_str()));
517            lines.push(Line::from(vec![
518                Span::styled(prefix, self.theme.muted()),
519                Span::styled(
520                    if available == 0 {
521                        String::new()
522                    } else {
523                        ellipsize_single_line(&entry.preview, available)
524                    },
525                    self.theme.muted(),
526                ),
527            ]));
528        }
529        if overflow > 0 {
530            lines.push(Line::from(Span::styled(
531                truncate_with_ellipsis(&format!("+{overflow} more"), usize::from(inner.width)),
532                self.theme.muted(),
533            )));
534        }
535        let line_count = lines.len();
536        frame.render_widget(Paragraph::new(lines), inner);
537        self.hit_map.queue_entries = (0..line_count)
538            .map(|index| QueueEntryHit {
539                rect: Rect::new(
540                    inner.x,
541                    inner.y.saturating_add(index as u16),
542                    inner.width,
543                    1,
544                ),
545                index,
546                kind: if index < shown {
547                    pending[index].kind
548                } else if pending
549                    .iter()
550                    .all(|entry| entry.kind == QueueEntryKind::User)
551                {
552                    QueueEntryKind::User
553                } else {
554                    QueueEntryKind::Overflow
555                },
556            })
557            .collect();
558    }

560    /// Copy text to the system clipboard via an OSC 52 escape written to
561    /// the terminal: no platform dependency, and it works over SSH (the
562    /// terminal emulator owns the clipboard, not the host). Terminals that
563    /// ignore OSC 52 simply leave the clipboard untouched.
564    pub(super) fn copy_to_clipboard(&mut self, text: String) {
565        let characters = text.chars().count();
566        let result = match &self.clipboard_sink {
567            ClipboardSink::Osc52 => io::stdout()
568                .lock()
569                .write_all(osc52_sequence(&text).as_bytes())
570                .and_then(|()| io::stdout().flush()),
571            #[cfg(test)]
572            ClipboardSink::Capture(copied) => {
573                copied.lock().expect("clipboard capture").push(text);
574                Ok(())
575            }
576        };
577        self.status = match result {
578            Ok(()) => format!("copied {characters} characters to the clipboard"),
579            Err(error) => format!("clipboard write failed: {error}"),
580        };
581    }

583    /// Prepend restored text to the composer, preserving FIFO order for
584    /// multiple entries. Never called with an empty batch.
585    pub(super) fn restore_composer_text(&mut self, texts: Vec<String>) {
586        debug_assert!(!texts.is_empty());
587        let mut restored = texts.join("\n");
588        let existing = self.input.as_str();
589        if !existing.is_empty() {
590            restored.push('\n');
591            restored.push_str(existing);
592        }
593        self.mutate_input(|input| input.set_buffer(restored));
594        self.input_focused = true;
595    }

597    /// Restore any voided pending inputs of the viewed session into the
598    /// composer: run-end casualties and recalls that resolved while another
599    /// session was being viewed. The store holds them until this drain.
600    pub(super) fn restore_voided_inputs(&mut self) {
601        let Some(session_id) = self.selected else {
602            return;
603        };
604        let texts = self.store.take_voided_inputs(session_id);
605        if texts.is_empty() {

608        self.restore_composer_text(texts);
609        self.status = "unsent message restored to the composer".into();

612    /// Recall the engine's newest pending steered input so its text returns
613    /// to the composer for editing (`run.recall_steer`). Triggered by
614    /// clicking a strip entry or pressing Up in an empty composer; the
615    /// `UserInputRecalled` event removes the entry from the strip itself.
616    pub(in crate::ui) fn recall_newest_pending(&mut self) {
617        let Some(session_id) = self.selected else {
618            return;
619        };
620        let Some(state) = self.store.sessions.get(&session_id) else {
621            return;
622        };
623        let Some(run_id) = state.active_run else {
624            self.status = "no active run to recall from".into();
625            return;
626        };
627        if state.pending_inputs.is_empty() {
628            self.status = "no pending message to recall".into();
629            return;
630        }
631        let client = self.client.clone();
632        let updates = self.rpc_updates_tx.clone();
633        self.spawn_rpc(async move {
634            let update = match client.recall_steer(RunRecallSteerParams { run_id }).await {
635                Ok(result) => match result.recalled {
636                    Some(text) => {
637                        let _ = updates.send(RpcUpdate::SteerRecalled { session_id, text });
638                        return;
639                    }
640                    // The lane raced ahead (a promotion landed first):
641                    // nothing is owed; the strip is already catching up.
642                    None => RpcUpdate::Notice("nothing pending to recall".to_owned()),
643                },
644                Err(error) => RpcUpdate::Status(error.to_string()),
645            };
646            let _ = updates.send(update);
647        });
648    }

650    /// Open the copy/revert/fork menu for a clicked user-message row. The
651    /// message text is captured now: a rebuild (e.g. a concurrent revert)
652    /// can change the transcript before the action runs.
653    pub(super) fn open_user_menu(&mut self, hit: UserMessageHit) {
654        let Some(session_id) = self.selected else {
655            return;
656        };
657        let text = self.store.sessions.get(&session_id).and_then(|state| {
658            state.transcript.iter().find_map(|item| match item {
659                TranscriptItem::User { seq, text, .. } if *seq == hit.seq => Some(text.clone()),
660                _ => None,
661            })
662        });
663        let Some(text) = text else {
664            self.status = "message is no longer in the visible branch".into();
665            return;
666        };
667        self.user_menu = Some(UserMenuState {
668            session_id,
669            seq: hit.seq,
670            text,
671        });
672        self.picker_state.select(Some(0));
673        self.modal = Modal::UserMessage;
674    }

677        match key.code {
678            KeyCode::Esc => {
679                self.modal = Modal::None;
680                self.user_menu = None;
681            }
682            KeyCode::Up => {
683                move_picker_selection(&mut self.picker_state, USER_MENU_ITEMS.len(), true);
684            }
685            KeyCode::Down => {
686                move_picker_selection(&mut self.picker_state, USER_MENU_ITEMS.len(), false);
687            }
688            KeyCode::Enter => {
689                self.choose_picker_entry(self.picker_state.selected().unwrap_or(0))
690                    .await;
691            }
692            KeyCode::Char('c') => self.activate_user_menu_entry(0),
693            KeyCode::Char('r') => self.activate_user_menu_entry(1),
694            KeyCode::Char('f') => self.activate_user_menu_entry(2),
695            _ => {}
696        }
697    }

699    /// Run one menu row: copy, revert (behind its confirm guard), or fork.
700    pub(super) fn activate_user_menu_entry(&mut self, index: usize) {

718        }

736    /// Revert the session to just before the menu's message (`through_seq =
737    /// seq - 1`), voiding it and every later turn from the visible branch.
738    /// The physical log is append-only; the `SessionReverted` marker drives
739    /// the transcript rebuild through the normal event flow. On success the
740    /// message text restores into the composer for editing and resending.
741    pub(super) fn dispatch_session_revert(&mut self, menu: UserMenuState) {
742        // User messages always follow `SessionCreated` (sequence 1), so
743        // `seq - 1` is a positive existing physical sequence.
744        let through_seq = menu.seq.saturating_sub(1).max(1);
745        let session_id = menu.session_id;
746        let text = menu.text;
747        self.status = "reverting to before the message…".into();
748        let client = self.client.clone();
749        let updates = self.rpc_updates_tx.clone();
750        self.spawn_rpc(async move {
751            let message = match client
752                .revert_session(SessionRevertParams {
753                    session_id,
754                    through_seq,
755                })
756                .await
757            {
758                Ok(result) => {
759                    let text = result.instructions_override.unwrap_or(text);
760                    let _ = updates.send(RpcUpdate::Reverted { session_id, text });
761                    return;
762                }
763                Err(error) => format!("revert failed: {error}"),
764            };
765            let _ = updates.send(RpcUpdate::Status(message));
766        });
767    }

769    /// Fork the session at the menu's message (`through_seq = seq`, keeping
770    /// the message in the copied prefix), then switch to the new session.
771    pub(super) fn dispatch_session_fork(&mut self, menu: UserMenuState) {
772        let session_id = menu.session_id;
773        self.status = "forking the session from the message…".into();
774        let client = self.client.clone();
775        let updates = self.rpc_updates_tx.clone();
776        self.spawn_rpc(async move {
777            let message = match client
778                .fork_session(SessionForkParams {
779                    session_id,
780                    through_seq: menu.seq,
781                })
782                .await
783            {
784                Ok(result) => {
785                    let _ = updates.send(RpcUpdate::Forked {
786                        forked: result.session_id,
787                    });
788                    return;
789                }
790                Err(error) => format!("fork failed: {error}"),
791            };
792            let _ = updates.send(RpcUpdate::Status(message));
793        });
794    }

831    pub async fn send_stdin(&mut self, input: String, eof: bool) {
832        let Some((run_id, call_id)) = self.selected_running_tool() else {
833            self.status = "no running interactive tool".into();
834            return;
835        };
836        let data = (!input.is_empty()).then(|| STANDARD.encode(input.as_bytes()));
837        let client = self.client.clone();
838        let updates = self.rpc_updates_tx.clone();
839        let lane = {
840            let mut lanes = self.stdin_lanes.lock().await;
841            lanes
842                .entry(call_id)
843                .or_insert_with(|| Arc::new(tokio::sync::Mutex::new(())))
844                .clone()
845        };
846        self.spawn_rpc(async move {
847            let _guard = lane.lock().await;
848            let update = match tokio::time::timeout(
849                STDIN_RPC_TIMEOUT,
850                client.tool_stdin(RunToolStdinParams {
851                    run_id,
852                    call_id,
853                    data,
854                    eof,
855                }),
856            )
857            .await
858            {
859                Err(_) => RpcUpdate::Status("stdin request timed out".into()),
860                Ok(Ok(result)) if !result.accepted => {
861                    RpcUpdate::Status("stdin was rejected by the tool".into())
862                }
863                Ok(Ok(_)) => {
864                    if eof {
865                        RpcUpdate::Notice("tool stdin closed".into())
866                    } else {
867                        RpcUpdate::Notice("stdin sent".into())
868                    }
869                }
870                Ok(Err(error)) => RpcUpdate::Status(error.to_string()),
871            };
872            let _ = updates.send(update);
873        });
874    }

876    pub(in crate::ui) async fn run_command(&mut self, command: SlashCommand) {
877        match command {
878            SlashCommand::Quit => self.should_quit = true,
879            SlashCommand::ShowAgentPanel => {
880                self.agent_panel_mode = AgentPanelMode::Shown;
881                self.status = "agent panel shown; manual visibility override active".into();
882            }
883            SlashCommand::HideAgentPanel => {
884                self.agent_panel_mode = AgentPanelMode::Hidden;
885                self.status = "agent panel hidden; manual visibility override active".into();
886            }
887            SlashCommand::New => {
888                if self.runtime.is_empty() {
889                    self.status = EMPTY_RUNTIME_GUIDANCE.into();
890                    return;
891                }
892                self.new_session_draft =
893                    self.draft_selection_for_preset(self.selected_preset.as_deref(), None);
894                self.open_selection_modal(Modal::Agents);
895                if self.modal == Modal::Agents {
896                    self.status = "Select the agent for the new root session.".into();
897                }
898            }
899            SlashCommand::Agent => self.open_selection_modal(Modal::Agents),
900            SlashCommand::Model => {
901                self.open_selection_modal(Modal::Models);
902                // Only the palette's `/model` continues on to the variant
903                // step; the title's model segment keeps its one-step picker.
904                self.model_then_variant = self.modal == Modal::Models;
905            }
906            SlashCommand::Preset => {
907                self.modal = Modal::Presets;
908                self.picker_state.select(Some(0));
909                self.status =
910                    "Select the preset for the next root run and future new sessions.".into();
911            }
912            SlashCommand::Connect => {
913                self.clear_connect_secrets();
914                self.modal = Modal::ConnectProviders;
915                self.provider_search.reset();
916                self.picker_state.select(Some(0));
917                if self.providers.is_empty() {
918                    self.status = "No providers are available in the runtime snapshot.".into();
919                } else {
920                    self.status = "Search providers, then press Down or Tab to choose one.".into();
921                }
922            }
923            SlashCommand::Mcp => {
924                self.modal = Modal::Mcp;
925                self.mcp_panel.form = None;
926                self.poll_mcp();
927            }
928            SlashCommand::Permissions => {
929                if self.selected.is_none() {
930                    self.status = "select a session before editing permissions".into();
931                } else {
932                    self.modal = Modal::Permissions;
933                    self.permission_panel.begin_load();
934                    self.load_permissions();
935                }
936            }
937            SlashCommand::Usage => {
938                self.modal = Modal::Usage;
939                self.load_usage();
940            }
941            SlashCommand::Sessions => {
942                self.modal = Modal::Sessions;
943                self.session_search.reset();
944                self.picker_state.select(Some(0));
945            }
946            SlashCommand::Cancel => self.cancel_active_run(),
947        }
948    }

978            Ok(_) => self.status = "skill is not user-invocable".into(),
979            Err(error) => self.status = format!("load skill failed: {error}"),

983    pub(super) async fn compact_selected_session(&mut self, focus: Option<String>) {
984        let Some(session_id) = self.selected else {
985            self.status = "select a session before compacting context".into();
986            return;
987        };
988        let focus = match focus.map(SafeDisplayText::new).transpose() {
989            Ok(focus) => focus,
990            Err(_) => {
991                self.status = "compaction focus must be control-free and at most 1024 bytes".into();
992                return;
993            }
994        };
995        self.status = "compacting context…".into();
996        self.status = match self
997            .client
998            .compact_session(SessionCompactParams { session_id, focus })
999            .await
1000        {
1001            Ok(result) if result.compacted => "context compacted".into(),
1002            Ok(result) if result.cancellation_reason.is_some() => {
1003                result.cancellation_reason.expect("reason is present")
1004            }
1005            Ok(_) => "context did not require or could not produce a smaller checkpoint".into(),
1006            Err(error) => format!("context compaction failed: {error}"),
1007        };
1008    }
