6    /// Retire any composer-leg selection: its byte offsets address the
7    /// draft as it was when the drag happened, so a later buffer mutation
8    /// or cursor move would leave it painting and cutting the wrong slice.
9    /// Conversation selections address rendered lines and are unaffected.
10    pub(super) fn retire_composer_selection(&mut self) {
11        if matches!(self.selection, Some(TextSelection::Composer { .. })) {
12            self.selection = None;
13        }
14    }

17    pub(super) fn mutate_input(&mut self, mutation: impl FnOnce(&mut InputState)) {
18        mutation(&mut self.input);

29    pub(super) fn read_only_input_allowed(&self) -> bool {
30        if self.new_session_draft.is_some() {
31            return true;
32        }
33        let input = self.input.as_str().trim_start();
34        !input.is_empty() && ("/new".starts_with(input) || input.starts_with("/new "))
35    }

37    pub(in crate::ui) async fn handle_input_key(&mut self, key: KeyEvent) {
38        if self
39            .selected
40            .is_some_and(|session| self.read_only_sessions.contains(&session))
41            && !self.read_only_input_allowed()
42            && !(self.input.as_str().is_empty()
43                && key.code == KeyCode::Char('/')
44                && key.modifiers.is_empty())
45        {
46            self.input_focused = false;
47            self.status = "Session is owned by another cookie process; input is disabled.".into();
48            return;
49        }
50        if self.runtime.phase() == RuntimePhase::ErrorRetry
51            && key.code == KeyCode::Enter
52            && self.input.as_str().is_empty()
53        {
54            self.status = "Retrying runtime snapshot…".into();
55            self.refresh_coherent_lists().await;
56            return;
57        }
58        if self.runtime.phase() == RuntimePhase::Loading {
59            self.status = "loading runtime snapshot".into();
60            return;
61        }
62        if is_newline_key(key) {
63            self.input_focused = true;
64            self.mutate_input(|input| input.insert_newline());
65            return;
66        }
67        if key.code == KeyCode::Char('p') && key.modifiers == KeyModifiers::CONTROL {
68            self.input_focused = true;
69            self.mutate_input(|input| input.set_buffer("/".into()));
70            self.palette_dismissed = false;
71            self.palette_state.select(Some(0));
72            return;
73        }
74        if !self.input_focused {
75            match key.code {
76                KeyCode::Enter => self.input_focused = true,
77                KeyCode::Char(character) if is_printable_key(key) => {
78                    self.input_focused = true;
79                    if self.input.as_str().is_empty() && character == '/' {
80                        self.palette_dismissed = false;
81                    }
82                    self.mutate_input(|input| input.insert(character));
83                }
84                _ => {}
85            }
86            return;
87        }
88        match key.code {
89            KeyCode::Enter => self.submit_input().await,
90            KeyCode::Backspace if key.modifiers.contains(KeyModifiers::CONTROL) => {
91                self.mutate_input(|input| input.delete_word_left());
92            }
93            KeyCode::Backspace => {
94                self.mutate_input(|input| input.backspace());
95            }
96            KeyCode::Delete if key.modifiers.contains(KeyModifiers::CONTROL) => {
97                self.mutate_input(|input| input.delete_word_right());
98            }
99            KeyCode::Delete => {
100                self.mutate_input(|input| input.delete());
101            }
102            KeyCode::Left if key.modifiers.contains(KeyModifiers::CONTROL) => {
103                self.navigate_input(|input| input.move_word_left());
104            }
105            KeyCode::Left => self.navigate_input(|input| input.move_left()),
106            KeyCode::Right if key.modifiers.contains(KeyModifiers::CONTROL) => {
107                self.navigate_input(|input| input.move_word_right());
108            }
109            KeyCode::Right => self.navigate_input(|input| input.move_right()),
110            KeyCode::Up => {
111                // Recall gesture: with an empty composer and a non-empty
112                // pending lane, Up withdraws the newest pending message back
113                // for editing instead of moving a cursor that has nothing
114                // to move through.
115                if self.input.as_str().is_empty() && self.selected_pending_inputs().is_some() {
116                    self.recall_newest_pending();
117                } else {
118                    self.navigate_input(|input| input.move_up());
119                }
120            }
121            KeyCode::Down => self.navigate_input(|input| input.move_down()),
122            KeyCode::PageUp => {
123                let page = self
124                    .hit_map
125                    .conversation
126                    .map_or(1, |rect| rect.height.max(1));
127                self.conversation_scroll.up(usize::from(page));
128            }
129            KeyCode::PageDown => {
130                let page = self
131                    .hit_map
132                    .conversation
133                    .map_or(1, |rect| rect.height.max(1));
134                self.conversation_scroll.down(usize::from(page));
135            }
136            KeyCode::Home if key.modifiers.contains(KeyModifiers::CONTROL) => {
137                self.navigate_input(|input| input.move_buffer_home());
138            }
139            KeyCode::Home => self.navigate_input(|input| input.move_home()),
140            KeyCode::End if key.modifiers.contains(KeyModifiers::CONTROL) => {
141                self.navigate_input(|input| input.move_buffer_end());
142            }
143            KeyCode::End => self.navigate_input(|input| input.move_end()),
144            KeyCode::Char('a') if key.modifiers == KeyModifiers::CONTROL => {
145                self.navigate_input(|input| input.move_home());
146            }
147            KeyCode::Char('e') if key.modifiers == KeyModifiers::CONTROL => {
148                self.navigate_input(|input| input.move_end());
149            }
150            KeyCode::Char(character) if is_printable_key(key) => {
151                if self.input.as_str().is_empty() && character == '/' {
152                    self.palette_dismissed = false;
153                }
154                self.mutate_input(|input| input.insert(character));
155            }
156            _ => {}
157        }
158    }

160    pub(in crate::ui) fn handle_paste(&mut self, text: &str) {
161        if self.modal == Modal::Sessions {
162            let sanitized = text.replace(['\r', '\n'], "");
163            self.session_search.focus_input();
164            self.session_search.input_mut().insert_text(&sanitized);
165            self.session_search_changed();
166            return;
167        }
168        if self.modal == Modal::ConnectProviders {
169            let sanitized = text.replace(['\r', '\n'], "");
170            self.provider_search.focus_input();
171            self.provider_search.input_mut().insert_text(&sanitized);
172            self.provider_search_changed();
173            return;
174        }
175        if self.modal == Modal::Models {
176            let sanitized = text.replace(['\r', '\n'], "");
177            self.model_search.focus_input();
178            self.model_search.input_mut().insert_text(&sanitized);
179            self.model_search_changed();
180            return;
181        }
182        if self.modal == Modal::Agents {
183            let sanitized = text.replace(['\r', '\n'], "");
184            self.agent_search.focus_input();
185            self.agent_search.input_mut().insert_text(&sanitized);
186            self.agent_search_changed();
187            return;
188        }
189        if self.modal == Modal::ConnectSetup {
190            let mut sanitized = Zeroizing::new(text.replace(['\r', '\n'], ""));
191            if let Some(form) = &mut self.provider_form {
192                match form.focus() {
193                    ProviderFormFocus::Credential(index) => {
194                        form.error = None;
195                        form.secrets[index]
196                            .input
197                            .insert_owned(std::mem::take(&mut *sanitized));
198                    }
199                    ProviderFormFocus::Setup(index) => {
200                        form.error = None;
201                        form.setup[index]
202                            .input
203                            .insert_owned(std::mem::take(&mut *sanitized));
204                    }
205                    ProviderFormFocus::AuthMethod
206                    | ProviderFormFocus::Submit
207                    | ProviderFormFocus::Cancel => {}
208                }
209            }
210            return;
211        }
212        if self.modal == Modal::Mcp {
213            if let Some(input) = self
214                .mcp_panel
215                .form
216                .as_mut()
217                .and_then(McpForm::focused_input)
218            {
219                input.insert_text(&text.replace(['\r', '\n'], ""));
220            }
221            return;
222        }
223        if self.modal == Modal::Permissions {
224            if let Some(form) = &mut self.permission_panel.form
225                && form.focus_pattern
226            {
227                form.pattern.insert_text(&text.replace(['\r', '\n'], ""));
228            }
229            return;
230        }
231        if self.modal != Modal::None {
232            return;
233        }
234        if self
235            .selected
236            .is_some_and(|session| self.read_only_sessions.contains(&session))
237            && !self.read_only_input_allowed()
238            && !text.trim_start().starts_with("/new")
239        {
240            self.input_focused = false;
241            self.status = "Session is owned by another cookie process; input is disabled.".into();
242            return;
243        }
244        self.input_focused = true;
245        if self.input.as_str().is_empty() && text.starts_with('/') {
246            self.palette_dismissed = false;
247        }
248        let normalized = text.replace("\r\n", "\n").replace('\r', "\n");
249        self.mutate_input(|input| input.insert_text(&normalized));
250    }

252    pub(in crate::ui) async fn submit_input(&mut self) {
253        if self
254            .selected
255            .is_some_and(|session| self.read_only_sessions.contains(&session))
256            && !self.read_only_input_allowed()
257        {
258            self.status = "Session is owned by another cookie process; input is disabled.".into();
259            return;
260        }
261        if self.input.as_str().trim().is_empty() {
262            return;
263        }
264        let skills = self
265            .skills
266            .iter()
267            .filter(|skill| skill.precedence_winner && skill.user_invocable)
268            .map(|skill| skill.name.clone())
269            .collect::<Vec<_>>();
270        let mut submission = parse_submission_with_skills(self.input.as_str(), &skills);
271        let could_be_pending_skill = self
272            .input
273            .as_str()
274            .strip_prefix('/')
275            .and_then(|input| input.split_whitespace().next())
276            .is_some_and(|name| {
277                !COMMANDS
278                    .iter()
279                    .any(|spec| spec.name == name || spec.aliases.contains(&name))
280            });
281        if submission.is_err() && self.new_session_draft.is_some() && could_be_pending_skill {
282            let selection = self
283                .new_session_draft
284                .clone()
285                .expect("pending new-session draft");
286            if self.create_root_session(selection).await {
287                let Some(session_id) = self.selected else {
288                    self.status = "new session was created without a selection".into();
289                    return;
290                };
291                match self
292                    .client
293                    .list_skills(cookie_agent_protocol::SkillsListParams { session_id })
294                    .await
295                {
296                    Ok(result) => {
297                        self.skills = result.skills.clone();
298                        self.skill_panel.install(result);
299                        let skill_names = self
300                            .skills
301                            .iter()
302                            .filter(|skill| skill.precedence_winner && skill.user_invocable)
303                            .map(|skill| skill.name.clone())
304                            .collect::<Vec<_>>();
305                        submission =
306                            parse_submission_with_skills(self.input.as_str(), &skill_names);
307                    }
308                    Err(error) => {
309                        self.status = format!("skill discovery failed: {error}");
310                        return;
311                    }
312                }
313            } else {
314                // Keep both the original slash input and pending selection for retry.
315                return;
316            }
317        }
318        let submission = match submission {
319            Ok(submission) => submission,
320            Err(error) => {
321                self.mutate_input(|input| {
322                    input.take();
323                });
324                self.palette_dismissed = false;
325                self.status = error;
326                return;
327            }
328        };
329        match submission {
330            Submission::Command(command) => {
331                self.mutate_input(|input| {
332                    input.take();
333                });
334                self.palette_dismissed = false;
335                self.run_command(command).await;
336            }
337            Submission::Prompt(prompt) => self.submit_prompt(prompt).await,
338        }
339    }

341    pub(in crate::ui) async fn submit_prompt(&mut self, input: String) {
342        if self.runtime.phase() == RuntimePhase::Loading {
343            self.status = "loading runtime snapshot".into();
344            return;
345        }
346        if self.runtime.phase() == RuntimePhase::ErrorRetry && self.runtime.snapshot().is_none() {
347            self.status = self
348                .runtime
349                .durable_explanation()
350                .unwrap_or("runtime snapshot unavailable; retry")
351                .into();
352            return;
353        }
354        if self.runtime.is_empty() {
355            self.status = EMPTY_RUNTIME_GUIDANCE.into();
356            return;
357        }
358        if let Some(selection) = self.new_session_draft.clone() {
359            // `/new` may be opened while an existing session remains
360            // selected. The draft is the authoritative signal that this
361            // first message belongs to a new root.
362            self.create_root_session(selection).await;
363            if self.new_session_draft.is_some() {
364                return;
365            }
366        }
367        let Some(session_id) = self.selected else {
368            self.status = "create or select a session first".into();
369            return;
370        };
371        let client = self.client.clone();
372        let updates = self.rpc_updates_tx.clone();
373        let active_run = self
374            .store
375            .sessions
376            .get(&session_id)
377            .and_then(|state| state.active_run);
378        let selection = if active_run.is_none() {
379            self.validated_draft_selection()
380        } else {
381            None
382        };
383        let reset_fallback = self.draft_reset_fallback;
384        if active_run.is_none() && selection.is_none() {
385            self.status = "select a draft agent/model before submitting".into();
386            return;
387        }
388        self.mutate_input(|input| {
389            input.take();
390        });
391        self.palette_dismissed = false;
392        let submitted_id = client_run_id();
393        let draft_generation = self.draft_generation;
394        if active_run.is_none() {
395            self.track_fallback_reset(session_id, &submitted_id);
396        }
397        self.spawn_rpc(async move {
398            if let Some(run_id) = active_run {
399                // The engine admits steered inputs even across compaction
400                // reservations now and reports the pending lane through
401                // events; only a transport failure can strand the text, and
402                // that is owed back to the composer.
403                match client
404                    .steer_run(RunSteerParams {
405                        run_id,
406                        input: input.clone(),
407                    })
408                    .await
409                {
410                    Ok(result) if !result.accepted => {
411                        let error = result
412                            .handled_reason
413                            .unwrap_or_else(|| "steer request was rejected".into());
414                        let _ = updates.send(RpcUpdate::SteerFailed {
415                            session_id,
416                            input,
417                            error,
418                        });
419                    }
420                    Ok(result) if result.handled_reason.is_some() => {
421                        let _ = updates.send(RpcUpdate::Notice(
422                            result.handled_reason.expect("reason is present"),
423                        ));
424                    }
425                    Ok(_) => {}
426                    Err(error) => {
427                        let _ = updates.send(RpcUpdate::SteerFailed {
428                            session_id,
429                            input,
430                            error: error.to_string(),
431                        });
432                    }
433                }
434            } else {
435                let result = client
436                    .start_run(RunStartParams {
437                        reset_fallback,
438                        session_id,
439                        client_run_id: submitted_id.clone(),
440                        selection: selection.expect("draft selection checked"),
441                        input: input.clone(),
442                    })
443                    .await
444                    .map(|_| ())
445                    .map_err(|error| error.to_string());
446                let _ = updates.send(RpcUpdate::RunStartFinished {
447                    session_id,
448                    client_run_id: submitted_id,
449                    draft_generation,
450                    reset_fallback,
451                    input,
452                    result,
453                });
454            }
455        });
456    }

458    /// The viewed session's pending steered inputs, when the lane is

468    pub(in crate::ui) fn selected_queue_entries(&self) -> Vec<PendingQueueEntry> {
469        use crate::state::ProducerMessageStatus;
470
471        let Some(state) = self.selected.and_then(|id| self.store.sessions.get(&id)) else {
472            return Vec::new();
473        };
474        let mut entries = state
475            .pending_inputs
476            .iter()
477            .map(|input| PendingQueueEntry {
478                kind: QueueEntryKind::User,
479                seq: input.admission_seq,
480                accepted_at: input.admitted_at,
481                preview: input.text.clone(),
482            })
483            .collect::<Vec<_>>();
484        for item in &state.transcript {
485            let TranscriptItem::ProducerMessage {
486                seq,
487                accepted_at,
488                message_id,
489                producer_owner,
490                mode,
491                summary,
492                status,
493                ..
494            } = item
495            else {
496                continue;
497            };
498            match status {
499                ProducerMessageStatus::Pending | ProducerMessageStatus::Admitted => {}
500                ProducerMessageStatus::Claimed
501                | ProducerMessageStatus::Consumed
502                | ProducerMessageStatus::Discarded => continue,
503            };
504            entries.push(PendingQueueEntry {
505                kind: QueueEntryKind::Producer(*message_id),
506                seq: *seq,
507                accepted_at: *accepted_at,
508                preview: crate::ui::transcript::producer_summary(
509                    producer_owner,
510                    *mode,
511                    summary.as_deref(),
512                ),
513            });
514        }
515        entries.sort_by_key(|entry| entry.seq);
516        entries
517    }

519    /// Strip height in rows for the selected session's pending lane: zero
520    /// while empty so the layout never leaves a stray border behind.
521    pub(in crate::ui) fn queue_strip_height(&self) -> u16 {
522        let pending = self.selected_queue_entries();
523        if pending.is_empty() {
524            return 0;
525        }
526        // Visible entries plus block borders; the "+N more" folding row
527        // shares the entry budget, so the cap never grows past it.
528        (pending.len().min(MAX_VISIBLE_QUEUE_ROWS) as u16).saturating_add(2)
529    }

531    /// Render the pending-input strip between the conversation pane and the
532    /// status line. Only user rows can recall input for editing.
533    pub(in crate::ui) fn render_queue_strip(&mut self, frame: &mut ratatui::Frame, area: Rect) {
534        self.hit_map.queue_entries.clear();
535        if area.height == 0 || area.width < 3 {
536            return;
537        }
538        let pending = self.selected_queue_entries();
539        if pending.is_empty() {
540            return;
541        }
542        let oldest_age = jiff::Timestamp::now()
543            .duration_since(pending[0].accepted_at)
544            .as_secs()
545            .max(0);
546        let title = truncate_with_ellipsis(
547            &format!("Pending · oldest {}", queue_age_label(oldest_age)),
548            usize::from(area.width.saturating_sub(2)),
549        );
550        let block = Block::bordered()
551            .title(Span::styled(title, self.theme.muted()))
552            .border_style(self.theme.panel_border())
553            .style(self.theme.panel());
554        let inner = block.inner(area);
555        frame.render_widget(block, area);
556        let entry_rows = inner.height as usize;
557        if entry_rows == 0 {
558            return;
559        }
560        // Entries fill the budget; one row folds the remainder into
561        // "+N more" so the strip never grows past its cap.
562        let shown = if pending.len() > entry_rows {
563            entry_rows.saturating_sub(1)
564        } else {
565            pending.len()
566        };
567        let overflow = pending.len() - shown;
568        let mut lines = Vec::new();
569        for (index, entry) in pending.iter().enumerate().take(shown) {
570            let prefix =
571                truncate_with_ellipsis(&format!("⏳ {} ", index + 1), usize::from(inner.width));
572            let available =
573                usize::from(inner.width).saturating_sub(UnicodeWidthStr::width(prefix.as_str()));
574            lines.push(Line::from(vec![
575                Span::styled(prefix, self.theme.muted()),
576                Span::styled(
577                    if available == 0 {
578                        String::new()
579                    } else {
580                        ellipsize_single_line(&entry.preview, available)
581                    },
582                    self.theme.muted(),
583                ),
584            ]));
585        }
586        if overflow > 0 {
587            lines.push(Line::from(Span::styled(
588                truncate_with_ellipsis(&format!("+{overflow} more"), usize::from(inner.width)),
589                self.theme.muted(),
590            )));
591        }
592        let line_count = lines.len();
593        frame.render_widget(Paragraph::new(lines), inner);
594        self.hit_map.queue_entries = (0..line_count)
595            .map(|index| QueueEntryHit {
596                rect: Rect::new(
597                    inner.x,
598                    inner.y.saturating_add(index as u16),
599                    inner.width,
600                    1,
601                ),
602                index,
603                kind: if index < shown {
604                    pending[index].kind
605                } else if pending
606                    .iter()
607                    .all(|entry| entry.kind == QueueEntryKind::User)
608                {
609                    QueueEntryKind::User
610                } else {
611                    QueueEntryKind::Overflow
612                },
613            })
614            .collect();
615    }

617    /// Copy text to the system clipboard via an OSC 52 escape written to
618    /// the terminal: no platform dependency, and it works over SSH (the
619    /// terminal emulator owns the clipboard, not the host). Terminals that
620    /// ignore OSC 52 simply leave the clipboard untouched.
621    pub(super) fn copy_to_clipboard(&mut self, text: String) {
622        let characters = text.chars().count();
623        let result = match &self.clipboard_sink {
624            ClipboardSink::Osc52 => io::stdout()
625                .lock()
626                .write_all(osc52_sequence(&text).as_bytes())
627                .and_then(|()| io::stdout().flush()),
628            #[cfg(test)]
629            ClipboardSink::Capture(copied) => {
630                copied.lock().expect("clipboard capture").push(text);
631                Ok(())
632            }
633        };
634        self.status = match result {
635            Ok(()) => format!("copied {characters} characters to the clipboard"),
636            Err(error) => format!("clipboard write failed: {error}"),
637        };
638    }

640    /// Prepend restored text to the composer, preserving FIFO order for
641    /// multiple entries. Never called with an empty batch.
642    pub(super) fn restore_composer_text(&mut self, texts: Vec<String>) {
643        debug_assert!(!texts.is_empty());
644        let mut restored = texts.join("\n");
645        let existing = self.input.as_str();
646        if !existing.is_empty() {
647            restored.push('\n');
648            restored.push_str(existing);
649        }
650        self.mutate_input(|input| input.set_buffer(restored));
651        self.input_focused = true;
652    }

654    /// Restore any voided pending inputs of the viewed session into the
655    /// composer: run-end casualties and recalls that resolved while another
656    /// session was being viewed. The store holds them until this drain.
657    pub(super) fn restore_voided_inputs(&mut self) {
658        let Some(session_id) = self.selected else {
659            return;
660        };
661        let texts = self.store.take_voided_inputs(session_id);
662        if texts.is_empty() {
663            return;
664        }
665        self.restore_composer_text(texts);
666        self.status = "unsent message restored to the composer".into();

669    /// Recall the engine's newest pending steered input so its text returns
670    /// to the composer for editing (`run.recall_steer`). Triggered by
671    /// clicking a strip entry or pressing Up in an empty composer; the
672    /// `UserInputRecalled` event removes the entry from the strip itself.
673    pub(in crate::ui) fn recall_newest_pending(&mut self) {
674        let Some(session_id) = self.selected else {
675            return;
676        };
677        let Some(state) = self.store.sessions.get(&session_id) else {
678            return;
679        };
680        let Some(run_id) = state.active_run else {
681            self.status = "no active run to recall from".into();
682            return;
683        };
684        if state.pending_inputs.is_empty() {
685            self.status = "no pending message to recall".into();
686            return;
687        }
688        let client = self.client.clone();
689        let updates = self.rpc_updates_tx.clone();
690        self.spawn_rpc(async move {
691            let update = match client.recall_steer(RunRecallSteerParams { run_id }).await {
692                Ok(result) => match result.recalled {
693                    Some(text) => {
694                        let _ = updates.send(RpcUpdate::SteerRecalled { session_id, text });
695                        return;
696                    }
697                    // The lane raced ahead (a promotion landed first):
698                    // nothing is owed; the strip is already catching up.
699                    None => RpcUpdate::Notice("nothing pending to recall".to_owned()),
700                },
701                Err(error) => RpcUpdate::Status(error.to_string()),
702            };
703            let _ = updates.send(update);
704        });
705    }

793    /// Revert the session to just before the menu's message (`through_seq =
794    /// seq - 1`), voiding it and every later turn from the visible branch.
795    /// The physical log is append-only; the `SessionReverted` marker drives
796    /// the transcript rebuild through the normal event flow. On success the
797    /// message text restores into the composer for editing and resending.
798    pub(super) fn dispatch_session_revert(&mut self, menu: UserMenuState) {
799        // User messages always follow `SessionCreated` (sequence 1), so
800        // `seq - 1` is a positive existing physical sequence.
801        let through_seq = menu.seq.saturating_sub(1).max(1);
802        let session_id = menu.session_id;
803        let text = menu.text;
804        self.status = "reverting to before the message…".into();
805        let client = self.client.clone();
806        let updates = self.rpc_updates_tx.clone();
807        self.spawn_rpc(async move {
808            let message = match client
809                .revert_session(SessionRevertParams {
810                    session_id,
811                    through_seq,
812                })
813                .await
814            {
815                Ok(result) => {
816                    let text = result.instructions_override.unwrap_or(text);
817                    let _ = updates.send(RpcUpdate::Reverted { session_id, text });
818                    return;
819                }
820                Err(error) => format!("revert failed: {error}"),
821            };
822            let _ = updates.send(RpcUpdate::Status(message));
823        });
824    }

826    /// Fork the session at the menu's message (`through_seq = seq`, keeping
827    /// the message in the copied prefix), then switch to the new session.
828    pub(super) fn dispatch_session_fork(&mut self, menu: UserMenuState) {
829        let session_id = menu.session_id;
830        self.status = "forking the session from the message…".into();
831        let client = self.client.clone();
832        let updates = self.rpc_updates_tx.clone();
833        self.spawn_rpc(async move {
834            let message = match client
835                .fork_session(SessionForkParams {
836                    session_id,
837                    through_seq: menu.seq,
838                })
839                .await
840            {
841                Ok(result) => {
842                    let _ = updates.send(RpcUpdate::Forked {
843                        forked: result.session_id,
844                    });
845                    return;
846                }
847                Err(error) => format!("fork failed: {error}"),
848            };
849            let _ = updates.send(RpcUpdate::Status(message));
850        });
851    }

889    pub async fn send_stdin(&mut self, input: String, eof: bool) {
890        let Some((run_id, call_id)) = self.selected_running_tool() else {
891            self.status = "no running interactive tool".into();
892            return;
893        };
894        let data = (!input.is_empty()).then(|| STANDARD.encode(input.as_bytes()));
895        let client = self.client.clone();
896        let updates = self.rpc_updates_tx.clone();
897        let lane = {
898            let mut lanes = self.stdin_lanes.lock().await;
899            lanes
900                .entry(call_id)
901                .or_insert_with(|| Arc::new(tokio::sync::Mutex::new(())))
902                .clone()
903        };
904        self.spawn_rpc(async move {
905            let _guard = lane.lock().await;
906            let update = match tokio::time::timeout(
907                STDIN_RPC_TIMEOUT,
908                client.tool_stdin(RunToolStdinParams {
909                    run_id,
910                    call_id,
911                    data,
912                    eof,
913                }),
914            )
915            .await
916            {
917                Err(_) => RpcUpdate::Status("stdin request timed out".into()),
918                Ok(Ok(result)) if !result.accepted => {
919                    RpcUpdate::Status("stdin was rejected by the tool".into())
920                }
921                Ok(Ok(_)) => {
922                    if eof {
923                        RpcUpdate::Notice("tool stdin closed".into())
924                    } else {
925                        RpcUpdate::Notice("stdin sent".into())
926                    }
927                }
928                Ok(Err(error)) => RpcUpdate::Status(error.to_string()),
929            };
930            let _ = updates.send(update);
931        });
932    }

984                    self.permission_panel.begin_load();
985                    self.load_permissions();
986                }
987            }

1004                    Err(error) => self.status = format!("list skills failed: {error}"),
1005                }

1010                    return;

1015                        session_id,
1016                        name: name.clone(),
1017                        args: args.clone(),
1018                    })
1019                    .await
1020                {
1021                    Ok(result) if result.skill.user_invocable => {
1022                        self.submit_prompt(cookie_agent_protocol::encode_skill_submission(
1023                            &name, &args,
1024                        ))
1025                        .await;
1026                    }
1027                    Ok(_) => self.status = "skill is not user-invocable".into(),
1028                    Err(error) => self.status = format!("load skill failed: {error}"),
1029                }
1030            }
1031            SlashCommand::Usage => {
1032                self.modal = Modal::Usage;
1033                self.load_usage();
1034            }
1035            SlashCommand::Sessions => {
1036                self.modal = Modal::Sessions;
1037                self.session_search.reset();
1038                self.picker_state.select(Some(0));
1039            }
1040            SlashCommand::Cancel => self.cancel_active_run(),
1041            SlashCommand::Goal(command) => self.run_goal_command(command),
1042            SlashCommand::Compact(focus) => self.compact_selected_session(focus).await,
1043            SlashCommand::Approve(decision) => self.answer_approval(decision).await,
1044            SlashCommand::Events(level) => {
1045                // View-only threshold change: the TOML is not rewritten and
1046                // hidden rows stay in the session projection.
1047                self.tui_config.minimum_event_level = level;
1048                self.status = format!("diagnostic event filter: {}", level.name());
1049            }
1050            SlashCommand::Help => self.show_help(),
1051        }
1052    }

1054    pub(super) async fn compact_selected_session(&mut self, focus: Option<String>) {
1055        let Some(session_id) = self.selected else {
1056            self.status = "select a session before compacting context".into();
1057            return;
1058        };
1059        let focus = match focus.map(SafeDisplayText::new).transpose() {
1060            Ok(focus) => focus,
1061            Err(_) => {
1062                self.status = "compaction focus must be control-free and at most 1024 bytes".into();
1063                return;
1064            }
1065        };
1066        self.status = "compacting context…".into();
1067        self.status = match self
1068            .client
1069            .compact_session(SessionCompactParams { session_id, focus })
1070            .await
1071        {
1072            Ok(result) if result.compacted => "context compacted".into(),
1073            Ok(result) if result.cancellation_reason.is_some() => {
1074                result.cancellation_reason.expect("reason is present")
1075            }
1076            Ok(_) => "context did not require or could not produce a smaller checkpoint".into(),
1077            Err(error) => format!("context compaction failed: {error}"),
1078        };
1079    }

1081    pub(in crate::ui) fn show_help(&mut self) {
1082        // One line per command in the transcript; the status line stays a
1083        // short pointer instead of a truncated wall of text.
1084        let notice = std::iter::once("Available commands:".to_owned())
1085            .chain(
1086                COMMANDS
1087                    .iter()
1088                    .filter(|spec| self.command_is_available(spec))
1089                    .map(|spec| format!("{} — {}", spec.usage, spec.description)),
1090            )
1091            .chain(std::iter::once(
1092                "Use // to send a prompt beginning with /.".to_owned(),
1093            ))
1094            .collect::<Vec<_>>()
1095            .join("\n");
1096        self.status = "commands listed in the conversation".into();
1097        self.transient_notices.push(notice);
1098        if self.transient_notices.len() > MAX_TRANSIENT_NOTICES {
1099            let excess = self.transient_notices.len() - MAX_TRANSIENT_NOTICES;
1100            self.transient_notices.drain(..excess);
1101        }
1102    }
