21/// The text a new agent document starts from.
22pub(super) const NEW_AGENT: &str = "---\ndescription: \nmode: subagent\nenabled: true\nmodels: [{ model: \"\" }]\npermissions: {}\n---\n";

24/// A document to open in the editor; the event loop runs the editor and
25/// hands the result to [`App::finish_config_edit`].
26#[derive(Clone, Debug, Eq, PartialEq)]
27pub(in crate::ui) struct EditorRequest {
28    pub(in crate::ui) path: String,
29    pub(in crate::ui) text: String,
30    /// The document's text on disk, `None` for a new one.
31    pub(in crate::ui) original: Option<String>,
32    pub(in crate::ui) expected_fingerprint: Option<String>,
33    pub(in crate::ui) revision: ConfigRevision,
34}
35
36/// What `config.apply` answered for an edit.
37#[derive(Debug)]
38pub(in crate::ui) enum ConfigApplyOutcome {
39    Applied(ConfigStatus),
40    Refused(ConfigApplyError),
41    Failed(String),
42}
43
44impl ConfigApplyOutcome {
45    fn from_result(result: Result<cookie_agent_protocol::ConfigApplyResult, ClientError>) -> Self {
46        match result {
47            Ok(result) => Self::Applied(result.status),
48            Err(ClientError::Rpc(rpc))
49                if rpc.code == cookie_agent_protocol::CONFIG_APPLY_ERROR_CODE =>
50            {
51                rpc.data
52                    .and_then(|data| serde_json::from_value(data).ok())
53                    .map_or_else(|| Self::Failed(rpc.message), Self::Refused)
54            }
55            Err(error) => Self::Failed(error.to_string()),
56        }
57    }
58}

84impl ConfigPanelState {
85    pub(in crate::ui) fn rows(&self) -> Vec<ConfigRow> {
86        let Some(config) = &self.config else {
87            return Vec::new();
88        };
89        let mut rows = vec![ConfigRow {
90            label: if config
91                .documents
92                .iter()
93                .any(|document| document.path == "config.toml")
94            {
95                "config.toml".into()
96            } else {
97                "config.toml (not created)".into()
98            },
99            path: Some("config.toml".into()),
100            preset: None,
101        }];
102        let agent = |path: &str| -> Option<(Option<String>, String)> {
103            let rest = path.strip_prefix("agents/")?.strip_suffix(".md")?;
104            Some(match rest.split_once('/') {
105                Some((preset, id)) => (Some(preset.to_owned()), id.to_owned()),
106                None => (None, rest.to_owned()),
107            })
108        };
109        let mut presets = std::collections::BTreeMap::<String, Vec<(String, String)>>::new();
110        for preset in &config.empty_presets {
111            presets.entry(preset.clone()).or_default();
112        }
113        for document in &config.documents {
114            match agent(&document.path) {
115                Some((None, id)) => rows.push(ConfigRow {
116                    label: format!("agent {id}"),
117                    path: Some(document.path.clone()),
118                    preset: None,
119                }),
120                Some((Some(preset), id)) => presets
121                    .entry(preset)
122                    .or_default()
123                    .push((id, document.path.clone())),
124                None => {}
125            }
126        }
127        for (preset, agents) in presets {
128            rows.push(ConfigRow {
129                label: format!("preset {preset}"),
130                path: None,
131                preset: Some(preset.clone()),
132            });
133            rows.extend(agents.into_iter().map(|(id, path)| ConfigRow {
134                label: format!("  agent {id}"),
135                path: Some(path),
136                preset: Some(preset.clone()),
137            }));
138        }
139        rows
140    }
141
142    fn selected_row(&self) -> Option<ConfigRow> {
143        self.selection
144            .selected()
145            .and_then(|index| self.rows().into_iter().nth(index))
146    }
147}

150    pub(super) fn open_config_panel(&mut self) {
151        let drafts = std::mem::take(&mut self.config_panel.drafts);
152        self.config_panel = ConfigPanelState {
153            drafts,
154            ..ConfigPanelState::default()
155        };
156        self.modal = Modal::Config;
157        self.poll_config();
158    }

160    pub(super) fn poll_config(&mut self) {
161        if self.modal != Modal::Config || self.config_panel.refresh_in_flight {
162            return;
163        }
164        self.config_panel.refresh_in_flight = true;
165        let client = self.client.clone();
166        let updates = self.rpc_updates_tx.clone();
167        self.spawn_rpc(async move {
168            let result = client.get_config().await.map_err(|error| error.to_string());
169            let _ = updates.send(RpcUpdate::ConfigLoaded { result });
170        });
171    }

173    pub(super) fn install_config(&mut self, result: Result<ConfigGetResult, String>) {
174        self.config_panel.refresh_in_flight = false;
175        match result {
176            Ok(config) => self.config_panel.config = Some(config),
177            Err(error) => self.status = format!("configuration unavailable: {error}"),
178        }
179    }

181    /// Opens `path` in the editor: a refused edit's text when there is one,
182    /// else the document, else a new document's template.
183    pub(in crate::ui) fn begin_config_edit(&mut self, path: &str) {
184        if let Some(draft) = self.config_panel.drafts.get(path) {
185            self.pending_editor = Some(draft.clone());
186            return;
187        }
188        let Some(config) = &self.config_panel.config else {
189            return;
190        };
191        let document = config
192            .documents
193            .iter()
194            .find(|document| document.path == path);
195        let original = document.map(|document| document.text.clone());
196        self.pending_editor = Some(EditorRequest {
197            path: path.to_owned(),
198            text: original.clone().unwrap_or_else(|| {
199                if path.ends_with(".md") {
200                    NEW_AGENT.to_owned()
201                } else {
202                    String::new()
203                }
204            }),
205            original,
206            expected_fingerprint: document.map(|document| document.fingerprint.clone()),
207            revision: config.revision.clone(),
208        });
209    }

211    /// What the editor left for `request`: applied unless unchanged.
212    pub(in crate::ui) fn finish_config_edit(
213        &mut self,
214        request: EditorRequest,
215        edited: Result<String, String>,
216    ) {
217        let text = match edited {
218            Ok(text) => text,
219            Err(error) => {
220                self.status = error;
221                return;
222            }
223        };
224        if request.original.as_deref() == Some(text.as_str()) {
225            self.config_panel.drafts.remove(&request.path);
226            self.status = format!("No changes to {}.", request.path);
227            return;
228        }
229        let request = EditorRequest { text, ..request };
230        self.status = format!("applying {}…", request.path);
231        let change = ConfigChange::Put {
232            path: request.path.clone(),
233            text: request.text.clone(),
234            expected_fingerprint: request.expected_fingerprint.clone(),
235        };
236        self.dispatch_config_apply(request.revision.clone(), change, Some(request));
237    }
238
239    fn dispatch_config_apply(
240        &mut self,
241        revision: ConfigRevision,
242        change: ConfigChange,
243        edit: Option<EditorRequest>,
244    ) {
245        let client = self.client.clone();
246        let updates = self.rpc_updates_tx.clone();
247        let path = match &change {
248            ConfigChange::Put { path, .. } | ConfigChange::Delete { path, .. } => path.clone(),
249            ConfigChange::Set { key, .. } => key.join("."),
250        };
251        self.spawn_rpc(async move {
252            let outcome = ConfigApplyOutcome::from_result(
253                client
254                    .apply_config(ConfigApplyParams {
255                        expected_revision: revision,
256                        changes: vec![change],
257                    })
258                    .await,
259            );
260            let _ = updates.send(RpcUpdate::ConfigApplied {
261                path,
262                edit,
263                outcome,
264            });
265        });
266    }

268    pub(super) fn install_config_apply(
269        &mut self,
270        path: String,
271        edit: Option<EditorRequest>,
272        outcome: ConfigApplyOutcome,
273    ) {
274        match outcome {
275            ConfigApplyOutcome::Applied(status) => {
276                self.config_panel.drafts.remove(&path);
277                self.config_panel.error = None;
278                self.status = format!("Applied {path}.{}", restart_note(&status));
279            }
280            ConfigApplyOutcome::Refused(ConfigApplyError {
281                code: ConfigApplyErrorCode::Conflict,
282                message,
283            }) => {
284                self.config_panel.drafts.remove(&path);
285                self.config_panel.error = Some(format!(
286                    "{path} was not changed: {message}. The list is reloaded; edit it again."
287                ));
288                self.status = format!("{path} changed meanwhile; edit it again");
289            }
290            ConfigApplyOutcome::Refused(error) => {
291                if let Some(edit) = edit {
292                    self.config_panel.drafts.insert(path.clone(), edit);
293                }
294                self.config_panel.error =
295                    Some(format!("{path} was not changed: {}", error.message));
296                self.status = format!("Not applied; Enter edits {path} again.");
297            }
298            ConfigApplyOutcome::Failed(error) => self.status = error,
299        }
300        self.poll_config();
301    }

303    pub(super) fn handle_config_key(&mut self, key: KeyEvent) {
304        if let Some(input) = self.config_panel.new_path.as_mut() {
305            match key.code {
306                KeyCode::Esc => self.config_panel.new_path = None,
307                KeyCode::Enter => {
308                    let path = input.as_str().trim().to_owned();
309                    self.config_panel.new_path = None;
310                    let exists = self.config_panel.config.as_ref().is_some_and(|config| {
311                        config
312                            .documents
313                            .iter()
314                            .any(|document| document.path == path)
315                    });
316                    if !path.starts_with("agents/") || !path.ends_with(".md") {
317                        self.status =
318                            "a new agent is agents/<id>.md or agents/<preset>/<id>.md".into();
319                    } else if exists {
320                        self.status = format!("{path} exists; Enter edits it");
321                    } else {
322                        self.begin_config_edit(&path);
323                    }
324                }
325                _ => super::keys::edit_plain_input(input, key),
326            }
327            return;
328        }
329        if let Some(path) = self.config_panel.confirm_delete.take() {
330            if matches!(key.code, KeyCode::Char('y' | 'Y')) {
331                self.delete_config_document(&path);
332            } else {
333                self.status = "Delete cancelled.".into();
334            }
335            return;
336        }
337        let count = self.config_panel.rows().len();
338        match key.code {
339            KeyCode::Esc => self.modal = Modal::None,
340            KeyCode::Up => {
341                super::move_picker_selection(&mut self.config_panel.selection, count, true);
342            }
343            KeyCode::Down => {
344                super::move_picker_selection(&mut self.config_panel.selection, count, false);
345            }
346            KeyCode::Enter => {
347                if let Some(path) = self.config_panel.selected_row().and_then(|row| row.path) {
348                    self.begin_config_edit(&path);
349                }
350            }
351            KeyCode::Char('n') => {
352                let prefix = match self.config_panel.selected_row().and_then(|row| row.preset) {
353                    Some(preset) => format!("agents/{preset}/"),
354                    None => "agents/".to_owned(),
355                };
356                let mut input = InputState::default();
357                input.set_buffer(prefix);
358                self.config_panel.new_path = Some(input);
359            }
360            KeyCode::Char('d') => match self.config_panel.selected_row().and_then(|row| row.path) {
361                Some(path) if path != "config.toml" => {
362                    self.config_panel.confirm_delete = Some(path);
363                }
364                _ => self.status = "only an agent document can be deleted".into(),
365            },
366            KeyCode::Char('r') => self.reload_config(),
367            _ => {}
368        }
369    }

371    fn delete_config_document(&mut self, path: &str) {
372        let Some(config) = &self.config_panel.config else {
373            return;
374        };
375        let Some(document) = config
376            .documents
377            .iter()
378            .find(|document| document.path == path)
379        else {
380            return;
381        };
382        self.status = format!("deleting {path}…");
383        let change = ConfigChange::Delete {
384            path: path.to_owned(),
385            expected_fingerprint: document.fingerprint.clone(),
386        };
387        self.dispatch_config_apply(config.revision.clone(), change, None);
388    }

390    fn reload_config(&mut self) {
391        let client = self.client.clone();
392        let updates = self.rpc_updates_tx.clone();
393        self.status = "reloading the configuration…".into();
394        self.spawn_rpc(async move {
395            let update = match client.reload_config().await {
396                Ok(result) => RpcUpdate::ConfigReloaded {
397                    status: result.status,
398                },
399                Err(error) => RpcUpdate::Status(error.to_string()),

401            let _ = updates.send(update);
402        });
403    }

413    pub(super) fn render_config_panel(&mut self, frame: &mut ratatui::Frame) {
414        let area = centered(frame.area(), 80, 70);
415        paint_panel(frame, area, &self.theme);
416        let rows = self.config_panel.rows();
417        let count = rows.len();
418        self.config_panel.selection.select((count > 0).then(|| {
419            self.config_panel
420                .selection
421                .selected()
422                .unwrap_or(0)
423                .min(count - 1)
424        }));
425        let chunks = Layout::default()
426            .direction(Direction::Vertical)
427            .constraints([Constraint::Percentage(55), Constraint::Percentage(45)])
428            .split(area);
429        let items = rows
430            .iter()
431            .map(|row| {
432                let style = if row.path.is_some() {
433                    self.theme.body()
434                } else {
435                    self.theme.muted()
436                };
437                let draft = row
438                    .path
439                    .as_ref()
440                    .is_some_and(|path| self.config_panel.drafts.contains_key(path));
441                ListItem::new(Line::from(vec![
442                    Span::styled(row.label.clone(), style),
443                    Span::styled(
444                        if draft { "  (not applied)" } else { "" },
445                        self.theme.warning(),
446                    ),
447                ]))
448            })
449            .collect::<Vec<_>>();
450        frame.render_stateful_widget(
451            List::new(items)
452                .highlight_symbol("> ")
453                .highlight_style(self.theme.selected())
454                .block(
455                    crate::ui::panel_block()
456                        .border_style(self.theme.panel_border())
457                        .title(crate::ui::panel_title("Configuration"))
458                        .title_bottom(crate::ui::panel_title(
459                            Line::from(Span::styled(
460                                "enter edit | n new agent | d delete | r reload | esc close",
461                                self.theme.internal(),
462                            ))
463                            .right_aligned(),
464                        )),
465                ),
466            chunks[0],
467            &mut self.config_panel.selection,
468        );
469        let detail = self.config_detail();
470        frame.render_widget(
471            Paragraph::new(detail).wrap(Wrap { trim: false }).block(
472                crate::ui::panel_block()
473                    .border_style(self.theme.panel_border())
474                    .title(crate::ui::panel_title("Status")),
475            ),
476            chunks[1],
477        );
478    }

480    fn config_detail(&self) -> String {
481        let panel = &self.config_panel;
482        if let Some(input) = &panel.new_path {
483            return format!(
484                "New agent document (agents/<id>.md or agents/<preset>/<id>.md):\n{}\n\nEnter opens it in the editor; Esc cancels.",
485                input.as_str()
486            );
487        }
488        if let Some(path) = &panel.confirm_delete {
489            return format!("Delete {path}? y deletes; any other key cancels.");
490        }
491        let Some(config) = &panel.config else {
492            return "Loading the configuration...".into();
493        };
494        let mut lines = vec![format!(
495            "Applied revision {}",
496            short_revision(&config.status.applied_revision)
497        )];
498        if config.revision != config.status.applied_revision {
499            lines.push("The files differ from the applied configuration; r reloads them.".into());
500        }
501        if let Some(error) = &config.status.error {
502            lines.push(format!("The last reload failed: {error}"));
503        }
504        let restart = restart_note(&config.status);
505        if !restart.is_empty() {
506            lines.push(restart.trim().to_owned());
507        }
508        if let Some(error) = &panel.error {
509            lines.push(String::new());
510            lines.push(error.clone());
511        }
512        lines.push(String::new());
513        lines.push(
514            "Enter edits the selected document in $VISUAL or $EDITOR; saving applies it.".into(),
515        );
516        lines.join("\n")
517    }
518}

520fn short_revision(revision: &ConfigRevision) -> &str {
521    let hex = revision.as_str().trim_start_matches("sha256:");
522    &hex[..hex.len().min(12)]
523}

525/// " Restart the daemon to apply: …" when something needs it.
526fn restart_note(status: &ConfigStatus) -> String {
527    if status.restart_required.is_empty() {
528        return String::new();
529    }
530    let parts = status
531        .restart_required
532        .iter()
533        .map(|part| match part {
534            ConfigRestartPart::Server => "server",
535            ConfigRestartPart::Plugins => "plugins",
536            ConfigRestartPart::Runtimes => "runtimes",
537        })
538        .collect::<Vec<_>>()
539        .join(", ");
540    format!(" Restart the daemon to apply: {parts}.")
541}

543/// Runs the editor on `request`'s text with the terminal handed over to it;
544/// the edited text, or why there is none.
545pub(super) fn run_editor(request: &EditorRequest) -> Result<String, String> {
546    let suffix = if request.path.ends_with(".md") {
547        ".md"
548    } else {
549        ".toml"
550    };
551    let mut file = tempfile::Builder::new()
552        .prefix("cookie-config-")
553        .suffix(suffix)
554        .tempfile()
555        .map_err(|error| format!("create a file to edit: {error}"))?;
556    std::io::Write::write_all(&mut file, request.text.as_bytes())
557        .map_err(|error| format!("write the file to edit: {error}"))?;
558    let editor = std::env::var("VISUAL")
559        .or_else(|_| std::env::var("EDITOR"))
560        .ok()
561        .filter(|editor| !editor.trim().is_empty())
562        .unwrap_or_else(|| "vi".to_owned());
563    let status = editor_command(&editor, file.path())
564        .status()
565        .map_err(|error| format!("run the editor `{editor}`: {error}"))?;
566    if !status.success() {
567        return Err(format!("the editor `{editor}` exited with {status}"));
568    }
569    std::fs::read_to_string(file.path()).map_err(|error| format!("read the edited file: {error}"))
570}

572/// The editor, which may carry arguments (`code --wait`), on `path`.
573fn editor_command(editor: &str, path: &std::path::Path) -> std::process::Command {
574    #[cfg(unix)]
575    {
576        let mut command = std::process::Command::new("sh");
577        command
578            .arg("-c")
579            .arg(format!("{editor} \"$1\""))
580            .arg("sh")
581            .arg(path);
582        command
583    }
584    #[cfg(not(unix))]
585    {
586        let mut words = editor.split_whitespace();
587        let mut command = std::process::Command::new(words.next().unwrap_or("notepad"));
588        command.args(words).arg(path);
589        command
590    }
