1//! The `/plugins` panel: every configured plugin with its state, what it
2//! declares, and why it last failed, and restarting one.
3
4use cookie_agent_client_state::plugins::{plugin_detail, state_label};
5use cookie_agent_protocol::PluginInfo;
6use crossterm::event::{KeyCode, KeyEvent};
7use ratatui::{
8    layout::{Constraint, Direction, Layout},
9    text::{Line, Span},
10    widgets::{List, ListItem, ListState, Paragraph, Wrap},
11};
12
13use super::{App, Modal, RpcUpdate, centered, paint_panel};
14
15#[derive(Debug, Default)]
16pub(super) struct PluginPanelState {
17    /// `None` until `plugins.list` answers.
18    pub(super) plugins: Option<Vec<PluginInfo>>,
19    pub(super) selection: ListState,
20    pub(super) refresh: super::ListingRefresh,
21}
22
23impl App {
24    pub(super) fn open_plugin_panel(&mut self) {
25        self.plugin_panel = PluginPanelState::default();
26        self.modal = Modal::Plugins;
27        self.poll_plugins();
28    }
29
30    /// Refetches `plugins.list` while the panel is open.
31    pub(super) fn poll_plugins(&mut self) {
32        if self.modal != Modal::Plugins || !self.plugin_panel.refresh.start() {
33            return;
34        }
35        let client = self.client.clone();
36        let updates = self.rpc_updates_tx.clone();
37        self.spawn_rpc(async move {
38            let result = client
39                .list_plugins()
40                .await
41                .map(|listing| listing.plugins)
42                .map_err(|error| error.to_string());
43            let _ = updates.send(RpcUpdate::PluginsRefreshed { result });
44        });
45    }
46
47    pub(super) fn install_plugins(&mut self, result: Result<Vec<PluginInfo>, String>) {
48        let again = self.plugin_panel.refresh.landed();
49        match result {
50            Ok(plugins) => self.plugin_panel.plugins = Some(plugins),
51            Err(error) => self.status = format!("plugin refresh failed: {error}"),
52        }
53        if again {
54            self.poll_plugins();
55        }
56    }
57
58    pub(super) fn render_plugin_panel(&mut self, frame: &mut ratatui::Frame) {
59        let area = centered(frame.area(), 80, 70);
60        paint_panel(frame, area, &self.theme);
61        let plugins = self.plugin_panel.plugins.as_deref().unwrap_or_default();
62        let count = plugins.len();
63        self.plugin_panel.selection.select((count > 0).then(|| {
64            self.plugin_panel
65                .selection
66                .selected()
67                .unwrap_or(0)
68                .min(count - 1)
69        }));
70        let chunks = Layout::default()
71            .direction(Direction::Vertical)
72            .constraints([Constraint::Percentage(45), Constraint::Percentage(55)])
73            .split(area);
74        let items = plugins
75            .iter()
76            .map(|plugin| {
77                ListItem::new(Line::from(vec![
78                    Span::styled(plugin.name.clone(), self.theme.body()),
79                    Span::styled(
80                        format!("  {}", state_label(plugin.state)),
81                        self.theme.muted(),
82                    ),
83                ]))
84            })
85            .collect::<Vec<_>>();
86        frame.render_stateful_widget(
87            List::new(items)
88                .highlight_symbol("> ")
89                .highlight_style(self.theme.selected())
90                .block(
91                    crate::ui::panel_block()
92                        .border_style(self.theme.panel_border())
93                        .title(crate::ui::panel_title("Plugins"))
94                        .title_bottom(crate::ui::panel_title(
95                            Line::from(Span::styled(
96                                "r restart | esc close",
97                                self.theme.internal(),
98                            ))
99                            .right_aligned(),
100                        )),
101                ),
102            chunks[0],
103            &mut self.plugin_panel.selection,
104        );
105        let detail = match (
106            self.plugin_panel.plugins.as_deref(),
107            self.plugin_panel.selection.selected(),
108        ) {
109            (None, _) => "Loading plugins...".to_owned(),
110            (Some([]), _) => {
111                "No plugins are configured. Add one under [plugins.<name>] in the config."
112                    .to_owned()
113            }
114            (Some(plugins), Some(index)) => {
115                plugins.get(index).map(plugin_detail).unwrap_or_default()
116            }
117            (Some(_), None) => String::new(),
118        };
119        frame.render_widget(
120            Paragraph::new(detail).wrap(Wrap { trim: false }).block(
121                crate::ui::panel_block()
122                    .border_style(self.theme.panel_border())
123                    .title(crate::ui::panel_title("Plugin details")),
124            ),
125            chunks[1],
126        );
127    }
128
129    pub(super) fn handle_plugin_key(&mut self, key: KeyEvent) {
130        let count = self.plugin_panel.plugins.as_ref().map_or(0, Vec::len);
131        match key.code {
132            KeyCode::Esc => self.modal = Modal::None,
133            KeyCode::Up => {
134                super::move_picker_selection(&mut self.plugin_panel.selection, count, true);
135            }
136            KeyCode::Down => {
137                super::move_picker_selection(&mut self.plugin_panel.selection, count, false);
138            }
139            KeyCode::Char('r') => self.restart_selected_plugin(),
140            _ => {}
141        }
142    }
143
144    fn restart_selected_plugin(&mut self) {
145        let Some(plugin) = self
146            .plugin_panel
147            .selection
148            .selected()
149            .and_then(|index| self.plugin_panel.plugins.as_ref()?.get(index))
150        else {
151            return;
152        };
153        let name = plugin.name.clone();
154        let client = self.client.clone();
155        let updates = self.rpc_updates_tx.clone();
156        self.status = format!("restarting plugin `{name}`…");
157        self.spawn_rpc(async move {
158            let update = match client.restart_plugin(name.clone()).await {
159                Ok(_) => RpcUpdate::Notice(format!("plugin `{name}` is restarting")),
160                Err(error) => RpcUpdate::failure(&error),
161            };
162            let _ = updates.send(update);
163        });
164    }
165}

167#[cfg(test)]
168mod tests {
169    use cookie_agent_protocol::PluginState;
170
171    use super::*;
172
173    fn plugin(state: PluginState) -> PluginInfo {
174        PluginInfo {
175            name: "issues".into(),
176            state,
177            reason: None,
178            tools: Vec::new(),
179            hooks: Vec::new(),
180            subscribe_events: Vec::new(),
181            subscribe_bus: false,
182            publish_bus: false,
183            producer_messaging: false,
184            dropped_events: 0,
185            stderr: None,
186        }
187    }
188
189    #[test]
190    fn detail_shows_what_a_plugin_declares_and_why_it_failed() {
191        let mut failed = plugin(PluginState::Failed);
192        failed.reason = Some("exited with status 1".into());
193        failed.tools = vec!["read_issue".into(), "close_issue".into()];
194        failed.hooks = vec![cookie_agent_protocol::HookName::new("tool_before_call").unwrap()];
195        failed.subscribe_events = vec![
196            cookie_agent_protocol::EventPattern::new("run_*").unwrap(),
197            cookie_agent_protocol::EventPattern::new("tool_call_*").unwrap(),
198        ];
199        failed.dropped_events = 3;
200        failed.stderr = Some("token missing".into());
201        let detail = plugin_detail(&failed);
202        for expected in [
203            "state: failed",
204            "reason: exited with status 1",
205            "tools: read_issue, close_issue",
206            "hooks: tool_before_call",
207            "also: events (run_*, tool_call_*)",
208            "dropped notifications: 3",
209            "stderr:\ntoken missing",
210        ] {
211            assert!(
212                detail.contains(expected),
213                "{expected:?} missing from {detail}"
214            );
215        }
216
217        let quiet = plugin_detail(&plugin(PluginState::Connected));
218        assert!(quiet.contains("tools: none"));
219        assert!(!quiet.contains("reason:"));
220        assert!(!quiet.contains("dropped"));
221    }
