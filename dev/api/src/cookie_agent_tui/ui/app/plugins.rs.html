1//! The `/plugins` panel: every configured plugin with its state, what it
2//! declares, and why it last failed, and restarting one.

4use cookie_agent_protocol::{PluginInfo, PluginState};

12use super::{App, Modal, RpcUpdate, centered, paint_panel};

14#[derive(Debug, Default)]
15pub(super) struct PluginPanelState {
16    /// `None` until `plugins.list` answers.
17    pub(super) plugins: Option<Vec<PluginInfo>>,
18    pub(super) selection: ListState,
19    pub(super) refresh_in_flight: bool,
20}
21
22impl App {
23    pub(super) fn open_plugin_panel(&mut self) {
24        self.plugin_panel = PluginPanelState::default();
25        self.modal = Modal::Plugins;
26        self.poll_plugins();
27    }
28
29    /// Refetches `plugins.list` while the panel is open.
30    pub(super) fn poll_plugins(&mut self) {
31        if self.modal != Modal::Plugins || self.plugin_panel.refresh_in_flight {
32            return;
33        }
34        self.plugin_panel.refresh_in_flight = true;
35        let client = self.client.clone();
36        let updates = self.rpc_updates_tx.clone();
37        self.spawn_rpc(async move {
38            let result = client
39                .list_plugins()
40                .await
41                .map(|listing| listing.plugins)
42                .map_err(|error| error.to_string());
43            let _ = updates.send(RpcUpdate::PluginsRefreshed { result });
44        });
45    }
46
47    pub(super) fn install_plugins(&mut self, result: Result<Vec<PluginInfo>, String>) {
48        self.plugin_panel.refresh_in_flight = false;
49        match result {
50            Ok(plugins) => self.plugin_panel.plugins = Some(plugins),
51            Err(error) => self.status = format!("plugin refresh failed: {error}"),
52        }
53    }
54
55    pub(super) fn render_plugin_panel(&mut self, frame: &mut ratatui::Frame) {
56        let area = centered(frame.area(), 80, 70);
57        paint_panel(frame, area, &self.theme);
58        let plugins = self.plugin_panel.plugins.as_deref().unwrap_or_default();
59        let count = plugins.len();
60        self.plugin_panel.selection.select((count > 0).then(|| {
61            self.plugin_panel
62                .selection
63                .selected()
64                .unwrap_or(0)
65                .min(count - 1)
66        }));
67        let chunks = Layout::default()
68            .direction(Direction::Vertical)
69            .constraints([Constraint::Percentage(45), Constraint::Percentage(55)])
70            .split(area);
71        let items = plugins
72            .iter()
73            .map(|plugin| {
74                ListItem::new(Line::from(vec![
75                    Span::styled(plugin.name.clone(), self.theme.body()),
76                    Span::styled(
77                        format!("  {}", state_label(plugin.state)),
78                        self.theme.muted(),
79                    ),
80                ]))
81            })
82            .collect::<Vec<_>>();
83        frame.render_stateful_widget(
84            List::new(items)
85                .highlight_symbol("> ")
86                .highlight_style(self.theme.selected())
87                .block(
88                    crate::ui::panel_block()
89                        .border_style(self.theme.panel_border())
90                        .title(crate::ui::panel_title("Plugins"))
91                        .title_bottom(crate::ui::panel_title(
92                            Line::from(Span::styled(
93                                "r restart | esc close",
94                                self.theme.internal(),
95                            ))
96                            .right_aligned(),
97                        )),
98                ),
99            chunks[0],
100            &mut self.plugin_panel.selection,
101        );
102        let detail = match (
103            self.plugin_panel.plugins.as_deref(),
104            self.plugin_panel.selection.selected(),
105        ) {
106            (None, _) => "Loading plugins...".to_owned(),
107            (Some([]), _) => {
108                "No plugins are configured. Add one under [plugins.<name>] in the config."
109                    .to_owned()
110            }
111            (Some(plugins), Some(index)) => {
112                plugins.get(index).map(plugin_detail).unwrap_or_default()
113            }
114            (Some(_), None) => String::new(),
115        };
116        frame.render_widget(
117            Paragraph::new(detail).wrap(Wrap { trim: false }).block(
118                crate::ui::panel_block()
119                    .border_style(self.theme.panel_border())
120                    .title(crate::ui::panel_title("Plugin details")),
121            ),
122            chunks[1],
123        );
124    }
125
126    pub(super) fn handle_plugin_key(&mut self, key: KeyEvent) {
127        let count = self.plugin_panel.plugins.as_ref().map_or(0, Vec::len);
128        match key.code {
129            KeyCode::Esc => self.modal = Modal::None,
130            KeyCode::Up => {
131                super::move_picker_selection(&mut self.plugin_panel.selection, count, true);
132            }
133            KeyCode::Down => {
134                super::move_picker_selection(&mut self.plugin_panel.selection, count, false);
135            }
136            KeyCode::Char('r') => self.restart_selected_plugin(),
137            _ => {}
138        }
139    }
140
141    fn restart_selected_plugin(&mut self) {
142        let Some(plugin) = self
143            .plugin_panel
144            .selection
145            .selected()
146            .and_then(|index| self.plugin_panel.plugins.as_ref()?.get(index))
147        else {
148            return;
149        };
150        let name = plugin.name.clone();
151        let client = self.client.clone();
152        let updates = self.rpc_updates_tx.clone();
153        self.status = format!("restarting plugin `{name}`…");
154        self.spawn_rpc(async move {
155            let update = match client.restart_plugin(name.clone()).await {
156                Ok(_) => RpcUpdate::Notice(format!("plugin `{name}` is restarting")),
157                Err(error) => RpcUpdate::Status(error.to_string()),
158            };
159            let _ = updates.send(update);
160        });
161    }
162}

164const fn state_label(state: PluginState) -> &'static str {
165    match state {
166        PluginState::Disabled => "disabled",
167        PluginState::Starting => "starting",
168        PluginState::Connected => "connected",
169        PluginState::Reconnecting => "reconnecting",
170        PluginState::Failed => "failed",
171        PluginState::Stopped => "stopped",
172    }
173}
174
175fn plugin_detail(plugin: &PluginInfo) -> String {
176    let mut text = format!("{}\nstate: {}", plugin.name, state_label(plugin.state));
177    if let Some(reason) = &plugin.reason {
178        text.push_str(&format!("\nreason: {reason}"));
179    }
180    let list = |items: &[String]| {
181        if items.is_empty() {
182            "none".to_owned()
183        } else {
184            items.join(", ")
185        }
186    };
187    text.push_str(&format!("\ntools: {}", list(&plugin.tools)));
188    let hooks = plugin
189        .hooks
190        .iter()
191        .map(|hook| {
192            serde_json::to_value(hook)
193                .ok()
194                .and_then(|value| value.as_str().map(str::to_owned))
195                .unwrap_or_default()
196        })
197        .collect::<Vec<_>>();
198    text.push_str(&format!("\nhooks: {}", list(&hooks)));
199    let mut streams = Vec::new();
200    if plugin.subscribe_events {
201        streams.push("events".to_owned());
202    }
203    if plugin.subscribe_bus {
204        streams.push("bus".to_owned());
205    }
206    if plugin.publish_bus {
207        streams.push("publishes bus".to_owned());
208    }
209    if plugin.producer_messaging {
210        streams.push("producers".to_owned());
211    }
212    text.push_str(&format!("\nalso: {}", list(&streams)));
213    if plugin.dropped_events > 0 {
214        text.push_str(&format!(
215            "\ndropped notifications: {}",
216            plugin.dropped_events
217        ));
218    }
219    if let Some(stderr) = &plugin.stderr {
220        text.push_str("\n\nstderr:\n");
221        text.push_str(stderr);
222    }
223    text
224}

226#[cfg(test)]
227mod tests {
228    use super::*;
229
230    fn plugin(state: PluginState) -> PluginInfo {
231        PluginInfo {
232            name: "issues".into(),
233            state,
234            reason: None,
235            tools: Vec::new(),
236            hooks: Vec::new(),
237            subscribe_events: false,
238            subscribe_bus: false,
239            publish_bus: false,
240            producer_messaging: false,
241            dropped_events: 0,
242            stderr: None,
243        }
244    }
245
246    #[test]
247    fn detail_shows_what_a_plugin_declares_and_why_it_failed() {
248        let mut failed = plugin(PluginState::Failed);
249        failed.reason = Some("exited with status 1".into());
250        failed.tools = vec!["read_issue".into(), "close_issue".into()];
251        failed.hooks = vec![cookie_agent_protocol::ExtensionInterceptionHook::ToolBeforeCall];
252        failed.subscribe_events = true;
253        failed.dropped_events = 3;
254        failed.stderr = Some("token missing".into());
255        let detail = plugin_detail(&failed);
256        for expected in [
257            "state: failed",
258            "reason: exited with status 1",
259            "tools: read_issue, close_issue",
260            "hooks: tool_before_call",
261            "also: events",
262            "dropped notifications: 3",
263            "stderr:\ntoken missing",
264        ] {
265            assert!(
266                detail.contains(expected),
267                "{expected:?} missing from {detail}"
268            );
269        }
270
271        let quiet = plugin_detail(&plugin(PluginState::Connected));
272        assert!(quiet.contains("tools: none"));
273        assert!(!quiet.contains("reason:"));
274        assert!(!quiet.contains("dropped"));
275    }
