36/// A job form to open in the editor; the event loop runs the editor and
37/// hands the result to [`App::finish_cron_edit`].

39pub(in crate::ui) struct CronEditRequest {

41    pub(in crate::ui) job: Option<CronJob>,
42    /// The session `target: session` fires into, and whether it is a

44    /// session on screen.

46    /// What `target: new` starts from: the job's template, else the
47    /// current session's agent, project, and directory.
48    pub(in crate::ui) template: Option<CronTemplate>,
49    pub(in crate::ui) text: String,
50    /// The form as it was opened.

130    /// The actions the panel offers now.
131    pub(in crate::ui) fn actions(&self) -> Vec<CronAction> {
132        if self.pending.is_some() {
133            return vec![CronAction::EditAgain, CronAction::Cancel];
134        }
135        if self.confirm_delete.is_some() {
136            return vec![CronAction::ConfirmDelete, CronAction::Cancel];
137        }
138        if !matches!(self.listing, Some(Ok(_))) {
139            return Vec::new();
140        }
141        match self.selected_job() {
142            Some(job) => vec![
143                CronAction::Edit,
144                CronAction::New,
145                if job.enabled {
146                    CronAction::Disable
147                } else {
148                    CronAction::Enable
149                },
150                CronAction::RunNow,
151                CronAction::Delete,
152            ],
153            None => vec![CronAction::New],
154        }
155    }

157    fn install(&mut self, listing: Result<CronsListResult, String>) {
158        self.listing = Some(listing);
159        self.refresh_in_flight = false;
160        let count = self.jobs().len();
161        if let Some(index) = self
162            .focus
163            .as_ref()
164            .and_then(|id| self.jobs().iter().position(|job| job.id == *id))
165        {
166            self.focus = None;
167            self.selection.select(Some(index));
168            return;
169        }
170        self.selection
171            .select((count > 0).then(|| self.selection.selected().unwrap_or(0).min(count - 1)));
172    }

175/// The permission modes a form names, the default first.
176const PERMISSION_MODES: [(&str, PermissionMode); 5] = [
177    ("auto_approve_n", PermissionMode::AutoApproveN),
178    ("auto_approve", PermissionMode::AutoApprove),
179    ("auto_approve_y", PermissionMode::AutoApproveY),
180    ("ask", PermissionMode::Ask),
181    ("yolo", PermissionMode::Yolo),
182];

184fn mode_name(mode: PermissionMode) -> &'static str {
185    PERMISSION_MODES
186        .iter()
187        .find(|(_, candidate)| *candidate == mode)
188        .map_or("", |(name, _)| name)

233/// Reads a form back as a draft for `request`'s job, or says what is wrong.
234pub(in crate::ui) fn parse_form(
235    text: &str,
236    request: &CronEditRequest,
237) -> Result<CronDraft, String> {
238    let (header, prompt) = match text.split_once("\n---\n") {
239        Some((header, prompt)) => (header, prompt),
240        None => match text.strip_suffix("\n---") {
241            Some(header) => (header, ""),
242            None => return Err("the form needs the --- line before the prompt".to_owned()),
243        },
244    };
245    let mut fields: Vec<(&str, &str)> = Vec::new();
246    for line in header.lines() {
247        let line = line.trim();
248        if line.is_empty() || line.starts_with('#') {
249            continue;
250        }
251        let Some((key, value)) = line.split_once(':') else {
252            return Err(format!("`{line}` is not `key: value`"));
253        };
254        let key = key.trim();
255        if !matches!(
256            key,
257            "name"
258                | "schedule"
259                | "max_runs"
260                | "target"
261                | "delivery"
262                | "agent"
263                | "model"
264                | "permission_mode"
265        ) {
266            return Err(format!("`{key}` is not a field of a job"));
267        }
268        if fields.iter().any(|(seen, _)| *seen == key) {
269            return Err(format!("`{key}` is given twice"));
270        }
271        fields.push((key, value.trim()));
272    }
273    let field = |key: &str| {
274        fields
275            .iter()
276            .find(|(seen, _)| *seen == key)
277            .map_or("", |(_, value)| value)
278    };
279    let delivery = match field("delivery") {
280        "" | "queue" => ProducerDeliveryMode::Queue,
281        "steer" => ProducerDeliveryMode::Steer,
282        other => return Err(format!("delivery is queue or steer, not `{other}`")),
283    };
284    let permission_mode = match field("permission_mode") {
285        "" => None,
286        name => Some(
287            PERMISSION_MODES
288                .iter()
289                .find(|(candidate, _)| *candidate == name)
290                .map(|(_, mode)| *mode)
291                .ok_or_else(|| format!("`{name}` is not a permission mode"))?,
292        ),
293    };
294    let agent = field("agent");
295    let target = match field("target") {
296        "session" => {
297            let (session, subagent) = request
298                .session
299                .ok_or("target: session needs an open session; use target: new")?;
300            CronDraftTarget::Session { session, subagent }
301        }
302        "new" => {
303            let template = match &request.template {
304                Some(template) => template.clone(),
305                None => bare_template(
306                    agent
307                        .parse()
308                        .map_err(|_| "a new-session job needs an agent".to_owned())?,
309                ),
310            };
311            CronDraftTarget::New { template }
312        }
313        other => return Err(format!("target is session or new, not `{other}`")),
314    };
315    Ok(CronDraft {
316        name: field("name").to_owned(),
317        prompt: prompt.to_owned(),
318        schedule: field("schedule").to_owned(),
319        max_runs: field("max_runs").to_owned(),
320        target,
321        delivery,
322        permission_mode,
323        model: field("model").to_owned(),
324        agent: agent.to_owned(),
325    })
326}

328/// A job's row: title, schedule reading, next fire, target, state, and
329/// last outcome.
330fn row_line(job: &CronJob, target: &str, theme: &Theme) -> Line<'static> {
331    let now = jiff::Timestamp::now();
332    let zone = TimeZone::system();
333    let next = job.next_fire.map_or_else(
334        || "no next fire".to_owned(),
335        |next| format!("next {}", until(next, now)),
336    );
337    let outcome = last_outcome(job);
338    let outcome_style = if outcome == "failed" {
339        theme.error()
340    } else {
341        theme.muted()
342    };
343    Line::from(vec![
344        Span::styled(job_title(job), theme.body()),
345        Span::styled(
346            format!("  {}", schedule_reading(&job.schedule, job.max_runs, &zone)),
347            theme.muted(),
348        ),
349        Span::styled(format!("  {next}"), theme.muted()),
350        Span::styled(format!("  → {target}"), theme.muted()),
351        Span::styled(
352            format!("  {}", state_label(job)),
353            if job.enabled {
354                theme.muted()
355            } else {
356                theme.warning()
357            },
358        ),
359        Span::styled(format!("  last {outcome}"), outcome_style),
360    ])
361}

363fn detail_text(panel: &CronPanelState, target: Option<&str>) -> String {
364    if let Some((_, message)) = &panel.pending {
365        return format!("Not saved: {message}\n\ne edits your form again; Esc drops it.");
366    }
367    if let Some(job) = &panel.confirm_delete {
368        return format!(
369            "Delete {}? Its run sessions stay. y deletes; Esc cancels.",
370            job_title(job)
371        );
372    }
373    match &panel.listing {
374        None => return "Loading cron jobs...".into(),
375        Some(Err(error)) => return format!("Cron jobs are unavailable: {error}"),
376        Some(Ok(_)) => {}
377    }
378    let Some(job) = panel.selected_job() else {
379        return "No cron jobs yet. n schedules a prompt: into this session when one is open, \
380                or into a new session per fire."
381            .into();
382    };
383    let zone = TimeZone::system();
384    let mut lines: Vec<String> =
385        detail_lines(job, target.unwrap_or(""), jiff::Timestamp::now(), &zone)
386            .into_iter()
387            .map(|(label, value)| format!("{label}: {value}"))
388            .collect();
389    lines.push(String::new());
390    lines.push(job.prompt.clone());
391    lines.join("\n")
392}

394fn history_text(panel: &CronPanelState) -> String {
395    let Some(job) = panel.selected_job() else {
396        return String::new();
397    };
398    if job.history.is_empty() {
399        return "No fires yet.".into();
400    }
401    let zone = TimeZone::system();
402    job.history
403        .iter()
404        .rev()
405        .map(|fire| {
406            format!(
407                "{}  {}",
408                local_minute(fire.fired_at, &zone),
409                fire_outcome(fire)
410            )
411        })
412        .collect::<Vec<_>>()
413        .join("\n")
414}

519    /// Opens a new job's form: into the session on screen, or into a new
520    /// session per fire when none is.
521    fn begin_new_cron(&mut self) {
522        let session = self.viewed_cron_session();
523        let template = self.default_cron_template();
524        let mut draft = match (session, &template) {
525            (Some((session, subagent)), _) => CronDraft::for_session(session, subagent),
526            (None, Some(template)) => CronDraft::for_new_session(template.clone()),
527            (None, None) => {
528                self.status = "No agent is available to run a new-session job.".into();
529                return;
530            }
531        };
532        if draft.agent.is_empty()
533            && let Some(template) = &template
534        {
535            draft.agent = template.agent.to_string();
536        }
537        let label = session.map(|session| self.session_target_label(session));
538        let text = form_text(&draft, label.as_deref());
539        self.status = "editing a new cron job".into();
540        self.pending_cron_editor = Some(CronEditRequest {
541            job: None,
542            session,
543            template,
544            text: text.clone(),
545            original: text,
546        });
547    }

550        let zone = TimeZone::system();
551        let mut draft = CronDraft::of_job(&job, &zone);
552        let session = job
553            .target
554            .session()
555            .map(|session| {

562        let template = match &job.target {

564            _ => self.default_cron_template(),

567            && let Some(template) = &template

569            draft.agent = template.agent.to_string();
570        }
571        let label = session.map(|session| self.session_target_label(session));
572        let text = form_text(&draft, label.as_deref());
573        self.status = format!("editing {}", job_title(&job));
574        self.pending_cron_editor = Some(CronEditRequest {
575            job: Some(job),
576            session,
577            template,
578            text: text.clone(),
579            original: text,

583    /// What the editor left for `request`: sent unless unchanged.
584    pub(in crate::ui) fn finish_cron_edit(
585        &mut self,
586        request: CronEditRequest,
587        edited: Result<String, String>,
588    ) {
589        let text = match edited {
590            Ok(text) => text,
591            Err(error) => {
592                self.status = error;
593                return;
594            }
595        };
596        let title = request.job.as_ref().map(job_title);
597        if text == request.original {
598            self.cron_panel.pending = None;
599            self.status = match &title {
600                Some(title) => format!("No changes to {title}."),
601                None => "No changes; the job was not created.".into(),
602            };
603            return;
604        }
605        let request = CronEditRequest { text, ..request };
606        let draft = match parse_form(&request.text, &request) {
607            Ok(draft) => draft,
608            Err(message) => return self.refuse_cron_form(request, message),
609        };
610        let zone = TimeZone::system();
611        match &request.job {
612            None => match draft.create_params() {
613                Ok(params) => self.dispatch_cron_save(request, CronSave::Create(params)),
614                Err(message) => self.refuse_cron_form(request, message),
615            },
616            Some(job) => match draft.update_params(job, &zone) {
617                Ok(params) if !update_changes_anything(&params) => {
618                    self.cron_panel.pending = None;
619                    self.status = format!("No changes to {}.", job_title(job));
620                }
621                Ok(params) => self.dispatch_cron_save(request, CronSave::Update(params)),
622                Err(message) => self.refuse_cron_form(request, message),
623            },
624        }
625    }
626
627    fn refuse_cron_form(&mut self, request: CronEditRequest, message: String) {
628        self.status = format!("Not saved: {message}");
629        self.cron_panel.pending = Some((request, message));
630    }

632    fn dispatch_cron_save(&mut self, request: CronEditRequest, save: CronSave) {
633        self.status = "saving the cron job…".into();
634        let client = self.client.clone();
635        let updates = self.rpc_updates_tx.clone();
636        self.spawn_rpc(async move {
637            let result = match save {
638                CronSave::Create(params) => client.create_cron(params).await,
639                CronSave::Update(params) => client.update_cron(params).await,
640            }
641            .map(|result| result.job)
642            .map_err(|error| message_of(&error));
643            let _ = updates.send(RpcUpdate::CronSaved {
644                request: Box::new(request),
645                result: Box::new(result),
646            });
647        });
648    }

650    pub(in crate::ui) fn install_cron_save(

652        request: CronEditRequest,
653        result: Result<CronJob, String>,
654    ) {
655        match result {
656            Ok(job) => {
657                self.cron_panel.pending = None;
658                let zone = TimeZone::system();
659                let next = job.next_fire.map_or_else(String::new, |next| {
660                    format!("; next fire {}", local_minute(next, &zone))
661                });
662                let verb = if request.job.is_some() {
663                    "Saved"
664                } else {
665                    "Created"
666                };
667                self.status = format!("{verb} {}{next}.", job_title(&job));
668                self.cron_panel.focus = Some(job.id);
669            }
670            Err(message) => self.refuse_cron_form(request, message),
671        }
672        self.poll_crons();
673    }

675    /// Runs a change of one job whose answer is a status line either way.
676    fn dispatch_cron_change(&mut self, pending: String, change: CronChange) {
677        self.status = pending;
678        let client = self.client.clone();
679        let updates = self.rpc_updates_tx.clone();
680        self.spawn_rpc(async move {
681            let result = match change {
682                CronChange::Enabled { title, id, enabled } => {
683                    let mut params = CronUpdateParams::new(id);
684                    params.enabled = Some(enabled);
685                    let verb = if enabled { "Enabled" } else { "Disabled" };
686                    client
687                        .update_cron(params)
688                        .await
689                        .map(|_| format!("{verb} {title}."))
690                        .map_err(|error| format!("{title} was not changed: {}", message_of(&error)))
691                }
692                CronChange::RunNow { title, id } => client
693                    .run_cron(id)
694                    .await
695                    .map(|run| {
696                        let outcome = run
697                            .job
698                            .history
699                            .iter()
700                            .rev()
701                            .find(|fire| fire.run_now)
702                            .map_or_else(|| "delivered".to_owned(), fire_outcome);
703                        format!("Ran {title} now: {outcome}.")
704                    })
705                    .map_err(|error| format!("{title} did not run: {}", message_of(&error))),
706                CronChange::Delete { title, id } => client
707                    .delete_cron(id)
708                    .await
709                    .map(|_| format!("Deleted {title}."))
710                    .map_err(|error| format!("{title} was not deleted: {}", message_of(&error))),
711            };
712            let _ = updates.send(RpcUpdate::CronChanged { result });
713        });
714    }

717        self.status = match result {
718            Ok(done) | Err(done) => done,
719        };
720        self.poll_crons();

723    /// Acts on `action`, from its key or its button.
724    pub(in crate::ui) fn activate_cron_action(&mut self, action: CronAction) {
725        if !self.cron_panel.actions().contains(&action) {
726            return;
727        }
728        let selected = self.cron_panel.selected_job().cloned();
729        match action {
730            CronAction::Edit => {
731                if let Some(job) = selected {
732                    self.begin_cron_edit(job);
733                }
734            }
735            CronAction::New => self.begin_new_cron(),
736            CronAction::Enable | CronAction::Disable => {
737                if let Some(job) = selected {
738                    let enabled = action == CronAction::Enable;
739                    let title = job_title(&job);
740                    let verb = if enabled { "enabling" } else { "disabling" };
741                    self.dispatch_cron_change(
742                        format!("{verb} {title}…"),
743                        CronChange::Enabled {
744                            title,
745                            id: job.id,
746                            enabled,
747                        },
748                    );
749                }
750            }
751            CronAction::RunNow => {
752                if let Some(job) = selected {
753                    let title = job_title(&job);
754                    self.dispatch_cron_change(
755                        format!("running {title}…"),
756                        CronChange::RunNow { title, id: job.id },
757                    );
758                }
759            }
760            CronAction::Delete => self.cron_panel.confirm_delete = selected,
761            CronAction::ConfirmDelete => {
762                if let Some(job) = self.cron_panel.confirm_delete.take() {
763                    let title = job_title(&job);
764                    self.dispatch_cron_change(
765                        format!("deleting {title}…"),
766                        CronChange::Delete { title, id: job.id },
767                    );
768                }
769            }
770            CronAction::Cancel => {
771                if self.cron_panel.confirm_delete.take().is_some() {
772                    self.status = "Delete cancelled.".into();
773                } else if self.cron_panel.pending.take().is_some() {
774                    self.status = "Your form was dropped.".into();
775                }
776            }
777            CronAction::EditAgain => {
778                if let Some((request, _)) = self.cron_panel.pending.take() {
779                    self.pending_cron_editor = Some(request);
780                }
781            }
782        }
783    }

785    pub(super) fn handle_cron_key(&mut self, key: KeyEvent) {
786        let actions = self.cron_panel.actions();
787        let action = match key.code {
788            KeyCode::Esc if actions.contains(&CronAction::Cancel) => Some(CronAction::Cancel),
789            KeyCode::Esc => {
790                self.modal = Modal::None;
791                return;
792            }
793            KeyCode::Enter => Some(CronAction::Edit),
794            KeyCode::Char(character) => actions
795                .iter()
796                .copied()
797                .find(|action| action.key_label() == character.to_string()),
798            KeyCode::Up | KeyCode::Down => {
799                if self.cron_panel.confirm_delete.is_none() && self.cron_panel.pending.is_none() {
800                    let count = self.cron_panel.jobs().len();
801                    super::move_picker_selection(
802                        &mut self.cron_panel.selection,
803                        count,
804                        key.code == KeyCode::Up,
805                    );
806                }
807                return;
808            }
809            _ => None,
810        };
811        if let Some(action) = action {
812            self.activate_cron_action(action);
813        }
814    }

852/// The panel: the jobs, the selected job's details and history, and the
853/// actions.
854fn render_cron(
855    frame: &mut Frame,
856    area: Rect,
857    panel: &mut CronPanelState,
858    targets: &[String],
859    theme: &Theme,
860) -> CronHits {
861    let mut hits = CronHits::default();
862    let count = panel.jobs().len();
863    panel
864        .selection
865        .select((count > 0).then(|| panel.selection.selected().unwrap_or(0).min(count - 1)));
866    let [list_area, bottom_area, actions_area] = Layout::default()
867        .direction(Direction::Vertical)
868        .constraints([
869            Constraint::Min(4),
870            Constraint::Length(14),
871            Constraint::Length(1),
872        ])
873        .areas(area);
874    let [detail_area, history_area] = Layout::default()
875        .direction(Direction::Horizontal)
876        .constraints([Constraint::Percentage(55), Constraint::Percentage(45)])
877        .areas(bottom_area);
878    let items = panel
879        .jobs()
880        .iter()
881        .zip(targets)
882        .map(|(job, target)| ListItem::new(row_line(job, target, theme)))
883        .collect::<Vec<_>>();
884    let firing = !matches!(&panel.listing, Some(Ok(listing)) if !listing.firing);
885    let title = if firing {
886        "Cron jobs"
887    } else {
888        "Cron jobs · firing is off ([cron] enabled = false)"
889    };
890    let block = crate::ui::panel_block()
891        .border_style(theme.panel_border())
892        .title(crate::ui::panel_title(title))
893        .title_bottom(crate::ui::panel_title(
894            Line::from(Span::styled(
895                "e edit | n new | t enable/disable | r run now | d delete | esc close",
896                theme.internal(),
897            ))
898            .right_aligned(),
899        ));
900    let inner = block.inner(list_area);
901    frame.render_stateful_widget(
902        List::new(items)
903            .highlight_symbol("> ")
904            .highlight_style(theme.selected())
905            .block(block),
906        list_area,
907        &mut panel.selection,
908    );
909    let quiet = panel.confirm_delete.is_some() || panel.pending.is_some();
910    if !quiet {
911        let offset = panel.selection.offset();
912        for (line, index) in (offset..count).take(usize::from(inner.height)).enumerate() {
913            let y = inner.y + u16::try_from(line).unwrap_or(u16::MAX);
914            hits.rows
915                .push((Rect::new(inner.x, y, inner.width, 1), index));
916        }
917    }
918    let target = panel
919        .selection
920        .selected()
921        .and_then(|index| targets.get(index))
922        .map(String::as_str);
923    frame.render_widget(
924        Paragraph::new(detail_text(panel, target))
925            .wrap(Wrap { trim: false })
926            .block(
927                crate::ui::panel_block()
928                    .border_style(theme.panel_border())
929                    .title(crate::ui::panel_title("Details")),
930            ),
931        detail_area,
932    );
933    frame.render_widget(
934        Paragraph::new(history_text(panel))
935            .wrap(Wrap { trim: false })
936            .block(
937                crate::ui::panel_block()
938                    .border_style(theme.panel_border())
939                    .title(crate::ui::panel_title("History, newest first")),
940            ),
941        history_area,
942    );
943    let mut spans = Vec::new();
944    let mut x = actions_area.x;
945    for action in panel.actions() {
946        let label = format!(" {} ({}) ", action.label(), action.key_label());
947        let width = u16::try_from(label.chars().count()).unwrap_or(u16::MAX);
948        if x.saturating_add(width) <= actions_area.right() {
949            hits.actions
950                .push((Rect::new(x, actions_area.y, width, 1), action));
951        }
952        spans.push(Span::styled(label, theme.link()));
953        spans.push(Span::raw(" "));
954        x = x.saturating_add(width + 1);
955    }
956    frame.render_widget(Paragraph::new(Line::from(spans)), actions_area);
957    hits
958}
