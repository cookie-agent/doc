11/// A button on the session picker's action row.
12#[derive(Clone, Copy, Debug, Eq, PartialEq)]
13pub(in crate::ui) enum SessionPickerAction {
14    /// Opens a title field for the highlighted session (Ctrl+R).
15    Rename,
16    /// Asks to delete the highlighted session (Ctrl+D).
17    Delete,
18    /// Switches between this directory's sessions and every directory's
19    /// (Ctrl+A).
20    ToggleScope,
21    /// Cycles the project filter: all, each project, no project (Ctrl+P).
22    CycleProject,
23}
24
25impl SessionPickerAction {
26    fn label(self, all_directories: bool, project: &str) -> String {
27        match self {
28            Self::Rename => " Rename (ctrl+r) ".to_owned(),
29            Self::Delete => " Delete (ctrl+d) ".to_owned(),
30            Self::ToggleScope if all_directories => " This directory (ctrl+a) ".to_owned(),
31            Self::ToggleScope => " All directories (ctrl+a) ".to_owned(),
32            Self::CycleProject => format!(" {project} (ctrl+p) "),
33        }
34    }
35}

37/// A session delete waiting for the user to confirm it.
38#[derive(Clone, Debug)]
39pub(in crate::ui) struct SessionDeletePrompt {
40    /// The root session as the picker showed it; its last event seq guards
41    /// the delete.
42    pub(in crate::ui) session: SessionMeta,
43    /// How many subagent sessions go with it, once `session.tree` answered.
44    pub(in crate::ui) subagents: Option<usize>,
45}

47impl App {
48    /// The session picker's title for the directories and the project it
49    /// lists.
50    pub(super) fn session_scope_title(&self) -> String {
51        let scope = if self.session_scope_all {
52            "Sessions — all directories"
53        } else {
54            "Sessions — this directory"
55        };
56        if self.session_project_filter.is_some() {
57            format!("{scope} · {}", self.session_project_filter_label())
58        } else {
59            scope.to_owned()
60        }
61    }

63    /// Ctrl+A in the picker: lists every directory's sessions, or this
64    /// directory's again.
65    pub(in crate::ui) async fn toggle_session_scope(&mut self) {
66        self.session_scope_all = !self.session_scope_all;
67        self.note_sessions_changed();
68        self.refresh_session_list().await;
69        self.session_search_changed();
70    }

72    /// Ctrl+D, Delete, or the Delete button: asks to delete the highlighted
73    /// session. Nothing is sent before the user confirms, and nothing at all
74    /// to a daemon older than client protocol 1.5.
75    pub(in crate::ui) fn begin_session_delete(&mut self) {
76        let index = self.picker_state.selected().unwrap_or(0);
77        let Some(session_id) = self.session_search_ids().get(index).copied() else {
78            self.status = "There is no session to delete.".into();
79            return;
80        };
81        if !session_delete::daemon_can_delete_sessions(self.client.daemon_version()) {
82            self.status = DAEMON_TOO_OLD_TO_DELETE.into();
83            return;
84        }
85        let Some(session) = self
86            .sessions
87            .iter()
88            .find(|session| session.session_id == session_id)
89            .cloned()
90        else {
91            return;
92        };
93        self.session_delete = Some(SessionDeletePrompt {
94            session,
95            subagents: None,
96        });
97        let client = self.client.clone();
98        let updates = self.rpc_updates_tx.clone();
99        self.spawn_rpc(async move {
100            if let Ok(result) = client
101                .session_tree(cookie_agent_protocol::SessionTreeParams { session_id })
102                .await
103            {
104                let _ = updates.send(RpcUpdate::SessionDeleteSubagents {
105                    session_id,
106                    subagents: subagent_count(&result.tree),
107                });
108            }
109        });
110    }

112    /// Records how many subagents the waiting delete takes with it.
113    pub(super) fn note_session_delete_subagents(&mut self, session_id: SessionId, count: usize) {
114        if let Some(prompt) = self.session_delete.as_mut()
115            && prompt.session.session_id == session_id
116        {
117            prompt.subagents = Some(count);
118        }
119    }
120
121    /// Answers the waiting delete from its keys or its buttons.
122    pub(in crate::ui) fn answer_session_delete(&mut self, button: ConfirmButton) {
123        let Some(prompt) = self.session_delete.take() else {
124            return;
125        };
126        let title = session_title(&prompt.session);
127        if button == ConfirmButton::Cancel {
128            self.status = format!("Kept {title}.");
129            return;
130        }
131        self.status = format!("Deleting {title}…");
132        let client = self.client.clone();
133        let updates = self.rpc_updates_tx.clone();
134        self.spawn_rpc(async move {
135            let outcome = session_delete::delete_session(&client, &prompt.session).await;
136            let _ = updates.send(RpcUpdate::SessionDeleted { title, outcome });
137        });
138    }
139
140    /// Shows how a delete ended. The sessions themselves leave every view
141    /// through the `sessions.changed` removals the daemon sends.
142    pub(super) fn finish_session_delete(&mut self, title: String, outcome: SessionDeleteOutcome) {
143        match outcome {
144            SessionDeleteOutcome::Deleted(deleted) => {
145                for session_id in deleted {
146                    self.forget_removed_session(session_id);
147                }
148                self.status = format!("Deleted {title}.");
149            }
150            SessionDeleteOutcome::DaemonTooOld => self.status = DAEMON_TOO_OLD_TO_DELETE.into(),
151            SessionDeleteOutcome::Failed(error) => {
152                self.handle_rpc_update(RpcUpdate::failure(&error));
153            }
154        }
155    }

157    /// Forgets a session the daemon removed. When it is the viewed one, or
158    /// the root of the viewed tree, the TUI leaves it for a new session's
159    /// draft, as it starts without one.
160    pub(super) fn forget_removed_session(&mut self, session_id: SessionId) {
161        let before = self.sessions.len();
162        self.sessions
163            .retain(|session| session.session_id != session_id);
164        if self.sessions.len() != before {
165            self.note_sessions_changed();
166        }
167        self.subscriptions.release(session_id);
168        self.store.forget_session(session_id);
169        if self
170            .session_delete
171            .as_ref()
172            .is_some_and(|prompt| prompt.session.session_id == session_id)
173        {
174            self.session_delete = None;
175        }
176        if self.tree_root == Some(session_id) {
177            self.leave_deleted_session();
178        } else if self.selected == Some(session_id) {
179            match self.tree_root {
180                Some(root) => self.watch_session(root),
181                None => self.leave_deleted_session(),
182            }
183        }
184    }

186    fn leave_deleted_session(&mut self) {
187        self.selected = None;
188        self.tree_root = None;
189        self.tree = None;
190        self.tree_cursor = None;
191        self.tree_refresh_in_flight = None;
192        self.tree_refresh_pending = false;
193        self.selection_generation = self.selection_generation.wrapping_add(1);
194        self.conversation_scroll = ConversationScroll::default();
195        self.scrollbar_geometry = None;
196        self.scrollbar_drag = None;
197        if matches!(self.selection, Some(TextSelection::Conversation { .. })) {
198            self.selection = None;
199        }
200        self.new_session_draft = self.default_draft_selection();
201        self.status = "The session was deleted.".into();
202    }

204    /// The picker's action row along its bottom edge, or the waiting
205    /// delete's confirmation over the picker.
206    pub(super) fn render_session_picker_actions(
207        &mut self,
208        frame: &mut ratatui::Frame,
209        row: Rect,
210        screen: Rect,
211    ) {
212        self.hit_map.session_picker_actions.clear();
213        if let Some(prompt) = self.session_delete.clone() {
214            self.render_session_delete_confirm(frame, &prompt, screen);
215            return;
216        }
217        let mut spans = Vec::new();
218        let mut x = row.x;
219        let project = self.session_project_filter_label();
220        let actions = [
221            SessionPickerAction::Rename,
222            SessionPickerAction::Delete,
223            SessionPickerAction::ToggleScope,
224            SessionPickerAction::CycleProject,
225        ];
226        let shown = if self.daemon_has_projects() {
227            &actions[..]
228        } else {
229            &actions[..3]
230        };
231        for &action in shown {
232            let label = action.label(self.session_scope_all, &project);
233            let width = u16::try_from(UnicodeWidthStr::width(label.as_str())).unwrap_or(u16::MAX);
234            if x.saturating_add(width) <= row.right() {
235                self.hit_map
236                    .session_picker_actions
237                    .push((Rect::new(x, row.y, width, 1), action));
238            }
239            spans.push(Span::styled(label, self.theme.link()));
240            spans.push(Span::raw(" "));
241            x = x.saturating_add(width + 1);
242        }
243        spans.push(Span::styled(
244            "enter open · esc close",
245            self.theme.internal(),
246        ));
247        frame.render_widget(Paragraph::new(Line::from(spans)), row);
248    }

250    fn render_session_delete_confirm(
251        &mut self,
252        frame: &mut ratatui::Frame,
253        prompt: &SessionDeletePrompt,
254        screen: Rect,
255    ) {
256        let theme = &self.theme;
257        let title = session_title(&prompt.session);
258        let subagents = match prompt.subagents {
259            Some(0) => "It has no subagents.".to_owned(),
260            Some(1) => "Its 1 subagent session goes with it.".to_owned(),
261            Some(count) => format!("Its {count} subagent sessions go with it."),
262            None => "Every subagent session of it goes with it.".to_owned(),
263        };
264        let body = |width: u16| {
265            let mut lines = vec![
266                Line::styled(format!("Delete {title}?"), theme.heading()),
267                Line::default(),
268            ];
269            lines.extend(wrapped_line(
270                Line::styled(
271                    format!(
272                        "{subagents} Its messages, tool output, and attachments are removed \
273                         from disk, and this cannot be undone. To keep a copy, fork it first."
274                    ),
275                    theme.muted_text(),
276                ),
277                width,
278            ));
279            lines
280        };
281        self.hit_map.confirm_buttons =
282            render_confirm_panel(frame, screen, "Delete session", body, theme);
283    }

285    /// Keys while a delete waits for its answer: y or Enter deletes, n or
286    /// Esc keeps the session. Returns whether a prompt took the key.
287    pub(super) fn handle_session_delete_key(&mut self, key: KeyEvent) -> bool {
288        if self.session_delete.is_none() {
289            return false;
290        }
291        match key.code {
292            KeyCode::Esc | KeyCode::Char('n' | 'N') => {
293                self.answer_session_delete(ConfirmButton::Cancel);
294            }
295            KeyCode::Enter | KeyCode::Char('y' | 'Y') => {
296                self.answer_session_delete(ConfirmButton::Confirm);
297            }
298            _ => {}
299        }
300        true
301    }

303    /// Acts on a picker action from its button.
304    pub(in crate::ui) async fn activate_session_picker_action(
305        &mut self,
306        action: SessionPickerAction,
307    ) {
308        match action {
309            SessionPickerAction::Rename => self.begin_session_rename(),
310            SessionPickerAction::Delete => self.begin_session_delete(),
311            SessionPickerAction::ToggleScope => self.toggle_session_scope().await,
312            SessionPickerAction::CycleProject => self.cycle_session_project_filter(),
313        }
314    }

317/// A session's title as the picker names it.
318fn session_title(session: &SessionMeta) -> String {
319    session.title.as_ref().map_or_else(
320        || {
321            format!(
322                "the untitled session {}",
323                crate::ui::pickers::short_id(session)
324            )
325        },
326        |title| format!("“{title}”"),
327    )
328}

330/// How many sessions `tree` holds below its root.
331fn subagent_count(tree: &cookie_agent_protocol::SessionTree) -> usize {
332    tree.children
333        .iter()
334        .map(|child| 1 + subagent_count(child))
335        .sum()
