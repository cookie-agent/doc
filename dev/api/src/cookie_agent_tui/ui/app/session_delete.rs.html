11/// A button on the session picker's action row.
12#[derive(Clone, Copy, Debug, Eq, PartialEq)]
13pub(in crate::ui) enum SessionPickerAction {
14    /// Opens a title field for the highlighted session (Ctrl+R).
15    Rename,
16    /// Asks to delete the highlighted session (Ctrl+D).
17    Delete,
18    /// Marks the highlighted session read, or unread when it is read
19    /// (Ctrl+K; `docs/site/specs/unread-sessions.md`).
20    ToggleRead,
21    /// Switches between this directory's sessions and every directory's
22    /// (Ctrl+A).
23    ToggleScope,
24    /// Cycles the project filter: all, each project, no project (Ctrl+P).
25    CycleProject,
26}
27
28impl SessionPickerAction {
29    fn label(self, all_directories: bool, project: &str, unread: bool) -> String {
30        match self {
31            Self::Rename => " Rename (ctrl+r) ".to_owned(),
32            Self::Delete => " Delete (ctrl+d) ".to_owned(),
33            Self::ToggleRead if unread => " Mark read (ctrl+k) ".to_owned(),
34            Self::ToggleRead => " Mark unread (ctrl+k) ".to_owned(),
35            Self::ToggleScope if all_directories => " This directory (ctrl+a) ".to_owned(),
36            Self::ToggleScope => " All directories (ctrl+a) ".to_owned(),
37            Self::CycleProject => format!(" {project} (ctrl+p) "),
38        }
39    }
40}

42/// A session delete waiting for the user to confirm it.
43#[derive(Clone, Debug)]
44pub(in crate::ui) struct SessionDeletePrompt {
45    /// The root session as the picker showed it; its last event seq guards
46    /// the delete.
47    pub(in crate::ui) session: SessionMeta,
48    /// How many subagent sessions go with it, once `session.tree` answered.
49    pub(in crate::ui) subagents: Option<usize>,

53    /// The session picker's title for the directories and the project it
54    /// lists.
55    pub(super) fn session_scope_title(&self) -> String {
56        let scope = if self.session_scope_all {
57            "Sessions — all directories"
58        } else {
59            "Sessions — this directory"
60        };
61        if self.session_project_filter.is_some() {
62            format!("{scope} · {}", self.session_project_filter_label())
63        } else {
64            scope.to_owned()
65        }
66    }

68    /// Ctrl+A in the picker: lists every directory's sessions, or this
69    /// directory's again.
70    pub(in crate::ui) async fn toggle_session_scope(&mut self) {
71        self.session_scope_all = !self.session_scope_all;
72        self.note_sessions_changed();
73        self.refresh_session_list().await;
74        self.session_search_changed();
75    }

77    /// Ctrl+D, Delete, or the Delete button: asks to delete the highlighted
78    /// session. Nothing is sent before the user confirms, and nothing at all
79    /// to a daemon older than client protocol 1.5.
80    /// The session the picker highlights.
81    fn highlighted_session(&mut self) -> Option<SessionMeta> {
82        let index = self.picker_state.selected().unwrap_or(0);
83        let session_id = self.session_search_ids().get(index).copied()?;
84        self.sessions
85            .iter()
86            .find(|session| session.session_id == session_id)
87            .cloned()
88    }
89
90    /// Ctrl+K or the button: marks the highlighted session read, or unread
91    /// when it is read. The daemon announces the change to every client.
92    pub(in crate::ui) fn toggle_session_read(&mut self) {
93        if !cookie_agent_client_state::unread::daemon_keeps_unread(self.client.daemon_version()) {
94            self.status = "This daemon keeps no unread sessions; update it.".into();
95            return;
96        }
97        let Some(session) = self.highlighted_session() else {
98            self.status = "There is no session to mark.".into();
99            return;
100        };
101        let (session_id, unread) = (session.session_id, !session.unread);
102        let client = self.client.clone();
103        let updates = self.rpc_updates_tx.clone();
104        self.spawn_rpc(async move {
105            let update = match client.set_session_unread(vec![session_id], unread).await {
106                Ok(result) => RpcUpdate::Notice(cookie_agent_client_state::unread::marked_message(
107                    result.changed.len(),
108                    unread,
109                )),
110                Err(error) => RpcUpdate::failure(&error),
111            };
112            let _ = updates.send(update);
113        });
114    }
115
116    pub(in crate::ui) fn begin_session_delete(&mut self) {
117        let index = self.picker_state.selected().unwrap_or(0);
118        let Some(session_id) = self.session_search_ids().get(index).copied() else {
119            self.status = "There is no session to delete.".into();
120            return;
121        };
122        if !session_delete::daemon_can_delete_sessions(self.client.daemon_version()) {
123            self.status = DAEMON_TOO_OLD_TO_DELETE.into();
124            return;
125        }
126        let Some(session) = self
127            .sessions
128            .iter()
129            .find(|session| session.session_id == session_id)
130            .cloned()
131        else {
132            return;
133        };
134        self.session_delete = Some(SessionDeletePrompt {
135            session,
136            subagents: None,
137        });
138        let client = self.client.clone();
139        let updates = self.rpc_updates_tx.clone();
140        self.spawn_rpc(async move {
141            if let Ok(result) = client
142                .session_tree(cookie_agent_protocol::SessionTreeParams { session_id })
143                .await
144            {
145                let _ = updates.send(RpcUpdate::SessionDeleteSubagents {
146                    session_id,
147                    subagents: subagent_count(&result.tree),
148                });
149            }
150        });
151    }

153    /// Records how many subagents the waiting delete takes with it.
154    pub(super) fn note_session_delete_subagents(&mut self, session_id: SessionId, count: usize) {
155        if let Some(prompt) = self.session_delete.as_mut()
156            && prompt.session.session_id == session_id
157        {
158            prompt.subagents = Some(count);
159        }
160    }

162    /// Answers the waiting delete from its keys or its buttons.
163    pub(in crate::ui) fn answer_session_delete(&mut self, button: ConfirmButton) {
164        let Some(prompt) = self.session_delete.take() else {
165            return;
166        };
167        let title = session_title(&prompt.session);
168        if button == ConfirmButton::Cancel {
169            self.status = format!("Kept {title}.");
170            return;
171        }
172        self.status = format!("Deleting {title}…");
173        let client = self.client.clone();
174        let updates = self.rpc_updates_tx.clone();
175        self.spawn_rpc(async move {
176            let outcome = session_delete::delete_session(&client, &prompt.session).await;
177            let _ = updates.send(RpcUpdate::SessionDeleted { title, outcome });
178        });
179    }
180
181    /// Shows how a delete ended. The sessions themselves leave every view
182    /// through the `sessions.changed` removals the daemon sends.
183    pub(super) fn finish_session_delete(&mut self, title: String, outcome: SessionDeleteOutcome) {
184        match outcome {
185            SessionDeleteOutcome::Deleted(deleted) => {
186                for session_id in deleted {
187                    self.forget_removed_session(session_id);
188                }
189                self.status = format!("Deleted {title}.");
190            }
191            SessionDeleteOutcome::DaemonTooOld => self.status = DAEMON_TOO_OLD_TO_DELETE.into(),
192            SessionDeleteOutcome::Failed(error) => {
193                self.handle_rpc_update(RpcUpdate::failure(&error));
194            }
195        }
196    }

198    /// Forgets a session the daemon removed. When it is the viewed one, or
199    /// the root of the viewed tree, the TUI leaves it for a new session's
200    /// draft, as it starts without one.
201    pub(super) fn forget_removed_session(&mut self, session_id: SessionId) {
202        let before = self.sessions.len();
203        self.sessions
204            .retain(|session| session.session_id != session_id);
205        if self.sessions.len() != before {
206            self.note_sessions_changed();
207        }
208        self.subscriptions.release(session_id);
209        self.store.forget_session(session_id);
210        if self
211            .session_delete
212            .as_ref()
213            .is_some_and(|prompt| prompt.session.session_id == session_id)
214        {
215            self.session_delete = None;
216        }
217        if self.tree_root == Some(session_id) {
218            self.leave_deleted_session();
219        } else if self.selected == Some(session_id) {
220            match self.tree_root {
221                Some(root) => self.watch_session(root),
222                None => self.leave_deleted_session(),
223            }
224        }
225        self.follow_composer_slot();
226        self.forget_composer_draft(session_id);
227    }

229    fn leave_deleted_session(&mut self) {
230        self.selected = None;
231        self.tree_root = None;
232        self.tree = None;
233        self.tree_cursor = None;
234        self.tree_refresh_in_flight = None;
235        self.tree_refresh_pending = false;
236        self.selection_generation = self.selection_generation.wrapping_add(1);
237        self.conversation_scroll = ConversationScroll::default();
238        self.scrollbar_geometry = None;
239        self.scrollbar_drag = None;
240        if matches!(self.selection, Some(TextSelection::Conversation { .. })) {
241            self.selection = None;
242        }
243        self.new_session_draft = self.default_draft_selection();
244        self.status = "The session was deleted.".into();
245    }

247    /// The picker's action row along its bottom edge, or the waiting
248    /// delete's confirmation over the picker.
249    pub(super) fn render_session_picker_actions(
250        &mut self,
251        frame: &mut ratatui::Frame,
252        row: Rect,
253        screen: Rect,
254    ) {
255        self.hit_map.session_picker_actions.clear();
256        if let Some(prompt) = self.session_delete.clone() {
257            self.render_session_delete_confirm(frame, &prompt, screen);
258            return;
259        }
260        let mut spans = Vec::new();
261        let mut x = row.x;
262        let project = self.session_project_filter_label();
263        let unread = self
264            .highlighted_session()
265            .is_some_and(|session| session.unread);
266        let keeps_unread =
267            cookie_agent_client_state::unread::daemon_keeps_unread(self.client.daemon_version());
268        let has_projects = self.daemon_has_projects();
269        let shown = [
270            SessionPickerAction::Rename,
271            SessionPickerAction::Delete,
272            SessionPickerAction::ToggleRead,
273            SessionPickerAction::ToggleScope,
274            SessionPickerAction::CycleProject,
275        ]
276        .into_iter()
277        .filter(|action| match action {
278            SessionPickerAction::ToggleRead => keeps_unread,
279            SessionPickerAction::CycleProject => has_projects,
280            _ => true,
281        });
282        for action in shown {
283            let label = action.label(self.session_scope_all, &project, unread);
284            let width = u16::try_from(UnicodeWidthStr::width(label.as_str())).unwrap_or(u16::MAX);
285            if x.saturating_add(width) <= row.right() {
286                self.hit_map
287                    .session_picker_actions
288                    .push((Rect::new(x, row.y, width, 1), action));
289            }
290            spans.push(Span::styled(label, self.theme.link()));
291            spans.push(Span::raw(" "));
292            x = x.saturating_add(width + 1);
293        }
294        spans.push(Span::styled(
295            "enter open · esc close",
296            self.theme.internal(),
297        ));
298        frame.render_widget(Paragraph::new(Line::from(spans)), row);
299    }

301    fn render_session_delete_confirm(
302        &mut self,
303        frame: &mut ratatui::Frame,
304        prompt: &SessionDeletePrompt,
305        screen: Rect,
306    ) {
307        let theme = &self.theme;
308        let title = session_title(&prompt.session);
309        let subagents = match prompt.subagents {
310            Some(0) => "It has no subagents.".to_owned(),
311            Some(1) => "Its 1 subagent session goes with it.".to_owned(),
312            Some(count) => format!("Its {count} subagent sessions go with it."),
313            None => "Every subagent session of it goes with it.".to_owned(),
314        };
315        let body = |width: u16| {
316            let mut lines = vec![
317                Line::styled(format!("Delete {title}?"), theme.heading()),
318                Line::default(),
319            ];
320            lines.extend(wrapped_line(
321                Line::styled(
322                    format!(
323                        "{subagents} Its messages, tool output, and attachments are removed \
324                         from disk, and this cannot be undone. To keep a copy, fork it first."
325                    ),
326                    theme.muted_text(),
327                ),
328                width,
329            ));
330            lines
331        };
332        self.hit_map.confirm_buttons =
333            render_confirm_panel(frame, screen, "Delete session", body, theme);
334    }

336    /// Keys while a delete waits for its answer: y or Enter deletes, n or
337    /// Esc keeps the session. Returns whether a prompt took the key.
338    pub(super) fn handle_session_delete_key(&mut self, key: KeyEvent) -> bool {
339        if self.session_delete.is_none() {
340            return false;
341        }
342        match key.code {
343            KeyCode::Esc | KeyCode::Char('n' | 'N') => {
344                self.answer_session_delete(ConfirmButton::Cancel);
345            }
346            KeyCode::Enter | KeyCode::Char('y' | 'Y') => {
347                self.answer_session_delete(ConfirmButton::Confirm);
348            }
349            _ => {}
350        }
351        true
352    }

354    /// Acts on a picker action from its button.
355    pub(in crate::ui) async fn activate_session_picker_action(
356        &mut self,
357        action: SessionPickerAction,
358    ) {
359        match action {
360            SessionPickerAction::Rename => self.begin_session_rename(),
361            SessionPickerAction::Delete => self.begin_session_delete(),
362            SessionPickerAction::ToggleRead => self.toggle_session_read(),
363            SessionPickerAction::ToggleScope => self.toggle_session_scope().await,
364            SessionPickerAction::CycleProject => self.cycle_session_project_filter(),
365        }
366    }

369/// A session's title as the picker names it.
370fn session_title(session: &SessionMeta) -> String {
371    session.title.as_ref().map_or_else(
372        || {
373            format!(
374                "the untitled session {}",
375                crate::ui::pickers::short_id(session)
376            )
377        },
378        |title| format!("“{title}”"),
379    )
380}

382/// How many sessions `tree` holds below its root.
383fn subagent_count(tree: &cookie_agent_protocol::SessionTree) -> usize {
384    tree.children
385        .iter()
386        .map(|child| 1 + subagent_count(child))
387        .sum()
