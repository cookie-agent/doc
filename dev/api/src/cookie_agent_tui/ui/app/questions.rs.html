1//! The question panel: an agent's question to the user (ACP elicitation,
2//! `docs/site/specs/acp-agents.md` section 15).
3//!
4//! It docks where the approval panel does, above the status bar in place
5//! of the composer, whenever the viewed session has a question waiting and
6//! no approval is on top. A form shows one block per field, in the form's
7//! order; Tab moves between the fields and the buttons, ↑↓ between a
8//! field's choices, Space picks a choice or flips a box (Enter picks and
9//! moves on), and Enter on a button answers. A URL question shows the whole URL and its host,
10//! which `c` copies. Esc cancels the question and Ctrl-C the run. Like an
11//! approval's hotkeys, the answers wait out a short grace after the
12//! question appears.
13

27/// A button of the question panel.
28#[derive(Clone, Copy, Debug, Eq, PartialEq)]
29pub(in crate::ui) enum QuestionButton {
30    /// Send the form, or agree to open the URL (`accept`).
31    Submit,
32    /// Copy a URL question's URL.
33    CopyLink,
34    Decline,
35    Cancel,
36}

38impl QuestionButton {
39    fn label(self, kind: QuestionKind<'_>) -> &'static str {
40        match (self, kind) {
41            (Self::Submit, QuestionKind::Url(_)) => "✓ Done",
42            (Self::Submit, _) => "✓ Submit",
43            (Self::CopyLink, _) => "⧉ Copy link (c)",
44            (Self::Decline, _) => "✗ Decline",
45            (Self::Cancel, _) => "Cancel (esc)",
46        }
47    }
48
49    fn tone(self) -> crate::theme::DecisionTone {
50        match self {
51            Self::Submit => crate::theme::DecisionTone::Allow,
52            Self::Decline => crate::theme::DecisionTone::Deny,
53            Self::CopyLink | Self::Cancel => crate::theme::DecisionTone::Neutral,
54        }
55    }
56}

58/// The buttons a question offers, left to right.
59pub(in crate::ui) fn question_buttons(kind: QuestionKind<'_>) -> Vec<QuestionButton> {
60    match kind {

62            QuestionButton::Submit,
63            QuestionButton::Decline,
64            QuestionButton::Cancel,

85/// The panel's interaction state; it starts fresh for each question.
86#[derive(Default)]
87pub(in crate::ui) struct QuestionPanel {
88    question: Option<ElicitationId>,
89    /// When the question was first drawn on top.
90    shown_at: Option<Instant>,
91    pub(in crate::ui) draft: Option<FormDraft>,
92    /// The text of each text-like field, index-aligned with the form.
93    pub(in crate::ui) inputs: Vec<InputState>,
94    /// The highlighted choice of each choice field.
95    pub(in crate::ui) cursors: Vec<usize>,
96    pub(in crate::ui) focus: Option<QuestionFocus>,
97    /// The field that did not fit, from the last submit.
98    pub(in crate::ui) problem: Option<FieldProblem>,
99    /// A message beside the buttons.
100    pub(in crate::ui) notice: Option<String>,
101    /// An answer is on its way.
102    pub(in crate::ui) sending: bool,
103    pub(in crate::ui) scroll: u16,
104    max_scroll: u16,
105}

107impl QuestionPanel {
108    /// Reset for `question` when it was not the one shown.
109    pub(super) fn sync(&mut self, question: &QuestionState) {
110        if self.question == Some(question.request.elicitation_id) {
111            return;
112        }
113        let kind = question.kind();
114        *self = Self {
115            question: Some(question.request.elicitation_id),
116            ..Self::default()
117        };
118        match kind {
119            QuestionKind::Form(form) => {
120                let draft = FormDraft::new(form);
121                self.inputs = draft
122                    .values
123                    .iter()
124                    .map(|value| {
125                        let mut input = InputState::default();
126                        if let DraftValue::Text(text) = value {
127                            input.insert_text(text);
128                        }
129                        input
130                    })
131                    .collect();
132                self.cursors = draft
133                    .values
134                    .iter()
135                    .map(|value| match value {
136                        DraftValue::Choice(Some(index)) => *index,
137                        _ => 0,
138                    })
139                    .collect();
140                self.focus = Some(if form.fields.is_empty() {
141                    QuestionFocus::Button(0)
142                } else {
143                    QuestionFocus::Field(0)
144                });
145                self.draft = Some(draft);
146            }
147            // A URL question starts on Copy link, never on its answer.
148            QuestionKind::Url(_) | QuestionKind::Unsupported => {
149                self.focus = Some(QuestionFocus::Button(0));
150            }
151        }
152    }
153
154    /// The draft with the typed texts in it.
155    fn current_draft(&self) -> Option<FormDraft> {
156        let mut draft = self.draft.clone()?;
157        for (value, input) in draft.values.iter_mut().zip(&self.inputs) {
158            if let DraftValue::Text(text) = value {
159                input.as_str().clone_into(text);
160            }
161        }
162        Some(draft)
163    }
164}

166/// The marker of a choice row: a radio for a single choice, a box for a
167/// multiple one.
168fn choice_marker(multiple: bool, chosen: bool) -> &'static str {
169    match (multiple, chosen) {
170        (false, true) => "(•)",
171        (false, false) => "( )",
172        (true, true) => "[x]",
173        (true, false) => "[ ]",
174    }
175}

177/// The panel body's rows, and the row each field starts at.
178fn question_body(
179    question: &QuestionState,
180    panel: &QuestionPanel,
181    width: u16,
182    theme: &Theme,
183) -> (Vec<Line<'static>>, Vec<usize>) {
184    let mut rows = Vec::new();
185    let wrap = |rows: &mut Vec<Line<'static>>, line: Line<'static>, indent: usize| {
186        let room = width
187            .saturating_sub(u16::try_from(indent).unwrap_or(0))
188            .max(1);
189        for mut wrapped in wrapped_line(line, room) {
190            wrapped.spans.insert(0, Span::raw(" ".repeat(indent)));
191            rows.push(wrapped);
192        }
193    };
194    for line in question.request.message.lines() {
195        wrap(
196            &mut rows,
197            Line::from(Span::styled(line.to_owned(), theme.body())),
198            0,
199        );
200    }
201    let mut starts = Vec::new();
202    match question.kind() {
203        QuestionKind::Url(url) => {
204            rows.push(Line::default());
205            let host = url_host(url).unwrap_or("(no host)").to_owned();
206            rows.push(Line::from(vec![
207                Span::styled("Opens  ", theme.muted()),
208                Span::styled(host, theme.heading()),
209            ]));
210            wrap(
211                &mut rows,
212                Line::from(Span::styled(url.to_owned(), theme.link())),
213                7,
214            );
215            for warning in url_warnings(url) {
216                wrap(
217                    &mut rows,
218                    Line::from(Span::styled(format!("⚠ {warning}"), theme.warning())),
219                    7,
220                );
221            }
222            rows.push(Line::from(Span::styled(
223                "Open it in your browser, then choose Done when you have finished there.",
224                theme.muted(),
225            )));
226        }
227        QuestionKind::Unsupported => {
228            rows.push(Line::from(Span::styled(
229                "This client cannot show this kind of question: decline or cancel it.",
230                theme.warning(),
231            )));
232        }
233        QuestionKind::Form(form) => {
234            for line in form.title.iter().chain(&form.description) {
235                wrap(
236                    &mut rows,
237                    Line::from(Span::styled(line.clone(), theme.muted())),
238                    0,
239                );
240            }
241            let draft = panel.current_draft();
242            for (index, field) in form.fields.iter().enumerate() {
243                rows.push(Line::default());
244                starts.push(rows.len());
245                let focused = panel.focus == Some(QuestionFocus::Field(index));
246                let pointer = if focused { "› " } else { "  " };
247                let mut heading = vec![
248                    Span::styled(pointer, theme.warning()),
249                    Span::styled(field.label().to_owned(), theme.heading()),
250                ];
251                let hint = field_hint(field);
252                if !hint.is_empty() {
253                    heading.push(Span::styled(format!("  {hint}"), theme.muted()));
254                }
255                wrap(&mut rows, Line::from(heading), 0);
256                if let Some(description) = &field.description {
257                    wrap(
258                        &mut rows,
259                        Line::from(Span::styled(description.clone(), theme.muted())),
260                        4,
261                    );
262                }
263                let value = draft
264                    .as_ref()
265                    .and_then(|draft| draft.values.get(index))
266                    .cloned()
267                    .unwrap_or(DraftValue::Unsupported);
268                let cursor = panel.cursors.get(index).copied().unwrap_or(0);
269                match (field.input.known(), value) {
270                    (
271                        Some(
272                            ElicitationInput::Choice { choices, .. }
273                            | ElicitationInput::MultiChoice { choices, .. },
274                        ),
275                        value,
276                    ) => {
277                        let multiple = matches!(value, DraftValue::MultiChoice(_));
278                        for (choice_index, choice) in choices.iter().enumerate() {
279                            let chosen = match &value {
280                                DraftValue::Choice(chosen) => *chosen == Some(choice_index),
281                                DraftValue::MultiChoice(chosen) => {
282                                    chosen.get(choice_index).copied().unwrap_or(false)
283                                }
284                                _ => false,
285                            };
286                            let highlighted = focused && cursor == choice_index;
287                            let style = if highlighted {
288                                theme.selected()
289                            } else {
290                                theme.body()
291                            };
292                            let mut spans = vec![Span::styled(
293                                format!("{} {}", choice_marker(multiple, chosen), choice.title),
294                                style,
295                            )];
296                            if let Some(description) = &choice.description {
297                                spans
298                                    .push(Span::styled(format!(" — {description}"), theme.muted()));
299                            }
300                            wrap(&mut rows, Line::from(spans), 4);
301                        }
302                    }
303                    (Some(ElicitationInput::Boolean { .. }), DraftValue::Boolean(on)) => {
304                        let style = if focused {
305                            theme.selected()
306                        } else {
307                            theme.body()
308                        };
309                        rows.push(Line::from(vec![
310                            Span::raw("    "),
311                            Span::styled(
312                                format!(
313                                    "{} {}",
314                                    choice_marker(true, on),
315                                    if on { "yes" } else { "no" }
316                                ),
317                                style,
318                            ),
319                        ]));
320                    }
321                    (Some(_), DraftValue::Text(_)) => {
322                        let text = panel
323                            .inputs
324                            .get(index)
325                            .map(|input| input.as_str().to_owned())
326                            .unwrap_or_default();
327                        let caret = if focused { "▏" } else { "" };
328                        let shown = if text.is_empty() && !focused {
329                            Span::styled("(empty)", theme.internal())
330                        } else {
331                            Span::styled(format!("{text}{caret}"), theme.body())
332                        };
333                        rows.push(Line::from(vec![
334                            Span::raw("    "),
335                            Span::styled("[ ", theme.muted()),
336                            shown,
337                            Span::styled(" ]", theme.muted()),
338                        ]));
339                    }
340                    _ => rows.push(Line::from(vec![
341                        Span::raw("    "),
342                        Span::styled("this client cannot fill it in", theme.internal()),
343                    ])),
344                }
345                if let Some(problem) = panel
346                    .problem
347                    .as_ref()
348                    .filter(|problem| problem.index == index)
349                {
350                    wrap(
351                        &mut rows,
352                        Line::from(Span::styled(
353                            format!("✗ {}", problem.message),
354                            theme.error(),
355                        )),
356                        4,
357                    );
358                }
359            }
360        }
361    }
362    (rows, starts)
363}

365/// The footer hint for the bottom border, the longest that fits.
366fn question_hints(kind: QuestionKind<'_>, width: u16) -> String {
367    let variants: &[&str] = match kind {
368        QuestionKind::Form(_) => &[
369            "tab next · ↑↓ choose · space pick · ⏎ next/press · esc cancel",
370            "tab · ↑↓ · space · ⏎ · esc",
371        ],
372        _ => &["←→ choose · ⏎ press · esc cancel", "←→ ⏎ · esc"],
373    };
374    let room = usize::from(width.saturating_sub(4));
375    variants
376        .iter()
377        .find(|hint| UnicodeWidthStr::width(**hint) <= room)
378        .map_or_else(String::new, |hint| (*hint).to_owned())
379}

381/// Whether a field of `form` takes typed text.
382fn is_text_field(form: &ElicitationForm, index: usize) -> bool {
383    matches!(
384        form.fields.get(index).and_then(|field| field.input.known()),
385        Some(
386            ElicitationInput::Text { .. }
387                | ElicitationInput::Number { .. }
388                | ElicitationInput::Integer { .. }
389        )
390    )
391}

393/// How many choices field `index` of `form` has, for a choice field.
394fn choice_count(form: &ElicitationForm, index: usize) -> Option<usize> {
395    match form.fields.get(index)?.input.known()? {
396        ElicitationInput::Choice { choices, .. }
397        | ElicitationInput::MultiChoice { choices, .. } => Some(choices.len()),
398        _ => None,
399    }

414    /// Whether the question on top has been shown for the full grace, so
415    /// keys aimed at the composer cannot answer it.
416    fn question_armed(&mut self, now: Instant) -> bool {
417        let shown = *self.question_panel.shown_at.get_or_insert(now);
418        now.duration_since(shown) >= APPROVAL_HOTKEY_GRACE
419    }

421    /// Backdate the question's appearance past the grace.
422    #[cfg(test)]
423    pub(crate) fn arm_question_for_test(&mut self) {
424        if let Some(question) = self.current_question().cloned() {
425            self.question_panel.sync(&question);
426        }
427        self.question_panel.shown_at = Instant::now().checked_sub(APPROVAL_HOTKEY_GRACE * 2);

430    /// Keys while a question is on top: the panel owns the keyboard.
431    pub(super) async fn handle_question_key(&mut self, key: KeyEvent) {
432        let Some(question) = self.current_question().cloned() else {
433            return;
434        };
435        self.question_panel.sync(&question);
436        self.question_panel.notice = None;
437        if key.code == KeyCode::Char('c') && key.modifiers == KeyModifiers::CONTROL {
438            self.cancel_active_run();
439            return;
440        }
441        let kind = question.kind();
442        let buttons = question_buttons(kind);
443        let fields = match kind {
444            QuestionKind::Form(form) => form.fields.len(),
445            _ => 0,
446        };
447        let focus = self
448            .question_panel
449            .focus
450            .unwrap_or(QuestionFocus::Button(0));
451        let stops = fields + buttons.len();
452        let position = match focus {
453            QuestionFocus::Field(index) => index,
454            QuestionFocus::Button(index) => fields + index,
455        };
456        let at = |position: usize| {
457            if position < fields {
458                QuestionFocus::Field(position)
459            } else {
460                QuestionFocus::Button(position - fields)
461            }
462        };
463        let step = |forward: bool| {
464            if forward {
465                at((position + 1) % stops)
466            } else {
467                at((position + stops - 1) % stops)
468            }
469        };
470        match (key.code, focus) {
471            (KeyCode::Esc, _) => {
472                self.answer_question(&question, QuestionButton::Cancel)
473                    .await
474            }
475            (KeyCode::Tab, _) => self.question_panel.focus = Some(step(true)),
476            (KeyCode::BackTab, _) => self.question_panel.focus = Some(step(false)),
477            (KeyCode::Left, QuestionFocus::Button(_)) => {
478                self.question_panel.focus = Some(step(false));
479            }
480            (KeyCode::Right, QuestionFocus::Button(_)) => {
481                self.question_panel.focus = Some(step(true));
482            }
483            (KeyCode::Enter, QuestionFocus::Button(index)) => {
484                if let Some(button) = buttons.get(index) {
485                    self.answer_question(&question, *button).await;
486                }
487            }
488            (KeyCode::Char('c'), QuestionFocus::Button(_))
489                if matches!(kind, QuestionKind::Url(_)) =>
490            {
491                self.answer_question(&question, QuestionButton::CopyLink)
492                    .await;
493            }
494            (KeyCode::Up | KeyCode::Down, QuestionFocus::Button(_)) => {
495                self.question_panel.focus = Some(step(key.code == KeyCode::Down));
496            }
497            (_, QuestionFocus::Field(index)) => {
498                let QuestionKind::Form(form) = kind else {
499                    return;
500                };
501                self.question_field_key(form, index, key, step(true), step(false));
502            }
503            _ => {}
504        }
505    }

507    /// A key on form field `index`: `next` and `previous` are the focus
508    /// stops around it.
509    fn question_field_key(
510        &mut self,
511        form: &ElicitationForm,
512        index: usize,
513        key: KeyEvent,
514        next: QuestionFocus,
515        previous: QuestionFocus,
516    ) {
517        let panel = &mut self.question_panel;
518        panel.problem = None;
519        if is_text_field(form, index) {
520            match key.code {
521                KeyCode::Enter | KeyCode::Down => panel.focus = Some(next),
522                KeyCode::Up => panel.focus = Some(previous),
523                _ => {
524                    if let Some(input) = panel.inputs.get_mut(index) {
525                        super::keys::edit_plain_input(input, key);
526                    }
527                }
528            }
529            return;
530        }
531        if let Some(count) = choice_count(form, index) {
532            let cursor = panel.cursors.get(index).copied().unwrap_or(0);
533            match key.code {
534                KeyCode::Up if cursor == 0 => panel.focus = Some(previous),
535                KeyCode::Up => panel.cursors[index] = cursor - 1,
536                KeyCode::Down if cursor + 1 >= count => panel.focus = Some(next),
537                KeyCode::Down => panel.cursors[index] = cursor + 1,
538                KeyCode::Char(' ') | KeyCode::Enter => {
539                    let Some(value) = panel
540                        .draft
541                        .as_mut()
542                        .and_then(|draft| draft.values.get_mut(index))
543                    else {
544                        return;
545                    };
546                    match value {
547                        DraftValue::Choice(chosen) => {
548                            *chosen = Some(cursor);
549                            if key.code == KeyCode::Enter {
550                                panel.focus = Some(next);
551                            }
552                        }
553                        DraftValue::MultiChoice(chosen) => {
554                            if let Some(chosen) = chosen.get_mut(cursor) {
555                                *chosen = !*chosen;
556                            }
557                        }
558                        _ => {}
559                    }
560                }
561                _ => {}
562            }
563            return;
564        }
565        // A yes-or-no box, or a field this client cannot fill in.
566        match key.code {
567            KeyCode::Down | KeyCode::Enter => panel.focus = Some(next),
568            KeyCode::Up => panel.focus = Some(previous),
569            KeyCode::Char(' ') => {
570                if let Some(DraftValue::Boolean(on)) = panel
571                    .draft
572                    .as_mut()
573                    .and_then(|draft| draft.values.get_mut(index))
574                {
575                    *on = !*on;
576                }
577            }
578            _ => {}
579        }
580    }

582    /// Presses `button` of `question`: answers it, or copies its URL.
583    async fn answer_question(&mut self, question: &QuestionState, button: QuestionButton) {
584        if button == QuestionButton::CopyLink {
585            if let QuestionKind::Url(url) = question.kind() {
586                self.copy_to_clipboard(url.to_owned());
587                self.question_panel.notice = Some("link copied".into());
588            }
589            return;
590        }
591        if self.question_panel.sending {
592            return;
593        }
594        if !self.question_armed(Instant::now()) {
595            self.question_panel.notice = Some("just appeared · press again".into());
596            return;
597        }
598        let (action, content) = match button {
599            QuestionButton::Submit => {
600                let content = match question.kind() {
601                    QuestionKind::Form(form) => {
602                        let draft = self
603                            .question_panel
604                            .current_draft()
605                            .unwrap_or_else(|| FormDraft::new(form));
606                        match draft.content(form) {
607                            Ok(content) => Some(content),
608                            Err(problem) => {
609                                self.question_panel.notice = Some(problem.message.clone());
610                                self.question_panel.focus =
611                                    Some(QuestionFocus::Field(problem.index));
612                                self.question_panel.problem = Some(problem);
613                                return;
614                            }
615                        }
616                    }
617                    QuestionKind::Url(_) => None,
618                    QuestionKind::Unsupported => return,
619                };
620                (ElicitationAction::Accept, content)
621            }
622            QuestionButton::Decline => (ElicitationAction::Decline, None),
623            QuestionButton::Cancel | QuestionButton::CopyLink => (ElicitationAction::Cancel, None),
624        };
625        self.question_panel.sending = true;
626        let client = self.client.clone();
627        let updates = self.rpc_updates_tx.clone();
628        let params = ElicitationRespondParams {
629            session_id: question.session_id,
630            elicitation_id: question.request.elicitation_id,
631            action,
632            content,
633        };
634        self.spawn_rpc(async move {
635            let session_id = params.session_id;
636            let elicitation_id = params.elicitation_id;
637            let result = client
638                .respond_elicitation(params)
639                .await
640                .map(|_| ())
641                .map_err(|error| (error.error_code().map(str::to_owned), error.to_string()));
642            let _ = updates.send(RpcUpdate::QuestionAnswered {
643                session_id,
644                elicitation_id,
645                result,
646            });
647        });
648    }

650    /// `elicitation.respond` answered. The question leaves when its
651    /// `elicitation_resolved` arrives; one no longer waiting leaves now.
652    pub(super) fn finish_question_answer(
653        &mut self,
654        session_id: SessionId,
655        elicitation_id: ElicitationId,
656        result: Result<(), (Option<String>, String)>,
657    ) {
658        if self.question_panel.question == Some(elicitation_id) {
659            self.question_panel.sending = false;
660        }
661        match result {
662            Ok(()) => {}
663            Err((code, message)) => {
664                if code.as_deref() == Some("elicitation_not_pending") {
665                    if let Some(state) = self.store.sessions.get_mut(&session_id) {
666                        state
667                            .questions
668                            .retain(|question| question.request.elicitation_id != elicitation_id);
669                    }
670                    self.status = "the question is no longer waiting".into();
671                } else {
672                    self.status = format!("answer not sent: {message}");
673                    if self.question_panel.question == Some(elicitation_id) {
674                        self.question_panel.notice = Some(format!("not sent: {message}"));
675                    }
676                }
677            }
678        }
679    }

681    /// Pastes `text`, on one line, into the question's focused text field.
682    pub(super) fn paste_into_question(&mut self, text: &str) {
683        let Some(question) = self.current_question().cloned() else {
684            return;
685        };
686        self.question_panel.sync(&question);
687        let QuestionKind::Form(form) = question.kind() else {
688            return;
689        };
690        if let Some(QuestionFocus::Field(index)) = self.question_panel.focus
691            && is_text_field(form, index)
692            && let Some(input) = self.question_panel.inputs.get_mut(index)
693        {
694            input.insert_text(&text.replace(['\r', '\n'], " "));
695        }
696    }

698    pub(super) fn scroll_question(&mut self, up: bool, lines: u16) {
699        let panel = &mut self.question_panel;
700        panel.scroll = if up {
701            panel.scroll.saturating_sub(lines)
702        } else {
703            panel.scroll.saturating_add(lines).min(panel.max_scroll)
704        };
705    }

707    /// Draws the question panel docked with its bottom edge at
708    /// `dock_bottom`, sized to its content and capped so a few
709    /// conversation rows stay visible.
710    pub(in crate::ui) fn render_question(
711        &mut self,
712        frame: &mut ratatui::Frame,
713        question: &QuestionState,
714        screen: Rect,
715        dock_bottom: u16,
716    ) {
717        self.question_panel.sync(question);
718        self.question_panel
719            .shown_at
720            .get_or_insert_with(Instant::now);
721        let theme = &self.theme;
722        let kind = question.kind();
723        let text_width = screen.width.saturating_sub(4).max(1);
724        let (rows, starts) = question_body(question, &self.question_panel, text_width, theme);
725        let footer_height = 2;
726        let desired = u16::try_from(rows.len())
727            .unwrap_or(u16::MAX)
728            .saturating_add(3 + footer_height);
729        let available = dock_bottom.saturating_sub(screen.y);
730        let cap = available
731            .saturating_sub(QUESTION_MIN_CONVERSATION_ROWS)
732            .max(available.min(8 + footer_height));
733        let height = desired.min(cap);
734        let area = Rect::new(screen.x, dock_bottom - height, screen.width, height);
735        self.hit_map.question = Some(area);
736        paint_panel(frame, area, theme);
737        let inner = inner_rect(area);
738        let content = Rect::new(
739            inner.x.saturating_add(1),
740            inner.y,
741            inner.width.saturating_sub(2),
742            inner.height,
743        );
744        let body = Rect::new(
745            content.x,
746            content.y.saturating_add(1).min(content.bottom()),
747            content.width,
748            content.height.saturating_sub(1 + footer_height),
749        );
750        let line_count = u16::try_from(rows.len()).unwrap_or(u16::MAX);
751        let panel = &mut self.question_panel;
752        panel.max_scroll = line_count.saturating_sub(body.height);
753        // Keep the focused field in view.
754        if let Some(QuestionFocus::Field(index)) = panel.focus
755            && let Some(start) = starts.get(index)
756        {
757            let start = u16::try_from(*start).unwrap_or(u16::MAX);
758            let end = starts
759                .get(index + 1)
760                .map_or(line_count, |next| u16::try_from(*next).unwrap_or(u16::MAX));
761            if start < panel.scroll {
762                panel.scroll = start;
763            } else if end > panel.scroll.saturating_add(body.height) {
764                panel.scroll = end.saturating_sub(body.height).min(start);
765            }
766        }
767        panel.scroll = panel.scroll.min(panel.max_scroll);
768        let scroll = panel.scroll;
769        let queued = self
770            .selected
771            .and_then(|id| self.store.sessions.get(&id))
772            .map_or(0, |state| state.questions.len());
773        let mut block = crate::ui::panel_block()
774            .border_style(theme.warning())
775            .title(crate::ui::fitted_panel_title(
776                Span::styled(
777                    format!("? {} · waiting on you", question.heading()),
778                    theme.warning(),
779                ),
780                area.width,
781            ))
782            .style(theme.panel());
783        if queued > 1 {
784            block = block.title(crate::ui::panel_title(
785                Line::from(Span::styled(format!("1 of {queued}"), theme.muted())).right_aligned(),
786            ));
787        }
788        if panel.max_scroll > 0 {
789            block = block.title_bottom(crate::ui::panel_title(Line::from(Span::styled(
790                format!(
791                    "{}–{}/{line_count}",
792                    scroll.saturating_add(1),
793                    scroll.saturating_add(body.height).min(line_count)
794                ),
795                theme.internal(),
796            ))));
797        }
798        let hints = question_hints(kind, area.width);
799        if !hints.is_empty() {
800            block = block.title_bottom(crate::ui::panel_title(
801                Line::from(Span::styled(hints, theme.internal())).right_aligned(),
802            ));
803        }
804        frame.render_widget(block, area);
805        for (offset, row) in rows
806            .into_iter()
807            .skip(usize::from(scroll))
808            .take(usize::from(body.height))
809            .enumerate()
810        {
811            let y = body.y + offset as u16;
812            frame.render_widget(Paragraph::new(row), Rect::new(body.x, y, body.width, 1));
813        }
814        let buttons_row = Rect::new(
815            content.x,
816            content.bottom().saturating_sub(1),
817            content.width,
818            1,
819        );
820        let mut x = buttons_row.x;
821        for (index, button) in question_buttons(kind).into_iter().enumerate() {
822            let focused = self.question_panel.focus == Some(QuestionFocus::Button(index));
823            let tone = theme.decision(button.tone(), focused);
824            let style = if focused {
825                tone
826            } else {
827                theme
828                    .code_background()
829                    .map_or_else(Style::default, |background| Style::default().bg(background))
830                    .patch(tone)
831            };
832            let label = format!(" {} ", button.label(kind));
833            let width = u16::try_from(UnicodeWidthStr::width(label.as_str())).unwrap_or(u16::MAX);
834            let visible = width.min(buttons_row.right().saturating_sub(x));
835            if visible == 0 {
836                break;
837            }
838            frame.render_widget(
839                Paragraph::new(Span::styled(label, style)),
840                Rect::new(x, buttons_row.y, visible, 1),
841            );
842            x = x.saturating_add(width).saturating_add(2);
843        }
844        let notice = if self.question_panel.sending {
845            Some("sending…".to_owned())
846        } else {
847            self.question_panel.notice.clone()
848        };
849        if let Some(notice) = notice {
850            let room = buttons_row.right().saturating_sub(x);
851            let notice = truncate_with_ellipsis(&notice, usize::from(room));
852            let width = u16::try_from(UnicodeWidthStr::width(notice.as_str())).unwrap_or(room);
853            frame.render_widget(
854                Paragraph::new(Span::styled(notice, theme.warning())),
855                Rect::new(buttons_row.right() - width, buttons_row.y, width, 1),
856            );
857        }
858    }
