1//! Background calls (`docs/site/specs/background-tools.md` section 9): the
2//! title-row button that moves a running call to the background, the bar
3//! above the composer while the session has background calls, and the panel
4//! that lists them and stops one.
5
6use cookie_agent_protocol::{BackgroundActor, BackgroundCall, ToolCallId};

8use ratatui::{

11    widgets::Paragraph,

13use unicode_width::UnicodeWidthStr;

17/// The bar's control that opens the panel.
18const DETAILS: &str = " [Details]";
19/// Each panel row's control that stops its call.
20const STOP: &str = " [stop]";
21
22#[derive(Debug, Default)]
23pub(super) struct BackgroundPanelState {
24    pub(super) selected: usize,
25}
26
27impl App {
28    /// The selected session's background calls.
29    pub(super) fn background_calls(&self) -> &[BackgroundCall] {
30        self.selected
31            .and_then(|session_id| self.store.sessions.get(&session_id))
32            .map_or(&[], |state| state.background_calls.as_slice())
33    }
34
35    pub(super) fn background_bar_visible(&self) -> bool {
36        !self.background_calls().is_empty()
37    }
38
39    /// Asks the daemon to move the running call `call_id` to the background.
40    pub(super) fn request_background(&mut self, call_id: ToolCallId) {
41        let Some(run_id) = self.selected_active_run() else {
42            self.status = "the call is no longer running".into();
43            return;
44        };
45        let client = self.client.clone();
46        let updates = self.rpc_updates_tx.clone();
47        self.status = "moving the call to the background…".into();
48        self.spawn_rpc(async move {
49            let update = match client
50                .background_tool(cookie_agent_protocol::RunToolBackgroundParams { run_id, call_id })
51                .await
52            {
53                Ok(result) if result.accepted => {
54                    RpcUpdate::Notice("the call moves to the background once it can".into())
55                }
56                Ok(_) => RpcUpdate::Notice("this call cannot move to the background".into()),
57                Err(error) => RpcUpdate::Status(error.to_string()),
58            };
59            let _ = updates.send(update);
60        });
61    }

63    pub(super) fn render_background_bar(&mut self, frame: &mut ratatui::Frame, area: Rect) {
64        self.hit_map.background_bar = None;
65        if area.width == 0 || area.height == 0 {
66            return;
67        }
68        let calls = self.background_calls();
69        if calls.is_empty() {
70            return;
71        }
72        let summary = calls
73            .iter()
74            .map(|call| match &call.display {
75                Some(display) => format!("{}: {}", call.tool, single_line(display)),
76                None => call.tool.clone(),
77            })
78            .collect::<Vec<_>>()
79            .join(" · ");
80        let width = usize::from(area.width);
81        let controls = if width >= UnicodeWidthStr::width(DETAILS) + 16 {
82            DETAILS
83        } else {
84            ""
85        };
86        let details_width = width.saturating_sub(UnicodeWidthStr::width(controls));
87        let mut text = truncate_with_ellipsis(
88            &format!("⏳ {} in background: {summary}", calls.len()),
89            details_width,
90        );
91        text.push_str(
92            &" ".repeat(details_width.saturating_sub(UnicodeWidthStr::width(text.as_str()))),
93        );
94        frame.render_widget(
95            Paragraph::new(Line::from(vec![
96                Span::styled(text, self.theme.assistant()),
97                Span::styled(controls, self.theme.link()),
98            ]))
99            .style(self.theme.panel()),
100            area,
101        );
102        self.hit_map.background_bar = Some(area);
103    }

105    pub(super) fn open_background_panel(&mut self) {
106        if self.background_calls().is_empty() {
107            self.status = "no calls are running in the background".into();
108            return;
109        }
110        self.background_panel = BackgroundPanelState::default();
111        self.modal = Modal::Background;
112    }

114    pub(super) fn render_background_panel(&mut self, frame: &mut ratatui::Frame) {
115        self.hit_map.background_rows.clear();
116        self.hit_map.background_stops.clear();
117        let calls = self.background_calls().to_vec();
118        if calls.is_empty() {
119            self.modal = Modal::None;
120            return;
121        }
122        self.background_panel.selected = self.background_panel.selected.min(calls.len() - 1);
123        let frame_area = frame.area();
124        if frame_area.width < 8 || frame_area.height < 4 {
125            return;
126        }
127        let width = frame_area.width.min(90);
128        let height = frame_area.height.min(
129            u16::try_from(calls.len())
130                .unwrap_or(u16::MAX)
131                .saturating_add(2),
132        );
133        let area = Rect::new(
134            frame_area.x + (frame_area.width - width) / 2,
135            frame_area.y + (frame_area.height - height) / 2,
136            width,
137            height,
138        );
139        paint_panel(frame, area, &self.theme);
140        let inner_width = usize::from(area.width.saturating_sub(2));
141        let row_width = inner_width.saturating_sub(UnicodeWidthStr::width(STOP));
142        let now = jiff::Timestamp::now();
143        let lines = calls
144            .iter()
145            .enumerate()
146            .take(usize::from(area.height.saturating_sub(2)))
147            .map(|(index, call)| {
148                let elapsed = now.duration_since(call.started_at).as_secs().max(0);
149                let by = match call.by {
150                    BackgroundActor::Model => "by the model",
151                    BackgroundActor::User => "by you",
152                };
153                let row = format!(
154                    "{} {}{} · {} · {by}",
155                    call.job,
156                    call.tool,
157                    call.display
158                        .as_deref()
159                        .map(|display| format!(": {}", single_line(display)))
160                        .unwrap_or_default(),
161                    elapsed_label(elapsed),
162                );
163                let mut row = truncate_with_ellipsis(&row, row_width);
164                row.push_str(
165                    &" ".repeat(row_width.saturating_sub(UnicodeWidthStr::width(row.as_str()))),
166                );
167                let style = if index == self.background_panel.selected {
168                    self.theme.selected()
169                } else {
170                    self.theme.body()
171                };
172                let y = area.y + 1 + u16::try_from(index).unwrap_or(u16::MAX);
173                self.hit_map
174                    .background_rows
175                    .push((Rect::new(area.x + 1, y, row_width as u16, 1), index));
176                self.hit_map.background_stops.push((
177                    Rect::new(
178                        area.x + 1 + row_width as u16,
179                        y,
180                        UnicodeWidthStr::width(STOP) as u16,
181                        1,
182                    ),
183                    index,
184                ));
185                Line::from(vec![
186                    Span::styled(row, style),
187                    Span::styled(STOP, self.theme.link()),
188                ])
189            })
190            .collect::<Vec<_>>();
191        frame.render_widget(
192            Paragraph::new(lines).block(
193                crate::ui::panel_block()
194                    .border_style(self.theme.panel_border())
195                    .title(crate::ui::panel_title(Span::styled(
196                        truncate_with_ellipsis("Background calls", inner_width),
197                        self.theme.heading(),
198                    )))
199                    .title_bottom(crate::ui::panel_title(
200                        Line::from(Span::styled(
201                            truncate_with_ellipsis("Up/Down · s: stop · Esc: close", inner_width),
202                            self.theme.internal(),
203                        ))
204                        .right_aligned(),
205                    )),
206            ),
207            area,
208        );
209    }

211    pub(super) fn handle_background_key(&mut self, key: KeyEvent) {
212        let count = self.background_calls().len();
213        match key.code {
214            KeyCode::Esc | KeyCode::Enter => self.modal = Modal::None,
215            KeyCode::Up => {
216                self.background_panel.selected = self.background_panel.selected.saturating_sub(1);
217            }
218            KeyCode::Down if count > 0 => {
219                self.background_panel.selected =
220                    (self.background_panel.selected + 1).min(count - 1);
221            }
222            KeyCode::Char('s') | KeyCode::Delete => {
223                self.stop_background_call(self.background_panel.selected);
224            }
225            _ => {}
226        }
227    }
228
229    /// Stops the background call at `index` of the selected session's list,
230    /// as the user. It still reports to the model.
231    pub(super) fn stop_background_call(&mut self, index: usize) {
232        let (Some(session_id), Some(call)) =
233            (self.selected, self.background_calls().get(index).cloned())
234        else {
235            return;
236        };
237        self.background_panel.selected = index;
238        let client = self.client.clone();
239        let updates = self.rpc_updates_tx.clone();
240        self.status = format!("stopping {}…", call.job);
241        self.spawn_rpc(async move {
242            let update = match client.stop_background(session_id, call.job.clone()).await {
243                Ok(_) => RpcUpdate::Notice(format!("stopped {}", call.job)),
244                Err(error) => RpcUpdate::Status(error.to_string()),
245            };
246            let _ = updates.send(update);
247        });
248    }
249}
250
251fn single_line(value: &str) -> String {
252    value.split_whitespace().collect::<Vec<_>>().join(" ")
253}
254
255fn elapsed_label(seconds: i64) -> String {
256    match seconds {
257        0..60 => format!("{seconds}s"),
258        60..3600 => format!("{}m{:02}s", seconds / 60, seconds % 60),
259        _ => format!("{}h{:02}m", seconds / 3600, seconds % 3600 / 60),
260    }
