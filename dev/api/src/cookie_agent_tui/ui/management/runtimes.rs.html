30/// A runtime a new-session draft attaches. Its instance is opened as soon as
31/// it joins the draft, so its status, and why it failed, show before the
32/// session exists.
33#[derive(Clone, Debug, Eq, PartialEq)]
34pub(in crate::ui) struct DraftRuntime {
35    /// Matches the answer of the open request to this entry.
36    pub(in crate::ui) id: u64,
37    pub(in crate::ui) request: RuntimeAttachRequest,
38    /// The instance the open request answered, or why it failed.
39    pub(in crate::ui) opened: Option<Result<RuntimeInstanceInfo, String>>,
40}

42impl DraftRuntime {
43    pub(in crate::ui) fn is_global(&self) -> bool {
44        self.request.runtime_id.as_str() == RuntimeId::GLOBAL
45    }
46
47    /// The opened instance's canonical workspace, else the one requested.
48    pub(in crate::ui) fn workspace(&self) -> Option<&str> {
49        match &self.opened {
50            Some(Ok(instance)) => instance.workspace.as_deref(),
51            _ => self.request.workspace.as_deref(),
52        }
53    }
54
55    /// Whether `other` names the same instance as this entry.
56    pub(in crate::ui) fn same_instance(&self, other: &RuntimeAttachRequest) -> bool {
57        self.request.runtime_id == other.runtime_id
58            && self.request.capability == other.capability
59            && (self.request.workspace == other.workspace
60                || self.workspace() == other.workspace.as_deref())
61    }
62}

64/// One thing the attach step offers: a capability of a runtime, or a runtime
65/// that has not said what it offers yet, which attaching connects and takes
66/// its only capability of.
67#[derive(Clone, Debug, Eq, PartialEq)]
68pub(in crate::ui) struct RuntimeChoice {
69    pub(in crate::ui) runtime_id: RuntimeId,
70    pub(in crate::ui) capability: Option<RuntimeCapabilityInfo>,
71    pub(in crate::ui) connected: bool,
72    /// Why the runtime last failed to connect.
73    pub(in crate::ui) message: Option<String>,
74}
75
76impl RuntimeChoice {
77    /// Whether attaching asks for a workspace: the capability takes one, or
78    /// the runtime has not said what it offers yet (then it is optional).
79    pub(in crate::ui) fn asks_workspace(&self) -> bool {
80        self.capability
81            .as_ref()
82            .is_none_or(|capability| capability.takes_workspace)
83    }
84
85    fn label(&self) -> String {
86        match &self.capability {
87            Some(capability) => format!("{}/{}", self.runtime_id, capability.name),
88            None => self.runtime_id.to_string(),
89        }
90    }
91
92    /// Case-insensitive substring match over the runtime ID and the
93    /// capability's name, kind, and description.
94    fn matches(&self, query: &str) -> bool {
95        let query = query.trim().to_lowercase();
96        if query.is_empty() {
97            return true;
98        }
99        let mut haystack = self.runtime_id.to_string();
100        if let Some(capability) = &self.capability {
101            haystack.push(' ');
102            haystack.push_str(capability.name.as_str());
103            haystack.push(' ');
104            haystack.push_str(capability.kind.as_str());
105            haystack.push(' ');
106            haystack.push_str(&capability.description);
107        }
108        haystack.to_lowercase().contains(&query)
109    }
110}

145pub(in crate::ui) struct RuntimePanel {
146    pub(in crate::ui) listing: Option<RuntimesListResult>,
147    pub(in crate::ui) selection: ListState,
148    pub(in crate::ui) form: Option<RuntimeAttachForm>,
149    pub(in crate::ui) refresh_in_flight: bool,

152impl RuntimePanel {
153    pub(in crate::ui) fn install(&mut self, listing: RuntimesListResult) {
154        self.listing = Some(listing);
155        self.refresh_in_flight = false;
156    }
157
158    pub(in crate::ui) fn runtimes(&self) -> &[RuntimeListing] {
159        self.listing
160            .as_ref()
161            .map_or(&[], |listing| listing.runtimes.as_slice())
162    }
163
164    /// Opens the attach step; there is nothing to attach before
165    /// `runtimes.list` answered.
166    pub(in crate::ui) fn begin_attach(&mut self) -> Result<(), String> {
167        if self.runtimes().is_empty() {
168            return Err("no runtimes are available yet".into());
169        }
170        let mut form = RuntimeAttachForm::default();
171        form.selection.select(Some(0));
172        self.form = Some(form);
173        Ok(())
174    }
175
176    /// Every runtime's capabilities, in listing order, that match the
177    /// search; a runtime that has not said what it offers is one choice.
178    pub(in crate::ui) fn choices(&self) -> Vec<RuntimeChoice> {
179        let query = self.form.as_ref().map_or("", |form| form.search.as_str());
180        self.runtimes()
181            .iter()
182            .flat_map(|runtime| {
183                let choice = |capability| RuntimeChoice {
184                    runtime_id: runtime.runtime_id.clone(),
185                    capability,
186                    connected: runtime.connected,
187                    message: runtime.message.clone(),
188                };
189                if runtime.capabilities.is_empty() {
190                    vec![choice(None)]
191                } else {
192                    runtime
193                        .capabilities
194                        .iter()
195                        .cloned()
196                        .map(|capability| choice(Some(capability)))
197                        .collect()
198                }
199            })
200            .filter(|choice| choice.matches(query))
201            .collect()
202    }
203
204    /// Keeps the highlighted choice within the filtered list.
205    pub(in crate::ui) fn clamp_choice(&mut self) {
206        let count = self.choices().len();
207        if let Some(form) = self.form.as_mut() {
208            form.selection
209                .select((count > 0).then(|| form.selection.selected().unwrap_or(0).min(count - 1)));
210        }
211    }
212
213    pub(in crate::ui) fn move_choice(&mut self, up: bool) {
214        let count = self.choices().len();
215        if let Some(form) = self.form.as_mut()
216            && count > 0
217        {
218            let index = form.selection.selected().unwrap_or(0);
219            form.selection.select(Some(cycle(index, count, up)));
220        }
221    }
222
223    /// Takes the highlighted choice: on to its parameters when it needs
224    /// any, else straight to attaching it.
225    pub(in crate::ui) fn choose(&mut self) -> Result<AttachStep, String> {
226        let choices = self.choices();
227        let form = self.form.as_mut().ok_or("no attach in progress")?;
228        let choice = form
229            .selection
230            .selected()
231            .and_then(|index| choices.get(index))
232            .ok_or("no runtime matches the search")?
233            .clone();
234        if choice.asks_workspace() {
235            form.workspace = InputState::default();
236            form.chosen = Some(choice);
237            return Ok(AttachStep::Parameters);
238        }
239        Ok(AttachStep::Attach(
240            choice.runtime_id,
241            choice.capability.map(|capability| capability.name),
242            None,
243        ))
244    }
245
246    /// Esc: from the parameters back to the list, from the list out of the
247    /// attach step.
248    pub(in crate::ui) fn back(&mut self) {
249        match self.form.as_mut() {
250            Some(form) if form.chosen.is_some() => form.chosen = None,
251            _ => self.form = None,
252        }
253    }
254
255    /// The attach request the parameters describe.
256    pub(in crate::ui) fn attach_request(
257        &self,
258    ) -> Result<(RuntimeId, Option<CapabilityName>, Option<String>), String> {
259        let form = self.form.as_ref().ok_or("no attach in progress")?;
260        let choice = form.chosen.as_ref().ok_or("choose a runtime")?;
261        let workspace = form.workspace.as_str().trim();
262        let workspace = match (&choice.capability, workspace.is_empty()) {
263            (Some(capability), true) => {
264                return Err(format!(
265                    "capability `{}` of runtime `{}` needs a workspace path",
266                    capability.name, choice.runtime_id
267                ));
268            }
269            (None, true) => None,
270            (_, false) => Some(workspace.to_owned()),
271        };
272        Ok((
273            choice.runtime_id.clone(),
274            choice
275                .capability
276                .as_ref()
277                .map(|capability| capability.name.clone()),
278            workspace,
279        ))
280    }
281
282    /// The `runtimes.reconnect` request that retries `row`'s instance: its
283    /// workspace goes along exactly when it has one.
284    pub(in crate::ui) fn reconnect_request(
285        &self,
286        row: &RuntimeRow,
287    ) -> Result<(RuntimeId, CapabilityName, Option<String>), String> {
288        if !self
289            .runtimes()
290            .iter()
291            .any(|runtime| runtime.runtime_id == row.attachment.runtime_id)
292        {
293            return Err(format!(
294                "runtime `{}` is not a known runtime",
295                row.attachment.runtime_id
296            ));
297        }
298        Ok((
299            row.attachment.runtime_id.clone(),
300            row.attachment.capability.clone(),
301            row.attachment.workspace.clone(),
302        ))
303    }
304
305    pub(in crate::ui) fn clamp(&mut self, rows: usize) {
306        self.selection
307            .select((rows > 0).then(|| self.selection.selected().unwrap_or(0).min(rows - 1)));
308    }
309}

311const fn cycle(index: usize, count: usize, backward: bool) -> usize {
312    if backward {
313        (index + count - 1) % count
314    } else {
315        (index + 1) % count
316    }
317}

319/// The rows for `attachments`, each matched to its instance in `listing`.
320/// `None` attachments is a session created before runtimes: `global` and
321/// `local`.
322pub(in crate::ui) fn runtime_rows(
323    attachments: Option<&[RuntimeAttachment]>,
324    listing: Option<&RuntimesListResult>,
325) -> Vec<RuntimeRow> {
326    let legacy;
327    let attachments = match attachments {
328        Some(attachments) => attachments,
329        None => {
330            legacy = legacy_attachments(listing);
331            &legacy
332        }
333    };
334    attachments
335        .iter()
336        .map(|attachment| RuntimeRow {
337            attachment: attachment.clone(),
338            instance: listing.and_then(|listing| {
339                listing
340                    .instances
341                    .iter()
342                    .find(|instance| {
343                        instance.runtime_id == attachment.runtime_id
344                            && instance.capability == attachment.capability
345                            && instance.workspace == attachment.workspace
346                    })
347                    .cloned()
348            }),
349        })
350        .collect()
351}

353/// The rows for a new-session draft: each entry with its live instance from
354/// `listing`, else the one its open request answered. Until the instance
355/// reports its attachment name and kind, the row shows the runtime ID and the
356/// capability.
357pub(in crate::ui) fn draft_runtime_rows(
358    entries: &[DraftRuntime],
359    listing: Option<&RuntimesListResult>,
360) -> Vec<RuntimeRow> {
361    entries
362        .iter()
363        .filter_map(|entry| {
364            let capability = match (&entry.request.capability, &entry.opened) {
365                (Some(capability), _) => capability.clone(),
366                (None, Some(Ok(instance))) => instance.capability.clone(),
367                (None, _) => cookie_agent_protocol::default_capability(&entry.request.runtime_id),
368            };
369            let workspace = entry.workspace().map(str::to_owned);
370            let live = listing.and_then(|listing| {
371                listing
372                    .instances
373                    .iter()
374                    .find(|instance| {
375                        instance.runtime_id == entry.request.runtime_id
376                            && instance.capability == capability
377                            && instance.workspace == workspace
378                    })
379                    .cloned()
380            });
381            let instance = live.or_else(|| match &entry.opened {
382                Some(Ok(instance)) => Some(instance.clone()),
383                Some(Err(message)) => Some(RuntimeInstanceInfo {
384                    runtime_id: entry.request.runtime_id.clone(),
385                    capability: capability.clone(),
386                    attachment_name: None,
387                    kind: None,
388                    workspace: workspace.clone(),
389                    platform: None,
390                    status: RuntimeInstanceStatus::Failed,
391                    message: Some(message.clone()),
392                }),
393                None => None,
394            });
395            Some(RuntimeRow {
396                attachment: RuntimeAttachment {
397                    name: match instance
398                        .as_ref()
399                        .and_then(|instance| instance.attachment_name.clone())
400                    {
401                        Some(name) => name,
402                        None => AttachmentName::new(entry.request.runtime_id.as_str()).ok()?,
403                    },
404                    kind: match instance.as_ref().and_then(|instance| instance.kind.clone()) {
405                        Some(kind) => kind,
406                        None => RuntimeKind::new(capability.as_str()).ok()?,
407                    },
408                    runtime_id: entry.request.runtime_id.clone(),
409                    capability,
410                    workspace,
411                },
412                instance,
413            })
414        })
415        .collect()
416}

418/// What a session created before runtimes is attached to.
419fn legacy_attachments(listing: Option<&RuntimesListResult>) -> Vec<RuntimeAttachment> {
420    [RuntimeId::GLOBAL, RuntimeId::LOCAL]
421        .into_iter()
422        .filter_map(|id| {
423            let runtime_id = RuntimeId::new(id).ok()?;
424            let instance = listing.and_then(|listing| {
425                listing
426                    .instances
427                    .iter()
428                    .find(|instance| instance.runtime_id == runtime_id)
429            });
430            Some(RuntimeAttachment {
431                capability: cookie_agent_protocol::default_capability(&runtime_id),
432                name: instance
433                    .and_then(|instance| instance.attachment_name.clone())
434                    .or_else(|| cookie_agent_protocol::AttachmentName::new(id).ok())?,
435                // Until the instance reports its kind, label the two built-ins
436                // with the kinds they declare.
437                kind: match instance.and_then(|instance| instance.kind.clone()) {
438                    Some(kind) => kind,
439                    None => RuntimeKind::new(if id == RuntimeId::GLOBAL {
440                        "global"
441                    } else {
442                        "coding"
443                    })
444                    .ok()?,
445                },
446                workspace: instance.and_then(|instance| instance.workspace.clone()),
447                runtime_id,
448            })
449        })
450        .collect()
451}

453const fn status_label(status: Option<RuntimeInstanceStatus>) -> &'static str {
454    match status {
455        Some(RuntimeInstanceStatus::Starting) => "starting",
456        Some(RuntimeInstanceStatus::Ready) => "ready",
457        Some(RuntimeInstanceStatus::Reconnecting) => "reconnecting",
458        Some(RuntimeInstanceStatus::Failed) => "failed",
459        Some(RuntimeInstanceStatus::Stopped) => "stopped",
460        None => "not running",
461    }
462}

464pub(in crate::ui) fn render_runtimes(
465    frame: &mut Frame,
466    area: Rect,
467    panel: &mut RuntimePanel,
468    rows: &[RuntimeRow],
469    theme: &Theme,
470) {
471    paint_panel(frame, area, theme);
472    if panel.form.is_some() {
473        render_attach_form(frame, area, panel, theme);
474        return;
475    }
476    panel.clamp(rows.len());
477    let chunks = Layout::default()
478        .direction(Direction::Vertical)
479        .constraints([Constraint::Percentage(55), Constraint::Percentage(45)])
480        .split(area);
481    let items = rows
482        .iter()
483        .map(|row| {
484            ListItem::new(format!(
485                "{}  {}/{}  {}  {}",
486                row.attachment.name,
487                row.attachment.runtime_id,
488                row.attachment.capability,
489                row.attachment
490                    .workspace
491                    .as_deref()
492                    .unwrap_or("(no workspace)"),
493                status_label(row.instance.as_ref().map(|instance| instance.status)),
494            ))
495        })
496        .collect::<Vec<_>>();
497    frame.render_stateful_widget(
498        List::new(items)
499            .highlight_symbol("> ")
500            .highlight_style(theme.selected())
501            .block(
502                crate::ui::panel_block()
503                    .border_style(theme.panel_border())
504                    .title(crate::ui::panel_title("Attached runtimes"))
505                    .title_bottom(crate::ui::panel_title(
506                        Line::from(Span::styled(
507                            "a attach | d detach | r retry | esc close",
508                            theme.internal(),
509                        ))
510                        .right_aligned(),
511                    )),
512            ),
513        chunks[0],
514        &mut panel.selection,
515    );
516    let detail = panel
517        .selection
518        .selected()
519        .and_then(|index| rows.get(index))
520        .map_or_else(
521            || {
522                if panel.listing.is_none() {
523                    "Loading runtimes...".to_owned()
524                } else {
525                    "No runtimes attached. Press a to attach one.".to_owned()
526                }
527            },
528            row_detail,
529        );
530    frame.render_widget(
531        Paragraph::new(detail).wrap(Wrap { trim: false }).block(
532            crate::ui::panel_block()
533                .border_style(theme.panel_border())
534                .title(crate::ui::panel_title("Runtime details")),
535        ),
536        chunks[1],
537    );
538}

540fn row_detail(row: &RuntimeRow) -> String {
541    let attachment = &row.attachment;
542    let mut text = format!(
543        "{}\nruntime: {}\ncapability: {} ({})",
544        attachment.name,
545        attachment.runtime_id,
546        attachment.capability,
547        attachment.kind.as_str()
548    );
549    if let Some(workspace) = &attachment.workspace {
550        text.push_str(&format!("\nworkspace: {workspace}"));
551    }
552    text.push_str(&format!(
553        "\nstatus: {}",
554        status_label(row.instance.as_ref().map(|instance| instance.status))
555    ));
556    if let Some(platform) = row
557        .instance
558        .as_ref()
559        .and_then(|instance| instance.platform.as_deref())
560    {
561        text.push_str(&format!("\nplatform: {platform}"));
562    }
563    if let Some(message) = row
564        .instance
565        .as_ref()
566        .and_then(|instance| instance.message.as_deref())
567    {
568        text.push_str("\n\n");
569        text.push_str(message);
570    }
571    text
572}

574fn render_attach_form(frame: &mut Frame, area: Rect, panel: &mut RuntimePanel, theme: &Theme) {
575    let choices = panel.choices();
576    let total = panel
577        .runtimes()
578        .iter()
579        .map(|runtime| runtime.capabilities.len().max(1))
580        .sum::<usize>();
581    let Some(form) = panel.form.as_mut() else {
582        return;
583    };
584    if let Some(choice) = form.chosen.clone() {
585        render_attach_parameters(frame, area, &choice, form, theme);
586        return;
587    }
588    let block = |title: String, hint: &'static str| {
589        crate::ui::panel_block()
590            .border_style(theme.panel_border())
591            .title(crate::ui::panel_title(title))
592            .title_bottom(crate::ui::panel_title(
593                Line::from(Span::styled(hint, theme.internal())).right_aligned(),
594            ))
595    };
596    let chunks = Layout::default()
597        .direction(Direction::Vertical)
598        .constraints([
599            Constraint::Length(3),
600            Constraint::Min(3),
601            Constraint::Length(4),
602        ])
603        .split(area);
604    crate::ui::input::render(
605        frame,
606        chunks[0],
607        &mut form.search,
608        true,
609        "Search runtimes",
610        Some("Filter…"),
611        theme,
612    );
613    let items = choices
614        .iter()
615        .map(|choice| {
616            let mut spans = vec![Span::styled(choice.label(), theme.body())];
617            match &choice.capability {
618                Some(capability) if !capability.description.is_empty() => spans.push(Span::styled(
619                    format!("  {}", capability.description),
620                    theme.muted(),
621                )),
622                Some(_) => {}
623                None => spans.push(Span::styled(
624                    "  capabilities known once it connects",
625                    theme.muted(),
626                )),
627            }
628            if !choice.connected {
629                spans.push(Span::styled("  (not connected)", theme.muted()));
630            }
631            ListItem::new(Line::from(spans))
632        })
633        .collect::<Vec<_>>();
634    frame.render_stateful_widget(
635        List::new(items)
636            .highlight_symbol("> ")
637            .highlight_style(theme.selected())
638            .block(block(
639                format!("Attach runtime ({}/{total})", choices.len()),
640                "type to filter | arrows choose | enter next | esc cancel",
641            )),
642        chunks[1],
643        &mut form.selection,
644    );
645    let detail = form
646        .selection
647        .selected()
648        .and_then(|index| choices.get(index))
649        .map_or_else(
650            || "No runtime matches the search.".to_owned(),
651            |choice| {
652                let mut text = match &choice.capability {
653                    Some(capability) => format!(
654                        "{} ({}){}",
655                        choice.label(),
656                        capability.kind.as_str(),
657                        if capability.takes_workspace {
658                            ", takes a workspace"
659                        } else {
660                            ", takes no workspace"
661                        }
662                    ),
663                    None => format!(
664                        "{}: attaching connects it and takes its only capability",
665                        choice.runtime_id
666                    ),
667                };
668                if let Some(message) = choice.message.as_deref().filter(|_| !choice.connected) {
669                    text.push_str(&format!("\nLast connection failed: {message}"));
670                }
671                text
672            },
673        );
674    frame.render_widget(
675        Paragraph::new(detail).wrap(Wrap { trim: false }).block(
676            crate::ui::panel_block()
677                .border_style(theme.panel_border())
678                .title(crate::ui::panel_title("Details")),
679        ),
680        chunks[2],
681    );
682}

684fn render_attach_parameters(
685    frame: &mut Frame,
686    area: Rect,
687    choice: &RuntimeChoice,
688    form: &mut RuntimeAttachForm,
689    theme: &Theme,
690) {
691    let mut lines = vec![Line::from(vec![
692        Span::styled("Runtime: ", theme.muted()),
693        Span::styled(choice.runtime_id.to_string(), theme.body()),
694    ])];
695    lines.push(match &choice.capability {
696        Some(capability) => Line::from(vec![
697            Span::styled("Capability: ", theme.muted()),
698            Span::styled(
699                if capability.description.is_empty() {
700                    capability.name.to_string()
701                } else {
702                    format!("{} - {}", capability.name, capability.description)
703                },
704                theme.body(),
705            ),
706        ]),
707        None => Line::from(Span::styled(
708            "Its only capability, known once it connects; the workspace is optional.",
709            theme.muted(),
710        )),
711    });
712    if let Some(message) = choice.message.as_deref().filter(|_| !choice.connected) {
713        lines.push(Line::from(Span::styled(
714            format!("Last connection failed: {message}"),
715            theme.muted(),
716        )));
717    }
718    let chunks = Layout::default()
719        .direction(Direction::Vertical)
720        .constraints([Constraint::Min(4), Constraint::Length(3)])
721        .split(area);
722    frame.render_widget(
723        Paragraph::new(lines).wrap(Wrap { trim: false }).block(
724            crate::ui::panel_block()
725                .border_style(theme.panel_border())
726                .title(crate::ui::panel_title(format!("Attach {}", choice.label())))
727                .title_bottom(crate::ui::panel_title(
728                    Line::from(Span::styled("enter attach | esc back", theme.internal()))
729                        .right_aligned(),
730                )),
731        ),
732        chunks[0],
733    );
734    crate::ui::input::render(
735        frame,
736        chunks[1],
737        &mut form.workspace,
738        true,
739        "Workspace path (on the runtime's machine)",
740        Some("~/project"),
741        theme,
742    );
743}
