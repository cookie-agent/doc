33/// The description `listing` has for `capability` of `runtime_id`.
34fn capability_description(
35    listing: Option<&RuntimesListResult>,
36    runtime_id: &RuntimeId,
37    capability: &CapabilityName,
38) -> Option<String> {
39    listing?
40        .runtimes
41        .iter()
42        .find(|runtime| runtime.runtime_id == *runtime_id)?
43        .capabilities
44        .iter()
45        .find(|offered| offered.name == *capability)
46        .map(|offered| offered.description.clone())
47        .filter(|description| !description.is_empty())
48}

84/// One thing the attach step offers: a capability of a runtime, or a runtime
85/// that has not said what it offers yet, which attaching connects and takes
86/// its only capability of.
87#[derive(Clone, Debug, Eq, PartialEq)]
88pub(in crate::ui) struct RuntimeChoice {
89    pub(in crate::ui) runtime_id: RuntimeId,
90    pub(in crate::ui) capability: Option<RuntimeCapabilityInfo>,
91    pub(in crate::ui) connected: bool,
92    /// Why the runtime last failed to connect.
93    pub(in crate::ui) message: Option<String>,
94}

96impl RuntimeChoice {
97    /// Whether attaching asks for a workspace: the capability takes one, or
98    /// the runtime has not said what it offers yet (then it is optional).
99    pub(in crate::ui) fn asks_workspace(&self) -> bool {
100        self.capability
101            .as_ref()
102            .is_none_or(|capability| capability.takes_workspace)
103    }
104
105    fn label(&self) -> String {
106        match &self.capability {
107            Some(capability) => format!("{}/{}", self.runtime_id, capability.name),
108            None => self.runtime_id.to_string(),
109        }
110    }
111
112    /// Case-insensitive substring match over the runtime ID and the
113    /// capability's name, kind, and description.
114    fn matches(&self, query: &str) -> bool {
115        let query = query.trim().to_lowercase();
116        if query.is_empty() {
117            return true;
118        }
119        let mut haystack = self.runtime_id.to_string();
120        if let Some(capability) = &self.capability {
121            haystack.push(' ');
122            haystack.push_str(capability.name.as_str());
123            haystack.push(' ');
124            haystack.push_str(capability.kind.as_str());
125            haystack.push(' ');
126            haystack.push_str(&capability.description);
127        }
128        haystack.to_lowercase().contains(&query)
129    }
130}

342    /// The machines view's rows: each machine, then its connected
343    /// runtimes and the runtimes pinned to it that are not connected.
344    pub(in crate::ui) fn machine_entries(&self) -> Vec<MachineEntry> {
345        let Some(Ok(listing)) = &self.machines else {
346            return Vec::new();
347        };
348        listing
349            .machines
350            .iter()
351            .flat_map(|machine| {
352                let mut entries = vec![MachineEntry::Machine(machine.name.clone())];
353                for runtime in &machine.connected {
354                    entries.push(MachineEntry::Runtime {
355                        machine: machine.name.clone(),
356                        runtime: runtime.runtime_id.clone(),
357                        assigned_by: Some(runtime.assigned_by),
358                        pinned: machine.pinned.contains(&runtime.runtime_id),
359                    });
360                }
361                for runtime in machine.pinned.iter().filter(|pinned| {
362                    !machine
363                        .connected
364                        .iter()
365                        .any(|connected| connected.runtime_id == **pinned)
366                }) {
367                    entries.push(MachineEntry::Runtime {
368                        machine: machine.name.clone(),
369                        runtime: runtime.clone(),
370                        assigned_by: None,
371                        pinned: true,
372                    });
373                }
374                entries
375            })
376            .collect()
377    }

396    /// Starts `action` on the selected row: a request ready to send, or
397    /// `None` when the form now asks for a machine name.
398    pub(in crate::ui) fn begin_machine_action(
399        &mut self,
400        action: MachineAction,
401    ) -> Result<Option<MachineRequest>, String> {
402        let entry = self
403            .selected_machine_entry()
404            .ok_or("select a machine or a runtime first")?;
405        if !entry.actions().contains(&action) {
406            return Err(match (&entry, action) {
407                (MachineEntry::Machine(_), _) => format!("select a runtime to {}", action.label()),
408                (MachineEntry::Runtime { .. }, MachineAction::Unpin) => {
409                    "only a pinned runtime can be unpinned".to_owned()
410                }
411                (MachineEntry::Runtime { .. }, _) => {
412                    format!("select a machine to {}", action.label())
413                }
414            });
415        }
416        if action.asks_name() {
417            self.machine_form = Some(MachineForm {
418                action,
419                entry,
420                input: InputState::default(),
421            });
422            return Ok(None);
423        }
424        Ok(Some(match (action, entry) {
425            (MachineAction::Forget, MachineEntry::Machine(machine)) => {
426                MachineRequest::Forget { machine }
427            }
428            (MachineAction::Unpin, MachineEntry::Runtime { runtime, .. }) => {
429                MachineRequest::Unassign { runtime }
430            }
431            _ => return Err("nothing to do".into()),
432        }))
433    }

435    /// The request the open form describes.
436    pub(in crate::ui) fn submit_machine_form(&mut self) -> Result<MachineRequest, String> {
437        let form = self
438            .machine_form
439            .as_ref()
440            .ok_or("no machine change in progress")?;
441        let typed = form.input.as_str().trim();
442        let name = MachineName::new(typed).map_err(|_| {
443            format!(
444                "`{typed}` is not a machine name: a lowercase letter, then up to 31 lowercase \
445                 letters, digits, `_`, or `-`"
446            )
447        })?;
448        let request = match (&form.action, &form.entry) {
449            (MachineAction::Rename, MachineEntry::Machine(machine)) => MachineRequest::Rename {
450                machine: machine.clone(),
451                name,
452            },
453            (MachineAction::Merge, MachineEntry::Machine(machine)) => MachineRequest::Merge {
454                from: machine.clone(),
455                into: name,
456            },
457            (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => {
458                MachineRequest::Assign {
459                    runtime: runtime.clone(),
460                    machine: name,
461                }
462            }
463            _ => return Err("nothing to do".into()),
464        };
465        self.machine_form = None;
466        Ok(request)
467    }

487    /// Every runtime's capabilities, in listing order, that match the
488    /// search; a runtime that has not said what it offers is one choice.
489    pub(in crate::ui) fn choices(&self) -> Vec<RuntimeChoice> {
490        let query = self.form.as_ref().map_or("", |form| form.search.as_str());
491        self.runtimes()
492            .iter()
493            .flat_map(|runtime| {
494                let choice = |capability| RuntimeChoice {
495                    runtime_id: runtime.runtime_id.clone(),
496                    capability,
497                    connected: runtime.connected,
498                    message: runtime.message.clone(),
499                };
500                if runtime.capabilities.is_empty() {
501                    vec![choice(None)]
502                } else {
503                    runtime
504                        .capabilities
505                        .iter()
506                        .cloned()
507                        .map(|capability| choice(Some(capability)))
508                        .collect()
509                }
510            })
511            .filter(|choice| choice.matches(query))
512            .collect()
513    }

534    /// Takes the highlighted choice: on to its parameters when it needs
535    /// any, else straight to attaching it.
536    pub(in crate::ui) fn choose(&mut self) -> Result<AttachStep, String> {
537        let choices = self.choices();
538        let form = self.form.as_mut().ok_or("no attach in progress")?;
539        let choice = form
540            .selection
541            .selected()
542            .and_then(|index| choices.get(index))
543            .ok_or("no runtime matches the search")?
544            .clone();
545        if choice.asks_workspace() {
546            form.workspace = InputState::default();
547            form.chosen = Some(choice);
548            return Ok(AttachStep::Parameters);
549        }
550        Ok(AttachStep::Attach(
551            choice.runtime_id,
552            choice.capability.map(|capability| capability.name),
553            None,
554        ))
555    }

566    /// The attach request the parameters describe.
567    pub(in crate::ui) fn attach_request(
568        &self,
569    ) -> Result<(RuntimeId, Option<CapabilityName>, Option<String>), String> {
570        let form = self.form.as_ref().ok_or("no attach in progress")?;
571        let choice = form.chosen.as_ref().ok_or("choose a runtime")?;
572        let workspace = form.workspace.as_str().trim();
573        let workspace = match (&choice.capability, workspace.is_empty()) {
574            (Some(capability), true) => {
575                return Err(format!(
576                    "capability `{}` of runtime `{}` needs a workspace path",
577                    capability.name, choice.runtime_id
578                ));
579            }
580            (None, true) => None,
581            (_, false) => Some(workspace.to_owned()),
582        };
583        Ok((
584            choice.runtime_id.clone(),
585            choice
586                .capability
587                .as_ref()
588                .map(|capability| capability.name.clone()),
589            workspace,
590        ))
591    }

593    /// The `runtimes.reconnect` request that retries `row`'s instance: its
594    /// workspace goes along exactly when it has one.
595    pub(in crate::ui) fn reconnect_request(
596        &self,
597        row: &RuntimeRow,
598    ) -> Result<(RuntimeId, CapabilityName, Option<String>), String> {
599        if !self
600            .runtimes()
601            .iter()
602            .any(|runtime| runtime.runtime_id == row.attachment.runtime_id)
603        {
604            return Err(format!(
605                "runtime `{}` is not a known runtime",
606                row.attachment.runtime_id
607            ));
608        }
609        Ok((
610            row.attachment.runtime_id.clone(),
611            row.attachment.capability.clone(),
612            row.attachment.workspace.clone(),
613        ))
614    }

622const fn cycle(index: usize, count: usize, backward: bool) -> usize {
623    if backward {
624        (index + count - 1) % count
625    } else {
626        (index + 1) % count
627    }

659/// The rows for a new-session draft: each entry with its live instance from
660/// `listing`, else the one its open request answered. Until the instance
661/// reports its attachment name and kind, the row shows the runtime ID and the
662/// capability.
663pub(in crate::ui) fn draft_runtime_rows(
664    entries: &[DraftRuntime],
665    listing: Option<&RuntimesListResult>,
666) -> Vec<RuntimeRow> {
667    entries
668        .iter()
669        .filter_map(|entry| {
670            let capability = match (&entry.request.capability, &entry.opened) {
671                (Some(capability), _) => capability.clone(),
672                (None, Some(Ok(instance))) => instance.capability.clone(),
673                (None, _) => cookie_agent_protocol::default_capability(&entry.request.runtime_id),
674            };
675            let workspace = entry.workspace().map(str::to_owned);
676            let live = listing.and_then(|listing| {
677                listing
678                    .instances
679                    .iter()
680                    .find(|instance| {
681                        instance.runtime_id == entry.request.runtime_id
682                            && instance.capability == capability
683                            && instance.workspace == workspace
684                    })
685                    .cloned()
686            });
687            let instance = live.or_else(|| match &entry.opened {
688                Some(Ok(instance)) => Some(instance.clone()),
689                Some(Err(message)) => Some(RuntimeInstanceInfo {
690                    runtime_id: entry.request.runtime_id.clone(),
691                    capability: capability.clone(),
692                    attachment_name: None,
693                    kind: None,
694                    workspace: workspace.clone(),
695                    platform: None,
696                    status: RuntimeInstanceStatus::Failed,
697                    message: Some(message.clone()),
698                }),
699                None => None,
700            });
701            Some(RuntimeRow {
702                description: capability_description(
703                    listing,
704                    &entry.request.runtime_id,
705                    &capability,
706                ),
707                attachment: RuntimeAttachment {
708                    name: match instance
709                        .as_ref()
710                        .and_then(|instance| instance.attachment_name.clone())
711                    {
712                        Some(name) => name,
713                        None => AttachmentName::new(entry.request.runtime_id.as_str()).ok()?,
714                    },
715                    kind: match instance.as_ref().and_then(|instance| instance.kind.clone()) {
716                        Some(kind) => kind,
717                        None => RuntimeKind::new(capability.as_str()).ok()?,
718                    },
719                    runtime_id: entry.request.runtime_id.clone(),
720                    capability,
721                    workspace,
722                },
723                instance,
724            })
725        })
726        .collect()
727}

729const fn status_label(status: Option<RuntimeInstanceStatus>) -> &'static str {
730    match status {
731        Some(RuntimeInstanceStatus::Starting) => "starting",
732        Some(RuntimeInstanceStatus::Ready) => "ready",
733        Some(RuntimeInstanceStatus::Reconnecting) => "reconnecting",
734        Some(RuntimeInstanceStatus::Failed) => "failed",
735        Some(RuntimeInstanceStatus::Stopped) => "stopped",
736        Some(RuntimeInstanceStatus::Unknown) => "unknown",
737        None => "not running",
738    }
739}

741pub(in crate::ui) fn render_runtimes(
742    frame: &mut Frame,
743    area: Rect,
744    panel: &mut RuntimePanel,
745    rows: &[RuntimeRow],
746    theme: &Theme,
747) -> RuntimeHits {
748    let mut hits = RuntimeHits::default();
749    paint_panel(frame, area, theme);
750    if panel.form.is_some() {
751        render_attach_form(frame, area, panel, theme);

755        .direction(Direction::Vertical)

766/// The view tabs, each a click target while no form is open.
767fn render_tabs(
768    frame: &mut Frame,
769    area: Rect,
770    panel: &RuntimePanel,
771    theme: &Theme,
772    hits: &mut RuntimeHits,
773) {
774    let mut spans = Vec::new();
775    let mut x = area.x;
776    for view in RuntimeView::ALL {
777        let label = view.label();
778        let width = u16::try_from(label.len()).unwrap_or(u16::MAX);
779        let style = if view == panel.view {
780            theme.selected()
781        } else {
782            theme.link()
783        };
784        spans.push(Span::styled(label, style));
785        spans.push(Span::raw(" "));
786        if panel.machine_form.is_none() && x.saturating_add(width) <= area.right() {
787            hits.tabs.push((Rect::new(x, area.y, width, 1), view));
788        }
789        x = x.saturating_add(width + 1);
790    }
791    spans.push(Span::styled("m switch", theme.internal()));
792    frame.render_widget(Paragraph::new(Line::from(spans)), area);
793}

870/// The machines view: each machine with its runtimes, the selected row's
871/// details, the actions it offers, and the form an action that needs a
872/// machine name opens.
873fn render_machines(
874    frame: &mut Frame,
875    area: Rect,
876    panel: &mut RuntimePanel,
877    theme: &Theme,
878    hits: &mut RuntimeHits,
879) {
880    let entries = panel.machine_entries();
881    let form_height = if panel.machine_form.is_some() { 3 } else { 0 };
882    let [list_area, detail_area, actions_area, form_area] = Layout::default()
883        .direction(Direction::Vertical)
884        .constraints([
885            Constraint::Min(3),
886            Constraint::Length(6),
887            Constraint::Length(1),
888            Constraint::Length(form_height),
889        ])
890        .areas(area);
891    let machines = match &panel.machines {
892        Some(Ok(listing)) => listing.machines.as_slice(),
893        _ => &[],
894    };
895    let items = entries
896        .iter()
897        .map(|entry| ListItem::new(machine_entry_line(entry, machines, theme)))
898        .collect::<Vec<_>>();
899    let block = crate::ui::panel_block()
900        .border_style(theme.panel_border())
901        .title(crate::ui::panel_title("Machines"))
902        .title_bottom(crate::ui::panel_title(
903            Line::from(Span::styled(
904                "n rename | v move | u unpin | g merge | f forget | m attachments | esc close",
905                theme.internal(),
906            ))
907            .right_aligned(),
908        ));
909    let inner = block.inner(list_area);
910    frame.render_stateful_widget(
911        List::new(items)
912            .highlight_symbol("> ")
913            .highlight_style(theme.selected())
914            .block(block),
915        list_area,
916        &mut panel.machine_selection,
917    );
918    if panel.machine_form.is_none() {
919        let offset = panel.machine_selection.offset();
920        for (line, index) in (offset..entries.len())
921            .take(usize::from(inner.height))
922            .enumerate()
923        {
924            let y = inner.y + u16::try_from(line).unwrap_or(u16::MAX);
925            hits.machine_rows
926                .push((Rect::new(inner.x, y, inner.width, 1), index));
927        }
928    }
929    let selected = panel.selected_machine_entry();
930    let detail = match (&panel.machines, &selected) {
931        (None, _) => "Loading machines...".to_owned(),
932        (Some(Err(error)), _) => format!("Machines are unavailable: {error}"),
933        (Some(Ok(_)), None) => "No machines yet.".to_owned(),
934        (Some(Ok(_)), Some(entry)) => machine_entry_detail(entry, machines),
935    };
936    frame.render_widget(
937        Paragraph::new(detail).wrap(Wrap { trim: false }).block(
938            crate::ui::panel_block()
939                .border_style(theme.panel_border())
940                .title(crate::ui::panel_title("Details")),
941        ),
942        detail_area,
943    );
944    let mut spans = Vec::new();
945    let mut x = actions_area.x;
946    for action in selected.iter().flat_map(MachineEntry::actions) {
947        let label = format!(" {} ({}) ", action.label(), action.key());
948        let width = u16::try_from(label.len()).unwrap_or(u16::MAX);
949        if panel.machine_form.is_none() && x.saturating_add(width) <= actions_area.right() {
950            hits.machine_actions
951                .push((Rect::new(x, actions_area.y, width, 1), action));
952        }
953        spans.push(Span::styled(label, theme.link()));
954        spans.push(Span::raw(" "));
955        x = x.saturating_add(width + 1);
956    }
957    frame.render_widget(Paragraph::new(Line::from(spans)), actions_area);
958    if let Some(form) = panel.machine_form.as_mut() {
959        let title = match (&form.action, &form.entry) {
960            (MachineAction::Rename, MachineEntry::Machine(machine)) => {
961                format!("New name for {machine}")
962            }
963            (MachineAction::Merge, MachineEntry::Machine(machine)) => {
964                format!("Merge {machine} into (its memories move too)")
965            }
966            (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => {
967                format!("Machine for runtime {runtime} (a new name creates it)")
968            }
969            _ => "Machine name".to_owned(),
970        };
971        crate::ui::input::render(
972            frame,
973            form_area,
974            &mut form.input,
975            true,
976            title,
977            Some("enter apply | esc cancel"),
978            theme,
979        );
980    }
981}

983fn machine_entry_line(
984    entry: &MachineEntry,
985    machines: &[MachineInfo],
986    theme: &Theme,
987) -> Line<'static> {
988    match entry {
989        MachineEntry::Machine(name) => {
990            let machine = machines.iter().find(|machine| machine.name == *name);
991            let mut spans = vec![Span::styled(name.to_string(), theme.body())];
992            if let Some(platform) = machine.and_then(|machine| machine.platform.as_deref()) {
993                spans.push(Span::styled(format!(" ({platform})"), theme.muted()));
994            }
995            let memories = machine.map_or(0, |machine| machine.memories);
996            spans.push(Span::styled(
997                match memories {
998                    1 => "  1 memory".to_owned(),
999                    count => format!("  {count} memories"),
1000                },
1001                theme.muted(),
1002            ));
1003            Line::from(spans)
1004        }
1005        MachineEntry::Runtime {
1006            runtime,
1007            assigned_by,
1008            pinned,
1009            ..
1010        } => Line::from(vec![
1011            Span::styled(format!("  {runtime}"), theme.body()),
1012            Span::styled(
1013                format!("  {}", runtime_placement(*assigned_by, *pinned)),
1014                theme.muted(),
1015            ),
1016        ]),
1017    }
1018}

1020/// Why a runtime row sits under its machine.
1021fn runtime_placement(assigned_by: Option<MachineAssignment>, pinned: bool) -> &'static str {
1022    match (assigned_by, pinned) {
1023        (None, _) => "pinned, not connected",
1024        (Some(MachineAssignment::Pinned), _) | (Some(_), true) => "pinned",
1025        (Some(MachineAssignment::MachineId), false) => "by machine ID",
1026        (Some(MachineAssignment::Unknown), false) => "connected",
1027    }
1028}

1030fn machine_entry_detail(entry: &MachineEntry, machines: &[MachineInfo]) -> String {
1031    match entry {
1032        MachineEntry::Machine(name) => {
1033            let Some(machine) = machines.iter().find(|machine| machine.name == *name) else {
1034                return name.to_string();
1035            };
1036            let mut text = format!(
1037                "{}\nmemories: {}\nconnected runtimes: {}",
1038                machine.name,
1039                machine.memories,
1040                machine.connected.len()
1041            );
1042            if !machine.pinned.is_empty() {
1043                let pinned = machine
1044                    .pinned
1045                    .iter()
1046                    .map(RuntimeId::to_string)
1047                    .collect::<Vec<_>>();
1048                text.push_str(&format!("\npinned: {}", pinned.join(", ")));
1049            }
1050            if !machine.ids.is_empty() {
1051                text.push_str(&format!("\nmachine IDs: {}", machine.ids.join(", ")));
1052            }
1053            text
1054        }
1055        MachineEntry::Runtime {
1056            machine,
1057            runtime,
1058            assigned_by,
1059            pinned,
1060        } => format!(
1061            "runtime {runtime} on machine {machine}: {}",
1062            runtime_placement(*assigned_by, *pinned)
1063        ),
1064    }
1065}

1067fn row_detail(row: &RuntimeRow, machine: Option<&MachineName>) -> String {
1068    let attachment = &row.attachment;
1069    let mut text = format!(
1070        "{}\nruntime: {}\ncapability: {} ({})",
1071        attachment.name,
1072        attachment.runtime_id,
1073        attachment.capability,
1074        attachment.kind.as_str()
1075    );
1076    if let Some(machine) = machine {
1077        text.push_str(&format!("\nmachine: {machine}"));
1078    }
1079    if let Some(description) = &row.description {
1080        text.push_str(&format!("\ndescription: {description}"));
1081    }
1082    if let Some(workspace) = &attachment.workspace {
1083        text.push_str(&format!("\nworkspace: {workspace}"));
1084    }
1085    text.push_str(&format!(
1086        "\nstatus: {}",
1087        status_label(row.instance.as_ref().map(|instance| instance.status))
1088    ));
1089    if let Some(platform) = row
1090        .instance
1091        .as_ref()
1092        .and_then(|instance| instance.platform.as_deref())
1093    {
1094        text.push_str(&format!("\nplatform: {platform}"));
1095    }
1096    if let Some(message) = row
1097        .instance
1098        .as_ref()
1099        .and_then(|instance| instance.message.as_deref())
1100    {
1101        text.push_str("\n\n");
1102        text.push_str(message);
1103    }
1104    text
1105}
