53pub(in crate::ui) fn permission_rows(result: &SessionPermissionGetResult) -> Vec<PermissionRow> {
54    result
55        .permissions
56        .iter()
57        .flat_map(|action| {
58            std::iter::once(PermissionRow {
59                action: action.action,
60                resource: "*".into(),
61                effect: action.effect,
62                source: action.source,
63            })
64            .chain(action.patterns.iter().map(|rule| PermissionRow {
65                action: action.action,
66                resource: rule.resource.as_str().to_owned(),
67                effect: rule.effect,
68                source: rule.source,
69            }))
70        })
71        .collect()
72}

81/// The new-rule form, or the edit form of one session rule.
82pub(in crate::ui) struct PermissionForm {
83    pub(in crate::ui) action: PermissionAction,
84    pub(in crate::ui) pattern: InputState,
85    pub(in crate::ui) effect: PermissionEffect,
86    pub(in crate::ui) focus: PermissionFormFocus,
87    /// The session rule being edited, which a changed pattern replaces.
88    pub(in crate::ui) editing: Option<(PermissionAction, String)>,
89    /// Why the last submit was refused.
90    pub(in crate::ui) error: Option<String>,
91}
92
93impl PermissionForm {
94    pub(in crate::ui) fn new(action: PermissionAction) -> Self {
95        Self {
96            action,
97            pattern: InputState::default(),
98            effect: PermissionEffect::Ask,
99            focus: PermissionFormFocus::Pattern,
100            editing: None,
101            error: None,
102        }
103    }
104
105    pub(in crate::ui) fn edit(row: &PermissionRow) -> Self {
106        let mut pattern = InputState::default();
107        pattern.set_buffer(row.resource.clone());
108        Self {
109            action: row.action,
110            pattern,
111            effect: row.effect,
112            focus: PermissionFormFocus::Pattern,
113            editing: Some((row.action, row.resource.clone())),
114            error: None,
115        }
116    }
117
118    pub(in crate::ui) fn cycle_focus(&mut self, backward: bool) {
119        use PermissionFormFocus::*;
120        self.focus = match (self.focus, backward) {
121            (Action, false) | (Effect, true) => Pattern,
122            (Pattern, false) | (Action, true) => Effect,
123            (Effect, false) | (Pattern, true) => Action,
124        };
125    }
126
127    pub(in crate::ui) fn cycle_action(&mut self, backward: bool) {
128        let index = PERMISSION_ACTIONS
129            .iter()
130            .position(|action| *action == self.action)
131            .unwrap_or(0);
132        let len = PERMISSION_ACTIONS.len();
133        self.action = PERMISSION_ACTIONS[if backward {
134            (index + len - 1) % len
135        } else {
136            (index + 1) % len
137        }];
138    }
139}

153impl PermissionPanel {
154    pub(in crate::ui) fn begin_load(&mut self) {
155        self.result = None;
156        self.form = None;
157        self.notice = None;
158        self.offset = 0;
159        self.selection.select(None);
160    }
161
162    pub(in crate::ui) fn rows(&self) -> Vec<PermissionRow> {
163        self.result
164            .as_ref()
165            .map(permission_rows)
166            .unwrap_or_default()
167    }
168
169    /// Selectable rows: every rule plus the trailing `new rule` row.
170    pub(in crate::ui) fn row_count(&self) -> usize {
171        if self.result.is_some() {
172            self.rows().len() + 1
173        } else {
174            0
175        }
176    }
177
178    pub(in crate::ui) fn selected(&self) -> Option<PermissionRow> {
179        self.selection
180            .selected()
181            .and_then(|index| self.rows().get(index).cloned())
182    }
183
184    pub(in crate::ui) fn add_row_selected(&self) -> bool {
185        self.result.is_some() && self.selection.selected() == Some(self.rows().len())
186    }
187
188    /// Install a fresh result, keeping the selection on the same rule when
189    /// it still exists (a changed effect keeps its row).
190    pub(in crate::ui) fn install(&mut self, result: SessionPermissionGetResult) {
191        let previous = self.selected();
192        self.result = Some(result);
193        let rows = self.rows();
194        let index = previous
195            .and_then(|previous| {
196                rows.iter().position(|row| {
197                    row.action == previous.action && row.resource == previous.resource
198                })
199            })
200            .or_else(|| self.selection.selected())
201            .unwrap_or(0)
202            .min(rows.len());
203        self.selection.select(Some(index));
204    }
205
206    pub(in crate::ui) fn move_selection(&mut self, delta: isize) {
207        let count = self.row_count();
208        if count == 0 {
209            return;
210        }
211        let current = self.selection.selected().unwrap_or(0);
212        let next = current.saturating_add_signed(delta).min(count - 1);
213        self.selection.select(Some(next));
214    }
215
216    pub(in crate::ui) fn select_last(&mut self) {
217        if let Some(last) = self.row_count().checked_sub(1) {
218            self.selection.select(Some(last));
219        }
220    }
221}

223/// One step through allow → ask → deny, stopping at either end the way a
224/// segmented control does.
225pub(in crate::ui) fn step_effect(effect: PermissionEffect, backward: bool) -> PermissionEffect {
226    let index = EFFECTS
227        .iter()
228        .position(|candidate| *candidate == effect)
229        .unwrap_or(1);
230    let next = if backward {
231        index.saturating_sub(1)
232    } else {
233        (index + 1).min(EFFECTS.len() - 1)
234    };
235    EFFECTS[next]
236}

311/// The selected row in plain words: what it matches, what happens, where
312/// it comes from, and what editing it does.
313fn row_explanation(row: &PermissionRow, has_patterns: bool) -> String {
314    let noun = resource_noun(row.action);
315    let matched = if row.resource == "*" {
316        if has_patterns {
317            format!("Other {noun} (no pattern below matches)")
318        } else {
319            format!("All {noun}")
320        }
321    } else {
322        format!("{} matching `{}`", capitalized(noun), row.resource)
323    };
324    let effect = effect_phrase(row.effect);
325    let origin = match row.source {
326        PermissionRuleSource::SessionOverlay => {
327            "A session override, checked before the agent's rules; d removes it, Enter edits it."
328        }
329        PermissionRuleSource::AgentDocument => {
330            "From the agent; changing it adds a session override."
331        }
332        PermissionRuleSource::Default => {
333            "No rule covers this, so it is denied and its tools are hidden; changing it adds a session override."
334        }
335    };
336    format!("{matched} {effect}. {origin}")
337}

339fn capitalized(text: &str) -> String {
340    let mut characters = text.chars();
341    characters.next().map_or_else(String::new, |first| {
342        first.to_uppercase().chain(characters).collect()
343    })
344}

346/// Where the panel's clickable parts landed this frame.
347#[derive(Clone, Debug, Default)]
348pub(in crate::ui) struct PermissionHits {
349    /// (row rect, selectable row index); the last index is `new rule`.
350    pub(in crate::ui) rows: Vec<(Rect, usize)>,
351    /// The selected row's effect segments.
352    pub(in crate::ui) effects: Vec<(Rect, PermissionEffect)>,
353    /// The form's effect segments.
354    pub(in crate::ui) form_effects: Vec<(Rect, PermissionEffect)>,
355    pub(in crate::ui) mode: Option<Rect>,
356}

358const ACTION_COLUMNS: usize = 13;
359const SOURCE_COLUMNS: usize = 9;

361/// The effect column: a glyph and label, or on the selected row the full
362/// segmented control whose segments are returned with their offsets.
363fn effect_cells(
364    effect: PermissionEffect,
365    selected: bool,
366    theme: &Theme,
367) -> (Vec<Span<'static>>, Vec<(usize, usize, PermissionEffect)>) {
368    if !selected {
369        return (
370            vec![Span::styled(
371                format!("{} {}", effect_glyph(effect), effect_label(effect)),
372                effect_style(effect, false, theme),
373            )],
374            Vec::new(),
375        );
376    }
377    let mut spans = Vec::new();
378    let mut segments = Vec::new();
379    let mut column = 0;
380    for (index, candidate) in EFFECTS.into_iter().enumerate() {
381        if index > 0 {
382            spans.push(Span::styled("│", theme.panel_border()));
383            column += 1;
384        }
385        let text = format!(" {} ", effect_label(candidate));
386        let width = text.len();
387        let style = if candidate == effect {
388            effect_style(candidate, true, theme)
389        } else {
390            theme.muted()
391        };
392        spans.push(Span::styled(text, style));
393        segments.push((column, width, candidate));
394        column += width;
395    }
396    (spans, segments)
397}

401/// Render the panel centered in `screen`, sized to its rules, and return
402/// where its clickable parts landed.
403pub(in crate::ui) fn render_permissions(
404    frame: &mut Frame,
405    screen: Rect,
406    panel: &mut PermissionPanel,
407    mode: Option<PermissionMode>,
408    theme: &Theme,
409) -> PermissionHits {
410    let mut hits = PermissionHits::default();
411    let rows = panel.rows();
412    // Sized to the longest pattern (plus its tree glyph and a gap), within
413    // a readable minimum and nine tenths of the screen.
414    let longest_pattern = rows
415        .iter()
416        .map(|row| UnicodeWidthStr::width(row.resource.as_str()))
417        .max()
418        .unwrap_or(0);
419    let natural =
420        4 + ACTION_COLUMNS + (longest_pattern + 4).max(18) + EFFECT_COLUMNS + SOURCE_COLUMNS;
421    let width = u16::try_from(natural).unwrap_or(u16::MAX).clamp(
422        screen.width.min(64),
423        (screen.width.saturating_mul(9) / 10).max(screen.width.min(64)),
424    );
425    let explanation_width = width.saturating_sub(4);
426    let explanation = panel
427        .selected()
428        .map(|row| {
429            let has_patterns = rows
430                .iter()
431                .any(|other| other.action == row.action && other.resource != "*");
432            row_explanation(&row, has_patterns)
433        })
434        .or_else(|| {
435            panel.add_row_selected().then(|| {
436                "Add a session rule: it is checked before the agent's rules and lasts for \
437                 this session and its revert/fork branches."
438                    .to_owned()
439            })
440        });
441    let mut footer = Vec::new();
442    if let Some(notice) = &panel.notice {
443        footer.extend(wrapped_line(
444            Line::styled(notice.clone(), theme.warning()),
445            explanation_width,
446        ));
447    }
448    if let Some(explanation) = explanation {
449        footer.extend(wrapped_line(
450            Line::styled(explanation, theme.muted_text()),
451            explanation_width,
452        ));
453    }
454    // The form is five rows under a one-row gap.
455    let form_height = if panel.form.is_some() { 6 } else { 0 };
456    let table_rows = panel.row_count().max(1);
457    // Border, blank, header, table, blank, explanation, form, border.
458    let desired = table_rows + footer.len() + form_height + 5;
459    let height = u16::try_from(desired)
460        .unwrap_or(u16::MAX)
461        .min(screen.height.saturating_mul(9) / 10)
462        .min(screen.height);
463    let area = Rect::new(
464        screen.x + screen.width.saturating_sub(width) / 2,
465        screen.y + screen.height.saturating_sub(height) / 2,
466        width,
467        height,
468    );
469    paint_panel(frame, area, theme);
470
471    let hints = if panel.form.is_some() {
472        "tab field · ←→ change · ⏎ save · esc cancel"
473    } else {
474        "↑↓ select · ←→ effect · n new · d remove · m mode · esc close"
475    };
476    let mut block = crate::ui::panel_block()
477        .border_style(theme.panel_border())
478        .title(crate::ui::panel_title(Span::styled(
479            "Permissions",
480            theme.heading(),
481        )))
482        .style(theme.panel());
483    let mode_text = mode.map(|mode| format!("mode {} · m", permission_mode_label(mode)));
484    if let Some(text) = &mode_text {
485        block = block.title(crate::ui::panel_title(
486            Line::from(Span::styled(text.clone(), theme.muted())).right_aligned(),
487        ));
488    }
489    if usize::from(area.width) > UnicodeWidthStr::width(hints) + 4 {
490        block = block.title_bottom(crate::ui::panel_title(
491            Line::from(Span::styled(hints, theme.internal())).right_aligned(),
492        ));
493    }
494    frame.render_widget(block, area);
495    if let Some(text) = mode_text {
496        let text_width = UnicodeWidthStr::width(text.as_str()) as u16;
497        let x = area.right().saturating_sub(2 + text_width);
498        hits.mode = Some(Rect::new(x, area.y, text_width, 1));
499    }
500
501    let inner = Rect::new(
502        area.x + 2,
503        area.y + 1,
504        area.width.saturating_sub(4),
505        area.height.saturating_sub(2),
506    );
507    if inner.height < 2 {
508        return hits;
509    }
510    if panel.result.is_none() {
511        frame.render_widget(
512            Paragraph::new(Span::styled("Loading permissions…", theme.muted())),
513            Rect::new(inner.x, inner.y + 1, inner.width, 1),
514        );
515        return hits;
516    }
517
518    let pattern_columns = usize::from(inner.width)
519        .saturating_sub(ACTION_COLUMNS + EFFECT_COLUMNS + SOURCE_COLUMNS)
520        .max(8);
521    let header = Line::from(vec![Span::styled(
522        format!(
523            "{:<ACTION_COLUMNS$}{:<pattern_columns$}{:<EFFECT_COLUMNS$}{}",
524            "ACTION", "PATTERN", "EFFECT", "SOURCE"
525        ),
526        theme.internal(),
527    )]);
528    frame.render_widget(
529        Paragraph::new(header),
530        Rect::new(inner.x, inner.y + 1, inner.width, 1),
531    );
532
533    let footer_rows = u16::try_from(footer.len()).unwrap_or(u16::MAX);
534    let table_top = inner.y + 2;
535    let table_height = inner
536        .height
537        .saturating_sub(2 + footer_rows + 1 + form_height as u16);
538    let visible = usize::from(table_height).max(1);
539    let selected = panel.selection.selected().unwrap_or(0);
540    if selected < panel.offset {
541        panel.offset = selected;
542    } else if selected >= panel.offset + visible {
543        panel.offset = selected + 1 - visible;
544    }
545    panel.offset = panel.offset.min(panel.row_count().saturating_sub(visible));
546
547    for (offset, index) in (panel.offset..panel.row_count()).take(visible).enumerate() {
548        let y = table_top + offset as u16;
549        let row_rect = Rect::new(inner.x.saturating_sub(1), y, inner.width + 2, 1);
550        let is_selected = index == selected;
551        if is_selected {
552            let clip = row_rect.intersection(frame.area());
553            frame.buffer_mut().set_style(clip, theme.selected_overlay());
554        }
555        let text_rect = Rect::new(inner.x, y, inner.width, 1);
556        hits.rows.push((row_rect, index));
557        let Some(row) = rows.get(index) else {
558            frame.render_widget(
559                Paragraph::new(Line::from(vec![
560                    Span::styled("+ ", theme.assistant()),
561                    Span::styled("new session rule", theme.assistant()),
562                    Span::styled("  n", theme.internal()),
563                ])),
564                text_rect,
565            );
566            continue;
567        };
568        let first_of_action = index == 0 || rows[index - 1].action != row.action;
569        let last_of_action = rows
570            .get(index + 1)
571            .is_none_or(|next| next.action != row.action);
572        let mut spans = if first_of_action {
573            vec![
574                Span::raw(action_icon(row.action)),
575                Span::raw(" "),
576                Span::styled(
577                    format!(
578                        "{:<width$}",
579                        action_label(row.action),
580                        width = ACTION_COLUMNS - 3
581                    ),
582                    theme.heading(),
583                ),
584            ]
585        } else {
586            vec![Span::raw(" ".repeat(ACTION_COLUMNS))]
587        };
588        // An action's patterns hang off its `*` row as a small tree.
589        let branch = if first_of_action {
590            ""
591        } else if last_of_action {
592            "└ "
593        } else {
594            "├ "
595        };
596        spans.push(Span::styled(branch, theme.panel_border()));
597        let pattern = truncate_with_ellipsis(
598            &row.resource,
599            pattern_columns.saturating_sub(2 + branch.chars().count()),
600        );
601        let pattern_width = UnicodeWidthStr::width(pattern.as_str()) + branch.chars().count();
602        let pattern_style = if row.resource == "*" {
603            theme.muted_text()
604        } else {
605            theme.body()
606        };
607        spans.push(Span::styled(pattern, pattern_style));
608        spans.push(Span::raw(
609            " ".repeat(pattern_columns.saturating_sub(pattern_width)),
610        ));
611        let (effect_spans, segments) = effect_cells(row.effect, is_selected, theme);
612        let effect_width = effect_spans
613            .iter()
614            .map(|span| UnicodeWidthStr::width(span.content.as_ref()))
615            .sum::<usize>();
616        let effect_x = inner.x + (ACTION_COLUMNS + pattern_columns) as u16;
617        hits.effects
618            .extend(segments.into_iter().map(|(column, width, effect)| {
619                (
620                    Rect::new(effect_x + column as u16, y, width as u16, 1),
621                    effect,
622                )
623            }));
624        spans.extend(effect_spans);
625        spans.push(Span::raw(
626            " ".repeat(EFFECT_COLUMNS.saturating_sub(effect_width)),
627        ));
628        spans.push(Span::styled(
629            source_label(row.source),
630            source_style(row.source, theme),
631        ));
632        let mut line = Line::from(spans);
633        if is_selected {
634            for span in &mut line.spans {
635                if span.style.bg.is_none() {
636                    span.style = span.style.patch(theme.selected_overlay());
637                }
638            }
639        }
640        frame.render_widget(Paragraph::new(line), text_rect);
641    }
642    if panel.row_count() > visible {
643        let label = format!(
644            "{}–{}/{}",
645            panel.offset + 1,
646            (panel.offset + visible).min(panel.row_count()),
647            panel.row_count()
648        );
649        // The row range sits in the bottom border, opposite the hints.
650        let label_width = u16::try_from(label.len()).unwrap_or(u16::MAX);
651        frame.render_widget(
652            Paragraph::new(Span::styled(label, theme.internal())),
653            Rect::new(
654                area.x + 2,
655                area.bottom() - 1,
656                label_width.min(inner.width),
657                1,
658            ),
659        );
660    }
661
662    let footer_top = table_top + table_height + 1;
663    for (offset, line) in footer.into_iter().enumerate() {
664        frame.render_widget(
665            Paragraph::new(line),
666            Rect::new(inner.x, footer_top + offset as u16, inner.width, 1),
667        );
668    }
669    if let Some(form) = &mut panel.form {
670        let form_area = Rect::new(
671            inner.x,
672            area.bottom().saturating_sub(form_height as u16),
673            inner.width,
674            form_height as u16 - 1,
675        );
676        hits.form_effects = render_form(frame, form_area, form, theme);
677    }
678    hits
679}

681/// The rule form: action, pattern, and effect on their own rows, the
682/// focused one marked, any validation error in the bottom border.
683fn render_form(
684    frame: &mut Frame,
685    area: Rect,
686    form: &mut PermissionForm,
687    theme: &Theme,
688) -> Vec<(Rect, PermissionEffect)> {
689    let title = if form.editing.is_some() {
690        "Edit session rule"
691    } else {
692        "New session rule"
693    };
694    let mut block = crate::ui::panel_block()
695        .border_style(theme.input_border(true))
696        .title(crate::ui::panel_title(Span::styled(title, theme.heading())))
697        .style(theme.panel());
698    if let Some(error) = &form.error {
699        block = block.title_bottom(crate::ui::panel_title(Span::styled(
700            error.clone(),
701            theme.error(),
702        )));
703    }
704    frame.render_widget(block, area);
705    let inner = Rect::new(
706        area.x + 2,
707        area.y + 1,
708        area.width.saturating_sub(4),
709        area.height.saturating_sub(2),
710    );
711    let label = |text: &'static str, focused: bool| {
712        let style = if focused { theme.user() } else { theme.muted() };
713        Span::styled(
714            format!("{} {text:<9}", if focused { "›" } else { " " }),
715            style,
716        )
717    };
718    let rows = [
719        PermissionFormFocus::Action,
720        PermissionFormFocus::Pattern,
721        PermissionFormFocus::Effect,
722    ];
723    let mut segments = Vec::new();
724    for (offset, field) in rows.into_iter().enumerate() {
725        let y = inner.y + offset as u16;
726        if y >= inner.bottom() {
727            break;
728        }
729        let focused = form.focus == field;
730        let row = Rect::new(inner.x, y, inner.width, 1);
731        let mut spans = vec![label(
732            match field {
733                PermissionFormFocus::Action => "action",
734                PermissionFormFocus::Pattern => "pattern",
735                PermissionFormFocus::Effect => "effect",
736            },
737            focused,
738        )];
739        let label_width = 11;
740        match field {
741            PermissionFormFocus::Action => {
742                let arrows = if focused {
743                    theme.user()
744                } else {
745                    theme.internal()
746                };
747                spans.extend([
748                    Span::styled("‹ ", arrows),
749                    Span::raw(action_icon(form.action)),
750                    Span::raw(" "),
751                    Span::styled(action_label(form.action), theme.heading()),
752                    Span::styled(" ›", arrows),
753                ]);
754            }
755            PermissionFormFocus::Pattern => {
756                let text = form.pattern.as_str();
757                if text.is_empty() {
758                    spans.push(Span::styled(
759                        match form.action {
760                            PermissionAction::Bash => "e.g. git log*",
761                            PermissionAction::Webfetch => "e.g. https://docs.rs/*",
762                            PermissionAction::Delegate | PermissionAction::Message => {
763                                "e.g. explore"
764                            }
765                            _ => "e.g. src/** or *",
766                        },
767                        theme.internal(),
768                    ));
769                } else {
770                    spans.push(Span::styled(text.to_owned(), theme.body()));
771                }
772                if focused {
773                    let cursor = form.pattern.cursor_visual_position(u16::MAX).1;
774                    frame.set_cursor_position((
775                        row.x + label_width + cursor.min(row.width.saturating_sub(label_width + 1)),
776                        y,
777                    ));
778                }
779            }
780            PermissionFormFocus::Effect => {
781                let (effect_spans, effect_segments) = effect_cells(form.effect, true, theme);
782                let x = row.x + label_width;
783                segments.extend(effect_segments.into_iter().map(|(column, width, effect)| {
784                    (Rect::new(x + column as u16, y, width as u16, 1), effect)
785                }));
786                spans.extend(effect_spans);
787            }
788        }
789        frame.render_widget(Paragraph::new(Line::from(spans)), row);
790    }
791    segments
792}
