16/// The agents view of one runtime.
17pub(in crate::ui) struct AgentsView {
18    pub(in crate::ui) runtime: RuntimeId,
19    /// `runtimes.agents.list`, or why it failed; `None` until it answers.
20    pub(in crate::ui) declarations: Option<Result<Vec<RuntimeAgentDeclaration>, String>>,
21    pub(in crate::ui) selection: ListState,
22    pub(in crate::ui) form: Option<AcpAgentForm>,
23}
24
25impl AgentsView {
26    pub(in crate::ui) fn new(runtime: RuntimeId) -> Self {
27        Self {
28            runtime,
29            declarations: None,
30            selection: ListState::default(),
31            form: None,
32        }
33    }
34
35    pub(in crate::ui) fn install(
36        &mut self,
37        declarations: Result<Vec<RuntimeAgentDeclaration>, String>,
38    ) {
39        let count = declarations.as_ref().map_or(0, Vec::len);
40        self.declarations = Some(declarations);
41        self.selection.select((count > 0).then(|| {
42            self.selection
43                .selected()
44                .unwrap_or(0)
45                .min(count.saturating_sub(1))
46        }));
47    }
48
49    pub(in crate::ui) fn selected(&self) -> Option<&RuntimeAgentDeclaration> {
50        let declarations = self.declarations.as_ref()?.as_ref().ok()?;
51        declarations.get(self.selection.selected()?)
52    }
53
54    pub(in crate::ui) fn move_selection(&mut self, up: bool) {
55        let count = self
56            .declarations
57            .as_ref()
58            .and_then(|declarations| declarations.as_ref().ok())
59            .map_or(0, Vec::len);
60        if count == 0 {
61            return;
62        }
63        let selected = self.selection.selected().unwrap_or(0);
64        self.selection.select(Some(if up {
65            selected.saturating_sub(1)
66        } else {
67            (selected + 1).min(count - 1)
68        }));
69    }
70}

72#[derive(Clone, Copy, Debug, Eq, PartialEq)]
73pub(in crate::ui) enum AcpFormFocus {
74    Agent,
75    Command,
76    Environment,
77    Description,
78    Enabled,
79    Submit,
80}
81
82const FOCUS_ORDER: [AcpFormFocus; 6] = [
83    AcpFormFocus::Agent,
84    AcpFormFocus::Command,
85    AcpFormFocus::Environment,
86    AcpFormFocus::Description,
87    AcpFormFocus::Enabled,
88    AcpFormFocus::Submit,
89];

91/// The form of one declaration.
92pub(in crate::ui) struct AcpAgentForm {
93    /// The key of the declaration it edits; a changed key renames it.
94    pub(in crate::ui) editing: Option<AgentId>,
95    agent: InputState,
96    command: InputState,
97    environment: InputState,
98    description: InputState,
99    pub(in crate::ui) enabled: bool,
100    pub(in crate::ui) focus: AcpFormFocus,
101}
102
103fn input(value: &str) -> InputState {
104    let mut input = InputState::default();
105    input.set_buffer(value.to_owned());
106    input
107}
108
109impl AcpAgentForm {
110    pub(in crate::ui) fn new(fields: &AcpAgentFields, editing: Option<AgentId>) -> Self {
111        Self {
112            editing,
113            agent: input(&fields.agent),
114            command: input(&fields.command),
115            environment: input(&fields.environment),
116            description: input(&fields.description),
117            enabled: fields.enabled,
118            focus: AcpFormFocus::Agent,
119        }
120    }
121
122    pub(in crate::ui) fn fields(&self) -> AcpAgentFields {
123        AcpAgentFields {
124            agent: self.agent.as_str().to_owned(),
125            command: self.command.as_str().to_owned(),
126            environment: self.environment.as_str().to_owned(),
127            description: self.description.as_str().to_owned(),
128            enabled: self.enabled,
129        }
130    }
131
132    pub(in crate::ui) fn move_focus(&mut self, backward: bool) {
133        let index = FOCUS_ORDER
134            .iter()
135            .position(|focus| *focus == self.focus)
136            .unwrap_or(0);
137        let count = FOCUS_ORDER.len();
138        self.focus = FOCUS_ORDER[if backward {
139            (index + count - 1) % count
140        } else {
141            (index + 1) % count
142        }];
143    }
144
145    pub(in crate::ui) fn focused_input(&mut self) -> Option<&mut InputState> {
146        match self.focus {
147            AcpFormFocus::Agent => Some(&mut self.agent),
148            AcpFormFocus::Command => Some(&mut self.command),
149            AcpFormFocus::Environment => Some(&mut self.environment),
150            AcpFormFocus::Description => Some(&mut self.description),
151            AcpFormFocus::Enabled | AcpFormFocus::Submit => None,
152        }
153    }
154}

156/// The agents view: the form while one is open, else the declarations and
157/// the selected one's command and documents.
158pub(in crate::ui) fn render_agents(
159    frame: &mut Frame,
160    area: Rect,
161    view: &mut AgentsView,
162    theme: &Theme,
163) {
164    if let Some(form) = view.form.as_ref() {
165        render_form(frame, area, &view.runtime, form, theme);
166        return;
167    }
168    let [list_area, detail_area] = Layout::default()
169        .direction(Direction::Vertical)
170        .constraints([Constraint::Min(3), Constraint::Length(7)])
171        .areas(area);
172    let rows = match &view.declarations {
173        None => vec![ListItem::new(Line::styled(
174            "Loading ACP agents...",
175            theme.muted(),
176        ))],
177        Some(Err(error)) => vec![ListItem::new(Line::styled(
178            format!("ACP agents are unavailable: {error}"),
179            theme.muted(),
180        ))],
181        Some(Ok(declarations)) if declarations.is_empty() => vec![ListItem::new(Line::styled(
182            format!(
183                "[runtimes.{}.acp.agents] declares none; n adds one.",
184                view.runtime
185            ),
186            theme.muted(),
187        ))],
188        Some(Ok(declarations)) => declarations
189            .iter()
190            .map(|declaration| {
191                let state = if !declaration.definition.enabled {
192                    "disabled"
193                } else if declaration.available {
194                    "available"
195                } else {
196                    "unavailable"
197                };
198                ListItem::new(Line::from(vec![
199                    Span::styled(format!("  {}", declaration.agent), theme.body()),
200                    Span::styled(format!("  {state}"), theme.muted()),
201                ]))
202            })
203            .collect(),
204    };
205    frame.render_stateful_widget(
206        List::new(rows)
207            .highlight_symbol("> ")
208            .highlight_style(theme.selected())
209            .block(
210                crate::ui::panel_block()
211                    .border_style(theme.panel_border())
212                    .title(crate::ui::panel_title(format!("ACP agents on {}", view.runtime)))
213                    .title_bottom(crate::ui::panel_title(
214                        Line::from(Span::styled(
215                            "n add | e edit | d remove | space toggle | o document | t template | esc machines",
216                            theme.internal(),
217                        ))
218                        .right_aligned(),
219                    )),
220            ),
221        list_area,
222        &mut view.selection,
223    );
224    let detail = view.selected().map_or_else(String::new, |declaration| {
225        let mut text = format!("command: {}", declaration.definition.command.join(" "));
226        if !declaration.definition.description.is_empty() {
227            text.push_str(&format!(
228                "\ndescription: {}",
229                declaration.definition.description
230            ));
231        }
232        if !declaration.definition.env.is_empty() {
233            let names = declaration
234                .definition
235                .env
236                .keys()
237                .map(String::as_str)
238                .collect::<Vec<_>>();
239            text.push_str(&format!("\nenvironment: {}", names.join(", ")));
240        }
241        text.push_str(&if declaration.documents.is_empty() {
242            "\ndocuments: none yet; o creates one".to_owned()
243        } else {
244            format!("\ndocuments: {}", declaration.documents.join(", "))
245        });
246        text
247    });
248    frame.render_widget(
249        Paragraph::new(detail).wrap(Wrap { trim: false }).block(
250            crate::ui::panel_block()
251                .border_style(theme.panel_border())
252                .title(crate::ui::panel_title("Details")),
253        ),
254        detail_area,
255    );
256}

258fn render_form(
259    frame: &mut Frame,
260    area: Rect,
261    runtime: &RuntimeId,
262    form: &AcpAgentForm,
263    theme: &Theme,
264) {
265    let rows = [
266        (
267            AcpFormFocus::Agent,
268            "Agent ID",
269            form.agent.as_str().to_owned(),
270        ),
271        (
272            AcpFormFocus::Command,
273            "Command JSON array",
274            form.command.as_str().to_owned(),
275        ),
276        (
277            AcpFormFocus::Environment,
278            "Env JSON object",
279            form.environment.as_str().to_owned(),
280        ),
281        (
282            AcpFormFocus::Description,
283            "Description",
284            form.description.as_str().to_owned(),
285        ),
286        (AcpFormFocus::Enabled, "Enabled", form.enabled.to_string()),
287        (AcpFormFocus::Submit, "Submit", "press enter".into()),
288    ];
289    let mut lines = rows
290        .into_iter()
291        .map(|(focus, label, value)| {
292            let marker = if form.focus == focus { ">" } else { " " };
293            Line::from(vec![
294                Span::styled(format!("{marker} {label}: "), theme.muted()),
295                Span::styled(
296                    value,
297                    if form.focus == focus {
298                        theme.selected()
299                    } else {
300                        theme.body()
301                    },
302                ),
303            ])
304        })
305        .collect::<Vec<_>>();
306    if form.editing.is_some() {
307        lines.push(Line::styled(
308            "  A new ID renames the declaration, and its document when no other runtime declares the old ID.",
309            theme.muted(),
310        ));
311    }
312    let title = match &form.editing {
313        Some(agent) => format!("Edit ACP agent {agent} on {runtime}"),
314        None => format!("New ACP agent on {runtime}"),
315    };
316    frame.render_widget(
317        Paragraph::new(lines).wrap(Wrap { trim: false }).block(
318            crate::ui::panel_block()
319                .border_style(theme.panel_border())
320                .title(crate::ui::panel_title(title))
321                .title_bottom(crate::ui::panel_title(
322                    Line::from(Span::styled(
323                        "tab next | space toggle | enter save | esc cancel",
324                        theme.internal(),
325                    ))
326                    .right_aligned(),
327                )),
328        ),
329        area,
330    );
