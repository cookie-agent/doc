25/// Narrowest row where `tool_block_lines` can still keep the assistant gutter.
26pub(super) const TOOL_HEADER_GUTTER_MIN_COLUMNS: u16 = 8;
27
28/// The `"│ "` assistant gutter a header row hangs behind at normal widths.
29pub(super) const TOOL_HEADER_GUTTER_COLUMNS: usize = 2;
30
31/// The full `"[T…] "` role label that replaces the gutter below
32/// `TOOL_HEADER_GUTTER_MIN_COLUMNS`, leaving the row no room for a wide icon.
33pub(super) const TOOL_HEADER_LABEL_COLUMNS: usize = 5;
34
35/// Columns the compact role label may spend on a narrow row. It shortens with
36/// the row and always leaves `reserve` columns for text — at least one, and as
37/// much more as the widest grapheme the row has to carry whole needs — so
38/// neither the label nor its wrapped continuation can exceed the viewport.
39pub(super) fn tool_row_label_columns(width: u16, reserve: usize) -> usize {
40    usize::from(width)
41        .saturating_sub(reserve.max(1))
42        .clamp(1, TOOL_HEADER_LABEL_COLUMNS)
43}
44
45/// The bracketed role label itself, shortened to that budget: `"[T…] "` while it
46/// fits, then `"[T…]"`, `"[T"`, and `"[…"` down to a single column.
47pub(super) fn tool_row_prefix(short: &str, width: u16, reserve: usize) -> String {
48    let room = tool_row_label_columns(width, reserve);
49    let full = format!("[{short}] ");
50    if UnicodeWidthStr::width(full.as_str()) <= room {
51        return full;
52    }
53    crate::ui::app::truncate_with_ellipsis(&format!("[{short}]"), room)
54}

56/// Columns the header text may use at `width`, after the prefix
57/// `tool_block_lines` puts in front of the row. The header never wraps, so the
58/// markers, label, argument, and status suffix share exactly this budget.
59pub(super) fn tool_header_content_width(width: u16) -> usize {
60    let gutter = if width >= TOOL_HEADER_GUTTER_MIN_COLUMNS {
61        TOOL_HEADER_GUTTER_COLUMNS
62    } else {
63        // The header is cut with `truncate_with_ellipsis`, which already counts
64        // display widths, so it only needs a single column to hang its tail on.
65        tool_row_label_columns(width, 1)
66    };
67    usize::from(width).saturating_sub(gutter)
68}

70/// The `{icon} {chevron} ` markers in front of the header title. A row too
71/// narrow for the icon drops it (its role colour and `[T…]` label already say
72/// what ran), then the separating space, and keeps the bare chevron last because
73/// that is the only expand/collapse cue the row has.
74pub(super) fn tool_header_chrome(icon: &str, chevron: char, width: u16) -> String {
75    let room = tool_header_content_width(width);
76    [
77        format!("{icon} {chevron} "),
78        format!("{chevron} "),
79        chevron.to_string(),
80    ]
81    .into_iter()
82    .find(|chrome| UnicodeWidthStr::width(chrome.as_str()) <= room)
83    .unwrap_or_default()
84}

86/// The finished header row and its `{label} {argument}` title, which the
87/// expanded `path:` row compares against the raw path to see whether anything
88/// was lost. Text that does not fit drops out rather than wrapping: the status
89/// suffix goes first (the row style already carries the status), then the
90/// argument, then the label.
91pub(super) fn tool_header_row(
92    tool: &crate::state::ToolCallState,
93    chevron: char,
94    suffix: &str,
95    width: u16,
96) -> (String, String) {
97    let chrome = tool_header_chrome(tool_icon(tool), chevron, width);
98    let content =
99        tool_header_content_width(width).saturating_sub(UnicodeWidthStr::width(chrome.as_str()));
100    let (budget, suffix) = if UnicodeWidthStr::width(suffix) <= content {
101        (content - UnicodeWidthStr::width(suffix), suffix)
102    } else {
103        (content, "")
104    };
105    let title = tool_header_title(tool, budget);
106    // An empty title must not leave the markers' separator space behind.
107    let chrome = if title.is_empty() {
108        chrome.trim_end().to_owned()
109    } else {
110        chrome
111    };
112    (
113        format!("{chrome}{title}{suffix}").trim_end().to_owned(),
114        title,
115    )
116}

118/// Columns the primary argument may occupy inside the header's `budget`: what
119/// the `{label} ` prefix leaves over. Zero means the label alone fills the row,
120/// so the argument must be dropped to stay on one line.
121pub(super) fn header_argument_width(label: &str, budget: usize) -> usize {
122    budget.saturating_sub(UnicodeWidthStr::width(label) + 1)
123}

131pub(super) fn tool_header_title(tool: &crate::state::ToolCallState, budget: usize) -> String {
132    let title = tool.presentation.title.as_str();
133    let argument_width = header_argument_width(title, budget);
134    let Some(argument) = header_argument(tool) else {
135        return crate::ui::app::truncate_with_ellipsis(title, budget);
136    };
137    if argument_width == 0 {
138        // The label alone fills the row (long plugin names at narrow widths):
139        // keep the label legible and drop the argument rather than wrap.
140        return crate::ui::app::truncate_with_ellipsis(title, budget);
141    }
142    format!(
143        "{title} {}",
144        crate::ui::app::truncate_with_ellipsis(&argument, argument_width)
145    )
146}

154/// A compact or expanded tool row inside its owning assistant item. Running
155/// pulses a suffix; terminal failures retain their exact concise markers.
156/// `pending_name` identifies a committed placeholder whose execution has not
157/// started yet: it renders a neutral pending row, never an error.
158pub(super) fn tool_child_layout(
159    state: &SessionState,
160    call_id: Option<cookie_agent_protocol::ToolCallId>,
161    block_key: impl Into<BlockKey>,
162    pending_name: Option<&str>,
163    context: &mut TranscriptRenderContext<'_>,
164) -> ItemLayout {
165    let block_id = match block_key.into() {
166        BlockKey::Call(call) => BlockId::Tool(call),
167        BlockKey::CommittedTool {
168            turn_seq,
169            content_index,
170        } => BlockId::CommittedTool {
171            turn_seq,
172            content_index,
173        },
174    };
175    let is_expanded = context
176        .expanded
177        .is_some_and(|blocks| blocks.contains(&block_id));
178    let tool = call_id.and_then(|call_id| state.tools.get(&call_id));
179    let Some(tool) = tool else {
180        let (role, text) = match pending_name {
181            // The turn committed this call but execution has not published
182            // its start yet; the placeholder links by content index shortly.
183            Some(name) => (
184                Role::ToolRunning,
185                format!("{} ▸ {} · pending", hint_icon(None), name),
186            ),
187            None => (Role::Error, "tool: unavailable payload".to_owned()),
188        };
189        let lines = if pending_name.is_some() {
190            tool_block_lines(
191                role,
192                vec![ToolBodyLine::wrapped(Line::from(text))],
193                context.width,
194                context.theme,
195            )
196            .lines
197        } else {
198            role_block(role, vec![Line::from(text)], context.width, context.theme)
199        };
200        return ItemLayout {
201            regions: vec![BlockRegion {
202                id: block_id,
203                start_line: 0,
204                end_line: lines.len(),
205                header_lines: None,
206                header_gutter: None,
207            }],
208            lines,
209            user_seq: None,
210        };
211    };
212    let (suffix, role) = match tool.status {
213        // The running marker breathes with the animation clock: a resting
214        // ellipsis, then growing dots. Subtle liveness, never busy.
215        ToolStatus::Running => (
216            match context.clock_bucket {
217                0 => " …".to_owned(),
218                dots => format!(" {}", ".".repeat(usize::from(dots))),
219            },
220            Role::ToolRunning,
221        ),
222        ToolStatus::Completed => (String::new(), Role::ToolSuccess),
223        ToolStatus::Failed => (" failed".to_owned(), Role::ToolFailure),
224        ToolStatus::Cancelled => (" cancelled".to_owned(), Role::ToolFailure),
225        ToolStatus::Interrupted => (" interrupted".to_owned(), Role::ToolFailure),
226    };
227    let output_block_id = |section| call_id.map(|call_id| BlockId::ToolOutput { call_id, section });
228    let section_expanded = |section| {
229        output_block_id(section)
230            .is_some_and(|id| context.expanded.is_some_and(|blocks| blocks.contains(&id)))
231    };
232    let has_body = tool.failure.is_some()
233        || (tool.status == ToolStatus::Running && tool.progress.is_some())
234        || !tool.shown_display().is_empty();
235    let any_output_expanded = [
236        ToolOutputSection::Detail,
237        ToolOutputSection::Stdout,
238        ToolOutputSection::Stderr,
239    ]
240    .into_iter()
241    .any(section_expanded);
242    let limits = if any_output_expanded {
243        EXPANDED_TOOL_OUTPUT_LIMITS
244    } else {
245        COLLAPSED_TOOL_OUTPUT_LIMITS
246    };
247    let mut budget = RenderBudget::new(limits);
248    let chevron = if is_expanded { '▾' } else { '▸' };
249    // A title keeps a margin column on each side where the theme has them.
250    let header_width = if panel_margin(context.width) {
251        context.width - 2
252    } else {
253        context.width
254    };
255    // A backgrounded call says so; a running call of a background-capable
256    // tool offers the move at the right end of its title row.
257    let suffix = if tool.backgrounded {
258        format!("{suffix} · background")
259    } else {
260        suffix
261    };
262    // A running call's latest progress message reads as its status; an
263    // expanded call shows it above its output instead.
264    let suffix = match tool.progress.as_deref() {
265        Some(progress) if tool.status == ToolStatus::Running && !is_expanded => {
266            format!(" · {}{suffix}", flatten_header_argument(progress))
267        }
268        _ => suffix,
269    };
270    let button_width = UnicodeWidthStr::width(BACKGROUND_BUTTON);
271    let content_width = tool_header_content_width(header_width);
272    let offer_background = tool.status == ToolStatus::Running
273        && tool.backgroundable
274        && content_width >= button_width + 1 + MIN_TITLE_BESIDE_BUTTON;
275    let row_width = if offer_background {
276        header_width.saturating_sub(u16::try_from(button_width + 1).unwrap_or(u16::MAX))
277    } else {
278        header_width
279    };
280    let (mut header, title) = tool_header_row(tool, chevron, &suffix, row_width);
281    if offer_background {
282        let used = UnicodeWidthStr::width(header.as_str());
283        header.push_str(&" ".repeat(content_width.saturating_sub(used + button_width)));
284        header.push_str(BACKGROUND_BUTTON);
285    }
286    let mut body = Vec::new();
287    if is_expanded {
288        body.extend(expanded_title_lines(
289            tool,
290            ExpandedHeader {
291                row: header,
292                title: &title,
293                chevron,
294                suffix: &suffix,
295            },
296            context.width,
297        ));
298        // The output band keeps a blank row above and below its text so
299        // nothing touches the band's edge.
300        let padded = panel_margin(context.width);
301        // Everything under the title: the call's arguments, then what it
302        // shows.
303        let style = BlockStyle {
304            theme: context.theme,
305            highlighter: context.highlighter,
306            width: panel_text_width(context.width),
307        };
308        let mut panel =
309            argument_section_lines(tool, &style, context.minimum_event_level, !padded, has_body);
310        if has_body {
311            panel.extend(tool_body_lines(
312                tool,
313                &style,
314                section_expanded(ToolOutputSection::Detail),
315                &mut budget,
316            ));
317        }
318        // Every expanded tool is one panel: a grey title block over a lighter
319        // grey output band.
320        for line in &mut body {
321            line.band = Some(ToolBand::Title);
322        }
323        if padded && !panel.is_empty() {
324            panel.insert(0, ToolBodyLine::wrapped(Line::default()));
325            panel.push(ToolBodyLine::wrapped(Line::default()));
326        }
327        for line in &mut panel {
328            line.band = Some(ToolBand::Terminal);
329        }
330        body.extend(panel);
331        // One clear row between the panel and whatever follows it.
332        body.push(ToolBodyLine::wrapped(Line::default()));
333    } else {
334        // Collapsed rows are muted body text; only a failure's status suffix
335        // keeps the error colour, so failed calls still stand out.
336        let line = match header.strip_suffix(suffix.as_str()) {
337            Some(head) if matches!(role, Role::ToolFailure) && !suffix.is_empty() => {
338                Line::from(vec![
339                    Span::raw(head.to_owned()),
340                    Span::styled(suffix.clone(), context.theme.tool_failure()),
341                ])
342            }
343            _ => Line::from(header),
344        };
345        body.push(ToolBodyLine::wrapped(line));
346    }
347    // Collapsed rows are muted body text; expanded blocks are ordinary text
348    // whose header suffix still names a failure.
349    let text_style = if is_expanded {
350        context.theme.body()
351    } else {
352        context.theme.muted_text()
353    };
354    let rendered = tool_block_lines_with(text_style, role, body, context.width, context.theme);
355    let mut regions = vec![BlockRegion {
356        id: block_id,
357        start_line: 0,
358        end_line: rendered.lines.len(),
359        header_lines: Some(rendered.header_lines),
360        header_gutter: Some(header_gutter_columns(&rendered)),
361    }];
362    if let Some(call_id) = call_id {
363        let chrome = rendered.chrome;
364        regions.extend(
365            rendered
366                .actions
367                .into_iter()
368                .map(|(action, start_line, end_line)| BlockRegion {
369                    id: action.block_id(call_id),
370                    start_line,
371                    end_line,
372                    header_lines: None,
373                    // A notice row hangs behind the same gutter as the rows it
374                    // stands in for.
375                    header_gutter: chrome.get(start_line).copied(),
376                }),
377        );
378    }
379    ItemLayout {
380        regions,
381        lines: rendered.lines,
382        user_seq: None,
383    }
384}

392/// Columns a panel row's text gets inside the block's `│ ` gutter and, where
393/// the panel has them, its margins.
394pub(super) fn panel_text_width(width: u16) -> u16 {
395    let margins = if panel_margin(width) { 2 } else { 0 };
396    width.saturating_sub(2 + margins).max(1)
397}

399/// The call's arguments, set apart from what it shows: a blank row, a muted
400/// `arguments` label, the presentation's argument blocks, and a blank row
401/// before any output. A call whose presentation has no argument blocks shows
402/// its raw arguments instead, as diagnostic detail at the `info` filter and
403/// below.
404fn argument_section_lines(
405    tool: &crate::state::ToolCallState,
406    style: &BlockStyle<'_>,
407    minimum_event_level: crate::state::EventLevel,
408    leading_blank: bool,
409    output_follows: bool,
410) -> Vec<ToolBodyLine> {
411    let theme = style.theme;
412    let muted = theme.muted();
413    let limits = RenderLimits {
414        lines: MAX_ARGUMENT_LINES,
415        bytes: MAX_ARGUMENT_BYTES,
416    };
417    let arguments = tool.shown_arguments();
418    let rows = if arguments.is_empty() {
419        // A presentation with argument blocks or a preview is complete as
420        // given; only one with neither falls back to the raw arguments.
421        if minimum_event_level > crate::state::EventLevel::Info
422            || !tool.presentation.arguments.is_empty()
423            || !tool.presentation.preview.is_empty()
424        {
425            return Vec::new();
426        }
427        let raw = serde_json::from_str::<serde_json::Value>(&tool.arguments)
428            .ok()
429            .and_then(|value| serde_json::to_string_pretty(&value).ok())
430            .unwrap_or_else(|| tool.arguments.clone());
431        bounded_safe_display_text(&raw, muted, limits.lines, limits.bytes)
432            .into_iter()
433            .map(ToolBodyLine::wrapped)
434            .collect::<Vec<_>>()
435    } else {
436        let mut budget = RenderBudget::new(limits);
437        let mut sink = LineSink::new(&mut budget, limits);
438        let mut rows = display_rows(&arguments, &[], style, &mut sink);
439        let omitted = sink.omitted();
440        if omitted > 0 {
441            rows.push(ToolBodyLine::wrapped(Line::styled(
442                format!("… {omitted} more lines not shown"),
443                muted,
444            )));
445        }
446        rows
447    };
448    if rows.is_empty() {
449        return Vec::new();
450    }
451    let mut lines = Vec::new();
452    // A padded panel already opens on a blank row.
453    if leading_blank {
454        lines.push(ToolBodyLine::wrapped(Line::default()));
455    }
456    lines.push(ToolBodyLine::wrapped(Line::styled("arguments", muted)));
457    lines.extend(rows);
458    if output_follows {
459        lines.push(ToolBodyLine::wrapped(Line::default()));
460    }
461    lines
462}

472/// An expanded tool's title block: the header row, or, when the collapsed
473/// header had to shorten its title or argument, the whole of them on rows
474/// hanging under where the title starts. Every row is part of the clickable
475/// header.
476fn expanded_title_lines(
477    tool: &crate::state::ToolCallState,
478    header: ExpandedHeader<'_>,
479    width: u16,
480) -> Vec<ToolBodyLine> {
481    let name = flatten_header_argument(tool.presentation.title.as_str());
482    let chrome = tool_header_chrome(tool_icon(tool), header.chevron, width);
483    let full = match header_argument(tool) {
484        Some(argument) => format!("{name} {argument}"),
485        None => name.clone(),
486    };
487    if header.title == full {
488        let indent =
489            UnicodeWidthStr::width(chrome.as_str()) + UnicodeWidthStr::width(name.as_str()) + 1;
490        return vec![ToolBodyLine::title(Line::from(header.row), indent)];
491    }
492    let indent = UnicodeWidthStr::width(chrome.as_str());
493    vec![ToolBodyLine::title(
494        Line::from(format!("{chrome}{full}{}", header.suffix)),
495        indent,
496    )]
497}

523/// Paint one banded row out to the block's full `width`, not its widest row,
524/// so short output still reads as one solid panel. Spans behind the row's
525/// first `gutter_spans` take `tint` or else the band; spans that already carry
526/// a background (the left margin) keep it, and a `padded` row ends on one
527/// plain band column, so a tinted row sits inset in the band. A theme without
528/// band colours gets the same layout, uncoloured.
529pub(super) fn paint_band(
530    line: &mut Line<'static>,
531    gutter_spans: usize,
532    width: u16,
533    background: Option<Color>,
534    tint: Option<Color>,
535    padded: bool,
536) {
537    let fill = usize::from(width).saturating_sub(line.width());
538    let right = usize::from(padded).min(fill);
539    let start = gutter_spans.min(line.spans.len());
540    if fill > right {
541        line.spans
542            .push(Span::raw(crate::markdown::spaces(fill - right)));
543    }
544    if let Some(row) = tint.or(background) {
545        for span in &mut line.spans[start..] {
546            if span.style.bg.is_none() {
547                span.style = span.style.bg(row);
548            }
549        }
550    }
551    if right > 0 {
552        let mut margin = Span::raw(crate::markdown::spaces(right));
553        if let Some(background) = background {
554            margin.style = margin.style.bg(background);
555        }
556        line.spans.push(margin);
557    }
558}

560/// The one-column left margin behind a panel row's `│ ` gutter. It wears the
561/// gutter's own style, which is how copying recognises it as chrome in every
562/// theme, plus the band's background when the row sits on one.
563pub(super) fn margin_span(theme: &Theme, background: Option<Color>) -> Span<'static> {
564    let mut style = theme.assistant();
565    if let Some(background) = background {
566        style = style.bg(background);
567    }
568    Span::styled(" ", style)
569}

603impl ToolRowAction {
604    /// The clickable region the row becomes in `call_id`'s block.
605    pub(super) fn block_id(self, call_id: cookie_agent_protocol::ToolCallId) -> BlockId {
606        match self {
607            Self::Toggle(section) => BlockId::ToolOutput { call_id, section },
608            Self::LoadFull(block) => BlockId::ToolFull {
609                call_id,
610                block: u32::try_from(block).unwrap_or(u32::MAX),
611            },
612            Self::OpenSession(session) => BlockId::ToolSession { call_id, session },
613            Self::OpenMemory(block) => BlockId::ToolMemory {
614                call_id,
615                block: u32::try_from(block).unwrap_or(u32::MAX),
616            },
617        }
618    }
619}

702pub(super) const OUTPUT_NOTICE_RESERVE_BYTES: usize = 96;
703
704pub(super) struct RenderBudget {
705    pub(super) remaining: RenderLimits,
706}
707
708impl RenderBudget {
709    pub(super) fn new(limits: RenderLimits) -> Self {
710        Self { remaining: limits }
711    }
712
713    pub(super) fn section_capacity(
714        &self,
715        limits: RenderLimits,
716        future_sections: usize,
717    ) -> RenderLimits {
718        let reserved_sections = future_sections.saturating_add(1);
719        RenderLimits {
720            lines: limits
721                .lines
722                .saturating_sub(1)
723                .min(self.remaining.lines.saturating_sub(reserved_sections)),
724            bytes: limits
725                .bytes
726                .saturating_sub(OUTPUT_NOTICE_RESERVE_BYTES)
727                .min(
728                    self.remaining
729                        .bytes
730                        .saturating_sub(reserved_sections * OUTPUT_NOTICE_RESERVE_BYTES),
731                ),
732        }
733    }
734
735    pub(super) fn consume(
736        &mut self,
737        capacity: &mut RenderLimits,
738        text: &str,
739    ) -> Option<(String, bool)> {
740        if capacity.lines == 0 || capacity.bytes == 0 {
741            return None;
742        }
743        let available = capacity.bytes.min(self.remaining.bytes);
744        let (text, complete) = sanitized_display_prefix(text, available);
745        if text.is_empty() && !complete {
746            return None;
747        }
748        capacity.lines -= 1;
749        capacity.bytes = capacity.bytes.saturating_sub(text.len());
750        self.remaining.lines = self.remaining.lines.saturating_sub(1);
751        self.remaining.bytes = self.remaining.bytes.saturating_sub(text.len());
752        Some((text, complete))
753    }
754
755    pub(super) fn consume_notice(&mut self, text: &str) -> bool {
756        if self.remaining.lines == 0 || self.remaining.bytes < text.len() {
757            return false;
758        }
759        self.remaining.lines -= 1;
760        self.remaining.bytes -= text.len();
761        true
762    }
763}
764
765pub(super) fn output_section_limits(expanded: bool) -> RenderLimits {
766    if expanded {
767        EXPANDED_TOOL_OUTPUT_LIMITS
768    } else {
769        COLLAPSED_TOOL_OUTPUT_LIMITS
770    }
771}

773/// What an expanded call shows under its arguments: why it failed, the
774/// running call's progress message, and its display blocks, all from one
775/// render budget, then the row that expands or collapses them.
776pub(super) fn tool_body_lines(
777    tool: &crate::state::ToolCallState,
778    style: &BlockStyle<'_>,
779    expanded: bool,
780    budget: &mut RenderBudget,
781) -> Vec<ToolBodyLine> {
782    let theme = style.theme;
783    let mut sink = LineSink::new(budget, output_section_limits(expanded));
784    let mut output = Vec::new();
785    if let Some(failure) = &tool.failure {
786        for line in failure.split('\n') {
787            // Expanded text is ordinary text; the header suffix already
788            // names the failure.
789            if let Some(text) = sink.line(line) {
790                output.push(ToolBodyLine::wrapped(Line::from(text)));
791            }
792        }
793    }
794    if tool.status == ToolStatus::Running
795        && let Some(progress) = &tool.progress
796        && let Some(text) = sink.line(progress)
797    {
798        output.push(ToolBodyLine::wrapped(Line::styled(
799            format!("⋯ {text}"),
800            theme.muted(),
801        )));
802    }
803    let blocks = display_rows(tool.shown_display(), &tool.full, style, &mut sink);
804    if !output.is_empty() && !blocks.is_empty() {
805        output.push(ToolBodyLine::wrapped(Line::default()));
806    }
807    output.extend(blocks);
808    let omitted = sink.omitted();
809    append_output_notice(
810        &mut output,
811        omitted,
812        expanded,
813        ToolOutputSection::Detail,
814        sink.budget(),
815        theme,
816    );
817    output
818}

820/// Tool children render inside the assistant item without a standalone
821/// `TOOL` header: the compact/expanded rows keep the assistant gutter and
822/// take only their status style.
823pub(super) struct ToolBlockLayout {
824    pub(super) lines: Vec<Line<'static>>,
825    /// Rows that act when clicked, by their start and end line.
826    pub(super) actions: Vec<(ToolRowAction, usize, usize)>,
827    pub(super) header_lines: usize,
828    /// Chrome columns per row, indexed alongside `lines`: the width of the
829    /// spans this builder hung in front of each row, counted from what it
830    /// built rather than from what the row says.
831    pub(super) chrome: Vec<u16>,
832}
833
834/// A collapsed-style tool block: muted body text. `role` still picks the
835/// narrow-row status label.
836pub(super) fn tool_block_lines(
837    role: Role,
838    body: Vec<ToolBodyLine>,
839    width: u16,
840    theme: &Theme,
841) -> ToolBlockLayout {
842    tool_block_lines_with(theme.muted_text(), role, body, width, theme)
843}
844
845/// [`tool_block_lines`] with an explicit base text style: an expanded block
846/// uses ordinary text rather than its status colour.
847pub(super) fn tool_block_lines_with(
848    style: Style,
849    role: Role,
850    body: Vec<ToolBodyLine>,
851    width: u16,
852    theme: &Theme,
853) -> ToolBlockLayout {
854    let mut lines = Vec::new();
855    let mut actions = Vec::new();
856    let mut banded_rows = Vec::new();
857    // Row index → how many of its leading spans are chrome this builder put
858    // there: the block's `│ ` gutter, a narrow-mode label, a diff's line
859    // number and marker. Counted while the row is assembled because it is a
860    // fact about where the spans came from, not about what they say — the same
861    // characters arriving as command output are content, and command output
862    // that reads like a gutter (`│ `) is still content.
863    let mut gutters = Vec::new();
864    let mut header_lines = 0;
865    for (index, body_line) in body.into_iter().enumerate() {
866        let band = body_line.band;
867        let tint = body_line.tint;
868        let header = body_line.header;
869        let action = body_line.action;
870        let line_style = body_line.line.style;
871        let spans = body_line
872            .line
873            .spans
874            .into_iter()
875            .map(|mut span| {
876                span.style = style.patch(span.style);
877                span
878            })
879            .collect::<Vec<_>>();
880        let line = Line::from(spans).style(line_style);
881        let start = lines.len();
882        let background = match band {
883            Some(ToolBand::Title) => theme.tool_title_background(),
884            Some(ToolBand::Terminal) => theme.terminal_background(),
885            None => None,
886        };
887        // Banded rows and title rows keep one margin column on each side of
888        // their text, so a title sits in the same column collapsed or
889        // expanded. The left one rides in the gutter (chrome, so hover and
890        // copy skip it); the right one is reserved from the wrap width.
891        let margin = panel_margin(width) && (band.is_some() || index == 0 || header);
892        let wrap_width = if margin { width - 1 } else { width };
893        let gutter = || {
894            let mut gutter = vec![Span::styled("│ ", theme.assistant())];
895            if margin {
896                gutter.push(margin_span(theme, background));
897            }
898            gutter
899        };
900        match body_line.kind {
901            ToolBodyLineKind::Wrapped | ToolBodyLineKind::Hanging { .. } if width < 8 => {
902                // The label and its aligned indent share one budget, so every
903                // wrapped row of the block stays inside the viewport.
904                let reserve = unbreakable_columns(&line);
905                let label_width = tool_row_label_columns(width, reserve);
906                let prefix = if index == 0 {
907                    tool_row_prefix(tool_row_short(role), width, reserve)
908                } else {
909                    " ".repeat(label_width)
910                };
911                // Every row keeps exactly one label column, even when the
912                // label itself would not fit and the span goes empty.
913                lines.extend(prefixed_wrapped_line(prefix, style, line, width));
914                gutters.resize(lines.len(), 1);
915            }
916            // Only the `│ ` span is chrome here: the margin behind it takes the
917            // hover fill with the row's text, so a highlighted title or notice
918            // covers its padding. Copying still skips it by its style.
919            ToolBodyLineKind::Wrapped => {
920                let gutter = gutter();
921                let chrome = usize::from(gutter_fits(&gutter, &line, wrap_width));
922                lines.extend(repeated_prefixed_wrapped_line(gutter, line, wrap_width));
923                gutters.resize(lines.len(), chrome);
924            }
925            ToolBodyLineKind::Hanging { indent } => {
926                let gutter = gutter();
927                let chrome = usize::from(gutter_fits(&gutter, &line, wrap_width));
928                lines.extend(repeated_prefixed_hanging_line(
929                    gutter, line, wrap_width, indent,
930                ));
931                gutters.resize(lines.len(), chrome);
932            }
933            ToolBodyLineKind::Code {
934                mut first_gutter,
935                mut continuation_gutter,
936            } => {
937                let prefix = if width >= 3 { gutter() } else { Vec::new() };
938                let prefix_width = prefix
939                    .iter()
940                    .map(|span| UnicodeWidthStr::width(span.content.as_ref()))
941                    .sum::<usize>();
942                let available = usize::from(wrap_width.max(1)).saturating_sub(prefix_width);
943                let mut first_gutter_width = first_gutter
944                    .iter()
945                    .map(|span| UnicodeWidthStr::width(span.content.as_ref()))
946                    .sum::<usize>();
947                let mut continuation_gutter_width = continuation_gutter
948                    .iter()
949                    .map(|span| UnicodeWidthStr::width(span.content.as_ref()))
950                    .sum::<usize>();
951                if first_gutter_width >= available || continuation_gutter_width >= available {
952                    first_gutter.clear();
953                    continuation_gutter.clear();
954                    first_gutter_width = 0;
955                    continuation_gutter_width = 0;
956                }
957                for (wrapped_index, content) in crate::markdown::wrap_code_spans(
958                    line.spans,
959                    available.saturating_sub(first_gutter_width).max(1),
960                    available.saturating_sub(continuation_gutter_width).max(1),
961                )
962                .into_iter()
963                .enumerate()
964                {
965                    let mut spans = prefix.clone();
966                    if wrapped_index == 0 {
967                        spans.extend(first_gutter.clone());
968                    } else {
969                        spans.extend(continuation_gutter.clone());
970                    }
971                    spans.extend(content);
972                    gutters.push(
973                        prefix.len()
974                            + if wrapped_index == 0 {
975                                first_gutter.len()
976                            } else {
977                                continuation_gutter.len()
978                            },
979                    );
980                    lines.push(Line::from(spans).style(line_style));
981                }
982            }
983        }
984        // An expanded title block's rows are all header: one click target,
985        // one hover highlight.
986        if index == 0 || header {
987            header_lines = lines.len();
988        }
989        if let Some(action) = action {
990            actions.push((action, start, lines.len()));
991        }
992        if band.is_some() {
993            banded_rows.extend(
994                (start..lines.len()).map(|row| (row, gutters[row], background, tint, margin)),
995            );
996        }
997    }
998    for (index, chrome, background, tint, padded) in banded_rows {
999        // The band stops at the block's own `│ ` gutter, its first chrome
1000        // span; line numbers and diff markers behind it are band too.
1001        // Counted, never guessed — `│ ` arriving as command output is a tree
1002        // row that must be banded, and a gutter welded to its text by
1003        // `append_span` is chrome that must not be.
1004        paint_band(
1005            &mut lines[index],
1006            chrome.min(1),
1007            width,
1008            background,
1009            tint,
1010            padded,
1011        );
1012    }
1013    for line in &mut lines {
1014        line.style = line
1015            .style
1016            .remove_modifier(ratatui::style::Modifier::UNDERLINED);
1017        for span in &mut line.spans {
1018            span.style = span
1019                .style
1020                .remove_modifier(ratatui::style::Modifier::UNDERLINED);
1021        }
1022    }
1023    // Measured last, from the counted spans: a row's chrome is whatever the
1024    // builder put in front of it, so command output that reads like a gutter
1025    // (`│ ├── src`, a tree listing) is measured as the content it is.
1026    let chrome = gutters
1027        .iter()
1028        .zip(&lines)
1029        .map(|(spans, line)| {
1030            u16::try_from(
1031                line.spans
1032                    .iter()
1033                    .take(*spans)
1034                    .map(|span| UnicodeWidthStr::width(span.content.as_ref()))
1035                    .sum::<usize>(),
1036            )
1037            .unwrap_or(u16::MAX)
1038        })
1039        .collect();
1040    ToolBlockLayout {
1041        lines,
1042        actions,
1043        header_lines,
1044        chrome,
1045    }
1046}

1048/// Chrome columns of a tool block's header rows: the widest chrome row of the
1049/// header, since one highlight spans them all.
1050pub(super) fn header_gutter_columns(rendered: &ToolBlockLayout) -> u16 {
1051    rendered
1052        .chrome
1053        .iter()
1054        .take(rendered.header_lines)
1055        .copied()
1056        .max()
1057        .unwrap_or(0)
