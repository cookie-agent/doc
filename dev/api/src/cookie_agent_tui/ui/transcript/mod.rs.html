59/// Scrollbar geometry over the total rendered line height.
60///
61/// Thumb **height** is strictly a function of the total content height and
62/// the viewport (track) height — `ceil(viewport² / content)`, clamped to
63/// `[1, track]` — and never of the scroll offset, position, or follow state.
64/// Thumb **top** is the only position-dependent value: offset 0 maps to the
65/// first track row and the maximum valid top offset maps the thumb flush
66/// against the last track row, so top/bottom are exact. One helper is shared
67/// by render, hit testing, and drag math; no ratatui `ScrollbarState`
68/// position/content-length folding is involved.
69#[derive(Clone, Copy, Debug, Eq, PartialEq)]
70pub(super) struct ScrollbarGeometry {
71    pub(super) track: Rect,
72    pub(super) thumb: Rect,
73    pub(super) content_height: usize,
74    pub(super) viewport_height: usize,
75    pub(super) max_offset: usize,
76}
77
78impl ScrollbarGeometry {
79    pub(super) fn resolve(track: Rect, content_height: usize) -> Option<Self> {
80        if track.height == 0 || content_height <= usize::from(track.height) {
81            return None;
82        }
83        let viewport_height = usize::from(track.height);
84        let max_offset = content_height - viewport_height;
85        Some(Self {
86            track,
87            thumb: Rect::default(),
88            content_height,
89            viewport_height,
90            max_offset,
91        })
92    }
93
94    /// Clamp any top offset into the valid range for this geometry.
95    pub(super) fn clamp_offset(&self, offset: usize) -> usize {
96        offset.min(self.max_offset)
97    }
98
99    /// Thumb height in rows: the exact visible fraction of the track,
100    /// independent of scroll offset/position/following.
101    pub(super) fn thumb_size(&self) -> usize {
102        let track = usize::from(self.track.height);
103        (self.viewport_height * track)
104            .div_ceil(self.content_height)
105            .clamp(1, track)
106    }
107
108    /// Thumb top row within the track for a top offset. Linear over the
109    /// available travel `track − thumb_size`, so the thumb sits flush at the
110    /// track top at offset 0 and flush at the track bottom at max offset,
111    /// always at full height.
112    pub(super) fn thumb_top(&self, offset: usize) -> usize {
113        let travel = usize::from(self.track.height) - self.thumb_size();
114        if travel == 0 || self.max_offset == 0 {
115            return 0;
116        }
117        (self.clamp_offset(offset) * travel + self.max_offset / 2) / self.max_offset
118    }
119
120    pub(super) fn with_thumb(mut self, offset: usize) -> Self {
121        let top = self.thumb_top(offset);
122        let size = self.thumb_size();
123        self.thumb = Rect::new(
124            self.track.x,
125            self.track.y + u16::try_from(top).unwrap_or(u16::MAX),
126            self.track.width,
127            u16::try_from(size).unwrap_or(u16::MAX),
128        );
129        self
130    }
131
132    /// Offset for a thumb dragged so its grab anchor sits at `row`. Exact
133    /// inverse of `thumb_top`, clamped to the valid range.
134    pub(super) fn offset_for_thumb_anchor(&self, row: u16, grab: u16) -> usize {
135        let travel = usize::from(self.track.height) - self.thumb_size();
136        if travel == 0 {
137            return 0;
138        }
139        let row = usize::from(row.saturating_sub(self.track.y).min(self.track.height - 1));
140        let top = row.saturating_sub(usize::from(grab));
141        (top * self.max_offset + travel / 2) / travel
142    }
143
144    /// Offset whose viewport is centered on the track position of `row`.
145    pub(super) fn offset_for_track_row(&self, row: u16) -> usize {
146        self.offset_for_thumb_anchor(row, (self.thumb_size() / 2) as u16)
147    }
148}

150/// Columns every scrollable pane keeps free at the right edge of its
151/// interior: [`SCROLLBAR_MARGIN`] blank columns, then the
152/// [`SCROLLBAR_TRACK_COLUMNS`] track flush against the right border.
153///
154/// The reservation is **constant** — it never depends on whether content
155/// actually overflows. Wrap width is part of [`LayoutCacheKey`], so a
156/// conditional reservation re-wrapped the whole transcript the moment the
157/// "can this overflow?" heuristic flipped: a phantom reflow with no content
158/// change. Only the *thumb* stays conditional on real overflow; reserving is
159/// about width, drawing is about need.
160pub(super) const SCROLLBAR_RESERVE: u16 = 2;
161
162/// Track columns inside [`SCROLLBAR_RESERVE`].
163pub(super) const SCROLLBAR_TRACK_COLUMNS: u16 = 1;
164
165/// Blank columns between a pane's text and its scrollbar track.
166pub(super) const SCROLLBAR_MARGIN: u16 = SCROLLBAR_RESERVE - SCROLLBAR_TRACK_COLUMNS;
167
168/// Text width of a bordered pane: both borders, then the constant scrollbar
169/// reservation. Shared by the conversation pane and the message composer so
170/// both lay out, wrap, highlight and hit-test on the same columns.
171pub(super) fn pane_text_width(outer_width: u16) -> u16 {
172    outer_width
173        .saturating_sub(2)
174        .saturating_sub(SCROLLBAR_RESERVE)
175}

177/// The reserved scrollbar track of a bordered pane, given the pane `area` and
178/// the text `viewport` carved out of it: the rightmost reserved column, one
179/// blank margin away from the text. Collapses to zero width when the pane is
180/// too narrow to hold both the margin and the track.
181pub(super) fn pane_scrollbar_track(area: Rect, viewport: Rect) -> Rect {
182    Rect::new(
183        viewport
184            .x
185            .saturating_add(viewport.width)
186            .saturating_add(SCROLLBAR_MARGIN),
187        viewport.y,
188        SCROLLBAR_TRACK_COLUMNS.min(
189            area.width
190                .saturating_sub(2)
191                .saturating_sub(viewport.width)
192                .saturating_sub(SCROLLBAR_MARGIN),
193        ),
194        viewport.height,
195    )
196}

198/// Where the conversation viewport rests, and whether it follows live output.
199///
200/// Following pins the view to the bottom on every frame. Any scroll the user
201/// makes away from the bottom stops it at once, and it stays stopped, however
202/// output streams in or content changes under the view, until the user's own
203/// scrolling (wheel, keys, the scrollbar) or a block they fold brings the view
204/// back to the bottom.
205#[derive(Debug)]
206pub struct ConversationScroll {
207    pub(super) offset: usize,
208    pub(super) following: bool,
209    max_offset: Option<usize>,
210    /// The user moved the view, or folded or opened a block, since the last
211    /// clamp: resting on the bottom then means they came back to it, while
212    /// content shrinking under a scrolled-up view only pushes it there.
213    moved: bool,
214}
215
216impl Default for ConversationScroll {
217    fn default() -> Self {
218        let mut scroll = Self {
219            offset: 0,
220            following: false,
221            max_offset: None,
222            moved: false,
223        };
224        scroll.bottom();
225        scroll
226    }
227}

229impl ConversationScroll {
230    pub(super) fn max_offset(total_lines: usize, viewport_height: u16) -> usize {
231        total_lines.saturating_sub(usize::from(viewport_height))
232    }
233
234    pub(super) fn clamp(&mut self, total_lines: usize, viewport_height: u16) {
235        let max_offset = Self::max_offset(total_lines, viewport_height);
236        self.max_offset = Some(max_offset);
237        // Content that shrank below a scrolled-up view pushes it onto the new
238        // bottom without the user coming back there.
239        let pushed = !self.moved && self.offset > max_offset;
240        self.moved = false;
241        self.offset = if self.following {
242            max_offset
243        } else {
244            self.offset.min(max_offset)
245        };
246        // A non-following view the user brought exactly onto the last valid
247        // top offset is the live bottom; following resumes from there.
248        if self.offset == max_offset && !pushed {
249            self.following = true;
250        }
251    }
252
253    pub(super) fn up(&mut self, lines: usize) {
254        let target = self.offset.saturating_sub(lines);
255        self.reveal(
256            BlockRegion {
257                id: BlockId::Thinking(0),
258                start_line: target,
259                end_line: target,
260                header_lines: None,
261                header_gutter: None,
262            },
263            1,
264        );
265    }
266    pub(super) fn down(&mut self, lines: usize) {
267        self.scroll_to(self.offset.saturating_add(lines));
268    }
269    #[cfg(test)]
270    pub fn top(&mut self) {
271        self.following = false;
272        self.moved = true;
273        self.offset = 0;
274    }
275    pub fn bottom(&mut self) {
276        self.following = true;
277    }
278
279    /// Keeps the view where it is while the user folds or opens a block:
280    /// what they read stays put instead of following the tail.
281    pub(super) fn hold(&mut self) {
282        self.following = false;
283        self.moved = true;
284    }
285
286    /// Absolute top offset from a scrollbar thumb/track gesture.
287    pub(super) fn scroll_to(&mut self, offset: usize) {
288        // Resolve the gesture against the last rendered content, before an
289        // output append can move the bottom on the next frame.
290        self.offset = offset;
291        self.moved = true;
292        self.following = self.max_offset.is_some_and(|max| offset >= max);
293    }
294
295    pub fn reveal(&mut self, region: BlockRegion, viewport_height: u16) {
296        let height = usize::from(viewport_height.max(1));
297        self.following = false;
298        self.moved = true;
299        if region.start_line < self.offset {
300            self.offset = region.start_line;
301        } else if region.end_line > self.offset.saturating_add(height) {
302            self.offset = region.end_line.saturating_sub(height);
303        }
304    }
305}

420/// A line sequence assembled from shared [`Chunk`]s: appending an unchanged
421/// item, part, or tool row costs a pointer, never a copy of its rows.
422#[derive(Clone, Default)]
423pub(super) struct LineChunks {
424    chunks: Vec<Chunk>,
425    /// The line index each chunk starts at.
426    starts: Vec<usize>,
427    len: usize,
428}
429
430impl LineChunks {
431    pub(super) fn len(&self) -> usize {
432        self.len
433    }
434
435    pub(super) fn is_empty(&self) -> bool {
436        self.len == 0
437    }
438
439    pub(super) fn push_chunk(&mut self, chunk: Chunk) {
440        if chunk.is_empty() {
441            return;
442        }
443        self.starts.push(self.len);
444        self.len += chunk.len();
445        self.chunks.push(chunk);
446    }
447
448    fn push_lines(&mut self, lines: Vec<Line<'static>>) {
449        if !lines.is_empty() {
450            self.push_chunk(lines.into());
451        }
452    }
453
454    pub(super) fn get(&self, index: usize) -> Option<&Line<'static>> {
455        if index >= self.len {
456            return None;
457        }
458        let chunk = self.starts.partition_point(|start| *start <= index) - 1;
459        self.chunks[chunk].get(index - self.starts[chunk])
460    }
461
462    /// Keep the first `len` lines.
463    fn truncate(&mut self, len: usize) {
464        if len >= self.len {
465            return;
466        }
467        let kept = self.starts.partition_point(|start| *start < len);
468        self.chunks.truncate(kept);
469        self.starts.truncate(kept);
470        self.len = len;
471        if let (Some(last), Some(start)) = (self.chunks.last_mut(), self.starts.last())
472            && start + last.len() > len
473        {
474            *last = last[..len - start].into();
475        }
476    }
477
478    pub(super) fn iter(&self) -> impl Iterator<Item = &Line<'static>> {
479        self.chunks.iter().flat_map(|chunk| chunk.iter())
480    }
481
482    pub(super) fn to_vec(&self) -> Vec<Line<'static>> {
483        self.iter().cloned().collect()
484    }
485
486    #[cfg(test)]
487    pub(super) fn chunks(&self) -> &[Chunk] {
488        &self.chunks
489    }
490}

589    BlockStart(BlockId),

593impl LayoutCache {
594    /// The closest stable boundary before the viewport, plus its row offset.
595    /// Block ends also anchor text following a tool within one assistant item.
596    fn scroll_anchor(&self, offset: usize) -> Option<(ScrollAnchorPoint, usize)> {
597        self.items
598            .iter()
599            .zip(&self.item_offsets)
600            .map(|(item, start)| (ScrollAnchorPoint::Item(item.key.id), start.lines))
601            .chain(
602                self.layout
603                    .regions
604                    .iter()
605                    // Output notices move after the newly revealed lines;
606                    // anchor their containing tool's content instead.
607                    .filter(|region| {
608                        !matches!(
609                            region.id,
610                            BlockId::ToolOutput { .. }
611                                | BlockId::ToolFull { .. }
612                                | BlockId::ToolSession { .. }
613                                | BlockId::ToolMemory { .. }
614                        )
615                    })
616                    .flat_map(|region| {
617                        [
618                            (ScrollAnchorPoint::BlockStart(region.id), region.start_line),
619                            (ScrollAnchorPoint::BlockEnd(region.id), region.end_line),
620                        ]
621                    }),
622            )
623            .filter(|(_, line)| *line <= offset)
624            .max_by_key(|(_, line)| *line)
625            .map(|(point, line)| (point, offset - line))
626    }
627
628    /// Ask for the code rendered plain on `lines` (of the assembled layout)
629    /// to be highlighted, marking each item holding some for its next
630    /// layout. True when any was asked for.
631    fn request_highlights(&mut self, lines: std::ops::Range<usize>) -> bool {
632        let total = self.layout.lines.len();
633        let first = self
634            .item_offsets
635            .partition_point(|offset| offset.lines <= lines.start)
636            .saturating_sub(1);
637        let mut requested = false;
638        for index in first..self.items.len().min(self.item_offsets.len()) {
639            let start = self.item_offsets[index].lines;
640            if start >= lines.end {
641                break;
642            }
643            let item = &mut self.items[index];
644            if item.deferred.is_empty() {
645                continue;
646            }
647            let end = self
648                .item_offsets
649                .get(index + 1)
650                .map_or(total, |next| next.lines);
651            // The item's rows end the span its separator opens.
652            let body = end.saturating_sub(item.layout.lines.len());
653            for (rows, highlight) in &item.deferred {
654                if body + rows.start < lines.end && body + rows.end > lines.start {
655                    highlight.request();
656                    item.highlight_requested = true;
657                    requested = true;
658                }
659            }
660        }
661        requested
662    }
663
664    fn anchor_line(&self, point: ScrollAnchorPoint) -> Option<usize> {
665        match point {
666            ScrollAnchorPoint::Item(id) => self
667                .items
668                .iter()
669                .zip(&self.item_offsets)
670                .find_map(|(item, start)| (item.key.id == id).then_some(start.lines)),
671            ScrollAnchorPoint::BlockStart(id) | ScrollAnchorPoint::BlockEnd(id) => self
672                .layout
673                .regions
674                .iter()
675                .find(|region| region.id == id)
676                .map(|region| match point {
677                    ScrollAnchorPoint::BlockStart(_) => region.start_line,
678                    _ => region.end_line,
679                }),
680        }
681    }

741/// [`ItemLayout`] over shared chunks: an assistant item holds one chunk per
742/// child, so re-laying it out after one child changed reuses every other
743/// child's rows by reference.
744#[derive(Clone, Default)]
745struct ChunkedItem {
746    lines: LineChunks,
747    regions: Vec<BlockRegion>,
748    user_seq: Option<u64>,
749}
750
751impl ChunkedItem {
752    /// Append one child's rows and its regions, shifted to where they land.
753    fn push(&mut self, lines: &Chunk, regions: &[BlockRegion]) {
754        let start = self.lines.len();
755        self.regions
756            .extend(regions.iter().map(|region| BlockRegion {
757                start_line: start + region.start_line,
758                end_line: start + region.end_line,
759                ..*region
760            }));
761        self.lines.push_chunk(Chunk::clone(lines));
762    }
763
764    /// [`Self::push`] for a multi-chunk child.
765    fn push_chunks(&mut self, lines: &LineChunks, regions: &[BlockRegion]) {
766        let start = self.lines.len();
767        self.regions
768            .extend(regions.iter().map(|region| BlockRegion {
769                start_line: start + region.start_line,
770                end_line: start + region.end_line,
771                ..*region
772            }));
773        for chunk in &lines.chunks {
774            self.lines.push_chunk(Chunk::clone(chunk));
775        }
776    }
777
778    fn push_layout(&mut self, layout: ItemLayout) {
779        let start = self.lines.len();
780        self.regions
781            .extend(layout.regions.into_iter().map(|region| BlockRegion {
782                start_line: start + region.start_line,
783                end_line: start + region.end_line,
784                ..region
785            }));
786        self.lines.push_lines(layout.lines);
787    }
788
789    fn flatten(self) -> ItemLayout {
790        ItemLayout {
791            lines: self.lines.to_vec(),
792            regions: self.regions,
793            user_seq: self.user_seq,
794        }
795    }
796}

908impl BlockHit {
909    /// The cells a hover paints: the block's header content rect, clipped to
910    /// its hit rect and to `viewport` (which already stops before the
911    /// reserved scrollbar columns), minus a background button with the blank
912    /// column before it and the rest of its row, and minus a memory link
913    /// with a blank column on each side. The header highlight leaves out
914    /// what hovers on its own, so the highlights never touch.
915    pub(super) fn hover_cells(&self, viewport: Option<Rect>) -> Vec<Rect> {
916        let Some(rect) = self.hover_rect.map(|rect| rect.intersection(self.rect)) else {
917            return Vec::new();
918        };
919        let rect = viewport.map_or(rect, |viewport| rect.intersection(viewport));
920        let on_header = |button: &Rect| button.y >= rect.y && button.y < rect.bottom();
921        // Each target on the title row takes the columns `[start, end)`.
922        let mut taken = self
923            .action_rect
924            .filter(on_header)
925            .map(|button| (button.y, button.x.saturating_sub(1), rect.right()))
926            .into_iter()
927            .chain(self.link_rect.filter(on_header).map(|link| {
928                (
929                    link.y,
930                    link.x.saturating_sub(1),
931                    link.right().saturating_add(1),
932                )
933            }))
934            .collect::<Vec<_>>();
935        let Some(&(row, _, _)) = taken.first() else {
936            return vec![rect];
937        };
938        taken.retain(|(y, _, _)| *y == row);
939        taken.sort_by_key(|(_, start, _)| *start);
940        let mut cells = vec![Rect::new(rect.x, rect.y, rect.width, row - rect.y)];
941        let mut x = rect.x;
942        for (_, start, end) in taken {
943            let start = start.clamp(x, rect.right());
944            cells.push(Rect::new(x, row, start - x, 1));
945            x = end.clamp(x, rect.right());
946        }
947        cells.push(Rect::new(x, row, rect.right() - x, 1));
948        cells.push(Rect::new(
949            rect.x,
950            row + 1,
951            rect.width,
952            rect.bottom() - row - 1,
953        ));
954        cells.retain(|rect| !rect.is_empty());
955        cells
956    }

959// Layout cache validity depends on each independent render input; grouping them
960// would obscure invalidation semantics without reducing call-site complexity.
961#[allow(clippy::too_many_arguments)]
962pub(super) fn ensure_cached_transcript_layout(
963    cache: &mut LayoutCache,
964    session_id: SessionId,
965    state: &SessionState,
966    draft_agent: Option<&AgentId>,
967    expanded: Option<&HashSet<BlockId>>,
968    width: u16,
969    theme: &Theme,
970    highlighter: &dyn Highlighter,
971    minimum_event_level: crate::state::EventLevel,
972    clock_bucket: u8,
973    model_names: &ModelDisplayNames,
974) -> bool {
975    let key = LayoutCacheKey {
976        session_id,
977        session_generation: state.generation,
978        width,
979        theme: theme.key(),
980        highlighter: highlighter as *const dyn Highlighter as *const () as usize,
981        minimum_event_level,
982    };
983    let full_rebuild = cache.key != Some(key);
984    if full_rebuild {
985        cache.key = Some(key);
986        cache.layout = ChunkedLayout::default();
987        cache.items.clear();
988        cache.item_offsets.clear();
989        cache.assistant_parts.clear();
990        cache.tool_rows.clear();
991        cache.system_prompt = None;
992    }
993
994    let prompt_key = state
995        .run_snapshot
996        .as_deref()
997        .map(|snapshot| SystemPromptLayoutKey {
998            fingerprint: snapshot.prompt_fingerprint.clone(),
999            snapshot_agent: snapshot.agent.clone(),
1000            draft_agent: draft_agent.cloned(),
1001            expanded: expanded.is_some_and(|blocks| blocks.contains(&BlockId::SystemPrompt)),
1002            model_tools: state.model_tools.clone(),
1003            model_deferred_tools: state.model_deferred_tools.clone(),
1004            model_tools_expanded: expanded
1005                .is_some_and(|blocks| blocks.contains(&BlockId::ModelTools)),
1006            acp_agent: state.acp.agent.clone(),
1007        });
1008    let system_prompt_changed =
1009        cache.system_prompt.as_ref().map(|cached| &cached.key) != prompt_key.as_ref();
1010    if system_prompt_changed {
1011        cache.layout = ChunkedLayout::default();
1012        cache.item_offsets.clear();
1013        cache.system_prompt = if let (Some(snapshot), Some(prompt_key)) =
1014            (state.run_snapshot.as_deref(), prompt_key)
1015        {
1016            let layout = system_prompt_layout(snapshot, draft_agent, expanded, width, theme);
1017            append_chunked_item(&mut cache.layout, &layout.into());
1018            if let Some(tools) = state.model_tools.as_deref() {
1019                append_chunked_item(
1020                    &mut cache.layout,
1021                    &model_tools_layout(state, tools, expanded, width, theme).into(),
1022                );
1023            }
1024            Some(CachedSystemPromptLayout { key: prompt_key })
1025        } else {
1026            None
1027        };
1028    }
1029
1030    let first_dirty = if full_rebuild || system_prompt_changed {
1031        Some(0)
1032    } else {
1033        state
1034            .transcript
1035            .iter()
1036            .enumerate()
1037            .find_map(|(index, item)| {
1038                (!cache.items.get(index).is_some_and(|cached| {
1039                    !cached.highlight_requested
1040                        && item_layout_key_matches(
1041                            &cached.key,
1042                            state,
1043                            item,
1044                            expanded,
1045                            clock_bucket,
1046                            model_names,
1047                        )
1048                }))
1049                .then_some(index)
1050            })
1051            .or_else(|| {
1052                (cache.items.len() != state.transcript.len()).then_some(state.transcript.len())
1053            })
1054    };
1055    let Some(first_dirty) = first_dirty else {
1056        return true;
1057    };
1058
1059    if !full_rebuild && !system_prompt_changed {
1060        let offset = cache
1061            .item_offsets
1062            .get(first_dirty)
1063            .copied()
1064            .unwrap_or_else(|| ItemAssemblyOffset::at_end(&cache.layout));
1065        cache.layout.lines.truncate(offset.lines);
1066        cache.layout.regions.truncate(offset.regions);
1067        cache.layout.user_regions.truncate(offset.user_regions);
1068        cache.item_offsets.truncate(first_dirty);
1069    }
1070
1071    // Items from the first dirty one on are matched by id, not position: a
1072    // row moved to the end (a folded warning) shifts every later item by
1073    // one, and each of them is still exactly what it was.
1074    let mut previous = cache
1075        .items
1076        .drain(first_dirty.min(cache.items.len())..)
1077        .map(|cached| (cached.key.id, cached))
1078        .collect::<HashMap<_, _>>();
1079    for item in state.transcript.iter().skip(first_dirty) {
1080        cache
1081            .item_offsets
1082            .push(ItemAssemblyOffset::at_end(&cache.layout));
1083        let reusable = previous.remove(&item.id()).filter(|cached| {
1084            !cached.highlight_requested
1085                && item_layout_key_matches(
1086                    &cached.key,
1087                    state,
1088                    item,
1089                    expanded,
1090                    clock_bucket,
1091                    model_names,
1092                )
1093        });
1094        if let Some(cached) = reusable {
1095            cache.items.push(cached);
1096        } else {
1097            let item_key = item_layout_key(state, item, expanded, clock_bucket, model_names);
1098            let mut assistant_child_lines = Vec::new();
1099            let mut deferred = Vec::new();
1100            let mut context = TranscriptRenderContext {
1101                expanded,
1102                width,
1103                theme,
1104                highlighter,
1105                minimum_event_level,
1106                clock_bucket,
1107                model_names,
1108                assistant_part_cache: &mut cache.assistant_parts,
1109                assistant_part_layout_passes: &mut cache.assistant_part_layout_passes,
1110                tool_cache: &mut cache.tool_rows,
1111                assistant_child_lines: &mut assistant_child_lines,
1112                defer_highlights: true,
1113                deferred_highlights: &mut deferred,
1114            };
1115            // Unchanged parts and tool rows come back from their caches by
1116            // reference, so this re-renders only what changed.
1117            let layout = chunked_item_layout(state, item, &mut context);
1118            let cached = CachedItemLayout {
1119                key: item_key,
1120                highlight_requested: false,
1121                deferred,
1122                layout,
1123                child_lines: assistant_child_lines,
1124            };
1125            cache.items.push(cached);
1126            #[cfg(test)]
1127            {
1128                cache.item_layout_passes = cache.item_layout_passes.wrapping_add(1);
1129            }
1130        }
1131        let cached = cache.items.last().expect("item pushed");
1132        append_chunked_item(&mut cache.layout, &cached.layout);
1133        #[cfg(test)]
1134        {
1135            cache.item_assembly_passes = cache.item_assembly_passes.wrapping_add(1);
1136        }
1137    }
1138    cache.items.truncate(state.transcript.len());
1139    cache.item_offsets.truncate(state.transcript.len());
1140    cache.revision = next_layout_revision();
1141    false
1142}

1160/// Descendant rows spliced into a rendered transcript, kept apart from the
1161/// transcript's own lines: the view reads through both, so no frame copies
1162/// the transcript to show a handful of rows.
1163#[derive(Clone, Debug, Default)]
1164pub(super) struct SplicedRows {
1165    /// Every inserted row, in order.
1166    rows: Vec<Line<'static>>,
1167    /// Per splice point, ascending: its original-line position and the
1168    /// number of rows inserted there.
1169    pub(super) shifts: Vec<(usize, usize)>,
1170    /// Per splice point: the spliced-line index of its first row.
1171    starts: Vec<usize>,
1172    /// The inserted rows' own collapsible regions, in spliced coordinates.
1173    pub(super) regions: Vec<BlockRegion>,
1174}
1175
1176impl SplicedRows {
1177    fn is_empty(&self) -> bool {
1178        self.shifts.is_empty()
1179    }
1180
1181    /// Spliced line count over `base_len` original lines.
1182    fn len(&self, base_len: usize) -> usize {
1183        base_len + self.rows.len()
1184    }
1185
1186    /// Spliced line `index` over the original `base` lines.
1187    pub(super) fn line<'a, L: LineSource + ?Sized>(
1188        &'a self,
1189        base: &'a L,
1190        index: usize,
1191    ) -> Option<&'a Line<'static>> {
1192        let splice = self.starts.partition_point(|start| *start <= index);
1193        let Some(splice) = splice.checked_sub(1) else {
1194            return base.line_at(index);
1195        };
1196        let start = self.starts[splice];
1197        let (position, inserted) = self.shifts[splice];
1198        let inserted_before = start - position;
1199        if index < start + inserted {
1200            self.rows.get(inserted_before + index - start)
1201        } else {
1202            base.line_at(index - inserted_before - inserted)
1203        }
1204    }
1205
1206    /// Shift one original line coordinate by the rows spliced in before it;
1207    /// see [`splice_shifted`].
1208    fn shift(&self, line: usize, inclusive: bool) -> usize {
1209        let splices = self
1210            .shifts
1211            .partition_point(|(position, _)| *position < line || (inclusive && *position == line));
1212        let Some(last) = splices.checked_sub(1) else {
1213            return line;
1214        };
1215        let (position, inserted) = self.shifts[last];
1216        line + (self.starts[last] - position) + inserted
1217    }
1218
1219    fn shift_region(&self, mut region: BlockRegion) -> BlockRegion {
1220        let start = self.shift(region.start_line, true);
1221        region.end_line = self.shift(region.end_line, false).max(start);
1222        region.start_line = start;
1223        region
1224    }
1225
1226    fn shift_user_region(&self, mut region: UserRegion) -> UserRegion {
1227        let start = self.shift(region.start_line, true);
1228        region.end_line = self.shift(region.end_line, false).max(start);
1229        region.start_line = start;
1230        region
1231    }
1232
1233    /// Every spliced line, copied out.
1234    pub(super) fn materialize<L: LineSource + ?Sized>(&self, base: &L) -> Vec<Line<'static>> {
1235        (0..self.len(base.line_count()))
1236            .filter_map(|index| self.line(base, index).cloned())
1237            .collect()
1238    }
1239}

1295/// Shift one unspliced line coordinate by the rows spliced in before it. A
1296/// region start (`inclusive`) moves with rows inserted exactly at it; an
1297/// exclusive region end does not, so a region ending where rows are
1298/// inserted never grows over them.
1299fn splice_shifted(line: usize, splice_shifts: &[(usize, usize)], inclusive: bool) -> usize {
1300    splice_shifts
1301        .iter()
1302        .filter(|(position, _)| *position < line || (inclusive && *position == line))
1303        .map(|(_, inserted)| *inserted)
1304        .sum::<usize>()
1305        .saturating_add(line)
1306}
1307
1308/// Map a spliced line back to the unspliced layout. A line inside spliced-in
1309/// rows maps to their splice point, with its depth into them.
1310fn unsplice_line(line: usize, splice_shifts: &[(usize, usize)]) -> (usize, Option<usize>) {
1311    let mut inserted_before = 0;
1312    for (position, inserted) in splice_shifts {
1313        let start = position + inserted_before;
1314        if line < start {
1315            break;
1316        }
1317        if line < start + inserted {
1318            return (*position, Some(line - start));
1319        }
1320        inserted_before += inserted;
1321    }
1322    (line - inserted_before, None)
1323}

1325/// The header that resumes an assistant block below descendant rows spliced
1326/// in before child `child`: the block's producer with the model in effect at
1327/// that point, exactly as a block split by an in-session row would read.
1328fn continuation_header(
1329    item: &TranscriptItem,
1330    child: usize,
1331    width: u16,
1332    theme: &Theme,
1333    model_names: &ModelDisplayNames,
1334) -> Vec<Line<'static>> {
1335    let TranscriptItem::Assistant {
1336        attribution,
1337        children,
1338        ..
1339    } = item
1340    else {
1341        return Vec::new();
1342    };
1343    let resolved_model = children[..child.min(children.len())]
1344        .iter()
1345        .rev()
1346        .find_map(|child| match child {
1347            AssistantChild::Attribution { resolved_model } => Some(resolved_model),

1510    /// Splice aggregated descendant warning and error blocks into a rendered
1511    /// transcript at their chronological position.
1512    ///
1513    /// The viewed transcript is a sequence of timed segments in render
1514    /// order: each item at its durable insertion time, except that an
1515    /// assistant block contributes one segment per child at the child's own
1516    /// time (one run's turns and tool calls share a block, so the block's
1517    /// start time alone would push a row that happened mid-run below
1518    /// everything the run did afterwards). Each row lands just before the
1519    /// first segment that happened after it, so nothing rendered above a row
1520    /// happened later than the row; rows newer than every segment (or when
1521    /// nothing is timed) land at the bottom.
1522    ///
1523    /// Before an item, a row takes that item's separator slot. Between two
1524    /// children of an assistant block, it breaks the block: the rows render
1525    /// below the earlier children and a continuation header resumes the
1526    /// block, as an in-session row splitting a block would.
1527    ///
1528    /// `item_offsets` holds one [`ItemAssemblyOffset`] per transcript item
1529    /// and `items` the cached layouts they were assembled from. One pass
1530    /// walks the segments against the time-ordered rows.
1531    ///
1532    /// Returns the spliced lines plus, for each splice point, the
1533    /// original-line position and the number of lines inserted there, so
1534    /// callers can shift scroll anchors and hit regions that address the
1535    /// unspliced layout.
1536    // Each input is independent render state (layout, projection, rows,
1537    // width, theme, display names); bundling them would only add a type.
1538    #[allow(clippy::too_many_arguments)]
1539    fn splice_descendant_events(
1540        lines: &(impl LineSource + ?Sized),
1541        items: &[CachedItemLayout],
1542        item_offsets: &[ItemAssemblyOffset],
1543        state: &SessionState,
1544        events: &[DescendantEvent],
1545        expanded: Option<&HashSet<BlockId>>,
1546        width: u16,
1547        theme: &Theme,
1548        model_names: &ModelDisplayNames,
1549        row_layouts: &mut EventRowLayouts,
1550    ) -> SplicedRows {
1551        let mut placements: Vec<Placement<'_>> = Vec::with_capacity(events.len());
1552        let mut next = 0;
1553        for (index, item) in state.transcript.iter().enumerate() {
1554            if next == events.len() {
1555                break;
1556            }
1557            let Some(offset) = item_offsets.get(index) else {
1558                break;
1559            };
1560            let item_end = item_offsets
1561                .get(index + 1)
1562                .map_or(lines.line_count(), |next| next.lines);
1563            let child_segments = match (item, items.get(index)) {
1564                (
1565                    TranscriptItem::Assistant {
1566                        children,
1567                        child_times,
1568                        ..
1569                    },
1570                    Some(cached),
1571                ) if !children.is_empty()
1572                    && child_times.len() == children.len()
1573                    && cached.child_lines.len() == children.len()
1574                    && cached.key.id == item.id()
1575                    && cached.key.version == item.version() =>
1576                {
1577                    Some((
1578                        child_times,
1579                        &cached.child_lines,
1580                        item_end.saturating_sub(cached.layout.lines.line_count()),
1581                    ))
1582                }
1583                _ => None,
1584            };
1585            if let Some((child_times, child_lines, body_start)) = child_segments {
1586                for (child, time) in child_times.iter().enumerate() {
1587                    while next < events.len() && events[next].time < *time {
1588                        let placement = if child == 0 {
1589                            (offset.lines, &events[next], None)
1590                        } else {
1591                            (
1592                                body_start + child_lines[child],
1593                                &events[next],
1594                                Some((index, child)),
1595                            )
1596                        };
1597                        placements.push(placement);
1598                        next += 1;
1599                    }
1600                }
1601            } else if let Some(time) = state.item_time(item.id()) {
1602                while next < events.len() && events[next].time < time {
1603                    placements.push((offset.lines, &events[next], None));
1604                    next += 1;
1605                }
1606            }
1607        }
1608        placements.extend(
1609            events[next..]
1610                .iter()
1611                .map(|event| (lines.line_count(), event, None)),
1612        );
1613        // Positions already ascend with segment order; the stable sort only
1614        // guards stale cached offsets.
1615        placements.sort_by_key(|(position, _, _)| *position);
1616
1617        let mut spliced = SplicedRows::default();
1618        let mut placements = placements.into_iter().peekable();
1619        while let Some((position, event, resume)) = placements.next() {
1620            let position = position.min(lines.line_count());
1621            let before = spliced.rows.len();
1622            let mut resume = resume;
1623            let mut event = Some(event);
1624            while let Some(row) = event.take() {
1625                // The line above: this splice's previous row, else the
1626                // transcript line before the splice point.
1627                let above = if spliced.rows.len() > before {
1628                    spliced.rows.last()
1629                } else {
1630                    position.checked_sub(1).and_then(|line| lines.line_at(line))
1631                };
1632                let needs_separator = above.is_some_and(|line| {
1633                    line.spans
1634                        .iter()
1635                        .any(|span| !span.content.trim().is_empty())
1636                });
1637                if needs_separator {
1638                    spliced.rows.push(Line::default());
1639                }
1640                let block_id = BlockId::DescendantEvent {
1641                    session: row.session,
1642                    item: row.item,
1643                };
1644                let key = EventRowKey {
1645                    level: row.level,
1646                    repeat: row.repeat,
1647                    expanded: expanded.is_some_and(|blocks| blocks.contains(&block_id)),
1648                    width,
1649                    theme: theme.key(),
1650                };
1651                let cached = match row_layouts.entry((row.session, row.item)) {
1652                    std::collections::hash_map::Entry::Occupied(entry)
1653                        if entry.get().key == key && entry.get().text == row.text =>
1654                    {
1655                        entry.into_mut()
1656                    }
1657                    entry => entry
1658                        .insert_entry(CachedEventRow {
1659                            key,
1660                            text: row.text.clone(),
1661                            layout: event_row_layout(
1662                                block_id, row.level, &row.text, row.repeat, expanded, width, theme,
1663                            ),
1664                        })
1665                        .into_mut(),
1666                };
1667                let row_layout = &cached.layout;
1668                let row_start = position + spliced.rows.len();
1669                spliced
1670                    .regions
1671                    .extend(row_layout.regions.iter().map(|region| BlockRegion {
1672                        start_line: row_start + region.start_line,
1673                        end_line: row_start + region.end_line,
1674                        ..*region
1675                    }));
1676                spliced.rows.extend(row_layout.lines.iter().cloned());
1677                if let Some((_, row, row_resume)) =
1678                    placements.next_if(|(next, _, _)| (*next).min(lines.line_count()) == position)
1679                {
1680                    resume = resume.or(row_resume);
1681                    event = Some(row);
1682                }
1683            }
1684            if let Some((item, child)) = resume
1685                && let Some(item) = state.transcript.get(item)
1686            {
1687                spliced.rows.push(Line::default());
1688                spliced
1689                    .rows
1690                    .extend(continuation_header(item, child, width, theme, model_names));
1691            }
1692            let inserted = spliced.rows.len() - before;
1693            if inserted > 0 {
1694                spliced.shifts.push((position, inserted));
1695                spliced.starts.push(position + before);
1696            }
1697        }
1698        spliced
1699    }

1763    pub(super) fn render_conversation(&mut self, frame: &mut ratatui::Frame, area: Rect) {
1764        // The two rightmost inner columns always belong to the scrollbar: a
1765        // blank margin and the track. Content layout, block hit regions and
1766        // highlight rects are built at that reduced width and never extend
1767        // into it, so the track can be grabbed without hitting blocks — and
1768        // because the reservation is constant, the wrap width (a
1769        // `LayoutCacheKey` input) never flips when content starts or stops
1770        // overflowing.
1771        let width = pane_text_width(area.width);
1772        // A new-session draft shows the fresh session it will create, not the
1773        // conversation still selected behind it.
1774        let viewed = self.viewed_session();
1775        let session_present = self.new_session_draft.is_some()
1776            || viewed.is_some_and(|session_id| self.store.sessions.contains_key(&session_id));
1777        let empty_layout = ChunkedLayout::from(TranscriptLayout {
1778            lines: empty_conversation_lines(session_present, width, &self.theme),
1779            regions: Vec::new(),
1780            user_regions: Vec::new(),
1781        });
1782        let clock_bucket = self.clock_bucket();
1783        let draft_agent = self.draft.as_ref().map(|draft| draft.agent.clone());
1784        let anchor_key = self.layout_cache.key;
1785        // The offset addresses the last drawn (spliced) lines; the anchor
1786        // addresses the cached layout, so map the offset back first.
1787        let (unspliced_offset, spliced_depth) = unsplice_line(
1788            self.conversation_scroll.offset,
1789            &self.conversation_splice_shifts,
1790        );
1791        let anchor = (!self.conversation_scroll.following)
1792            .then(|| self.layout_cache.scroll_anchor(unspliced_offset))
1793            .flatten();
1794        let mut layout_changed = false;
1795        let mut cached_layout = false;
1796        if let Some(session_id) = viewed {
1797            self.select_layout_cache(session_id);
1798        }
1799        if let Some((changed, cached)) =
1800            self.ensure_viewed_layout(viewed, width, clock_bucket, draft_agent.as_ref())
1801        {
1802            layout_changed = changed;
1803            cached_layout = cached;
1804        }
1805        let layout = if cached_layout {
1806            &self.layout_cache.layout
1807        } else {
1808            &empty_layout
1809        };
1810        // Descendant rows read through to the layout instead of copying it;
1811        // the splice itself is rebuilt only when its inputs change.
1812        let mut splice = std::mem::take(&mut self.splice_cache);
1813        self.refresh_splice(&mut splice, layout, cached_layout, width);
1814        let mut notice_lines = self.notice_lines(width);
1815        // Notices follow the same rhythm as transcript items: one blank row
1816        // between real content and the first notice block.
1817        if !layout.lines.is_empty() && !notice_lines.is_empty() {
1818            notice_lines.insert(0, Line::default());
1819        }
1820        let spliced = &splice.rows;
1821        let splice_shifts = spliced.shifts.as_slice();
1822        let layout_len = spliced.len(layout.lines.len());
1823        let viewport = Rect::new(
1824            area.x.saturating_add(1),
1825            area.y.saturating_add(1),
1826            width,
1827            area.height.saturating_sub(2),
1828        );
1829        let scrollbar_track = pane_scrollbar_track(area, viewport);
1830        let content_height = layout_len + notice_lines.len();
1831        if layout_changed
1832            && anchor_key == self.layout_cache.key
1833            && let Some((point, row_offset)) = anchor
1834            && let Some(line) = self.layout_cache.anchor_line(point)
1835        {
1836            // `line` addresses the unspliced layout; rows spliced in above
1837            // the anchored row shift it down by their count. A view resting
1838            // inside spliced rows keeps its depth into them.
1839            let line = line.saturating_add(row_offset);
1840            self.conversation_scroll.offset = match spliced_depth {
1841                Some(depth) => splice_shifted(line, splice_shifts, false).saturating_add(depth),
1842                None => splice_shifted(line, splice_shifts, true),
1843            };
1844        }
1845        self.conversation_splice_shifts.clear();
1846        self.conversation_splice_shifts
1847            .extend_from_slice(splice_shifts);
1848        self.conversation_scroll
1849            .clamp(content_height, viewport.height);
1850        // Code rendered plain while out of view is highlighted as it comes
1851        // within a viewport of the visible rows, before this frame draws
1852        // it. Highlighting moves no row, so the scroll position holds.
1853        let (visible, _) = unsplice_line(
1854            self.conversation_scroll.offset,
1855            &self.conversation_splice_shifts,
1856        );
1857        let height = usize::from(viewport.height.max(1));
1858        if cached_layout
1859            && self
1860                .layout_cache
1861                .request_highlights(visible.saturating_sub(height)..visible + 2 * height)
1862        {
1863            self.ensure_viewed_layout(viewed, width, clock_bucket, draft_agent.as_ref());
1864            self.refresh_splice(&mut splice, &self.layout_cache.layout, true, width);
1865        }
1866        let layout = if cached_layout {
1867            &self.layout_cache.layout
1868        } else {
1869            &empty_layout
1870        };
1871        let spliced = &splice.rows;
1872        let line_at = |index: usize| spliced.line(&layout.lines, index);
1873        let layout_len = spliced.len(layout.lines.len());

2283/// Remove the oldest of the `parked` layout caches (oldest first) until at
2284/// most `max_caches` remain holding at most `max_lines` rows together, and
2285/// return them.
2286fn evict_parked_layouts(
2287    parked: &mut Vec<LayoutCache>,
2288    max_caches: usize,
2289    max_lines: usize,
2290) -> Vec<LayoutCache> {
2291    let mut lines = parked
2292        .iter()
2293        .map(|cache| cache.layout.lines.len())
2294        .sum::<usize>();
2295    let mut evicted = 0;
2296    while evicted < parked.len() && (parked.len() - evicted > max_caches || lines > max_lines) {
2297        lines -= parked[evicted].layout.lines.len();
2298        evicted += 1;
2299    }
2300    parked.drain(..evicted).collect()
2301}

2394/// [`block_hit`] over lines read by index, so a spliced view need not be
2395/// copied into one slice.
2396pub(super) fn block_hit_at<'a>(
2397    region: BlockRegion,
2398    line_at: impl Fn(usize) -> Option<&'a Line<'static>>,
2399    viewport: Rect,
2400    scroll_offset: usize,
2401) -> Option<BlockHit> {
2402    let viewport_end = scroll_offset.saturating_add(usize::from(viewport.height));
2403    let start = region.start_line.max(scroll_offset);
2404    let end = region.end_line.min(viewport_end);
2405    let header_end = region
2406        .header_lines
2407        .map_or(start.saturating_add(1), |height| {
2408            region.start_line.saturating_add(height)
2409        })
2410        .min(end);
2411    // The block's own content columns: everything right of its leading gutter
2412    // and left of the viewport edge, which already stops before the reserved
2413    // scrollbar columns. Highlights share this rect with the band and the
2414    // selection so no painted row ever covers a `│` border or the track.
2415    // Hit rectangles stay viewport-wide on purpose: the gutter and the blank
2416    // tail of a row must still click, toggle and drag.
2417    //
2418    // The block's header rows stand behind their gutter, and only behind it:
2419    // a builder that counted its own spans says how wide that is, which is the
2420    // only reading that cannot mistake `"│ "` arriving as command output — a
2421    // tree listing, a diff body — for a second gutter and hide real columns.
2422    // Regions whose builders do not count fall back to reading the row; the
2423    // walk is bounded to header rows, which carry the block's own gutter, and
2424    // resolves the one case a bare line cannot settle — leading spaces that
2425    // could be a continuation indent or indentation inside the content — as
2426    // content, matching `extract_line` rather than hiding columns a reader
2427    // could have selected.
2428    let hovered = start..header_end.max(start);
2429    let gutter = region.header_gutter.unwrap_or_else(|| {
2430        hovered
2431            .filter_map(&line_at)
2432            .map(leading_gutter_columns)
2433            .max()
2434            .unwrap_or(0)
2435    });
2436    let content = Rect::new(
2437        viewport.x.saturating_add(gutter),
2438        viewport.y,
2439        viewport.width.saturating_sub(gutter),
2440        viewport.height,
2441    );
2442    let action_rect = match region.id {
2443        BlockId::Tool(_) if region.start_line >= scroll_offset && region.start_line < end => {
2444            line_at(region.start_line).and_then(|line| {
2445                let text = line
2446                    .spans
2447                    .iter()
2448                    .map(|span| span.content.as_ref())
2449                    .collect::<String>();
2450                let at = text
2451                    .trim_end()
2452                    .strip_suffix(tool_rows::BACKGROUND_BUTTON)?
2453                    .len();
2454                let column = UnicodeWidthStr::width(&text[..at]);
2455                Some(Rect::new(
2456                    viewport
2457                        .x
2458                        .saturating_add(u16::try_from(column).unwrap_or(u16::MAX)),
2459                    viewport.y
2460                        + u16::try_from(region.start_line - scroll_offset).unwrap_or(u16::MAX),
2461                    u16::try_from(UnicodeWidthStr::width(tool_rows::BACKGROUND_BUTTON))
2462                        .unwrap_or(u16::MAX),
2463                    1,
2464                ))
2465            })
2466        }
2467        _ => None,
2468    };
2469    // A memory URL the title row names links to the Memory panel.
2470    let link_rect = match region.id {
2471        BlockId::Tool(_) if region.start_line >= scroll_offset && region.start_line < end => {
2472            line_at(region.start_line).and_then(|line| {
2473                let text = line
2474                    .spans
2475                    .iter()
2476                    .map(|span| span.content.as_ref())
2477                    .collect::<String>();
2478                let span = cookie_agent_client_state::tool_header::memory_url_span(&text)?;
2479                let column = UnicodeWidthStr::width(&text[..span.start]);
2480                Some(Rect::new(
2481                    viewport
2482                        .x
2483                        .saturating_add(u16::try_from(column).unwrap_or(u16::MAX)),
2484                    viewport.y
2485                        + u16::try_from(region.start_line - scroll_offset).unwrap_or(u16::MAX),
2486                    u16::try_from(UnicodeWidthStr::width(&text[span])).unwrap_or(u16::MAX),
2487                    1,
2488                ))
2489            })
2490        }
2491        _ => None,
2492    };
2493    (start < end).then(|| BlockHit {
2494        action_rect,
2495        link_rect,
2496        rect: Rect::new(
2497            viewport.x,
2498            viewport.y + u16::try_from(start - scroll_offset).unwrap_or(u16::MAX),
2499            viewport.width,
2500            u16::try_from(end - start).unwrap_or(u16::MAX),
2501        ),
2502        id: region.id,
2503        toggle_rect: if matches!(
2504            region.id,
2505            BlockId::ToolOutput { .. }
2506                | BlockId::ToolFull { .. }
2507                | BlockId::ToolSession { .. }
2508                | BlockId::ToolMemory { .. }
2509        ) {
2510            // Action regions are single clickable rows of a tool body.
2511            (region.start_line >= scroll_offset).then(|| {
2512                Rect::new(
2513                    viewport.x,
2514                    viewport.y + u16::try_from(start - scroll_offset).unwrap_or(u16::MAX),
2515                    viewport.width,
2516                    1,
2517                )
2518            })
2519        } else {
2520            (start < header_end).then(|| {
2521                Rect::new(
2522                    viewport.x,
2523                    viewport.y + u16::try_from(start - scroll_offset).unwrap_or(u16::MAX),
2524                    viewport.width,
2525                    u16::try_from(header_end - start).unwrap_or(u16::MAX),
2526                )
2527            })
2528        },
2529        hover_rect: (start < header_end).then(|| {
2530            Rect::new(
2531                content.x,
2532                viewport.y + u16::try_from(start - scroll_offset).unwrap_or(u16::MAX),
2533                content.width,
2534                u16::try_from(header_end - start).unwrap_or(u16::MAX),
2535            )
2536        }),
2537    })
2538}

2540/// The user-message analogue of [`block_hit`]: clip a message's logical-line
2541/// range to the visible window so its rows open the copy/revert/fork menu.
2542/// The hover paints the message's content columns only, right of the widest
2543/// leading gutter among its visible rows, like a block header's.
2544pub(super) fn user_message_hit<'a>(
2545    region: UserRegion,
2546    line_at: impl Fn(usize) -> Option<&'a Line<'static>>,
2547    viewport: Rect,
2548    scroll_offset: usize,
2549) -> Option<UserMessageHit> {
2550    let viewport_end = scroll_offset.saturating_add(usize::from(viewport.height));
2551    let start = region.start_line.max(scroll_offset);
2552    let end = region.end_line.min(viewport_end);
2553    let rect = Rect::new(
2554        viewport.x,
2555        viewport.y + u16::try_from(start.saturating_sub(scroll_offset)).unwrap_or(u16::MAX),
2556        viewport.width,
2557        u16::try_from(end.saturating_sub(start)).unwrap_or(u16::MAX),
2558    );
2559    let gutter = (start..end)
2560        .filter_map(line_at)
2561        .map(leading_gutter_columns)
2562        .max()
2563        .unwrap_or(0)
2564        .min(viewport.width);
2565    (start < end).then(|| UserMessageHit {
2566        rect,
2567        hover_rect: Rect {
2568            x: rect.x.saturating_add(gutter),
2569            width: rect.width - gutter,
2570            ..rect
2571        },
2572        seq: region.seq,
2573    })
2574}
