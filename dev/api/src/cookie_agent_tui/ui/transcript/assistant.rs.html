5pub(super) fn assistant_item_layout(
6    state: &SessionState,
7    item_id: u64,
8    attribution: &crate::state::FrozenAssistantAttribution,
9    children: &[AssistantChild],
10    context: &mut TranscriptRenderContext<'_>,
11) -> ItemLayout {
12    let mut layout = ItemLayout {
13        lines: assistant_header(attribution.header().as_str(), context.width, context.theme),
14        regions: Vec::new(),
15        user_seq: None,
16    };
17    let mut previous_is_prose = None;
18    for child in children {
19        // Prose and the compact tool/thinking rows are separated by one blank
20        // gutter row whenever the reply switches between them; runs of rows
21        // stay tight. Separators sit between parts, never inside a part's
22        // line range, so streaming splices are unaffected.
23        let is_prose = matches!(child, AssistantChild::Text { .. });
24        if previous_is_prose.is_some_and(|previous| previous != is_prose) {
25            layout.lines.extend(assistant_body_line(
26                Line::default(),
27                context.width,
28                context.theme,
29            ));
30        }
31        previous_is_prose = Some(is_prose);
32        match child {
33            AssistantChild::Text { .. } | AssistantChild::Thinking { .. } => {
34                let key = assistant_part_layout_key(state, item_id, child, context);
35                let part_layout = if context
36                    .assistant_part_cache
37                    .get(&child.id())
38                    .is_some_and(|cached| cached.key == key)
39                {
40                    context.assistant_part_cache[&child.id()].layout.clone()
41                } else {
42                    let part_layout = assistant_child_layout(
43                        child,
44                        key,
45                        context.width,
46                        context.theme,
47                        context.highlighter,
48                    );
49                    context.assistant_part_cache.insert(
50                        child.id(),
51                        CachedAssistantPartLayout {
52                            key,
53                            layout: part_layout.clone(),
54                        },
55                    );
56                    *context.assistant_part_layout_passes =
57                        context.assistant_part_layout_passes.wrapping_add(1);
58                    part_layout
59                };
60                let start_line = layout.lines.len();
61                let start_region = layout.regions.len();
62                layout.lines.extend(part_layout.lines);
63                layout
64                    .regions
65                    .extend(part_layout.regions.into_iter().map(|region| BlockRegion {
66                        id: region.id,
67                        start_line: start_line + region.start_line,
68                        end_line: start_line + region.end_line,
69                        ..region
70                    }));
71                context.assistant_part_ranges.push(AssistantPartRange {
72                    id: child.id(),
73                    key,
74                    lines: start_line..layout.lines.len(),
75                    regions: start_region..layout.regions.len(),
76                });
77            }
78            AssistantChild::Tool { call_id } => {
79                let child_layout =
80                    tool_child_layout(state, Some(*call_id), *call_id, None, context);
81                let start_line = layout.lines.len();
82                layout.lines.extend(child_layout.lines);
83                layout
84                    .regions
85                    .extend(child_layout.regions.into_iter().map(|region| BlockRegion {
86                        id: region.id,
87                        start_line: start_line + region.start_line,
88                        end_line: start_line + region.end_line,
89                        ..region
90                    }));
91            }
92            AssistantChild::Attribution { resolved_model } => {
93                layout.lines.extend(attribution_line(
94                    resolved_model,
95                    context.width,
96                    context.theme,
97                ));
98            }
99            AssistantChild::Notice { text } => {
100                layout.lines.extend(assistant_body_line(
101                    Line::from(Span::styled(text.clone(), context.theme.muted())),
102                    context.width,
103                    context.theme,
104                ));
105            }
106            AssistantChild::CommittedTool {
107                turn_seq,
108                content_index,
109                name,
110            } => {
111                let child_layout = tool_child_layout(
112                    state,
113                    None,
114                    BlockKey::CommittedTool {
115                        turn_seq: *turn_seq,
116                        content_index: *content_index,
117                    },
118                    Some(name.as_str()),
119                    context,
120                );
121                let start_line = layout.lines.len();
122                layout.lines.extend(child_layout.lines);
123                layout
124                    .regions
125                    .extend(child_layout.regions.into_iter().map(|region| BlockRegion {
126                        id: region.id,
127                        start_line: start_line + region.start_line,
128                        end_line: start_line + region.end_line,
129                        ..region
130                    }));
131            }
132            AssistantChild::MediaFile {
133                turn_seq,
134                content_index,
135                file,
136            } => {
137                let child_layout = media_file_layout(*turn_seq, *content_index, file, context);
138                let start_line = layout.lines.len();
139                layout.lines.extend(child_layout.lines);
140                layout
141                    .regions
142                    .extend(child_layout.regions.into_iter().map(|region| BlockRegion {
143                        id: region.id,
144                        start_line: start_line + region.start_line,
145                        end_line: start_line + region.end_line,
146                        ..region
147                    }));
148            }
149        }
150    }
151    // The block footer closes the run: one muted, gutter-aligned row with
152    // generation speed and context use, from committed-turn usage and
153    // durable event timestamps. Passive — no region, no hover — and absent
154    // entirely when the data is missing.
155    if let Some(footer) = assistant_footer_line(state, item_id, context.width, context.theme) {
156        layout.lines.extend(footer);
157    }
158    layout
159}

161pub(super) fn assistant_part_layout_key(
162    state: &SessionState,
163    item_id: u64,
164    child: &AssistantChild,
165    context: &TranscriptRenderContext<'_>,
166) -> AssistantPartLayoutKey {
167    let block_id = match child {
168        AssistantChild::Thinking { id, .. } => Some(BlockId::Thinking(*id)),
169        AssistantChild::Text { .. }
170        | AssistantChild::Tool { .. }
171        | AssistantChild::Attribution { .. }
172        | AssistantChild::CommittedTool { .. }
173        | AssistantChild::MediaFile { .. }
174        | AssistantChild::Notice { .. } => None,
175    };
176    let streaming = matches!(child, AssistantChild::Thinking { id, .. } if state.is_open_thinking(item_id, *id));
177    let duration = match child {
178        AssistantChild::Thinking { id, .. } if !streaming => state
179            .thinking_duration(item_id, *id)
180            .filter(|duration| duration.as_secs() >= 1),
181        _ => None,
182    };
183    AssistantPartLayoutKey {
184        version: child.version(),
185        expanded: block_id
186            .is_some_and(|id| context.expanded.is_some_and(|blocks| blocks.contains(&id))),
187        streaming,
188        dots: if streaming { context.clock_bucket } else { 0 },
189        duration,
190    }
191}

193pub(super) fn splice_active_assistant_part(
194    cached: &mut CachedItemLayout,
195    state: &SessionState,
196    item: &TranscriptItem,
197    context: &mut TranscriptRenderContext<'_>,
198) -> bool {
199    let TranscriptItem::Assistant { id, children, .. } = item else {
200        return false;
201    };
202    let parts = children
203        .iter()
204        .filter(|child| {
205            matches!(
206                child,
207                AssistantChild::Text { .. } | AssistantChild::Thinking { .. }
208            )
209        })
210        .collect::<Vec<_>>();
211    if parts.len() != cached.assistant_parts.len() {
212        return false;
213    }
214    let mut dirty = None;
215    for (index, (child, range)) in parts.iter().zip(&cached.assistant_parts).enumerate() {
216        if child.id() != range.id {
217            return false;
218        }
219        let key = assistant_part_layout_key(state, *id, child, context);
220        if key != range.key && dirty.replace((index, *child, key)).is_some() {
221            return false;
222        }
223    }
224    let Some((dirty_index, child, key)) = dirty else {
225        return false;
226    };
227    if !state.is_open_assistant_part(*id, child.id()) {
228        return false;
229    }
230
231    let part_layout = assistant_child_layout(
232        child,
233        key,
234        context.width,
235        context.theme,
236        context.highlighter,
237    );
238    context.assistant_part_cache.insert(
239        child.id(),
240        CachedAssistantPartLayout {
241            key,
242            layout: part_layout.clone(),
243        },
244    );
245    *context.assistant_part_layout_passes = context.assistant_part_layout_passes.wrapping_add(1);
246
247    let old = cached.assistant_parts[dirty_index].clone();
248    let old_line_len = old.lines.len();
249    let new_line_len = part_layout.lines.len();
250    let line_delta = isize::try_from(new_line_len).unwrap_or(isize::MAX)
251        - isize::try_from(old_line_len).unwrap_or(isize::MAX);
252    cached
253        .layout
254        .lines
255        .splice(old.lines.clone(), part_layout.lines);
256
257    let new_regions = part_layout
258        .regions
259        .into_iter()
260        .map(|region| BlockRegion {
261            id: region.id,
262            start_line: old.lines.start + region.start_line,
263            end_line: old.lines.start + region.end_line,
264            ..region
265        })
266        .collect::<Vec<_>>();
267    let old_region_len = old.regions.len();
268    let new_region_len = new_regions.len();
269    cached
270        .layout
271        .regions
272        .splice(old.regions.clone(), new_regions);
273    let region_delta = isize::try_from(new_region_len).unwrap_or(isize::MAX)
274        - isize::try_from(old_region_len).unwrap_or(isize::MAX);
275    let shifted_region_start = old.regions.start + new_region_len;
276    for region in &mut cached.layout.regions[shifted_region_start..] {
277        region.start_line = region
278            .start_line
279            .checked_add_signed(line_delta)
280            .expect("assistant region offset remains valid");
281        region.end_line = region
282            .end_line
283            .checked_add_signed(line_delta)
284            .expect("assistant region offset remains valid");
285    }
286
287    let range = &mut cached.assistant_parts[dirty_index];
288    range.key = key;
289    range.lines.end = range.lines.start + new_line_len;
290    range.regions.end = range.regions.start + new_region_len;
291    for range in &mut cached.assistant_parts[dirty_index + 1..] {
292        range.lines.start = range
293            .lines
294            .start
295            .checked_add_signed(line_delta)
296            .expect("assistant line offset remains valid");
297        range.lines.end = range
298            .lines
299            .end
300            .checked_add_signed(line_delta)
301            .expect("assistant line offset remains valid");
302        range.regions.start = range
303            .regions
304            .start
305            .checked_add_signed(region_delta)
306            .expect("assistant region index remains valid");
307        range.regions.end = range
308            .regions
309            .end
310            .checked_add_signed(region_delta)
311            .expect("assistant region index remains valid");
312    }
313    true
314}

316pub(super) fn assistant_child_layout(
317    child: &AssistantChild,
318    key: AssistantPartLayoutKey,
319    width: u16,
320    theme: &Theme,
321    highlighter: &dyn Highlighter,
322) -> ItemLayout {
323    match child {
324        AssistantChild::Text { markdown, .. } => ItemLayout {
325            lines: {
326                let markdown_width = width.saturating_sub(u16::from(width >= 3) * 2);
327                crate::markdown::render_markdown_lines_width(
328                    markdown,
329                    theme,
330                    highlighter,
331                    markdown_width,
332                )
333                .into_iter()
334                .flat_map(|line| assistant_markdown_body_line(line, width, theme))
335                .collect()
336            },
337            regions: Vec::new(),
338            user_seq: None,
339        },
340        AssistantChild::Thinking { id, text, .. } => {
341            let block_id = BlockId::Thinking(*id);
342            let (body, text_rows) = thinking_body_lines(text, width, theme);
343            let hidden_lines = text_rows.max(1);
344            // While thinking streams the header animates an ellipsis; once
345            // sealed it reads "thought", with the durable elapsed time when
346            // the projection recorded one. Exactly one chevron per thinking
347            // row: `▸` collapsed, `▾` expanded, after the thinking emoji.
348            let status = if key.streaming {
349                format!("thinking{}", ".".repeat(usize::from(key.dots)))
350            } else if let Some(duration) = key.duration {
351                format!("thought for {}", format_thinking_duration(duration))
352            } else {
353                "thought".to_owned()
354            };
355            let label = if key.expanded {
356                format!("💭 ▾ {status}")
357            } else {
358                let noun = if hidden_lines == 1 { "line" } else { "lines" };
359                format!("💭 ▸ {status} ({hidden_lines} {noun} hidden)")
360            };
361            // Thinking is secondary to the reply: its row reads as muted body
362            // text whether collapsed or expanded. Expanded, it heads a panel
363            // like an expanded tool's, on the title band.
364            // Collapsed, the title keeps the same margin, so it does not
365            // shift a column when it expands.
366            let label = Line::from(Span::styled(label, theme.muted_text()));
367            let (mut lines, header_gutter) = if panel_margin(width) {
368                let title_band = theme.tool_title_background().filter(|_| key.expanded);
369                let (mut rows, gutter) = margin_rows(label, width, theme, title_band);
370                if key.expanded {
371                    for row in &mut rows {
372                        paint_band(row, gutter.min(1), width, title_band, None, true);
373                    }
374                }
375                // Hover starts on the margin, as on a tool title, so the
376                // highlight covers the padding too.
377                (rows, (gutter > 0).then_some(2))
378            } else {
379                (assistant_body_line(label, width, theme), None)
380            };
381            let header_lines = lines.len();
382            if key.expanded {
383                lines.extend(body);
384            }
385            ItemLayout {
386                regions: vec![BlockRegion {
387                    id: block_id,
388                    start_line: 0,
389                    end_line: lines.len(),
390                    header_lines: Some(header_lines),
391                    header_gutter,
392                }],
393                lines,
394                user_seq: None,
395            }
396        }
397        AssistantChild::Tool { .. }
398        | AssistantChild::Attribution { .. }
399        | AssistantChild::CommittedTool { .. }
400        | AssistantChild::MediaFile { .. }
401        | AssistantChild::Notice { .. } => {
402            unreachable!("tool children use tool_child_layout")
403        }
404    }
405}

407/// A settled thinking duration as compact text: seconds under a minute,
408/// then minutes and seconds. Sub-second spans never reach the label (they
409/// are filtered to plain "thought" by the caller).
410pub(super) fn format_thinking_duration(duration: Duration) -> String {
411    let seconds = duration.as_secs();
412    if seconds >= 60 {
413        format!("{}m {}s", seconds / 60, seconds % 60)
414    } else {
415        format!("{seconds}s")
416    }
417}

419pub(super) fn assistant_header(attribution: &str, width: u16, theme: &Theme) -> Vec<Line<'static>> {
420    // The frozen `Agent • Model` attribution wraps at tiny widths and is
421    // never reduced to a tag: it is the sole producer identity. The agent
422    // leads in the bold role style; the model trails muted so the header
423    // reads as one name, not two competing labels.
424    let (agent, model) = attribution
425        .split_once(" • ")
426        .map_or((attribution, None), |(agent, model)| (agent, Some(model)));
427    let text = if width >= 8 {
428        format!("╭─ {agent}")
429    } else {
430        agent.to_owned()
431    };
432    let gutter = (width >= 4).then_some("│ ");
433    let gutter_width = gutter.map_or(0, unicode_width::UnicodeWidthStr::width);
434    // The continuation gutter's width is reserved before wrapping, so every
435    // rendered row including its prefix fits the panel width.
436    let wrap_width = u16::try_from(
437        usize::from(width.max(1))
438            .saturating_sub(gutter_width)
439            .max(1),
440    )
441    .unwrap_or(u16::MAX);
442    let mut spans = vec![Span::styled(text, theme.assistant())];
443    if let Some(model) = model {
444        spans.push(Span::styled(format!(" • {model}"), theme.muted()));
445    }
446    spans.push(Span::raw(" "));
447    wrapped_line(Line::from(spans), wrap_width)
448        .into_iter()
449        .enumerate()
450        .map(|(index, mut line)| {
451            if index > 0
452                && let Some(gutter) = gutter
453            {
454                line.spans
455                    .insert(0, Span::styled(gutter, theme.assistant()));
456            }
457            line
458        })
459        .collect()
460}

462pub(super) fn attribution_line(
463    resolved_model: &cookie_agent_protocol::ResolvedModelRef,
464    width: u16,
465    theme: &Theme,
466) -> Vec<Line<'static>> {
467    let variant = resolved_model
468        .selection
469        .variant
470        .as_ref()
471        .map_or_else(|| "base".to_owned(), ToString::to_string);
472    let prefix = (width >= 4).then(|| vec![Span::styled("├─ ", theme.muted())]);
473    repeated_prefixed_wrapped_line(
474        prefix.unwrap_or_default(),
475        Line::from(Span::styled(
476            format!("now using {}[{variant}]", resolved_model.selection.model),
477            theme.muted(),
478        )),
479        width,
480    )
481}

483/// The assistant block's closing footer:
484/// `╰─ ⚡ 42.1 tps · 12.5K ctx · $0.0040` in
485/// muted styling — visually subordinate to the body, closing the block's
486/// gutter tree. A block whose run was interrupted gains a trailing
487/// `· interrupted`. The rate is committed output tokens over generation wall
488/// time measured between durable event timestamps, so a replayed log yields
489/// the identical row; the ctx is the total context the turn left behind
490/// (`input_tokens + output_tokens`). `None` unless every input is present:
491/// at least one turn with a positive generation span and a known
492/// end-of-turn context total.
493pub(super) fn assistant_footer_line(
494    state: &SessionState,
495    item_id: u64,
496    width: u16,
497    theme: &Theme,
498) -> Option<Vec<Line<'static>>> {
499    let metrics = state.assistant_metrics.get(&item_id)?;
500    let context_tokens = metrics.context_tokens?;
501    if metrics.timed_output_tokens == 0 || metrics.generation.is_zero() {
502        return None;
503    }
504    let tps = metrics.timed_output_tokens as f64 / metrics.generation.as_secs_f64();
505    let cost = metrics
506        .estimated_cost_pico_usd
507        .map(|cost| crate::ui::app::format_cost_usd(cost as f64 / 1_000_000_000_000.0));
508    let cost = cost.map_or_else(String::new, |cost| format!(" · {cost}"));
509    let interrupted = if state.interrupted_assistant_items.contains(&item_id) {
510        " · interrupted"
511    } else {
512        ""
513    };
514    let prefix = (width >= 4).then(|| vec![Span::styled("╰─ ", theme.muted())]);
515    Some(repeated_prefixed_wrapped_line(
516        prefix.unwrap_or_default(),
517        Line::from(Span::styled(
518            format!(
519                "⚡ {tps:.1} tps · {} ctx{cost}{interrupted}",
520                crate::ui::app::format_token_count(context_tokens),
521            ),
522            theme.muted(),
523        )),
524        width,
525    ))
526}

531    theme: &Theme,

537pub(super) fn assistant_markdown_body_line(
538    line: MarkdownLine,
539    width: u16,
540    theme: &Theme,
541) -> Vec<Line<'static>> {
542    let prefix = (width >= 3).then(|| vec![Span::styled("│ ", theme.assistant())]);
543    let prefix = prefix.unwrap_or_default();
544    match line.kind {
545        MarkdownLineKind::Prose => repeated_prefixed_wrapped_line(prefix, line.line, width),
546        MarkdownLineKind::ListItem {
547            continuation_indent,
548        } => repeated_prefixed_hanging_line(prefix, line.line, width, continuation_indent),
549        MarkdownLineKind::Code => vec![prefixed_unwrapped_line(prefix, line.line, width)],
550        MarkdownLineKind::Table => vec![prefixed_unwrapped_line(prefix, line.line, width)],
551    }
552}

554/// Expanded thinking, with the number of rows its text wraps to: muted
555/// italic text on the output band, padded like a tool panel's and followed
556/// by a clear row. Blocks too narrow to pad mark each row with a muted `│`
557/// instead; a dashed `┆` renders as slanted strokes in some fonts (macOS
558/// Terminal).
559pub(super) fn thinking_body_lines(
560    text: &str,
561    width: u16,
562    theme: &Theme,
563) -> (Vec<Line<'static>>, usize) {
564    let style = theme
565        .muted_text()
566        .add_modifier(ratatui::style::Modifier::ITALIC);
567    let text_line = |text: &str| Line::styled(text.to_owned(), style);
568    if panel_margin(width) {
569        let background = theme.terminal_background();
570        let blank = || padded_band_rows(Line::default(), width, theme, background);
571        let text_rows = text
572            .split('\n')
573            .flat_map(|text| padded_band_rows(text_line(text), width, theme, background))
574            .collect::<Vec<_>>();
575        let rows = text_rows.len();
576        let mut lines = blank();
577        lines.extend(text_rows);
578        lines.extend(blank());
579        lines.extend(assistant_body_line(Line::default(), width, theme));
580        return (lines, rows);
581    }
582    let lines = text
583        .split('\n')
584        .flat_map(|text| {
585            let prefix = if width >= 5 {
586                vec![
587                    Span::styled("│ ", theme.assistant()),
588                    Span::styled("│ ", style),
589                ]
590            } else if width >= 3 {
591                vec![Span::styled("│ ", style)]
592            } else {
593                Vec::new()
594            };
595            repeated_prefixed_wrapped_line(prefix, text_line(text), width)
596        })
597        .collect::<Vec<_>>();
598    let rows = lines.len();
599    (lines, rows)
600}

602/// One line behind the assistant `│ ` gutter on a padded band: a band column
603/// on each side of the text, which wraps between them.
604fn padded_band_rows(
605    line: Line<'static>,
606    width: u16,
607    theme: &Theme,
608    background: Option<ratatui::style::Color>,
609) -> Vec<Line<'static>> {
610    let (mut rows, gutter) = margin_rows(line, width, theme, background);
611    for row in &mut rows {
612        paint_band(row, gutter.min(1), width, background, None, true);
613    }
614    rows
615}

617/// One line behind the assistant `│ ` gutter and its one-column margin
618/// (on `background` when the row sits on a band), wrapped one column short
619/// of `width` so the right side keeps its margin too, with the number of
620/// chrome spans each row starts with (none when the gutter did not fit).
621fn margin_rows(
622    line: Line<'static>,
623    width: u16,
624    theme: &Theme,
625    background: Option<ratatui::style::Color>,
626) -> (Vec<Line<'static>>, usize) {
627    let prefix = vec![
628        Span::styled("│ ", theme.assistant()),
629        margin_span(theme, background),
630    ];
631    let wrap_width = width - 1;
632    let gutter = if gutter_fits(&prefix, &line, wrap_width) {
633        prefix.len()
634    } else {
635        0
636    };
637    (
638        repeated_prefixed_wrapped_line(prefix, line, wrap_width),
639        gutter,
640    )
641}
