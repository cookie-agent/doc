46pub(super) fn bounded_safe_display_lines<'a>(
47    lines: impl IntoIterator<Item = &'a str>,
48    style: Style,
49    max_lines: usize,
50    max_bytes: usize,
51) -> Vec<Line<'static>> {
52    let mut rendered = Vec::new();
53    let mut rendered_bytes = 0;
54    let mut fully_rendered_lines = 0usize;
55    let mut total_lines = 0usize;
56
57    for line in lines {
58        total_lines += 1;
59        if rendered.len() >= max_lines || rendered_bytes >= max_bytes {
60            continue;
61        }
62        let available = max_bytes - rendered_bytes;
63        let (sanitized, complete) = sanitized_display_prefix(line, available);
64        if complete {
65            rendered_bytes += sanitized.len();
66            rendered.push(Line::styled(sanitized, style));
67            fully_rendered_lines += 1;
68            continue;
69        }
70
71        if !sanitized.is_empty() {
72            rendered.push(Line::styled(sanitized, style));
73        }
74        rendered_bytes = max_bytes;
75    }
76
77    let omitted_lines = total_lines.saturating_sub(fully_rendered_lines);
78    if omitted_lines > 0 {
79        rendered.push(Line::styled(
80            format!("… truncated ({omitted_lines} more lines)"),
81            style,
82        ));
83    }
84    rendered
85}

91/// Control characters become U+FFFD and tabs expand to the next tab stop
92/// (see [`crate::markdown::expand_tabs_in_spans`]), so measured and drawn
93/// widths agree. `max_bytes` bounds the expanded output.
94pub(super) fn sanitized_display_prefix(text: &str, max_bytes: usize) -> (String, bool) {
95    let mut sanitized = String::with_capacity(text.len().min(max_bytes));
96    let mut column = 0;
97    for character in text.chars() {
98        if character == '\t' {
99            let advance = crate::markdown::tab_advance(column);
100            if sanitized.len().saturating_add(advance) > max_bytes {
101                return (sanitized, false);
102            }
103            sanitized.extend(std::iter::repeat_n(' ', advance));
104            column += advance;
105            continue;
106        }
107        let character = if character.is_control() {
108            '\u{FFFD}'
109        } else {
110            character
111        };
112        if sanitized.len().saturating_add(character.len_utf8()) > max_bytes {
113            return (sanitized, false);
114        }
115        sanitized.push(character);
116        column += unicode_width::UnicodeWidthChar::width(character).unwrap_or(0);
117    }
118    (sanitized, true)
119}

121pub(super) fn path_extension(path: &str) -> Option<&str> {
122    let name = path.rsplit(['/', '\\']).next()?;
123    name.rsplit_once('.')
124        .filter(|(stem, extension)| !stem.is_empty() && !extension.is_empty())
125        .map(|(_, extension)| extension)
126}

131    pub(super) metadata: &'a str,
132}

134pub(super) fn parse_read_output(detail: &str) -> Option<ReadOutput<'_>> {
135    const OPEN: &str = "<content>\n";
136    const CLOSE: &str = "</content>";
137    let open = detail.find(OPEN)?;
138    let content_start = open + OPEN.len();
139    let remaining = &detail[content_start..];
140    let close_offset = if remaining.starts_with(CLOSE) {
141        0
142    } else {
143        remaining.find("\n</content>")? + 1
144    };
145    let close = content_start + close_offset;
146    let content = detail[content_start..close].trim_end_matches('\n');
147    if !content
148        .lines()
149        .all(|line| parse_numbered_read_line(line).is_some())
150    {
151        return None;
152    }
153    Some(ReadOutput {
154        preamble: detail[..open].trim_end_matches('\n'),
155        content,
156        metadata: detail[close + CLOSE.len()..].trim_start_matches('\n'),
157    })
158}

160pub(super) fn parse_numbered_read_line(line: &str) -> Option<(usize, &str)> {
161    let (number, text) = line.split_once(": ")?;
162    Some((number.parse().ok()?, text))
163}

165pub(super) fn render_read_output(
166    read: ReadOutput<'_>,
167    language: Option<&str>,
168    section: ToolOutputSection,
169    expanded: bool,
170    budget: &mut RenderBudget,
171    future_sections: usize,
172    context: &TranscriptRenderContext<'_>,
173) -> Vec<ToolBodyLine> {
174    let preamble = || {
175        read.preamble
176            .lines()
177            .filter(|line| !line.starts_with('<') && !line.starts_with("Read file "))
178    };
179    let metadata = || read.metadata.lines().filter(|line| !line.is_empty());
180    let total_lines = preamble().count() + read.content.lines().count() + metadata().count();
181    let number_width = read
182        .content
183        .lines()
184        .filter_map(|line| parse_numbered_read_line(line).map(|(number, _)| number))
185        .max()
186        .unwrap_or(1)
187        .max(1)
188        .ilog10() as usize
189        + 1;
190    let mut renderer = SectionRenderer::new(budget, expanded, future_sections);
191    let mut output = Vec::new();
192    for line in preamble() {
193        let Some(line) = renderer.take(line) else {
194            break;
195        };
196        output.push(ToolBodyLine::wrapped(Line::from(Span::styled(
197            line,
198            context.theme.muted(),
199        ))));
200    }
201    let mut rows = Vec::new();
202    if !renderer.exhausted() {
203        for line in read.content.lines() {
204            let (number, text) = parse_numbered_read_line(line).expect("validated read row");
205            let Some(text) = renderer.take(text) else {
206                break;
207            };
208            rows.push((number, text));
209        }
210    }
211    let source = rows
212        .iter()
213        .map(|(_, text)| text.as_str())
214        .collect::<Vec<_>>()
215        .join("\n");
216    let highlighted = language.map(|language| {
217        context.highlighter.highlight_stable(
218            &crate::markdown::normalized_language(language),
219            &source,
220            context.theme,
221        )
222    });
223    for (index, (number, text)) in rows.into_iter().enumerate() {
224        let line = highlighted
225            .as_ref()
226            .and_then(|lines| lines.get(index))
227            .cloned()
228            .unwrap_or_else(|| Line::from(text));
229        output.push(ToolBodyLine::guttered_code(
230            line,
231            vec![
232                Span::styled(
233                    format!("{number:>number_width$}"),
234                    context.theme.code_gutter(),
235                ),
236                Span::styled(" │ ", context.theme.code_gutter()),
237            ],
238            vec![
239                Span::styled(" ".repeat(number_width), context.theme.code_gutter()),
240                Span::styled(" │ ", context.theme.code_gutter()),
241            ],
242        ));
243    }
244    if !renderer.exhausted() {
245        for line in metadata() {
246            let Some(line) = renderer.take(line) else {
247                break;
248            };
249            output.push(ToolBodyLine::wrapped(Line::from(Span::styled(
250                line,
251                context.theme.muted(),
252            ))));
253        }
254    }
255    let omitted = total_lines.saturating_sub(renderer.fully_rendered);
256    append_output_notice(
257        &mut output,
258        omitted,
259        expanded,
260        section,
261        renderer.budget,
262        context.theme,
263    );
264    output
265}

282pub(super) fn generic_output_lines(
283    heading: Option<&str>,
284    source: OutputText<'_>,
285    section: ToolOutputSection,
286    expanded: bool,
287    budget: &mut RenderBudget,
288    future_sections: usize,
289    theme: &Theme,
290) -> Vec<ToolBodyLine> {
291    let total_lines = usize::from(heading.is_some()) + source.original_lines;
292    let mut renderer = SectionRenderer::new(budget, expanded, future_sections);
293    let mut output = Vec::new();
294    if let Some(heading) = heading
295        && let Some(heading) = renderer.take(heading)
296    {
297        output.push(ToolBodyLine::wrapped(Line::from(heading)));
298    }
299    if !renderer.exhausted() {
300        for line in source.text.lines() {
301            let Some(line) = renderer.take(line) else {
302                break;
303            };
304            output.push(ToolBodyLine::wrapped(Line::from(line)));
305        }
306    }
307    let omitted = total_lines.saturating_sub(renderer.fully_rendered);
308    append_output_notice(
309        &mut output,
310        omitted,
311        expanded,
312        section,
313        renderer.budget,
314        theme,
315    );
316    output
317}

319pub(super) fn append_output_notice(
320    output: &mut Vec<ToolBodyLine>,
321    omitted: usize,
322    expanded: bool,
323    section: ToolOutputSection,
324    budget: &mut RenderBudget,
325    theme: &Theme,
326) {
327    let notice = if omitted > 0 && expanded {
328        format!("… {omitted} more lines (maximum shown; click to collapse)")
329    } else if omitted > 0 {
330        format!("… {omitted} more lines (click to expand)")
331    } else if expanded {
332        "▴ click to collapse".to_owned()
333    } else {
334        return;
335    };
336    if budget.consume_notice(&notice) {
337        output.push(ToolBodyLine::toggle(
338            Line::from(Span::styled(notice, theme.muted())),
339            section,
340        ));
341    }
342}
