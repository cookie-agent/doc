46pub(super) fn bounded_safe_display_lines<'a>(
47    lines: impl IntoIterator<Item = &'a str>,
48    style: Style,
49    max_lines: usize,
50    max_bytes: usize,
51) -> Vec<Line<'static>> {
52    let mut rendered = Vec::new();
53    let mut rendered_bytes = 0;
54    let mut fully_rendered_lines = 0usize;
55    let mut total_lines = 0usize;
56
57    for line in lines {
58        total_lines += 1;
59        if rendered.len() >= max_lines || rendered_bytes >= max_bytes {
60            continue;
61        }
62        let available = max_bytes - rendered_bytes;
63        let (sanitized, complete) = sanitized_display_prefix(line, available);
64        if complete {
65            rendered_bytes += sanitized.len();
66            rendered.push(Line::styled(sanitized, style));
67            fully_rendered_lines += 1;
68            continue;
69        }
70
71        if !sanitized.is_empty() {
72            rendered.push(Line::styled(sanitized, style));
73        }
74        rendered_bytes = max_bytes;
75    }
76
77    let omitted_lines = total_lines.saturating_sub(fully_rendered_lines);
78    if omitted_lines > 0 {
79        rendered.push(Line::styled(
80            format!("… truncated ({omitted_lines} more lines)"),
81            style,
82        ));
83    }
84    rendered
85}

91/// Control characters become U+FFFD and tabs expand to the next tab stop
92/// (see [`crate::markdown::expand_tabs_in_spans`]), so measured and drawn
93/// widths agree. `max_bytes` bounds the expanded output.
94pub(super) fn sanitized_display_prefix(text: &str, max_bytes: usize) -> (String, bool) {
95    let mut sanitized = String::with_capacity(text.len().min(max_bytes));
96    let mut column = 0;
97    for character in text.chars() {
98        if character == '\t' {
99            let advance = crate::markdown::tab_advance(column);
100            if sanitized.len().saturating_add(advance) > max_bytes {
101                return (sanitized, false);
102            }
103            sanitized.extend(std::iter::repeat_n(' ', advance));
104            column += advance;
105            continue;
106        }
107        let character = if character.is_control() {
108            '\u{FFFD}'
109        } else {
110            character
111        };
112        if sanitized.len().saturating_add(character.len_utf8()) > max_bytes {
113            return (sanitized, false);
114        }
115        sanitized.push(character);
116        column += unicode_width::UnicodeWidthChar::width(character).unwrap_or(0);
117    }
118    (sanitized, true)
119}

121pub(super) fn path_extension(path: &str) -> Option<&str> {
122    let name = path.rsplit(['/', '\\']).next()?;
123    name.rsplit_once('.')
124        .filter(|(stem, extension)| !stem.is_empty() && !extension.is_empty())
125        .map(|(_, extension)| extension)
126}

128pub(super) fn append_output_notice(
129    output: &mut Vec<ToolBodyLine>,
130    omitted: usize,
131    expanded: bool,
132    section: ToolOutputSection,
133    budget: &mut RenderBudget,
134    theme: &Theme,
135) {
136    let notice = if omitted > 0 && expanded {
137        format!("… {omitted} more lines (maximum shown; click to collapse)")
138    } else if omitted > 0 {
139        format!("… {omitted} more lines (click to expand)")
140    } else if expanded {
141        "▴ click to collapse".to_owned()
142    } else {
143        return;
144    };
145    if budget.consume_notice(&notice) {
146        output.push(ToolBodyLine::toggle(
147            Line::from(Span::styled(notice, theme.muted())),
148            section,
149        ));
150    }
151}
