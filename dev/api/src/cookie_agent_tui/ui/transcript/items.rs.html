7pub(super) fn item_layout_key(
8    state: &SessionState,
9    item: &TranscriptItem,
10    expanded: Option<&HashSet<BlockId>>,
11    clock_bucket: u8,
12    model_names: &ModelDisplayNames,
13    transcript: TranscriptMode,
14    fold: u64,
15) -> ItemLayoutKey {
16    ItemLayoutKey {
17        id: item.id(),
18        version: item.version(),
19        fold,
20        interaction: item_interaction(item, expanded),
21        clock: if item_is_live(state, item, transcript) {
22            clock_bucket
23        } else {
24            0
25        },
26        model_names: item_model_names_revision(item, model_names),
27    }
28}

30// Each argument is an independent input of the cached layout, as for
31// `ensure_cached_transcript_layout_in`.
32#[allow(clippy::too_many_arguments)]
33pub(super) fn item_layout_key_matches(
34    key: &ItemLayoutKey,
35    state: &SessionState,
36    item: &TranscriptItem,
37    expanded: Option<&HashSet<BlockId>>,
38    clock_bucket: u8,
39    model_names: &ModelDisplayNames,
40    transcript: TranscriptMode,
41    fold: u64,
42) -> bool {
43    key.id == item.id()
44        && key.version == item.version()
45        && key.fold == fold
46        && key.model_names == item_model_names_revision(item, model_names)
47        && key.clock
48            == if item_is_live(state, item, transcript) {
49                clock_bucket
50            } else {
51                0
52            }
53        && item_interaction_matches(&key.interaction, item, expanded)
54}

56/// What an item's layout reads of the reduced mode's folds: where it stands
57/// in one, whether the fold is open, and on a fold's first block what its
58/// work row says. Zero outside the folds.
59pub(super) fn item_fold_key(
60    folds: &FoldPlan,
61    work_rows: &WorkRows,
62    item: &TranscriptItem,
63    expanded: Option<&HashSet<BlockId>>,
64) -> u64 {
65    use std::hash::{Hash as _, Hasher as _};
66    let Some(fold) = folds.get(item.id()) else {
67        return 0;
68    };
69    let mut hasher = std::hash::DefaultHasher::new();
70    fold.hash(&mut hasher);
71    let keys = match fold {
72        ItemFold::Row { key } => vec![(*key, false)],
73        ItemFold::Block(block) => block
74            .children
75            .iter()
76            .filter_map(|child| match child {
77                ChildFold::Work { key, starts } => Some((*key, *starts)),
78                ChildFold::Shown => None,
79            })
80            .collect(),
81    };
82    for (key, starts) in keys {
83        expanded
84            .is_some_and(|blocks| blocks.contains(&BlockId::work(key)))
85            .hash(&mut hasher);
86        if starts && let Some((row, fresh)) = work_rows.get(&key) {
87            row.summary.label().hash(&mut hasher);
88            row.summary.failed.hash(&mut hasher);
89            row.lines.hash(&mut hasher);
90            fresh.hash(&mut hasher);
91        }
92    }
93    hasher.finish()
94}

96/// Only assistant items render model labels (their headers and "now using"
97/// rows), so only they relayout when display names change; their cached
98/// prose and thinking parts are reused.
99fn item_model_names_revision(item: &TranscriptItem, model_names: &ModelDisplayNames) -> u64 {
100    if matches!(item, TranscriptItem::Assistant { .. }) {
101        model_names.revision()
102    } else {
103        0
104    }

125pub(super) fn for_each_item_block_id(
126    item: &TranscriptItem,
127    mut visit: impl FnMut(BlockId) -> bool,
128) -> bool {
129    match item {
130        TranscriptItem::Assistant { children, .. } => {
131            for child in children {
132                let id = match child {
133                    AssistantChild::Thinking { id, .. } => Some(BlockId::Thinking(*id)),
134                    AssistantChild::Tool { call_id } => Some(BlockId::Tool(*call_id)),
135                    AssistantChild::MediaFile {
136                        turn_seq,
137                        content_index,
138                        ..
139                    } => Some(BlockId::MediaFile {
140                        turn_seq: *turn_seq,
141                        content_index: *content_index,
142                    }),
143                    AssistantChild::Text { .. }
144                    | AssistantChild::Attribution { .. }
145                    | AssistantChild::CommittedTool { .. }
146                    | AssistantChild::Notice { .. } => None,
147                };
148                let Some(id) = id else {
149                    continue;
150                };
151                if !visit(id) {
152                    return false;
153                }
154                if let BlockId::Tool(call_id) = id {
155                    for section in [
156                        ToolOutputSection::Detail,
157                        ToolOutputSection::Stdout,
158                        ToolOutputSection::Stderr,
159                    ] {
160                        if !visit(BlockId::ToolOutput { call_id, section }) {
161                            return false;
162                        }
163                    }
164                }
165            }
166            true
167        }
168        TranscriptItem::Compacting { seq, .. } => visit(BlockId::Compacting(*seq)),
169        TranscriptItem::Compaction { seq, .. } | TranscriptItem::AgentCompaction { seq, .. } => {
170            visit(BlockId::Compaction(*seq))
171        }
172        TranscriptItem::AgentSessionMessage { seq, .. } => {
173            visit(BlockId::AgentSessionMessage(*seq))
174        }
175        TranscriptItem::AgentBackgroundEnded { seq, .. } => {
176            visit(BlockId::AgentBackgroundEnded(*seq))
177        }
178        TranscriptItem::PluginMessage { seq, .. } => visit(BlockId::PluginMessage(*seq)),
179        TranscriptItem::AgentMd { seq, .. } => visit(BlockId::AgentMd(*seq)),
180        TranscriptItem::SkillLoaded { seq, .. } => visit(BlockId::SkillLoaded(*seq)),
181        TranscriptItem::ModelInput { seq, .. } => visit(BlockId::ModelInput(*seq)),
182        TranscriptItem::MemoryChange { seq, .. } => visit(BlockId::MemoryChange(*seq)),
183        TranscriptItem::ProducerMessage { message_id, .. } => {
184            visit(BlockId::ProducerMessage(*message_id))
185        }
186        TranscriptItem::Event { id, .. } => visit(BlockId::Event(*id)),
187        TranscriptItem::User { .. }
188        | TranscriptItem::Goal { .. }
189        | TranscriptItem::Plan { .. }
190        | TranscriptItem::AcpNotice { .. } => true,
191    }
192}

194pub(super) fn item_interaction_matches(
195    expected: &[(BlockId, bool)],
196    item: &TranscriptItem,
197    expanded: Option<&HashSet<BlockId>>,
198) -> bool {
199    let mut index = 0;
200    let complete = for_each_item_block_id(item, |id| {
201        let matches = expected.get(index).is_some_and(|entry| {
202            *entry == (id, expanded.is_some_and(|blocks| blocks.contains(&id)))
203        });
204        index += 1;
205        matches
206    });
207    complete && index == expected.len()
208}

220pub(super) fn item_interaction(
221    item: &TranscriptItem,
222    expanded: Option<&HashSet<BlockId>>,
223) -> Vec<(BlockId, bool)> {
224    let mut interaction = Vec::new();
225    for_each_item_block_id(item, |id| {
226        interaction.push((id, expanded.is_some_and(|blocks| blocks.contains(&id))));
227        true
228    });
229    interaction
230}

232/// Whether one transcript item owns live content — a still-streaming
233/// thinking part, a running tool row, or in the reduced mode a work row
234/// whose time ticks — and so re-renders on each animation clock bucket.
235pub(super) fn item_is_live(
236    state: &SessionState,
237    item: &TranscriptItem,
238    transcript: TranscriptMode,
239) -> bool {
240    match item {
241        TranscriptItem::Assistant { id, .. }
242            if transcript == TranscriptMode::Reduced && state.assistant_live(*id) =>
243        {
244            true
245        }
246        TranscriptItem::Assistant { id, children, .. } => {
247            children.iter().any(|child| match child {
248                AssistantChild::Thinking { id: part_id, .. } => {
249                    state.is_open_thinking(*id, *part_id)
250                }
251                AssistantChild::Tool { call_id } => state
252                    .tools
253                    .get(call_id)
254                    .is_some_and(|tool| tool.status == ToolStatus::Running),
255                AssistantChild::Text { .. }
256                | AssistantChild::Attribution { .. }
257                | AssistantChild::CommittedTool { .. }
258                | AssistantChild::MediaFile { .. }
259                | AssistantChild::Notice { .. } => false,
260            })
261        }
262        TranscriptItem::Compacting { .. } => true,
263        TranscriptItem::User { .. }
264        | TranscriptItem::Event { .. }
265        | TranscriptItem::Compaction { .. }
266        | TranscriptItem::PluginMessage { .. }
267        | TranscriptItem::AgentMd { .. }
268        | TranscriptItem::SkillLoaded { .. }
269        | TranscriptItem::ModelInput { .. }
270        | TranscriptItem::MemoryChange { .. }
271        | TranscriptItem::Goal { .. }
272        | TranscriptItem::Plan { .. }
273        | TranscriptItem::AcpNotice { .. }
274        | TranscriptItem::AgentCompaction { .. }
275        | TranscriptItem::AgentSessionMessage { .. }
276        | TranscriptItem::AgentBackgroundEnded { .. }
277        | TranscriptItem::ProducerMessage { .. } => false,
278    }
279}

299pub(super) fn transcript_item_layout(
300    state: &SessionState,
301    item: &TranscriptItem,
302    context: &mut TranscriptRenderContext<'_>,
303) -> ItemLayout {
304    // A background row inside a folded run's work shows when it opens.
305    if let Some(ItemFold::Row { key }) = context.folds.get(item.id())
306        && !context
307            .expanded
308            .is_some_and(|blocks| blocks.contains(&BlockId::work(*key)))
309    {
310        return ItemLayout::default();
311    }
312    match item {
313        TranscriptItem::User { markdown, seq, .. } => ItemLayout {
314            lines: {
315                let width = context.width;
316                let theme = context.theme;
317                let mut lines = role_block(Role::User, Vec::new(), width, theme);
318                lines.extend(
319                    crate::markdown::render_markdown_lines_width(
320                        markdown,
321                        theme,
322                        context.highlighter,
323                        markdown_body_width(width),
324                    )
325                    .into_iter()
326                    .flat_map(|line| markdown_body_line(line, width, theme.user())),
327                );
328                lines
329            },
330            regions: Vec::new(),
331            user_seq: Some(*seq),
332        },
333        TranscriptItem::Assistant {
334            id,
335            attribution,
336            children,
337            ..
338        } => assistant_item_layout(state, *id, attribution, children, context).flatten(),
339        TranscriptItem::Event {
340            id,
341            level,
342            text,
343            repeat,
344            ..
345        } => {
346            // Level filtering is a pure view concern: the row stays in the
347            // session projection and reappears when the threshold is lowered.
348            if *level < context.minimum_event_level {
349                return ItemLayout::default();
350            }
351            event_row_layout(
352                BlockId::Event(*id),
353                *level,
354                text,
355                *repeat,
356                context.expanded,
357                context.width,
358                context.theme,
359            )
360        }
361        TranscriptItem::Compacting { seq, source, .. } => compacting_layout(*seq, source, context),
362        TranscriptItem::Compaction { seq, commit, .. } => compaction_layout(*seq, commit, context),
363        TranscriptItem::AgentCompaction { seq, summary, .. } => {
364            agent_compaction_layout(*seq, summary.as_deref(), context)
365        }
366        TranscriptItem::AgentSessionMessage { seq, message, .. } => {
367            agent_session_message_layout(*seq, message, context)
368        }
369        TranscriptItem::AgentBackgroundEnded {
370            seq,
371            ended_at,
372            ended,
373            ..
374        } => agent_background_ended_layout(*seq, *ended_at, ended, context),
375        TranscriptItem::PluginMessage {
376            seq, role, input, ..
377        } => plugin_message_layout(*seq, *role, input, context),
378        TranscriptItem::AgentMd {
379            seq, entries, text, ..
380        } => agent_md_layout(*seq, entries, text, context),
381        TranscriptItem::SkillLoaded {
382            seq,
383            name,
384            source_path,
385            args,
386            rendered_body,
387            ..
388        } => skill_loaded_layout(*seq, name, source_path, args, rendered_body, context),
389        TranscriptItem::ModelInput {
390            seq,
391            kind,
392            header,
393            text,
394            ..
395        } => model_input_layout(*seq, *kind, header, text, context),
396        TranscriptItem::MemoryChange {
397            seq, change, url, ..
398        } => memory_change_layout(*seq, *change, url, context),
399        TranscriptItem::Plan { entries, .. } => ItemLayout {
400            lines: plan_layout(entries, context.width, context.theme),
401            regions: Vec::new(),
402            user_seq: None,
403        },
404        TranscriptItem::AcpNotice { notice, .. } => ItemLayout {
405            lines: acp_notice_layout(notice, context.width, context.theme),
406            regions: Vec::new(),
407            user_seq: None,
408        },
409        TranscriptItem::Goal {
410            goal, activation, ..
411        } => ItemLayout {
412            lines: if *activation {
413                goal_activation_layout(goal, context.width, context.theme)
414            } else {
415                goal_layout(goal, context.width, context.theme)
416            },
417            regions: Vec::new(),
418            user_seq: None,
419        },
420        TranscriptItem::ProducerMessage {
421            message_id,
422            producer_owner,
423            mode,
424            body,
425            summary,
426            status,
427            ..
428        } => match status {
429            // A claim means the running request already carries the message,
430            // so it shows before the response streams in, not after the turn
431            // commits. Claims leave the queue before the request starts, and
432            // the reducer anchored the row at admission, so it lands above
433            // the response and never shows twice.
434            crate::state::ProducerMessageStatus::Pending
435            | crate::state::ProducerMessageStatus::Admitted => ItemLayout::default(),
436            crate::state::ProducerMessageStatus::Claimed
437            | crate::state::ProducerMessageStatus::Consumed => producer_message_layout(
438                *message_id,
439                producer_owner,
440                *mode,
441                body,
442                &producer_summary(producer_owner, *mode, summary.as_deref()),
443                *status,
444                context,
445            ),
446            crate::state::ProducerMessageStatus::Discarded
447                if context.minimum_event_level == crate::state::EventLevel::Debug =>
448            {
449                discarded_producer_message_layout(
450                    producer_owner,
451                    *mode,
452                    context.width,
453                    context.theme,
454                )
455            }
456            crate::state::ProducerMessageStatus::Discarded => ItemLayout::default(),
457        },
458    }
