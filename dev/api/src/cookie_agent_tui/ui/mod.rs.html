16/// The most Agents rows the panel ever shows at once; a longer tree scrolls
17/// within this viewport instead of growing the panel.
18pub(crate) const MAX_AGENT_PANEL_ROWS: usize = 4;
19/// Absolute ceiling for the whole Agents panel, borders included: it exists to
20/// keep a pathological layout from starving the conversation. The panel is
21/// already bounded by [`MAX_AGENT_PANEL_ROWS`] text rows plus two borders, so
22/// this stays deliberately looser than that (4 + 2 = 6 ≤ 10).
23const MAX_AGENT_HEIGHT: u16 = 10;
24/// Composer ceiling in total rows: five text rows plus top/bottom borders.
25const MAX_INPUT_HEIGHT: u16 = 7;
26
27#[derive(Clone, Copy, Debug, Eq, PartialEq)]
28pub(crate) struct UiLayout {
29    pub(crate) agent: Rect,
30    pub(crate) conversation: Rect,
31    /// The steer-queue strip between the conversation pane and the status
32    /// line; zero height while the selected session's queue is empty.
33    pub(crate) queue: Rect,
34    pub(crate) status: Rect,
35    /// The persistent goal bar directly above the composer; zero height when
36    /// it is not visible or the terminal has no space for it.
37    pub(crate) goal: Rect,
38    pub(crate) input: Rect,
39    pub(crate) bar: Rect,
40}
41
42/// Resolve the single top-to-bottom pane geometry for a known visible tree
43/// row count, queue-strip row demand, goal visibility, and composer text-row
44/// demand. The Agents panel is hidden when `visible_tree_row_count` is zero or
45/// one; otherwise it is exactly `clamp(visible_tree_row_count, 1,
46/// MAX_AGENT_PANEL_ROWS)` text rows — a longer tree scrolls inside that
47/// viewport instead of growing the panel — with its borders outside that count,
48/// so the conversation starts immediately below it. The composer takes one text
49/// row by default and grows with its content up to five; every added composer
50/// row is reclaimed from the conversation pane. The goal bar takes exactly one
51/// row when it is visible and space remains above the composer, with the
52/// ephemeral status line yielding first on short terminals. The queue strip
53/// reclaims conversation rows the same way and leaves the conversation at least
54/// one row whenever content space remains after fixed bars.
55pub(crate) fn terminal_layout_with_tree_rows(
56    area: Rect,
57    visible_tree_rows: usize,
58    queue_rows: u16,
59    goal_visible: bool,
60    input_text_rows: u16,
61) -> UiLayout {
62    let agent_height = if visible_tree_rows <= 1 {
63        0
64    } else {
65        (visible_tree_rows.clamp(1, MAX_AGENT_PANEL_ROWS) as u16)
66            .saturating_add(2)
67            .min(MAX_AGENT_HEIGHT)
68    };
69    let bar_height = u16::from(area.height > 0);
70    let input_height = input_text_rows
71        .clamp(1, input::MAX_TEXT_ROWS)
72        .saturating_add(2)
73        .min(MAX_INPUT_HEIGHT)
74        .min(area.height.saturating_sub(bar_height));
75    let available_height = area
76        .height
77        .saturating_sub(input_height)
78        .saturating_sub(bar_height);
79    let goal_height = u16::from(goal_visible && available_height > 0);
80    let status_height = u16::from(available_height.saturating_sub(goal_height) >= 3);
81    let content_height = available_height
82        .saturating_sub(goal_height)
83        .saturating_sub(status_height);
84    let agent_height = if content_height > 1 {
85        agent_height.min(content_height - 1)
86    } else {
87        content_height
88    };
89    let conversation_available = content_height.saturating_sub(agent_height);
90    // The strip never starves the conversation completely: on a cramped
91    // terminal it shrinks away rather than taking the last row.
92    let queue_height = queue_rows.min(conversation_available.saturating_sub(1));
93    let conversation_height = conversation_available.saturating_sub(queue_height);
94    let agent = Rect::new(area.x, area.y, area.width, agent_height);
95    let conversation = Rect::new(
96        area.x,
97        area.y.saturating_add(agent_height),
98        area.width,
99        conversation_height,
100    );
101    let queue = Rect::new(
102        area.x,
103        conversation.y.saturating_add(conversation_height),
104        area.width,
105        queue_height,
106    );
107    let status = Rect::new(
108        area.x,
109        queue.y.saturating_add(queue_height),
110        area.width,
111        status_height,
112    );
113    let goal = Rect::new(
114        area.x,
115        status.y.saturating_add(status_height),
116        area.width,
117        goal_height,
118    );
119    let input = Rect::new(
120        area.x,
121        area.y
122            .saturating_add(area.height.saturating_sub(bar_height + input_height)),
123        area.width,
124        input_height,
125    );
126    let bar = Rect::new(
127        area.x,
128        area.y
129            .saturating_add(area.height.saturating_sub(bar_height)),
130        area.width,
131        bar_height,
132    );
133    UiLayout {
134        agent,
135        conversation,
136        queue,
137        status,
138        goal,
139        input,
140        bar,
141    }
142}

144#[cfg(test)]
145mod tests {
146    use super::terminal_layout_with_tree_rows;
147    use ratatui::layout::Rect;

149    #[test]
150    fn hidden_goal_preserves_existing_geometry_without_a_blank_row() {
151        for width in [1, 3, 7, 8, 20, 40, 80] {
152            for height in 0..=24 {
153                let area = Rect::new(2, 3, width, height);
154                let layout = terminal_layout_with_tree_rows(area, 3, 5, false, 2);
155
156                assert_eq!(layout.goal.height, 0, "{width}x{height}");
157                assert_eq!(layout.goal.y, layout.input.y, "{width}x{height}");
158                assert_eq!(
159                    layout.status.y + layout.status.height,
160                    layout.input.y,
161                    "{width}x{height}"
162                );
163            }
164        }
165    }
166
167    #[test]
168    fn visible_goal_is_one_row_when_space_exists_and_never_overlaps() {
169        for width in [1, 3, 7, 8, 20, 40, 80] {
170            for height in 0..=24 {
171                let area = Rect::new(2, 3, width, height);
172                let hidden = terminal_layout_with_tree_rows(area, 3, 5, false, 1);
173                let shown = terminal_layout_with_tree_rows(area, 3, 5, true, 1);
174                let expected_goal_height = u16::from(
175                    area.height
176                        .saturating_sub(hidden.input.height)
177                        .saturating_sub(hidden.bar.height)
178                        > 0,
179                );
180
181                assert_eq!(shown.goal.height, expected_goal_height, "{width}x{height}");
182                assert_eq!(shown.goal.y + shown.goal.height, shown.input.y);
183                assert_eq!(shown.input, hidden.input, "{width}x{height}");
184                assert_eq!(shown.bar, hidden.bar, "{width}x{height}");
185
186                let panes = [
187                    shown.agent,
188                    shown.conversation,
189                    shown.queue,
190                    shown.status,
191                    shown.goal,
192                    shown.input,
193                    shown.bar,
194                ];
195                for pane in panes {
196                    assert_eq!(pane.x, area.x, "{width}x{height}: {pane:?}");
197                    assert_eq!(pane.width, area.width, "{width}x{height}: {pane:?}");
198                    assert!(
199                        pane.y >= area.y && pane.y + pane.height <= area.y + area.height,
200                        "{width}x{height}: {pane:?} outside {area:?}"
201                    );
202                }
203                for pair in panes.windows(2) {
204                    assert!(
205                        pair[0].y + pair[0].height <= pair[1].y,
206                        "{width}x{height}: {:?} overlaps {:?}",
207                        pair[0],
208                        pair[1]
209                    );
210                }
211            }
212        }
213    }
214
215    #[test]
216    fn visible_goal_is_reserved_before_ephemeral_status() {
217        let area = Rect::new(0, 0, 20, 7);
218        let hidden = terminal_layout_with_tree_rows(area, 0, 0, false, 1);
219        let shown = terminal_layout_with_tree_rows(area, 0, 0, true, 1);
220
221        assert_eq!(hidden.status.height, 1);
222        assert_eq!(hidden.goal.height, 0);
223        assert_eq!(shown.status.height, 0);
224        assert_eq!(shown.goal.height, 1);
225        assert_eq!(shown.goal.y + shown.goal.height, shown.input.y);
226        assert_eq!(shown.input, hidden.input);
227        assert_eq!(shown.bar, hidden.bar);
228    }
