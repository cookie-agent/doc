345/// How well `query` matches any of `candidates`: exact beats prefix beats
346/// substring. `None` means no match. An empty query matches everything.
347pub(crate) fn match_rank<'a>(
348    candidates: impl IntoIterator<Item = &'a str>,
349    query: &str,
350) -> Option<u8> {
351    if query.is_empty() {
352        return Some(0);
353    }
354    candidates
355        .into_iter()
356        .filter_map(|candidate| {
357            if candidate == query {
358                Some(0)
359            } else if candidate.starts_with(query) {
360                Some(1)
361            } else if candidate.contains(query) {
362                Some(2)
363            } else {
364                None
365            }
366        })
367        .min()
368}

370/// Commands matching `query`, best match first and registry order within a
371/// rank.
372pub(crate) fn entries(query: &str) -> Vec<&'static CommandSpec> {
373    let raw = query;
374    let query = normalized_query(query);
375    let mut ranked = COMMANDS
376        .iter()
377        .filter_map(|spec| {
378            if command_argument(spec, raw).is_some() {
379                return Some((0, spec));
380            }
381            match_rank(
382                std::iter::once(spec.name).chain(spec.aliases.iter().copied()),
383                &query,
384            )
385            .map(|rank| (rank, spec))
386        })
387        .collect::<Vec<_>>();
388    ranked.sort_by_key(|(rank, _)| *rank);
389    ranked.into_iter().map(|(_, spec)| spec).collect()
390}

392/// The text typed after `spec`'s name (or an alias) and a space in the
393/// palette's search, as typed, for a command that takes one.
394pub(crate) fn command_argument<'a>(spec: &CommandSpec, query: &'a str) -> Option<&'a str> {
395    if !spec.action.takes_argument() {
396        return None;
397    }
398    let query = query.trim_start();
399    let (head, rest) = query.split_once(char::is_whitespace)?;
400    std::iter::once(spec.name)
401        .chain(spec.aliases.iter().copied())
402        .any(|name| head.eq_ignore_ascii_case(name))
403        .then_some(rest)
404}

449/// The open palette: the command search plus a stack of follow-up steps.
450/// Esc pops one step; with no steps left it closes the palette.
451pub(crate) struct CommandPalette {
452    pub(crate) search: InputState,
453    pub(crate) list: ListState,
454    pub(crate) steps: Vec<PaletteStep>,
455}
456
457impl CommandPalette {
458    pub(crate) fn new() -> Self {
459        Self {
460            search: InputState::default(),
461            list: ListState::default().with_selected(Some(0)),
462            steps: Vec::new(),
463        }
464    }
465
466    /// The text field keys and pastes edit in the current step, if any.
467    pub(crate) fn active_input(&mut self) -> Option<&mut InputState> {
468        match self.steps.last_mut() {
469            None => Some(&mut self.search),
470            Some(PaletteStep::Text { input, .. }) => Some(input),
471            Some(PaletteStep::Skills { search, .. }) => Some(search),
472            Some(PaletteStep::EventLevel { .. }) => None,
473        }
474    }
475
476    /// The list the current step selects from, if any.
477    pub(crate) fn active_list(&mut self) -> Option<&mut ListState> {
478        match self.steps.last_mut() {
479            None => Some(&mut self.list),
480            Some(PaletteStep::EventLevel { list } | PaletteStep::Skills { list, .. }) => Some(list),
481            Some(PaletteStep::Text { .. }) => None,
482        }
483    }
484}

492pub(crate) fn move_selection(state: &mut ListState, len: usize, up: bool) {
493    if len == 0 {
494        state.select(Some(0));
495        return;
496    }
497    let selected = state.selected().unwrap_or(0);
498    state.select(Some(if up {
499        selected.saturating_sub(1)
500    } else {
501        (selected + 1).min(len - 1)
502    }));
503}

530/// [`render_list`] whose rows may start with a badge, such as the agent's
531/// commands' `agent`.
532pub(crate) fn render_badged_list(
533    frame: &mut Frame,
534    area: Rect,
535    chrome: ListChrome<'_>,
536    search: Option<(&mut InputState, &str)>,
537    entries: Vec<(String, Option<&'static str>)>,
538    state: &mut ListState,
539    theme: &crate::theme::Theme,
540) -> Vec<(Rect, usize)> {
541    super::app::paint_panel(frame, area, theme);
542    let search_height = if search.is_some() {
543        3.min(area.height)
544    } else {
545        0
546    };
547    if let Some((input, placeholder)) = search {
548        input::render(
549            frame,
550            Rect::new(area.x, area.y, area.width, search_height),
551            input,
552            true,
553            "Search",
554            Some(placeholder),
555            theme,
556        );
557    }
558    let detail_height = u16::from(chrome.detail.is_some() && area.height > search_height + 3);
559    let list_area = Rect::new(
560        area.x,
561        area.y.saturating_add(search_height),
562        area.width,
563        area.height
564            .saturating_sub(search_height)
565            .saturating_sub(detail_height),
566    );
567    if detail_height > 0
568        && let Some(detail) = chrome.detail
569    {
570        let width = usize::from(area.width.saturating_sub(2));
571        frame.render_widget(
572            Paragraph::new(Span::styled(
573                super::app::truncate_with_ellipsis(&detail, width),
574                theme.muted(),
575            )),
576            Rect::new(
577                area.x.saturating_add(1),
578                list_area.y.saturating_add(list_area.height),
579                area.width.saturating_sub(2),
580                1,
581            ),
582        );
583    }
584    // Labels end in an ellipsis rather than clip, after the selection
585    // marker's two columns and the badge.
586    let available = usize::from(list_area.width.saturating_sub(4));
587    let lines = entries
588        .into_iter()
589        .map(|(label, badge)| match badge {
590            Some(badge) => {
591                let badge = format!(" {badge} ");
592                let room = available.saturating_sub(badge.chars().count() + 1);
593                Line::from(vec![
594                    Span::styled(
595                        badge,
596                        theme
597                            .internal()
598                            .add_modifier(ratatui::style::Modifier::REVERSED),
599                    ),
600                    Span::raw(" "),
601                    Span::raw(super::app::truncate_with_ellipsis(&label, room)),
602                ])
603            }
604            None => Line::from(super::app::truncate_with_ellipsis(&label, available)),
605        })
606        .collect();
607    super::pickers::render_lines(
608        frame,
609        super::pickers::PickerChrome {
610            title: chrome.title,
611            empty_message: Some(chrome.empty_message),
612            hint: Some(chrome.hint),
613        },
614        lines,
615        list_area,
616        state,
617        theme,
618        theme.selected(),
619    )
620}

622/// A text step: one input box under a title, with a key hint below it.
623/// Only the rows the box needs are painted, so the step floats compactly at
624/// the top of `area`. Returns the painted rectangle.
625pub(crate) fn render_text(
626    frame: &mut Frame,
627    area: Rect,
628    title: &str,
629    placeholder: &str,
630    hint: &str,
631    input: &mut InputState,
632    theme: &crate::theme::Theme,
633) -> Rect {
634    let rows = u16::try_from(input.composer_rows(area.width.saturating_sub(2)))
635        .unwrap_or(u16::MAX)
636        .clamp(1, input::MAX_TEXT_ROWS);
637    let box_height = rows.saturating_add(2).min(area.height);
638    let painted = Rect::new(
639        area.x,
640        area.y,
641        area.width,
642        box_height.saturating_add(1).min(area.height),
643    );
644    super::app::paint_panel(frame, painted, theme);
645    input::render(
646        frame,
647        Rect::new(area.x, area.y, area.width, box_height),
648        input,
649        true,
650        title.to_owned(),
651        Some(placeholder),
652        theme,
653    );
654    if painted.height > box_height {
655        frame.render_widget(
656            Paragraph::new(
657                Line::from(Span::styled(hint.to_owned(), theme.internal())).right_aligned(),
658            ),
659            Rect::new(
660                area.x.saturating_add(1),
661                area.y.saturating_add(box_height),
662                area.width.saturating_sub(2),
663                1,
664            ),
665        );
666    }
667    painted
668}

680    #[test]
681    fn search_ranks_exact_then_prefix_then_substring() {
682        assert_eq!(names("").len(), COMMANDS.len());
683        // The exact name first; `/cron new` only contains it.
684        assert_eq!(names("new"), ["new", "cron new"]);
685        assert_eq!(names("cron"), ["cron", "cron new"]);
686        assert_eq!(names("model"), ["model"]);
687        // The exact name outranks the panel modes it merely prefixes.
688        assert_eq!(
689            names("agent"),
690            [
691                "agent",
692                "agent-panel show",
693                "agent-panel hide",
694                "agent-panel auto"
695            ]
696        );
697        assert_eq!(names("agent-panel h"), ["agent-panel hide"]);
698        // "sessions" contains "s" everywhere; the exact alias still wins.
699        assert_eq!(names("q").first(), Some(&"quit"));
700        assert_eq!(names("perms"), ["permissions"]);
701        assert_eq!(names("exit"), ["quit"]);
702        for alias in ["resume", "load", "continue"] {
703            assert_eq!(names(alias), ["sessions"], "{alias}");
704        }
705        // Aliases search but never show.
706        let sessions = entries("resume")[0];
707        assert_eq!(sessions.label(), "/sessions — choose a session");
708        assert_eq!(names("hide"), ["agent-panel hide"]);
709        assert_eq!(names("  GOAL "), ["goal"]);
710        let ranked = names("s");
711        assert_eq!(ranked.first(), Some(&"skills"), "{ranked:?}");
712        assert!(names("definitely-not-a-command").is_empty());
713        assert_eq!(match_rank(["compact"], "pact"), Some(2));
714        assert_eq!(match_rank(["compact"], "comp"), Some(1));
715        assert_eq!(match_rank(["compact"], "compact"), Some(0));
716        assert_eq!(match_rank(["compact"], "x"), None);
717    }

751        let rename = COMMANDS
752            .iter()
753            .find(|spec| spec.name == "rename")
754            .expect("rename");

759            Some("Fix The Parser")

761        assert_eq!(command_argument(rename, "  title --auto"), Some("--auto"));
762        assert_eq!(command_argument(rename, "rename"), None);
763        assert_eq!(command_argument(rename, "renamed x"), None);
764        // Other commands never read an argument.
765        let goal = COMMANDS.iter().find(|spec| spec.name == "goal").unwrap();
766        assert_eq!(command_argument(goal, "goal ship it"), None);
767        assert!(names("goal ship it").is_empty());

770    #[test]
771    fn list_rows_ellipsize_and_the_footer_explains_the_keys() {
772        let theme = crate::theme::Theme::default();
773        let mut terminal = Terminal::new(TestBackend::new(34, 12)).expect("terminal");
774        let mut state = ListState::default().with_selected(Some(0));
775        let mut search = crate::ui::input::InputState::default();
776        terminal
777            .draw(|frame| {
778                super::render_list(
779                    frame,
780                    frame.area(),
781                    super::ListChrome {
782                        title: "Commands",
783                        hint: "enter: choose · esc: close",
784                        empty_message: "No matching commands",
785                        detail: None,
786                    },
787                    Some((&mut search, "Filter commands…")),
788                    vec![
789                        "/goal — set a root-session goal (asks for the objective)".to_owned(),
790                        "/quit — exit the TUI".to_owned(),
791                    ],
792                    &mut state,
793                    &theme,
794                );
795            })
796            .expect("render palette");
797        let buffer = terminal.backend().buffer();
798        let text = (0..12)
799            .flat_map(|y| (0..34).map(move |x| buffer[(x, y)].symbol().to_owned()))
800            .collect::<String>();
801        assert!(text.contains('…'), "{text}");
802        assert!(!text.contains("asks for the objective"), "{text}");
803        assert!(text.contains("Search"), "{text}");
804        // No row spills over the right border.
805        assert_eq!(buffer[(33, 4)].symbol(), "│");
806        assert!(text.contains("enter: choose"), "{text}");
807    }
