1//! The error registry, as specified in `docs/site/specs/compatibility-v1.md`
2//! section 6: every error the four protocols send carries a stable
3//! `data.code` and a `message` a user can understand, both from this one
4//! table.
5//!
6//! An error object is built from its registry entry:
7//!
8//! ```
9//! use cookie_agent_protocol::errors;
10//!
11//! let error = errors::MEDIA_UNSUPPORTED
12//!     .error()
13//!     .field("kind", "image")
14//!     .field("model", "anthropic/claude-haiku-5-5")
15//!     .detail("the model declares input modalities: text")
16//!     .try_build()
17//!     .unwrap();
18//! assert_eq!(error.code, errors::APPLICATION_ERROR);
19//! assert_eq!(error.message, "The model anthropic/claude-haiku-5-5 does not accept image input.");
20//! assert_eq!(error.data.unwrap()["code"], "media_unsupported");
21//! ```
22//!
23//! Clients branch only on `data.code`, show `message`, then `hint` when
24//! present, and `detail` on request. The daemon sends
25//! `session_requires_newer_engine`; the rest of the daemon, the SDK, and the
26//! runtimes switch to these errors in later phases.
27

33/// JSON-RPC 2.0: the frame is not JSON.
34pub const PARSE_ERROR: i32 = -32700;
35/// JSON-RPC 2.0: the frame is not a request, notification, or reply.
36pub const INVALID_REQUEST: i32 = -32600;
37/// JSON-RPC 2.0: no such method.
38pub const METHOD_NOT_FOUND: i32 = -32601;
39/// JSON-RPC 2.0: the params do not match the method.
40pub const INVALID_PARAMS: i32 = -32602;
41/// JSON-RPC 2.0: the receiver failed in a way it cannot describe further.
42pub const INTERNAL_ERROR: i32 = -32603;
43/// Any application error; `data.code` says which.
44pub const APPLICATION_ERROR: i32 = -32000;
45/// The connection must complete its handshake first.
46pub const HANDSHAKE_REQUIRED: i32 = -32001;
47/// The peer speaks another major of the protocol.
48pub const UNSUPPORTED_PROTOCOL: i32 = -32002;
49/// The connection has not authenticated.
50pub const UNAUTHENTICATED: i32 = -32003;

52/// The most bytes of `message`.
53pub const MAX_MESSAGE_BYTES: usize = 1024;
54/// The most bytes of `data.hint`.
55pub const MAX_HINT_BYTES: usize = 512;
56/// The most bytes of `data.detail`.
57pub const MAX_DETAIL_BYTES: usize = 4096;

59/// `data` members every error may carry, which a code cannot declare as its
60/// own fields.
61pub const RESERVED_DATA_MEMBERS: &[&str] = &["code", "hint", "detail"];
62
63/// One structured `data` member an error code defines.
64#[derive(Clone, Copy, Debug, Eq, PartialEq)]
65pub struct ErrorField {
66    pub name: &'static str,
67    /// A required field is always present, so templates may name it.
68    pub required: bool,
69    pub description: &'static str,
70}
71
72const fn required(name: &'static str, description: &'static str) -> ErrorField {
73    ErrorField {
74        name,
75        required: true,
76        description,
77    }
78}
79
80const fn optional(name: &'static str, description: &'static str) -> ErrorField {
81    ErrorField {
82        name,
83        required: false,
84        description,
85    }
86}
87
88/// One registered error code.
89#[derive(Clone, Copy, Debug, Eq, PartialEq)]
90pub struct ErrorCode {
91    /// The stable snake-case `data.code`.
92    pub code: &'static str,
93    /// The JSON-RPC `code` it is sent with.
94    pub rpc_code: i32,
95    pub fields: &'static [ErrorField],
96    /// The `message`, with `{field}` replaced by required fields' values.
97    pub message: &'static str,
98    /// The `data.hint`, filled like `message`, where a next step helps.
99    pub hint: Option<&'static str>,
100    /// A one-line description for the reference docs.
101    pub description: &'static str,
102}
103
104impl ErrorCode {
105    /// Starts an error of this code.
106    pub fn error(&'static self) -> ErrorBuilder {
107        ErrorBuilder {
108            code: self,
109            fields: Map::new(),
110            detail: None,
111            problem: None,
112        }
113    }
114
115    #[must_use]
116    pub fn field(&self, name: &str) -> Option<&'static ErrorField> {
117        self.fields.iter().find(|field| field.name == name)
118    }
119}

121macro_rules! error_codes {
122    ($(
123        $(#[$meta:meta])*
124        $constant:ident {
125            code: $code:literal,
126            rpc: $rpc:expr,
127            fields: [$($field:expr),* $(,)?],
128            message: $message:literal,
129            hint: $hint:expr,
130            description: $description:literal $(,)?
131        }
132    )+) => {
133        $(
134            $(#[$meta])*
135            pub static $constant: ErrorCode = ErrorCode {
136                code: $code,
137                rpc_code: $rpc,
138                fields: &[$($field),*],
139                message: $message,
140                hint: $hint,
141                description: $description,
142            };
143        )+
144
145        /// Every registered code.
146        pub static REGISTRY: &[&ErrorCode] = &[$(&$constant),+];
147    };
148}

723/// The registry entry of `code`.
724#[must_use]
725pub fn lookup(code: &str) -> Option<&'static ErrorCode> {
726    REGISTRY.iter().copied().find(|entry| entry.code == code)
727}
728
729/// Builds one error object from its registry entry.
730#[derive(Clone, Debug)]
731#[must_use]
732pub struct ErrorBuilder {
733    code: &'static ErrorCode,
734    fields: Map<String, Value>,
735    detail: Option<String>,
736    problem: Option<ErrorBuildError>,
737}
738
739impl ErrorBuilder {
740    /// Sets a declared field.
741    pub fn field(mut self, name: &str, value: impl Serialize) -> Self {
742        if self.code.field(name).is_none() {
743            self.problem
744                .get_or_insert(ErrorBuildError::UndeclaredField {
745                    code: self.code.code,
746                    field: name.to_owned(),
747                });
748            return self;
749        }
750        match serde_json::to_value(value) {
751            Ok(Value::Null) => {
752                self.fields.remove(name);
753            }
754            Ok(value) => {
755                self.fields.insert(name.to_owned(), value);
756            }
757            Err(_) => {
758                self.problem.get_or_insert(ErrorBuildError::Unserializable {
759                    code: self.code.code,
760                    field: name.to_owned(),
761                });
762            }
763        }
764        self
765    }
766
767    /// Sets the technical detail shown on request.
768    pub fn detail(mut self, detail: impl Into<String>) -> Self {
769        self.detail = Some(detail.into());
770        self
771    }
772
773    /// The error object, or why it cannot be built: an undeclared or
774    /// missing field.
775    pub fn try_build(self) -> Result<JsonRpcError, ErrorBuildError> {
776        if let Some(problem) = self.problem {
777            return Err(problem);
778        }
779        if let Some(missing) = self
780            .code
781            .fields
782            .iter()
783            .find(|field| field.required && !self.fields.contains_key(field.name))
784        {
785            return Err(ErrorBuildError::MissingField {
786                code: self.code.code,
787                field: missing.name,
788            });
789        }
790        let message = bounded(
791            &fill(self.code.message, &self.fields),
792            MAX_MESSAGE_BYTES,
793            false,
794        );
795        let mut data = Map::new();
796        data.insert("code".into(), Value::String(self.code.code.into()));
797        if let Some(hint) = self.code.hint {
798            data.insert(
799                "hint".into(),
800                Value::String(bounded(&fill(hint, &self.fields), MAX_HINT_BYTES, false)),
801            );
802        }
803        if let Some(detail) = self.detail.filter(|detail| !detail.is_empty()) {
804            data.insert(
805                "detail".into(),
806                Value::String(bounded(&detail, MAX_DETAIL_BYTES, true)),
807            );
808        }
809        data.extend(self.fields);
810        Ok(JsonRpcError {
811            code: self.code.rpc_code,
812            message,
813            data: Some(Value::Object(data)),
814        })
815    }
816
817    /// The error object. A builder misused by its caller (a bug) yields an
818    /// `internal_error` naming the problem in its detail instead.
819    pub fn build(self) -> JsonRpcError {
820        self.try_build().unwrap_or_else(|problem| {
821            debug_assert!(false, "{problem}");
822            INTERNAL
823                .error()
824                .detail(problem.to_string())
825                .try_build()
826                .expect("internal_error declares no fields")
827        })
828    }
829}

831/// A caller misused [`ErrorBuilder`].
832#[derive(Clone, Debug, Eq, PartialEq, thiserror::Error)]
833pub enum ErrorBuildError {
834    #[error("error code {code} declares no field {field}")]
835    UndeclaredField { code: &'static str, field: String },
836    #[error("error code {code} requires field {field}")]
837    MissingField {
838        code: &'static str,
839        field: &'static str,
840    },
841    #[error("field {field} of error code {code} does not serialize")]
842    Unserializable { code: &'static str, field: String },
843}

845/// The `{name}` placeholders of a template.
846pub fn placeholders(template: &str) -> impl Iterator<Item = &str> {
847    template
848        .split('{')
849        .skip(1)
850        .filter_map(|rest| rest.split_once('}').map(|(name, _)| name))
851}
852
853fn fill(template: &str, fields: &Map<String, Value>) -> String {
854    let mut out = String::with_capacity(template.len());
855    let mut rest = template;
856    while let Some(start) = rest.find('{') {
857        out.push_str(&rest[..start]);
858        let after = &rest[start + 1..];
859        let Some(end) = after.find('}') else {
860            out.push_str(&rest[start..]);
861            return out;
862        };
863        let name = &after[..end];
864        match fields.get(name) {
865            Some(Value::String(text)) => out.push_str(text),
866            Some(value) => out.push_str(&value.to_string()),
867            None => out.push_str(&rest[start..=start + end + 1]),
868        }
869        rest = &after[end + 1..];
870    }
871    out.push_str(rest);
872    out
873}
874
875/// `text` made terminal-safe and cut to `limit` bytes at a character
876/// boundary, ending in `…` when cut. Newlines and tabs stay in a detail and
877/// become spaces elsewhere.
878fn bounded(text: &str, limit: usize, multiline: bool) -> String {
879    let mut safe = crate::sanitize_display_text(text).into_owned();
880    if !multiline {
881        safe = safe.replace(['\n', '\t'], " ");
882    }
883    if safe.len() <= limit {
884        return safe;
885    }
886    let ellipsis = '…';
887    let mut end = limit - ellipsis.len_utf8();
888    while !safe.is_char_boundary(end) {
889        end -= 1;
890    }
891    safe.truncate(end);
892    safe.push(ellipsis);
893    safe
894}
