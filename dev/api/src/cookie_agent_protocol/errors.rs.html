1//! The error registry, as specified in `docs/site/specs/compatibility-v1.md`
2//! section 6: every error the four protocols send carries a stable
3//! `data.code` and a `message` a user can understand, both from this one
4//! table.
5//!
6//! An error object is built from its registry entry:
7//!
8//! ```
9//! use cookie_agent_protocol::errors;
10//!
11//! let error = errors::MEDIA_UNSUPPORTED
12//!     .error()
13//!     .field("kind", "image")
14//!     .field("model", "anthropic/claude-haiku-5-5")
15//!     .detail("the model declares input modalities: text")
16//!     .try_build()
17//!     .unwrap();
18//! assert_eq!(error.code, errors::APPLICATION_ERROR);
19//! assert_eq!(error.message, "The model anthropic/claude-haiku-5-5 does not accept image input.");
20//! assert_eq!(error.data.unwrap()["code"], "media_unsupported");
21//! ```
22//!
23//! Clients branch only on `data.code`, show `message`, then `hint` when
24//! present, and `detail` on request. Nothing sends these errors yet; the
25//! daemon, the SDK, and the runtimes switch to them in later phases.
26

32/// JSON-RPC 2.0: the frame is not JSON.
33pub const PARSE_ERROR: i32 = -32700;
34/// JSON-RPC 2.0: the frame is not a request, notification, or reply.
35pub const INVALID_REQUEST: i32 = -32600;
36/// JSON-RPC 2.0: no such method.
37pub const METHOD_NOT_FOUND: i32 = -32601;
38/// JSON-RPC 2.0: the params do not match the method.
39pub const INVALID_PARAMS: i32 = -32602;
40/// JSON-RPC 2.0: the receiver failed in a way it cannot describe further.
41pub const INTERNAL_ERROR: i32 = -32603;
42/// Any application error; `data.code` says which.
43pub const APPLICATION_ERROR: i32 = -32000;
44/// The connection must complete its handshake first.
45pub const HANDSHAKE_REQUIRED: i32 = -32001;
46/// The peer speaks another major of the protocol.
47pub const UNSUPPORTED_PROTOCOL: i32 = -32002;
48/// The connection has not authenticated.
49pub const UNAUTHENTICATED: i32 = -32003;

51/// The most bytes of `message`.
52pub const MAX_MESSAGE_BYTES: usize = 1024;
53/// The most bytes of `data.hint`.
54pub const MAX_HINT_BYTES: usize = 512;
55/// The most bytes of `data.detail`.
56pub const MAX_DETAIL_BYTES: usize = 4096;

58/// `data` members every error may carry, which a code cannot declare as its
59/// own fields.
60pub const RESERVED_DATA_MEMBERS: &[&str] = &["code", "hint", "detail"];
61
62/// One structured `data` member an error code defines.
63#[derive(Clone, Copy, Debug, Eq, PartialEq)]
64pub struct ErrorField {
65    pub name: &'static str,
66    /// A required field is always present, so templates may name it.
67    pub required: bool,
68    pub description: &'static str,
69}
70
71const fn required(name: &'static str, description: &'static str) -> ErrorField {
72    ErrorField {
73        name,
74        required: true,
75        description,
76    }
77}
78
79const fn optional(name: &'static str, description: &'static str) -> ErrorField {
80    ErrorField {
81        name,
82        required: false,
83        description,
84    }
85}
86
87/// One registered error code.
88#[derive(Clone, Copy, Debug, Eq, PartialEq)]
89pub struct ErrorCode {
90    /// The stable snake-case `data.code`.
91    pub code: &'static str,
92    /// The JSON-RPC `code` it is sent with.
93    pub rpc_code: i32,
94    pub fields: &'static [ErrorField],
95    /// The `message`, with `{field}` replaced by required fields' values.
96    pub message: &'static str,
97    /// The `data.hint`, filled like `message`, where a next step helps.
98    pub hint: Option<&'static str>,
99    /// A one-line description for the reference docs.
100    pub description: &'static str,
101}
102
103impl ErrorCode {
104    /// Starts an error of this code.
105    pub fn error(&'static self) -> ErrorBuilder {
106        ErrorBuilder {
107            code: self,
108            fields: Map::new(),
109            detail: None,
110            problem: None,
111        }
112    }
113
114    #[must_use]
115    pub fn field(&self, name: &str) -> Option<&'static ErrorField> {
116        self.fields.iter().find(|field| field.name == name)
117    }
118}

120macro_rules! error_codes {
121    ($(
122        $(#[$meta:meta])*
123        $constant:ident {
124            code: $code:literal,
125            rpc: $rpc:expr,
126            fields: [$($field:expr),* $(,)?],
127            message: $message:literal,
128            hint: $hint:expr,
129            description: $description:literal $(,)?
130        }
131    )+) => {
132        $(
133            $(#[$meta])*
134            pub static $constant: ErrorCode = ErrorCode {
135                code: $code,
136                rpc_code: $rpc,
137                fields: &[$($field),*],
138                message: $message,
139                hint: $hint,
140                description: $description,
141            };
142        )+
143
144        /// Every registered code.
145        pub static REGISTRY: &[&ErrorCode] = &[$(&$constant),+];
146    };
147}

722/// The registry entry of `code`.
723#[must_use]
724pub fn lookup(code: &str) -> Option<&'static ErrorCode> {
725    REGISTRY.iter().copied().find(|entry| entry.code == code)
726}
727
728/// Builds one error object from its registry entry.
729#[derive(Clone, Debug)]
730#[must_use]
731pub struct ErrorBuilder {
732    code: &'static ErrorCode,
733    fields: Map<String, Value>,
734    detail: Option<String>,
735    problem: Option<ErrorBuildError>,
736}
737
738impl ErrorBuilder {
739    /// Sets a declared field.
740    pub fn field(mut self, name: &str, value: impl Serialize) -> Self {
741        if self.code.field(name).is_none() {
742            self.problem
743                .get_or_insert(ErrorBuildError::UndeclaredField {
744                    code: self.code.code,
745                    field: name.to_owned(),
746                });
747            return self;
748        }
749        match serde_json::to_value(value) {
750            Ok(Value::Null) => {
751                self.fields.remove(name);
752            }
753            Ok(value) => {
754                self.fields.insert(name.to_owned(), value);
755            }
756            Err(_) => {
757                self.problem.get_or_insert(ErrorBuildError::Unserializable {
758                    code: self.code.code,
759                    field: name.to_owned(),
760                });
761            }
762        }
763        self
764    }
765
766    /// Sets the technical detail shown on request.
767    pub fn detail(mut self, detail: impl Into<String>) -> Self {
768        self.detail = Some(detail.into());
769        self
770    }
771
772    /// The error object, or why it cannot be built: an undeclared or
773    /// missing field.
774    pub fn try_build(self) -> Result<JsonRpcError, ErrorBuildError> {
775        if let Some(problem) = self.problem {
776            return Err(problem);
777        }
778        if let Some(missing) = self
779            .code
780            .fields
781            .iter()
782            .find(|field| field.required && !self.fields.contains_key(field.name))
783        {
784            return Err(ErrorBuildError::MissingField {
785                code: self.code.code,
786                field: missing.name,
787            });
788        }
789        let message = bounded(
790            &fill(self.code.message, &self.fields),
791            MAX_MESSAGE_BYTES,
792            false,
793        );
794        let mut data = Map::new();
795        data.insert("code".into(), Value::String(self.code.code.into()));
796        if let Some(hint) = self.code.hint {
797            data.insert(
798                "hint".into(),
799                Value::String(bounded(&fill(hint, &self.fields), MAX_HINT_BYTES, false)),
800            );
801        }
802        if let Some(detail) = self.detail.filter(|detail| !detail.is_empty()) {
803            data.insert(
804                "detail".into(),
805                Value::String(bounded(&detail, MAX_DETAIL_BYTES, true)),
806            );
807        }
808        data.extend(self.fields);
809        Ok(JsonRpcError {
810            code: self.code.rpc_code,
811            message,
812            data: Some(Value::Object(data)),
813        })
814    }
815
816    /// The error object. A builder misused by its caller (a bug) yields an
817    /// `internal_error` naming the problem in its detail instead.
818    pub fn build(self) -> JsonRpcError {
819        self.try_build().unwrap_or_else(|problem| {
820            debug_assert!(false, "{problem}");
821            INTERNAL
822                .error()
823                .detail(problem.to_string())
824                .try_build()
825                .expect("internal_error declares no fields")
826        })
827    }
828}

830/// A caller misused [`ErrorBuilder`].
831#[derive(Clone, Debug, Eq, PartialEq, thiserror::Error)]
832pub enum ErrorBuildError {
833    #[error("error code {code} declares no field {field}")]
834    UndeclaredField { code: &'static str, field: String },
835    #[error("error code {code} requires field {field}")]
836    MissingField {
837        code: &'static str,
838        field: &'static str,
839    },
840    #[error("field {field} of error code {code} does not serialize")]
841    Unserializable { code: &'static str, field: String },
842}

844/// The `{name}` placeholders of a template.
845pub fn placeholders(template: &str) -> impl Iterator<Item = &str> {
846    template
847        .split('{')
848        .skip(1)
849        .filter_map(|rest| rest.split_once('}').map(|(name, _)| name))
850}
851
852fn fill(template: &str, fields: &Map<String, Value>) -> String {
853    let mut out = String::with_capacity(template.len());
854    let mut rest = template;
855    while let Some(start) = rest.find('{') {
856        out.push_str(&rest[..start]);
857        let after = &rest[start + 1..];
858        let Some(end) = after.find('}') else {
859            out.push_str(&rest[start..]);
860            return out;
861        };
862        let name = &after[..end];
863        match fields.get(name) {
864            Some(Value::String(text)) => out.push_str(text),
865            Some(value) => out.push_str(&value.to_string()),
866            None => out.push_str(&rest[start..=start + end + 1]),
867        }
868        rest = &after[end + 1..];
869    }
870    out.push_str(rest);
871    out
872}
873
874/// `text` made terminal-safe and cut to `limit` bytes at a character
875/// boundary, ending in `…` when cut. Newlines and tabs stay in a detail and
876/// become spaces elsewhere.
877fn bounded(text: &str, limit: usize, multiline: bool) -> String {
878    let mut safe = crate::sanitize_display_text(text).into_owned();
879    if !multiline {
880        safe = safe.replace(['\n', '\t'], " ");
881    }
882    if safe.len() <= limit {
883        return safe;
884    }
885    let ellipsis = '…';
886    let mut end = limit - ellipsis.len_utf8();
887    while !safe.is_char_boundary(end) {
888        end -= 1;
889    }
890    safe.truncate(end);
891    safe.push(ellipsis);
892    safe
893}
