24/// One source of the `AGENTS.md` turn: the attachment that loaded its
25/// files, whether they are a project's, and the files.
26#[derive(Clone, Copy, Debug)]
27pub struct AgentMdSource<'a> {
28    pub attachment: &'a str,
29    pub project: bool,
30    pub entries: &'a [AgentMdEntry],
31}
32
33/// The `AGENTS.md` turn, the one user turn holding every loaded file, from
34/// the latest load of each source in the latest run, in log order: the
35/// project's file first, then each workspace's (projects spec section 6).
36/// A workspace's file names its attachment when several attachments gave
37/// files; a project's file says where it is from in its source,
38/// `project:<name>/AGENTS.md`. Empty when no source has a file.
39#[must_use]
40pub fn agent_md_turn(loaded: &[AgentMdSource<'_>]) -> String {
41    let several = loaded.iter().filter(|source| !source.project).count() > 1;
42    let files = loaded
43        .iter()
44        .filter(|source| source.project)
45        .chain(loaded.iter().filter(|source| !source.project))
46        .flat_map(|source| {
47            source.entries.iter().map(move |entry| {
48                let from = escape_attribute(entry.source.as_str());
49                let runtime = if several && !source.project {
50                    format!(" runtime=\"{}\"", escape_attribute(source.attachment))
51                } else {
52                    String::new()
53                };
54                format!(
55                    "<contents from=\"{from}\"{runtime}>\n{}\n</contents>",
56                    entry.content
57                )
58            })
59        })
60        .collect::<Vec<_>>();
61    if files.is_empty() {
62        return String::new();
63    }
64    let files = files.join("\n\n");
65    format!("{AGENT_MD_PREAMBLE}\n\n{files}\n\n{AGENT_MD_FOOTNOTE}\n</system-reminder>")
66}

68/// `value` as an attribute of the elements the model reads: `&`, `"`,
69/// `<`, and `>` escaped, nothing else.
70#[must_use]
71pub fn escape_xml_attribute(value: &str) -> String {
72    value
73        .replace('&', "&amp;")
74        .replace('"', "&quot;")
75        .replace('<', "&lt;")
76        .replace('>', "&gt;")
77}
78
79/// `value` as the text of an element the model reads: `&`, `<`, and `>`
80/// escaped, so quotes and apostrophes read as written.
81#[must_use]
82pub fn escape_xml_text(value: &str) -> String {
83    value
84        .replace('&', "&amp;")
85        .replace('<', "&lt;")
86        .replace('>', "&gt;")
87}
88
89fn escape_attribute(value: &str) -> String {
90    escape_xml_attribute(value)
91}

105/// The user turn that follows a reply an interrupt cut off, from the next
106/// request on, when the reply kept any content.
107///
108/// The engine adds it when it assembles a request, never to the log: the
109/// log records what the model actually produced, and the marker tells the
110/// model that the fragment it sees is neither its ending nor a user's
111/// message. The wording names no cause: the user, a delegating parent, or
112/// shutdown may have cancelled the run.
113pub const INTERRUPTED_TURN_MARKER: &str =
114    "[Response interrupted — it was cut off before it finished]";

120/// Starts every loop warning the engine appends to a tool's output.
121pub const LOOP_WARNING_MARKER: &str = "\n\n<system-reminder>\nRepeated tool calls detected";

123/// The loop warning at the end of a tool's `output`, if it carries one,
124/// without the blank line that separates it: what the model read beyond
125/// the tool's own output.
126#[must_use]
127pub fn loop_warning(output: &str) -> Option<&str> {
128    output
129        .rfind(LOOP_WARNING_MARKER)
130        .map(|index| &output[index + 2..])
131}
132
133/// `output` without the loop warning at its end, if it carries one.
134#[must_use]
135pub fn without_loop_warning(output: &str) -> &str {
136    output
137        .rfind(LOOP_WARNING_MARKER)
138        .map_or(output, |index| &output[..index])
139}

141/// What the model reads in place of a tool output that left its context
142/// for an artifact: the artifact's ID (the hex SHA-256 of
143/// `artifact://sha256/<hex>`), the output's size, and how many messages
144/// the tool emitted with it, which go with it.
145#[must_use]
146pub fn tool_output_elision_marker(
147    artifact_id: &str,
148    original_bytes: u64,
149    additional_message_count: usize,
150) -> String {
151    let mut marker = format!(
152        "[tool output elided; {original_bytes} bytes retained; read it with `read_artifact(artifact=\"{artifact_id}\")`]"
153    );
154    if additional_message_count > 0 {
155        marker.push_str(&format!(
156            " {additional_message_count} tool-emitted message(s) were elided with this result and are not recoverable."
157        ));
158    }
159    marker
160}

162/// The user turn a changed instructions announcement renders as in the
163/// model's history (stable prefix spec, section 6.3). A block's text is the
164/// system prompt's own, so it is not escaped.
165/// One of the engine's own notices to the model, such as changed
166/// instructions or a goal the user paused, as the model reads it: inside
167/// `<system-reminder>` tags. What other agents and jobs send keeps its own
168/// element (`<subagent_report>`, `<agent_message>`, ...).
169#[must_use]
170pub fn system_reminder(body: &str) -> String {
171    format!("<system-reminder>\n{}\n</system-reminder>", body.trim_end())
172}

174/// What opens the notice of a reply's tags that attached nothing: each
175/// follows on a line of its own, `- <what it named>: <why>`.
176pub const ATTACHMENT_FAILURES_LEAD: &str = "Attachments of your last reply that were not attached:";
177
178/// The notice of a reply's tags that attached nothing, one line of
179/// `notes` each.
180#[must_use]
181pub fn attachment_failures(notes: &[String]) -> String {
182    let mut text = String::from(ATTACHMENT_FAILURES_LEAD);
183    for note in notes {
184        text.push_str("\n- ");
185        text.push_str(note);
186    }
187    system_reminder(&text)
188}
189
190/// The lines of an [`attachment_failures`] notice, each `- …`, or of one
191/// that earlier builds wrote (lines of `[attachment not attached] …`
192/// after [`TOOL_EMITTED_SYSTEM_USER_MARKER`]); `None` for any other text.
193#[must_use]
194pub fn attachment_failure_lines(text: &str) -> Option<&str> {
195    if let Some(notes) = text
196        .strip_prefix("<system-reminder>\n")
197        .and_then(|rest| rest.strip_prefix(ATTACHMENT_FAILURES_LEAD))
198        .and_then(|rest| rest.strip_prefix('\n'))
199        .and_then(|rest| rest.strip_suffix("\n</system-reminder>"))
200    {
201        return Some(notes);
202    }
203    let notes = text
204        .strip_prefix(TOOL_EMITTED_SYSTEM_USER_MARKER)?
205        .trim_start_matches('\n');
206    notes
207        .starts_with("[attachment not attached]")
208        .then_some(notes)
209}

211#[must_use]
212pub fn prompt_blocks_announcement(blocks: &[AnnouncedPromptBlock]) -> String {
213    let mut text = String::from("<instructions_changed>\n");
214    for block in blocks {
215        let name = escape_attribute(&block.key);
216        match &block.text {
217            Some(body) => {
218                text.push_str(&format!("<block name=\"{name}\">\n"));
219                text.push_str(body.trim_end());
220                text.push_str("\n</block>\n");
221            }
222            None => text.push_str(&format!("<block name=\"{name}\" removed=\"true\"/>\n")),
223        }
224    }
225    text.push_str(
226        "</instructions_changed>\nEach block replaces the part of the system prompt it names, and a removed block no longer applies: runtimes is where this session's tools work, attachments how to attach files to a reply, tool_instructions the <tool_instructions> sections, and skills the skill listing.",
227    );
228    system_reminder(&text)
229}

243/// The user turn a deferred tools announcement renders as in the model's
244/// history (deferred tools spec section 7.2): the groups that became
245/// loadable, each with its summary the first time, then those that left.
246/// `first` is whether no earlier announcement since the latest checkpoint
247/// added groups; only the first explains tool_search.
248#[must_use]
249pub fn deferred_tools_announcement(
250    added: &[AnnouncedToolGroup],
251    removed: &[AnnouncedToolGroup],
252    first: bool,
253) -> String {
254    fn group(text: &mut String, group: &AnnouncedToolGroup) {
255        text.push_str(&format!(
256            "<group name=\"{}\" source=\"{}\"",
257            escape_attribute(&group.name),
258            source_label(group.source)
259        ));
260        if let Some(summary) = &group.summary {
261            text.push_str(&format!(" summary=\"{}\"", escape_attribute(summary)));
262        }
263        text.push_str(">\n");
264        for tool in &group.tools {
265            text.push_str(tool);
266            text.push('\n');
267        }
268        text.push_str("</group>");
269    }
270    let mut parts = Vec::new();
271    if !added.is_empty() {
272        let header = if first {
273            DEFERRED_TOOLS_ADDED
274        } else {
275            DEFERRED_TOOLS_ADDED_MORE
276        };
277        let mut text = format!("{header}\n");
278        for (index, announced) in added.iter().enumerate() {
279            if index > 0 {
280                text.push('\n');
281            }
282            group(&mut text, announced);
283        }
284        parts.push(text);
285    }
286    if !removed.is_empty() {
287        let mut text = format!("{DEFERRED_TOOLS_REMOVED}\n");
288        for (index, announced) in removed.iter().enumerate() {
289            if index > 0 {
290                text.push('\n');
291            }
292            group(&mut text, announced);
293        }
294        parts.push(text);
295    }
296    system_reminder(&parts.join("\n\n"))
297}

299/// The user turn a `tools_availability_announced` event renders as (session
300/// tool groups spec, section 8.2): tools the round declared that left, on
301/// one machine (with where they still are) or everywhere, then those that
302/// came back.
303#[must_use]
304pub fn tools_availability_notice(
305    unavailable: &[UnavailableToolGroup],
306    available_again: &[UnavailableToolGroup],
307) -> String {
308    fn group(text: &mut String, group: &UnavailableToolGroup) {
309        text.push_str(&format!(
310            "<group name=\"{}\" source=\"{}\"",
311            escape_attribute(&group.name),
312            source_label(group.source)
313        ));
314        if !group.available_on.is_empty() {
315            text.push_str(&format!(
316                " available_on=\"{}\"",
317                escape_attribute(&group.available_on.join(", "))
318            ));
319        }
320        text.push_str(">\n");
321        for tool in &group.tools {
322            text.push_str(tool);
323            text.push('\n');
324        }
325        text.push_str("</group>");
326    }
327    /// The parts of `groups`, one per machine (every machine first), each
328    /// opened by `header(machine)`.
329    fn parts(
330        groups: &[UnavailableToolGroup],
331        header: impl Fn(Option<&str>) -> String,
332    ) -> Vec<String> {
333        let mut machines = Vec::<Option<&str>>::new();
334        for group in groups {
335            let machine = group.machine.as_deref();
336            if !machines.contains(&machine) {
337                machines.push(machine);
338            }
339        }
340        machines.sort();
341        machines
342            .into_iter()
343            .map(|machine| {
344                let mut text = format!("{}\n", header(machine));
345                for (index, member) in groups
346                    .iter()
347                    .filter(|group| group.machine.as_deref() == machine)
348                    .enumerate()
349                {
350                    if index > 0 {
351                        text.push('\n');
352                    }
353                    group(&mut text, member);
354                }
355                text
356            })
357            .collect()
358    }
359    let mut text = parts(unavailable, |machine| {
360        match machine {
361        Some(machine) => format!(
362            "The following tool groups are no longer available on the machine \"{machine}\" in this session. Calls to them with machine=\"{machine}\" will fail; they are still available on the machines each group lists:"
363        ),
364        None => "The following tool groups are no longer available in this session. Their tools stay listed until the context is next refreshed, but calls to them will fail:".to_owned(),
365    }
366    });
367    text.extend(parts(available_again, |machine| match machine {
368        Some(machine) => format!(
369            "The following tool groups are available again on the machine \"{machine}\" in this session:"
370        ),
371        None => "The following tool groups are available again in this session:".to_owned(),
372    }));
373    system_reminder(&text.join("\n\n"))
374}

376fn source_label(source: ToolGroupSource) -> &'static str {
377    match source {
378        ToolGroupSource::Mcp => "mcp",
379        ToolGroupSource::Plugin => "plugin",
380        ToolGroupSource::Builtin => "builtin",
381        ToolGroupSource::Unknown => "unknown",
382    }

389    #[test]
390    fn a_loop_warning_splits_from_the_output_it_ends() {
391        let output = format!("listing{LOOP_WARNING_MARKER}: you have made ...\n</system-reminder>");
392        assert_eq!(without_loop_warning(&output), "listing");

394            loop_warning(&output),

397            )

400        assert_eq!(without_loop_warning("listing"), "listing");
401    }
