22impl Serialize for JsonRpcVersion {
23    fn serialize<S>(&self, s: S) -> Result<S::Ok, S::Error>
24    where
25        S: serde::Serializer,
26    {
27        s.serialize_str("2.0")
28    }
29}
30impl<'de> Deserialize<'de> for JsonRpcVersion {
31    fn deserialize<D>(d: D) -> Result<Self, D::Error>
32    where
33        D: serde::Deserializer<'de>,
34    {
35        let value = String::deserialize(d)?;
36        if value == "2.0" {
37            Ok(Self::current())
38        } else {
39            Err(serde::de::Error::custom(
40                "JSON-RPC version must be exactly 2.0",
41            ))
42        }
43    }
44}

50        Cow::Borrowed("JsonRpcVersion")
51    }
52    fn json_schema(_: &mut SchemaGenerator) -> Schema {
53        json_schema!({"type":"string","const":"2.0"})
54    }

188/// A self-declared kind label: a lowercase ASCII letter followed by up to 15
189/// lowercase letters, digits, or hyphens. The daemon only displays it and
190/// never branches on it.
191macro_rules! kind_label {
192    ($(#[$meta:meta])* $name:ident, $what:literal, $description:literal) => {
193        $(#[$meta])*
194        #[derive(Clone, Debug, Eq, Hash, PartialEq)]
195        pub struct $name(String);
196        impl $name {
197            pub const MAX_LEN: usize = 16;
198
199            pub fn new(value: impl Into<String>) -> Result<Self, WireStringError> {
200                let value = value.into();
201                let valid = value.len() <= Self::MAX_LEN
202                    && value.bytes().enumerate().all(|(index, byte)| {
203                        byte.is_ascii_lowercase()
204                            || (index > 0 && (byte.is_ascii_digit() || byte == b'-'))
205                    })
206                    && !value.is_empty();
207                if valid {
208                    Ok(Self(value))
209                } else {
210                    Err(WireStringError::Invalid($what))
211                }
212            }
213
214            #[must_use]
215            pub fn as_str(&self) -> &str {
216                &self.0
217            }
218        }
219        impl fmt::Display for $name {
220            fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
221                formatter.write_str(&self.0)
222            }
223        }
224        impl Serialize for $name {
225            fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
226            where
227                S: serde::Serializer,
228            {
229                serializer.serialize_str(&self.0)
230            }
231        }
232        impl<'de> Deserialize<'de> for $name {
233            fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
234            where
235                D: serde::Deserializer<'de>,
236            {
237                Self::new(String::deserialize(deserializer)?).map_err(serde::de::Error::custom)
238            }
239        }
240        impl JsonSchema for $name {
241            fn inline_schema() -> bool {
242                true
243            }
244            fn schema_name() -> Cow<'static, str> {
245                Cow::Borrowed(stringify!($name))
246            }
247            fn json_schema(_: &mut SchemaGenerator) -> Schema {
248                json_schema!({
249                    "type":"string",
250                    "pattern":"^[a-z][a-z0-9-]{0,15}$",
251                    "description":$description
252                })
253            }
254        }
255    };
256}
