17pub struct JsonRpcVersion(());
18impl JsonRpcVersion {
19    #[must_use]
20    pub const fn current() -> Self {
21        Self(())
22    }
23}
24impl Serialize for JsonRpcVersion {
25    fn serialize<S>(&self, s: S) -> Result<S::Ok, S::Error>
26    where
27        S: serde::Serializer,
28    {
29        s.serialize_str("2.0")
30    }
31}
32impl<'de> Deserialize<'de> for JsonRpcVersion {
33    fn deserialize<D>(d: D) -> Result<Self, D::Error>
34    where
35        D: serde::Deserializer<'de>,
36    {
37        let value = String::deserialize(d)?;
38        if value == "2.0" {
39            Ok(Self::current())
40        } else {
41            Err(serde::de::Error::custom(
42                "JSON-RPC version must be exactly 2.0",
43            ))
44        }
45    }
46}

52        Cow::Borrowed("JsonRpcVersion")
53    }
54    fn json_schema(_: &mut SchemaGenerator) -> Schema {
55        json_schema!({"type":"string","const":"2.0"})
56    }

193/// A self-declared kind label: a lowercase ASCII letter followed by up to 15
194/// lowercase letters, digits, or hyphens. The daemon only displays it and
195/// never branches on it.
196macro_rules! kind_label {
197    ($(#[$meta:meta])* $name:ident, $what:literal, $description:literal) => {
198        $(#[$meta])*
199        #[derive(Clone, Debug, Eq, Hash, PartialEq)]
200        pub struct $name(String);
201        impl $name {
202            pub const MAX_LEN: usize = 16;
203
204            pub fn new(value: impl Into<String>) -> Result<Self, WireStringError> {
205                let value = value.into();
206                let valid = value.len() <= Self::MAX_LEN
207                    && value.bytes().enumerate().all(|(index, byte)| {
208                        byte.is_ascii_lowercase()
209                            || (index > 0 && (byte.is_ascii_digit() || byte == b'-'))
210                    })
211                    && !value.is_empty();
212                if valid {
213                    Ok(Self(value))
214                } else {
215                    Err(WireStringError::Invalid($what))
216                }
217            }
218
219            #[must_use]
220            pub fn as_str(&self) -> &str {
221                &self.0
222            }
223        }
224        impl fmt::Display for $name {
225            fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
226                formatter.write_str(&self.0)
227            }
228        }
229        impl Serialize for $name {
230            fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
231            where
232                S: serde::Serializer,
233            {
234                serializer.serialize_str(&self.0)
235            }
236        }
237        impl<'de> Deserialize<'de> for $name {
238            fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
239            where
240                D: serde::Deserializer<'de>,
241            {
242                Self::new(String::deserialize(deserializer)?).map_err(serde::de::Error::custom)
243            }
244        }
245        impl JsonSchema for $name {
246            fn inline_schema() -> bool {
247                true
248            }
249            fn schema_name() -> Cow<'static, str> {
250                Cow::Borrowed(stringify!($name))
251            }
252            fn json_schema(_: &mut SchemaGenerator) -> Schema {
253                json_schema!({
254                    "type":"string",
255                    "pattern":"^[a-z][a-z0-9-]{0,15}$",
256                    "description":$description
257                })
258            }
259        }
260    };
261}
