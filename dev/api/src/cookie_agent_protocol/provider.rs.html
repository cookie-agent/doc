12macro_rules! positive_counter {
13    ($name:ident, $description:literal) => {
14        #[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Serialize)]
15        #[serde(transparent)]
16        pub struct $name(u64);
17
18        impl $name {
19            pub fn new(value: u64) -> Result<Self, ProviderSchemaError> {
20                if (1..=MAX_SAFE_INTEGER).contains(&value) {
21                    Ok(Self(value))
22                } else {
23                    Err(ProviderSchemaError::InvalidCounter)
24                }
25            }
26
27            #[must_use]
28            pub const fn get(self) -> u64 { self.0 }
29        }
30
31        impl<'de> Deserialize<'de> for $name {
32            fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
33            where D: serde::Deserializer<'de> {
34                Self::new(u64::deserialize(deserializer)?).map_err(serde::de::Error::custom)
35            }
36        }
37
38        impl JsonSchema for $name {
39            fn inline_schema() -> bool { true }
40            fn schema_name() -> Cow<'static, str> { Cow::Borrowed(stringify!($name)) }
41            fn json_schema(_: &mut SchemaGenerator) -> Schema {
42                json_schema!({"type":"integer","minimum":1,"maximum":MAX_SAFE_INTEGER,"description":$description})
43            }
44        }
45    };
46}

359/// Why a provider's catalog model is not selectable.
360///
361/// A value this build does not know reads as `unsupported`.

363    Clone, Copy, Debug, Deserialize, Eq, Hash, JsonSchema, Ord, PartialEq, PartialOrd, Serialize,
364)]
365#[serde(rename_all = "snake_case")]
366#[schemars(extend("x-open" = true))]
367pub enum ModelUnavailableKind {
368    /// The catalog parser quarantined the model record.
369    Quarantined,
370    /// Provider setup values (region, resource, deployment) are missing.
371    NeedsSetup,
372    /// No usable credentials are configured or stored for the provider.
373    NeedsCredentials,

379/// Per-provider model availability counts over every catalog model row that
380/// is not deprecated, disabled, or non-text.
381#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
382pub struct ProviderModelCounts {
383    pub available: u32,
384    pub quarantined: u32,
385    pub unsupported: u32,
386    pub needs_setup: u32,
387    pub needs_credentials: u32,
388}
389
390impl ProviderModelCounts {
391    #[must_use]
392    pub const fn unavailable(&self) -> u32 {
393        self.quarantined
394            .saturating_add(self.unsupported)
395            .saturating_add(self.needs_setup)
396            .saturating_add(self.needs_credentials)
397    }
398
399    #[must_use]
400    pub const fn count(&self, kind: ModelUnavailableKind) -> u32 {
401        match kind {
402            ModelUnavailableKind::Quarantined => self.quarantined,
403            ModelUnavailableKind::Unsupported => self.unsupported,
404            ModelUnavailableKind::NeedsSetup => self.needs_setup,
405            ModelUnavailableKind::NeedsCredentials => self.needs_credentials,
406        }
407    }
408}

626impl<'de> Deserialize<'de> for ProviderCredentialValues {
627    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
628    where
629        D: serde::Deserializer<'de>,
630    {
631        struct Visitor;
632        impl<'de> serde::de::Visitor<'de> for Visitor {
633            type Value = BTreeMap<String, String>;
634            fn expecting(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
635                formatter.write_str("a unique credential field map")
636            }
637            fn visit_map<A>(self, mut access: A) -> Result<Self::Value, A::Error>
638            where
639                A: serde::de::MapAccess<'de>,
640            {
641                let mut raw = BTreeMap::new();
642                while let Some((key, value)) = access.next_entry::<String, String>()? {
643                    if raw.len() >= 32 {
644                        return Err(serde::de::Error::custom("auth values exceed 32 fields"));
645                    }
646                    if raw.insert(key, value).is_some() {
647                        return Err(serde::de::Error::custom("duplicate auth credential field"));
648                    }
649                }
650                Ok(raw)
651            }
652        }
653        let raw = deserializer.deserialize_map(Visitor)?;
654        if raw.len() > 32
655            || raw.iter().any(|(key, value)| {
656                AuthFieldName::new(key.clone()).is_err()
657                    || value.is_empty()
658                    || value.len() > 16 * 1024
659            })
660        {
661            return Err(serde::de::Error::custom(
662                "auth values must contain at most 32 strict bounded nonempty credential fields",
663            ));
664        }
665        Ok(Self(raw))
666    }
667}

720
721fn deserialize_setup_values<'de, D>(
722    d: D,
723) -> Result<BTreeMap<SetupFieldId, SafeSetupValue>, D::Error>
724where
725    D: serde::Deserializer<'de>,
726{
727    struct Visitor;
728    impl<'de> serde::de::Visitor<'de> for Visitor {
729        type Value = BTreeMap<SetupFieldId, SafeSetupValue>;
730        fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
731            f.write_str("a unique bounded setup field map")
732        }
733        fn visit_map<A>(self, mut access: A) -> Result<Self::Value, A::Error>
734        where
735            A: serde::de::MapAccess<'de>,
736        {
737            let mut values = BTreeMap::new();
738            while let Some((key, value)) = access.next_entry::<SetupFieldId, SafeSetupValue>()? {
739                if values.len() >= 32 {
740                    return Err(serde::de::Error::custom("setup values exceed 32 fields"));
741                }
742                if values.insert(key, value).is_some() {
743                    return Err(serde::de::Error::custom("duplicate setup field"));
744                }
745            }
746            Ok(values)
747        }
748    }
749    d.deserialize_map(Visitor)
750}
