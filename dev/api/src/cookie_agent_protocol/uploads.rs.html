1//! Uploading user media into a session tree's artifact store, and reading
2//! any artifact of a tree back, as specified in
3//! `docs/site/specs/compatibility-v1.md` section 11.3.
4//!
5//! An upload is begun, sent in chunks of at most [`ARTIFACT_CHUNK_BYTES`]
6//! in order, and finished with the digest of its bytes, which returns the
7//! [`ArtifactRef`] a `file` content part names. Bytes travel as base64.
8
9use std::{borrow::Cow, fmt};
10
11use schemars::{JsonSchema, Schema, SchemaGenerator, json_schema};
12use serde::{Deserialize, Serialize};
13
14use crate::{ArtifactRef, SessionId, WireStringError};
15
16pub const ARTIFACT_UPLOAD_BEGIN_METHOD: &str = "artifact.upload.begin";
17pub const ARTIFACT_UPLOAD_CHUNK_METHOD: &str = "artifact.upload.chunk";
18pub const ARTIFACT_UPLOAD_FINISH_METHOD: &str = "artifact.upload.finish";
19pub const ARTIFACT_UPLOAD_CANCEL_METHOD: &str = "artifact.upload.cancel";
20pub const ARTIFACT_READ_METHOD: &str = "artifact.read";
21
22/// The most decoded bytes one `artifact.upload.chunk` carries or one
23/// `artifact.read` returns.
24pub const ARTIFACT_CHUNK_BYTES: u64 = 1024 * 1024;
25
26/// The longest base64 text of [`ARTIFACT_CHUNK_BYTES`] bytes.
27const MAX_CHUNK_BASE64_LEN: u64 = ARTIFACT_CHUNK_BYTES.div_ceil(3) * 4;
28
29/// An upload in progress, unique within the daemon and valid only on the
30/// connection that began it.
31#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Serialize)]
32#[serde(transparent)]
33pub struct UploadId(String);
34
35impl UploadId {
36    pub const MAX_LEN: usize = 64;
37
38    pub fn new(value: impl Into<String>) -> Result<Self, WireStringError> {
39        let value = value.into();
40        let valid = !value.is_empty()
41            && value.len() <= Self::MAX_LEN
42            && value
43                .bytes()
44                .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_');
45        if valid {
46            Ok(Self(value))
47        } else {
48            Err(WireStringError::Invalid(
49                "an upload ID is 1 to 64 ASCII letters, digits, `-`, or `_`",
50            ))
51        }
52    }
53
54    #[must_use]
55    pub fn as_str(&self) -> &str {
56        &self.0
57    }
58}
59
60impl fmt::Display for UploadId {
61    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
62        formatter.write_str(&self.0)
63    }
64}
65
66impl<'de> Deserialize<'de> for UploadId {
67    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
68    where
69        D: serde::Deserializer<'de>,
70    {
71        Self::new(String::deserialize(deserializer)?).map_err(serde::de::Error::custom)
72    }
73}
74
75impl JsonSchema for UploadId {
76    fn inline_schema() -> bool {
77        true
78    }
79    fn schema_name() -> Cow<'static, str> {
80        Cow::Borrowed("UploadId")
81    }
82    fn json_schema(_: &mut SchemaGenerator) -> Schema {
83        json_schema!({
84            "type": "string",
85            "pattern": "^[A-Za-z0-9_-]{1,64}$",
86            "description": "An upload in progress on this connection."
87        })
88    }
89}

91/// `artifact.upload.begin`: starts an upload of `byte_length` bytes into
92/// the store of `session_id`'s tree.
93#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
94#[serde(deny_unknown_fields)]
95pub struct ArtifactUploadBeginParams {
96    pub session_id: SessionId,
97    /// The upload's size; at most the largest attachment cap (25 MiB).
98    #[schemars(range(min = 1))]
99    pub byte_length: u64,
100    /// The name to attach the file under, without a directory. It never
101    /// decides the media type.
102    #[serde(default, skip_serializing_if = "Option::is_none")]
103    #[schemars(length(min = 1, max = 255))]
104    pub filename: Option<String>,
105}
106
107#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
108pub struct ArtifactUploadBeginResult {
109    pub upload_id: UploadId,
110    /// The most decoded bytes one chunk may carry.
111    pub chunk_bytes: u64,
112}
113
114/// `artifact.upload.chunk`: the bytes at `offset`, which must be exactly
115/// the bytes received so far. Send a chunk once the previous one is
116/// answered: requests run concurrently, so pipelined chunks may arrive out
117/// of order and be refused.
118#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
119#[serde(deny_unknown_fields)]
120pub struct ArtifactUploadChunkParams {
121    pub upload_id: UploadId,
122    pub offset: u64,
123    /// Base64 (standard alphabet, padded) of at most `chunk_bytes` bytes.
124    #[schemars(length(min = 1, max = MAX_CHUNK_BASE64_LEN))]
125    pub data: String,
126}
127
128#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
129pub struct ArtifactUploadChunkResult {}
130
131/// `artifact.upload.finish`: checks the digest, recognizes and validates
132/// the content, and stores it.
133#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
134#[serde(deny_unknown_fields)]
135pub struct ArtifactUploadFinishParams {
136    pub upload_id: UploadId,
137    /// Lowercase hex SHA-256 of every byte sent.
138    #[schemars(length(min = 64, max = 64), regex(pattern = "^[0-9a-f]{64}$"))]
139    pub sha256: String,
140}

168/// `artifact.read`: `length` bytes from `offset` of an artifact in the
169/// store of `session_id`'s tree: a user upload, a tool attachment, or the
170/// `full` content of a display block.
171#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
172#[serde(deny_unknown_fields)]
173pub struct ArtifactReadParams {
174    pub session_id: SessionId,
175    #[schemars(length(min = 64, max = 64), regex(pattern = "^[0-9a-f]{64}$"))]
176    pub sha256: String,
177    /// At most `total_bytes`; reading at `total_bytes` returns no data.
178    pub offset: u64,
179    #[schemars(range(min = 1, max = ARTIFACT_CHUNK_BYTES))]
180    pub length: u64,
181}
182
183#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
184pub struct ArtifactReadResult {
185    /// Base64 of the bytes read: fewer than `length` at the end.
186    pub data: String,
187    pub total_bytes: u64,
188    /// The media type recognized from the content; `text/plain` for UTF-8
189    /// text and `application/octet-stream` for anything else.
190    pub media_type: String,
191}

194mod tests {
195    use serde_json::json;
196
197    use super::*;

199    #[test]
200    fn upload_ids_are_short_url_safe_labels() {
201        assert!(UploadId::new("upl_0192-abc").is_ok());
202        for invalid in ["", "has space", "slash/", &"a".repeat(65)] {
203            assert!(UploadId::new(invalid).is_err(), "{invalid:?}");
204        }
205        assert!(serde_json::from_value::<UploadId>(json!("a b")).is_err());
206    }
207
208    #[test]
209    fn a_chunk_of_the_largest_size_fits_its_base64_bound() {
210        // Padded base64 is four characters per three bytes, rounded up.
211        assert_eq!(MAX_CHUNK_BASE64_LEN, 1_398_104);
212    }
213
214    #[test]
215    fn finish_results_nest_the_reference_a_file_part_names() {
216        let result = ArtifactUploadFinishResult {
217            artifact: ArtifactRef {
218                sha256: "ab".repeat(32),
219                byte_length: 3,
220                media_type: "image/png".into(),
221                filename: Some("a.png".into()),
222            },
223            image: Some(ImageSize {
224                width: 2,
225                height: 1,
226            }),
227        };
228        assert_eq!(
229            serde_json::to_value(&result).unwrap(),
230            json!({
231                "artifact": {"sha256": "ab".repeat(32), "byte_length": 3,
232                             "media_type": "image/png", "filename": "a.png"},
233                "image": {"width": 2, "height": 1}
234            })
235        );
236    }
