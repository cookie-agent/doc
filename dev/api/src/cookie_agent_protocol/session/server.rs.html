53/// Per-connection facilities available to a server implementation.
54#[derive(Clone)]
55pub struct ServerContext {
56    connection_id: u64,
57    /// The origin derived from the handshake's client identity.
58    client_origin: Arc<std::sync::OnceLock<EventOrigin>>,
59    /// Serialized frames, written to the transport in queue order.
60    notifications: mpsc::Sender<String>,
61    shutdown: CancellationToken,
62    subscribed_sessions: Arc<Mutex<HashSet<SessionId>>>,
63    /// The task forwarding each session's live tail on this connection.
64    session_tails: Arc<Mutex<HashMap<SessionId, tokio::task::JoinHandle<()>>>>,
65    /// Opens once the response of the request this context was handed to has
66    /// been queued. `None` for connection-scoped contexts.
67    response_sent: Option<CancellationToken>,
68}

70impl ServerContext {
71    fn new(notifications: mpsc::Sender<String>, shutdown: CancellationToken) -> Self {
72        Self {
73            connection_id: NEXT_CONNECTION_ID.fetch_add(1, Ordering::Relaxed),
74            client_origin: Arc::new(std::sync::OnceLock::new()),
75            notifications,
76            shutdown,
77            subscribed_sessions: Arc::new(Mutex::new(HashSet::new())),
78            session_tails: Arc::new(Mutex::new(HashMap::new())),
79            response_sent: None,
80        }
81    }
82
83    /// This connection's identity, unique within the process. A server
84    /// keys the live tail it registers for a session by it, so that a
85    /// connection holds at most one tail per session.
86    #[must_use]
87    pub fn connection_id(&self) -> u64 {
88        self.connection_id
89    }
90
91    /// The origin stamped on events this connection's requests cause,
92    /// derived from the client identity sent in the handshake. Requests run
93    /// only after the handshake, so it is always set when a request sees it.
94    #[must_use]
95    pub fn client_origin(&self) -> EventOrigin {
96        self.client_origin
97            .get()
98            .cloned()
99            .expect("requests run only after the handshake set the client origin")
100    }
101
102    /// Records the handshake identity. A repeated handshake keeps the first.
103    fn hello(&self, hello: &ClientHello) -> ServerHello {
104        let origin = self
105            .client_origin
106            .get_or_init(|| hello.client.origin())
107            .clone();
108        ServerHello {
109            protocol_version: crate::ProtocolVersion::current(),
110            origin,
111            machine_id: crate::paths::machine_id(),
112        }
113    }
114
115    /// Forward a session's live tail to this connection as
116    /// `events.subscription` notifications until the tail or the connection
117    /// ends.
118    ///
119    /// The connection holds one tail per session, so the producer must have
120    /// closed any tail this one replaces (see [`Self::connection_id`]): the
121    /// messages that tail still holds are forwarded first, then this one's,
122    /// keeping the connection in publication order. Two tails of one session
123    /// would each carry every message, and live-only output, which has no
124    /// sequence to de-duplicate by, would reach the client twice.
125    pub fn forward_session_tail(
126        &self,
127        session_id: SessionId,
128        mut receiver: mpsc::Receiver<EventSubscriptionMessage>,
129    ) {
130        let mut tails = self
131            .session_tails
132            .lock()
133            .unwrap_or_else(|poisoned| poisoned.into_inner());
134        let replaced = tails.remove(&session_id);
135        let context = self.clone();
136        let forwarder = tokio::spawn(async move {
137            if let Some(replaced) = replaced {
138                let _ = replaced.await;
139            }
140            let shutdown = context.shutdown();
141            loop {
142                let message = tokio::select! {
143                    () = shutdown.cancelled() => return,
144                    message = receiver.recv() => match message {
145                        Some(message) => message,
146                        None => return,
147                    },
148                };
149                if context
150                    .notify("events.subscription", &message)
151                    .await
152                    .is_err()
153                {
154                    return;
155                }
156            }
157        });
158        tails.insert(session_id, forwarder);
159    }
160
161    /// Emit one JSON-RPC notification in connection order.
162    ///
163    /// A notification from work a request started (for example a live tail
164    /// registered by `events.subscribe`) is held until that request's
165    /// response is queued, so the client never sees the tail before the
166    /// snapshot it continues.
167    ///
168    /// The notification is serialized once, straight to its frame text.
169    pub async fn notify<T: Serialize + ?Sized>(
170        &self,
171        method: &str,
172        params: &T,
173    ) -> Result<(), TransportError> {
174        let notification = NotificationRef::new(method, params).to_text()?;
175        if let Some(response_sent) = &self.response_sent {
176            tokio::select! {
177                () = self.shutdown.cancelled() => return Err(TransportError::Closed),
178                () = response_sent.cancelled() => {}
179            }
180        }
181        tokio::select! {
182            () = self.shutdown.cancelled() => Err(TransportError::Closed),
183            result = self.notifications.send(notification) => result.map_err(|_| TransportError::Closed),
184        }
185    }
186
187    fn for_request(&self) -> (Self, CancellationToken) {
188        let response_sent = CancellationToken::new();
189        (
190            Self {
191                response_sent: Some(response_sent.clone()),
192                ..self.clone()
193            },
194            response_sent,
195        )
196    }
197
198    /// A cancellation token scoped to this protocol connection.
199    #[must_use]
200    pub fn shutdown(&self) -> CancellationToken {
201        self.shutdown.clone()
202    }
203
204    /// Marks a session as authorized for connection-scoped live notifications.
205    pub fn register_session_subscription(&self, session_id: SessionId) {
206        self.subscribed_sessions
207            .lock()
208            .unwrap_or_else(|poisoned| poisoned.into_inner())
209            .insert(session_id);
210    }
211
212    /// Stops this connection's live tail for a session: no further
213    /// `events.subscription` or plugin notification for it is sent. Returns
214    /// whether the connection was subscribed.
215    pub fn unregister_session_subscription(&self, session_id: SessionId) -> bool {
216        if let Some(forwarder) = self
217            .session_tails
218            .lock()
219            .unwrap_or_else(|poisoned| poisoned.into_inner())
220            .remove(&session_id)
221        {
222            forwarder.abort();
223        }
224        self.subscribed_sessions
225            .lock()
226            .unwrap_or_else(|poisoned| poisoned.into_inner())
227            .remove(&session_id)
228    }
229
230    /// The sessions this connection is subscribed to.
231    #[must_use]
232    pub fn subscribed_sessions(&self) -> Vec<SessionId> {
233        self.subscribed_sessions
234            .lock()
235            .unwrap_or_else(|poisoned| poisoned.into_inner())
236            .iter()
237            .copied()
238            .collect()
239    }
240
241    /// Returns whether this connection has successfully subscribed to the session.
242    #[must_use]
243    pub fn is_session_subscribed(&self, session_id: SessionId) -> bool {
244        self.subscribed_sessions
245            .lock()
246            .unwrap_or_else(|poisoned| poisoned.into_inner())
247            .contains(&session_id)
248    }
249}

573/// Run one complete server-side protocol session over a transport.
574///
575/// After the handshake, requests run concurrently and each response is
576/// written as soon as it is ready; clients match responses by id. A slow
577/// request, such as a large replay, never holds up the ones behind it.
578pub async fn serve<T, S>(
579    server: Arc<S>,
580    mut transport: T,
581    shutdown: CancellationToken,
582) -> Result<(), TransportError>
583where
584    T: Transport,
585    S: ServerProtocol,
586{
587    let (outbound, mut outbound_rx) = mpsc::channel(OUTBOUND_QUEUE_CAPACITY);
588    let context = ServerContext::new(outbound, shutdown);
589    let _guard = ConnectionShutdown(context.shutdown());
590    let in_flight = Arc::new(Semaphore::new(MAX_IN_FLIGHT_REQUESTS));
591    let mut permit = None;
592    let mut handshaken = false;
593    loop {
594        tokio::select! {
595            () = context.shutdown.cancelled() => return Ok(()),
596            acquired = in_flight.clone().acquire_owned(), if permit.is_none() => {
597                permit = Some(acquired.expect("the in-flight semaphore is never closed"));
598            }
599            incoming = transport.recv(), if permit.is_some() => {
600                let Some(frame) = incoming? else { return Ok(()); };
601                let incoming = match parse_incoming(frame).and_then(classify_incoming) {
602                    Ok(incoming) => incoming,
603                    Err(fault) => {
604                        transport.send(MessageFrame::Text(error_response(None, fault)?)).await?;
605                        continue;
606                    }
607                };
608                if !handshaken {
609                    let Incoming::Request { id, method, params } = incoming else {
610                        continue;
611                    };
612                    let response = if method != "handshake" {
613                        error_response(Some(id), handshake_required())?
614                    } else {
615                        match decode_handshake(params) {
616                            Ok(hello) => {
617                                handshaken = true;
618                                success_response(&id, &server_hello(&context, &hello))?
619                            }
620                            Err(fault) => error_response(Some(id), fault)?,
621                        }
622                    };
623                    transport.send(MessageFrame::Text(response)).await?;
624                    if handshaken {
625                        server.connected(context.clone()).await;
626                    }
627                    continue;
628                }
629                let permit = permit.take();
630                let server = server.clone();
631                let (request_context, response_sent) = context.for_request();
632                tokio::spawn(async move {
633                    let _permit = permit;
634                    // Opens the notification gate on every exit, a panic included.
635                    let _response_sent = response_sent.drop_guard();
636                    let Some(response) = handle(server.as_ref(), incoming, &request_context).await
637                    else {
638                        return;
639                    };
640                    tokio::select! {
641                        () = request_context.shutdown.cancelled() => {}
642                        _ = request_context.notifications.send(response) => {}
643                    }
644                });
645            }
646            Some(frame) = outbound_rx.recv() => {
647                transport.send(MessageFrame::Text(frame)).await?;
648            }
649        }
650    }
651}

653/// Runs one post-handshake message and builds its response, if it has one.
654async fn handle<S: ServerProtocol>(
655    server: &S,
656    incoming: Incoming,
657    context: &ServerContext,
658) -> Option<String> {
659    match incoming {
660        Incoming::Request { id, method, params } => {
661            let result = if method == "handshake" {
662                decode_handshake(params).map(|hello| server_hello(context, &hello))
663            } else {
664                dispatch(server, &method, params, context, true).await
665            };
666            let response = match result {
667                Ok(result) => success_response(&id, &result),
668                Err(fault) => error_response(Some(id.clone()), fault),
669            };
670            Some(response.unwrap_or_else(|_| {
671                error_response(Some(id), ServerFault::internal())
672                    .expect("an error response without data serializes")
673            }))
674        }
675        Incoming::Notification { method, params } => {
676            if method == "handshake" {
677                let _ = decode_handshake(params);
678            } else {
679                let _ = dispatch(server, &method, params, context, false).await;
680            }
681            None
682        }
683    }
684}

686fn server_hello(context: &ServerContext, hello: &ClientHello) -> Box<RawValue> {
687    serde_json::value::to_raw_value(&context.hello(hello)).expect("server hello serializes")
688}

825#[derive(Deserialize)]
826#[serde(deny_unknown_fields)]
827struct RawSessionRenameParams {
828    session_id: SessionId,
829    client_rename_id: ClientRenameId,
830    change: RawSessionRenameChange,
831}
832
833#[derive(Deserialize)]
834#[serde(tag = "type", rename_all = "snake_case", deny_unknown_fields)]
835enum RawSessionRenameChange {
836    Set { title: String },
837    Clear,
838    Reset,
839}
840
841fn decode_rename(params: Option<Value>) -> Result<SessionRenameParams, ServerFault> {
842    let params = params.unwrap_or(Value::Null);
843    if let Ok(request) = serde_json::from_value::<SessionRenameParams>(params.clone()) {
844        return Ok(request);
845    }
846    if serde_json::to_vec(&params)
847        .map_err(|_| ServerFault::invalid_params())?
848        .len()
849        > MAX_RAW_RENAME_PARAMS_BYTES
850    {
851        return Err(ServerFault::invalid_params());
852    }
853    let raw: RawSessionRenameParams =
854        serde_json::from_value(params).map_err(|_| ServerFault::invalid_params())?;
855    let invalid_title = rename_invalid_title(&raw);
856    let change = match raw.change {
857        RawSessionRenameChange::Set { title } => SessionRenameChange::Set {
858            title: SessionTitle::new(title).map_err(|_| invalid_title)?,
859        },
860        RawSessionRenameChange::Clear => SessionRenameChange::Clear,
861        RawSessionRenameChange::Reset => SessionRenameChange::Reset,
862    };
863    Ok(SessionRenameParams {
864        session_id: raw.session_id,
865        client_rename_id: raw.client_rename_id,
866        change,
867    })
868}
869
870fn rename_invalid_title(raw: &RawSessionRenameParams) -> ServerFault {
871    ServerFault::new(
872        -32602,
873        "session rename rejected",
874        serde_json::to_value(SessionRenameError {
875            code: SessionRenameErrorCode::InvalidTitle,
876            session_id: raw.session_id,
877            client_rename_id: raw.client_rename_id.clone(),
878        })
879        .ok(),
880    )
881}

900/// Parses a request frame. The frame's own buffer is wiped once parsed: a
901/// `provider.connect` request carries credentials, and the typed params that
902/// end up holding them zeroize when dropped.
903fn parse_incoming(frame: MessageFrame) -> Result<Value, ServerFault> {
904    let parsed = match frame {
905        MessageFrame::Value(value) => return Ok(value),
906        MessageFrame::Text(mut text) => {
907            let parsed = serde_json::from_str(&text);
908            text.zeroize();
909            parsed
910        }
911        MessageFrame::Bytes(bytes) => {
912            let parsed = serde_json::from_slice(&bytes);
913            // Shared bytes another holder still reads are left alone.
914            if let Ok(mut owned) = bytes.try_into_mut() {
915                owned.as_mut().zeroize();
916            }
917            parsed
918        }
919    };
920    parsed.map_err(|_| ServerFault::new(-32700, "parse error", None))
921}

923fn classify_incoming(value: Value) -> Result<Incoming, ServerFault> {
924    let object = value.as_object().ok_or_else(|| {
925        ServerFault::new(
926            -32600,
927            "invalid request",
928            Some(json!({ "detail": "a batch or scalar is not a request" })),
929        )
930    })?;
931    if object.contains_key("id") {
932        let request: Request = serde_json::from_value(value).map_err(|_| {
933            ServerFault::new(
934                -32600,
935                "invalid request",
936                Some(json!({ "detail": "invalid JSON-RPC request envelope" })),
937            )
938        })?;
939        if request.method.is_empty() {
940            return Err(ServerFault::new(
941                -32600,
942                "invalid request",
943                Some(json!({ "detail": "method must be non-empty" })),
944            ));
945        }
946        Ok(Incoming::Request {
947            id: request.id,
948            method: request.method,
949            params: request.params,
950        })
951    } else {
952        let notification: Notification = serde_json::from_value(value).map_err(|_| {
953            ServerFault::new(
954                -32600,
955                "invalid request",
956                Some(json!({ "detail": "invalid JSON-RPC notification envelope" })),
957            )
958        })?;
959        if notification.method.is_empty() {
960            return Err(ServerFault::new(
961                -32600,
962                "invalid request",
963                Some(json!({ "detail": "method must be non-empty" })),
964            ));
965        }
966        Ok(Incoming::Notification {
967            method: notification.method,
968            params: notification.params,
969        })
970    }
971}

982/// A success response borrowing its id and its already-serialized result.
983#[derive(Serialize)]
984struct SuccessResponseRef<'a> {
985    jsonrpc: JsonRpcVersion,
986    id: &'a JsonRpcId,
987    result: &'a RawValue,
988}
989
990/// Wraps an already-serialized result without re-serializing it: a replay
991/// page can run to megabytes, and it is copied into the frame as text.
992fn success_response(id: &JsonRpcId, result: &RawValue) -> Result<String, TransportError> {
993    Ok(serde_json::to_string(&SuccessResponseRef {
994        jsonrpc: JsonRpcVersion::current(),
995        id,
996        result,
997    })?)
998}
999
1000fn error_response(id: Option<JsonRpcId>, fault: ServerFault) -> Result<String, TransportError> {
1001    match id {
1002        Some(id) => Ok(serde_json::to_string(&Response::Error(ErrorResponse {
1003            jsonrpc: JsonRpcVersion::current(),
1004            id,
1005            error: JsonRpcError {
1006                code: fault.code,
1007                message: fault.message.into(),
1008                data: fault.data,
1009            },
1010        }))?),
1011        None => Ok(json!({
1012            "jsonrpc": "2.0",
1013            "id": null,
1014            "error": { "code": fault.code, "message": fault.message, "data": fault.data },
1015        })
1016        .to_string()),
1017    }
1018}
