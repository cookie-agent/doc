8/// The largest message, and so the largest frame, a WebSocket end accepts.
9/// Each protocol message goes out as a single frame, so this must fit the
10/// largest single event (a native context window carries up to 32 MiB) in
11/// its notification or replay page. Replay pages are otherwise kept to a few
12/// MiB by the daemon. Both ends are local and authenticated.
13pub const MAX_WEBSOCKET_MESSAGE_BYTES: usize = 128 * 1024 * 1024;
14
15/// One complete JSON-RPC message exchanged by a protocol transport.
16#[derive(Clone, PartialEq)]
17pub enum MessageFrame {
18    /// JSON text this process serialized.
19    Text(String),
20    /// JSON text as received from a socket, shared rather than copied. A
21    /// reader parses it directly from these bytes; they are not checked to
22    /// be UTF-8 until then.
23    Bytes(bytes::Bytes),
24    Value(Value),
25}
26
27impl MessageFrame {
28    /// The frame's JSON text, for a transport that writes text. A
29    /// [`Self::Value`] is serialized; a [`Self::Bytes`] that is not UTF-8
30    /// is an invalid frame.
31    pub fn into_text(self) -> Result<String, TransportError> {
32        match self {
33            Self::Text(text) => Ok(text),
34            Self::Bytes(bytes) => String::from_utf8(bytes.into()).map_err(|error| {
35                TransportError::Other(format!("frame is not UTF-8 text: {error}"))
36            }),
37            Self::Value(value) => Ok(serde_json::to_string(&value)?),
38        }
39    }
40}
41
42impl fmt::Debug for MessageFrame {
43    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
44        match self {
45            Self::Text(_) => formatter.write_str("MessageFrame::Text(<redacted>)"),
46            Self::Bytes(_) => formatter.write_str("MessageFrame::Bytes(<redacted>)"),
47            Self::Value(_) => formatter.write_str("MessageFrame::Value(<redacted>)"),
48        }
49    }
50}

52/// Frame-level channel with no JSON-RPC semantics.
53#[async_trait]
54pub trait Transport: Send {
55    async fn send(&mut self, frame: MessageFrame) -> Result<(), TransportError>;
56    async fn recv(&mut self) -> Result<Option<MessageFrame>, TransportError>;
57}
58
59pub use Transport as MessageStream;
60
61#[derive(Debug, Error)]
62pub enum TransportError {
63    #[error("transport closed")]
64    Closed,
65    #[error("invalid transport frame: {0}")]
66    InvalidFrame(#[from] serde_json::Error),
67    #[error("transport error: {0}")]
68    Other(String),
69}

71/// One end of an in-memory frame channel created by [`in_process_pair`].
72pub struct InProcessStream {
73    sender: mpsc::Sender<MessageFrame>,
74    receiver: mpsc::Receiver<MessageFrame>,
75}
76
77/// Creates two connected in-memory transports, typically a client end and the
78/// server end handed to `serve`. Each direction buffers up to `capacity`
79/// frames.
80#[must_use]
81pub fn in_process_pair(capacity: usize) -> (InProcessStream, InProcessStream) {
82    let (client_to_server_tx, client_to_server_rx) = mpsc::channel(capacity);
83    let (server_to_client_tx, server_to_client_rx) = mpsc::channel(capacity);
84    (
85        InProcessStream {
86            sender: client_to_server_tx,
87            receiver: server_to_client_rx,
88        },
89        InProcessStream {
90            sender: server_to_client_tx,
91            receiver: client_to_server_rx,
92        },
93    )
94}
95
96#[async_trait]
97impl Transport for InProcessStream {
98    async fn send(&mut self, frame: MessageFrame) -> Result<(), TransportError> {
99        self.sender
100            .send(frame)
101            .await
102            .map_err(|_| TransportError::Closed)
103    }
104
105    async fn recv(&mut self) -> Result<Option<MessageFrame>, TransportError> {
106        Ok(self.receiver.recv().await)
107    }
108}
