1//! Restarting the daemon from a client (`docs/site/specs/engine-configuration.md`
2//! section 11). Client protocol 1.19.0.
3//!
4//! A client asks with `daemon.restart`; the daemon tells every client with
5//! `daemon.restarting` while the restart waits and then closes their
6//! connections with WebSocket close code [`RESTART_CLOSE_CODE`], and they
7//! reconnect as after any dropped connection.
8
9use schemars::JsonSchema;
10use serde::{Deserialize, Serialize};
11
12pub const DAEMON_RESTART_METHOD: &str = "daemon.restart";
13pub const DAEMON_RESTARTING_METHOD: &str = "daemon.restarting";
14
15/// The WebSocket close code a restarting daemon closes client connections
16/// with: 1012, Service Restart.
17pub const RESTART_CLOSE_CODE: u16 = 1012;
18/// The close reason that goes with [`RESTART_CLOSE_CODE`].
19pub const RESTART_CLOSE_REASON: &str = "restarting";

21crate::open_enum! {
22    /// When the daemon restarts.
23    #[derive(Clone, Copy, Debug, Default, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
24    #[serde(rename_all = "snake_case")]
25    pub enum RestartWhen {
26        /// Once no run is active in any session and no background call is
27        /// pending. Runs keep starting meanwhile, and the restart waits for
28        /// them too; only once it is due are new runs refused.
29        #[default]
30        Idle,
31        /// At once: running turns end as interrupted.
32        Now,
33    }
34}
35
36/// Restart the daemon: it relaunches its own executable with the same
37/// arguments and environment. A second call while a restart is scheduled
38/// changes its `when`; `now` wins over `idle`.
39#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
40#[serde(deny_unknown_fields)]
41pub struct DaemonRestartParams {
42    #[serde(default)]
43    pub when: RestartWhen,
44}
45
46#[derive(Clone, Copy, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
47pub struct DaemonRestartResult {
48    /// Whether this call scheduled the restart; `false` when one was already
49    /// scheduled and this call only changed its `when`.
50    pub scheduled: bool,
51    /// The runs active when the daemon answered: what `idle` waits for and
52    /// what `now` interrupts.
53    pub active_runs: u32,
54    /// The background calls still running when the daemon answered, which
55    /// `idle` waits for too and `now` ends (client protocol 1.35.0; absent
56    /// from older daemons, which did not wait for them).
57    #[serde(default, skip_serializing_if = "is_zero")]
58    pub background_calls: u32,
59}

61/// A restart is scheduled, sent to every client when it is and again as the
62/// wait progresses.
63#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
64pub struct DaemonRestartingNotification {
65    pub when: RestartWhen,
66    /// The runs still active.
67    pub active_runs: u32,
68    /// The background calls still running (client protocol 1.35.0).
69    #[serde(default, skip_serializing_if = "is_zero")]
70    pub background_calls: u32,
71    /// What the restart waits for or is doing, for people to read.
72    #[serde(default, skip_serializing_if = "Option::is_none")]
73    pub reason: Option<String>,
74}

81// The daemon's status (`docs/site/specs/daemon-startup.md`). Client
82// protocol 1.65.0.
83
84pub const DAEMON_STATUS_METHOD: &str = "daemon.status";
85
86/// The first client protocol that reads [`DaemonStatus`]: the daemon answers
87/// its handshake while it starts, sends it `daemon.status`, and no longer
88/// `daemon.restarting`.
89pub const DAEMON_STATUS_SINCE: crate::SemVer = crate::SemVer::new(1, 65, 0);
90
91/// Whether a client of protocol `version` reads [`DaemonStatus`].
92#[must_use]
93pub fn reads_daemon_status(version: crate::SemVer) -> bool {
94    version >= DAEMON_STATUS_SINCE
95}

97crate::open_enum! {
98    /// What the daemon accepts now.
99    #[derive(Clone, Copy, Debug, Default, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
100    #[serde(rename_all = "snake_case")]
101    pub enum DaemonPhase {
102        /// Starting: only `handshake` and `daemon.status` are answered; any
103        /// other request is refused with `daemon_starting`.
104        Starting,
105        /// Everything works, whatever tasks are running.
106        #[default]
107        Ready,
108        /// Going down, for a stop or a restart that is due: new runs are
109        /// refused, and connections close soon.
110        Stopping,
111        /// Startup failed, as `error` says; the daemon exits.
112        Failed,
113    }
114}
115
116/// The daemon's phase and what it is busy with: the `daemon.status` result,
117/// the `daemon.status` notification, and [`crate::ServerHello::status`].
118#[derive(Clone, Debug, Default, Deserialize, JsonSchema, PartialEq, Serialize)]
119pub struct DaemonStatus {
120    pub phase: DaemonPhase,
121    /// In phase `stopping`: the daemon relaunches.
122    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
123    pub restart: bool,
124    /// What the daemon is doing that a user would want to see, oldest
125    /// first. Free-form: clients show any task the same way.
126    #[serde(default, skip_serializing_if = "Vec::is_empty")]
127    pub tasks: Vec<DaemonTask>,
128    /// In phase `failed`: why, as a registry error.
129    #[serde(default, skip_serializing_if = "Option::is_none")]
130    pub error: Option<crate::JsonRpcError>,
131}
132
133impl DaemonStatus {
134    /// The status of a daemon that is ready and busy with nothing.
135    #[must_use]
136    pub fn ready() -> Self {
137        Self::default()
138    }
139
140    /// The tasks the daemon is not ready until they end.
141    pub fn blocking(&self) -> impl Iterator<Item = &DaemonTask> {
142        self.tasks.iter().filter(|task| task.blocking)
143    }
144}

146/// One thing the daemon is doing.
147#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
148pub struct DaemonTask {
149    /// Stable for the task's life; opaque.
150    pub id: String,
151    /// What it is, for people: "Moving sessions to the new storage layout".
152    pub label: String,
153    /// What it is on now, for people.
154    #[serde(default, skip_serializing_if = "Option::is_none")]
155    pub detail: Option<String>,
156    #[serde(default, skip_serializing_if = "Option::is_none")]
157    pub progress: Option<TaskProgress>,
158    /// The daemon is not ready until this task ends.
159    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
160    pub blocking: bool,
161    pub started_at: jiff::Timestamp,
162}

164/// How far a task is.
165#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
166pub struct TaskProgress {
167    pub done: u64,
168    /// Absent while the task does not know how much there is.
169    #[serde(default, skip_serializing_if = "Option::is_none")]
170    pub total: Option<u64>,
171    /// What `done` and `total` count, for people: "sessions".
172    #[serde(default, skip_serializing_if = "Option::is_none")]
173    pub unit: Option<String>,
174}
