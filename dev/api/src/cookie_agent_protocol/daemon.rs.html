1//! Restarting the daemon from a client (`docs/site/specs/engine-configuration.md`
2//! section 11). Client protocol 1.19.0.
3//!
4//! A client asks with `daemon.restart`; the daemon tells every client with
5//! `daemon.restarting` while the restart waits and then closes their
6//! connections with WebSocket close code [`RESTART_CLOSE_CODE`], and they
7//! reconnect as after any dropped connection.
8
9use schemars::JsonSchema;
10use serde::{Deserialize, Serialize};
11
12pub const DAEMON_RESTART_METHOD: &str = "daemon.restart";
13pub const DAEMON_RESTARTING_METHOD: &str = "daemon.restarting";
14
15/// The WebSocket close code a restarting daemon closes client connections
16/// with: 1012, Service Restart.
17pub const RESTART_CLOSE_CODE: u16 = 1012;
18/// The close reason that goes with [`RESTART_CLOSE_CODE`].
19pub const RESTART_CLOSE_REASON: &str = "restarting";
20
21crate::open_enum! {
22    /// When the daemon restarts.
23    #[derive(Clone, Copy, Debug, Default, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
24    #[serde(rename_all = "snake_case")]
25    pub enum RestartWhen {
26        /// Once no run is active in any session. New runs are refused
27        /// meanwhile.
28        #[default]
29        Idle,
30        /// At once: running turns end as interrupted.
31        Now,
32    }
33}
34
35/// Restart the daemon: it relaunches its own executable with the same
36/// arguments and environment. A second call while a restart is scheduled
37/// changes its `when`; `now` wins over `idle`.
38#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
39#[serde(deny_unknown_fields)]
40pub struct DaemonRestartParams {
41    #[serde(default)]
42    pub when: RestartWhen,
43}
44
45#[derive(Clone, Copy, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
46pub struct DaemonRestartResult {
47    /// Whether this call scheduled the restart; `false` when one was already
48    /// scheduled and this call only changed its `when`.
49    pub scheduled: bool,
50    /// The runs active when the daemon answered: what `idle` waits for and
51    /// what `now` interrupts.
52    pub active_runs: u32,
53    /// The background calls still running when the daemon answered, which
54    /// `idle` waits for too and `now` ends (client protocol 1.35.0; absent
55    /// from older daemons, which did not wait for them).
56    #[serde(default, skip_serializing_if = "is_zero")]
57    pub background_calls: u32,
58}
59
60/// A restart is scheduled, sent to every client when it is and again as the
61/// wait progresses.
62#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
63pub struct DaemonRestartingNotification {
64    pub when: RestartWhen,
65    /// The runs still active.
66    pub active_runs: u32,
67    /// The background calls still running (client protocol 1.35.0).
68    #[serde(default, skip_serializing_if = "is_zero")]
69    pub background_calls: u32,
70    /// What the restart waits for or is doing, for people to read.
71    #[serde(default, skip_serializing_if = "Option::is_none")]
72    pub reason: Option<String>,
73}
74
75#[allow(clippy::trivially_copy_pass_by_ref)]
76const fn is_zero(count: &u32) -> bool {
77    *count == 0
