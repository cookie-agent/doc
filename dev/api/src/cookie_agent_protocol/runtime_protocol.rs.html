1//! Wire types of the runtime protocol between the daemon and a runtime, as
2//! specified in `docs/site/specs/runtimes.md` section 9 and documented in
3//! `docs/site/reference/runtime-protocol.md`.
4//!
5//! The protocol is JSON-RPC 2.0, one message per frame: a line over stdio or
6//! a frame of an in-memory transport. Every message is bounded by
7//! [`MAX_FRAME_BYTES`]. Both ends send requests: the daemon drives tools, and
8//! a runtime reports progress back while a call runs. [`crate::peer`] carries
9//! the messages.
10//!
11//! Every message is information except the requests and notifications a
12//! runtime or plugin sends that ask the daemon to act (`tool/progress`,
13//! `tool/detachable`, and `producer/*`): information ignores members it does
14//! not know and reads an open enum's unknown value as its documented
15//! fallback (`docs/site/specs/compatibility-v1.md` section 4); the daemon
16//! decodes commands with [`crate::strict::decode_command`], which refuses
17//! what their schema does not list. Command types also deny unknown fields
18//! themselves.
19

29/// The largest frame either end sends or accepts. It fits an `execute`
30/// reply carrying [`MAX_INLINE_MEDIA_BYTES`] of base64 media next to a full
31/// tool result.
32pub const MAX_FRAME_BYTES: usize = 48 * 1024 * 1024;
33
34/// The most media bytes, before base64, one `tool/execute` reply carries:
35/// the largest single attachment a tool may return.
36pub const MAX_INLINE_MEDIA_BYTES: u64 = 25 * 1024 * 1024;
37
38/// The most calls one `tool/prepare` request carries.
39pub const MAX_PREPARE_BATCH: usize = 256;

41/// How the daemon refuses a runtime that dials in on
42/// `/runtime?id=<runtime ID>`, since protocol 1.2.0. A refusal before the
43/// WebSocket upgrade is an HTTP status whose body is the reason in plain
44/// text; one after it is a close frame whose reason is the reason. A
45/// runtime stops dialing in on a permanent refusal and dials in again,
46/// after a wait, on any other.

50pub mod dial_in {
51    /// HTTP 400: the runtime ID is malformed. Permanent.
52    pub const STATUS_INVALID_ID: u16 = 400;
53    /// HTTP 401: the bearer token is missing or wrong. Permanent.
54    pub const STATUS_UNAUTHORIZED: u16 = 401;
55    /// HTTP 403: the runtime ID is reserved (`global`, `local`) or
56    /// configured on the daemon, or the request carries an `Origin` header.
57    /// Permanent.
58    pub const STATUS_FORBIDDEN: u16 = 403;
59    /// HTTP 409: a runtime is connected, or being accepted, under the ID.
60    /// Transient: the daemon may not have noticed yet that an earlier
61    /// connection of the same runtime ended.
62    pub const STATUS_CONNECTED: u16 = 409;
63    /// Close code 1000, normal closure: the end that closes is done with
64    /// the connection, such as a runtime that stops.
65    pub const CLOSE_NORMAL: u16 = 1000;
66    /// Close code 1011: the runtime did not complete `runtime/initialize`,
67    /// such as one of an incompatible version or one that did not answer in
68    /// time. Transient.
69    pub const CLOSE_INITIALIZE_FAILED: u16 = 1011;
70
71    /// Whether a refusal with HTTP `status` is permanent.
72    #[must_use]
73    pub const fn is_permanent_status(status: u16) -> bool {
74        matches!(
75            status,
76            STATUS_INVALID_ID | STATUS_UNAUTHORIZED | STATUS_FORBIDDEN
77        )
78    }
79}

652/// The machine a runtime runs on, as `runtime/initialize` reports it.
653#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
654pub struct RuntimeMachine {
655    /// A stable, opaque ID of the machine, such as
656    /// [`crate::paths::machine_id`]: 1 to [`RuntimeMachine::MAX_ID_BYTES`]
657    /// bytes of printable ASCII. Runtimes that report one ID run on one
658    /// machine.
659    pub id: String,
660    /// The machine's host name, which names a machine the daemon has not
661    /// seen before; at most [`RuntimeMachine::MAX_HOST_NAME_BYTES`] bytes.
662    #[serde(default, skip_serializing_if = "Option::is_none")]
663    pub host_name: Option<String>,
664}
665
666impl RuntimeMachine {
667    /// The longest machine ID.
668    pub const MAX_ID_BYTES: usize = 128;
669    /// The longest host name.
670    pub const MAX_HOST_NAME_BYTES: usize = 255;
671
672    /// This machine: [`crate::paths::machine_id`] and
673    /// [`crate::paths::host_name`]. `None` when it has no machine ID.
674    #[cfg(not(target_arch = "wasm32"))]
675    #[must_use]
676    pub fn this_machine() -> Option<Self> {
677        let host_name = crate::paths::host_name();
678        Some(Self {
679            id: crate::paths::machine_id()?,
680            host_name: (!host_name.is_empty()).then_some(host_name),
681        })
682        .filter(Self::is_valid)
683    }
684
685    /// Whether the ID and host name keep their bounds: a reader treats a
686    /// machine that does not as unreported.
687    #[must_use]
688    pub fn is_valid(&self) -> bool {
689        (1..=Self::MAX_ID_BYTES).contains(&self.id.len())
690            && self.id.bytes().all(|byte| byte.is_ascii_graphic())
691            && self.host_name.as_ref().is_none_or(|host| {
692                host.len() <= Self::MAX_HOST_NAME_BYTES && !host.chars().any(char::is_control)
693            })
694    }
695}
696
697/// The longest home directory a runtime reports, in bytes.
698pub const MAX_HOME_BYTES: usize = 4096;
699
700/// Whether `home` is a home directory a reader takes: an absolute path in
701/// POSIX (`/home/user`) or Windows (`C:\Users\user`, `C:/Users/user`,
702/// `\\server\share\user`) form, of 1 to [`MAX_HOME_BYTES`] bytes and
703/// without control characters. Only the spelling is checked: the path is on
704/// the runtime's machine, which need not be the reader's.
705#[must_use]
706pub fn is_valid_home(home: &str) -> bool {
707    let bytes = home.as_bytes();
708    let absolute = bytes.first() == Some(&b'/')
709        || home.starts_with(r"\\")
710        || (bytes.len() >= 3
711            && bytes[0].is_ascii_alphabetic()
712            && bytes[1] == b':'
713            && matches!(bytes[2], b'\\' | b'/'));
714    absolute && home.len() <= MAX_HOME_BYTES && !home.chars().any(char::is_control)
715}

1124/// diagnostic, before it checks the rest of the command
1125/// (`docs/site/specs/compatibility-v1.md` section 12.4).
1126pub fn take_unknown_progress_blocks(params: &mut Value) -> Vec<String> {
1127    let Some(display) = params
1128        .pointer_mut("/progress/display")
1129        .filter(|display| display.get("type").and_then(Value::as_str) == Some("replace"))
1130    else {
1131        return Vec::new();
1132    };
1133    let Some(blocks) = display.get_mut("blocks").and_then(Value::as_array_mut) else {
1134        return Vec::new();
1135    };
1136    let mut unknown = Vec::new();
1137    blocks.retain(
1138        |block| match serde_json::from_value::<Open<DisplayBlock>>(block.clone()) {
1139            Ok(Open::Unknown { tag, .. }) => {
1140                unknown.push(tag);
1141                false
1142            }
1143            _ => true,
1144        },
1145    );
1146    unknown
1147}

1195/// Lowercase hex of `bytes`.
1196#[must_use]
1197pub fn hex_encode(bytes: &[u8]) -> String {
1198    const DIGITS: &[u8; 16] = b"0123456789abcdef";
1199    let mut text = String::with_capacity(bytes.len() * 2);
1200    for byte in bytes {
1201        text.push(char::from(DIGITS[usize::from(byte >> 4)]));
1202        text.push(char::from(DIGITS[usize::from(byte & 0x0f)]));
1203    }
1204    text
1205}
1206
1207/// The bytes of lowercase hex `text`.
1208#[must_use]
1209pub fn hex_decode(text: &str) -> Option<Vec<u8>> {
1210    fn digit(byte: u8) -> Option<u8> {
1211        match byte {
1212            b'0'..=b'9' => Some(byte - b'0'),
1213            b'a'..=b'f' => Some(byte - b'a' + 10),
1214            _ => None,
1215        }
1216    }
1217    let mut bytes = text.bytes();
1218    let mut decoded = Vec::with_capacity(text.len() / 2);
1219    while let Some(high) = bytes.next() {
1220        let low = bytes.next()?;
1221        decoded.push(digit(high)? << 4 | digit(low)?);
1222    }
1223    Some(decoded)
1224}
