1//! Wire types of the runtime protocol between the daemon and a runtime, as
2//! specified in `docs/site/specs/runtimes.md` section 9 and documented in
3//! `docs/site/reference/runtime-protocol.md`.
4//!
5//! The protocol is JSON-RPC 2.0, one message per frame: a line over stdio or
6//! a frame of an in-memory transport. Every message is bounded by
7//! [`MAX_FRAME_BYTES`]. Both ends send requests: the daemon drives tools, and
8//! a runtime reports progress back while a call runs. [`crate::peer`] carries
9//! the messages.
10//!
11//! Every message is information except the requests and notifications a
12//! runtime or plugin sends that ask the daemon to act (`tool/progress`,
13//! `tool/detachable`, and `producer/*`): information ignores members it does
14//! not know and reads an open enum's unknown value as its documented
15//! fallback (`docs/site/specs/compatibility-v1.md` section 4); the daemon
16//! decodes commands with [`crate::strict::decode_command`], which refuses
17//! what their schema does not list. Command types also deny unknown fields
18//! themselves.
19

29/// The largest frame either end sends or accepts. It fits an `execute`
30/// reply carrying [`MAX_INLINE_MEDIA_BYTES`] of base64 media next to a full
31/// tool result.
32pub const MAX_FRAME_BYTES: usize = 48 * 1024 * 1024;
33
34/// The most media bytes, before base64, one `tool/execute` reply carries:
35/// the largest single attachment a tool may return.
36pub const MAX_INLINE_MEDIA_BYTES: u64 = 25 * 1024 * 1024;
37
38/// The most calls one `tool/prepare` request carries.
39pub const MAX_PREPARE_BATCH: usize = 256;

41/// How the daemon refuses a runtime that dials in on
42/// `/runtime?id=<runtime ID>`, since protocol 1.2.0. A refusal before the
43/// WebSocket upgrade is an HTTP status whose body is the reason in plain
44/// text; one after it is a close frame whose reason is the reason. A
45/// runtime stops dialing in on a permanent refusal and dials in again,
46/// after a wait, on any other.
47pub mod dial_in {
48    /// HTTP 400: the runtime ID is malformed. Permanent.
49    pub const STATUS_INVALID_ID: u16 = 400;
50    /// HTTP 401: the bearer token is missing or wrong. Permanent.
51    pub const STATUS_UNAUTHORIZED: u16 = 401;
52    /// HTTP 403: the runtime ID is reserved (`global`, `local`) or
53    /// configured on the daemon, or the request carries an `Origin` header.
54    /// Permanent.
55    pub const STATUS_FORBIDDEN: u16 = 403;
56    /// HTTP 409: a runtime is connected, or being accepted, under the ID.
57    /// Transient: the daemon may not have noticed yet that an earlier
58    /// connection of the same runtime ended.
59    pub const STATUS_CONNECTED: u16 = 409;
60    /// Close code 1000, normal closure: the end that closes is done with
61    /// the connection, such as a runtime that stops.
62    pub const CLOSE_NORMAL: u16 = 1000;
63    /// Close code 1011: the runtime did not complete `runtime/initialize`,
64    /// such as one of an incompatible version or one that did not answer in
65    /// time. Transient.
66    pub const CLOSE_INITIALIZE_FAILED: u16 = 1011;
67
68    /// Whether a refusal with HTTP `status` is permanent.
69    #[must_use]
70    pub const fn is_permanent_status(status: u16) -> bool {
71        matches!(
72            status,
73            STATUS_INVALID_ID | STATUS_UNAUTHORIZED | STATUS_FORBIDDEN
74        )
75    }
76}

358/// The machine a runtime runs on, as `runtime/initialize` reports it.
359#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
360pub struct RuntimeMachine {
361    /// A stable, opaque ID of the machine, such as
362    /// [`crate::paths::machine_id`]: 1 to [`RuntimeMachine::MAX_ID_BYTES`]
363    /// bytes of printable ASCII. Runtimes that report one ID run on one
364    /// machine.
365    pub id: String,
366    /// The machine's host name, which names a machine the daemon has not
367    /// seen before; at most [`RuntimeMachine::MAX_HOST_NAME_BYTES`] bytes.
368    #[serde(default, skip_serializing_if = "Option::is_none")]
369    pub host_name: Option<String>,
370}
371
372impl RuntimeMachine {
373    /// The longest machine ID.
374    pub const MAX_ID_BYTES: usize = 128;
375    /// The longest host name.
376    pub const MAX_HOST_NAME_BYTES: usize = 255;
377
378    /// This machine: [`crate::paths::machine_id`] and
379    /// [`crate::paths::host_name`]. `None` when it has no machine ID.
380    #[cfg(not(target_arch = "wasm32"))]
381    #[must_use]
382    pub fn this_machine() -> Option<Self> {
383        let host_name = crate::paths::host_name();
384        Some(Self {
385            id: crate::paths::machine_id()?,
386            host_name: (!host_name.is_empty()).then_some(host_name),
387        })
388        .filter(Self::is_valid)
389    }
390
391    /// Whether the ID and host name keep their bounds: a reader treats a
392    /// machine that does not as unreported.
393    #[must_use]
394    pub fn is_valid(&self) -> bool {
395        (1..=Self::MAX_ID_BYTES).contains(&self.id.len())
396            && self.id.bytes().all(|byte| byte.is_ascii_graphic())
397            && self.host_name.as_ref().is_none_or(|host| {
398                host.len() <= Self::MAX_HOST_NAME_BYTES && !host.chars().any(char::is_control)
399            })
400    }
401}
402
403/// The longest home directory a runtime reports, in bytes.
404pub const MAX_HOME_BYTES: usize = 4096;
405
406/// Whether `home` is a home directory a reader takes: an absolute path in
407/// POSIX (`/home/user`) or Windows (`C:\Users\user`, `C:/Users/user`,
408/// `\\server\share\user`) form, of 1 to [`MAX_HOME_BYTES`] bytes and
409/// without control characters. Only the spelling is checked: the path is on
410/// the runtime's machine, which need not be the reader's.
411#[must_use]
412pub fn is_valid_home(home: &str) -> bool {
413    let bytes = home.as_bytes();
414    let absolute = bytes.first() == Some(&b'/')
415        || home.starts_with(r"\\")
416        || (bytes.len() >= 3
417            && bytes[0].is_ascii_alphabetic()
418            && bytes[1] == b':'
419            && matches!(bytes[2], b'\\' | b'/'));
420    absolute && home.len() <= MAX_HOME_BYTES && !home.chars().any(char::is_control)
421}

826/// Removes from raw `tool/progress` params the display blocks of a `replace`
827/// whose kind this build does not know, and returns their tags. `tool/progress`
828/// is a command, and a command refuses what its schema does not list, but a
829/// display block is only shown: the daemon drops such a block, with a
830/// diagnostic, before it checks the rest of the command
831/// (`docs/site/specs/compatibility-v1.md` section 12.4).
832pub fn take_unknown_progress_blocks(params: &mut Value) -> Vec<String> {
833    let Some(display) = params
834        .pointer_mut("/progress/display")
835        .filter(|display| display.get("type").and_then(Value::as_str) == Some("replace"))
836    else {
837        return Vec::new();
838    };
839    let Some(blocks) = display.get_mut("blocks").and_then(Value::as_array_mut) else {
840        return Vec::new();
841    };
842    let mut unknown = Vec::new();
843    blocks.retain(
844        |block| match serde_json::from_value::<Open<DisplayBlock>>(block.clone()) {
845            Ok(Open::Unknown { tag, .. }) => {
846                unknown.push(tag);
847                false
848            }
849            _ => true,
850        },
851    );
852    unknown
853}

901/// Lowercase hex of `bytes`.
902#[must_use]
903pub fn hex_encode(bytes: &[u8]) -> String {
904    const DIGITS: &[u8; 16] = b"0123456789abcdef";
905    let mut text = String::with_capacity(bytes.len() * 2);
906    for byte in bytes {
907        text.push(char::from(DIGITS[usize::from(byte >> 4)]));
908        text.push(char::from(DIGITS[usize::from(byte & 0x0f)]));
909    }
910    text
911}
912
913/// The bytes of lowercase hex `text`.
914#[must_use]
915pub fn hex_decode(text: &str) -> Option<Vec<u8>> {
916    fn digit(byte: u8) -> Option<u8> {
917        match byte {
918            b'0'..=b'9' => Some(byte - b'0'),
919            b'a'..=b'f' => Some(byte - b'a' + 10),
920            _ => None,
921        }
922    }
923    let mut bytes = text.bytes();
924    let mut decoded = Vec::with_capacity(text.len() / 2);
925    while let Some(high) = bytes.next() {
926        let low = bytes.next()?;
927        decoded.push(digit(high)? << 4 | digit(low)?);
928    }
929    Some(decoded)
930}
