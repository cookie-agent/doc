1//! Wire types of the runtime protocol between the daemon and a runtime, as
2//! specified in `docs/site/specs/runtimes.md` section 9 and documented in
3//! `docs/site/reference/runtime-protocol.md`.
4//!
5//! The protocol is JSON-RPC 2.0, one message per frame: a line over stdio or
6//! a frame of an in-memory transport. Every message is bounded by
7//! [`MAX_FRAME_BYTES`] and every type rejects unknown fields. Both ends send
8//! requests: the daemon drives tools, and a runtime reports progress and
9//! background jobs back while a call runs. [`crate::peer`] carries the
10//! messages.
11
12use crate::{
13    AdaptorId, AgentId, AgentMdEntry, AgentMdSkipped, ModelCapabilities, ModelKey,
14    PermissionAction, PersistedToolResult, PreparedOperationIdentity, RunId, RuntimeKind,
15    SessionId, SkillSource, ToolCallId, ToolCallPresentation, ToolCompletionOutput,
16    ToolOutputChunk, ToolOutputDeclaration,
17};
18use serde::{Deserialize, Serialize};
19use serde_json::Value;

21/// The runtime protocol version. Both ends must speak exactly this version.
22pub const RUNTIME_PROTOCOL_VERSION: u32 = 5;
23
24/// The largest frame either end sends or accepts. It fits an `execute`
25/// reply carrying [`MAX_INLINE_MEDIA_BYTES`] of base64 media next to a full
26/// tool result.
27pub const MAX_FRAME_BYTES: usize = 48 * 1024 * 1024;
28
29/// The most media bytes, before base64, one `tool/execute` reply carries:
30/// the largest single attachment a tool may return.
31pub const MAX_INLINE_MEDIA_BYTES: u64 = 25 * 1024 * 1024;
32
33/// The most calls one `tool/prepare` request carries.
34pub const MAX_PREPARE_BATCH: usize = 256;

58    pub const TOOL_EXECUTE: &str = "tool/execute";
59    /// daemon → runtime (notification): discard a prepared call.
60    pub const TOOL_RELEASE: &str = "tool/release";
61    /// runtime → daemon: display and output of an executing call.
62    pub const TOOL_PROGRESS: &str = "tool/progress";
63    /// daemon → runtime (notification): cancel an executing call.

91/// JSON-RPC error codes the runtime protocol uses besides the standard ones.
92pub mod error_code {
93    /// The params do not match the method.
94    pub const INVALID_PARAMS: i32 = -32602;
95    /// No such method.
96    pub const METHOD_NOT_FOUND: i32 = -32601;
97    /// The versions of the two ends differ.
98    pub const VERSION_MISMATCH: i32 = -32010;
99    /// The runtime rejects the initialize request or settings.
100    pub const INITIALIZE_REJECTED: i32 = -32011;
101    /// A request other than initialize arrived before initialize.
102    pub const NOT_INITIALIZED: i32 = -32012;
103    /// The prepared call or job the request names is unknown.
104    pub const UNKNOWN_HANDLE: i32 = -32013;
105    /// Reading the runtime's context failed; `data.path` names the file.
106    pub const CONTEXT_FAILED: i32 = -32014;
107    /// The reply would exceed the frame bound.
108    pub const REPLY_TOO_LARGE: i32 = -32015;
109    /// The daemon rejected a producer operation: messaging is not open on
110    /// the connection, the session is not attached to the runtime, or the
111    /// operation itself failed.
112    pub const PRODUCER_REJECTED: i32 = -32016;
113}

626/// Lowercase hex of `bytes`.
627#[must_use]
628pub fn hex_encode(bytes: &[u8]) -> String {
629    const DIGITS: &[u8; 16] = b"0123456789abcdef";
630    let mut text = String::with_capacity(bytes.len() * 2);
631    for byte in bytes {
632        text.push(char::from(DIGITS[usize::from(byte >> 4)]));
633        text.push(char::from(DIGITS[usize::from(byte & 0x0f)]));
634    }
635    text
636}

638/// The bytes of lowercase hex `text`.
639#[must_use]
640pub fn hex_decode(text: &str) -> Option<Vec<u8>> {
641    fn digit(byte: u8) -> Option<u8> {
642        match byte {
643            b'0'..=b'9' => Some(byte - b'0'),
644            b'a'..=b'f' => Some(byte - b'a' + 10),
645            _ => None,
646        }
647    }
648    let mut bytes = text.bytes();
649    let mut decoded = Vec::with_capacity(text.len() / 2);
650    while let Some(high) = bytes.next() {
651        let low = bytes.next()?;
652        decoded.push(digit(high)? << 4 | digit(low)?);
653    }
654    Some(decoded)
655}
