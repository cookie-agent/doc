1//! A symmetric JSON-RPC 2.0 peer for the runtime protocol and the plugin
2//! protocol.
3//!
4//! Both ends of the runtime protocol send requests and notifications, so one
5//! [`Peer`] type serves the daemon's client and the runtime's server. It runs
6//! over any frame [`Transport`]: an in-memory pair, or [`LineTransport`] over
7//! a child's stdio. Every frame is bounded by
8//! [`MAX_FRAME_BYTES`] and parsed strictly;
9//! a malformed frame ends the connection.
10//!
11//! The transport's `recv` must be cancel-safe: the peer's I/O task selects
12//! between receiving and sending.
13

38/// Why a request failed.
39#[derive(Clone, Debug, thiserror::Error)]
40pub enum RpcError {
41    /// The connection ended; carries why.
42    #[error("connection closed: {0}")]
43    Closed(String),
44    /// The other end answered with an error.
45    #[error("{message}")]
46    Remote {
47        code: i32,
48        message: String,
49        data: Option<Value>,
50    },
51    /// A message could not be built or read.
52    #[error("invalid runtime protocol message: {0}")]
53    Protocol(String),
54}
55
56impl RpcError {
57    /// An error answer with `code` and `message`.
58    #[must_use]
59    pub fn remote(code: i32, message: impl Into<String>) -> Self {
60        Self::Remote {
61            code,
62            message: message.into(),
63            data: None,
64        }
65    }
66}

68type PendingReply = oneshot::Sender<Result<Box<RawValue>, RpcError>>;

70struct Pending {
71    replies: HashMap<u64, PendingReply>,
72    /// Set once the connection ended; no reply arrives after that.
73    closed: Option<String>,
74}

76struct Shared {
77    outbound: mpsc::Sender<String>,
78    pending: Mutex<Pending>,
79    next_id: AtomicU64,
80    /// Stops the I/O task.

86/// One end of a runtime protocol connection. Clones share it. The
87/// connection ends when the transport ends or [`Self::close`] is called;
88/// dropping clones does not end it.
89#[derive(Clone)]

123fn parse_params<T: DeserializeOwned>(params: Option<&RawValue>) -> Result<T, RpcError> {
124    let text = params.map_or("null", RawValue::get);
125    serde_json::from_str(text).map_err(|error| {
126        RpcError::remote(
127            error_code::INVALID_PARAMS,
128            format!("invalid params: {error}"),
129        )
130    })
131}

136        parse_params(self.params.as_deref())
137    }

146    /// Answers with `result`. A result too large for one frame is answered
147    /// with an error instead.
148    pub async fn respond<T: Serialize>(self, result: &T) -> Result<(), RpcError> {
149        let text = match encode(&ResponseRef {
150            jsonrpc: JsonRpcVersion::current(),
151            id: self.id,
152            result,
153        }) {
154            Ok(text) => text,
155            Err(error) => {
156                return self
157                    .fail(RpcError::remote(
158                        error_code::REPLY_TOO_LARGE,
159                        error.to_string(),
160                    ))
161                    .await;
162            }
163        };
164        send(&self.shared, text).await
165    }
166
167    /// Answers with an error. A non-[`RpcError::Remote`] error is sent as an
168    /// internal error.
169    pub async fn fail(self, error: RpcError) -> Result<(), RpcError> {
170        let (code, message, data) = match error {
171            RpcError::Remote {
172                code,
173                message,
174                data,
175            } => (code, message, data),
176            other => (-32603, other.to_string(), None),
177        };
178        let text = encode(&ErrorResponseRef {
179            jsonrpc: JsonRpcVersion::current(),
180            id: self.id,
181            error: WireRpcError {
182                code,
183                message,
184                data,
185            },
186        })?;
187        send(&self.shared, text).await
188    }
189}

229/// Any message, before it is classified. Exactly one shape is valid: a
230/// request (`id`, `method`), a notification (`method`), or a reply (`id`
231/// with `result` or `error`).
232#[derive(Deserialize)]
233#[serde(deny_unknown_fields)]
234struct Envelope {
235    #[allow(dead_code)]
236    jsonrpc: JsonRpcVersion,
237    #[serde(default)]
238    id: Option<u64>,
239    #[serde(default)]
240    method: Option<String>,
241    #[serde(default)]
242    params: Option<Box<RawValue>>,
243    #[serde(default)]
244    result: Option<Box<RawValue>>,
245    #[serde(default)]
246    error: Option<WireRpcError>,
247}
248
249fn encode(message: &impl Serialize) -> Result<String, RpcError> {
250    let text = serde_json::to_string(message)
251        .map_err(|error| RpcError::Protocol(format!("message does not serialize: {error}")))?;
252    if text.len() > MAX_FRAME_BYTES {
253        return Err(RpcError::Protocol(format!(
254            "message of {} bytes exceeds the {MAX_FRAME_BYTES}-byte frame bound",
255            text.len()
256        )));
257    }
258    Ok(text)
259}

261async fn send(shared: &Shared, text: String) -> Result<(), RpcError> {
262    shared
263        .outbound
264        .send(text)
265        .await
266        .map_err(|_| RpcError::Closed(closed_reason(shared)))
267}
268
269fn closed_reason(shared: &Shared) -> String {
270    shared
271        .pending
272        .lock()
273        .unwrap_or_else(|poisoned| poisoned.into_inner())
274        .closed
275        .clone()
276        .unwrap_or_else(|| "connection closed".to_owned())
277}

279impl Peer {
280    /// Starts a peer on `transport`. Returns the peer and the stream of
281    /// requests and notifications the other end sends, which ends with the
282    /// connection.
283    #[must_use]
284    pub fn start(transport: Box<dyn Transport>) -> (Self, mpsc::Receiver<Incoming>) {
285        let (outbound, outbound_rx) = mpsc::channel(64);
286        let (incoming, incoming_rx) = mpsc::channel(64);
287        let shared = Arc::new(Shared {
288            outbound,
289            pending: Mutex::new(Pending {
290                replies: HashMap::new(),
291                closed: None,
292            }),
293            next_id: AtomicU64::new(1),
294            stop: CancellationToken::new(),
295            ended: CancellationToken::new(),
296        });
297        tokio::spawn(run_io(
298            Arc::clone(&shared),
299            transport,
300            outbound_rx,
301            incoming,
302        ));
303        (Self { shared }, incoming_rx)
304    }
305
306    /// Sends a request and waits for its reply.
307    pub async fn request<P, R>(&self, method: &str, params: &P) -> Result<R, RpcError>
308    where
309        P: Serialize + ?Sized,
310        R: DeserializeOwned,
311    {
312        let id = self.shared.next_id.fetch_add(1, Ordering::Relaxed);
313        let text = encode(&RequestRef {
314            jsonrpc: JsonRpcVersion::current(),
315            id,
316            method,
317            params,
318        })?;
319        let (sender, receiver) = oneshot::channel();
320        {
321            let mut pending = self
322                .shared
323                .pending
324                .lock()
325                .unwrap_or_else(|poisoned| poisoned.into_inner());
326            if let Some(reason) = &pending.closed {
327                return Err(RpcError::Closed(reason.clone()));
328            }
329            pending.replies.insert(id, sender);
330        }
331        let guard = PendingGuard {
332            shared: &self.shared,
333            id,
334        };
335        send(&self.shared, text).await?;
336        let reply = receiver
337            .await
338            .map_err(|_| RpcError::Closed(closed_reason(&self.shared)))??;
339        drop(guard);
340        serde_json::from_str(reply.get())
341            .map_err(|error| RpcError::Protocol(format!("invalid reply to `{method}`: {error}")))
342    }
343
344    /// Sends a notification.
345    pub async fn notify<P: Serialize + ?Sized>(
346        &self,
347        method: &str,
348        params: &P,
349    ) -> Result<(), RpcError> {
350        let text = encode(&NotificationRef {
351            jsonrpc: JsonRpcVersion::current(),
352            method,
353            params,
354        })?;
355        send(&self.shared, text).await
356    }
357
358    /// Sends a notification without waiting, for use from `Drop`. It is
359    /// lost if the outbound queue is full and no runtime can take it.
360    pub fn notify_detached<P: Serialize + ?Sized>(&self, method: &str, params: &P) {
361        let Ok(text) = encode(&NotificationRef {
362            jsonrpc: JsonRpcVersion::current(),
363            method,
364            params,
365        }) else {
366            return;
367        };
368        match self.shared.outbound.try_send(text) {
369            Ok(()) | Err(mpsc::error::TrySendError::Closed(_)) => {}
370            Err(mpsc::error::TrySendError::Full(text)) => {
371                if let Ok(handle) = tokio::runtime::Handle::try_current() {
372                    let shared = Arc::clone(&self.shared);
373                    handle.spawn(async move {
374                        let _ = send(&shared, text).await;
375                    });
376                }
377            }
378        }
379    }
380
381    /// Ends the connection.
382    pub fn close(&self) {
383        self.shared.stop.cancel();
384    }
385
386    /// Whether the connection has ended.
387    #[must_use]
388    pub fn is_closed(&self) -> bool {
389        self.shared.ended.is_cancelled()
390    }
391
392    /// Waits for the connection to end and returns why it did.
393    pub async fn closed(&self) -> String {
394        self.shared.ended.cancelled().await;
395        closed_reason(&self.shared)
396    }
397}

399/// Forgets a pending reply when its request future is dropped.
400struct PendingGuard<'a> {
401    shared: &'a Shared,
402    id: u64,
403}
404
405impl Drop for PendingGuard<'_> {
406    fn drop(&mut self) {
407        self.shared
408            .pending
409            .lock()
410            .unwrap_or_else(|poisoned| poisoned.into_inner())
411            .replies
412            .remove(&self.id);
413    }
414}

416async fn run_io(
417    shared: Arc<Shared>,
418    mut transport: Box<dyn Transport>,
419    mut outbound: mpsc::Receiver<String>,
420    incoming: mpsc::Sender<Incoming>,
421) {
422    let reason = loop {
423        tokio::select! {
424            () = shared.stop.cancelled() => {
425                // Messages queued before the close, such as the reply to
426                // `runtime/shutdown`, still go out.
427                while let Ok(text) = outbound.try_recv() {
428                    if transport.send(MessageFrame::Text(text)).await.is_err() {
429                        break;
430                    }
431                }
432                break "connection closed locally".to_owned();
433            }
434            text = outbound.recv() => {
435                let Some(text) = text else {
436                    break "connection closed locally".to_owned();
437                };
438                if let Err(error) = transport.send(MessageFrame::Text(text)).await {
439                    break transport_reason(&error);
440                }
441            }
442            frame = transport.recv() => match frame {
443                Ok(Some(frame)) => {
444                    if let Err(reason) = dispatch(&shared, frame, &incoming).await {
445                        break reason;
446                    }
447                }
448                Ok(None) => break "the other end closed the connection".to_owned(),
449                Err(error) => break transport_reason(&error),
450            },
451        }
452    };
453    let replies = {
454        let mut pending = shared
455            .pending
456            .lock()
457            .unwrap_or_else(|poisoned| poisoned.into_inner());
458        pending.closed = Some(reason.clone());
459        std::mem::take(&mut pending.replies)
460    };
461    for (_, reply) in replies {
462        let _ = reply.send(Err(RpcError::Closed(reason.clone())));
463    }
464    shared.ended.cancel();
465}
466
467fn transport_reason(error: &TransportError) -> String {
468    match error {
469        TransportError::Closed => "the other end closed the connection".to_owned(),
470        other => other.to_string(),
471    }
472}

474/// Routes one frame. An error ends the connection.
475async fn dispatch(
476    shared: &Arc<Shared>,
477    frame: MessageFrame,
478    incoming: &mpsc::Sender<Incoming>,
479) -> Result<(), String> {
480    let envelope: Envelope = match frame {
481        MessageFrame::Text(text) => parse_frame(text.as_bytes())?,
482        MessageFrame::Bytes(bytes) => parse_frame(&bytes)?,
483        MessageFrame::Value(value) => {
484            let text =
485                serde_json::to_string(&value).map_err(|error| format!("invalid frame: {error}"))?;
486            parse_frame(text.as_bytes())?
487        }
488    };
489    match envelope {
490        Envelope {
491            id: Some(id),
492            method: Some(method),
493            params,
494            result: None,
495            error: None,
496            ..
497        } => {
498            let request = IncomingRequest {
499                method,
500                params,
501                id,
502                shared: Arc::clone(shared),
503            };
504            if let Err(mpsc::error::SendError(Incoming::Request(request))) =
505                incoming.send(Incoming::Request(request)).await
506            {
507                let _ = request
508                    .fail(RpcError::remote(-32603, "the receiver is shutting down"))
509                    .await;
510            }
511            Ok(())
512        }
513        Envelope {
514            id: None,
515            method: Some(method),
516            params,
517            result: None,
518            error: None,
519            ..
520        } => {
521            let _ = incoming
522                .send(Incoming::Notification(IncomingNotification {
523                    method,
524                    params,
525                }))
526                .await;
527            Ok(())
528        }
529        Envelope {
530            id: Some(id),
531            method: None,
532            params: None,
533            result,
534            error,
535            ..
536        } => {
537            let reply = match (result, error) {
538                (Some(result), None) => Ok(result),
539                (None, Some(error)) => Err(RpcError::Remote {
540                    code: error.code,
541                    message: error.message,
542                    data: error.data,
543                }),
544                _ => return Err("a reply needs exactly one of result and error".to_owned()),
545            };
546            let sender = shared
547                .pending
548                .lock()
549                .unwrap_or_else(|poisoned| poisoned.into_inner())
550                .replies
551                .remove(&id);
552            // A reply to a request whose caller gave up is dropped.
553            if let Some(sender) = sender {
554                let _ = sender.send(reply);
555            }
556            Ok(())
557        }
558        _ => Err("message is neither a request, a notification, nor a reply".to_owned()),
559    }
560}

562fn parse_frame(bytes: &[u8]) -> Result<Envelope, String> {
563    if bytes.len() > MAX_FRAME_BYTES {
564        return Err(format!(
565            "frame of {} bytes exceeds the {MAX_FRAME_BYTES}-byte bound",
566            bytes.len()
567        ));
568    }
569    serde_json::from_slice(bytes).map_err(|error| format!("invalid frame: {error}"))
570}

572/// Newline-delimited frames over a byte stream, such as a child's stdio:
573/// one JSON message per line, each at most
574/// [`MAX_FRAME_BYTES`].
575pub struct LineTransport<R, W> {
576    reader: FramedRead<R, LinesCodec>,
577    writer: W,
578}
579
580impl<R: AsyncRead + Unpin, W: AsyncWrite + Unpin> LineTransport<R, W> {
581    #[must_use]
582    pub fn new(reader: R, writer: W) -> Self {
583        Self {
584            reader: FramedRead::new(reader, LinesCodec::new_with_max_length(MAX_FRAME_BYTES)),
585            writer,
586        }
587    }
588}
589
590#[async_trait]
591impl<R, W> Transport for LineTransport<R, W>
592where
593    R: AsyncRead + Unpin + Send,
594    W: AsyncWrite + Unpin + Send,
595{
596    async fn send(&mut self, frame: MessageFrame) -> Result<(), TransportError> {
597        let mut text = frame.into_text()?;
598        if text.contains('\n') {
599            return Err(TransportError::Other("frame contains a newline".into()));
600        }
601        text.push('\n');
602        self.writer
603            .write_all(text.as_bytes())
604            .await
605            .map_err(|error| TransportError::Other(error.to_string()))?;
606        self.writer
607            .flush()
608            .await
609            .map_err(|error| TransportError::Other(error.to_string()))
610    }
611
612    async fn recv(&mut self) -> Result<Option<MessageFrame>, TransportError> {
613        match self.reader.next().await {
614            Some(Ok(line)) => Ok(Some(MessageFrame::Text(line))),
615            Some(Err(LinesCodecError::MaxLineLengthExceeded)) => Err(TransportError::Other(
616                format!("frame exceeds the {MAX_FRAME_BYTES}-byte bound"),
617            )),
618            Some(Err(LinesCodecError::Io(error))) => Err(TransportError::Other(error.to_string())),
619            None => Ok(None),
620        }
621    }
