1//! Session-local model continuation derived from existing durable events.
2use std::borrow::Borrow;
3
4use crate::{EventPayload, ModelFinishReason, ModelSelection, RunId, RunSelection, StoredEvent};
5
6#[derive(Clone, Debug, Default, PartialEq)]
7pub struct SessionModelState {
8    pub selection: Option<RunSelection>,
9    pub successful_fallback: bool,
10    run: Option<RunId>,
11    chain: Vec<ModelSelection>,
12}
13
14impl SessionModelState {
15    pub fn from_events<E: Borrow<StoredEvent>>(events: &[E]) -> Self {
16        let mut state = Self::default();
17        for event in events {
18            let event: &StoredEvent = event.borrow();
19            state.apply(event.run_id, &event.payload);
20        }
21        state
22    }
23
24    pub fn apply(&mut self, run: Option<RunId>, event: &EventPayload) {
25        match event {
26            EventPayload::SessionCreated {
27                creation_selection, ..
28            } => {
29                self.selection = Some(creation_selection.clone());
30            }
31            EventPayload::RunStarted {
32                selection,
33                selected_suffix,
34                ..
35            } => {
36                self.successful_fallback &= self.selection.as_ref() == Some(selection);
37                self.selection = Some(selection.clone());
38                self.chain = selected_suffix
39                    .iter()
40                    .map(|binding| binding.selection.clone())
41                    .collect();
42                self.run = run;
43            }
44            EventPayload::ModelTurnCommitted {
45                resolved_model,
46                turn,
47                ..
48            } if run.is_some() && run == self.run && successful_finish(&turn.finish_reason) => {
49                // Temporary skill overrides outside the frozen chain do not change
50                // the session default. ModelFallback alone is only an attempted move.
51                if self
52                    .chain
53                    .iter()
54                    .position(|model| *model == resolved_model.selection)
55                    .is_some_and(|index| index > 0)
56                    && let Some(selection) = &mut self.selection
57                {
58                    selection.model = resolved_model.selection.clone();
59                    self.successful_fallback = true;
60                }
61            }
62            _ => {}
63        }
64    }
65
66    /// Resolve a stale/default selection against a freshly frozen chain. Exact
67    /// model+variant identities, not an index from an earlier configuration, match.
68    pub fn continuation(
69        &self,
70        requested: &RunSelection,
71        chain: &[ModelSelection],
72        reset: bool,
73    ) -> RunSelection {
74        if !reset
75            && self.successful_fallback
76            && let Some(selected) = &self.selection
77            && selected.agent == requested.agent
78            && selected.preset == requested.preset
79            && let Some(requested_index) = chain.iter().position(|model| *model == requested.model)
80            && let Some(selected_index) = chain.iter().position(|model| *model == selected.model)
81            && requested_index < selected_index
82        {
83            return selected.clone();
84        }
85        requested.clone()
86    }
87}
88
89fn successful_finish(reason: &ModelFinishReason) -> bool {
90    match reason {
91        // A tool-calling or output-limited turn still establishes a working model.
92        ModelFinishReason::Stop | ModelFinishReason::ToolCalls | ModelFinishReason::Length => true,
93        ModelFinishReason::ContentFilter | ModelFinishReason::Cancelled | ModelFinishReason::Error
94        | ModelFinishReason::Aborted | ModelFinishReason::Timeout | ModelFinishReason::Refused
95        // Unknown and vendor-specific reasons are not positive success evidence.
96        | ModelFinishReason::Unknown | ModelFinishReason::Other(_) => false,
97    }
98}

140    #[test]
141    fn only_successful_finish_reasons_advance_fallback_and_later_failure_keeps_progress() {
142        let a = selection("a");
143        let b = selection("b");
144        let c = selection("c");
145        let run = Some(RunId::new_v7());
146        let initial = SessionModelState {
147            selection: Some(a.clone()),
148            run,
149            chain: vec![a.model.clone(), b.model.clone(), c.model.clone()],
150            successful_fallback: false,
151        };
152        for (reason, success) in [
153            (ModelFinishReason::Stop, true),
154            (ModelFinishReason::ToolCalls, true),
155            (ModelFinishReason::Length, true),
156            (ModelFinishReason::Cancelled, false),
157            (ModelFinishReason::Aborted, false),
158            (ModelFinishReason::Error, false),
159            (ModelFinishReason::Timeout, false),
160            (ModelFinishReason::ContentFilter, false),
161            (ModelFinishReason::Refused, false),
162            (ModelFinishReason::Unknown, false),
163            (ModelFinishReason::Other("vendor-specific".into()), false),
164        ] {
165            let mut state = initial.clone();
166            state.apply(run, &committed(&b, reason.clone()));
167            assert_eq!(
168                state.selection,
169                Some(if success { b.clone() } else { a.clone() }),
170                "{reason:?}"
171            );
172            assert_eq!(state.successful_fallback, success, "{reason:?}");
173            if !success {
174                state.apply(run, &committed(&b, ModelFinishReason::ToolCalls));
175                state.apply(run, &committed(&c, reason));
176                state.apply(
177                    run,
178                    &EventPayload::RunFailed {
179                        error: crate::SafeErrorMessage::new("later run failure").unwrap(),
180                        model_error: None,
181                        resolved_model: None,
182                    },
183                );
184                state.apply(run, &EventPayload::RunCancelled { reason: None });
185                assert_eq!(state.selection, Some(b.clone()));
186                assert!(state.successful_fallback);
187                assert_eq!(state.continuation(&a, &initial.chain, false), b);
188            }
189        }
190    }

192    #[test]
193    fn continuation_uses_exact_identity_and_respects_scope_and_reset() {
194        let a = selection("a");
195        let b = selection("b");
196        let c = selection("c");
197        let state = SessionModelState {
198            selection: Some(b.clone()),
199            successful_fallback: true,
200            ..Default::default()
201        };
202        let chain = vec![a.model.clone(), b.model.clone(), c.model.clone()];
203        assert_eq!(state.continuation(&a, &chain, false), b);
204        assert_eq!(state.continuation(&a, &chain, true), a);
205        assert_eq!(state.continuation(&c, &chain, false), c);
206        assert_eq!(
207            state.continuation(
208                &a,
209                &[a.model.clone(), c.model.clone(), b.model.clone()],
210                false
211            ),
212            b
213        );
214        assert_eq!(
215            state.continuation(&a, &[a.model.clone(), c.model], false),
216            a
217        );
218        let mut other = a.clone();
219        other.agent = "other".parse().unwrap();
220        assert_eq!(state.continuation(&other, &chain, false), other);
221        other = a.clone();
222        other.preset = Some("python".into());
223        assert_eq!(state.continuation(&other, &chain, false), other);
224        let mut variant_b = b;
225        variant_b.model.variant = Some("fast".parse().unwrap());
226        let state = SessionModelState {
227            selection: Some(variant_b.clone()),
228            successful_fallback: true,
229            ..Default::default()
230        };
231        assert_eq!(state.continuation(&a, &chain, false), a);
232        assert_eq!(
233            state.continuation(&a, &[a.model.clone(), variant_b.model.clone()], false),
234            variant_b
235        );
236    }
