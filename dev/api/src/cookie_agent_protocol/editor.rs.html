1//! The text editor the CLI and the TUI hand a document to.
2
3use std::{path::Path, process::Command};
4
5/// The editor to run, which may carry arguments (`code --wait`):
6/// `$VISUAL`, else `$EDITOR`, else `vi`. Windows has no `vi`: there the
7/// default is VS Code, waiting for its tab to close, when `code` is on
8/// `PATH`, else Notepad, which waits for its window.
9#[must_use]
10pub fn editor() -> String {
11    std::env::var("VISUAL")
12        .or_else(|_| std::env::var("EDITOR"))
13        .ok()
14        .filter(|editor| !editor.trim().is_empty())
15        .unwrap_or_else(default_editor)
16}
17
18#[cfg(windows)]
19fn default_editor() -> String {
20    if crate::paths::find_executable("code").is_some() {
21        "code --wait".to_owned()
22    } else {
23        "notepad".to_owned()
24    }
25}
26
27#[cfg(not(windows))]
28fn default_editor() -> String {
29    "vi".to_owned()
30}
31
32/// `editor`, which may carry arguments (`code --wait`), on `path`.
33#[must_use]
34pub fn editor_command(editor: &str, path: &Path) -> Command {
35    #[cfg(unix)]
36    {
37        let mut command = Command::new("sh");
38        command
39            .arg("-c")
40            .arg(format!("{editor} \"$1\""))
41            .arg("sh")
42            .arg(path);
43        command
44    }
45    #[cfg(not(unix))]
46    {
47        let mut words = editor.split_whitespace();
48        let program = words.next().unwrap_or("notepad");
49        // `Command` finds only `.exe` files on `PATH`; editors such as
50        // VS Code install a `.cmd` launcher.
51        let mut command = Command::new(crate::paths::resolve_program(
52            program,
53            std::iter::empty::<(&str, &str)>(),
54        ));
55        command.args(words).arg(path);
56        command
57    }
58}

60#[cfg(all(test, windows))]
61mod tests {
62    #[test]
63    fn a_cmd_editor_runs_with_its_arguments_on_the_file() {
64        let directory = tempfile::tempdir().unwrap();
65        let editor = directory.path().join("cookie-editor.cmd");
66        std::fs::write(&editor, "@echo %1>%2\r\n").unwrap();
67        let file = directory.path().join("document.toml");
68        std::fs::write(&file, "before").unwrap();
69        let status = super::editor_command(&format!("{} --wait", editor.display()), &file)
70            .status()
71            .unwrap();
72        assert!(status.success());
73        assert_eq!(std::fs::read_to_string(&file).unwrap().trim(), "--wait");
74    }
75
76    #[test]
77    fn notepad_is_on_path() {
78        assert!(crate::paths::find_executable("notepad").is_some());
79    }
