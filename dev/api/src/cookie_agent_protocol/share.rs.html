1//! Share links (`docs/site/specs/share-links.md`, client protocol 1.86.0):
2//! public, unguessable URLs that show one asset to anyone who has them,
3//! without the daemon token.
4//!
5//! The daemon serves a link's landing page at `s/<id>` below its share
6//! base; clients make, list, change, and revoke links with the methods
7//! here and follow `shares.changed`.
8
9use schemars::JsonSchema;
10use serde::{Deserialize, Serialize};
11
12use crate::{AgentId, AssetId, AssetStore, SessionId, ShareId};

15pub const SHARE_CREATE_METHOD: &str = "share.create";

18pub const SHARES_REVOKE_ALL_METHOD: &str = "shares.revoke_all";
19pub const SHARES_CHANGED_METHOD: &str = "shares.changed";

21/// The path below the share base that a link's landing page is at, with
22/// the ID after it.
23pub const SHARE_PATH_PREFIX: &str = "s/";
24
25/// One share link (section 3).
26#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
27pub struct Share {
28    pub id: ShareId,
29    /// The landing page's path relative to the share base, `s/<id>`.
30    pub path: String,
31    /// The landing page's full URL, when the daemon knows its share base
32    /// (`[share] url` or `[server] page_url`). Absent, a client puts the
33    /// address it reached the daemon's page at before `path`.
34    #[serde(default, skip_serializing_if = "Option::is_none")]
35    pub url: Option<String>,
36    /// The asset it shows, always as the asset is now.
37    pub store: AssetStore,
38    pub asset: AssetId,
39    /// What the landing page is headed with; the asset's title when the
40    /// link was made, unless changed since.
41    pub title: String,
42    /// The asset's current type and size.
43    pub media_type: String,
44    pub byte_length: u64,
45    pub created: ShareCreated,
46    /// When the link stops working; never when absent.
47    #[serde(default, skip_serializing_if = "Option::is_none")]
48    pub expires_at: Option<jiff::Timestamp>,
49    /// How many times the landing page was opened. Approximate: chat apps
50    /// that preview links open it too.
51    pub views: u64,
52    #[serde(default, skip_serializing_if = "Option::is_none")]
53    pub last_viewed: Option<jiff::Timestamp>,
54}

56/// When a link was made, and by whom: the user when `session` is absent,

67/// When a link stops working.
68#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
69#[serde(tag = "type", rename_all = "snake_case")]
70pub enum ShareExpiry {
71    Never,
72    At { at: jiff::Timestamp },
73}
74
75impl ShareExpiry {
76    /// The time, `None` for never.
77    #[must_use]
78    pub const fn at(&self) -> Option<jiff::Timestamp> {
79        match self {
80            Self::Never => None,
81            Self::At { at } => Some(*at),
82        }
83    }
84}

163/// The landing page URL of the link `id` under `base`, a share base such
164/// as `https://share.example.com` or `http://127.0.0.1:7419/cookie/`.
165#[must_use]
166pub fn share_url(base: &str, id: &ShareId) -> String {
167    format!("{}/{SHARE_PATH_PREFIX}{id}", base.trim_end_matches('/'))
168}

170#[cfg(test)]
171mod tests {
172    use serde_json::json;
173
174    use super::*;
175
176    #[test]
177    fn ids_are_22_base64url_characters() {
178        assert!(ShareId::new("q3Vt0x9LmA2bCdEfGh1i-_").is_ok());
179        for invalid in [
180            "q3Vt0x9LmA2bCdEfGh1i-",
181            "q3Vt0x9LmA2bCdEfGh1i-__",
182            "q3Vt0x9LmA2bCdEfGh1i+/",
183            "",
184        ] {
185            assert!(ShareId::new(invalid).is_err(), "{invalid}");
186        }
187    }
188
189    #[test]
190    fn an_expiry_is_tagged() {
191        assert_eq!(
192            serde_json::to_value(ShareExpiry::Never).unwrap(),
193            json!({"type": "never"})
194        );
195        let at: jiff::Timestamp = "2026-11-10T00:00:00Z".parse().unwrap();
196        assert_eq!(
197            serde_json::from_value::<ShareExpiry>(json!({"type": "at", "at": at})).unwrap(),
198            ShareExpiry::At { at }
199        );
200    }
201
202    #[test]
203    fn a_url_joins_the_base_and_the_path() {
204        let id = ShareId::new("q3Vt0x9LmA2bCdEfGh1iJk").unwrap();
205        assert_eq!(
206            share_url("https://share.example.com", &id),
207            "https://share.example.com/s/q3Vt0x9LmA2bCdEfGh1iJk"
208        );
209        assert_eq!(
210            share_url("http://127.0.0.1:7419/cookie/", &id),
211            "http://127.0.0.1:7419/cookie/s/q3Vt0x9LmA2bCdEfGh1iJk"
212        );
213    }
