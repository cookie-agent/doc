1//! Share links (`docs/site/specs/share-links.md`, client protocol 1.86.0):
2//! public, unguessable URLs that show one asset to anyone who has them,
3//! without the daemon token. Conversation shares
4//! (`docs/site/specs/conversation-shares.md`, client protocol 1.88.0) are
5//! the same kind of URL for one session: whoever opens one is a guest in it.
6//!
7//! The daemon serves a link's page at `s/<id>` below its share base; clients
8//! make, list, change, and revoke links with the methods here and follow
9//! `shares.changed`.

80pub enum ShareExpiry {

83}

85impl ShareExpiry {
86    /// The time, `None` for never.
87    #[must_use]
88    pub const fn at(&self) -> Option<jiff::Timestamp> {
89        match self {
90            Self::Never => None,
91            Self::At { at } => Some(*at),
92        }

193/// Whether `label` is a label a conversation share may have: on one line,
194/// not blank, and at most [`SHARE_LABEL_MAX_BYTES`] bytes.
195#[must_use]
196pub fn is_share_label(label: &str) -> bool {
197    label.len() <= SHARE_LABEL_MAX_BYTES
198        && !label.trim().is_empty()
199        && !label.chars().any(char::is_control)
200}

202/// The prefix of a guest's origin below `client:`.
203const GUEST_ORIGIN_PREFIX: &str = "guest-";
204/// The letters of the origin's encoding of a label: an origin takes only
205/// lowercase letters, digits, and dashes.
206const BASE32: &[u8; 32] = b"abcdefghijklmnopqrstuvwxyz234567";
207
208/// The origin stamped on what a guest of the link labelled `label` causes:
209/// `client:guest-<label in base 32>-<8 hex digits of the connection>`. Every
210/// event carries its origin, so the label lives on with the messages after
211/// the link is gone, and the label is not the link's secret.
212#[must_use]
213pub fn guest_origin(label: &str, connection: u32) -> EventOrigin {
214    let mut encoded = String::new();
215    let (mut bits, mut held) = (0_u32, 0_u32);
216    for byte in label.bytes().take(SHARE_LABEL_MAX_BYTES) {
217        bits = (bits << 8) | u32::from(byte);
218        held += 8;
219        while held >= 5 {
220            held -= 5;
221            encoded.push(char::from(BASE32[((bits >> held) & 31) as usize]));
222        }
223        bits &= (1 << held) - 1;
224    }
225    if held > 0 {
226        encoded.push(char::from(BASE32[((bits << (5 - held)) & 31) as usize]));
227    }
228    EventOrigin::new(format!(
229        "client:{GUEST_ORIGIN_PREFIX}{encoded}-{connection:08x}"
230    ))
231    .expect("a label of at most 30 bytes in base 32 forms a valid origin")
232}

234/// The label of the link a guest's origin came through; `None` for any
235/// other origin.
236#[must_use]
237pub fn guest_label(origin: &EventOrigin) -> Option<String> {
238    let slug = origin
239        .as_str()
240        .strip_prefix("client:")?
241        .strip_prefix(GUEST_ORIGIN_PREFIX)?;
242    let (encoded, connection) = slug.rsplit_once('-')?;
243    if connection.len() != 8 || !connection.bytes().all(|byte| byte.is_ascii_hexdigit()) {
244        return None;
245    }
246    let mut bytes = Vec::with_capacity(encoded.len() * 5 / 8);
247    let (mut bits, mut held) = (0_u32, 0_u32);
248    for letter in encoded.bytes() {
249        let value = BASE32.iter().position(|known| *known == letter)?;
250        bits = (bits << 5) | u32::try_from(value).ok()?;
251        held += 5;
252        if held >= 8 {
253            held -= 8;
254            bytes.push(u8::try_from((bits >> held) & 255).ok()?);
255        }
256        bits &= (1 << held) - 1;
257    }
258    String::from_utf8(bytes)
259        .ok()
260        .filter(|label| is_share_label(label))
261}

350    #[test]
351    fn a_guest_origin_keeps_its_label() {
352        for label in [
353            "Alice",
354            "Guest",
355            "a",
356            "Zoë · lab",
357            &"x".repeat(SHARE_LABEL_MAX_BYTES),
358        ] {
359            let origin = guest_origin(label, 0xdead_beef);
360            assert!(origin.as_str().starts_with("client:guest-"), "{origin:?}");
361            assert_eq!(guest_label(&origin).as_deref(), Some(label));
362        }
363    }
