1//! Semver protocol versions, as specified in
2//! `docs/site/specs/compatibility-v1.md` section 3.
3//!
4//! Each surface's version is the string `"MAJOR.MINOR.PATCH"`: three decimal
5//! integers without leading zeros, and no pre-release or build suffix. Peers
6//! with different majors do not connect; peers with the same major share the
7//! lower of their two minors, and the patch never matters. The pre-1.0
8//! versions, whose every minor could break the wire, never connect to a 1.x
9//! peer.
10//!
11//! The client, plugin, and runtime handshakes send
12//! [`CLIENT_PROTOCOL_VERSION`], [`PLUGIN_PROTOCOL_VERSION`], and
13//! [`RUNTIME_PROTOCOL_VERSION`] and read the peer's with
14//! [`negotiate_version`]. All three froze at `1.0.0`; from there a minor only
15//! adds and a major breaks.
16

23/// The client protocol version this build speaks.
24pub const CLIENT_PROTOCOL_VERSION: SemVer = SemVer::new(1, 26, 0);
25/// The plugin protocol version this build speaks.
26pub const PLUGIN_PROTOCOL_VERSION: SemVer = SemVer::new(1, 14, 0);
27/// The runtime protocol version this build speaks.
28pub const RUNTIME_PROTOCOL_VERSION: SemVer = SemVer::new(1, 8, 0);
29
30/// The handshake member that carries a peer's version.
31pub const PROTOCOL_VERSION_MEMBER: &str = "protocol_version";
32
33/// A `MAJOR.MINOR.PATCH` protocol version.
34#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
35pub struct SemVer {
36    major: u32,
37    minor: u32,
38    patch: u32,
39}
40
41impl SemVer {
42    #[must_use]
43    pub const fn new(major: u32, minor: u32, patch: u32) -> Self {
44        Self {
45            major,
46            minor,
47            patch,
48        }
49    }
50
51    /// Parses `"MAJOR.MINOR.PATCH"`.
52    pub fn parse(text: &str) -> Result<Self, SemVerError> {
53        let mut parts = text.split('.');
54        let mut next = || parts.next().ok_or(SemVerError::Malformed).and_then(part);
55        let version = Self::new(next()?, next()?, next()?);
56        if parts.next().is_some() {
57            return Err(SemVerError::Malformed);
58        }
59        Ok(version)
60    }
61
62    #[must_use]
63    pub const fn major(self) -> u32 {
64        self.major
65    }
66
67    #[must_use]
68    pub const fn minor(self) -> u32 {
69        self.minor
70    }
71
72    #[must_use]
73    pub const fn patch(self) -> u32 {
74        self.patch
75    }
76
77    /// Whether a peer at `other` may connect: the majors are equal.
78    #[must_use]
79    pub const fn is_compatible_with(self, other: Self) -> bool {
80        self.major == other.major
81    }
82
83    /// The minor two compatible peers share: the lower of the two. `None`
84    /// when they are not compatible.
85    #[must_use]
86    pub fn shared_minor(self, other: Self) -> Option<u32> {
87        self.is_compatible_with(other)
88            .then(|| self.minor.min(other.minor))
89    }
90}
91
92fn part(text: &str) -> Result<u32, SemVerError> {
93    let digits = !text.is_empty() && text.bytes().all(|byte| byte.is_ascii_digit());
94    if !digits || (text.len() > 1 && text.starts_with('0')) {
95        return Err(SemVerError::Malformed);
96    }
97    text.parse().map_err(|_| SemVerError::Malformed)
98}
99
100#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
101pub enum SemVerError {
102    #[error("a protocol version is MAJOR.MINOR.PATCH without leading zeros or suffixes")]
103    Malformed,
104}
105
106impl fmt::Display for SemVer {
107    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
108        write!(formatter, "{}.{}.{}", self.major, self.minor, self.patch)
109    }
110}
111
112impl FromStr for SemVer {
113    type Err = SemVerError;
114
115    fn from_str(text: &str) -> Result<Self, Self::Err> {
116        Self::parse(text)
117    }
118}

120impl Serialize for SemVer {
121    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
122    where
123        S: serde::Serializer,
124    {
125        serializer.collect_str(self)
126    }
127}

129impl<'de> Deserialize<'de> for SemVer {
130    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
131    where
132        D: serde::Deserializer<'de>,
133    {
134        let text = Cow::<str>::deserialize(deserializer)?;
135        Self::parse(&text).map_err(serde::de::Error::custom)
136    }

157/// What a handshake's raw `protocol_version` member holds.
158#[derive(Clone, Debug, Eq, PartialEq, thiserror::Error)]
159pub enum PeerVersionError {
160    #[error("the handshake has no protocol_version")]
161    Missing,
162    /// An integer: a build from before semver versions.
163    #[error("the peer speaks the old exact protocol version {0}")]
164    Legacy(u64),
165    #[error("the handshake's protocol_version is not a MAJOR.MINOR.PATCH string")]
166    Malformed,
167}
168
169/// Reads `protocol_version` from a handshake's raw params before anything
170/// else is decoded, so that a peer of another major is answered with
171/// `unsupported_protocol` rather than with a decoding error of a shape it
172/// does not share.
173pub fn peer_protocol_version(params: &Value) -> Result<SemVer, PeerVersionError> {
174    match params.get(PROTOCOL_VERSION_MEMBER) {
175        None => Err(PeerVersionError::Missing),
176        Some(Value::String(text)) => SemVer::parse(text).map_err(|_| PeerVersionError::Malformed),
177        Some(Value::Number(number)) => number
178            .as_u64()
179            .map_or(Err(PeerVersionError::Malformed), |legacy| {
180                Err(PeerVersionError::Legacy(legacy))
181            }),
182        Some(_) => Err(PeerVersionError::Malformed),
183    }
184}
185
186/// Why a handshake's peer version is refused.
187#[derive(Clone, Debug, Eq, PartialEq)]
188pub enum VersionRefusal {
189    /// A version this build cannot speak with: another major, or the
190    /// integer version of a build from before semver.
191    Unsupported { supported: SemVer, peer: String },
192    /// The handshake carries no readable version.
193    Malformed(PeerVersionError),
194}
195
196impl VersionRefusal {
197    /// The answer to the handshake: the registry's `unsupported_protocol`
198    /// naming both versions, or `invalid_params`.
199    #[must_use]
200    pub fn to_rpc_error(&self) -> crate::JsonRpcError {
201        match self {
202            Self::Unsupported { supported, peer } => crate::errors::PROTOCOL_UNSUPPORTED
203                .error()
204                .field("supported_version", supported.to_string())
205                .field("peer_version", peer)
206                .build(),
207            Self::Malformed(error) => crate::errors::BAD_PARAMS
208                .error()
209                .field("field", PROTOCOL_VERSION_MEMBER)
210                .detail(error.to_string())
211                .build(),
212        }
213    }
214}
215
216impl fmt::Display for VersionRefusal {
217    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
218        match self {
219            Self::Unsupported { .. } => formatter.write_str(&self.to_rpc_error().message),
220            Self::Malformed(error) => write!(formatter, "the handshake is malformed: {error}"),
221        }
222    }
223}
224
225impl std::error::Error for VersionRefusal {}
226
227/// Reads the peer's version from a handshake's raw params or reply, before
228/// anything else of it is decoded, and checks that a build speaking
229/// `supported` can talk to it. Returns the peer's version.
230pub fn negotiate_version(supported: SemVer, handshake: &Value) -> Result<SemVer, VersionRefusal> {
231    match peer_protocol_version(handshake) {
232        Ok(peer) if supported.is_compatible_with(peer) => Ok(peer),
233        Ok(peer) => Err(VersionRefusal::Unsupported {
234            supported,
235            peer: peer.to_string(),
236        }),
237        Err(PeerVersionError::Legacy(peer)) => Err(VersionRefusal::Unsupported {
238            supported,
239            peer: peer.to_string(),
240        }),
241        Err(error) => Err(VersionRefusal::Malformed(error)),
242    }
243}

251    #[test]
252    fn versions_parse_and_print_as_three_parts() {
253        let version = SemVer::parse("1.20.3").unwrap();
254        assert_eq!(
255            (version.major(), version.minor(), version.patch()),
256            (1, 20, 3)
257        );
258        assert_eq!(version.to_string(), "1.20.3");
259        assert_eq!("0.0.0".parse::<SemVer>().unwrap(), SemVer::new(0, 0, 0));
260        assert_eq!(
261            SemVer::parse("4294967295.0.10").unwrap(),
262            SemVer::new(u32::MAX, 0, 10)
263        );
264    }

322    #[test]
323    fn a_handshake_of_another_version_is_refused_with_both_versions() {
324        let supported = SemVer::new(1, 3, 0);
325        for peer in ["1.3.9", "1.0.0", "1.7.2"] {
326            assert_eq!(
327                negotiate_version(supported, &json!({"protocol_version": peer, "x": 1})),
328                Ok(SemVer::parse(peer).unwrap())
329            );
330        }
331        for (peer, named) in [
332            (json!("0.52.0"), "0.52.0"),
333            (json!("2.3.0"), "2.3.0"),
334            (json!(2), "2"),
335        ] {
336            let refusal =
337                negotiate_version(supported, &json!({"protocol_version": peer})).unwrap_err();
338            let error = refusal.to_rpc_error();
339            assert_eq!(error.code, crate::errors::UNSUPPORTED_PROTOCOL);
340            let data = error.data.unwrap();
341            assert_eq!(data["code"], "unsupported_protocol");
342            assert_eq!(data["supported_version"], "1.3.0");
343            assert_eq!(data["peer_version"], named);
344            assert!(
345                refusal
346                    .to_string()
347                    .contains(&format!("1.3.0 here and {named}"))
348            );
349        }
350        for malformed in [json!({}), json!({"protocol_version": "3"})] {
351            let refusal = negotiate_version(supported, &malformed).unwrap_err();
352            assert!(matches!(refusal, VersionRefusal::Malformed(_)));
353            assert_eq!(refusal.to_rpc_error().code, crate::errors::INVALID_PARAMS);
354        }
355    }

391        assert_eq!(

393            Err(PeerVersionError::Missing)
