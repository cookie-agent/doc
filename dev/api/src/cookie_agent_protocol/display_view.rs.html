1//! The display view of a session's history and its newest-first windows
2//! (`docs/site/specs/session-storage-v3.md` section 3, client protocol
3//! 1.59.0).
4//!
5//! The display view is what a transcript shows of a session's events:
6//!
7//! - **Bookkeeping is left out**: `model_request_prepared`, which no
8//!   client reads, and a `model_replay_evaluated` whose decisions discard
9//!   nothing (its rows are below the warning level).
10//! - **Superseded settings are left out of a page**: a
11//!   `model_tools_published` a later one in the page replaces, and in an
12//!   `acp_session_updated` a command list or starting options a later one
13//!   replaces, or a tool presentation a later one of the same call
14//!   replaces. An update left empty is dropped.
15//! - **Long tool results are cut**: a result whose output, display, and
16//!   metadata take more than [`CUT_ABOVE_BYTES`] keeps the first
17//!   [`CUT_KEEP_BYTES`] of its output, a display fit to the same budget,
18//!   and no metadata; a [`CutOutput`] names it and the whole result's size,
19//!   and `event.read` returns it whole.
20//! - **A tool result's metadata is left out**: no transcript reads it, and
21//!   an ACP agent's results repeat their whole output there.
22//!
23//! Every other event is sent as stored. `model_attempt_started`,
24//! `model_output_started`, and `model_usage_recorded` stay: a transcript
25//! builds its replies, their thinking times, and their costs from them.
26//!
27//! A newest-first window ([`tail_window`]) holds the newest display events
28//! within a byte budget starting at a run, the events before it its state
29//! depends on ([`state_events`]), and, when the newest run alone exceeds the
30//! budget, the run's start, a [`HistoryGap`], and its newest model
31//! attempts. [`older_page`] reads the history before a window the same way.
32

43/// A tool result larger than this, output, display, and metadata
44/// together, is cut in the display view.
45pub const CUT_ABOVE_BYTES: usize = 4 * 1024;
46/// How much of a cut result's output, and of its display, the view keeps.
47pub const CUT_KEEP_BYTES: usize = 2 * 1024;
48
49/// Whether the display view leaves out an event of this payload wherever it
50/// stands: bookkeeping no transcript shows.
51#[must_use]
52pub fn leaves_out(payload: &EventPayload) -> bool {
53    match payload {
54        EventPayload::ModelRequestPrepared { .. } => true,
55        payload if payload.is_omitted_live_state() => true,
56        // Its replayed and reconstructed entries are debug rows, and an
57        // empty list an info row; only a discard warns.
58        EventPayload::ModelReplayEvaluated {
59            ordered_decisions, ..
60        } => !ordered_decisions.iter().any(|decision| {
61            matches!(
62                decision.disposition,
63                ReplayDisposition::DiscardedForeignAdapter { .. }
64                    | ReplayDisposition::DiscardedForeignModelSelection { .. }
65                    | ReplayDisposition::DiscardedForeignVariant { .. }
66                    | ReplayDisposition::DiscardedInvalidPayload { .. }
67            )
68        }),
69        _ => false,
70    }
71}

73/// One event as the display view sends it: `None` when the view leaves it
74/// out, else the event, cut when its tool result is long, with what the
75/// cut left out. The same event is shared when nothing changes.
76#[must_use]
77pub fn shown(event: &Arc<StoredEvent>) -> Option<(Arc<StoredEvent>, Option<CutOutput>)> {
78    if leaves_out(&event.payload) {
79        return None;
80    }
81    Some(match cut_result(event) {
82        Some((event, cut)) => (Arc::new(event), Some(cut)),
83        None => match without_metadata(event) {
84            Some(event) => (Arc::new(event), None),
85            None => (Arc::clone(event), None),
86        },
87    })
88}

90/// A tool result's event without its `metadata`; `None` when it has none.
91fn without_metadata(event: &StoredEvent) -> Option<StoredEvent> {
92    let EventPayload::ToolCallTerminated { termination } = &event.payload else {
93        return None;
94    };
95    if termination
96        .result
97        .as_ref()
98        .is_none_or(|result| result.metadata.is_null())
99    {
100        return None;
101    }
102    let mut termination = termination.clone();
103    if let Some(result) = termination.result.as_mut() {
104        result.metadata = serde_json::Value::Null;
105    }
106    Some(with_payload(
107        event,
108        EventPayload::ToolCallTerminated { termination },
109    ))
110}

112/// A `tool_call_terminated` event whose result is long, cut as the display
113/// view sends it.
114#[must_use]
115pub fn cut_result(event: &StoredEvent) -> Option<(StoredEvent, CutOutput)> {
116    let EventPayload::ToolCallTerminated { termination } = &event.payload else {
117        return None;
118    };
119    let result = termination.result.as_ref()?;
120    // The output alone decides most of them without measuring the rest.
121    if result.output.len() <= CUT_ABOVE_BYTES {
122        let display = serialized_len(&result.display);
123        let metadata = serialized_len(&result.metadata);
124        if result.output.len() + display + metadata <= CUT_ABOVE_BYTES {
125            return None;
126        }
127    }
128    let bytes = serialized_len(result) as u64;
129    let mut termination = termination.clone();
130    let result = termination.result.as_mut().expect("checked above");
131    result
132        .output
133        .truncate(floor_char_boundary(&result.output, CUT_KEEP_BYTES));
134    let blocks = std::mem::take(&mut result.display)
135        .into_iter()
136        .filter_map(Open::into_known)
137        .collect::<Vec<DisplayBlock>>();
138    result.display = crate::fit_display_budget(blocks, CUT_KEEP_BYTES)
139        .into_iter()
140        .map(Open::Known)
141        .collect();
142    result.metadata = serde_json::Value::Null;
143    Some((
144        StoredEvent {
145            engine_version: event.engine_version.clone(),
146            origin: event.origin.clone(),
147            session_id: event.session_id,
148            run_id: event.run_id,
149            seq: event.seq,
150            timestamp: event.timestamp,
151            payload: EventPayload::ToolCallTerminated { termination },
152        },
153        CutOutput {
154            seq: event.seq,
155            bytes,
156        },
157    ))
158}

160/// Events in `view`: [`display_events`] for the display view, else the
161/// events as stored.

163pub fn view_events(

174/// A contiguous run of a session's events in the display view, in order,
175/// with the results it cut. With `keep_last`, the last event is sent even
176/// when the view would leave it out, so a reader that pages forward from
177/// it continues after the right sequence.
178#[must_use]
179pub fn display_events(
180    events: &[Arc<StoredEvent>],
181    keep_last: bool,
182) -> (Vec<Arc<StoredEvent>>, Vec<CutOutput>) {
183    let superseded = Superseded::of(events);
184    let mut shown_events = Vec::with_capacity(events.len());
185    let mut cuts = Vec::new();
186    for (index, event) in events.iter().enumerate() {
187        let last = keep_last && index + 1 == events.len();
188        let event = match superseded.apply(index, event) {
189            Slim::Whole => Arc::clone(event),
190            Slim::Changed(event) => Arc::new(*event),
191            Slim::Dropped if last => Arc::clone(event),
192            Slim::Dropped => continue,
193        };
194        match shown(&event) {
195            Some((event, cut)) => {
196                cuts.extend(cut);
197                shown_events.push(event);
198            }
199            None if last => shown_events.push(event),
200            None => {}
201        }
202    }
203    (shown_events, cuts)
204}
205
206/// What [`Superseded::apply`] does to one event.
207enum Slim {
208    Whole,
209    Changed(Box<StoredEvent>),
210    Dropped,
211}
212
213/// The settings in a run of events that a later event of the same run
214/// replaces.
215struct Superseded {
216    /// The last `model_tools_published`.
217    tools: Option<usize>,
218    /// The last `acp_session_updated` with commands, and with starting
219    /// options.
220    commands: Option<usize>,
221    starting_options: Option<usize>,
222    /// The last tool presentation of each call.
223    presentations: HashMap<ToolCallId, usize>,
224}
225
226impl Superseded {
227    fn of(events: &[Arc<StoredEvent>]) -> Self {
228        let mut superseded = Self {
229            tools: None,
230            commands: None,
231            starting_options: None,
232            presentations: HashMap::new(),
233        };
234        for (index, event) in events.iter().enumerate() {
235            match &event.payload {
236                EventPayload::ModelToolsPublished { .. } => superseded.tools = Some(index),
237                EventPayload::AcpSessionUpdated { update } => {
238                    if update.commands.is_some() {
239                        superseded.commands = Some(index);
240                    }
241                    if update.starting_options.is_some() {
242                        superseded.starting_options = Some(index);
243                    }
244                    if let Some(presented) = &update.tool_presentation {
245                        superseded
246                            .presentations
247                            .insert(presented.tool_call_id, index);
248                    }
249                }
250                _ => {}
251            }
252        }
253        superseded
254    }
255
256    fn apply(&self, index: usize, event: &StoredEvent) -> Slim {
257        match &event.payload {
258            EventPayload::ModelToolsPublished { .. } if self.tools != Some(index) => Slim::Dropped,
259            EventPayload::AcpSessionUpdated { update } => {
260                let commands = update.commands.is_some() && self.commands != Some(index);
261                let starting =
262                    update.starting_options.is_some() && self.starting_options != Some(index);
263                let presentation = update.tool_presentation.as_ref().is_some_and(|presented| {
264                    self.presentations.get(&presented.tool_call_id) != Some(&index)
265                });
266                if !(commands || starting || presentation) {
267                    return Slim::Whole;
268                }
269                let mut update = update.clone();
270                if commands {
271                    update.commands = None;
272                }
273                if starting {
274                    update.starting_options = None;
275                }
276                if presentation {
277                    update.tool_presentation = None;
278                }
279                if update.is_empty() {
280                    Slim::Dropped
281                } else {
282                    Slim::Changed(Box::new(with_payload(
283                        event,
284                        EventPayload::AcpSessionUpdated { update },
285                    )))
286                }
287            }
288            _ => Slim::Whole,
289        }
290    }
291}

293/// The events of `prefix`, a log's events from its start, that the
294/// session's state after them depends on, in sequence order: what a client
295/// folds before a window that starts after `prefix` so that its state ends
296/// where a full replay's would.
297///
298/// They are the latest `run_started` and its `run_finished`, the latest
299/// `memory_loaded`,
300/// `model_tools_published`, `acp_session_started`, and
301/// `session_model_selected`; the session's `session_created`; the latest ACP
302/// commands, options, starting options, and context use, each event holding
303/// only those of them it is the latest for; the latest
304/// `goal_activated` and the goal's changes after it; each runtime
305/// attachment still attached; a hold no run or input cleared since; the
306/// tools announced since the latest compaction; and each input still
307/// queued.
308#[must_use]
309pub fn state_events(prefix: &[Arc<StoredEvent>]) -> Vec<Arc<StoredEvent>> {
310    let mut picked: Vec<Arc<StoredEvent>> = Vec::new();
311    let mut take = |event: &Arc<StoredEvent>| picked.push(Arc::clone(event));
312    if let Some(first) = prefix.first()
313        && matches!(first.payload, EventPayload::SessionCreated { .. })
314    {
315        take(first);
316    }
317    let mut latest = HashSet::<&'static str>::new();
318    let mut acp_members = HashSet::<&'static str>::new();
319    let mut attachments = HashSet::new();
320    let mut goal_found = false;
321    let mut checkpoint_found = false;
322    let mut hold_settled = false;
323    let mut acp_updates = Vec::new();
324    for event in prefix.iter().rev() {
325        match &event.payload {
326            EventPayload::RunStarted { .. } if latest.insert("run_started") => take(event),
327            // The latest run's end, when it ended: its run no longer runs.
328            EventPayload::RunFinished { .. } if !latest.contains("run_started") => {
329                if latest.insert("run_finished") {
330                    take(event);
331                }
332            }
333            EventPayload::MemoryLoaded { .. } if latest.insert("memory_loaded") => take(event),
334            EventPayload::ModelToolsPublished { .. } if latest.insert("model_tools") => take(event),
335            EventPayload::AcpSessionStarted { .. } if latest.insert("acp_session_started") => {
336                take(event);
337            }
338            EventPayload::SessionModelSelected { .. } if latest.insert("model_selected") => {
339                take(event)
340            }
341            EventPayload::AcpSessionUpdated { update } => {
342                let mut members = Vec::new();
343                for (member, present) in [
344                    ("commands", update.commands.is_some()),
345                    ("options", update.options.is_some()),
346                    ("starting_options", update.starting_options.is_some()),
347                    ("context", update.context.is_some()),
348                ] {
349                    if present && acp_members.insert(member) {
350                        members.push(member);
351                    }
352                }
353                if !members.is_empty() {
354                    acp_updates.push((event, members));
355                }
356            }
357            EventPayload::GoalChecklistRevised { .. }
358            | EventPayload::GoalLifecycleChanged { .. }
359                if !goal_found =>
360            {
361                take(event);
362            }
363            EventPayload::GoalActivated { .. } if !goal_found => {
364                goal_found = true;
365                take(event);
366            }
367            EventPayload::RuntimeAttached { name, .. } => {
368                if attachments.insert(name.clone()) {
369                    take(event);
370                }
371            }
372            EventPayload::RuntimeDetached { name } => {
373                attachments.insert(name.clone());
374            }
375            EventPayload::AutomaticRunsHeld { .. } if !hold_settled => {
376                hold_settled = true;
377                take(event);
378            }
379            EventPayload::AutomaticRunsResumed {} | EventPayload::UserInputSubmitted { .. } => {
380                hold_settled = true;
381            }
382            EventPayload::ContextCheckpointCommitted { .. } => checkpoint_found = true,
383            EventPayload::DeferredToolsAnnounced { .. } if !checkpoint_found => take(event),
384            _ => {}
385        }
386    }
387    for (event, members) in acp_updates {
388        let EventPayload::AcpSessionUpdated { update } = &event.payload else {
389            continue;
390        };
391        let mut kept = crate::AcpSessionUpdate::default();
392        for member in members {
393            match member {
394                "commands" => kept.commands.clone_from(&update.commands),
395                "options" => kept.options.clone_from(&update.options),
396                "starting_options" => kept.starting_options.clone_from(&update.starting_options),
397                _ => kept.context.clone_from(&update.context),
398            }
399        }
400        picked.push(Arc::new(with_payload(
401            event,
402            EventPayload::AcpSessionUpdated { update: kept },
403        )));
404    }
405    let queued = QueuedInputs::from_events(prefix);
406    for entry in queued.entries() {
407        if let Some(event) = event_at(prefix, entry.queued_seq) {
408            picked.push(Arc::clone(event));
409        }
410    }
411    picked.sort_by_key(|event| event.seq);
412    picked.dedup_by_key(|event| event.seq);
413    picked
414}

416/// The events of `skipped` a reader that holds what comes before and after
417/// it still needs: the settings ([`state_events`]) and what is still
418/// waiting when the log ends with `rest`: an approval or a question not
419/// answered, an input admitted but not applied, and a tool call still
420/// running.
421fn kept_in_gap(skipped: &[Arc<StoredEvent>], rest: &[Arc<StoredEvent>]) -> Vec<Arc<StoredEvent>> {
422    let mut answered_approvals = HashSet::<ApprovalId>::new();
423    let mut answered_questions = HashSet::<ElicitationId>::new();
424    let mut applied_inputs = HashSet::<u64>::new();
425    let mut ended_calls = HashSet::<ToolCallId>::new();
426    for event in skipped.iter().chain(rest) {
427        match &event.payload {
428            EventPayload::ApprovalFinalized { approval_id, .. } => {
429                answered_approvals.insert(*approval_id);
430            }
431            EventPayload::ElicitationResolved { resolution } => {
432                answered_questions.insert(resolution.elicitation_id);
433            }
434            EventPayload::UserInputApplied { user_input_seq }
435            | EventPayload::UserInputRecalled { user_input_seq, .. } => {
436                applied_inputs.insert(*user_input_seq);
437            }
438            EventPayload::ToolCallTerminated { termination } => {
439                ended_calls.insert(termination.tool_call_id);
440            }
441            _ => {}
442        }
443    }
444    // The gap's settings, without what the session's start holds: the
445    // window's header has it.
446    let mut kept = state_events(skipped)
447        .into_iter()
448        .filter(|event| !matches!(event.payload, EventPayload::SessionCreated { .. }))
449        .collect::<Vec<_>>();
450    for event in skipped {
451        let waiting = match &event.payload {
452            EventPayload::ApprovalRequested { request } => {
453                !answered_approvals.contains(&request.approval_id())
454            }
455            EventPayload::ApprovalStep { approval_id, .. } => {
456                !answered_approvals.contains(approval_id)
457            }
458            EventPayload::ElicitationRequested { request } => {
459                !answered_questions.contains(&request.elicitation_id)
460            }
461            EventPayload::UserInputAdmitted { .. } => !applied_inputs.contains(&event.seq),
462            EventPayload::ToolCallStarted { start } => !ended_calls.contains(&start.tool_call_id),
463            _ => false,
464        };
465        if waiting {
466            kept.push(Arc::clone(event));
467        }
468    }
469    kept.sort_by_key(|event| event.seq);
470    kept.dedup_by_key(|event| event.seq);
471    kept
472}

474/// A newest-first answer: [`tail_window`] or [`older_page`].
475#[derive(Clone, Debug, Default, PartialEq)]
476pub struct Window {
477    /// The display events, in sequence order.
478    pub events: Vec<Arc<StoredEvent>>,
479    /// The results among them the view cut.
480    pub cut: Vec<CutOutput>,
481    /// [`state_events`] of everything before the first event.
482    pub header: Vec<Arc<StoredEvent>>,
483    /// The sequence of the first event, from which older history is read
484    /// (`events.page`'s `before`); `None` when it is the log's first.
485    pub older_cursor: Option<u64>,
486    /// Display events left out inside the window.
487    pub gap: Option<HistoryGap>,
488}

490/// The newest display events of a log, `events`, within `budget` bytes of
491/// JSON: the newest runs that fit whole, starting at a `run_started` or
492/// `user_input_submitted`. When the newest run alone does not fit, its
493/// start (through its first input), the settings and waiting events of
494/// what it leaves out, a gap, and the run's newest model attempts that fit.
495/// The log's last event is always sent, as a forward page sends it.
496#[must_use]
497pub fn tail_window(events: &[Arc<StoredEvent>], view: EventsView, budget: usize) -> Window {
498    newest(events, None, view, budget).unwrap_or_default()
499}

501/// [`tail_window`] of a log read from a run boundary on: `suffix` holds its
502/// events from a `run_started` or `user_input_submitted` to its end, and
503/// `header` is [`state_events`] of every event before that one. The answer
504/// is the one [`tail_window`] gives for the whole log, or `None` when that
505/// window would reach before `suffix`, so the whole log has to be read.
506///
507/// It needs no more than `header`: [`state_events`] of `header` followed by
508/// later events is [`state_events`] of the log through them.
509#[must_use]
510pub fn tail_window_after(
511    header: &[Arc<StoredEvent>],
512    suffix: &[Arc<StoredEvent>],
513    view: EventsView,
514    budget: usize,
515) -> Option<Window> {
516    if suffix.first().is_none_or(|first| !is_run_boundary(first)) {
517        return None;
518    }
519    newest(suffix, Some(header), view, budget)
520}

522/// Where a cache of a log's newest-first view starts: the index of the run
523/// boundary from which [`tail_window_after`] answers every request of up to
524/// `budget` bytes, or `None` when the whole log fits in it.
525#[must_use]
526pub fn cache_start(events: &[Arc<StoredEvent>], budget: usize) -> Option<usize> {
527    let sizes = Sizes::new(events, EventsView::Display);
528    let fit = sizes.fit_before(events.len(), budget);
529    if fit == 0 {
530        return None;
531    }
532    // The window starts at a boundary at or after `fit`, or at the last one
533    // before it when the newest run does not fit: either way after the last
534    // boundary before `fit`, whose events from there on do not fit. A
535    // window in the full view, a smaller one, and a later log's window
536    // start later still.
537    (0..fit)
538        .rev()
539        .find(|&index| is_run_boundary(&events[index]))
540}

542/// [`tail_window`] of `events`, the whole log when `before` is `None`, else
543/// a suffix starting at a run boundary with `before` the header of what
544/// precedes it. `None` when the window would reach before such a suffix.
545fn newest(
546    events: &[Arc<StoredEvent>],
547    before: Option<&[Arc<StoredEvent>]>,
548    view: EventsView,
549    budget: usize,
550) -> Option<Window> {
551    let Some(last) = events.len().checked_sub(1) else {
552        return before.is_none().then(Window::default);
553    };
554    // The settings before `events[end]`, and where older history starts.
555    let header = |end: usize| match before {
556        None => state_events(&events[..end]),
557        Some(before) => state_events(
558            &before
559                .iter()
560                .chain(&events[..end])
561                .cloned()
562                .collect::<Vec<_>>(),
563        ),
564    };
565    let older_cursor = |start: usize| (start > 0 || before.is_some()).then(|| events[start].seq);
566    let sizes = Sizes::new(events, view);
567    let fit = sizes.fit_before(events.len(), budget);
568    if fit == 0 && before.is_some() {
569        // Everything held fits: older runs may too.
570        return None;
571    }
572    let run_start = (fit..=last).find(|&index| is_run_boundary(&events[index]));
573    if fit == 0 || run_start.is_some() {
574        let start = if fit == 0 {
575            0
576        } else {
577            run_start.unwrap_or(fit)
578        };
579        return Some(whole(
580            events,
581            view,
582            start,
583            events.len(),
584            true,
585            header(start),
586            older_cursor(start),
587        ));
588    }
589    // The newest run does not fit: its head, a gap, and its newest
590    // attempts.
591    let run = (0..=last)
592        .rev()
593        .find(|&index| is_run_boundary(&events[index]))
594        .unwrap_or(0);
595    let head_end = (run..=last)
596        .take_while(|&index| {
597            index == run
598                || !matches!(
599                    events[index].payload,
600                    EventPayload::ModelAttemptStarted { .. }
601                )
602        })
603        .find(|&index| {
604            matches!(
605                events[index].payload,
606                EventPayload::UserInputSubmitted { .. }
607            )
608        })
609        .unwrap_or(run)
610        + 1;
611    let head_bytes = sizes.sum(run, head_end);
612    let tail_fit = sizes
613        .fit_before(events.len(), budget.saturating_sub(head_bytes))
614        .max(head_end);
615    let attempts = |range: std::ops::RangeInclusive<usize>| {
616        range
617            .into_iter()
618            .filter(|&index| {
619                matches!(
620                    events[index].payload,
621                    EventPayload::ModelAttemptStarted { .. }
622                )
623            })
624            .collect::<Vec<_>>()
625    };
626    let tail_start = attempts(tail_fit..=last)
627        .first()
628        .copied()
629        .or_else(|| attempts(head_end..=last).last().copied())
630        .unwrap_or(tail_fit)
631        .min(last);
632    if tail_start <= head_end {
633        return Some(whole(
634            events,
635            view,
636            run,
637            events.len(),
638            true,
639            header(run),
640            older_cursor(run),
641        ));
642    }
643    let skipped = &events[head_end..tail_start];
644    let kept = kept_in_gap(skipped, &events[tail_start..]);
645    let sent = events[run..head_end]
646        .iter()
647        .chain(&kept)
648        .chain(&events[tail_start..])
649        .cloned()
650        .collect::<Vec<_>>();
651    let (window_events, cut) = view_events(&sent, view, true);
652    let skipped_count = skipped
653        .iter()
654        .filter(|event| view.is_full() || !leaves_out(&event.payload))
655        .count()
656        .saturating_sub(kept.len());
657    Some(Window {
658        events: window_events,
659        cut,
660        header: header(run),
661        older_cursor: older_cursor(run),
662        gap: Some(HistoryGap {
663            after_seq: events[head_end - 1].seq,
664            before_seq: events[tail_start].seq,
665            skipped: skipped_count as u64,
666        }),
667    })
668}

670/// The newest display events of `events`, a log, before sequence `before`,
671/// within `budget` bytes: starting at the earliest run or model attempt
672/// that fits, else at the earliest event that does, and at least one event
673/// while any precede `before`.
674#[must_use]
675pub fn older_page(
676    events: &[Arc<StoredEvent>],
677    before: u64,
678    view: EventsView,
679    budget: usize,
680) -> Window {
681    let end = events.partition_point(|event| event.seq < before);
682    if end == 0 {
683        return Window::default();
684    }
685    let sizes = Sizes::new(&events[..end], view);
686    let fit = sizes.fit_before(end, budget).min(end - 1);
687    let start = (fit..end)
688        .find(|&index| {
689            is_run_boundary(&events[index])
690                || matches!(
691                    events[index].payload,
692                    EventPayload::ModelAttemptStarted { .. }
693                )
694        })
695        .unwrap_or(fit);
696    whole(
697        events,
698        view,
699        start,
700        end,
701        false,
702        state_events(&events[..start]),
703        (start > 0).then(|| events[start].seq),
704    )
705}

707/// What a client that holds `held`, the header of a window or page, lacks
708/// of `header`, the header of an older page: its events that `held` holds
709/// no identical event of the same sequence for (`events.page` with
710/// `header: "delta"`, client protocol 1.64.0). [`merge_header`] puts them
711/// together again.
712#[must_use]
713pub fn header_delta(
714    held: &[Arc<StoredEvent>],
715    header: Vec<Arc<StoredEvent>>,
716) -> Vec<Arc<StoredEvent>> {
717    header
718        .into_iter()
719        .filter(|event| {
720            event_at(held, event.seq)
721                .is_none_or(|held| !Arc::ptr_eq(held, event) && **held != **event)
722        })
723        .collect()
724}

726/// The header of an older page from `held`, the header the client held,
727/// and `delta`, what the page's answer sent of the new one
728/// ([`header_delta`]): the held events before `older_cursor`, the page's
729/// first sequence, that `delta` holds no event of the same sequence for,
730/// with `delta`'s, in sequence order. Every event of the older header
731/// before `older_cursor` is in the newer one as it stands, or is sent, so
732/// this is the page's whole header.
733#[must_use]
734pub fn merge_header<E: std::borrow::Borrow<StoredEvent>>(
735    held: Vec<E>,
736    delta: Vec<E>,
737    older_cursor: Option<u64>,
738) -> Vec<E> {
739    let Some(start) = older_cursor else {
740        return Vec::new();
741    };
742    let sent = delta
743        .iter()
744        .map(|event| event.borrow().seq)
745        .collect::<HashSet<_>>();
746    let mut merged = held
747        .into_iter()
748        .filter(|event| {
749            let seq = event.borrow().seq;
750            seq < start && !sent.contains(&seq)
751        })
752        .chain(delta)
753        .collect::<Vec<_>>();
754    merged.sort_by_key(|event| event.borrow().seq);
755    merged
756}

758/// Display texts shorter than this stay in the event: a span would not be
759/// much smaller.
760const MIN_SPAN_BYTES: usize = 64;
761
762/// A tool result's event with the display texts that repeat its output
763/// left empty, and the [`OutputSpan`]s that name them (client protocol
764/// 1.64.0); `None` when no text repeats it.
765#[must_use]
766pub fn output_spans(event: &StoredEvent) -> Option<(StoredEvent, Vec<OutputSpan>)> {
767    let EventPayload::ToolCallTerminated { termination } = &event.payload else {
768        return None;
769    };
770    let result = termination.result.as_ref()?;
771    let output = result.output.as_str();
772    let found = |text: &str, from: usize| -> Option<(usize, usize)> {
773        if text.len() < MIN_SPAN_BYTES {
774            return None;
775        }
776        let start = from + output.get(from..)?.find(text)?;
777        Some((start, start + text.len()))
778    };
779    let span = |block: usize, segment: Option<usize>, (start, end): (usize, usize)| OutputSpan {
780        seq: event.seq,
781        block: block as u32,
782        segment: segment.map(|segment| segment as u32),
783        start: start as u64,
784        end: end as u64,
785    };
786    let mut display = result.display.clone();
787    let mut spans = Vec::new();
788    // A stream's segments occur in `output` in order.
789    let mut cursors = HashMap::<String, usize>::new();
790    for (index, block) in display.iter_mut().enumerate() {
791        let Open::Known(block) = block else {
792            continue;
793        };
794        match block {
795            DisplayBlock::Text { text, .. }
796            | DisplayBlock::Markdown { text, .. }
797            | DisplayBlock::Code { text, .. } => {
798                if let Some(range) = found(text, 0) {
799                    spans.push(span(index, None, range));
800                    text.clear();
801                }
802            }
803            DisplayBlock::Terminal { segments, .. } => {
804                for (number, segment) in segments.iter_mut().enumerate() {
805                    let cursor = cursors.entry(format!("{:?}", segment.stream)).or_default();
806                    if let Some(range) = found(&segment.text, *cursor) {
807                        *cursor = range.1;
808                        spans.push(span(index, Some(number), range));
809                        segment.text.clear();
810                    }
811                }
812            }
813            DisplayBlock::FileExcerpt {
814                start_line, lines, ..
815            } => {
816                if lines.is_empty() || lines.iter().any(|line| line.contains('\n')) {
817                    continue;
818                }
819                let numbered = crate::stored_display::numbered_lines(*start_line, lines);
820                if let Some(range) = found(&numbered, 0) {
821                    spans.push(span(index, None, range));
822                    lines.clear();
823                }
824            }
825            _ => {}
826        }
827    }
828    if spans.is_empty() {
829        return None;
830    }
831    let mut termination = termination.clone();
832    if let Some(result) = termination.result.as_mut() {
833        result.display = display;
834    }
835    Some((
836        with_payload(event, EventPayload::ToolCallTerminated { termination }),
837        spans,
838    ))
839}

841/// Puts back the display texts [`output_spans`] left empty in `event`, a
842/// `tool_call_terminated` event as JSON, from `spans`, those of its
843/// sequence. Fails when a span is not a range of the output or names a
844/// block or segment the display does not hold.
845pub fn restore_output_spans(
846    event: &mut serde_json::Value,
847    spans: &[OutputSpan],
848) -> Result<(), &'static str> {
849    const NOT_A_RESULT: &str = "an output span names no tool result's display";
850    let result = event
851        .get_mut("payload")
852        .and_then(|payload| payload.get_mut("result"))
853        .and_then(serde_json::Value::as_object_mut)
854        .ok_or(NOT_A_RESULT)?;
855    let output = result
856        .get("output")
857        .and_then(serde_json::Value::as_str)
858        .ok_or(NOT_A_RESULT)?
859        .to_owned();
860    let display = result
861        .get_mut("display")
862        .and_then(serde_json::Value::as_array_mut)
863        .ok_or(NOT_A_RESULT)?;
864    // Each span becomes the session log's encoding of the same text, which
865    // decoding expands.
866    for span in spans {
867        let block = display
868            .get_mut(span.block as usize)
869            .and_then(serde_json::Value::as_object_mut)
870            .ok_or(NOT_A_RESULT)?;
871        let file_excerpt =
872            block.get("type").and_then(serde_json::Value::as_str) == Some("file_excerpt");
873        let target = match span.segment {
874            Some(segment) => block
875                .get_mut("segments")
876                .and_then(|segments| segments.get_mut(segment as usize))
877                .and_then(serde_json::Value::as_object_mut)
878                .ok_or(NOT_A_RESULT)?,
879            None => block,
880        };
881        let (field, member) = if file_excerpt {
882            ("lines", "lines_span")
883        } else {
884            ("text", "text_span")
885        };
886        target.remove(field);
887        target.insert(member.to_owned(), serde_json::json!([span.start, span.end]));
888    }
889    crate::stored_display::decode(&output, display)
890        .map_err(|_| "an output span is not a range of its result's output")
891}

921/// The size of each event as the display view sends it, measured when a
922/// window first needs it.
923struct Sizes<'a> {
924    events: &'a [Arc<StoredEvent>],
925    view: EventsView,
926    measured: Vec<std::cell::Cell<Option<usize>>>,
927}
928
929impl<'a> Sizes<'a> {
930    fn new(events: &'a [Arc<StoredEvent>], view: EventsView) -> Self {
931        Self {
932            events,
933            view,
934            measured: vec![std::cell::Cell::new(None); events.len()],
935        }
936    }
937
938    /// An upper bound of event `index`'s size in the display view: what
939    /// [`display_events`] leaves out of a superseded setting is counted.
940    fn of(&self, index: usize) -> usize {
941        if let Some(size) = self.measured[index].get() {
942            return size;
943        }
944        let event = &self.events[index];
945        let size = match self.view {
946            EventsView::Full => serialized_len(&**event) + 1,
947            EventsView::Display => shown(event).map_or(0, |(event, _)| serialized_len(&*event) + 1),
948        };
949        self.measured[index].set(Some(size));
950        size
951    }
952
953    fn sum(&self, start: usize, end: usize) -> usize {
954        (start..end).map(|index| self.of(index)).sum()
955    }
956
957    /// The smallest index from which the events before `end` take at most
958    /// `budget` bytes; `end` when not even the one before it fits.
959    fn fit_before(&self, end: usize, budget: usize) -> usize {
960        let mut used = 0_usize;
961        let mut start = end;
962        while start > 0 {
963            used = used.saturating_add(self.of(start - 1));
964            if used > budget {
965                break;
966            }
967            start -= 1;
968        }
969        start
970    }
971}

973/// The event at `seq` in a log's events, which are in sequence order.
974#[must_use]
975pub fn event_at(events: &[Arc<StoredEvent>], seq: u64) -> Option<&Arc<StoredEvent>> {
976    events
977        .binary_search_by_key(&seq, |event| event.seq)
978        .ok()
979        .map(|index| &events[index])
980}

994/// The largest index at most `index` that is a char boundary of `text`.
995fn floor_char_boundary(text: &str, index: usize) -> usize {
996    if index >= text.len() {
997        return text.len();
998    }
999    (0..=index)
1000        .rev()
1001        .find(|&boundary| text.is_char_boundary(boundary))
1002        .unwrap_or(0)
1003}

1005/// A value's serialized size, counted without building its JSON.
1006fn serialized_len<T: serde::Serialize + ?Sized>(value: &T) -> usize {
1007    struct Counter(usize);
1008    impl std::io::Write for Counter {
1009        fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
1010            self.0 += bytes.len();
1011            Ok(bytes.len())
1012        }
1013        fn flush(&mut self) -> std::io::Result<()> {
1014            Ok(())
1015        }
1016    }
1017    let mut counter = Counter(0);
1018    match serde_json::to_writer(&mut counter, value) {
1019        Ok(()) => counter.0,
1020        Err(_) => usize::MAX / 4,
1021    }
1022}
