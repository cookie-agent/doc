75pub struct EventOrigin(String);
76impl EventOrigin {
77    pub fn new(value: impl Into<String>) -> Result<Self, EventSchemaError> {
78        let value = value.into();
79        if value == "user" {
80            return Ok(Self(value));
81        }
82        let Some((class, slug)) = value.split_once(':') else {
83            return Err(EventSchemaError::InvalidOrigin);
84        };
85        if !matches!(class, "engine" | "plugin" | "client")
86            || slug.is_empty()
87            || slug.len() > 64
88            || !slug.bytes().enumerate().all(|(index, byte)| {
89                byte.is_ascii_lowercase() || byte.is_ascii_digit() || (index > 0 && byte == b'-')
90            })
91        {
92            return Err(EventSchemaError::InvalidOrigin);
93        }
94        Ok(Self(value))
95    }
96
97    #[must_use]
98    pub fn as_str(&self) -> &str {
99        &self.0
100    }
101
102    #[must_use]
103    pub fn plugin_name(&self) -> Option<&str> {
104        self.0.strip_prefix("plugin:")
105    }
106}

268            || self.prompt.is_empty()
269            || (self.resume_session_id.is_some() && self.inherit_context)
270            || (!self.inherit_context && !self.seeded_context.is_empty())
271            || self.seeded_context.len() > 65_536
272            || self.seeded_context.iter().any(|turn| turn.text.is_empty())
273            || (writer_size_caps()
274                && self
275                    .seeded_context
276                    .iter()
277                    .map(|turn| turn.text.len())

285                || skill.name.is_empty()
286                || skill.rendered_body.is_empty()
287                || skill.source_path.is_empty()
288                || skill.base_dir.is_empty()
289                || skill.supporting_files.len() > 10
290                || skill.grants.len() > 256

693/// A tool result as sessions store it and clients receive it. `display` is
694/// what clients show, a list of display blocks sharing one 64 KiB budget;
695/// empty means nothing to show. A session log stores the display's text that
696/// repeats `output` as byte ranges of `output` instead of a second copy
697/// (`crate::stored_display`); decoding expands them, so nothing outside the
698/// engine meets them.
699#[derive(Clone, Debug, JsonSchema, PartialEq)]
700pub struct PersistedToolResult {
701    pub title: SafeDisplayText,
702    pub output: String,
703    pub display: Vec<Open<DisplayBlock>>,
704    /// Present on the result a call committed when it moved to the
705    /// background; the daemon alone sets it.
706    #[serde(default, skip_serializing_if = "Option::is_none")]
707    pub background: Option<ToolBackground>,
708    #[serde(default, skip_serializing_if = "Option::is_none")]
709    pub retained_output: Option<crate::RetainedToolOutput>,
710    pub metadata: Value,
711    #[schemars(
712        description = "Direct result files. Together with files in additional_messages, at most 256 are allowed."
713    )]
714    #[schemars(length(max = 256))]
715    pub attachments: Vec<ToolAttachment>,
716    #[schemars(
717        description = "Emitted model-history messages. Their files share the 256-file limit with attachments."
718    )]
719    #[serde(default, skip_serializing_if = "Vec::is_empty")]
720    #[schemars(length(max = 4))]
721    pub additional_messages: Vec<ToolEmittedMessage>,
722    /// Deferred tools this result loads: they become callable from it on
723    /// (client protocol 1.44.0). Only `tool_load` sets it.
724    #[serde(default, skip_serializing_if = "Vec::is_empty")]
725    #[schemars(length(max = 1024))]
726    pub loaded_tools: Vec<String>,
727}
728impl PersistedToolResult {
729    pub const MAX_OUTPUT_BYTES: usize = 2 * 1024 * 1024;
730    pub const MAX_METADATA_BYTES: usize = 2 * 1024 * 1024;
731    pub const MAX_ADDITIONAL_MESSAGES: usize = 4;
732    pub const MAX_ATTACHMENTS: usize = 256;
733
734    pub fn all_attachments(&self) -> impl Iterator<Item = &ToolAttachment> {
735        self.attachments.iter().chain(
736            self.additional_messages
737                .iter()
738                .flat_map(|message| message.content.iter())
739                .filter_map(|part| match part {
740                    ToolEmittedContent::Text(_) => None,
741                    ToolEmittedContent::File(attachment) => Some(attachment),
742                }),
743        )
744    }
745
746    pub fn validate(&self) -> Result<(), EventSchemaError> {
747        if writer_size_caps() && self.output.len() > Self::MAX_OUTPUT_BYTES {
748            return Err(EventSchemaError::ToolOutputTooLarge);
749        }
750        match crate::display::validate_display(&self.display) {
751            Ok(()) => {}
752            Err(crate::DisplayBlockError::TooLarge) if !writer_size_caps() => {}
753            Err(crate::DisplayBlockError::TooLarge) => {
754                return Err(EventSchemaError::ToolOutputTooLarge);
755            }
756            Err(_) => return Err(EventSchemaError::InvalidDisplay),
757        }
758        if let Some(retained) = &self.retained_output {
759            retained.validate()?;
760        }
761        if self.background.as_ref().is_some_and(|background| {
762            background.job.as_ref().is_some_and(|job| {
763                job.is_empty() || job.len() > 64 || job.chars().any(char::is_control)
764            }) || match background.state {
765                ToolBackgroundState::Job => background.job.is_none(),
766                ToolBackgroundState::SelfReporting => background.job.is_some(),
767                ToolBackgroundState::Unknown => false,
768            }
769        }) {
770            return Err(EventSchemaError::InvalidToolBackground);
771        }
772        if writer_size_caps()
773            && json_len_exceeds(&self.metadata, Self::MAX_METADATA_BYTES)
774                .map_err(|_| EventSchemaError::InvalidJson)?
775        {
776            return Err(EventSchemaError::ToolMetadataTooLarge);
777        }
778        let emitted_attachment_count = self
779            .additional_messages
780            .iter()
781            .flat_map(|message| &message.content)
782            .filter(|part| matches!(part, ToolEmittedContent::File(_)))
783            .count();
784        if self
785            .attachments
786            .len()
787            .saturating_add(emitted_attachment_count)
788            > Self::MAX_ATTACHMENTS
789        {
790            return Err(EventSchemaError::TooManyAttachments);
791        }
792        if self.additional_messages.len() > Self::MAX_ADDITIONAL_MESSAGES {
793            return Err(EventSchemaError::TooManyToolEmittedMessages);
794        }
795        if !valid_model_tool_names(&self.loaded_tools) {
796            return Err(EventSchemaError::InvalidModelTools);
797        }
798        for message in &self.additional_messages {
799            message.validate()?;
800        }
801        for attachment in self.all_attachments() {
802            attachment.validate()?;
803        }
804        Ok(())
805    }
806}

1342/// Runs `read` as a reader of stored history (`docs/site/specs/compatibility-v1.md`
1343/// section 8.5): the size caps a writer keeps to are not applied, so a
1344/// record reads as its writer bounded it. They are a tool result's output,
1345/// metadata, and display budget, a model turn's serialized size, native
1346/// replay payloads, and a diagnostic's detail. Every other check runs.
1347pub fn read_stored_history<T>(read: impl FnOnce() -> T) -> T {
1348    struct Restore(bool);
1349    impl Drop for Restore {
1350        fn drop(&mut self) {
1351            READING_HISTORY.with(|reading| reading.set(self.0));
1352        }
1353    }
1354    let _restore = Restore(READING_HISTORY.with(|reading| reading.replace(true)));
1355    read()
1356}

1362/// Runs `write` as the writer of a session log: values serialize in the

1364/// display text that repeats its output as ranges of it
1365/// (`docs/site/specs/compatibility-v1.md` section 12.3); every other value
1366/// serializes as it does anywhere else, and decoding reads either form.

1369    impl Drop for Restore {
1370        fn drop(&mut self) {

1372        }
1373    }
1374    let _restore = Restore(WRITING_HISTORY.with(|writing| writing.replace(true)));
1375    write()
1376}

1390/// Counts serialized bytes instead of buffering them, optionally failing
1391/// once a limit is passed so an oversized value stops serializing early.
1392struct ByteCounter {
1393    count: usize,
1394    limit: usize,
1395}
1396
1397impl std::io::Write for ByteCounter {
1398    fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
1399        self.count = self.count.saturating_add(bytes.len());
1400        if self.count > self.limit {
1401            return Err(std::io::Error::other("serialized value exceeds its limit"));
1402        }
1403        Ok(bytes.len())
1404    }
1405
1406    fn flush(&mut self) -> std::io::Result<()> {
1407        Ok(())
1408    }
1409}

1411/// The length of `value` serialized as JSON, without buffering it.
1412fn json_len<T: Serialize + ?Sized>(value: &T) -> Result<usize, serde_json::Error> {
1413    let mut counter = ByteCounter {
1414        count: 0,
1415        limit: usize::MAX,
1416    };
1417    serde_json::to_writer(&mut counter, value)?;
1418    Ok(counter.count)
1419}
1420
1421/// Whether `value` serializes to more than `maximum` bytes of JSON. Nothing
1422/// is buffered, and serialization stops as soon as the limit is passed.
1423pub(crate) fn json_len_exceeds<T: Serialize + ?Sized>(
1424    value: &T,
1425    maximum: usize,
1426) -> Result<bool, serde_json::Error> {
1427    let mut counter = ByteCounter {
1428        count: 0,
1429        limit: maximum,
1430    };
1431    match serde_json::to_writer(&mut counter, value) {
1432        Ok(()) => Ok(false),
1433        Err(_) if counter.count > maximum => Ok(true),
1434        Err(error) => Err(error),
1435    }
1436}

1524        if object.get("additionalProperties") == Some(&Value::Bool(false)) {
1525            object.remove("additionalProperties");

1527        object.values_mut().for_each(open);

2600/// The latest runtime revisions a log records, folded in log order.
2601///
2602/// `session_created` records them, and a later `run_started` or
2603/// `delegation_reserved` records them again only when they changed, so an
2604/// event without its own revisions was frozen against the latest before it.
2605#[derive(Clone, Debug, Default, Eq, PartialEq)]
2606pub struct RevisionFold {
2607    latest: Option<RuntimeRevisions>,
2608}
2609
2610impl RevisionFold {
2611    /// Folds one event and returns the revisions it was frozen against, for
2612    /// the events that record them.
2613    pub fn observe(&mut self, payload: &EventPayload) -> Option<&RuntimeRevisions> {
2614        match payload {
2615            EventPayload::SessionCreated { revisions, .. } => {
2616                self.latest = Some(revisions.clone());
2617            }
2618            EventPayload::RunStarted { revisions, .. }
2619            | EventPayload::DelegationReserved { revisions, .. } => {
2620                if let Some(revisions) = revisions {
2621                    self.latest = Some(revisions.clone());
2622                }
2623            }
2624            _ => return None,
2625        }
2626        self.latest.as_ref()
2627    }
2628
2629    /// The latest revisions folded so far.
2630    #[must_use]
2631    pub const fn latest(&self) -> Option<&RuntimeRevisions> {
2632        self.latest.as_ref()
2633    }
2634
2635    /// Drops `payload`'s own revisions when they equal the latest folded,
2636    /// which is how a writer records them only where they changed.
2637    pub fn elide_unchanged(&self, payload: &mut EventPayload) {
2638        if let EventPayload::RunStarted { revisions, .. }
2639        | EventPayload::DelegationReserved { revisions, .. } = payload
2640            && revisions.is_some()
2641            && revisions.as_ref() == self.latest.as_ref()
2642        {
2643            *revisions = None;
2644        }
2645    }
2646}

4612/// Decodes an event payload, replacing only leaves which Serde identifies as
4613/// nullable fields. Required-field and unknown-variant failures remain errors.
4614/// The original event log is not rewritten.
4615///
4616/// A payload that decodes strictly is read in place; only a payload that
4617/// needs repair is copied. Each repair locates its leaf first and then decodes
4618/// the payload once more, so repairing stays linear in the payload's size:
4619/// `serde_path_to_error` names the leaf where it can, and where serde buffered
4620/// the value first (inside an internally tagged enum, which every payload is)
4621/// the leaf is found by the value serde's error names.
4622pub fn deserialize_event_payload_best_effort(value: &Value) -> Result<EventPayloadRead, String> {
4623    if let Ok(payload) = EventPayload::deserialize(value) {
4624        return Ok(EventPayloadRead {
4625            payload,
4626            degraded_fields: Vec::new(),
4627        });
4628    }
4629    let mut degraded_fields = Vec::new();
4630    let mut value = value.clone();
4631    let mut previous_path = None;
4632    for _ in 0..=MAX_DEGRADED_FIELDS {
4633        count_degradation_decode();
4634        let error = match serde_path_to_error::deserialize::<_, EventPayload>(&value) {
4635            Ok(payload) => {
4636                return Ok(EventPayloadRead {
4637                    payload,
4638                    degraded_fields,
4639                });
4640            }
4641            Err(error) => error,
4642        };
4643        let path = error.path().to_string();
4644        if path != "." && previous_path.as_deref() == Some(path.as_str()) {
4645            return Err(error.to_string());
4646        }
4647        let degraded = if path == "." {
4648            replace_named_leaf(&mut value, &error.inner().to_string())
4649        } else if replace_path_with_null(&mut value, error.path()) {
4650            Some(path.clone())
4651        } else {
4652            None
4653        };
4654        let Some(degraded) = degraded else {
4655            return Err(error.to_string());
4656        };
4657        previous_path = Some(path);
4658        degraded_fields.push(degraded);
4659    }
4660    Err(format!(
4661        "event payload exceeds the {MAX_DEGRADED_FIELDS}-field degradation limit"
4662    ))
4663}

4686/// What serde's error names as the value it could not take.
4687enum NamedValue {
4688    /// `invalid type` or `invalid value`: a leaf equal to this one.
4689    Leaf(Value),
4690    /// `invalid type` for a sequence or a map: any array or object.
4691    Array,
4692    Object,
4693    /// `invalid length N`: an array of that length.
4694    Length(usize),
4695    /// `missing field`: an object without it, which takes a null.
4696    Missing(String),
4697}

4699impl NamedValue {
4700    /// Reads serde's standard messages, which spell the value out with
4701    /// [`serde::de::Unexpected`]; a custom message names none.
4702    fn parse(message: &str) -> Option<Self> {
4703        if let Some(rest) = message.strip_prefix("missing field `") {
4704            return rest
4705                .split('`')
4706                .next()
4707                .map(|field| Self::Missing(field.to_owned()));
4708        }
4709        if let Some(rest) = message.strip_prefix("unknown variant `") {
4710            return rest
4711                .split('`')
4712                .next()
4713                .map(|variant| Self::Leaf(Value::String(variant.to_owned())));
4714        }
4715        if let Some(rest) = message.strip_prefix("invalid length ") {
4716            return rest
4717                .split(',')
4718                .next()
4719                .and_then(|length| length.trim().parse().ok())
4720                .map(Self::Length);
4721        }
4722        let unexpected = message
4723            .strip_prefix("invalid type: ")
4724            .or_else(|| message.strip_prefix("invalid value: "))?;
4725        let quoted = |prefix: &str| {
4726            unexpected
4727                .strip_prefix(prefix)
4728                .and_then(|rest| rest.split('`').next())
4729        };
4730        if let Some(rest) = unexpected.strip_prefix("string ") {
4731            // The string as `Debug` spells it; decoding it as JSON covers the
4732            // escapes the two share, which is every one serde emits for
4733            // stored text.
4734            let end = rest.rfind("\", expected").or_else(|| rest.rfind('"'))?;
4735            return serde_json::from_str::<String>(&rest[..=end])
4736                .ok()
4737                .map(|text| Self::Leaf(Value::String(text)));
4738        }
4739        if let Some(number) = quoted("integer `") {
4740            return serde_json::from_str::<serde_json::Number>(number)
4741                .ok()
4742                .map(|number| Self::Leaf(Value::Number(number)));
4743        }
4744        if let Some(number) = quoted("floating point `") {
4745            return number
4746                .parse::<f64>()
4747                .ok()
4748                .and_then(serde_json::Number::from_f64)
4749                .map(|number| Self::Leaf(Value::Number(number)));
4750        }
4751        if let Some(boolean) = quoted("boolean `") {
4752            return boolean
4753                .parse()
4754                .ok()
4755                .map(|boolean| Self::Leaf(Value::Bool(boolean)));
4756        }
4757        if unexpected.starts_with("sequence") {
4758            return Some(Self::Array);
4759        }
4760        if unexpected.starts_with("map") {
4761            return Some(Self::Object);
4762        }
4763        None
4764    }

4766    fn matches(&self, value: &Value) -> bool {
4767        match (self, value) {
4768            (Self::Leaf(Value::Number(named)), Value::Number(number)) => {
4769                named == number
4770                    || named
4771                        .as_f64()
4772                        .is_some_and(|named| Some(named) == number.as_f64())
4773            }
4774            (Self::Leaf(named), value) => named == value,
4775            (Self::Array, Value::Array(_)) | (Self::Object, Value::Object(_)) => true,
4776            (Self::Length(length), Value::Array(items)) => items.len() == *length,
4777            (Self::Missing(field), Value::Object(members)) => !members.contains_key(field),
4778            _ => false,
4779        }
4780    }

4797            continue;

4805            Err(error)

4818/// The paths of the values `named` matches, never the payload's tag, at most
4819/// [`MAX_LEAF_CANDIDATES`] of them.
4820fn collect_named_paths(
4821    value: &Value,
4822    named: &NamedValue,
4823    current: &mut Vec<ValuePathSegment>,
4824    paths: &mut Vec<Vec<ValuePathSegment>>,
4825) {
4826    if paths.len() >= MAX_LEAF_CANDIDATES {
4827        return;
4828    }
4829    if !current.is_empty() && named.matches(value) {
4830        paths.push(current.clone());
4831    } else if current.is_empty() && matches!(named, NamedValue::Missing(_)) && named.matches(value)
4832    {
4833        paths.push(Vec::new());
4834    }
4835    match value {
4836        Value::Object(object) => {
4837            for (key, child) in object {
4838                if current.is_empty() && key == "type" {
4839                    continue;
4840                }
4841                current.push(ValuePathSegment::Key(key.clone()));
4842                collect_named_paths(child, named, current, paths);
4843                current.pop();
4844            }
4845        }
4846        Value::Array(array) => {
4847            for (index, child) in array.iter().enumerate() {
4848                current.push(ValuePathSegment::Index(index));
4849                collect_named_paths(child, named, current, paths);
4850                current.pop();
4851            }
4852        }
4853        _ => {}
4854    }

4877            .as_array_mut()

5024        if let Some(reference) = &self.reference {

5038                crate::validate_tool_stream_name(name)
5039                    .map_err(|_| EventSchemaError::InvalidArtifactReference)?;
5040            }
5041        }

5199impl crate::DisplayAudience {
5200    /// What the client receives of `payload`'s display blocks: the call's
5201    /// presentation, its result's display, and a progress `replace`. `None`
5202    /// when that is `payload` unchanged.
5203    #[must_use]
5204    pub fn payload(&self, payload: &EventPayload) -> Option<EventPayload> {
5205        match payload {
5206            EventPayload::ToolCallStarted { start } => {
5207                let arguments = self.blocks(&start.presentation.arguments);
5208                let preview = self.blocks(&start.presentation.preview);
5209                if arguments.is_none() && preview.is_none() {
5210                    return None;
5211                }
5212                let mut start = start.clone();
5213                if let Some(arguments) = arguments {
5214                    start.presentation.arguments = arguments;
5215                }
5216                if let Some(preview) = preview {
5217                    start.presentation.preview = preview;
5218                }
5219                Some(EventPayload::ToolCallStarted { start })
5220            }
5221            EventPayload::ToolCallTerminated { termination } => {
5222                let display = self.blocks(&termination.result.as_ref()?.display)?;
5223                let mut termination = termination.clone();
5224                if let Some(result) = termination.result.as_mut() {
5225                    result.display = display;
5226                }
5227                Some(EventPayload::ToolCallTerminated { termination })
5228            }
5229            EventPayload::ToolCallProgress {
5230                tool_call_id,
5231                message,
5232                display: Some(Open::Known(crate::DisplayDelta::Replace { blocks })),
5233            } => Some(EventPayload::ToolCallProgress {
5234                tool_call_id: *tool_call_id,
5235                message: message.clone(),
5236                display: Some(Open::Known(crate::DisplayDelta::Replace {
5237                    blocks: self.blocks(blocks)?,
5238                })),
5239            }),
5240            _ => None,
5241        }
5242    }

5248        match self.payload(&event.payload) {

5250            Some(payload) => std::sync::Arc::new(StoredEvent {
5251                payload,

5254        }
5255    }

5259    pub fn message(&self, message: EventSubscriptionMessage) -> EventSubscriptionMessage {
