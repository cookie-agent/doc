1//! The queued-input lane, as every reader of a session log reduces it.
2//!
3//! A queued input waits for the running run to complete. Once that happens,
4//! the engine starts a run with it, and that run's `run_started` carries the
5//! entry's `client_run_id`. The queue empties on any failed, cancelled, or
6//! interrupted run, and on `user_input_queue_voided`. Voided entries go back
7//! to their authors, who restore the text for editing.
8
9use std::borrow::Borrow;
10
11use crate::{ClientRunId, EventOrigin, EventPayload, RunSelection, StoredEvent, UserContent};
12
13/// One input waiting in a session's queue.
14#[derive(Clone, Debug, Eq, PartialEq)]
15pub struct QueuedInput {
16    /// The seq of its `user_input_queued` event.
17    pub queued_seq: u64,
18    /// Who queued it; absent only on records written without an origin.
19    pub origin: Option<EventOrigin>,
20    pub content: UserContent,
21    pub selection: RunSelection,
22    pub client_run_id: ClientRunId,
23    pub queued_at: jiff::Timestamp,
24}
25
26/// A session's queued inputs, oldest first.
27#[derive(Clone, Debug, Default, Eq, PartialEq)]
28pub struct QueuedInputs {
29    entries: Vec<QueuedInput>,
30}
31
32impl QueuedInputs {
33    /// The queue after a session's physical `events`, in order.
34    #[must_use]
35    pub fn from_events<E: Borrow<StoredEvent>>(events: &[E]) -> Self {
36        let mut queue = Self::default();
37        for event in events {
38            queue.apply(event.borrow());
39        }
40        queue
41    }
42
43    /// Folds one event and returns the entries it voided, oldest first.
44    pub fn apply(&mut self, event: &StoredEvent) -> Vec<QueuedInput> {
45        self.apply_payload(
46            event.seq,
47            event.timestamp,
48            event.origin.as_ref(),
49            &event.payload,
50        )
51    }
52
53    /// [`Self::apply`] for a reader that holds an event's parts.
54    pub fn apply_payload(
55        &mut self,
56        seq: u64,
57        timestamp: jiff::Timestamp,
58        origin: Option<&EventOrigin>,
59        payload: &EventPayload,
60    ) -> Vec<QueuedInput> {
61        match payload {
62            EventPayload::UserInputQueued {
63                content,
64                selection,
65                client_run_id,
66            } => self.entries.push(QueuedInput {
67                queued_seq: seq,
68                origin: origin.cloned(),
69                content: content.clone(),
70                selection: selection.clone(),
71                client_run_id: client_run_id.clone(),
72                queued_at: timestamp,
73            }),
74            EventPayload::RunStarted { client_run_id, .. } => {
75                self.entries
76                    .retain(|entry| &entry.client_run_id != client_run_id);
77            }
78            EventPayload::UserInputQueueRecalled { queued_seq } => {
79                self.entries.retain(|entry| entry.queued_seq != *queued_seq);
80            }
81            EventPayload::RunFinished { outcome }
82                if outcome.status() != crate::SessionStatus::Completed =>
83            {
84                return std::mem::take(&mut self.entries);
85            }
86            EventPayload::UserInputQueueVoided {} => {
87                return std::mem::take(&mut self.entries);
88            }
89            _ => {}
90        }
91        Vec::new()
92    }
93
94    #[must_use]
95    pub fn entries(&self) -> &[QueuedInput] {
96        &self.entries
97    }
98
99    #[must_use]
100    pub fn is_empty(&self) -> bool {
101        self.entries.is_empty()
102    }
103
104    /// The oldest entry: the next run the queue starts.
105    #[must_use]
106    pub fn head(&self) -> Option<&QueuedInput> {
107        self.entries.first()
108    }
109
110    #[must_use]
111    pub fn get(&self, queued_seq: u64) -> Option<&QueuedInput> {
112        self.entries
113            .iter()
114            .find(|entry| entry.queued_seq == queued_seq)
115    }
116}

118#[cfg(test)]
119mod tests {
120    use super::QueuedInputs;
121    use crate::{
122        AgentId, ClientRunId, EventOrigin, EventPayload, ModelSelection, RunId, RunSelection,

126    fn selection() -> RunSelection {
127        RunSelection {
128            agent: AgentId::new("primary").unwrap(),
129            model: ModelSelection {
130                model: "custom.test/model".parse().unwrap(),
131                variant: None,
132            },
133            preset: None,
134        }
135    }
136
137    fn event(seq: u64, origin: &str, payload: EventPayload) -> StoredEvent {
138        StoredEvent {
139            engine_version: None,
140            origin: Some(EventOrigin::new(origin).unwrap()),
141            session_id: SessionId::new_v7(),
142            run_id: None,
143            seq,
144            timestamp: jiff::Timestamp::new(seq as i64, 0).unwrap(),
145            payload,
146        }
147    }
148
149    fn queued(seq: u64, origin: &str, input: &str) -> StoredEvent {
150        event(
151            seq,
152            origin,
153            EventPayload::UserInputQueued {
154                content: crate::UserContent::text(input),
155                selection: selection(),
156                client_run_id: ClientRunId::new(format!("queued-{seq}")).unwrap(),
157            },
158        )
159    }

161    #[test]
162    fn recall_removes_only_its_entry_and_the_head_keeps_its_author() {
163        let mut queue = QueuedInputs::default();
164        for entry in [
165            queued(1, "client:tui-a", "first"),
166            queued(2, "client:tui-b", "second"),
167            queued(3, "client:tui-a", "third"),
168        ] {
169            assert!(queue.apply(&entry).is_empty());
170        }
171        let recalled = event(
172            4,
173            "client:tui-a",
174            EventPayload::UserInputQueueRecalled { queued_seq: 2 },
175        );
176        assert!(queue.apply(&recalled).is_empty());
177        assert_eq!(
178            queue
179                .entries()
180                .iter()
181                .map(|entry| entry.content.as_plain_text().into_owned())
182                .collect::<Vec<_>>(),
183            ["first", "third"]
184        );
185        assert_eq!(
186            queue.head().and_then(|entry| entry.origin.clone()),
187            Some(EventOrigin::new("client:tui-a").unwrap())
188        );
189        assert_eq!(queue.get(3).map(|entry| entry.queued_seq), Some(3));
190    }
191
192    #[test]
193    fn a_failed_cancelled_or_interrupted_run_voids_the_whole_queue() {
194        for terminal in [
195            EventPayload::run_finished(crate::RunOutcome::Cancelled { reason: None }),
196            EventPayload::run_finished(crate::RunOutcome::Interrupted { reason: None }),
197            EventPayload::UserInputQueueVoided {},
198        ] {
199            let mut queue = QueuedInputs::default();
200            queue.apply(&queued(1, "client:tui-a", "first"));
201            queue.apply(&queued(2, "client:tui-b", "second"));
202            let mut terminal = event(3, "engine:model-loop", terminal);
203            terminal.run_id = Some(RunId::new_v7());
204            let voided = queue.apply(&terminal);
205            assert_eq!(
206                voided
207                    .iter()
208                    .map(|entry| entry.content.as_plain_text().into_owned())
209                    .collect::<Vec<_>>(),
210                ["first", "second"]
211            );
212            assert!(queue.is_empty());
213        }
214    }

216    #[test]
217    fn completion_leaves_the_queue_for_the_engine_to_drain() {
218        let mut queue = QueuedInputs::default();
219        queue.apply(&queued(1, "client:tui-a", "first"));
220        let mut completed = event(
221            2,
222            "engine:model-loop",
223            EventPayload::run_finished(crate::RunOutcome::Completed { final_text: None }),
224        );
225        completed.run_id = Some(RunId::new_v7());
226        assert!(queue.apply(&completed).is_empty());
227        assert_eq!(queue.entries().len(), 1);
228    }
