1//! Cron jobs (`docs/site/specs/cron.md`): prompts the daemon fires on a
2//! schedule, into an existing session or into a new session per fire.
3//! Client protocol 1.15.0.
4//!
5//! The schedule, the target, and the actor are closed by design: a new kind
6//! of schedule or target is a new kind of job, which an older client could
7//! neither show nor edit. What a fire did and why a job is disabled are
8//! open: an older client shows a value it does not know as such.
9

57/// When a job fires. The daemon stores and returns `cron` and `at`; `in`
58/// is accepted on create and update only and becomes `at` at once.
59#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
60#[serde(rename_all = "snake_case")]
61#[schemars(transform = crate::event::open_object_schema)]
62pub enum CronSchedule {
63    /// A five-field cron expression (minute, hour, day of month, month, day
64    /// of week) in the daemon host's local time; day of month and day of
65    /// week combine with OR when both are restricted, as in Vixie cron.
66    Cron(String),
67    /// Once, this long from now: a whole number with `s`, `m`, `h`, or `d`,
68    /// from `1s` to `365d`.
69    In(String),
70    /// Once, at this time.
71    At(Timestamp),
72}

74/// How many fires a cron schedule delivers before the job finishes: a whole
75/// number from 1, or `"unlimited"`.
76#[derive(Clone, Copy, Debug, Eq, PartialEq)]
77pub enum CronMaxRuns {
78    Count(u32),
79    Unlimited,
80}
81
82impl CronMaxRuns {
83    const UNLIMITED: &'static str = "unlimited";
84
85    /// Whether `runs` fires reach this limit.
86    #[must_use]
87    pub fn reached(self, runs: u64) -> bool {
88        match self {
89            Self::Count(count) => runs >= u64::from(count),
90            Self::Unlimited => false,
91        }
92    }
93}
94
95impl Default for CronMaxRuns {
96    fn default() -> Self {
97        Self::Count(1)
98    }
99}
100
101impl fmt::Display for CronMaxRuns {
102    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
103        match self {
104            Self::Count(count) => count.fmt(formatter),
105            Self::Unlimited => formatter.write_str(Self::UNLIMITED),
106        }
107    }
108}
109
110impl Serialize for CronMaxRuns {
111    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
112        match self {
113            Self::Count(count) => serializer.serialize_u32(*count),
114            Self::Unlimited => serializer.serialize_str(Self::UNLIMITED),
115        }
116    }
117}
118
119impl<'de> Deserialize<'de> for CronMaxRuns {
120    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
121        #[derive(Deserialize)]
122        #[serde(untagged)]
123        enum Wire {
124            Count(u64),
125            Text(String),
126        }
127        match Wire::deserialize(deserializer)? {
128            Wire::Count(count) => u32::try_from(count)
129                .ok()
130                .filter(|count| *count >= 1)
131                .map(Self::Count)
132                .ok_or_else(|| serde::de::Error::custom("max_runs must be from 1 to 4294967295")),
133            Wire::Text(text) if text == Self::UNLIMITED => Ok(Self::Unlimited),
134            Wire::Text(_) => Err(serde::de::Error::custom(
135                "max_runs must be a whole number from 1 or \"unlimited\"",
136            )),
137        }
138    }
139}

160/// Where each fire goes.
161#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
162#[serde(rename_all = "snake_case")]
163#[schemars(transform = crate::event::open_object_schema)]
164pub enum CronTarget {
165    /// A root session.
166    Session(SessionId),
167    /// A delegated session.
168    Subagent(SessionId),
169    /// A new root session per fire, from this template.
170    New(CronTemplate),
171}
172
173impl CronTarget {
174    /// The target kind, which is the resource of the `cron` permission.
175    #[must_use]
176    pub const fn kind(&self) -> CronTargetKind {
177        match self {
178            Self::Session(_) => CronTargetKind::Session,
179            Self::Subagent(_) => CronTargetKind::Subagent,
180            Self::New(_) => CronTargetKind::New,
181        }
182    }
183
184    /// The session an in-session job fires into.
185    #[must_use]
186    pub const fn session(&self) -> Option<SessionId> {
187        match self {
188            Self::Session(session) | Self::Subagent(session) => Some(*session),
189            Self::New(_) => None,
190        }
191    }
192}

220/// What a new-session job's sessions start with. `runtimes`, `global`, and
221/// `cwd` have the shape of `session.create`'s members of the same names.
222#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
223pub struct CronTemplate {
224    /// The agent each run session starts with; it must be able to run as a
225    /// root (`primary` or `all`).
226    pub agent: AgentId,
227    #[serde(
228        default,
229        deserialize_with = "deserialize_optional_preset",
230        skip_serializing_if = "Option::is_none"
231    )]
232    #[schemars(with = "Option<AgentPresetNameSchema>")]
233    pub preset: Option<String>,

241    pub permission_rules: Vec<PermissionRule>,
242    /// The runtimes a run session attaches besides `global`, in order, as
243    /// `session.create`'s `runtimes` names them; as `session.create`
244    /// without `runtimes` when omitted.
245    #[serde(default, skip_serializing_if = "Option::is_none")]
246    #[schemars(length(max = 256))]
247    pub runtimes: Option<Vec<ProjectRuntime>>,
248    /// Whether a run session attaches the global runtime; `true` when
249    /// omitted.
250    #[serde(default = "attach_global", skip_serializing_if = "attaches_global")]
251    pub global: bool,
252    #[serde(default, skip_serializing_if = "Option::is_none")]
253    #[schemars(length(min = 1, max = 4096))]
254    pub cwd: Option<String>,
255    /// The project the run sessions belong to.
256    #[serde(default, skip_serializing_if = "Option::is_none")]
257    pub project: Option<ProjectId>,
258}

276/// Why a job is disabled.
277#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
278#[serde(tag = "kind", rename_all = "snake_case")]
279pub enum CronDisabledReason {
280    /// It ran `max_runs` times, or its one-shot time passed.
281    Finished,
282    /// It failed this many times in a row.
283    Failed { failures: u32 },
284    /// You or an agent disabled it.
285    Disabled { by: CronActor },
286}
287
288crate::open_enum!(tagged CronDisabledReason, tag = "kind", {
289    Finished => "finished",
290    Failed => "failed",
291    Disabled => "disabled",
292});

294crate::open_enum! {
295    /// Why a fire was skipped. A value this build does not know is still a
296    /// skip.
297    #[derive(Clone, Copy, Debug, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
298    #[serde(rename_all = "snake_case")]
299    pub enum CronSkipReason {
300        /// The job's previous fire into its session was still waiting.
301        PreviousFireNotDelivered,
302        /// The run session of the job's previous fire was still running.
303        PreviousRunStillRunning,
304        /// The daemon was not running at the fire time; `missed` counts the
305        /// fires missed.
306        DaemonWasDown,
307        /// The daemon stopped after recording the fire and before
308        /// delivering it.
309        Interrupted,
310        /// `[cron] enabled = false` held at the fire time.
311        FiringDisabled,
312        /// The daemon was waiting to restart (client protocol 1.19.0).
313        DaemonRestarting,
314        /// The target session's automatic runs were held (client protocol
315        /// 1.34.0; `docs/site/specs/held-sessions.md`).
316        SessionHeld,
317    }
318}

352/// One entry of a job's history.
353#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]

417    pub history: Vec<CronFire>,
