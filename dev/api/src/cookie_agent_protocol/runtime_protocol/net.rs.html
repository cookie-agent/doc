1//! The `net/*` messages of forwards (runtime protocol 1.19.0,
2//! `docs/site/specs/forwards.md` section 4). The daemon asks a runtime that
3//! reported `net` in its `runtime/initialize` result to listen on a port or
4//! connect to one, and relays the bytes between two runtimes, or between a
5//! runtime and a browser, over each runtime's connection.
6//!
7//! Listeners and connections are numbers the runtime chooses. Bytes travel
8//! as base64 in `net/data`, at most [`MAX_NET_DATA_BYTES`] at a time, and
9//! each side has at most [`NET_WINDOW_BYTES`] sent on a connection and not
10//! yet acknowledged with `net/ack`: it stops reading its socket until acks
11//! arrive, so a slow end slows the other.
12
13use serde::{Deserialize, Serialize};
14
15/// The most bytes one `net/data` or `net/datagram` carries.
16pub const MAX_NET_DATA_BYTES: usize = 64 * 1024;
17/// The most bytes a side has sent on a connection and not had
18/// acknowledged.
19pub const NET_WINDOW_BYTES: u64 = 1024 * 1024;
20
21/// A transport a forward carries.
22#[derive(Clone, Copy, Debug, Deserialize, Eq, Hash, PartialEq, Serialize, schemars::JsonSchema)]
23#[serde(rename_all = "snake_case")]
24pub enum NetProtocol {
25    Tcp,
26    Udp,
27}
28
29impl NetProtocol {
30    #[must_use]
31    pub const fn as_str(self) -> &'static str {
32        match self {
33            Self::Tcp => "tcp",
34            Self::Udp => "udp",
35        }
36    }
37}
38
39/// `net/listen`: listen on `bind:port` of the runtime's machine.
40#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
41pub struct NetListenParams {
42    pub protocol: NetProtocol,
43    /// An address of the machine, such as `127.0.0.1`.
44    pub bind: String,
45    /// `0` picks a free port.
46    pub port: u16,
47}
48
49#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
50pub struct NetListenResult {
51    pub listener: u64,
52    /// The port it listens on.
53    pub port: u16,
54}
55
56/// `net/unlisten`: stop listening. Connections it accepted stay until
57/// closed.
58#[derive(Clone, Copy, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
59pub struct NetListenerRef {
60    pub listener: u64,
61}
62
63/// `net/accepted` (runtime → daemon): a TCP listener accepted a
64/// connection.
65#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
66pub struct NetAcceptedParams {
67    pub listener: u64,
68    pub connection: u64,
69    /// The peer's address, `host:port`.
70    pub peer: String,
71}
72
73/// `net/connect`: connect to `host:port` from the runtime's machine.
74#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
75pub struct NetConnectParams {
76    pub protocol: NetProtocol,
77    pub host: String,
78    pub port: u16,
79}
80
81#[derive(Clone, Copy, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
82pub struct NetConnectResult {
83    pub connection: u64,
84}
85
86/// `net/data` (either direction): bytes of a TCP connection, in order.
87#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
88pub struct NetDataParams {
89    pub connection: u64,
90    /// Base64, at most [`MAX_NET_DATA_BYTES`] decoded.
91    pub data: String,
92}
93
94/// `net/ack` (either direction): bytes of `net/data` the receiver has
95/// written to its socket, which the sender may send again.
96#[derive(Clone, Copy, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
97pub struct NetAckParams {
98    pub connection: u64,
99    pub bytes: u64,
100}
101
102/// `net/close` (either direction): a connection's end. `eof` ends only the
103/// sender's direction; anything else ends the connection.
104#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
105pub struct NetCloseParams {
106    pub connection: u64,
107    #[serde(default, skip_serializing_if = "Option::is_none")]
108    pub reason: Option<String>,
109}
110
111/// What [`NetCloseParams::reason`] says when the sender only stopped
112/// writing.
113pub const NET_EOF: &str = "eof";
114
115/// `net/datagram` (either direction): one UDP datagram, of a UDP listener
116/// and its `peer`, or of a UDP connection.
117#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize, schemars::JsonSchema)]
118pub struct NetDatagramParams {
119    #[serde(default, skip_serializing_if = "Option::is_none")]
120    pub listener: Option<u64>,
121    #[serde(default, skip_serializing_if = "Option::is_none")]
122    pub connection: Option<u64>,
123    /// With a listener: the peer it came from, or goes to.
124    #[serde(default, skip_serializing_if = "Option::is_none")]
125    pub peer: Option<String>,
126    /// Base64, at most [`MAX_NET_DATA_BYTES`] decoded.
127    pub data: String,
128}
