15/// Errors returned while resolving per-user paths.
16#[derive(Debug, Error)]
17pub enum PathError {
18    #[error("home directory is unavailable")]
19    HomeUnavailable,
20}
21
22/// Returns the current user's home directory.
23///
24/// # Errors
25///
26/// Returns [`PathError::HomeUnavailable`] when the standard library cannot
27/// determine a home directory for the current user.
28pub fn home_dir() -> Result<PathBuf, PathError> {
29    std::env::home_dir().ok_or(PathError::HomeUnavailable)
30}
31
32/// Returns the current user's home directory canonicalized, as a runtime
33/// reports it (`home` of `runtime/initialize`): see [`canonicalize`]. The
34/// home as [`home_dir`] gives it when it cannot be canonicalized.
35///
36/// # Errors
37///
38/// Returns [`PathError::HomeUnavailable`] when the standard library cannot
39/// determine a home directory for the current user.
40pub fn canonical_home_dir() -> Result<PathBuf, PathError> {
41    let home = home_dir()?;
42    Ok(canonicalize(&home).unwrap_or(home))
43}

45/// Returns the unified Cookie Agent per-user data root.
46///
47/// # Errors
48///
49/// Returns [`PathError::HomeUnavailable`] when the current user's home
50/// directory cannot be determined.
51pub fn user_data_root() -> Result<PathBuf, PathError> {
52    Ok(home_dir()?.join(".cookie-agent"))
53}

55/// A stable, opaque identity of this machine, so a client can tell whether a
56/// daemon runs on its own machine: a SHA-256 of the operating system's machine
57/// ID (`/etc/machine-id` or `/var/lib/dbus/machine-id`) or, where there is
58/// none, of a random ID kept in `~/.cookie-agent/machine-id`, created on
59/// first use. The fallback is per user, so two users of one machine without
60/// an OS machine ID look like two machines. `None` when neither is
61/// available.
62#[must_use]
63pub fn machine_id() -> Option<String> {
64    static MACHINE_ID: std::sync::OnceLock<Option<String>> = std::sync::OnceLock::new();
65    MACHINE_ID
66        .get_or_init(|| {
67            // Debug builds under test: a runtime process of the test's own
68            // machine reports another, as a runtime of another computer would.
69            #[cfg(debug_assertions)]
70            if let Some(id) = std::env::var("COOKIE_AGENT_TEST_MACHINE_ID")
71                .ok()
72                .filter(|id| !id.is_empty())
73            {
74                return Some(id);
75            }
76            let raw = ["/etc/machine-id", "/var/lib/dbus/machine-id"]
77                .into_iter()
78                .find_map(|path| {
79                    std::fs::read_to_string(path)
80                        .ok()
81                        .map(|id| id.trim().to_owned())
82                        .filter(|id| !id.is_empty())
83                })
84                .or_else(user_machine_id)?;
85            let digest = <sha2::Sha256 as sha2::Digest>::digest(
86                format!("cookie-agent machine:{raw}").as_bytes(),
87            );
88            Some(
89                digest
90                    .iter()
91                    .take(16)
92                    .map(|byte| format!("{byte:02x}"))
93                    .collect(),
94            )
95        })
96        .clone()
97}

99/// This machine's host name; empty when it is unavailable.
100#[must_use]
101pub fn host_name() -> String {
102    #[cfg(unix)]
103    {
104        let mut buffer = [0_u8; 256];
105        // SAFETY: the buffer is writable for its whole length, and
106        // gethostname writes at most that many bytes.
107        let result = unsafe { libc::gethostname(buffer.as_mut_ptr().cast(), buffer.len()) };
108        if result != 0 {
109            return String::new();
110        }
111        let end = buffer
112            .iter()
113            .position(|byte| *byte == 0)
114            .unwrap_or(buffer.len());
115        String::from_utf8_lossy(&buffer[..end]).into_owned()
116    }
117    #[cfg(windows)]
118    {
119        std::env::var("COMPUTERNAME").unwrap_or_default()
120    }
121}

123/// The random machine ID kept in the user's data root, created when missing.
124fn user_machine_id() -> Option<String> {
125    let path = user_data_root().ok()?.join("machine-id");
126    if let Ok(id) = std::fs::read_to_string(&path) {
127        let id = id.trim().to_owned();
128        if !id.is_empty() {
129            return Some(id);
130        }
131    }
132    let id = uuid::Uuid::now_v7().simple().to_string();
133    std::fs::create_dir_all(path.parent()?).ok()?;
134    std::fs::write(&path, format!("{id}\n")).ok()?;
135    Some(id)
136}

138/// `path` canonicalized for showing and comparing: on Windows without the
139/// `\\?\` verbatim prefix `std::fs::canonicalize` adds, whenever the path
140/// means the same without it (see [`simplified`]); elsewhere exactly what
141/// `canonicalize` returns. Every canonical path a person or a model reads,
142/// such as a workspace root or a session's directory, comes from here, so
143/// they all compare equal.
144///
145/// # Errors
146///
147/// Fails as `canonicalize` does, for example when the path does not exist.
148pub fn canonicalize(path: &Path) -> std::io::Result<PathBuf> {
149    path.canonicalize().map(simplified)
150}
151
152/// `path` without the Windows verbatim prefix (`\\?\C:\…` becomes `C:\…`,
153/// `\\?\UNC\server\share\…` becomes `\\server\share\…`) when the path
154/// means the same without it: shorter than `MAX_PATH`, and with no component
155/// a non-verbatim path would read differently (a device name such as `CON`,
156/// or a name ending in a dot or a space). Other paths, and every path on
157/// other platforms, are returned unchanged.
158#[must_use]
159pub fn simplified(path: PathBuf) -> PathBuf {
160    if !cfg!(windows) {
161        return path;
162    }
163    match path.to_str().and_then(simplified_windows) {
164        Some(simple) => PathBuf::from(simple),
165        None => path,
166    }
167}
168
169/// The non-verbatim spelling of the Windows path `path`, when it has one
170/// that means the same.
171fn simplified_windows(path: &str) -> Option<String> {
172    const MAX_PATH: usize = 260;
173    let simple = if let Some(rest) = path.strip_prefix(r"\\?\UNC\") {
174        format!(r"\\{rest}")
175    } else {
176        let rest = path.strip_prefix(r"\\?\")?;
177        let bytes = rest.as_bytes();
178        if bytes.len() < 3 || !bytes[0].is_ascii_alphabetic() || &bytes[1..3] != b":\\" {
179            return None;
180        }
181        rest.to_owned()
182    };
183    if simple.len() >= MAX_PATH || simple.contains('/') {
184        return None;
185    }
186    let reserved = |name: &str| {
187        let stem = name
188            .split('.')
189            .next()
190            .unwrap_or(name)
191            .trim_end()
192            .to_ascii_uppercase();
193        matches!(
194            stem.as_str(),
195            "CON" | "PRN" | "AUX" | "NUL" | "CONIN$" | "CONOUT$"
196        ) || ((stem.starts_with("COM") || stem.starts_with("LPT"))
197            && stem.len() == 4
198            && stem.as_bytes()[3].is_ascii_digit())
199    };
200    let components = simple
201        .trim_start_matches('\\')
202        .split('\\')
203        .skip(1)
204        .filter(|component| !component.is_empty());
205    for component in components {
206        if component.ends_with('.') || component.ends_with(' ') || reserved(component) {
207            return None;
208        }
209    }
210    Some(simple)
211}

213/// The first `name` on `PATH`, found as [`search_path_candidates`] finds it.
214///
215/// `std::process::Command` looks only for `<name>.exe` on `PATH`, so it
216/// misses the `.cmd` launchers many tools install, such as VS Code's `code`.
217/// Like `where.exe` minus its current-directory search, which would let a
218/// repository plant an executable.
219#[cfg(windows)]
220#[must_use]
221pub fn find_executable(name: &str) -> Option<PathBuf> {
222    let path = std::env::var_os("PATH").unwrap_or_default();
223    let pathext = std::env::var_os("PATHEXT");
224    search_path_candidates(
225        name,
226        &path.to_string_lossy(),
227        pathext
228            .as_deref()
229            .map(|value| value.to_string_lossy())
230            .as_deref(),
231        Path::is_file,
232    )
233    .into_iter()
234    .next()
235}

237/// The program to hand `std::process::Command` (or tokio's) for a
238/// user-configured command: MCP stdio servers, ACP agents, runtime launch
239/// commands, and plugins all go through here. `env` is the child's
240/// configured environment; a `PATH` in it is searched instead of the
241/// daemon's, as `Command` itself does.
242///
243/// On Windows, `Command` appends only `.exe` to a bare name, so `npx`, which
244/// npm installs as `npx.cmd`, is "program not found". A `program` with no
245/// path separator that does not already end in a `PATHEXT` extension is
246/// therefore looked up on `PATH` with each `PATHEXT` extension, as
247/// [`search_path_candidates`] does, and the first hit `CreateProcess` can
248/// start (`.com`, `.exe`, `.bat`, `.cmd`; never `.ps1`) replaces it. A
249/// `.bat` or `.cmd` hit goes to `Command` as is: since Rust 1.77.2
250/// (CVE-2024-24576, "BatBadBut") the standard library runs a batch file
251/// through `cmd.exe /e:ON /v:OFF /d /c` with escaping that keeps every
252/// argument literal, and fails the spawn when an argument cannot be passed
253/// safely. Never build that `cmd.exe /c` line by hand.
254///
255/// Elsewhere, and when nothing is found, `program` comes back unchanged.
256#[must_use]
257pub fn resolve_program<K, V>(program: &str, env: impl IntoIterator<Item = (K, V)>) -> PathBuf
258where
259    K: AsRef<str>,
260    V: AsRef<str>,
261{
262    #[cfg(windows)]
263    {
264        let path = env
265            .into_iter()
266            .find(|(key, _)| key.as_ref().eq_ignore_ascii_case("PATH"))
267            .map(|(_, value)| value.as_ref().to_owned())
268            .or_else(|| std::env::var_os("PATH").map(|value| value.to_string_lossy().into_owned()))
269            .unwrap_or_default();
270        let pathext = std::env::var_os("PATHEXT");
271        windows_program(
272            program,
273            &path,
274            pathext
275                .as_deref()
276                .map(|value| value.to_string_lossy())
277                .as_deref(),
278            Path::is_file,
279        )
280        .unwrap_or_else(|| PathBuf::from(program))
281    }
282    #[cfg(not(windows))]
283    {
284        let _ = env;
285        PathBuf::from(program)
286    }
287}

289/// The Windows lookup behind [`resolve_program`]; `None` leaves `program`
290/// to `Command`.
291#[cfg(any(windows, test))]
292fn windows_program(
293    program: &str,
294    path: &str,
295    pathext: Option<&str>,
296    exists: impl Fn(&Path) -> bool,
297) -> Option<PathBuf> {
298    const RUNNABLE: [&str; 4] = ["com", "exe", "bat", "cmd"];
299    if program.is_empty() || program.contains(['/', '\\', ':']) {
300        return None;
301    }
302    let has_extension = pathext
303        .unwrap_or(".COM;.EXE;.BAT;.CMD")
304        .split(';')
305        .map(str::trim)
306        .filter(|extension| !extension.is_empty())
307        .any(|extension| {
308            program.len() > extension.len()
309                && program.is_char_boundary(program.len() - extension.len())
310                && program[program.len() - extension.len()..].eq_ignore_ascii_case(extension)
311        });
312    if has_extension {
313        return None;
314    }
315    search_path_candidates(program, path, pathext, exists)
316        .into_iter()
317        .find(|candidate| {
318            candidate
319                .extension()
320                .and_then(|extension| extension.to_str())
321                .is_some_and(|extension| {
322                    RUNNABLE
323                        .iter()
324                        .any(|runnable| extension.eq_ignore_ascii_case(runnable))
325                })
326        })
327}

329/// Pure candidate selection behind `where.exe`-style lookups: for each
330/// `PATH` entry in order, tries `name` with each `PATHEXT` extension in order
331/// (default `.COM;.EXE;.BAT;.CMD`). Like `where`, a name that already ends in
332/// a `PATHEXT` extension (case-insensitive) is tried as written instead.
333/// Entries are `;`-separated, may be double-quoted, and empty ones are
334/// skipped. No filtering happens here; callers vet the candidates.
335pub fn search_path_candidates(
336    name: &str,
337    path: &str,
338    pathext: Option<&str>,
339    exists: impl Fn(&Path) -> bool,
340) -> Vec<PathBuf> {
341    let extensions: Vec<&str> = pathext
342        .unwrap_or(".COM;.EXE;.BAT;.CMD")
343        .split(';')
344        .map(str::trim)
345        .filter(|extension| !extension.is_empty())
346        .collect();
347    let has_extension = extensions.iter().any(|extension| {
348        name.len() > extension.len()
349            && name.is_char_boundary(name.len() - extension.len())
350            && name[name.len() - extension.len()..].eq_ignore_ascii_case(extension)
351    });
352    let names: Vec<String> = if has_extension {
353        vec![name.to_owned()]
354    } else {
355        extensions
356            .iter()
357            .map(|extension| format!("{name}{extension}"))
358            .collect()
359    };
360
361    let mut entries = Vec::new();
362    let mut current = String::new();
363    let mut quoted = false;
364    for character in path.chars() {
365        match character {
366            '"' => quoted = !quoted,
367            ';' if !quoted => entries.push(std::mem::take(&mut current)),
368            _ => current.push(character),
369        }
370    }
371    entries.push(current);
372
373    let mut found = Vec::new();
374    for entry in entries.iter().filter(|entry| !entry.is_empty()) {
375        for file in &names {
376            let candidate = Path::new(entry).join(file);
377            if exists(&candidate) {
378                found.push(candidate);
379            }
380        }
381    }
382    found
383}

626                .to_string_lossy()
627                .to_lowercase()
628        );
629        // The BatBadBut shape: a quote that would end the argument and an
630        // `&` that would start a second command.
631        let status = std::process::Command::new(&program)
632            .args(["hello world", "x\" & echo injected> pwned.txt & \""])
633            .current_dir(directory.path())
634            .status()
635            .unwrap();
636        assert!(status.success());
637        let out = std::fs::read_to_string(directory.path().join("out.txt")).unwrap();
638        assert!(out.contains("hello world"), "{out}");
639        assert!(!directory.path().join("pwned.txt").exists(), "{out}");
640    }
