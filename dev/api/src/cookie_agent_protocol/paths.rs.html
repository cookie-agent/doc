1//! Cross-platform per-user path resolution for Cookie Agent.
2//!
3//! This module lives in `cookie_agent_protocol` because models, config, engine,
4//! server, TUI, and tools already depend on that low-level crate. Placing it in
5//! `cookie_agent_config` would create a dependency cycle with models.

7use std::path::{Path, PathBuf};

9use thiserror::Error;

11/// Errors returned while resolving per-user paths.
12#[derive(Debug, Error)]
13pub enum PathError {
14    #[error("home directory is unavailable")]
15    HomeUnavailable,
16}
17
18/// Returns the current user's home directory.
19///
20/// # Errors
21///
22/// Returns [`PathError::HomeUnavailable`] when the standard library cannot
23/// determine a home directory for the current user.
24pub fn home_dir() -> Result<PathBuf, PathError> {
25    std::env::home_dir().ok_or(PathError::HomeUnavailable)
26}
27
28/// Returns the current user's home directory canonicalized, as a runtime
29/// reports it (`home` of `runtime/initialize`): see [`canonicalize`]. The
30/// home as [`home_dir`] gives it when it cannot be canonicalized.
31///
32/// # Errors
33///
34/// Returns [`PathError::HomeUnavailable`] when the standard library cannot
35/// determine a home directory for the current user.
36pub fn canonical_home_dir() -> Result<PathBuf, PathError> {
37    let home = home_dir()?;
38    Ok(canonicalize(&home).unwrap_or(home))
39}

41/// Returns the unified Cookie Agent per-user data root.
42///
43/// # Errors
44///
45/// Returns [`PathError::HomeUnavailable`] when the current user's home
46/// directory cannot be determined.
47pub fn user_data_root() -> Result<PathBuf, PathError> {
48    Ok(home_dir()?.join(".cookie-agent"))
49}

51/// A stable, opaque identity of this machine, so a client can tell whether a
52/// daemon runs on its own machine: a SHA-256 of the operating system's machine
53/// ID (`/etc/machine-id` or `/var/lib/dbus/machine-id`) or, where there is
54/// none, of a random ID kept in `~/.cookie-agent/machine-id`, created on
55/// first use. The fallback is per user, so two users of one machine without
56/// an OS machine ID look like two machines. `None` when neither is
57/// available.
58#[must_use]
59pub fn machine_id() -> Option<String> {
60    static MACHINE_ID: std::sync::OnceLock<Option<String>> = std::sync::OnceLock::new();
61    MACHINE_ID
62        .get_or_init(|| {
63            // Debug builds under test: a runtime process of the test's own
64            // machine reports another, as a runtime of another computer would.
65            #[cfg(debug_assertions)]
66            if let Some(id) = std::env::var("COOKIE_AGENT_TEST_MACHINE_ID")
67                .ok()
68                .filter(|id| !id.is_empty())
69            {
70                return Some(id);
71            }
72            let raw = ["/etc/machine-id", "/var/lib/dbus/machine-id"]
73                .into_iter()
74                .find_map(|path| {
75                    std::fs::read_to_string(path)
76                        .ok()
77                        .map(|id| id.trim().to_owned())
78                        .filter(|id| !id.is_empty())
79                })
80                .or_else(user_machine_id)?;
81            let digest = <sha2::Sha256 as sha2::Digest>::digest(
82                format!("cookie-agent machine:{raw}").as_bytes(),
83            );
84            Some(
85                digest
86                    .iter()
87                    .take(16)
88                    .map(|byte| format!("{byte:02x}"))
89                    .collect(),
90            )
91        })
92        .clone()
93}

95/// This machine's host name; empty when it is unavailable.
96#[must_use]
97pub fn host_name() -> String {
98    #[cfg(unix)]
99    {
100        let mut buffer = [0_u8; 256];
101        // SAFETY: the buffer is writable for its whole length, and
102        // gethostname writes at most that many bytes.
103        let result = unsafe { libc::gethostname(buffer.as_mut_ptr().cast(), buffer.len()) };
104        if result != 0 {
105            return String::new();
106        }
107        let end = buffer
108            .iter()
109            .position(|byte| *byte == 0)
110            .unwrap_or(buffer.len());
111        String::from_utf8_lossy(&buffer[..end]).into_owned()
112    }
113    #[cfg(windows)]
114    {
115        std::env::var("COMPUTERNAME").unwrap_or_default()
116    }
117}

119/// The random machine ID kept in the user's data root, created when missing.
120fn user_machine_id() -> Option<String> {
121    let path = user_data_root().ok()?.join("machine-id");
122    if let Ok(id) = std::fs::read_to_string(&path) {
123        let id = id.trim().to_owned();
124        if !id.is_empty() {
125            return Some(id);
126        }
127    }
128    let id = uuid::Uuid::now_v7().simple().to_string();
129    std::fs::create_dir_all(path.parent()?).ok()?;
130    std::fs::write(&path, format!("{id}\n")).ok()?;
131    Some(id)
132}

134/// `path` canonicalized for showing and comparing: on Windows without the
135/// `\\?\` verbatim prefix `std::fs::canonicalize` adds, whenever the path
136/// means the same without it (see [`simplified`]); elsewhere exactly what
137/// `canonicalize` returns. Every canonical path a person or a model reads,
138/// such as a workspace root or a session's directory, comes from here, so
139/// they all compare equal.
140///
141/// # Errors
142///
143/// Fails as `canonicalize` does, for example when the path does not exist.
144pub fn canonicalize(path: &Path) -> std::io::Result<PathBuf> {
145    path.canonicalize().map(simplified)
146}
147
148/// `path` without the Windows verbatim prefix (`\\?\C:\…` becomes `C:\…`,
149/// `\\?\UNC\server\share\…` becomes `\\server\share\…`) when the path
150/// means the same without it: shorter than `MAX_PATH`, and with no component
151/// a non-verbatim path would read differently (a device name such as `CON`,
152/// or a name ending in a dot or a space). Other paths, and every path on
153/// other platforms, are returned unchanged.
154#[must_use]
155pub fn simplified(path: PathBuf) -> PathBuf {
156    if !cfg!(windows) {
157        return path;
158    }
159    match path.to_str().and_then(simplified_windows) {
160        Some(simple) => PathBuf::from(simple),
161        None => path,
162    }
163}
164
165/// The non-verbatim spelling of the Windows path `path`, when it has one
166/// that means the same.
167fn simplified_windows(path: &str) -> Option<String> {
168    const MAX_PATH: usize = 260;
169    let simple = if let Some(rest) = path.strip_prefix(r"\\?\UNC\") {
170        format!(r"\\{rest}")
171    } else {
172        let rest = path.strip_prefix(r"\\?\")?;
173        let bytes = rest.as_bytes();
174        if bytes.len() < 3 || !bytes[0].is_ascii_alphabetic() || &bytes[1..3] != b":\\" {
175            return None;
176        }
177        rest.to_owned()
178    };
179    if simple.len() >= MAX_PATH || simple.contains('/') {
180        return None;
181    }
182    let reserved = |name: &str| {
183        let stem = name
184            .split('.')
185            .next()
186            .unwrap_or(name)
187            .trim_end()
188            .to_ascii_uppercase();
189        matches!(
190            stem.as_str(),
191            "CON" | "PRN" | "AUX" | "NUL" | "CONIN$" | "CONOUT$"
192        ) || ((stem.starts_with("COM") || stem.starts_with("LPT"))
193            && stem.len() == 4
194            && stem.as_bytes()[3].is_ascii_digit())
195    };
196    let components = simple
197        .trim_start_matches('\\')
198        .split('\\')
199        .skip(1)
200        .filter(|component| !component.is_empty());
201    for component in components {
202        if component.ends_with('.') || component.ends_with(' ') || reserved(component) {
203            return None;
204        }
205    }
206    Some(simple)
207}

209/// The first `name` on `PATH`, found as [`search_path_candidates`] finds it.
210///
211/// `std::process::Command` looks only for `<name>.exe` on `PATH`, so it
212/// misses the `.cmd` launchers many tools install, such as VS Code's `code`.
213/// Like `where.exe` minus its current-directory search, which would let a
214/// repository plant an executable.
215#[cfg(windows)]
216#[must_use]
217pub fn find_executable(name: &str) -> Option<PathBuf> {
218    let path = std::env::var_os("PATH").unwrap_or_default();
219    let pathext = std::env::var_os("PATHEXT");
220    search_path_candidates(
221        name,
222        &path.to_string_lossy(),
223        pathext
224            .as_deref()
225            .map(|value| value.to_string_lossy())
226            .as_deref(),
227        Path::is_file,
228    )
229    .into_iter()
230    .next()
231}

233/// The program to hand `std::process::Command` (or tokio's) for a
234/// user-configured command: MCP stdio servers, ACP agents, runtime launch
235/// commands, and plugins all go through here. `env` is the child's
236/// configured environment; a `PATH` in it is searched instead of the
237/// daemon's, as `Command` itself does.
238///
239/// On Windows, `Command` appends only `.exe` to a bare name, so `npx`, which
240/// npm installs as `npx.cmd`, is "program not found". A `program` with no
241/// path separator that does not already end in a `PATHEXT` extension is
242/// therefore looked up on `PATH` with each `PATHEXT` extension, as
243/// [`search_path_candidates`] does, and the first hit `CreateProcess` can
244/// start (`.com`, `.exe`, `.bat`, `.cmd`; never `.ps1`) replaces it. A
245/// `.bat` or `.cmd` hit goes to `Command` as is: since Rust 1.77.2
246/// (CVE-2024-24576, "BatBadBut") the standard library runs a batch file
247/// through `cmd.exe /e:ON /v:OFF /d /c` with escaping that keeps every
248/// argument literal, and fails the spawn when an argument cannot be passed
249/// safely. Never build that `cmd.exe /c` line by hand.
250///
251/// Elsewhere, and when nothing is found, `program` comes back unchanged.
252#[must_use]
253pub fn resolve_program<K, V>(program: &str, env: impl IntoIterator<Item = (K, V)>) -> PathBuf
254where
255    K: AsRef<str>,
256    V: AsRef<str>,
257{
258    #[cfg(windows)]
259    {
260        let path = env
261            .into_iter()
262            .find(|(key, _)| key.as_ref().eq_ignore_ascii_case("PATH"))
263            .map(|(_, value)| value.as_ref().to_owned())
264            .or_else(|| std::env::var_os("PATH").map(|value| value.to_string_lossy().into_owned()))
265            .unwrap_or_default();
266        let pathext = std::env::var_os("PATHEXT");
267        windows_program(
268            program,
269            &path,
270            pathext
271                .as_deref()
272                .map(|value| value.to_string_lossy())
273                .as_deref(),
274            Path::is_file,
275        )
276        .unwrap_or_else(|| PathBuf::from(program))
277    }
278    #[cfg(not(windows))]
279    {
280        let _ = env;
281        PathBuf::from(program)
282    }
283}

285/// The Windows lookup behind [`resolve_program`]; `None` leaves `program`
286/// to `Command`.
287#[cfg(any(windows, test))]
288fn windows_program(
289    program: &str,
290    path: &str,
291    pathext: Option<&str>,
292    exists: impl Fn(&Path) -> bool,
293) -> Option<PathBuf> {
294    const RUNNABLE: [&str; 4] = ["com", "exe", "bat", "cmd"];
295    if program.is_empty() || program.contains(['/', '\\', ':']) {
296        return None;
297    }
298    let has_extension = pathext
299        .unwrap_or(".COM;.EXE;.BAT;.CMD")
300        .split(';')
301        .map(str::trim)
302        .filter(|extension| !extension.is_empty())
303        .any(|extension| {
304            program.len() > extension.len()
305                && program.is_char_boundary(program.len() - extension.len())
306                && program[program.len() - extension.len()..].eq_ignore_ascii_case(extension)
307        });
308    if has_extension {
309        return None;
310    }
311    search_path_candidates(program, path, pathext, exists)
312        .into_iter()
313        .find(|candidate| {
314            candidate
315                .extension()
316                .and_then(|extension| extension.to_str())
317                .is_some_and(|extension| {
318                    RUNNABLE
319                        .iter()
320                        .any(|runnable| extension.eq_ignore_ascii_case(runnable))
321                })
322        })
323}

325/// Pure candidate selection behind `where.exe`-style lookups: for each
326/// `PATH` entry in order, tries `name` with each `PATHEXT` extension in order
327/// (default `.COM;.EXE;.BAT;.CMD`). Like `where`, a name that already ends in
328/// a `PATHEXT` extension (case-insensitive) is tried as written instead.
329/// Entries are `;`-separated, may be double-quoted, and empty ones are
330/// skipped. No filtering happens here; callers vet the candidates.
331pub fn search_path_candidates(
332    name: &str,
333    path: &str,
334    pathext: Option<&str>,
335    exists: impl Fn(&Path) -> bool,
336) -> Vec<PathBuf> {
337    let extensions: Vec<&str> = pathext
338        .unwrap_or(".COM;.EXE;.BAT;.CMD")
339        .split(';')
340        .map(str::trim)
341        .filter(|extension| !extension.is_empty())
342        .collect();
343    let has_extension = extensions.iter().any(|extension| {
344        name.len() > extension.len()
345            && name.is_char_boundary(name.len() - extension.len())
346            && name[name.len() - extension.len()..].eq_ignore_ascii_case(extension)
347    });
348    let names: Vec<String> = if has_extension {
349        vec![name.to_owned()]
350    } else {
351        extensions
352            .iter()
353            .map(|extension| format!("{name}{extension}"))
354            .collect()
355    };
356
357    let mut entries = Vec::new();
358    let mut current = String::new();
359    let mut quoted = false;
360    for character in path.chars() {
361        match character {
362            '"' => quoted = !quoted,
363            ';' if !quoted => entries.push(std::mem::take(&mut current)),
364            _ => current.push(character),
365        }
366    }
367    entries.push(current);
368
369    let mut found = Vec::new();
370    for entry in entries.iter().filter(|entry| !entry.is_empty()) {
371        for file in &names {
372            let candidate = Path::new(entry).join(file);
373            if exists(&candidate) {
374                found.push(candidate);
375            }
376        }
377    }
378    found
379}

625        // The BatBadBut shape: a quote that would end the argument and an
626        // `&` that would start a second command.
627        let status = std::process::Command::new(&program)
628            .args(["hello world", "x\" & echo injected> pwned.txt & \""])
629            .current_dir(directory.path())
630            .status()
631            .unwrap();
632        assert!(status.success());
633        let out = std::fs::read_to_string(directory.path().join("out.txt")).unwrap();
634        assert!(out.contains("hello world"), "{out}");
635        assert!(!directory.path().join("pwned.txt").exists(), "{out}");
636    }
