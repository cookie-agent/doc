1//! What the client protocol and session history say about ACP agents
2//! (`docs/site/specs/acp-agents.md` sections 5, 10, and 11; client protocol
3//! 1.17.0, notices, steering, and tool presentations 1.20.0, context use
4//! and boolean options 1.23.0).
5//!
6//! The daemon records the ACP session a cookie session talks to
7//! (`acp_session_started`) and what the agent reports about it
8//! (`acp_session_updated`): its slash commands, its config options, and its
9//! plan. Clients fold the latest of each. These are information: a reader
10//! ignores members it does not know.
11

25crate::open_enum! {
26    /// How the daemon got the ACP session a turn runs in. An unknown way
27    /// reads as `new`.
28    #[derive(Clone, Copy, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
29    #[serde(rename_all = "snake_case")]
30    pub enum AcpSessionStart {
31        /// `session/new`: the session's first, or the agent could not
32        /// resume or load the one it had.
33        New,
34        /// `session/resume` of the session recorded before.
35        Resumed,
36        /// `session/load` of the session recorded before; its replay is
37        /// ignored.
38        Loaded,
39    }
40}

42/// One slash command an ACP agent offers in a session.

54crate::open_enum! {
55    /// The control an [`AcpConfigOption`] is (client protocol 1.23.0). An
56    /// unknown kind reads as a select, whose `values` a client lists.
57    #[derive(Clone, Copy, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
58    #[serde(rename_all = "snake_case")]
59    pub enum AcpConfigOptionKind {
60        /// One of `values`.
61        #[default]
62        Select,
63        /// On or off: `current_value` is `true` or `false`, and `values`
64        /// lists both (`true`, named On, and `false`, named Off), so a
65        /// client that reads every option as a select can still set it.
66        Boolean,
67    }
68}
69
70impl AcpConfigOptionKind {
71    fn is_select(&self) -> bool {
72        *self == Self::Select
73    }
74}

76/// One config option an ACP agent offers in a session, such as its model,
77/// mode, or reasoning effort. An agent that offers modes but no mode option
78/// is shown as an option `mode` of category `mode`.
79#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
80pub struct AcpConfigOption {
81    pub id: String,
82    pub name: String,
83    /// The option's control: a select (the default, left out) or a boolean
84    /// (client protocol 1.23.0).
85    #[serde(default, skip_serializing_if = "AcpConfigOptionKind::is_select")]
86    pub kind: AcpConfigOptionKind,
87    #[serde(default, skip_serializing_if = "Option::is_none")]
88    pub description: Option<String>,
89    /// The agent's label for what the option is, such as `model`, `mode`,
90    /// or `thought_level`.
91    #[serde(default, skip_serializing_if = "Option::is_none")]
92    pub category: Option<String>,
93    pub current_value: String,
94    #[serde(default)]
95    pub values: Vec<AcpConfigValue>,
96}

98/// One value an [`AcpConfigOption`] can take.
99#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
100pub struct AcpConfigValue {
101    pub value: String,
102    pub name: String,
103    #[serde(default, skip_serializing_if = "Option::is_none")]
104    pub description: Option<String>,

107/// The payload of `acp_session_started`.
108#[derive(Clone, Debug, Deserialize, JsonSchema, PartialEq, Serialize)]
109pub struct AcpSessionRecord {
110    /// The ACP agent, as the daemon registers it.
111    pub agent: AgentId,
112    /// The runtime that hosts it.
113    pub runtime_id: RuntimeId,
114    /// The agent's ID on the runtime.
115    pub runtime_agent: AgentId,
116    /// The agent's session ID.
117    pub acp_session_id: String,
118    /// The session's working directory on the runtime's machine.
119    pub cwd: String,
120    /// The ACP protocol version the agent answered `initialize` with.
121    pub protocol_version: u32,
122    /// `agentInfo` as the agent sent it, when it did.
123    #[serde(default, skip_serializing_if = "Value::is_null")]
124    pub agent_info: Value,
125    pub how: AcpSessionStart,
126    /// Whether the agent takes messages into its running turn (it
127    /// advertises the steering extension, `_session/steering`): input sent
128    /// to steer reaches the turn, else it waits for the agent's next turn
129    /// (client protocol 1.20.0).
130    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
131    pub steering: bool,
132}

157/// Input the daemon delivered into the agent's running turn through its
158/// steering extension, by the events that took it: the
159/// `user_input_submitted` events' sequence numbers and the producer

169/// A new presentation of one of the agent's own tool calls, whose title,

179/// How much of its context window the agent's conversation uses, as its

190/// The payload of `acp_session_updated`: what changed, each member the
191/// whole new value.
192#[derive(Clone, Debug, Default, Deserialize, JsonSchema, PartialEq, Serialize)]
193pub struct AcpSessionUpdate {
194    #[serde(default, skip_serializing_if = "Option::is_none")]
195    pub commands: Option<Vec<AcpCommand>>,
196    #[serde(default, skip_serializing_if = "Option::is_none")]
197    pub options: Option<Vec<AcpConfigOption>>,
198    /// The agent's plan, a `plan` display block.
199    #[serde(default, skip_serializing_if = "Option::is_none")]
200    pub plan: Option<Open<DisplayBlock>>,
201    /// A notice of the agent's, shown once where it arrives.
202    #[serde(default, skip_serializing_if = "Option::is_none")]
203    pub notice: Option<AcpNotice>,
204    /// Input delivered into the running turn.
205    #[serde(default, skip_serializing_if = "Option::is_none")]
206    pub steered: Option<AcpSteered>,
207    /// A tool call's new presentation.
208    #[serde(default, skip_serializing_if = "Option::is_none")]
209    pub tool_presentation: Option<AcpToolPresentation>,
210    /// The agent's context use, when it changed (client protocol 1.23.0).
211    #[serde(default, skip_serializing_if = "Option::is_none")]
212    pub context: Option<AcpContextUsage>,
213}
214
215impl AcpSessionUpdate {
216    /// Whether it carries nothing.
217    #[must_use]
218    pub fn is_empty(&self) -> bool {
219        self.commands.is_none()
220            && self.options.is_none()
221            && self.plan.is_none()
222            && self.notice.is_none()
223            && self.steered.is_none()
224            && self.tool_presentation.is_none()
225            && self.context.is_none()
226    }
227}
