1//! What the client protocol and session history say about ACP agents
2//! (`docs/site/specs/acp-agents.md` sections 5, 10, and 11; client protocol
3//! 1.17.0, notices, steering, and tool presentations 1.20.0, context use
4//! and boolean options 1.23.0, forked and detached sessions 1.24.0).
5//!
6//! The daemon records the ACP session a cookie session talks to
7//! (`acp_session_started`) and what the agent reports about it
8//! (`acp_session_updated`): its slash commands, its config options, and its
9//! plan. Clients fold the latest of each. These are information: a reader
10//! ignores members it does not know.
11

25crate::open_enum! {
26    /// How the daemon got the ACP session a turn runs in. An unknown way
27    /// reads as `new`.
28    #[derive(Clone, Copy, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
29    #[serde(rename_all = "snake_case")]
30    pub enum AcpSessionStart {
31        /// `session/new`: the session's first, or the agent could not
32        /// resume or load the one it had.
33        New,
34        /// `session/resume` of the session recorded before.
35        Resumed,
36        /// `session/load` of the session recorded before; its replay is
37        /// ignored.
38        Loaded,
39        /// `session/fork` of the session a fork or a revert left behind
40        /// ([`AcpSessionRecord::forked_from`]); client protocol 1.24.0.
41        Forked,
42    }
43}

45crate::open_enum! {
46    /// Why a cookie session stopped continuing an ACP session as it
47    /// stands (client protocol 1.24.0). An unknown reason reads as
48    /// `forked`, which never ends the agent's session.
49    #[derive(Clone, Copy, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
50    #[serde(rename_all = "snake_case")]
51    pub enum AcpDetachReason {
52        /// The cookie session is a fork: the ACP session stays its
53        /// source's.
54        Forked,
55        /// The cookie session was reverted to before the end of the ACP
56        /// session's conversation: the agent's session holds turns the
57        /// cookie session no longer has, and is deleted once the next
58        /// turn has branched off it.
59        Reverted,
60    }
61}

63/// A cookie session's ACP session left behind by a fork or a revert
64/// (client protocol 1.24.0): the next turn of its agent does not resume
65/// it but branches off it at `message_id`, the agent's last message the
66/// cookie session keeps (`session/fork`, else a new session given the
67/// cookie transcript), and the session recorded before is no longer the
68/// cookie session's.
69#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
70pub struct AcpDetached {
71    /// The agent's session ID ([`AcpSessionRecord::acp_session_id`]).
72    pub acp_session_id: String,
73    pub reason: AcpDetachReason,
74    /// The ACP `messageId` of the agent's last message the cookie session
75    /// keeps; none when it keeps none of the agent's messages.
76    #[serde(default, skip_serializing_if = "Option::is_none")]
77    pub message_id: Option<String>,
78}

92crate::open_enum! {
93    /// The control an [`AcpConfigOption`] is (client protocol 1.23.0). An
94    /// unknown kind reads as a select, whose `values` a client lists.
95    #[derive(Clone, Copy, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
96    #[serde(rename_all = "snake_case")]
97    pub enum AcpConfigOptionKind {
98        /// One of `values`.
99        #[default]
100        Select,
101        /// On or off: `current_value` is `true` or `false`, and `values`
102        /// lists both (`true`, named On, and `false`, named Off), so a
103        /// client that reads every option as a select can still set it.
104        Boolean,
105    }
106}
107
108impl AcpConfigOptionKind {
109    fn is_select(&self) -> bool {
110        *self == Self::Select
111    }
112}

114/// One config option an ACP agent offers in a session, such as its model,
115/// mode, or reasoning effort. An agent that offers modes but no mode option
116/// is shown as an option `mode` of category `mode`.
117#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
118pub struct AcpConfigOption {
119    pub id: String,
120    pub name: String,
121    /// The option's control: a select (the default, left out) or a boolean
122    /// (client protocol 1.23.0).
123    #[serde(default, skip_serializing_if = "AcpConfigOptionKind::is_select")]
124    pub kind: AcpConfigOptionKind,
125    #[serde(default, skip_serializing_if = "Option::is_none")]
126    pub description: Option<String>,
127    /// The agent's label for what the option is, such as `model`, `mode`,
128    /// or `thought_level`.
129    #[serde(default, skip_serializing_if = "Option::is_none")]
130    pub category: Option<String>,
131    pub current_value: String,
132    #[serde(default)]
133    pub values: Vec<AcpConfigValue>,
134}

145/// The payload of `acp_session_started`.
146#[derive(Clone, Debug, Deserialize, JsonSchema, PartialEq, Serialize)]
147pub struct AcpSessionRecord {
148    /// The ACP agent, as the daemon registers it.
149    pub agent: AgentId,
150    /// The runtime that hosts it.
151    pub runtime_id: RuntimeId,
152    /// The agent's ID on the runtime.
153    pub runtime_agent: AgentId,
154    /// The agent's session ID.
155    pub acp_session_id: String,
156    /// The session's working directory on the runtime's machine.
157    pub cwd: String,
158    /// The ACP protocol version the agent answered `initialize` with.
159    pub protocol_version: u32,
160    /// `agentInfo` as the agent sent it, when it did.
161    #[serde(default, skip_serializing_if = "Value::is_null")]
162    pub agent_info: Value,
163    pub how: AcpSessionStart,
164    /// Whether the agent takes messages into its running turn (it
165    /// advertises the steering extension, `_session/steering`): input sent
166    /// to steer reaches the turn, else it waits for the agent's next turn
167    /// (client protocol 1.20.0).
168    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
169    pub steering: bool,
170    /// The agent's session this one was forked from, for `how: forked`
171    /// (client protocol 1.24.0).
172    #[serde(default, skip_serializing_if = "Option::is_none")]
173    pub forked_from: Option<String>,
174    /// Whether the agent offers `session/fork` (client protocol 1.24.0);
175    /// absent in a record an older build wrote.
176    #[serde(default, skip_serializing_if = "Option::is_none")]
177    pub fork: Option<bool>,
178    /// Whether the agent's `session/fork` forks at a message (the JetBrains
179    /// AIR fork point), as the daemon found by asking it; absent until it
180    /// has, when `acp_session_updated.fork_point` may say later (client
181    /// protocol 1.24.0).
182    #[serde(default, skip_serializing_if = "Option::is_none")]
183    pub fork_point: Option<bool>,
184}

242/// The payload of `acp_session_updated`: what changed, each member the
243/// whole new value.
244#[derive(Clone, Debug, Default, Deserialize, JsonSchema, PartialEq, Serialize)]
245pub struct AcpSessionUpdate {
246    #[serde(default, skip_serializing_if = "Option::is_none")]
247    pub commands: Option<Vec<AcpCommand>>,
248    #[serde(default, skip_serializing_if = "Option::is_none")]
249    pub options: Option<Vec<AcpConfigOption>>,
250    /// The agent's plan, a `plan` display block.
251    #[serde(default, skip_serializing_if = "Option::is_none")]
252    pub plan: Option<Open<DisplayBlock>>,
253    /// A notice of the agent's, shown once where it arrives.
254    #[serde(default, skip_serializing_if = "Option::is_none")]
255    pub notice: Option<AcpNotice>,
256    /// Input delivered into the running turn.
257    #[serde(default, skip_serializing_if = "Option::is_none")]
258    pub steered: Option<AcpSteered>,
259    /// A tool call's new presentation.
260    #[serde(default, skip_serializing_if = "Option::is_none")]
261    pub tool_presentation: Option<AcpToolPresentation>,
262    /// The agent's context use, when it changed (client protocol 1.23.0).
263    #[serde(default, skip_serializing_if = "Option::is_none")]
264    pub context: Option<AcpContextUsage>,
265    /// The ACP session a fork or a revert left behind (client protocol
266    /// 1.24.0).
267    #[serde(default, skip_serializing_if = "Option::is_none")]
268    pub detached: Option<AcpDetached>,
269    /// Whether the agent's `session/fork` forks at a message, found after
270    /// the session's record ([`AcpSessionRecord::fork_point`]; client
271    /// protocol 1.24.0).
272    #[serde(default, skip_serializing_if = "Option::is_none")]
273    pub fork_point: Option<bool>,
274}
275
276impl AcpSessionUpdate {
277    /// Whether it carries nothing.
278    #[must_use]
279    pub fn is_empty(&self) -> bool {
280        self.commands.is_none()
281            && self.options.is_none()
282            && self.plan.is_none()
283            && self.notice.is_none()
284            && self.steered.is_none()
285            && self.tool_presentation.is_none()
286            && self.context.is_none()
287            && self.detached.is_none()
288            && self.fork_point.is_none()
289    }
290}
