1//! Runtime attachments: which runtimes a session uses, as specified in
2//! `docs/site/specs/runtimes.md`.
3
4use schemars::JsonSchema;
5use serde::{Deserialize, Serialize};
6
7use crate::{AttachmentName, CapabilityName, RuntimeId, RuntimeKind};
8
9/// One runtime a session is attached to.
10#[derive(Clone, Debug, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
11#[serde(deny_unknown_fields)]
12pub struct RuntimeAttachment {
13    /// Session-unique attachment name; the tool prefix on collisions.
14    pub name: AttachmentName,
15    /// The runtime the capability instance belongs to.
16    pub runtime_id: RuntimeId,
17    /// The capability of the runtime the session uses.
18    pub capability: CapabilityName,
19    /// The capability's self-declared kind.
20    pub kind: RuntimeKind,
21    /// Canonical workspace path the runtime reported; absent without one.
22    #[serde(default, skip_serializing_if = "Option::is_none")]
23    pub workspace: Option<String>,
24}
25
26/// The capability a runtime's attachment most likely refers to before its
27/// instance reports one: `global` for the global runtime, `coding` for any
28/// other. Clients use it to label a runtime they have not opened yet.
29#[must_use]
30pub fn default_capability(runtime_id: &RuntimeId) -> CapabilityName {
31    if runtime_id.as_str() == RuntimeId::GLOBAL {
32        CapabilityName::global()
33    } else {
34        CapabilityName::coding()
35    }
36}
37
38/// The attachment set a session's events describe: the fold of its
39/// `runtime_attached` and `runtime_detached` events in log order.
40#[derive(Clone, Debug, Default, Eq, PartialEq)]
41pub struct RuntimeAttachmentFold {
42    attachments: Vec<RuntimeAttachment>,
43}
44
45impl RuntimeAttachmentFold {
46    /// Folds one event payload.
47    pub fn observe(&mut self, payload: &crate::EventPayload) {
48        match payload {
49            crate::EventPayload::RuntimeAttached {
50                name,
51                runtime_id,
52                capability,
53                kind,
54                workspace,
55            } => {
56                self.attachments
57                    .retain(|attachment| &attachment.name != name);
58                self.attachments.push(RuntimeAttachment {
59                    name: name.clone(),
60                    runtime_id: runtime_id.clone(),
61                    capability: capability.clone(),
62                    kind: kind.clone(),
63                    workspace: workspace.clone(),
64                });
65            }
66            crate::EventPayload::RuntimeDetached { name } => {
67                self.attachments
68                    .retain(|attachment| &attachment.name != name);
69            }
70            _ => {}
71        }
72    }
73
74    /// The folded attachments, in attach order.
75    #[must_use]
76    pub fn attachments(&self) -> &[RuntimeAttachment] {
77        &self.attachments
78    }
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    #[test]
86    fn the_fold_follows_attach_and_detach_in_log_order() {
87        let attached =
88            |name: &str, runtime: &str, capability: &str| crate::EventPayload::RuntimeAttached {
89                name: AttachmentName::new(name).unwrap(),
90                runtime_id: RuntimeId::new(runtime).unwrap(),
91                capability: CapabilityName::new(capability).unwrap(),
92                kind: RuntimeKind::new(capability).unwrap(),
93                workspace: None,
94            };
95        let mut fold = RuntimeAttachmentFold::default();
96        assert!(fold.attachments().is_empty());
97        fold.observe(&attached("global", "global", "global"));
98        fold.observe(&attached("devbox_notes", "devbox", "notes"));
99        fold.observe(&crate::EventPayload::RuntimeDetached {
100            name: AttachmentName::new("global").unwrap(),
101        });
102        fold.observe(&attached("labpc_api", "labpc", "coding"));
103        let names = fold
104            .attachments()
105            .iter()
106            .map(|attachment| attachment.name.as_str())
107            .collect::<Vec<_>>();
108        assert_eq!(names, ["devbox_notes", "labpc_api"]);
109    }
110}
