1//! Runtime attachments: which runtimes a session uses, as specified in
2//! `docs/site/specs/runtimes.md`.
3
4use schemars::JsonSchema;
5use serde::{Deserialize, Serialize};
6
7use crate::{AttachmentName, RuntimeId, RuntimeKind};
8
9/// One runtime a session is attached to.
10#[derive(Clone, Debug, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
11#[serde(deny_unknown_fields)]
12pub struct RuntimeAttachment {
13    /// Session-unique attachment name; the tool prefix on collisions.
14    pub name: AttachmentName,
15    /// Template the runtime instance was started from.
16    pub runtime_id: RuntimeId,
17    pub kind: RuntimeKind,
18    /// Canonical workspace path the runtime reported; absent without one.
19    #[serde(default, skip_serializing_if = "Option::is_none")]
20    pub workspace: Option<String>,
21}
22
23/// The attachment set a session's events describe: the fold of its
24/// `runtime_attached` and `runtime_detached` events in log order.
25#[derive(Clone, Debug, Default, Eq, PartialEq)]
26pub struct RuntimeAttachmentFold {
27    /// `None` until the log holds an attach or detach event: such a session
28    /// was created before runtimes and reads as attached to `global` and
29    /// `local`.
30    attachments: Option<Vec<RuntimeAttachment>>,
31}
32
33impl RuntimeAttachmentFold {
34    /// Folds one event payload.
35    pub fn observe(&mut self, payload: &crate::EventPayload) {
36        match payload {
37            crate::EventPayload::RuntimeAttached {
38                name,
39                runtime_id,
40                kind,
41                workspace,
42            } => {
43                let attachments = self.attachments.get_or_insert_with(Vec::new);
44                attachments.retain(|attachment| &attachment.name != name);
45                attachments.push(RuntimeAttachment {
46                    name: name.clone(),
47                    runtime_id: runtime_id.clone(),
48                    kind: kind.clone(),
49                    workspace: workspace.clone(),
50                });
51            }
52            crate::EventPayload::RuntimeDetached { name } => {
53                self.attachments
54                    .get_or_insert_with(Vec::new)
55                    .retain(|attachment| &attachment.name != name);
56            }
57            _ => {}
58        }
59    }
60
61    /// The folded attachments, or `None` for a log created before runtimes.
62    #[must_use]
63    pub fn attachments(&self) -> Option<&[RuntimeAttachment]> {
64        self.attachments.as_deref()
65    }
66}
