1//! Asking for a GitHub star (`docs/site/specs/github-star.md`, client
2//! protocol 1.106.0): once the daemon has been used for 7 days the web page
3//! asks whether the user would star the project, and the daemon stars it
4//! with a GitHub token already on its machine when it has one.
5//!
6//! The daemon decides when the card shows (`star.status`) and records the
7//! user's answer (`star.answer`); `star.changed` tells every owner
8//! connection to ask again. Guests get none of them.
9
10use schemars::JsonSchema;
11use serde::{Deserialize, Serialize};
12
13pub const STAR_STATUS_METHOD: &str = "star.status";
14pub const STAR_ANSWER_METHOD: &str = "star.answer";
15pub const STAR_CHANGED_METHOD: &str = "star.changed";
16
17/// The repository the card asks to star.
18pub const STAR_REPOSITORY: &str = "cookie-agent/cookie-agent";
19/// The repository's page, which the card opens when the daemon cannot star.
20pub const STAR_REPOSITORY_URL: &str = "https://github.com/cookie-agent/cookie-agent";
21
22/// Whether the card shows now.
23#[derive(Clone, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
24#[serde(deny_unknown_fields)]
25pub struct StarStatusParams {}
26
27#[derive(Clone, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
28pub struct StarStatusResult {
29    /// The card shows.
30    pub show: bool,
31    /// The GitHub account the daemon stars as, when it can star: the card
32    /// offers **Star as @login**, and otherwise a link to
33    /// [`STAR_REPOSITORY_URL`].
34    #[serde(default, skip_serializing_if = "Option::is_none")]
35    pub login: Option<String>,
36}
37
38crate::open_enum! {
39    /// What the user did with the card.
40    #[derive(Clone, Copy, Debug, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
41    #[serde(rename_all = "snake_case")]
42    pub enum StarAnswer {
43        /// **Star as @login**: the daemon stars with its token. The card
44        /// never shows again, whether starring worked or not.
45        Star,
46        /// The page opened the repository's page. The card never shows
47        /// again.
48        Opened,
49        /// **Later**, or the card's ×: hidden for 7 days.
50        Later,
51        /// **Don't remind me**: the card never shows again.
52        Dismiss,
53    }
54}
55
56/// Records the user's answer.
57#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
58#[serde(deny_unknown_fields)]
59pub struct StarAnswerParams {
60    pub answer: StarAnswer,
61}
62
63#[derive(Clone, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
64pub struct StarAnswerResult {
65    /// GitHub accepted the star: only for [`StarAnswer::Star`].
66    pub starred: bool,
67}
68
69/// The card's state changed; ask `star.status` again.
70#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
71pub struct StarChangedNotification {}
