1//! Onboarding (`docs/site/specs/default-models-and-onboarding.md` section
2//! 8, client protocol 1.22.0): whether the daemon is set up, what installing
3//! the starter kit would do, and the one call that sets it up. Every client
4//! onboards with these calls and `provider.connect`; the daemon does the
5//! work and never branches on the kind of client asking.

22/// What onboarding still needs. A value this build does not know reads as
23/// `unknown`: something the daemon needs that a client cannot name, which
24/// keeps Finish disabled.
25#[derive(
26    Clone, Copy, Debug, Deserialize, Eq, Hash, JsonSchema, Ord, PartialEq, PartialOrd, Serialize,
27)]
28#[serde(rename_all = "snake_case")]
29#[schemars(extend("x-open" = true))]
30pub enum OnboardingRequirement {
31    /// No model provider is connected.
32    Provider,
33    /// `default_model` is unset or names no model a connected provider
34    /// offers.
35    DefaultModel,
36    /// `small_model` is unset or names no model a connected provider offers.
37    /// Not sent since 1.56.0: `small_model` is optional.
38    SmallModel,
39    /// `[agents] default` names no agent that can run as a root, such as
40    /// before the first refresh of the official agents (client protocol
41    /// 1.56.0).
42    DefaultAgent,
43    #[serde(other)]
44    Unknown,
45}

47/// One of the model fields, as an entry of another field names it (client
48/// protocol 1.69.0). A value this build does not know reads as `unknown`;

57    SmallModel,

63/// One entry of a model field's fallback list (client protocol 1.69.0):
64/// exactly one of `model`, a model key, and `field`, another model field
65/// whose models the entry stands for. `variant` absent means the model's
66/// default variant; beside `field`, it applies to each of that field's
67/// models that has no variant of its own and has a variant of that name
68/// (`docs/site/specs/model-fallback-lists.md`).

70pub struct OnboardingModelEntry {
71    #[serde(default, skip_serializing_if = "Option::is_none")]

73    #[serde(default, skip_serializing_if = "Option::is_none")]
74    pub field: Option<OnboardingModelField>,
75    #[serde(default, skip_serializing_if = "Option::is_none")]
76    pub variant: Option<ConfiguredVariantRef>,
77}

91/// Whether the daemon is set up: the result of `onboarding.get` and the
92/// payload of `onboarding.changed`. Computed from the live configuration and
93/// runtime; nothing is stored for it.
94#[derive(Clone, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
95pub struct OnboardingStatus {
96    /// `missing` is empty.
97    pub ready: bool,
98    #[schemars(length(max = 16))]
99    pub missing: Vec<OnboardingRequirement>,
100    /// The providers that offer at least one available model, sorted.
101    #[schemars(length(max = 4096))]
102    pub providers_connected: Vec<ProviderId>,
103    #[serde(default, skip_serializing_if = "Option::is_none")]
104    pub default_model: Option<OnboardingModel>,
105    #[serde(default, skip_serializing_if = "Option::is_none")]
106    pub small_model: Option<OnboardingModel>,
107    /// `utility_model` (client protocol 1.56.0).
108    #[serde(default, skip_serializing_if = "Option::is_none")]
109    pub utility_model: Option<OnboardingModel>,
110    /// `default_model` as configured, a list of one for a single model, its
111    /// references to other fields kept; absent when unset (client protocol
112    /// 1.69.0). `default_model` above is the first model it reaches.
113    #[serde(default, skip_serializing_if = "Option::is_none")]
114    #[schemars(length(min = 1, max = 32))]
115    pub default_model_list: Option<Vec<OnboardingModelEntry>>,
116    /// `small_model` as configured, like `default_model_list`.
117    #[serde(default, skip_serializing_if = "Option::is_none")]
118    #[schemars(length(min = 1, max = 32))]
119    pub small_model_list: Option<Vec<OnboardingModelEntry>>,
120    /// `utility_model` as configured, like `default_model_list`.
121    #[serde(default, skip_serializing_if = "Option::is_none")]
122    #[schemars(length(min = 1, max = 32))]
123    pub utility_model_list: Option<Vec<OnboardingModelEntry>>,
124    /// The two ways to set the daemon up, each with whether it is done
125    /// (client protocol 1.74.0, `docs/site/specs/acp-onboarding.md`
126    /// section 2). While the ACP path is complete, `missing` no longer lists
127    /// what only the models path needs.
128    #[serde(default, skip_serializing_if = "Option::is_none")]
129    pub paths: Option<OnboardingPaths>,
130    /// The ACP agents onboarding can link (client protocol 1.74.0).
131    #[serde(default, skip_serializing_if = "Vec::is_empty")]
132    #[schemars(length(max = 256))]
133    pub acp_presets: Vec<AcpPreset>,
134    /// The MCP servers the Tools step offers (client protocol 1.74.0).
135    #[serde(default, skip_serializing_if = "Vec::is_empty")]
136    #[schemars(length(max = 256))]
137    pub mcp_presets: Vec<McpPreset>,
138    /// The MCP servers configured now, by name and place (client protocol
139    /// 1.74.0).
140    #[serde(default, skip_serializing_if = "Vec::is_empty")]
141    #[schemars(length(max = 4096))]
142    pub mcp_servers: Vec<OnboardingMcpServer>,
143    /// `[agents] default` as configured; absent when unset, and new
144    /// sessions then start on `primary`, else the first agent that can run
145    /// as a root (client protocol 1.74.0).
146    #[serde(default, skip_serializing_if = "Option::is_none")]
147    pub default_agent: Option<crate::AgentId>,
148}
