1//! Onboarding (`docs/site/specs/default-models-and-onboarding.md` section
2//! 8, client protocol 1.22.0): whether the daemon is set up, what installing
3//! the starter kit would do, and the one call that sets it up. Every client
4//! onboards with these calls and `provider.connect`; the daemon does the
5//! work and never branches on the kind of client asking.
6
7use schemars::JsonSchema;
8use serde::{Deserialize, Serialize};
9
10use crate::{ModelKey, ProviderId};
11use cookie_agent_identity::ConfiguredVariantRef;

13pub const ONBOARDING_GET_METHOD: &str = "onboarding.get";
14pub const ONBOARDING_PREVIEW_METHOD: &str = "onboarding.preview";
15pub const ONBOARDING_RUN_METHOD: &str = "onboarding.run";
16pub const ONBOARDING_CHANGED_METHOD: &str = "onboarding.changed";

18/// The largest kit archive `onboarding.preview` and `onboarding.run` take,
19/// before base64.
20pub const MAX_ONBOARDING_ARCHIVE_BYTES: usize = 1024 * 1024;

22/// What onboarding still needs. A value this build does not know reads as
23/// `unknown`: something the daemon needs that a client cannot name, which
24/// keeps Finish disabled.
25#[derive(
26    Clone, Copy, Debug, Deserialize, Eq, Hash, JsonSchema, Ord, PartialEq, PartialOrd, Serialize,
27)]
28#[serde(rename_all = "snake_case")]
29#[schemars(extend("x-open" = true))]
30pub enum OnboardingRequirement {
31    /// No model provider is connected.
32    Provider,
33    /// `default_model` is unset or names no model a connected provider
34    /// offers.
35    DefaultModel,
36    /// `small_model` is unset or names no model a connected provider offers.
37    SmallModel,
38    #[serde(other)]
39    Unknown,
40}
41
42/// A model and its variant as the configuration holds them: `variant`
43/// absent means the model's default variant, `base` its base variant.
44#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
45pub struct OnboardingModel {
46    pub model: ModelKey,
47    #[serde(default, skip_serializing_if = "Option::is_none")]
48    pub variant: Option<ConfiguredVariantRef>,
49}

51/// Whether the daemon is set up: the result of `onboarding.get` and the
52/// payload of `onboarding.changed`. Computed from the live configuration and
53/// runtime; nothing is stored for it.
54#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
55pub struct OnboardingStatus {
56    /// `missing` is empty.
57    pub ready: bool,
58    #[schemars(length(max = 16))]
59    pub missing: Vec<OnboardingRequirement>,
60    /// The providers that offer at least one available model, sorted.
61    #[schemars(length(max = 4096))]
62    pub providers_connected: Vec<ProviderId>,
63    #[serde(default, skip_serializing_if = "Option::is_none")]
64    pub default_model: Option<OnboardingModel>,
65    #[serde(default, skip_serializing_if = "Option::is_none")]
66    pub small_model: Option<OnboardingModel>,
67}

74/// Where the starter kit comes from.
75#[derive(Clone, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
76#[serde(tag = "type", rename_all = "snake_case", deny_unknown_fields)]
77pub enum OnboardingKitSource {
78    /// The release of the daemon's version, else the latest release.
79    #[default]
80    Release,
81    /// A downloaded `cookie-agent-defaults.tar.gz`, sent whole.
82    Archive {
83        /// The archive's bytes in standard base64, at most
84        /// [`MAX_ONBOARDING_ARCHIVE_BYTES`] decoded.
85        #[schemars(length(min = 1, max = 1_398_104))]
86        data: String,
87    },
88}

99/// What installing a kit file would do. A value this build does not know
100/// reads as `unknown`.
101#[derive(Clone, Copy, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
102#[serde(rename_all = "snake_case")]
103#[schemars(extend("x-open" = true))]
104pub enum OnboardingFileAction {
105    /// The file does not exist: it is written.
106    Write,
107    /// The file is identical to the kit's: nothing is written.
108    Skip,
109    /// The file differs: it stays, and the kit's version is written beside
110    /// it as `<path>.new`.
111    WriteNew,
112    #[serde(other)]
113    Unknown,
114}
115
116/// One kit file against the daemon's user root.
117#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
118pub struct OnboardingKitFile {
119    /// Below the user root: `config.toml` or `agents/<id>.md`.
120    pub path: String,
121    pub action: OnboardingFileAction,
122    /// For `write_new`, a unified diff from the file to the kit's version.
123    #[serde(default, skip_serializing_if = "Option::is_none")]
124    pub diff: Option<String>,
125}

127#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
128pub struct OnboardingPreviewResult {
129    pub kit_version: u32,

157/// `onboarding.run`: sets the model fields given and installs the kit when
158/// asked, as one change. Refused with `onboarding_incomplete`, writing
159/// nothing, when the daemon would not be set up afterwards.
160#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
161#[serde(deny_unknown_fields)]
162pub struct OnboardingRunParams {
163    pub install_kit: bool,
164    /// `release` when absent; ignored without `install_kit`.
165    #[serde(default, skip_serializing_if = "Option::is_none")]
166    pub source: Option<OnboardingKitSource>,
167    #[serde(default, skip_serializing_if = "Option::is_none")]
168    pub default_model: Option<OnboardingModelInput>,
169    #[serde(default, skip_serializing_if = "Option::is_none")]
170    pub small_model: Option<OnboardingModelInput>,
171}
172
173#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
174pub struct OnboardingRunResult {
175    /// Kit files written.
176    #[schemars(length(max = 256))]
177    pub written: Vec<String>,
178    /// Kit files identical to the kit's.
179    #[schemars(length(max = 256))]
180    pub skipped: Vec<String>,
181    /// Kit files that differ, whose kit version was written beside them as
182    /// `<path>.new`.
183    #[schemars(length(max = 256))]
184    pub written_new: Vec<String>,
185    /// The configuration fields set.
186    #[schemars(length(max = 8))]
187    pub settings: Vec<String>,
188    pub status: OnboardingStatus,
189}
