1//! Assets (`docs/site/specs/assets.md`, client protocol 1.67.0): files that
2//! agents write for the user, kept by a session tree or a project and shown
3//! in the conversation as figures.
4//!
5//! Agents write and show assets with the `assets` tool group; clients list,
6//! read, rename, and delete them with the `asset.*` methods here and follow
7//! `assets.changed`. A figure is an [`crate::DisplayBlock::Asset`] block.

23/// The most bytes one asset holds.
24pub const ASSET_MAX_BYTES: u64 = 25 * 1024 * 1024;
25/// The most characters an asset's title has.
26pub const ASSET_TITLE_MAX_CHARS: usize = 200;
27/// The most characters a figure's caption has.
28pub const ASSET_CAPTION_MAX_CHARS: usize = 300;
29/// The most assets one `show_asset` call shows.
30pub const ASSET_SHOW_MAX: usize = 12;
31/// The most bytes one `asset.read` returns.
32pub const ASSET_CHUNK_BYTES: u64 = 1024 * 1024;
33/// The longest side of an `asset.thumbnail` picture, in pixels.
34pub const ASSET_THUMBNAIL_MAX_EDGE: u32 = 768;

36/// Where an asset is kept (section 4.1). A store's assets are listed,
37/// read, and changed together; an asset never moves between stores.
38#[derive(
39    Clone, Debug, Deserialize, Eq, Hash, JsonSchema, Ord, PartialEq, PartialOrd, Serialize,
40)]
41#[serde(tag = "type", rename_all = "snake_case")]
42pub enum AssetStore {
43    /// A session tree's own assets (scope `session`), named by the tree's
44    /// root. They are deleted with the tree.
45    Session { session: SessionId },
46    /// A project's assets (scope `project`).
47    Project { project: ProjectId },
48    /// The assets of every session in no project (scope `project`).
49    Shared,
50}
51
52impl AssetStore {
53    /// The scope an asset of this store has.
54    #[must_use]
55    pub const fn scope(&self) -> AssetScope {
56        match self {
57            Self::Session { .. } => AssetScope::Session,
58            Self::Project { .. } | Self::Shared => AssetScope::Project,
59        }
60    }
61}

63/// Who an asset belongs to: the session tree that wrote it, or every
64/// session of its project (section 4.1).
65#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
66#[serde(rename_all = "snake_case")]
67pub enum AssetScope {
68    #[default]
69    Session,
70    Project,
71}
72
73impl AssetScope {
74    #[must_use]
75    pub const fn as_str(self) -> &'static str {
76        match self {
77            Self::Session => "session",
78            Self::Project => "project",
79        }
80    }
81}

93/// One asset as it is now (section 3).
94#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
95pub struct Asset {
96    pub id: AssetId,
97    pub handle: AssetHandle,
98    pub store: AssetStore,
99    pub title: String,
100    /// The media type of the current content, such as `image/png`,
101    /// `text/html`, or `text/markdown`.
102    pub media_type: String,
103    pub byte_length: u64,
104    /// Lowercase hex SHA-256 of the current content. It changes when the
105    /// asset is written again, so a client knows to read it again.
106    pub sha256: String,
107    /// The name of the file the asset was last written from.
108    #[serde(default, skip_serializing_if = "Option::is_none")]
109    pub filename: Option<String>,
110    pub created: AssetStamp,
111    pub updated: AssetStamp,
112}
113
114impl Asset {
115    /// How the asset reads in one line: its handle, title, type, and size.
116    #[must_use]
117    pub fn summary(&self) -> String {
118        asset_summary(
119            &self.handle,
120            &self.title,
121            &self.media_type,
122            self.byte_length,
123        )
124    }
125
126    /// The asset as a figure shows it.
127    #[must_use]
128    pub fn block(&self) -> AssetBlock {
129        AssetBlock {
130            store: self.store.clone(),
131            asset: self.id,
132            handle: self.handle.clone(),
133            title: self.title.clone(),
134            media_type: self.media_type.clone(),
135            byte_length: self.byte_length,
136            sha256: self.sha256.clone(),
137            filename: self.filename.clone(),
138        }
139    }
140}

142/// `asset_3f9a1c2e "Loss curves" (text/html, 48 KB)`.
143#[must_use]
144pub fn asset_summary(
145    handle: &AssetHandle,
146    title: &str,
147    media_type: &str,
148    byte_length: u64,
149) -> String {
150    format!(
151        "{handle} {title:?} ({media_type}, {})",
152        human_bytes(byte_length)
153    )
154}

156/// An asset shown as a figure (section 10): which asset, and what it was
157/// when the call showed it. Clients show the asset as it is now, read
158/// with `asset.get`, and fall back to these fields when it is gone; a
159/// current `sha256` that differs from this one says it was replaced since.
160#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
161pub struct AssetBlock {
162    pub store: AssetStore,
163    pub asset: AssetId,
164    pub handle: AssetHandle,
165    pub title: String,
166    pub media_type: String,
167    pub byte_length: u64,
168    /// Lowercase hex SHA-256 of the content shown.
169    pub sha256: String,
170    #[serde(default, skip_serializing_if = "Option::is_none")]
171    pub filename: Option<String>,
172}
173
174impl AssetBlock {
175    /// The block's canonical text: its asset's one-line summary.
176    #[must_use]
177    pub fn to_text(&self) -> String {
178        format!(
179            "asset {}",
180            asset_summary(
181                &self.handle,
182                &self.title,
183                &self.media_type,
184                self.byte_length
185            )
186        )
187    }
188}

359/// Whether `title` is a title an asset may have: 1 to
360/// [`ASSET_TITLE_MAX_CHARS`] characters, on one line, with no control
361/// characters.
362#[must_use]
363pub fn is_asset_title(title: &str) -> bool {
364    let count = title.chars().count();
365    (1..=ASSET_TITLE_MAX_CHARS).contains(&count)
366        && !title.chars().any(char::is_control)
367        && !title.trim().is_empty()
368}

376    #[test]
377    fn handles_are_asset_and_eight_hex_digits() {
378        let handle = AssetHandle::random();
379        assert!(AssetHandle::new(handle.as_str()).is_ok(), "{handle}");
380        assert!(AssetHandle::new("asset_3f9a1c2e").is_ok());
381        for invalid in [
382            "asset_3F9A1C2E",
383            "asset_3f9a1c2",
384            "asset_3f9a1c2eff",
385            "assets_3f9a1c2e",
386            "3f9a1c2e",
387        ] {
388            assert!(AssetHandle::new(invalid).is_err(), "{invalid}");
389        }
390    }
391
392    #[test]
393    fn a_store_is_tagged_by_its_kind() {
394        let session = SessionId::new_v7();
395        let store = AssetStore::Session { session };
396        assert_eq!(
397            serde_json::to_value(&store).unwrap(),
398            json!({"type": "session", "session": session})
399        );
400        assert_eq!(
401            serde_json::from_value::<AssetStore>(json!({"type": "shared"})).unwrap(),
402            AssetStore::Shared
403        );
404        assert_eq!(store.scope(), AssetScope::Session);
405        assert_eq!(AssetStore::Shared.scope(), AssetScope::Project);
406    }
407
408    #[test]
409    fn titles_are_one_short_line() {
410        assert!(is_asset_title("Loss curves"));
411        assert!(is_asset_title(&"t".repeat(ASSET_TITLE_MAX_CHARS)));
412        assert!(!is_asset_title(&"t".repeat(ASSET_TITLE_MAX_CHARS + 1)));
413        assert!(!is_asset_title(""));
414        assert!(!is_asset_title("   "));
415        assert!(!is_asset_title("two\nlines"));
416    }
