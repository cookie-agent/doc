1//! The approval reviewer's own log (`docs/site/specs/forever-sessions.md`
2//! section 5.5; client protocol 1.92.0).
3//!
4//! The reviewer of a session keeps its conversation in a log of its own, a
5//! hidden child session filed in the reviewed session's folder, which no
6//! listing, tree, or client reads. Its records after the creation are
7//! [`crate::EventPayload::ReviewerTurnRecorded`], one per review: the `user`
8//! turn and the verdict as they were sent, what the turn holds as entries a
9//! later epoch renders again, and where the next review reads the session's
10//! log from.
11
12use schemars::JsonSchema;
13use serde::{Deserialize, Serialize};
14
15use crate::{ApprovalId, ProducerMessageId, RunId, ToolCallId};
16
17/// One review: the `user` turn of the call under review and the verdict that
18/// answered it, as sent, with what a later review needs.
19#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
20pub struct ReviewerTurn {
21    pub approval_id: ApprovalId,
22    /// The reviewed session's `approval_requested` of the call: the review's
23    /// cut. The turn holds what the log positions after the previous turn's
24    /// cut and up to this one.
25    #[schemars(range(min = 1))]
26    pub cut_seq: u64,
27    /// The reviewed session's `approval_step` that recorded the verdict. A
28    /// revert or fork of that session before it leaves the turn out.
29    #[schemars(range(min = 1))]
30    pub evaluated_seq: u64,
31    /// The name of the tool the reviewed call is of, as a later epoch's
32    /// carried verdicts name it.
33    pub tool: String,
34    /// The `user` turn as sent.
35    pub request: String,
36    /// The `assistant` turn as sent: the verdict as canonical JSON.
37    pub verdict: String,
38    /// The entries this turn renders since the previous turn, in the log's
39    /// order. Empty on an epoch's first turn, whose `epoch` holds them.
40    #[serde(default, skip_serializing_if = "Vec::is_empty")]
41    pub entries: Vec<ReviewerEntry>,
42    /// Set on the first turn of an epoch: the conversation started over
43    /// here, from what the reviewer's log kept, and holds nothing from
44    /// before. The reviewer's log lets go of the turns before it.
45    #[serde(default, skip_serializing_if = "Option::is_none")]
46    pub epoch: Option<ReviewerEpoch>,
47    /// What reading the reviewed session's log on from `cut_seq` needs of
48    /// what came before it.
49    pub cursor: ReviewerCursor,
50}
51
52/// An entry of the reviewer's conversation, as rendered, at its position in
53/// the reviewed session's log: a user message, or an entry of the agent's
54/// trajectory. A delegated session's inherited messages are at position 0.
55#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
56pub struct ReviewerEntry {
57    pub seq: u64,
58    pub text: String,
59    /// A user message, rather than a trajectory entry.
60    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
61    pub message: bool,
62}
63
64/// What an epoch's first turn kept for the epochs after it: the entries a
65/// later fresh turn can still render within its budgets, and what was left
66/// out before them.
67#[derive(Clone, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
68pub struct ReviewerEpoch {
69    /// The user messages a later epoch can keep, in order.
70    #[serde(default, skip_serializing_if = "Vec::is_empty")]
71    pub messages: Vec<ReviewerEntry>,
72    /// The user messages an epoch left out, which no later one keeps.
73    #[serde(default, skip_serializing_if = "Option::is_none")]
74    pub messages_omitted: Option<ReviewerOmitted>,
75    /// The newest trajectory entries, up to the first that puts them over
76    /// the trajectory's budget.
77    #[serde(default, skip_serializing_if = "Vec::is_empty")]
78    pub trajectory: Vec<ReviewerEntry>,
79    /// How many older trajectory entries were let go of.
80    #[serde(default, skip_serializing_if = "is_zero")]
81    pub trajectory_omitted: u64,
82    /// The verdicts before this turn that a later epoch carries forward, at
83    /// most five, oldest first.
84    #[serde(default, skip_serializing_if = "Vec::is_empty")]
85    pub verdicts: Vec<ReviewerVerdict>,
86}

106/// What the reviewed session's log says through a cut, as far as rendering
107/// what comes after it needs.
108#[derive(Clone, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
109pub struct ReviewerCursor {
110    /// How many runs the log named through the cut, in the order each first
111    /// appears: an entry says which run it came in by this count.
112    pub runs: u64,
113    /// The latest of those runs, at most 64, oldest first.
114    #[serde(default, skip_serializing_if = "Vec::is_empty")]
115    pub recent_runs: Vec<ReviewerRun>,
116    /// Calls started and not yet ended through the cut that the trajectory
117    /// shows, with the entry each renders as for each way it can end.
118    #[serde(default, skip_serializing_if = "Vec::is_empty")]
119    pub open_calls: Vec<ReviewerOpenCall>,
120    /// Producer messages accepted and not yet consumed or discarded that the
121    /// conversation shows, at most 256.
122    #[serde(default, skip_serializing_if = "Vec::is_empty")]
123    pub messages: Vec<ReviewerMessage>,
124    /// The runtimes the log attached, by attachment name.
125    #[serde(default, skip_serializing_if = "Vec::is_empty")]
126    pub attachments: Vec<ReviewerAttachment>,
127    /// The `AGENTS.md` texts of the newest run that loaded any. Empty for a
128    /// delegated session, whose `inherited` holds its root's.
129    #[serde(default, skip_serializing_if = "Vec::is_empty")]
130    pub agents_md: Vec<ReviewerAgentsMd>,
131    /// Set for a delegated session: what it inherited of its root.
132    #[serde(default, skip_serializing_if = "Option::is_none")]
133    pub inherited: Option<ReviewerInherited>,
134}

200impl ReviewerTurn {
201    /// Whether the turn is whole: its cut precedes its verdict, and neither
202    /// turn is empty.
203    #[must_use]
204    pub fn is_valid(&self) -> bool {
205        self.cut_seq >= 1
206            && self.evaluated_seq > self.cut_seq
207            && !self.request.is_empty()
208            && !self.verdict.is_empty()
209    }
