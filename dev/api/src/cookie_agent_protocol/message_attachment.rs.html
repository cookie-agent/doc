1//! Attachments in assistant messages (`docs/site/specs/message-attachments.md`,
2//! client protocol 1.73.0): an agent writes `<attach src="…"/>`,
3//! `<attach artifact="…"/>`, or `<attach asset="…"/>` in its reply, and the
4//! engine resolves each tag once, recording an `assistant_attachment` event.
5//!
6//! [`attach_tags`] is the one reading of which tags a text holds, in order,
7//! that the engine and every client share: only an HTML event of the reply's
8//! Markdown counts, so a tag in code is text.
9

18/// The most attachments one message holds; later tags fail.
19pub const ATTACH_MAX_PER_MESSAGE: usize = 12;
20/// The most characters a tag's title has.
21pub const ATTACH_TITLE_MAX_CHARS: usize = 200;

23/// The options every reader of a reply's Markdown parses it with, so that
24/// all of them see the same HTML events and so the same tags.
25#[must_use]
26pub fn markdown_options() -> pulldown_cmark::Options {
27    pulldown_cmark::Options::ENABLE_STRIKETHROUGH
28        | pulldown_cmark::Options::ENABLE_TABLES
29        | pulldown_cmark::Options::ENABLE_TASKLISTS
30        | pulldown_cmark::Options::ENABLE_GFM
31}

33/// What a tag attaches.
34#[derive(Clone, Debug, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
35#[serde(tag = "type", rename_all = "snake_case")]
36pub enum AttachSource {
37    /// A file on a machine: `src`, with the tag's `machine`.
38    Path {
39        path: String,
40        #[serde(default, skip_serializing_if = "Option::is_none")]
41        machine: Option<String>,
42    },
43    /// An artifact of the session tree, by its hex SHA-256 or the start of
44    /// it (8 or more digits) as a file label shows it.
45    Artifact { sha256: String },
46    /// An asset, by its handle.
47    Asset { handle: String },
48}
49
50/// One `<attach …/>` tag as written.
51#[derive(Clone, Debug, Deserialize, Eq, Hash, JsonSchema, PartialEq, Serialize)]
52pub struct AttachTag {
53    pub source: AttachSource,
54    #[serde(default, skip_serializing_if = "Option::is_none")]
55    pub title: Option<String>,
56}
57
58impl AttachTag {
59    /// What the tag names, for a card or a line before it resolves: its
60    /// title, else its file's name, its artifact's digest's start, or its
61    /// asset's handle.
62    #[must_use]
63    pub fn label(&self) -> String {
64        if let Some(title) = &self.title {
65            return title.clone();
66        }
67        match &self.source {
68            AttachSource::Path { path, .. } => path
69                .rsplit(['/', '\\'])
70                .next()
71                .filter(|name| !name.is_empty())
72                .unwrap_or(path)
73                .to_owned(),
74            AttachSource::Artifact { sha256 } => {
75                format!("artifact {}", sha256.get(..12).unwrap_or(sha256))
76            }
77            AttachSource::Asset { handle } => handle.clone(),
78        }
79    }
80}

102/// One piece of a stretch of raw HTML: HTML that stays text, or a tag.
103#[derive(Clone, Debug, Eq, PartialEq)]
104pub enum HtmlPiece<'a> {
105    Text(&'a str),
106    Tag(AttachTag),
107}
108
109/// `raw`, the text of one HTML event, cut into its attach tags and the
110/// rest, in order.
111#[must_use]
112pub fn split_attach_tags(raw: &str) -> Vec<HtmlPiece<'_>> {
113    let mut pieces = Vec::new();
114    let mut rest = raw;
115    let mut text_start = 0usize;
116    let mut offset = 0usize;
117    while let Some(at) = rest.find("<attach") {
118        let start = offset + at;
119        match parse_tag(&raw[start..]) {
120            Some((tag, length)) => {
121                if text_start < start {
122                    pieces.push(HtmlPiece::Text(&raw[text_start..start]));
123                }
124                pieces.push(HtmlPiece::Tag(tag));
125                offset = start + length;
126                text_start = offset;
127            }
128            None => offset = start + "<attach".len(),
129        }
130        rest = &raw[offset..];
131    }
132    if text_start < raw.len() {
133        pieces.push(HtmlPiece::Text(&raw[text_start..]));
134    }
135    pieces
136}

138/// Every attach tag of `markdown`, in order.
139#[must_use]
140pub fn attach_tags(markdown: &str) -> Vec<AttachTag> {
141    use pulldown_cmark::{Event, Parser};
142    if !markdown.contains("<attach") {
143        return Vec::new();
144    }
145    Parser::new_ext(markdown, markdown_options())
146        .filter_map(|event| match event {
147            Event::Html(raw) | Event::InlineHtml(raw) => Some(raw),
148            _ => None,
149        })
150        .flat_map(|raw| {
151            split_attach_tags(&raw)
152                .into_iter()
153                .filter_map(|piece| match piece {
154                    HtmlPiece::Tag(tag) => Some(tag),
155                    HtmlPiece::Text(_) => None,
156                })
157                .collect::<Vec<_>>()
158        })
159        .collect()
160}

162/// The tag `text` starts with, and its length: `<attach`, attributes, and
163/// `/>`, naming exactly one source.
164fn parse_tag(text: &str) -> Option<(AttachTag, usize)> {
165    let mut rest = text.strip_prefix("<attach")?;
166    if !rest.starts_with(|char: char| char.is_whitespace()) {
167        return None;
168    }
169    let (mut src, mut machine, mut artifact, mut asset, mut title) = (None, None, None, None, None);
170    loop {
171        rest = rest.trim_start();
172        if let Some(after) = rest.strip_prefix("/>") {
173            let length = text.len() - after.len();
174            let source = match (src, artifact, asset) {
175                (Some(path), None, None) => AttachSource::Path { path, machine },
176                (None, Some(sha256), None) if machine.is_none() => {
177                    AttachSource::Artifact { sha256 }
178                }
179                (None, None, Some(handle)) if machine.is_none() => AttachSource::Asset { handle },
180                _ => return None,
181            };
182            let title = title
183                .filter(|title: &String| !title.trim().is_empty())
184                .map(|title: String| title.chars().take(ATTACH_TITLE_MAX_CHARS).collect());
185            return Some((AttachTag { source, title }, length));
186        }
187        let name_end = rest.find(|char: char| !(char.is_ascii_alphanumeric() || char == '-'))?;
188        if name_end == 0 {
189            return None;
190        }
191        let name = &rest[..name_end];
192        rest = rest[name_end..]
193            .trim_start()
194            .strip_prefix('=')?
195            .trim_start();
196        let quote = rest
197            .chars()
198            .next()
199            .filter(|quote| matches!(quote, '"' | '\''))?;
200        let close = rest[1..].find(quote)? + 1;
201        let value = unescape(&rest[1..close]);
202        rest = &rest[close + 1..];
203        if value.trim().is_empty() || value.contains(['\n', '\r']) {
204            return None;
205        }
206        let slot = match name {
207            "src" => &mut src,
208            "machine" => &mut machine,
209            "artifact" => &mut artifact,
210            "asset" => &mut asset,
211            "title" => &mut title,
212            // An attribute the tag does not take is passed over.
213            _ => continue,
214        };
215        if slot.replace(value).is_some() {
216            return None;
217        }
218    }
219}

221/// `value` with the entities an author may write in an attribute decoded.
222fn unescape(value: &str) -> String {
223    value
224        .replace("&quot;", "\"")
225        .replace("&#39;", "'")
226        .replace("&apos;", "'")
227        .replace("&lt;", "<")
228        .replace("&gt;", ">")
229        .replace("&amp;", "&")
230}
