28#[derive(Clone, Debug, Eq, PartialEq, Serialize)]
29#[serde(untagged)]
30pub enum PermissionValue {
31    Effect(PermissionEffect),
32    Resources(IndexMap<WildcardPattern, PermissionEffect>),
33}
34
35impl<'de> Deserialize<'de> for PermissionValue {
36    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
37    where
38        D: serde::Deserializer<'de>,
39    {
40        struct Visitor;
41        impl<'de> serde::de::Visitor<'de> for Visitor {
42            type Value = PermissionValue;
43
44            fn expecting(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
45                formatter.write_str("a permission effect or ordered resource map")
46            }
47
48            fn visit_str<E>(self, value: &str) -> Result<Self::Value, E>
49            where
50                E: serde::de::Error,
51            {
52                let effect = match value {
53                    "allow" => PermissionEffect::Allow,
54                    "ask" => PermissionEffect::Ask,
55                    "deny" => PermissionEffect::Deny,
56                    _ => return Err(E::custom("invalid permission effect")),
57                };
58                Ok(PermissionValue::Effect(effect))
59            }
60
61            fn visit_map<A>(self, mut map: A) -> Result<Self::Value, A::Error>
62            where
63                A: serde::de::MapAccess<'de>,
64            {
65                let mut resources = IndexMap::new();
66                while let Some((resource, effect)) =
67                    map.next_entry::<WildcardPattern, PermissionEffect>()?
68                {
69                    if resources.insert(resource, effect).is_some() {
70                        return Err(serde::de::Error::custom("duplicate permission resource"));
71                    }
72                }
73                Ok(PermissionValue::Resources(resources))
74            }
75        }
76        deserializer.deserialize_any(Visitor)
77    }
78}

168fn deserialize_permissions<'de, D>(
169    deserializer: D,
170) -> Result<IndexMap<PermissionKey, PermissionValue>, D::Error>
171where
172    D: serde::Deserializer<'de>,
173{
174    struct Visitor;
175    impl<'de> serde::de::Visitor<'de> for Visitor {
176        type Value = IndexMap<PermissionKey, PermissionValue>;
177
178        fn expecting(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
179            formatter.write_str("an ordered action permission map")
180        }
181
182        fn visit_map<A>(self, mut map: A) -> Result<Self::Value, A::Error>
183        where
184            A: serde::de::MapAccess<'de>,
185        {
186            let mut permissions = IndexMap::new();
187            while let Some((key, value)) = map.next_entry::<PermissionKey, PermissionValue>()? {
188                if permissions.insert(key, value).is_some() {
189                    return Err(serde::de::Error::custom("duplicate permission action"));
190                }
191            }
192            Ok(permissions)
193        }
194    }
195    deserializer.deserialize_map(Visitor)
196}

198impl PermissionValue {
199    /// The rules this value declares under `key`, in declaration order.
200    pub fn rules(&self, key: impl Into<PermissionKey>) -> Vec<PermissionRule> {
201        let key = key.into();
202        match self {
203            Self::Effect(effect) => vec![PermissionRule {
204                action: key.action.clone(),
205                runtime: key.runtime.clone(),
206                resource: WildcardPattern::new("*").expect("static wildcard"),
207                effect: *effect,
208            }],
209            Self::Resources(resources) => resources
210                .iter()
211                .map(|(resource, effect)| PermissionRule {
212                    action: key.action.clone(),
213                    runtime: key.runtime.clone(),
214                    resource: resource.clone(),
215                    effect: *effect,
216                })
217                .collect(),
218        }
219    }
220}

222/// The model of a `models` entry: a model key, or one of the variables
223/// `${parent_model}` (internal agents only), `${default_model}`,
224/// `${small_model}`, and `${utility_model}`, which carry a model and its
225/// variant.
226#[derive(Clone, Debug, Eq, Ord, PartialEq, PartialOrd, Serialize)]
227pub enum AgentModelRef {
228    Model(ModelKey),
229    ParentModel,
230    /// `${default_model}`: the configuration's `default_model`.
231    DefaultModel,
232    /// `${small_model}`: the configuration's `small_model`.
233    SmallModel,
234    /// `${utility_model}`: the configuration's `utility_model`.
235    UtilityModel,
236}
237
238impl AgentModelRef {
239    /// The configuration field a variable names: `default_model`,
240    /// `small_model`, or `utility_model`; `None` for a model key and
241    /// `${parent_model}`.
242    #[must_use]
243    pub const fn config_field(&self) -> Option<&'static str> {
244        match self {
245            Self::DefaultModel => Some("default_model"),
246            Self::SmallModel => Some("small_model"),
247            Self::UtilityModel => Some("utility_model"),
248            Self::Model(_) | Self::ParentModel => None,
249        }
250    }
251}
252
253impl std::str::FromStr for AgentModelRef {
254    type Err = String;
255
256    fn from_str(value: &str) -> Result<Self, Self::Err> {
257        match value {
258            PARENT_MODEL_EXPRESSION => Ok(Self::ParentModel),
259            DEFAULT_MODEL_EXPRESSION => Ok(Self::DefaultModel),
260            SMALL_MODEL_EXPRESSION => Ok(Self::SmallModel),
261            UTILITY_MODEL_EXPRESSION => Ok(Self::UtilityModel),
262            _ => value
263                .parse::<ModelKey>()
264                .map(Self::Model)
265                .map_err(|error| error.to_string()),
266        }
267    }
268}

618fn authored_root_eligible(document: &AgentDocument) -> bool {
619    document.frontmatter.acp.is_none()
620        && document.frontmatter.enabled
621        && matches!(
622            document.frontmatter.mode,
623            AgentMode::Primary | AgentMode::All
624        )
625        && !document.frontmatter.models.is_empty()
626}

628/// An ACP agent's document: the runtime launches the agent, its `models`
629/// name the agent's own model and effort values, and it reads its own
630/// context files.
631fn validate_acp_settings(
632    document: &AgentDocument,
633    acp: &AcpDocumentSettings,
634) -> Result<(), ConfigError> {
635    let frontmatter = &document.frontmatter;
636    let mut models = BTreeSet::new();
637    let field = if !frontmatter.models.is_empty()
638        || acp.models.len() > MAX_LIST
639        || acp.models.iter().any(|choice| {
640            !valid_acp_label(&choice.model)
641                || choice
642                    .variant
643                    .as_deref()
644                    .is_some_and(|variant| !valid_acp_label(variant))
645                || !models.insert(choice.model.as_str())
646        }) {
647        Some("models")
648    } else if frontmatter.agent_md.is_some() {
649        Some("agent_md")
650    } else if frontmatter.mode == AgentMode::Internal {
651        Some("mode")
652    } else if frontmatter.limits != AgentLimits::default() {
653        Some("limits")
654    } else if acp.mode.as_ref().is_some_and(|mode| !valid_acp_label(mode))
655        || acp
656            .config
657            .iter()
658            .any(|(id, value)| !valid_acp_label(id) || !valid_acp_label(value))
659    {
660        Some("acp")
661    } else if acp.hide_tools.as_ref().is_some_and(|patterns| {
662        patterns.len() > MAX_LIST
663            || patterns.iter().any(|pattern| {
664                pattern.is_empty() || pattern.len() > 128 || pattern.chars().any(char::is_control)
665            })
666    }) {
667        Some("acp.hide_tools")
668    } else {
669        None
670    };
671    match field {
672        Some(field) => Err(ConfigError::AgentField {
673            agent: document.id.clone(),
674            field,
675        }),
676        None => Ok(()),
677    }
678}
679
680/// An ACP mode, option, or value ID as a document names it: a nonempty
681/// label of at most 256 bytes without control characters.
682fn valid_acp_label(label: &str) -> bool {
683    !label.is_empty() && label.len() <= 256 && !label.chars().any(char::is_control)
684}
685
686fn validate_agent_document(
687    document: &AgentDocument,
688    all: &BTreeMap<AgentId, AgentDocument>,
689) -> Result<(), ConfigError> {
690    let frontmatter = &document.frontmatter;
691    if matches!(
692        document.id.as_str(),
693        BUILT_IN_APPROVAL_AGENT_ID | BUILT_IN_COMPACTION_AGENT_ID | BUILT_IN_TITLE_AGENT_ID
694    ) && frontmatter.mode != AgentMode::Internal
695    {
696        return Err(ConfigError::AgentField {
697            agent: document.id.clone(),
698            field: "mode",
699        });
700    }
701    if let Some(allowed) =
702        crate::prompt_template::internal_prompt_placeholders(document.id.as_str())
703        && let Some(placeholder) =
704            crate::prompt_template::unknown_placeholder(&document.body, allowed)
705    {
706        return Err(ConfigError::UnknownPromptPlaceholder {
707            agent: document.id.clone(),
708            placeholder,
709        });
710    }
711    if frontmatter.description.is_empty()
712        || frontmatter.description.len() > 512
713        || frontmatter.description.chars().any(char::is_control)
714    {
715        return Err(ConfigError::AgentField {
716            agent: document.id.clone(),
717            field: "description",
718        });
719    }
720    if let Some(acp) = &frontmatter.acp {
721        validate_acp_settings(document, acp)?;
722    }
723    // Only root runs load AGENTS.md, so the override is meaningless on agents
724    // that can never run as a root.
725    if frontmatter.agent_md.is_some()
726        && matches!(frontmatter.mode, AgentMode::Subagent | AgentMode::Internal)
727    {
728        return Err(ConfigError::AgentField {
729            agent: document.id.clone(),
730            field: "agent_md",
731        });
732    }
733    if matches!(frontmatter.mode, AgentMode::Primary)
734        && frontmatter.models.is_empty()
735        && frontmatter.acp.is_none()
736    {
737        return Err(ConfigError::PrimaryFallback(document.id.clone()));
738    }
739    if frontmatter.models.len() > MAX_LIST {
740        return Err(ConfigError::AgentLimit(document.id.clone()));
741    }
742    let mut fallback_models = BTreeSet::new();
743    for fallback in &frontmatter.models {
744        // `${parent_model}` names the model of the run that owns an
745        // internal agent's run; other agents have none.
746        if matches!(fallback.model, AgentModelRef::ParentModel)
747            && frontmatter.mode != AgentMode::Internal
748        {
749            return Err(ConfigError::AgentField {
750                agent: document.id.clone(),
751                field: "models",
752            });
753        }
754        if !fallback_models.insert(fallback.model.clone()) {
755            return Err(ConfigError::AgentField {
756                agent: document.id.clone(),
757                field: "models",
758            });
759        }
760    }
761    if frontmatter.mode != AgentMode::Internal && frontmatter.limits.timeout_ms != 0 {
762        return Err(ConfigError::AgentTimeoutInternalOnly(document.id.clone()));
763    }
764    let permission_rules = frontmatter
765        .permissions
766        .values()
767        .map(|value| match value {
768            PermissionValue::Effect(_) => 1,
769            PermissionValue::Resources(resources) => resources.len(),
770        })
771        .sum::<usize>();
772    if permission_rules > MAX_LIST {
773        return Err(ConfigError::AgentLimit(document.id.clone()));
774    }
775    let delegate_key = PermissionKey::any(PermissionAction::DELEGATE);
776    if let Some(delegate) = frontmatter.permissions.get(&delegate_key) {
777        let rules = delegate.rules(&delegate_key);
778        // Expanded targets land in AgentDescriptor and FrozenDelegationPolicy,
779        // both limited to 256 IDs; validate the deduplicated union exactly as
780        // the runtime expansion produces it.
781        let mut expanded = BTreeSet::new();
782        for rule in rules {
783            if rule.effect == PermissionEffect::Deny {
784                continue;
785            }
786            let eligible = |target_document: &AgentDocument| {
787                target_document.frontmatter.enabled
788                    && matches!(
789                        target_document.frontmatter.mode,
790                        AgentMode::Subagent | AgentMode::All
791                    )
792            };
793            let pattern = rule.resource.as_str();
794            if pattern.contains('*') || pattern.contains('?') {
795                // Wildcard delegation uses the same runtime matching as every
796                // other permission. A pattern that matches no eligible agent
797                // yet may match an ACP agent a runtime reports later: the
798                // engine reports it as a configuration diagnostic.
799                expanded.extend(
800                    all.iter()
801                        .filter(|(id, target_document)| {
802                            crate::simple_wildcard_match(pattern, id.as_str())
803                                && eligible(target_document)
804                        })
805                        .map(|(id, _)| id.clone()),
806                );
807                continue;
808            }
809            let target =
810                AgentId::new(pattern).map_err(|_| ConfigError::Delegation(document.id.clone()))?;
811            // A target without a document may be an ACP agent a runtime
812            // reports later; the engine reports it as a configuration
813            // diagnostic until one does.
814            let Some(target_document) = all.get(&target) else {
815                continue;
816            };
817            if !eligible(target_document) {
818                return Err(ConfigError::IneligibleDelegationTarget {
819                    agent: document.id.clone(),
820                    target: target.clone(),
821                });
822            }
823            expanded.insert(target);
824        }
825        if expanded.len() > MAX_LIST {
826            return Err(ConfigError::AgentLimit(document.id.clone()));
827        }
828    }
829    for (key, value) in &frontmatter.permissions {
830        if value.rules(key).iter().any(|rule| {
831            !crate::agent_document::permission_expressions_valid(
832                key.action.as_str(),
833                rule.resource.as_str(),
834            )
835        }) {
836            return Err(ConfigError::AgentPermissionExpression(document.id.clone()));
837        }
838    }
839    Ok(())
840}
