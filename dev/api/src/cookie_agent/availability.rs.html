3use cookie_agent_protocol::{
4    AgentId, CredentialFieldType, ModelKey, ModelUnavailableKind, ProviderConfigurationState,
5    ProviderDescriptor, ProviderModelCounts, ProviderSupportState, UnavailableModelDescriptor,
6};

8const KINDS: [(ModelUnavailableKind, &str); 4] = [
9    (ModelUnavailableKind::Quarantined, "quarantined"),
10    (ModelUnavailableKind::Unsupported, "unsupported"),
11    (ModelUnavailableKind::NeedsSetup, "need setup"),
12    (ModelUnavailableKind::NeedsCredentials, "need credentials"),
13];
14
15/// Nonzero counts, e.g. `0 usable, 2 quarantined, 1 unsupported`.
16#[must_use]
17pub fn model_counts_text(counts: &ProviderModelCounts) -> String {
18    let mut parts = vec![format!("{} usable", counts.available)];
19    for (kind, label) in KINDS {
20        let count = counts.count(kind);
21        if count > 0 {
22            parts.push(format!("{count} {label}"));
23        }
24    }
25    parts.join(", ")
26}

28/// One unavailable model's reason, e.g. `quarantined: invalid_catalog_model_record`.
29#[must_use]
30pub fn unavailable_model_text(model: &UnavailableModelDescriptor) -> String {
31    let label = match model.kind {
32        ModelUnavailableKind::Quarantined => "quarantined",
33        ModelUnavailableKind::Unsupported => "unsupported",
34        ModelUnavailableKind::NeedsSetup => "needs setup",
35        ModelUnavailableKind::NeedsCredentials => "needs credentials",
36    };
37    model
38        .reason
39        .as_ref()
40        .map_or_else(|| label.to_owned(), |reason| format!("{label}: {reason}"))
41}

43/// Why a live model without tool calling cannot run `agent`, which publishes

45#[must_use]

50/// Why `key` is absent from the live model list, when its provider says so.
51#[must_use]
52pub fn model_unavailable_reason(
53    providers: &[ProviderDescriptor],
54    key: &ModelKey,
55) -> Option<String> {
56    let provider_id = key.provider_id();
57    let provider = providers
58        .iter()
59        .find(|provider| provider.id == provider_id)?;
60    if provider.support.state != ProviderSupportState::Supported {
61        return Some(provider.support.reason.as_ref().map_or_else(
62            || format!("provider `{provider_id}` is not supported"),
63            |reason| format!("provider `{provider_id}` is not supported: {reason}"),
64        ));
65    }
66    let model_id = key.model_id();
67    if let Some(model) = provider
68        .unavailable_models
69        .iter()
70        .find(|model| model.id == model_id)
71    {
72        return Some(unavailable_model_text(model));
73    }
74    (provider.model_counts.available == 0 && provider.model_counts.unavailable() > 0).then(|| {
75        format!(
76            "provider `{provider_id}` has no usable models ({})",
77            model_counts_text(&provider.model_counts)
78        )
79    })
80}

82/// Configured providers that yield no usable model, with their counts.
83#[must_use]
84pub fn unusable_configured_providers(providers: &[ProviderDescriptor]) -> Option<String> {
85    let names = providers
86        .iter()
87        .filter(|provider| {
88            (provider.configuration != ProviderConfigurationState::Unconfigured
89                || provider.durable_connection.is_some())
90                && provider.model_counts.available == 0
91                && provider.model_counts.unavailable() > 0
92        })
93        .map(|provider| {
94            format!(
95                "{} ({})",
96                provider.id,
97                model_counts_text(&provider.model_counts)
98            )
99        })
100        .collect::<Vec<_>>();
101    (!names.is_empty()).then(|| {
102        format!(
103            "configured providers without usable models: {}",
104            names.join("; ")
105        )
106    })
107}

109/// A config.toml alternative that reads the provider's single API key from
110/// the first catalog environment variable. Display-only: cookie never reads
111/// the variable unless the user authors this line.
112#[must_use]
113pub fn env_config_hint(provider: &ProviderDescriptor) -> Option<String> {
114    let single_key = provider.auth_methods.iter().any(|method| {
115        method.credentials.len() == 1
116            && method.credentials[0].credential_type == CredentialFieldType::ApiKey
117    });
118    if !single_key {
119        return None;
120    }
121    let name = provider
122        .environment
123        .iter()
124        .find(|name| name.as_str().contains("KEY"))
125        .or_else(|| provider.environment.first())?;
126    Some(format!(
127        "[providers.{}] source = \"models_dev\", api_key = \"${{env:{name}}}\"",
128        provider.id
129    ))
130}
