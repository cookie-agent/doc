12#[derive(Clone, Debug, Deserialize, Serialize)]
13pub struct SessionToolContext {
14    pub session: SessionId,
15    #[serde(skip)]
16    prompt_delegate_targets: Option<Vec<(AgentId, String)>>,

19impl SessionToolContext {
20    #[must_use]
21    pub const fn new(session: SessionId) -> Self {
22        Self {
23            session,
24            prompt_delegate_targets: None,
25        }
26    }

28    /// Context for composing a run's prompt, carrying the delegate targets
29    /// frozen for that run.
30    #[must_use]
31    pub fn for_prompt_composition(
32        session: SessionId,
33        delegate_targets: Vec<(AgentId, String)>,
34    ) -> Self {
35        Self {
36            session,
37            prompt_delegate_targets: Some(delegate_targets),
38        }
39    }
40
41    /// Delegate targets frozen for the run currently composing its prompt.
42    pub fn prompt_delegate_targets(
43        &self,
44    ) -> Option<impl ExactSizeIterator<Item = (&AgentId, &str)>> {
45        self.prompt_delegate_targets.as_ref().map(|targets| {
46            targets
47                .iter()
48                .map(|(id, description)| (id, description.as_str()))
49        })
50    }
51}

62#[derive(Clone, Debug, Deserialize, PartialEq, Serialize)]
63pub struct ToolSpec {
64    pub name: String,
65    /// What happens when another source publishes the same name.
66    #[serde(default)]
67    pub collision: ToolCollision,
68    /// The permission name the tool's calls are checked under; `None` for a
69    /// tool that needs no permission: always published when its source offers
70    /// it, and every call allowed without evaluating rules or asking.

72    pub permission_name: Option<String>,
73    pub description: String,
74    pub parameters: Value,
75    #[serde(default)]
76    pub concurrency: ToolConcurrency,
77    #[serde(default)]
78    pub result_truncation: ToolResultTruncationPolicy,
79    #[serde(default)]
80    pub output: cookie_agent_protocol::ToolOutputDeclaration,
81    /// The tool supports the background: the host adds a `background`
82    /// argument to its schema and may detach its calls.
83    #[serde(default)]
84    pub background: bool,
85    /// Whether a call runs in the background when the model omits the
86    /// argument. Only valid with `background`.
87    #[serde(default)]
88    pub background_default: bool,
89}

102#[async_trait]
103pub trait ToolProvider: Send + Sync {
104    /// Stable engine-facing identity used for system-prompt provenance labels.
105    fn provider_id(&self) -> &'static str;
106
107    /// Sections resolved once at run admission and frozen into the agent snapshot.
108    fn prompt_sections(&self, _ctx: &SessionToolContext) -> Result<Vec<PromptSection>, ToolError> {
109        Ok(Vec::new())
110    }
111
112    fn tools_for_session(&self, ctx: &SessionToolContext) -> Result<Vec<ToolSpec>, ToolError>;
113
114    /// The source this provider's tools compose under: the prefix a
115    /// colliding tool of it is published with. `None` composes it under the
116    /// attachment that offers it; a plugin names itself.
117    fn composition_source(&self) -> Option<String> {
118        None
119    }
120
121    /// Where `tool_name` comes from when it is not this provider's layer or
122    /// runtime itself: a plugin's tool, or one an MCP server publishes
123    /// through the provider. Clients see it on the call's presentation.
124    fn tool_source(&self, _tool_name: &str) -> Option<ProvidedToolSource> {
125        None
126    }
127    /// Claims a currently undiscovered dynamic tool, allowing preparation to make it available.
128    fn permission_for_unlisted_tool(
129        &self,
130        _tool_name: &str,
131    ) -> Result<Option<&'static str>, ToolError> {
132        Ok(None)
133    }
134    /// The permission name of `tool_name`; `None` when it needs no
135    /// permission.
136    fn get_permission_name(tool_name: &str) -> Result<Option<&'static str>, ToolError>
137    where
138        Self: Sized;
139    /// The permission name and resource of `arguments`. The host asks only
140    /// for a prepared call without a [`crate::PreparedDeclaration`]; a
141    /// provider that cannot answer synchronously, such as a runtime behind
142    /// the runtime protocol, reports both at preparation instead.
143    fn get_permission_resource(
144        &self,
145        tool_name: &str,
146        arguments: &Value,
147    ) -> Result<(Option<&'static str>, Option<String>), ToolError>;
148    fn get_display_argument(&self, name: &str, arguments: &Value) -> Result<String, ToolError>;
149
150    /// How clients show `call`. The host uses a prepared call's
151    /// [`crate::PreparedDeclaration`] when it has one, and asks here
152    /// otherwise, including for a call that failed to prepare.
153    fn presentation(&self, call: &ToolCall) -> PreparedPresentation {
154        match self.get_display_argument(&call.name, &call.arguments) {
155            Ok(display) => tool_presentation(&call.name, &display),
156            Err(_) => tool_title_only(&call.name),
157        }
158    }
159    async fn prepare(
160        &self,
161        ctx: ToolPreparationContext,
162        call: ToolCall,
163    ) -> Result<PreparedTool, ToolError>;
164
165    /// Prepare one turn's parallel call batch. Providers whose calls may target
166    /// the same mutable resource can chain preparation against successive states.
167    async fn prepare_parallel(
168        &self,
169        ctx: ToolPreparationContext,
170        calls: Vec<ToolCall>,
171    ) -> Vec<Result<PreparedTool, ToolError>> {
172        let mut prepared = Vec::with_capacity(calls.len());
173        for call in calls {
174            prepared.push(self.prepare(ctx.clone(), call).await);
175        }
176        prepared
177    }
178}

190/// A call's presentation with only its tool name as the title.
191#[must_use]
192pub fn tool_title_only(name: &str) -> PreparedPresentation {
193    PreparedPresentation::new(safe_display(name), None)
194}
195
196/// A call's presentation: the tool name as the title and `primary` as its
197/// byte-bounded display argument.
198#[must_use]
199pub fn tool_presentation(name: &str, primary: &str) -> PreparedPresentation {
200    let argument = truncate_utf8(primary.trim(), BoundedDisplayText::MAX_BYTES);
201    PreparedPresentation::new(
202        safe_display(name),
203        Some(
204            BoundedDisplayText::new(if argument.is_empty() {
205                "unavailable".into()
206            } else {
207                argument
208            })
209            .expect("byte-bounded display argument"),
210        ),
211    )
212}

214fn safe_display(value: &str) -> SafeDisplayText {
215    let mut output = String::new();
216    for character in value.chars() {
217        let character = if character.is_control() {
218            ' '
219        } else {
220            character
221        };
222        if output.len() + character.len_utf8() > SafeDisplayText::MAX_BYTES {
223            break;
224        }
225        output.push(character);
226    }
227    if output.is_empty() {
228        output.push_str("unavailable");
229    }
230    SafeDisplayText::new(output).expect("sanitized display text")
231}
232
233fn truncate_utf8(value: &str, maximum: usize) -> String {
234    if value.len() <= maximum {
235        return value.to_owned();
236    }
237    let mut end = maximum;
238    while end > 0 && !value.is_char_boundary(end) {
239        end -= 1;
240    }
241    value[..end].to_owned()
242}

250    #[test]
251    fn tool_concurrency_defaults_to_exclusive_when_omitted() {
252        let spec: ToolSpec = serde_json::from_value(serde_json::json!({
253            "name":"external_tool",
254            "permission_name":"external",
255            "description":"External tool",
256            "parameters":{"type":"object"}
257        }))
258        .expect("tool spec without concurrency");
259        assert_eq!(spec.concurrency, ToolConcurrency::Exclusive);
260    }
261
262    #[test]
263    fn presentation_byte_caps_arguments_and_preserves_controls() {
264        let presentation = tool_presentation("bash", "cat secret.txt; TOKEN=visible\u{1b}");
265        assert_eq!(presentation.title.as_str(), "bash");
266        let argument = presentation.primary_argument.unwrap();
267        assert_eq!(argument.as_str(), "cat secret.txt; TOKEN=visible\u{1b}");
268
269        let oversized = "x".repeat(BoundedDisplayText::MAX_BYTES + 8);
270        let argument = tool_presentation("bash", &oversized)
271            .primary_argument
272            .unwrap();
273        assert_eq!(argument.as_str().len(), BoundedDisplayText::MAX_BYTES);
274
275        // Three-byte graphemes do not divide the cap evenly: the cut stays on a
276        // character boundary rather than yielding an invalid UTF-8 prefix.
277        let multibyte = "→".repeat(BoundedDisplayText::MAX_BYTES);
278        let argument = tool_presentation("bash", &multibyte)
279            .primary_argument
280            .unwrap();
281        assert_eq!(
282            argument.as_str(),
283            "→".repeat(BoundedDisplayText::MAX_BYTES / 3)
284        );
285
286        assert_eq!(
287            tool_presentation("bash", "  \n\t ")
288                .primary_argument
289                .unwrap()
290                .as_str(),
291            "unavailable"
292        );
293    }
