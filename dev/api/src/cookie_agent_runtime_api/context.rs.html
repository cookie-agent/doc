1//! Workspace context a coding runtime contributes at run admission: the
2//! `AGENTS.md` lookup of `docs/site/specs/runtimes.md` section 8.
3//!
4//! The coding runtime runs it on its own machine and answers
5//! `runtime/context` with the result; the daemon runs the same lookup for an
6//! in-process coding runtime.
7
8use std::{
9    fs::File,
10    io::Read as _,
11    path::{Path, PathBuf},
12};
13
14use cookie_agent_protocol::{AgentMdEntry, AgentMdSkipped, SafeDisplayText};
15
16/// Largest `AGENTS.md` file the lookup reads; larger files are skipped and
17/// reported.
18pub const AGENT_MD_MAX_BYTES: u64 = AgentMdEntry::MAX_CONTENT_BYTES as u64;
19
20/// The `AGENTS.md` files found in one workspace.
21#[derive(Clone, Debug, Default, Eq, PartialEq)]
22pub struct AgentMdContext {
23    /// Files read, in prompt order.
24    pub entries: Vec<AgentMdEntry>,
25    /// Files skipped because they exceed the size limit.
26    pub skipped: Vec<AgentMdSkipped>,
27}
28
29/// An `AGENTS.md` file that exists but could not be read.
30#[derive(Debug, thiserror::Error)]
31#[error("AGENTS.md context read failed at {}: {source}", path.display())]
32pub struct AgentMdError {
33    pub path: PathBuf,
34    #[source]
35    pub source: std::io::Error,
36}
37
38/// Applies the `AGENTS.md` lookup to `workspace`: the preset's
39/// `.cookie-agent/agents/<preset>/AGENTS.md`, else
40/// `.cookie-agent/agents/AGENTS.md`, then `<workspace>/AGENTS.md`. Files over
41/// `max_bytes` (at most [`AGENT_MD_MAX_BYTES`]) are skipped.
42///
43/// # Errors
44///
45/// Fails when a file exists but cannot be read or is not UTF-8.
46pub fn load_agent_md(
47    workspace: &Path,
48    preset: Option<&str>,
49    max_bytes: u64,
50) -> Result<AgentMdContext, AgentMdError> {
51    let max_bytes = max_bytes.min(AGENT_MD_MAX_BYTES);
52    let agents_dir = workspace.join(".cookie-agent").join("agents");
53    let project_path = match preset {
54        Some(preset) => {
55            let preset_path = agents_dir.join(preset).join("AGENTS.md");
56            if preset_path.is_file() {
57                preset_path
58            } else {
59                agents_dir.join("AGENTS.md")
60            }
61        }
62        None => agents_dir.join("AGENTS.md"),
63    };
64    let mut context = AgentMdContext::default();
65    for path in [project_path, workspace.join("AGENTS.md")] {
66        if let Some(entry) = read_agent_md_file(&path, workspace, max_bytes)? {
67            context.entries.push(entry);
68        } else if let Some(skip) = skipped_agent_md_file(&path, max_bytes) {
69            context.skipped.push(skip);
70        }
71    }
72    Ok(context)
73}
74
75fn read_agent_md_file(
76    path: &Path,
77    workspace: &Path,
78    max_bytes: u64,
79) -> Result<Option<AgentMdEntry>, AgentMdError> {
80    let io_error = |source| AgentMdError {
81        path: path.to_owned(),
82        source,
83    };
84    let mut file = match File::open(path) {
85        Ok(file) => file,
86        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(None),
87        Err(source) => return Err(io_error(source)),
88    };
89    let metadata_len = file.metadata().map_err(io_error)?.len();
90    if metadata_len > max_bytes {
91        return Ok(None);
92    }
93    let mut bytes = Vec::with_capacity(metadata_len as usize);
94    file.read_to_end(&mut bytes).map_err(io_error)?;
95    let final_metadata_len = file.metadata().map_err(io_error)?.len();
96    let observed_bytes = metadata_len.max(final_metadata_len).max(bytes.len() as u64);
97    if observed_bytes > max_bytes {
98        return Ok(None);
99    }
100    let content = String::from_utf8(bytes).map_err(|error| {
101        io_error(std::io::Error::new(
102            std::io::ErrorKind::InvalidData,
103            error.utf8_error(),
104        ))
105    })?;
106    Ok(Some(AgentMdEntry {
107        source: SafeDisplayText::new(relative_display_path(path, workspace))
108            .expect("AGENTS.md relative path is bounded"),
109        byte_length: observed_bytes,
110        content,
111    }))
112}
113
114fn skipped_agent_md_file(path: &Path, max_bytes: u64) -> Option<AgentMdSkipped> {
115    let byte_length = std::fs::metadata(path).ok()?.len();
116    (byte_length > max_bytes).then(|| AgentMdSkipped {
117        path: SafeDisplayText::new(path.to_string_lossy().into_owned())
118            .expect("AGENTS.md absolute path is bounded"),
119        byte_length,
120    })
121}

123/// Display path relative to the runtime's workspace when possible, so
124/// model-facing context (e.g. AGENTS.md `from=` attributes) reinforces
125/// workspace-relative path style.
126/// How an AGENTS.md file is named: relative to the workspace with `/`
127/// between components on every platform, as the model reads paths in the
128/// workspace; else its absolute path as the platform spells it.
129fn relative_display_path(path: &Path, workspace: &Path) -> String {
130    path.strip_prefix(workspace).map_or_else(
131        |_| path.to_string_lossy().into_owned(),
132        |relative| {
133            relative
134                .components()
135                .map(|component| component.as_os_str().to_string_lossy())
136                .collect::<Vec<_>>()
137                .join("/")
138        },
139    )
140}
