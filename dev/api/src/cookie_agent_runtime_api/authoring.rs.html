25/// The JSON Schema of a tool's parameters type. Generated once per type:
26/// providers publish their tools at every model step.
27#[must_use]
28pub fn schema<T: JsonSchema + 'static>() -> serde_json::Value {
29    memoized::<T>(false, || {
30        serde_json::to_value(schemars::schema_for!(T)).expect("tool schemas serialize")
31    })
32}

34/// The schema of arguments [`parse_args_strict`] reads: every object that
35/// lists its properties refuses others, including types history also stores,
36/// which ignore them there.
37#[must_use]
38pub fn strict_schema<T: JsonSchema + 'static>() -> serde_json::Value {
39    memoized::<T>(true, || {
40        let mut schema = schema::<T>();
41        close_objects(&mut schema);
42        schema
43    })
44}

46/// The schema `build` makes for `T`, `strict` or not, built on first use and
47/// cloned from then on.
48fn memoized<T: 'static>(
49    strict: bool,
50    build: impl FnOnce() -> serde_json::Value,
51) -> serde_json::Value {
52    type Schemas = HashMap<(TypeId, bool), serde_json::Value>;
53    static SCHEMAS: OnceLock<RwLock<Schemas>> = OnceLock::new();
54    let schemas = SCHEMAS.get_or_init(RwLock::default);
55    let key = (TypeId::of::<T>(), strict);
56    if let Some(schema) = schemas
57        .read()
58        .unwrap_or_else(|poisoned| poisoned.into_inner())
59        .get(&key)
60    {
61        return schema.clone();
62    }
63    // Built outside the lock; two first uses may both build it.
64    let schema = build();
65    schemas
66        .write()
67        .unwrap_or_else(|poisoned| poisoned.into_inner())
68        .entry(key)
69        .or_insert(schema)
70        .clone()
71}
72fn close_objects(schema: &mut serde_json::Value) {
73    let Some(object) = schema.as_object_mut() else {
74        return;
75    };
76    if object.contains_key("properties") && !object.contains_key("additionalProperties") {
77        object.insert(
78            "additionalProperties".into(),
79            serde_json::Value::Bool(false),
80        );
81    }
82    for (keyword, value) in object {
83        match keyword.as_str() {
84            "properties" | "$defs" | "definitions" => {
85                if let Some(schemas) = value.as_object_mut() {
86                    schemas.values_mut().for_each(close_objects);
87                }
88            }
89            "oneOf" | "anyOf" | "allOf" | "prefixItems" => {
90                if let Some(schemas) = value.as_array_mut() {
91                    schemas.iter_mut().for_each(close_objects);
92                }
93            }
94            "items" | "additionalProperties" | "not" => close_objects(value),
95            _ => {}
96        }
97    }
98}

100/// A tool result title: control characters replaced, cut to the display
101/// bound, `Tool result` when nothing is left.
102#[must_use]
103pub fn safe_title(value: impl AsRef<str>) -> SafeDisplayText {
104    let mut safe = String::new();
105    for character in value.as_ref().chars() {
106        let character = if character.is_control() {
107            ' '
108        } else {
109            character
110        };
111        if safe.len() + character.len_utf8() > SafeDisplayText::MAX_BYTES {
112            break;
113        }
114        safe.push(character);
115    }
116    SafeDisplayText::new(if safe.is_empty() {
117        "Tool result".to_owned()
118    } else {
119        safe
120    })
121    .expect("sanitized tool title")
122}

124/// Deserializes a call's arguments, with an error the model can act on.
125pub fn parse_args<T: DeserializeOwned>(
126    tool: &str,
127    value: serde_json::Value,
128) -> Result<T, ToolError> {
129    serde_json::from_value(value).map_err(|error| {
130        tool_error(format!(
131            "The {tool} tool was called with invalid arguments: {error}.\nPlease rewrite the input so it satisfies the expected schema."
132        ))
133    })
134}

136/// Like [`parse_args`], for arguments that embed a type history also stores:
137/// such a type ignores unknown members, so they are refused here against the
138/// arguments' schema instead.
139pub fn parse_args_strict<T>(tool: &str, value: serde_json::Value) -> Result<T, ToolError>
140where
141    T: DeserializeOwned + schemars::JsonSchema + 'static,
142{
143    match cookie_agent_protocol::strict::decode_command(value) {
144        Ok(arguments) => Ok(arguments),
145        Err(error) => Err(tool_error(format!(
146            "The {tool} tool was called with invalid arguments: {error}.\nPlease rewrite the input so it satisfies the expected schema."
147        ))),
148    }
149}

151/// The identity of a prepared operation of `tool`: its normalized
152/// arguments, capabilities, resources, and execution context.
153pub fn prepared_operation<T: Serialize>(
154    tool: &str,
155    normalized_arguments: &T,
156    capabilities: Vec<(PermissionAction, &str)>,
157    resources: Vec<PreparedApprovalResource>,
158    execution_context_bytes: &[u8],
159) -> Result<PreparedOperationIdentity, ToolError> {
160    let arguments = serde_json::to_vec(normalized_arguments).map_err(tool_error)?;
161    let capabilities = capabilities
162        .into_iter()
163        .map(|(action, operation)| {
164            Ok(ApprovalCapability {
165                action,
166                operation: PreparedCapabilityOperation::new(format!("{tool}:{operation}"))
167                    .map_err(tool_error)?,
168            })
169        })
170        .collect::<Result<Vec<_>, ToolError>>()?;
171    PreparedOperationIdentity::new(
172        Sha256Digest::of_bytes(&arguments),
173        capabilities,
174        resources,
175        Sha256Digest::of_bytes(execution_context_bytes),
176    )
177    .map_err(tool_error)
178}

180/// One approval resource of a prepared operation. `bash` resources approve
181/// by command prefix; every other action approves the exact resource.
182pub fn prepared_resource(
183    action: PermissionAction,
184    logical_kind: &str,
185    stable_label_bytes: &[u8],
186    binding_bytes: &[u8],
187    lifetime: PreparedBindingLifetime,
188    source: ApprovalResourceSource,
189) -> Result<PreparedApprovalResource, ToolError> {
190    let label = Sha256Digest::of_bytes(stable_label_bytes);
191    let mut complete_binding = Vec::new();
192    complete_binding.extend_from_slice(logical_kind.as_bytes());
193    complete_binding.push(0);
194    complete_binding.extend_from_slice(stable_label_bytes);
195    complete_binding.push(0);
196    complete_binding.extend_from_slice(binding_bytes);
197    let boundary = if action == PermissionAction::BASH {
198        ApprovalBoundary::CommandPrefix {
199            prefix: String::from_utf8_lossy(stable_label_bytes).into_owned(),
200        }
201    } else {
202        ApprovalBoundary::Exact
203    };
204    Ok(PreparedApprovalResource {
205        capability: action,
206        canonical: PreparedResourceIdentity::new(format!("{logical_kind}:{}", label.as_str()))
207            .map_err(tool_error)?,
208        binding_digest: PreparedResourceDigest::from_canonical_binding_bytes(&complete_binding),
209        binding_lifetime: lifetime,
210        boundary,
211        source,
212    })
