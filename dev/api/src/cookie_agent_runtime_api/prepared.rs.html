14/// Completion supplies output once, or explicitly finalizes already accepted deltas.
15#[derive(Clone, Debug)]
16pub struct ToolCompletion {
17    pub output: cookie_agent_protocol::ToolCompletionOutput,
18    pub result: ToolResult,
19    pub failed: bool,
20}
21
22impl ToolCompletion {
23    #[cfg(any(test, feature = "test-support"))]
24    pub fn into_result_for_test(self) -> Result<ToolResult, ToolError> {
25        let mut result = self.result;
26        match self.output {
27            cookie_agent_protocol::ToolCompletionOutput::Single { text } => result.output = text,
28            _ => {
29                return Err(ToolError::execution(
30                    "streamed tests must use runtime capture",
31                ));
32            }
33        }
34        Ok(result)
35    }
36
37    /// A completion of `result`'s output. A result without display shows
38    /// its output.
39    pub fn single(mut result: ToolResult) -> Self {
40        let text = std::mem::take(&mut result.output);
41        if result.display.is_empty() {
42            result.display = bounded_tool_display(&text);
43        }
44        Self {
45            output: cookie_agent_protocol::ToolCompletionOutput::Single { text },
46            result,
47            failed: false,
48        }
49    }
50
51    pub fn streamed(result: ToolResult) -> Self {
52        Self {
53            output: cookie_agent_protocol::ToolCompletionOutput::Streamed,
54            result,
55            failed: false,
56        }
57    }
58}

60/// `text` as a result's display: one `text` block, sanitized for display and
61/// cut to whole lines within the display budget. An empty text shows
62/// nothing.
63#[must_use]
64pub fn bounded_tool_display(text: &str) -> Vec<Open<DisplayBlock>> {
65    cookie_agent_protocol::fit_result_display(cookie_agent_protocol::text_display(
66        sanitize_tool_display(text, usize::MAX),
67    ))
68}
69
70/// The head of `text` a display block shows: its first lines within
71/// `max_bytes` (the start of the first line when even that is longer), and
72/// the count of the lines it leaves out. A display then fits the head
73/// without measuring all of a long text.
74#[must_use]
75pub fn text_head(text: &str, max_bytes: usize) -> (&str, Option<u64>) {
76    if text.len() <= max_bytes {
77        return (text, None);
78    }
79    let mut end = max_bytes;
80    while !text.is_char_boundary(end) {
81        end -= 1;
82    }
83    let end = text[..end].rfind('\n').map_or(end, |newline| newline + 1);
84    let rest = &text[end..];
85    let omitted = rest.bytes().filter(|byte| *byte == b'\n').count() as u64
86        + u64::from(!rest.ends_with('\n'));
87    (&text[..end], Some(omitted))
88}

90/// `text` with control characters other than newline and tab replaced by
91/// spaces, cut on a character boundary to at most `maximum` bytes.
92#[must_use]
93pub fn sanitize_tool_display(text: &str, maximum: usize) -> String {
94    let mut display = String::new();
95    for character in text.chars() {
96        let character = if character.is_control() && !matches!(character, '\n' | '\t') {
97            ' '
98        } else {
99            character
100        };
101        if display.len() + character.len_utf8() > maximum {
102            break;
103        }
104        display.push(character);
105    }
106    display
107}

124#[async_trait]
125pub trait PreparedExecutor: Send + Sync {
126    /// Checks that the prepared operation can still run as approved. Runs
127    /// when an approval is answered and again, under the per-target
128    /// serialization lock, immediately before execution.
129    async fn revalidate(&self) -> Result<(), ToolError> {
130        Ok(())
131    }
132
133    async fn execute(
134        self: Box<Self>,
135        context: ToolExecutionContext,
136    ) -> Result<ToolCompletion, ToolError>;
137}

139/// The executor of a prepared call, taken exactly once by whoever runs it.
140pub type PreparedExecutorCell = Arc<tokio::sync::Mutex<Option<Box<dyn PreparedExecutor>>>>;

142/// What a provider that cannot answer the sync [`crate::ToolProvider`]
143/// metadata methods, such as a runtime behind the runtime protocol, reports
144/// about one prepared call.
145#[derive(Clone, Debug, PartialEq)]
146pub struct PreparedDeclaration {
147    /// The permission name the call is checked under; `None` when the tool
148    /// needs no permission.
149    pub permission_name: Option<String>,
150    /// The permission resource of the call's normalized arguments.
151    pub permission_resource: Option<String>,
152    /// How clients show the call.
153    pub presentation: PreparedPresentation,
154}

156/// A prepared call: its approval identity, normalized arguments, and
157/// executor.
158///
159/// A provider the host owns states the call's approval identity itself
160/// ([`Self::new`]). Every other provider, such as a runtime's tools or a
161/// plugin's, only declares the call ([`Self::declared`]): the host builds its
162/// identity from the tool's permission and resource and from the tool it
163/// knows the call goes to, and sets it with [`Self::identify`] before
164/// anything reads it.
165pub struct PreparedTool {
166    /// `None` until the host identifies a declared call.
167    operation: Option<PreparedOperationIdentity>,
168    policy_labels: Vec<Option<String>>,
169    normalized_arguments: serde_json::Value,
170    serialization_key: Option<PreparedSerializationKey>,
171    executor: PreparedExecutorCell,
172    declaration: Option<PreparedDeclaration>,
173    presentation: Option<PreparedPresentation>,
174}
175
176impl PreparedTool {
177    pub fn new(
178        operation: PreparedOperationIdentity,
179        normalized_arguments: serde_json::Value,
180        serialization_key: Option<PreparedSerializationKey>,
181        executor: Box<dyn PreparedExecutor>,
182    ) -> Result<Self, ToolError> {
183        if operation.resources().is_empty() {
184            return Err(ToolError::execution(
185                "prepared tool requires at least one permission resource",
186            ));
187        }
188        if normalized_arguments.is_null() {
189            return Err(ToolError::execution(
190                "prepared normalized arguments must not be null",
191            ));
192        }
193        let policy_labels = operation
194            .resources()
195            .iter()
196            .map(|resource| Some(resource.canonical.as_str().to_owned()))
197            .collect();
198        Ok(Self {
199            operation: Some(operation),
200            policy_labels,
201            normalized_arguments,
202            serialization_key,
203            executor: Arc::new(tokio::sync::Mutex::new(Some(executor))),
204            declaration: None,
205            presentation: None,
206        })
207    }
208
209    /// A call whose approval identity the host builds: the provider states
210    /// only its normalized arguments, and its permission resource as
211    /// [`crate::ToolProvider::get_permission_resource`] or a
212    /// [`PreparedDeclaration`] reports it.
213    pub fn declared(
214        normalized_arguments: serde_json::Value,
215        serialization_key: Option<PreparedSerializationKey>,
216        executor: Box<dyn PreparedExecutor>,
217    ) -> Result<Self, ToolError> {
218        if normalized_arguments.is_null() {
219            return Err(ToolError::execution(
220                "prepared normalized arguments must not be null",
221            ));
222        }
223        Ok(Self {
224            operation: None,
225            policy_labels: Vec::new(),
226            normalized_arguments,
227            serialization_key,
228            executor: Arc::new(tokio::sync::Mutex::new(Some(executor))),
229            declaration: None,
230            presentation: None,
231        })
232    }
233
234    /// Whether the host has yet to build the call's identity.
235    #[must_use]
236    pub const fn is_declared(&self) -> bool {
237        self.operation.is_none()
238    }
239
240    /// Sets the identity the host built for a declared call: one resource,
241    /// whose policy label is `resource` (`None` for the loose marker).
242    pub fn identify(
243        mut self,
244        operation: PreparedOperationIdentity,
245        resource: Option<String>,
246    ) -> Result<Self, ToolError> {
247        if self.operation.is_some() {
248            return Err(ToolError::execution(
249                "a prepared call that states its identity is not identified again",
250            ));
251        }
252        if operation.resources().len() != 1 {
253            return Err(ToolError::execution(
254                "a declared call's identity has exactly one resource",
255            ));
256        }
257        if resource.as_ref().is_some_and(String::is_empty) {
258            return Err(ToolError::execution(
259                "permission resource must not be empty",
260            ));
261        }
262        self.operation = Some(operation);
263        self.policy_labels = vec![resource];
264        Ok(self)
265    }
266
267    /// Attaches the permission and presentation the provider reported while
268    /// preparing; the host then uses them instead of asking the provider.
269    #[must_use]
270    pub fn with_declaration(mut self, declaration: PreparedDeclaration) -> Self {
271        self.declaration = Some(declaration);
272        self
273    }
274
275    /// Attaches how clients show the call, as the provider knows it only
276    /// once it prepared the call (such as a diff of the file it read). The
277    /// host uses it instead of [`crate::ToolProvider::presentation`].
278    #[must_use]
279    pub fn with_presentation(mut self, presentation: PreparedPresentation) -> Self {
280        self.presentation = Some(presentation);
281        self
282    }
283
284    /// How clients show the call, as the provider prepared it. A
285    /// [`PreparedDeclaration`]'s presentation takes precedence.
286    #[must_use]
287    pub fn presentation(&self) -> Option<&PreparedPresentation> {
288        self.declaration
289            .as_ref()
290            .map(|declaration| &declaration.presentation)
291            .or(self.presentation.as_ref())
292    }
293
294    /// The permission and presentation reported at preparation, if any.
295    #[must_use]
296    pub const fn declaration(&self) -> Option<&PreparedDeclaration> {
297        self.declaration.as_ref()
298    }
299
300    /// The call's approval identity.
301    ///
302    /// # Panics
303    ///
304    /// For a declared call the host has not identified yet.
305    #[must_use]
306    pub const fn operation(&self) -> &PreparedOperationIdentity {
307        match &self.operation {
308            Some(operation) => operation,
309            None => panic!("the host identifies a declared call before reading its identity"),
310        }
311    }
312
313    pub fn with_policy_labels(mut self, labels: Vec<String>) -> Result<Self, ToolError> {
314        let Some(operation) = &self.operation else {
315            return Err(ToolError::execution(
316                "a declared call's policy label is its permission resource",
317            ));
318        };
319        if labels.is_empty() || labels.len() != operation.resources().len() {
320            return Err(ToolError::execution(
321                "prepared policy labels must cover every resource",
322            ));
323        }
324        for (resource, label) in operation.resources().iter().zip(&labels) {
325            let expected = Sha256Digest::of_bytes(label.as_bytes());
326            if resource
327                .canonical
328                .as_str()
329                .rsplit_once(':')
330                .is_none_or(|(_, digest)| digest != expected.as_str())
331            {
332                return Err(ToolError::execution(
333                    "prepared policy label does not match its immutable resource identity",
334                ));
335            }
336        }
337        self.policy_labels = labels.into_iter().map(Some).collect();
338        Ok(self)
339    }
340
341    pub fn with_permission_resource(mut self, resource: Option<String>) -> Result<Self, ToolError> {
342        if resource.as_ref().is_some_and(String::is_empty) {
343            return Err(ToolError::execution(
344                "permission resource must not be empty",
345            ));
346        }
347        self.policy_labels.fill(resource);
348        Ok(self)
349    }

373    #[cfg(feature = "test-support")]
374    pub async fn execute_for_test(
375        self,
376        context: ToolExecutionContext,
377    ) -> Result<ToolResult, ToolError> {
378        let executor = self
379            .executor
380            .lock()
381            .await
382            .take()
383            .ok_or_else(|| ToolError::execution("prepared executor was already consumed"))?;
384        executor.execute(context).await?.into_result_for_test()
385    }
