1//! How a dropped connection is bridged (`docs/site/specs/runtimes.md`
2//! section 6.4).
3//!
4//! The same policy applies to MCP servers (in the daemon's global runtime and
5//! in coding runtimes) and to runtime instances: when a connected one drops,
6//! its host reconnects it (restarting the process for stdio servers and
7//! runtimes) a bounded number of times with growing backoff. While it reconnects its tools stay published and calls fail fast
8//! with `runtime_unavailable`, so a brief drop never changes the tool list.
9//! When every attempt fails, the caller drops the tools and reports the
10//! failure to clients.
11//!
12//! [`ReconnectPolicy::run`] owns only the attempt loop; the caller supplies
13//! one connection attempt and decides what the outcome means for its state.
14
15use std::{future::Future, sync::Arc, time::Duration};
16
17use tokio_util::sync::CancellationToken;
18
19/// The waits before each reconnect attempt; their count is the number of
20/// attempts.
21#[derive(Clone, Debug)]
22pub struct ReconnectPolicy {
23    delays: Arc<[Duration]>,
24}
25
26impl Default for ReconnectPolicy {
27    /// Three attempts after 1 s, 2 s, and 4 s.
28    fn default() -> Self {
29        let backoff =
30            cookie_agent_identity::Backoff::new(Duration::from_secs(1), Duration::from_secs(4));
31        Self::with_delays(backoff.delays().take(3).collect::<Vec<_>>())
32    }
33}
34
35/// How a reconnect loop ended.
36#[derive(Debug, Eq, PartialEq)]
37pub enum ReconnectOutcome<T> {
38    /// An attempt succeeded.
39    Reconnected(T),
40    /// Every attempt failed; `last_error` is the final attempt's reason.
41    Exhausted { attempts: usize, last_error: String },
42    /// The loop was cancelled (shutdown, or the connection was replaced).
43    Cancelled,
44}
45
46impl ReconnectPolicy {
47    /// A policy that waits `delays[i]` before attempt `i + 1`.
48    pub fn with_delays(delays: impl Into<Arc<[Duration]>>) -> Self {
49        Self {
50            delays: delays.into(),
51        }
52    }
53
54    /// Runs up to one attempt per delay, waiting that delay before each.
55    /// `attempt` receives the 1-based attempt number and returns the
56    /// connection or a human-readable reason.
57    pub async fn run<T, F, Fut>(
58        &self,
59        cancel: &CancellationToken,
60        mut attempt: F,
61    ) -> ReconnectOutcome<T>
62    where
63        F: FnMut(usize) -> Fut,
64        Fut: Future<Output = Result<T, String>>,
65    {
66        let mut last_error = String::new();
67        for (index, delay) in self.delays.iter().enumerate() {
68            tokio::select! {
69                () = tokio::time::sleep(*delay) => {}
70                () = cancel.cancelled() => return ReconnectOutcome::Cancelled,
71            }
72            let result = tokio::select! {
73                result = attempt(index + 1) => result,
74                () = cancel.cancelled() => return ReconnectOutcome::Cancelled,
75            };
76            match result {
77                Ok(value) => return ReconnectOutcome::Reconnected(value),
78                Err(error) => last_error = error,
79            }
80        }
81        ReconnectOutcome::Exhausted {
82            attempts: self.delays.len(),
83            last_error,
84        }
85    }
86}

88#[cfg(test)]
89mod tests {
90    use std::{
91        sync::atomic::{AtomicUsize, Ordering},
92        time::Duration,
93    };
94
95    use tokio_util::sync::CancellationToken;
96
97    use super::{ReconnectOutcome, ReconnectPolicy};
98
99    fn quick(attempts: usize) -> ReconnectPolicy {
100        ReconnectPolicy::with_delays(vec![Duration::from_millis(1); attempts])
101    }
102
103    #[tokio::test]
104    async fn the_first_successful_attempt_ends_the_loop() {
105        let calls = AtomicUsize::new(0);
106        let outcome = quick(3)
107            .run(&CancellationToken::new(), |attempt| {
108                calls.fetch_add(1, Ordering::Relaxed);
109                async move {
110                    if attempt == 2 {
111                        Ok(attempt)
112                    } else {
113                        Err(format!("attempt {attempt} failed"))
114                    }
115                }
116            })
117            .await;
118        assert_eq!(outcome, ReconnectOutcome::Reconnected(2));
119        assert_eq!(calls.load(Ordering::Relaxed), 2);
120    }
121
122    #[tokio::test]
123    async fn exhaustion_reports_every_attempt_and_the_last_reason() {
124        let outcome = quick(2)
125            .run(&CancellationToken::new(), |attempt| async move {
126                Err::<(), _>(format!("attempt {attempt} failed"))
127            })
128            .await;
129        assert_eq!(
130            outcome,
131            ReconnectOutcome::Exhausted {
132                attempts: 2,
133                last_error: "attempt 2 failed".into()
134            }
135        );
136    }
137
138    #[tokio::test]
139    async fn cancellation_stops_waiting_between_attempts() {
140        let cancel = CancellationToken::new();
141        cancel.cancel();
142        let outcome = ReconnectPolicy::with_delays(vec![Duration::from_secs(3600)])
143            .run(&cancel, |_| async { Ok::<_, String>(()) })
144            .await;
145        assert_eq!(outcome, ReconnectOutcome::Cancelled);
146    }
