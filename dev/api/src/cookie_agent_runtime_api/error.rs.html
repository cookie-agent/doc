3/// Why preparing or executing a tool call failed.
4#[derive(Debug, Error)]
5pub enum ToolError {
6    #[error("tool progress sink closed")]
7    ProgressSinkClosed,
8    #[error("tool failed: {0}")]
9    Failed(String),
10    #[error("prepared operation changed: {0}")]
11    OperationChanged(String),
12    #[error("unsupported prepared-operation security: {0}")]
13    UnsupportedSecurity(String),
14    #[error("prepared operation is unsupported on this platform: {0}")]
15    UnsupportedPlatform(String),
16    #[error("prepared capability resource limit exceeded: {0}")]
17    ResourceLimit(String),
18    #[error("invalid URL: {0}")]
19    InvalidUrl(String),
20    #[error("permission denied: {0}")]
21    PermissionDenied(String),
22    #[error("redirect error: {0}")]
23    RedirectError(String),
24    #[error("request timed out: {0}")]
25    Timeout(String),
26    #[error("transport error: {0}")]
27    TransportError(String),
28    #[error("unsupported content type: {0}")]
29    UnsupportedContentType(String),
30    /// The runtime or MCP server that owns the tool is reconnecting or
31    /// gone; the call was not attempted or did not complete.
32    #[error("runtime unavailable: {0}")]
33    RuntimeUnavailable(String),
34}
35
36impl ToolError {
37    #[must_use]
38    pub fn operation_changed(message: impl Into<String>) -> Self {
39        Self::OperationChanged(message.into())
40    }
41
42    #[must_use]
43    pub fn unsupported_security(message: impl Into<String>) -> Self {
44        Self::UnsupportedSecurity(message.into())
45    }
46
47    #[must_use]
48    pub fn unsupported_platform(message: impl Into<String>) -> Self {
49        Self::UnsupportedPlatform(message.into())
50    }
51
52    #[must_use]
53    pub fn resource_limit(message: impl Into<String>) -> Self {
54        Self::ResourceLimit(message.into())
55    }
56
57    #[must_use]
58    pub fn runtime_unavailable(message: impl Into<String>) -> Self {
59        Self::RuntimeUnavailable(message.into())
60    }
61
62    #[must_use]
63    pub fn execution(message: impl Into<String>) -> Self {
64        Self::Failed(cookie_agent_protocol::diagnostics::detail(&message.into()).to_string())
65    }
66
67    #[must_use]
68    pub fn message(&self) -> String {
69        match self {
70            Self::ProgressSinkClosed => "tool progress sink closed".into(),
71            Self::Failed(message)
72            | Self::OperationChanged(message)
73            | Self::UnsupportedSecurity(message)
74            | Self::UnsupportedPlatform(message)
75            | Self::InvalidUrl(message)
76            | Self::PermissionDenied(message)
77            | Self::RedirectError(message)
78            | Self::Timeout(message)
79            | Self::TransportError(message)
80            | Self::UnsupportedContentType(message)
81            | Self::RuntimeUnavailable(message)
82            | Self::ResourceLimit(message) => message.clone(),
83        }
84    }
85}
