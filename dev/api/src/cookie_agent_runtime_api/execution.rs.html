22/// Immutable harness context captured for one tool preparation/execution batch:
23/// what a tool knows about the turn, the same on either side of the runtime
24/// protocol (its `turn`).
25///
26/// This is tool-facing but harness-private metadata. Tool providers must not forward it to
27/// external systems, including MCP wrappers. Compaction rehydration uses the context of the owner
28/// policy and model binding that triggered the checkpoint.
29#[derive(Debug)]
30pub struct TurnAgentContext {
31    /// Agent that owns the tool call.
32    pub agent: AgentId,
33    /// Exact model selected for this turn.
34    pub model: ModelKey,
35    /// Frozen wire adapter family used to deliver tool results.
36    pub adapter: AdaptorId,
37    /// The media the model accepts from a tool, with effective limits and
38    /// delivery; see [`crate::media::tool_input_media`].
39    pub input_media: Vec<ToolInputMedia>,
40}

91impl ToolExecutionContext {
92    /// Tells the host the call may now be detached; see [`DetachPoint`].
93    pub fn detachable(&self, note: Option<String>) {
94        self.detach.reached(note);

97    /// Reads a page of an artifact in this call's session tree.
98    pub async fn read_artifact(
99        &self,
100        target: ArtifactReadTarget,
101        window: LineWindow,
102    ) -> Result<ArtifactReadPage, ToolError> {
103        Arc::clone(&self.artifacts)
104            .read_page(self.session, target, window)
105            .await
106    }
107
108    pub fn retain_attachment(
109        &self,
110        mime_type: impl Into<String>,
111        filename: Option<String>,
112        bytes: &[u8],
113    ) -> Result<ToolAttachment, ToolError> {
114        let mime_type = mime_type.into();
115        let path = filename.as_deref().map_or_else(PathBuf::new, PathBuf::from);
116        validate_attachment(&mime_type, &path, bytes)?;
117        self.retain_validated_attachment(mime_type, filename, bytes)
118    }
119
120    pub fn retain_validated_attachment(
121        &self,
122        mime_type: impl Into<String>,
123        filename: Option<String>,
124        bytes: &[u8],
125    ) -> Result<ToolAttachment, ToolError> {
126        let mime_type = mime_type.into();
127        let (reference, sha256) = self.artifacts.retain(self.session, bytes)?;
128        Ok(ToolAttachment {
129            mime_type: cookie_agent_protocol::MimeType::new(mime_type)
130                .map_err(|error| ToolError::execution(error.to_string()))?,
131            filename,
132            byte_length: bytes.len() as u64,
133            sha256: Sha256Digest::new(sha256)
134                .map_err(|error| ToolError::execution(error.to_string()))?,
135            reference,
136        })
137    }

140/// The session artifact store as tools see it. The host owns where and how
141/// the bytes are kept; a session only reaches its own tree's artifacts.
142#[async_trait]
143pub trait ToolArtifacts: std::fmt::Debug + Send + Sync {
144    /// Stores `content` in `session`'s tree, returning its reference and
145    /// lowercase hex SHA-256.
146    fn retain(
147        &self,
148        session: SessionId,
149        content: &[u8],
150    ) -> Result<(ArtifactReference, String), ToolError>;
151
152    /// Reads the lines of `window` of an artifact, or of one stream of a
153    /// named-stream manifest, in `session`'s tree, numbered as `read`
154    /// numbers a file's ([`crate::text_tools`]).
155    async fn read_page(
156        self: Arc<Self>,
157        session: SessionId,
158        target: ArtifactReadTarget,
159        window: LineWindow,
160    ) -> Result<ArtifactReadPage, ToolError>;
161}

182/// Where a call's validated progress goes when the host captures its output.
183#[async_trait]
184pub trait ProgressCapture: std::fmt::Debug + Send + Sync {
185    /// Captures `progress`, which [`ProgressSink`] has already bounded and

196/// A call's progress channel. It enforces the display and output bounds and
197/// hands output chunks to the host's [`ProgressCapture`].
198#[derive(Clone)]
199pub struct ProgressSink {
200    sender: mpsc::Sender<ToolProgress>,
201    capture: Option<Arc<dyn ProgressCapture>>,
202    budget: Arc<std::sync::Mutex<ProgressBudget>>,

204}

220/// What a call's progress has used of its display bounds: the bytes of its

228impl ProgressSink {

230    #[must_use]

232        Self {
233            sender,
234            capture: None,

236            unknown_blocks: None,

238    }

240    /// A sink whose output chunks go to `capture`.
241    #[must_use]
242    pub fn with_capture(

244        capture: Arc<dyn ProgressCapture>,

246        Self {
247            sender,

250            unknown_blocks: None,

254    /// The sink, telling `reporter` of every display block of a kind this
255    /// build does not know that it drops from a replacement, as one from a
256    /// newer plugin or runtime would carry.
257    #[must_use]
258    pub fn with_unknown_block_reporter(mut self, reporter: UnknownBlockReporter) -> Self {
259        self.unknown_blocks = Some(reporter);
260        self
261    }
262
263    /// Reports display blocks of kinds this build does not know, of tags
264    /// `tags`, that the host dropped from a progress update before it reached
265    /// the sink, as a runtime's `tool/progress` carries them.
266    pub fn report_unknown_blocks(&self, tags: Vec<String>) {
267        if let Some(reporter) = self.unknown_blocks.as_ref().filter(|_| !tags.is_empty()) {
268            reporter(tags);
269        }
270    }
271
272    pub async fn send(&self, mut progress: ToolProgress) -> Result<(), ToolError> {
273        if !progress.output.is_empty() && self.capture.is_none() {
274            return Err(ToolError::execution("output capture is unavailable"));
275        }
276        if progress.message.len() > cookie_agent_protocol::SafeDisplayText::MAX_BYTES
277            || progress
278                .display
279                .as_ref()
280                .is_some_and(|display| match display {
281                    cookie_agent_protocol::DisplayDelta::Append { text, .. } => {
282                        text.len() > cookie_agent_protocol::DisplayDelta::MAX_APPEND_BYTES
283                    }
284                    cookie_agent_protocol::DisplayDelta::Replace { .. } => false,
285                })
286        {
287            return Err(ToolError::resource_limit(
288                "tool display delta exceeds the display event bound",
289            ));
290        }
291        if progress.output.len() > cookie_agent_protocol::MAX_TOOL_STREAMS
292            || progress.output.iter().fold(0_usize, |total, chunk| {
293                total.saturating_add(chunk.text.len())
294            }) > cookie_agent_protocol::MAX_TOOL_DELTA_BYTES
295        {
296            return Err(ToolError::resource_limit(
297                "tool output delta exceeds 64 KiB or 8 chunks",
298            ));
299        }
300        for chunk in &progress.output {
301            if let Some(name) = &chunk.stream {
302                cookie_agent_protocol::validate_tool_stream_name(name)
303                    .map_err(ToolError::execution)?;
304            }
305        }
306        {
307            let mut budget = self.budget.lock().unwrap_or_else(|p| p.into_inner());
308            let remaining =
309                cookie_agent_protocol::MAX_TOOL_DISPLAY_BYTES.saturating_sub(budget.message_bytes);
310            progress.message = crate::sanitize_tool_display(&progress.message, remaining);
311            budget.message_bytes += progress.message.len();
312            if let Some(cookie_agent_protocol::DisplayDelta::Replace { blocks }) =
313                &mut progress.display
314            {
315                let unknown = cookie_agent_protocol::take_unknown_blocks(blocks);
316                if let Some(reporter) = self.unknown_blocks.as_ref().filter(|_| !unknown.is_empty())
317                {
318                    reporter(unknown);
319                }
320            }
321            progress.display = match progress.display.take().map(sanitize_delta) {
322                Some(delta) => budget
323                    .display
324                    .admit(delta)
325                    .map_err(|error| ToolError::execution(error.to_string()))?,
326                None => None,
327            };
328        }
329        let permit = if progress.message.is_empty() && progress.display.is_none() {
330            None
331        } else {
332            Some(
333                self.sender
334                    .clone()
335                    .reserve_owned()
336                    .await
337                    .map_err(|_| ToolError::ProgressSinkClosed)?,
338            )
339        };
340        if let Some(capture) = &self.capture {
341            return capture.emit(progress, permit).await;
342        }
343        if let Some(permit) = permit {
344            permit.send(progress);
345        }
346        Ok(())
347    }
348}

350/// `delta` with the control characters its text may not carry removed: an
351/// append's text as [`crate::sanitize_tool_display`] cleans it (keeping SGR
352/// colour when it names a terminal stream), a replacement's blocks as the
353/// daemon sanitizes blocks it receives.
354fn sanitize_delta(
355    delta: cookie_agent_protocol::DisplayDelta,
356) -> cookie_agent_protocol::DisplayDelta {
357    use cookie_agent_protocol::DisplayDelta;
358    match delta {
359        DisplayDelta::Append {
360            index,
361            text,
362            stream,
363        } => DisplayDelta::Append {
364            index,
365            text: match stream {
366                Some(_) => cookie_agent_protocol::sanitize_terminal_text(&text).into_owned(),
367                None => crate::sanitize_tool_display(&text, usize::MAX),
368            },
369            stream,
370        },
371        DisplayDelta::Replace { blocks } => DisplayDelta::Replace {
372            blocks: blocks
373                .into_iter()
374                .filter_map(cookie_agent_protocol::Open::into_known)
375                .map(|mut block| {
376                    block.sanitize();
377                    cookie_agent_protocol::Open::Known(block)
378                })
379                .collect(),
380        },
381    }

384/// The point at which an executing call of a background-capable tool may be
385/// detached, as `docs/site/specs/background-tools.md` section 4 specifies.
386///
387/// A background-capable tool calls [`DetachPoint::reached`] once it has
388/// started its work (bash right after spawning its command), with an
389/// optional note for the immediate result the host commits when it detaches
390/// the call. The host detaches only at or after that point: a call the model
391/// started in the background detaches as soon as it reaches it, and a user's
392/// request waits for it. A call that finishes first simply returns.
393#[derive(Clone, Debug)]
394pub struct DetachPoint {
395    sender: Arc<tokio::sync::watch::Sender<Option<Detachable>>>,
396}
397
398/// What a call reports when it reaches its detach point.
399#[derive(Clone, Debug, Default, Eq, PartialEq)]
400pub struct Detachable {
401    /// A line for the immediate result, such as a process ID.
402    pub note: Option<String>,
403    /// The work reports its own results once detached (a subagent reports
404    /// each finished turn), so the host only stops waiting: it drops the
405    /// call's execution and sends no `<background_result>`.
406    pub reports_itself: bool,
407    /// The result the host commits, in place of its generic one, when the
408    /// user interrupts the run while the call runs in the foreground at or
409    /// past this point: at once for work that reports itself, otherwise
410    /// once the call winds down within the cleanup grace. Other
411    /// cancellations, and a call that misses the grace, get the host's
412    /// generic result, which names their cause.
413    pub on_user_interrupt: Option<String>,
414}
415
416/// The host's side of a [`DetachPoint`].
417#[derive(Debug)]
418pub struct DetachWatch {
419    receiver: tokio::sync::watch::Receiver<Option<Detachable>>,
420}
421
422impl DetachPoint {
423    /// A detach point and the watch the host reads it through.
424    #[must_use]
425    pub fn channel() -> (Self, DetachWatch) {
426        let (sender, receiver) = tokio::sync::watch::channel(None);
427        (
428            Self {
429                sender: Arc::new(sender),
430            },
431            DetachWatch { receiver },
432        )
433    }
434
435    /// Marks the call detachable with `note`. Only the first call counts.
436    pub fn reached(&self, note: Option<String>) {
437        self.reached_with(Detachable {
438            note,
439            ..Detachable::default()
440        });
441    }
442
443    /// What the call reported at its detach point, once it reached it.
444    #[must_use]
445    pub fn current(&self) -> Option<Detachable> {
446        self.sender.borrow().clone()
447    }
448
449    /// Marks the call detachable as `detachable`. Only the first call counts.
450    pub fn reached_with(&self, detachable: Detachable) {
451        self.sender.send_if_modified(|current| {
452            if current.is_some() {
453                return false;
454            }
455            *current = Some(detachable);
456            true
457        });
458    }
459}
460
461impl DetachWatch {
462    /// The detach point, once the call has reached it.
463    #[must_use]
464    pub fn current(&self) -> Option<Detachable> {
465        self.receiver.borrow().clone()
466    }
467
468    /// Waits until the call reaches its detach point. Pending forever if the
469    /// call never does and its point is dropped.
470    pub async fn reached(&mut self) -> Detachable {
471        loop {
472            if let Some(detachable) = self.receiver.borrow_and_update().clone() {
473                return detachable;
474            }
475            if self.receiver.changed().await.is_err() {
476                std::future::pending::<()>().await;
477            }
478        }
479    }
