1//! What the file tools and the memory tools share: the numbered, 1-based
2//! pages of text that `read`, `memory_read`, and `read_artifact` return,
3//! and the wording of their results and errors.
4//!
5//! A page is a [`LineWindow`] of lines: `offset` is the 1-based number of
6//! its first line and `limit` how many lines it holds at most. Each line is
7//! returned as `N: text`, numbered from 1, cut after [`MAX_LINE_BYTES`] with
8//! a `... [line truncated: N more bytes]` marker, and the page stops before
9//! it outgrows its byte budget. A page that leaves lines out ends with a
10//! continuation marker naming the `offset` to read next
11//! ([`NumberedPage::continuation`]).
12
13use std::{borrow::Cow, io::BufRead};
14
15use cookie_agent_protocol::{DISPLAY_BUDGET_BYTES, DisplayBlock, PersistedToolResult};

19/// Lines one page holds when a call gives no `limit`.
20pub const DEFAULT_LIMIT: usize = 2_000;
21/// Longest line returned in full; the rest of a longer line is replaced by a
22/// marker, so one huge minified line cannot fill or fail the result.
23pub const MAX_LINE_BYTES: usize = 4 * 1024;
24/// Output budget for one page of text, under the 2 MiB result limit.
25pub const MAX_PAGE_BYTES: usize = PersistedToolResult::MAX_OUTPUT_BYTES - 64 * 1024;
26/// The most characters of a call's text an error repeats.
27pub const MAX_ECHO_CHARS: usize = 200;
28
29/// The lines a call asks for: from line `offset`, numbered from 1, at most
30/// `limit` of them. Directory pages count entries the same way.
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32pub struct LineWindow {
33    /// The first line, 1-based.
34    pub offset: usize,
35    /// The most lines the page holds; positive.
36    pub limit: usize,
37}
38
39impl LineWindow {
40    /// The window a call's `offset` and `limit` give, from line 1 and
41    /// [`DEFAULT_LIMIT`] lines when they are left out. An `offset` of 0 and a
42    /// `limit` of 0 are refused.
43    pub fn new(offset: Option<usize>, limit: Option<usize>) -> Result<Self, ToolError> {
44        let offset = offset.unwrap_or(1);
45        if offset == 0 {
46            return Err(ToolError::execution(
47                "offset is 1-based: the first line is offset=1",
48            ));
49        }
50        let limit = limit.unwrap_or(DEFAULT_LIMIT);
51        if limit == 0 {
52            return Err(ToolError::execution("limit must be positive"));
53        }
54        Ok(Self { offset, limit })
55    }
56
57    /// The zero-based index of the first line.
58    #[must_use]
59    pub const fn first_index(&self) -> usize {
60        self.offset - 1
61    }
62
63    /// The zero-based index just past the last line the window can hold.
64    #[must_use]
65    pub const fn end_index(&self) -> usize {
66        self.first_index().saturating_add(self.limit)
67    }
68
69    /// Fails when the window starts past the last of `total` items of
70    /// `target`, counted as `noun` (`line` or `entry`). Line 1 of nothing is
71    /// an empty page, not an error.
72    pub fn check_in_range(&self, total: usize, noun: &str, target: &str) -> Result<(), ToolError> {
73        if self.offset > 1 && self.offset > total {
74            return Err(ToolError::execution(format!(
75                "offset={} is past the end of {target}, which has {}",
76                self.offset,
77                counted(total, noun)
78            )));
79        }
80        Ok(())
81    }
82
83    /// The continuation marker of a page of `total` items, counted as
84    /// `noun`, that this window cuts short; `None` when the page reaches the
85    /// last item.
86    #[must_use]
87    pub fn remaining_marker(&self, total: usize, noun: &str) -> Option<String> {
88        let end = self.end_index();
89        (total > end).then(|| {
90            format!(
91                "[{} more {}; continue with offset={}]",
92                total - end,
93                noun_for(total - end, noun),
94                end + 1
95            )
96        })
97    }
98}
99
100/// `count` followed by `noun` (`line`, `entry`, `byte`, ...), plural unless
101/// `count` is 1.
102#[must_use]
103pub fn counted(count: usize, noun: &str) -> String {
104    format!("{count} {}", noun_for(count, noun))
105}
106
107/// `noun` as `count` of it reads: singular for 1, else plural.
108fn noun_for(count: usize, noun: &str) -> Cow<'_, str> {
109    if count == 1 {
110        return Cow::Borrowed(noun);
111    }
112    Cow::Owned(match noun.strip_suffix('y') {
113        Some(stem) => format!("{stem}ies"),
114        None => format!("{noun}s"),
115    })
116}

118/// Why a page could not be read.
119#[derive(Debug)]
120pub enum PageError {
121    Io(std::io::Error),
122    /// A line on the page is not UTF-8.
123    NotText,
124}

126/// One page of numbered lines.
127#[derive(Clone, Debug, Eq, PartialEq)]
128pub struct NumberedPage {
129    /// The page's lines, each `N: text\n`.
130    pub text: String,
131    /// The page's first line, 1-based.
132    pub start_line: usize,
133    /// The page's lines as clients see them, without their numbers, up to
134    /// the display budget.
135    pub excerpt: Vec<String>,
136    /// Lines of the page past the display budget.
137    pub excerpt_omitted: u64,
138    /// The lines of the whole text.
139    pub total_lines: usize,
140    /// The first line left out, 1-based, when lines follow the page.
141    pub next_offset: Option<usize>,
142    /// Whether the page stopped at its byte budget before its `limit`.
143    pub cut: bool,
144}
145
146impl NumberedPage {
147    /// The marker that ends a page leaving lines out: where to continue,
148    /// and why the page stopped. `None` when the page reaches the last line.
149    #[must_use]
150    pub fn continuation(&self) -> Option<String> {
151        let next = self.next_offset?;
152        Some(if self.cut {
153            format!("[page cut at the result size limit; continue with offset={next}]")
154        } else {
155            let remaining = self.total_lines.saturating_sub(next - 1);
156            format!(
157                "[{remaining} more {}; continue with offset={next}]",
158                noun_for(remaining, "line")
159            )
160        })
161    }
162
163    /// Fails when the page starts past the last line of `target`.
164    pub fn check_in_range(&self, target: &str) -> Result<(), ToolError> {
165        LineWindow {
166            offset: self.start_line,
167            limit: 1,
168        }
169        .check_in_range(self.total_lines, "line", target)
170    }
171
172    /// The numbered lines followed by the continuation marker, if any, on a
173    /// line of its own.
174    #[must_use]
175    pub fn text_with_continuation(&self) -> String {
176        match self.continuation() {
177            Some(marker) => format!("{}{marker}", self.text),
178            None => self.text.clone(),
179        }
180    }
181
182    /// The page as clients show it: an excerpt of `path` in `language`.
183    #[must_use]
184    pub fn excerpt_block(&self, path: String, language: Option<String>) -> DisplayBlock {
185        DisplayBlock::FileExcerpt {
186            path,
187            language,
188            start_line: self.start_line as u64,
189            lines: self.excerpt.clone(),
190            total_lines: Some(self.total_lines as u64),
191            omitted_lines: (self.excerpt_omitted > 0).then_some(self.excerpt_omitted),
192            full: None,
193        }
194    }
195}

197/// Builds a [`NumberedPage`] from text fed in chunks of any size, in one
198/// streaming pass that also counts the text's lines.
199#[derive(Debug)]
200pub struct LinePager {
201    window: LineWindow,
202    /// The most bytes the numbered lines may take.
203    budget: usize,
204    /// Whether a line that is not UTF-8 is shown with replacement
205    /// characters instead of failing the page.
206    lossy: bool,
207    text: String,
208    /// The current line's bytes, up to [`MAX_LINE_BYTES`].
209    line: Vec<u8>,
210    /// Bytes of the current line beyond [`MAX_LINE_BYTES`].
211    cut: usize,
212    /// The first line, zero-based, left out because the page reached its
213    /// budget.
214    stopped_at: Option<usize>,
215    excerpt: Vec<String>,
216    excerpt_bytes: usize,
217    excerpt_omitted: u64,
218    /// Lines ended so far, and whether bytes follow the last newline.
219    lines: usize,
220    partial: bool,
221}
222
223impl LinePager {
224    /// A pager of `window` whose numbered lines take at most `budget`
225    /// bytes; a line that is not UTF-8 fails the page.
226    #[must_use]
227    pub fn new(window: LineWindow, budget: usize) -> Self {
228        Self {
229            window,
230            budget,
231            lossy: false,
232            text: String::new(),
233            line: Vec::new(),
234            cut: 0,
235            stopped_at: None,
236            excerpt: Vec::new(),
237            excerpt_bytes: 0,
238            excerpt_omitted: 0,
239            lines: 0,
240            partial: false,
241        }
242    }
243
244    /// The same pager, showing a line that is not UTF-8 with replacement
245    /// characters instead of failing.
246    #[must_use]
247    pub fn lossy(mut self) -> Self {
248        self.lossy = true;
249        self
250    }
251
252    fn wanted(&self) -> bool {
253        self.lines >= self.window.first_index()
254            && self.lines < self.window.end_index()
255            && self.stopped_at.is_none()
256    }
257
258    /// Feeds the next bytes of the text.
259    pub fn push(&mut self, chunk: &[u8]) -> Result<(), PageError> {
260        let mut rest = chunk;
261        while !rest.is_empty() {
262            let wanted = self.wanted();
263            if !wanted && self.lines >= self.window.first_index() {
264                // Past the page: only count what remains.
265                self.lines += memchr::memchr_iter(b'\n', rest).count();
266                self.partial = rest.last() != Some(&b'\n');
267                return Ok(());
268            }
269            let newline = memchr::memchr(b'\n', rest);
270            if wanted {
271                self.keep(&rest[..newline.unwrap_or(rest.len())]);
272            }
273            match newline {
274                Some(at) => {
275                    if wanted {
276                        self.finish_line(true)?;
277                    }
278                    self.lines += 1;
279                    self.partial = false;
280                    rest = &rest[at + 1..];
281                }
282                None => {
283                    self.partial = true;
284                    rest = &[];
285                }
286            }
287        }
288        Ok(())
289    }
290
291    /// The page, once the whole text was fed.
292    pub fn finish(mut self) -> Result<NumberedPage, PageError> {
293        if self.partial {
294            if self.wanted() {
295                self.finish_line(false)?;
296            }
297            self.lines += 1;
298        }
299        let next_offset = match self.stopped_at {
300            Some(index) => Some(index + 1),
301            None => (self.lines > self.window.end_index()).then(|| self.window.end_index() + 1),
302        };
303        Ok(NumberedPage {
304            text: self.text,
305            start_line: self.window.offset,
306            excerpt: self.excerpt,
307            excerpt_omitted: self.excerpt_omitted,
308            total_lines: self.lines,
309            next_offset,
310            cut: self.stopped_at.is_some(),
311        })
312    }
313
314    fn keep(&mut self, segment: &[u8]) {
315        let take = MAX_LINE_BYTES
316            .saturating_sub(self.line.len())
317            .min(segment.len());
318        self.line.extend_from_slice(&segment[..take]);
319        self.cut += segment.len() - take;
320    }
321
322    fn finish_line(&mut self, terminated: bool) -> Result<(), PageError> {
323        if terminated && self.cut == 0 && self.line.last() == Some(&b'\r') {
324            self.line.pop();
325        }
326        let line = std::mem::take(&mut self.line);
327        let text: Cow<'_, str> = match std::str::from_utf8(&line) {
328            Ok(text) => Cow::Borrowed(text),
329            // A cut can split the last character; drop its leading bytes.
330            Err(error) if self.cut > 0 && error.error_len().is_none() => {
331                self.cut += line.len() - error.valid_up_to();
332                Cow::Borrowed(
333                    std::str::from_utf8(&line[..error.valid_up_to()]).expect("valid prefix"),
334                )
335            }
336            Err(_) if self.lossy => String::from_utf8_lossy(&line),
337            Err(_) => return Err(PageError::NotText),
338        };
339        let number = self.lines + 1;
340        let rendered = if self.cut == 0 {
341            format!("{number}: {text}\n")
342        } else {
343            format!(
344                "{number}: {text}... [line truncated: {} more bytes]\n",
345                self.cut
346            )
347        };
348        if self.text.len() + rendered.len() > self.budget {
349            self.stopped_at = Some(self.lines);
350        } else {
351            self.text.push_str(&rendered);
352            // The excerpt line is the numbered line without its number.
353            let shown = &rendered[number.to_string().len() + 2..rendered.len() - 1];
354            if self.excerpt_bytes + shown.len() <= DISPLAY_BUDGET_BYTES {
355                self.excerpt_bytes += shown.len();
356                self.excerpt.push(shown.to_owned());
357            } else {
358                self.excerpt_omitted += 1;
359            }
360        }
361        self.line = line;
362        self.line.clear();
363        self.cut = 0;
364        Ok(())
365    }
366}

368/// The page `window` of everything `reader` holds, its numbered lines in
369/// at most `budget` bytes.
370pub fn page_reader(
371    reader: &mut impl BufRead,
372    window: LineWindow,
373    budget: usize,
374) -> Result<NumberedPage, PageError> {
375    let mut pager = LinePager::new(window, budget);
376    loop {
377        let chunk = reader.fill_buf().map_err(PageError::Io)?;
378        if chunk.is_empty() {
379            break;
380        }
381        let consumed = chunk.len();
382        pager.push(chunk)?;
383        reader.consume(consumed);
384    }
385    pager.finish()
386}

388/// The page `window` of `text`, its numbered lines in at most `budget`
389/// bytes.
390#[must_use]
391pub fn page_text(text: &str, window: LineWindow, budget: usize) -> NumberedPage {
392    let mut pager = LinePager::new(window, budget);
393    pager
394        .push(text.as_bytes())
395        .and_then(|()| pager.finish())
396        .expect("a str is UTF-8")
397}

399/// `text` as an error repeats it: its first [`MAX_ECHO_CHARS`] characters,
400/// and an ellipsis when there are more.
401#[must_use]
402pub fn echo(text: &str) -> Cow<'_, str> {
403    match text.char_indices().nth(MAX_ECHO_CHARS) {
404        Some((at, _)) => Cow::Owned(format!("{}…", &text[..at])),
405        None => Cow::Borrowed(text),
406    }
407}

409/// The error of an edit whose `old` text occurs `count` times, other than
410/// once, in `target`; `replace_all` adds that the edit may replace every
411/// occurrence instead, for a tool that can.
412#[must_use]
413pub fn occurrence_error(old: &str, count: usize, target: &str, replace_all: bool) -> String {
414    format!(
415        "`{}` occurs {count} {} in {target}; include more context so it occurs once{}",
416        echo(old),
417        if count == 1 { "time" } else { "times" },
418        if replace_all {
419            " or set replaceAll"
420        } else {
421            ""
422        }
423    )
424}
