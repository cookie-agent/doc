1//! When each memory was last read (`docs/site/specs/memory.md` section
2//! 16.9.6), kept outside the memory files in `memory/.reads.json`, a map
3//! from memory URL to time. A model's `memory_read` and a snapshot that
4//! shows a memory whole count as reads; a client's does not.
5//!
6//! The times are advisory: the file is read once, kept in memory, and
7//! rewritten through a temporary file after each change, without the
8//! memory lock. Two daemons sharing a data root can each drop the other's
9//! latest times, and a URL whose memory moves or goes keeps its entry.
10
11use std::{collections::BTreeMap, fs};
12
13use jiff::Timestamp;
14
15use super::{MemoryRoot, READS_FILE, file::to_second, store::write_file, url::MemoryUrl};
16
17/// The last-read times, loaded from the file on first use.
18#[derive(Debug, Default)]
19pub struct ReadTimes(std::sync::Mutex<Option<BTreeMap<String, Timestamp>>>);
20
21impl MemoryRoot {
22    fn with_reads<T>(&self, work: impl FnOnce(&mut BTreeMap<String, Timestamp>) -> T) -> T {
23        let mut guard = self
24            .reads
25            .0
26            .lock()
27            .unwrap_or_else(std::sync::PoisonError::into_inner);
28        let times = guard.get_or_insert_with(|| {
29            fs::read(self.root.join(READS_FILE))
30                .ok()
31                .and_then(|bytes| serde_json::from_slice(&bytes).ok())
32                .unwrap_or_default()
33        });
34        work(times)
35    }
36
37    /// Records that `urls` were read at `now`. Nothing is written while the
38    /// memory folder does not exist; a failure to write is reported on
39    /// standard error, since the times are advisory.
40    pub fn record_reads<'a>(&self, urls: impl IntoIterator<Item = &'a MemoryUrl>, now: Timestamp) {
41        let now = to_second(now);
42        let text = self.with_reads(|times| {
43            let mut changed = false;
44            for url in urls {
45                changed |= times.insert(url.to_string(), now) != Some(now);
46            }
47            changed.then(|| serde_json::to_vec(times).expect("read times serialize"))
48        });
49        if let Some(text) = text
50            && self.root.is_dir()
51            && let Err(error) = write_file(&self.root.join(READS_FILE), &text)
52        {
53            eprintln!("cookie-agent: memory read times: {error}");
54        }
55    }
56
57    /// When the memory at `url` was last read, if the daemon knows.
58    pub fn last_read(&self, url: &MemoryUrl) -> Option<Timestamp> {
59        self.with_reads(|times| times.get(&url.to_string()).copied())
60    }
61}

63#[cfg(test)]
64mod tests {
65    use super::*;
66
67    #[test]
68    fn read_times_persist_beside_the_memories() {
69        let data = tempfile::tempdir().unwrap();
70        let memory = MemoryRoot::new(data.path());
71        assert!(memory.create_defaults_if_absent().unwrap());
72        let url = MemoryUrl::parse("user://feedback").unwrap();
73        assert_eq!(memory.last_read(&url), None);
74        let at = Timestamp::from_second(1_000_000_000).unwrap();
75        memory.record_reads([&url], at);
76        assert_eq!(memory.last_read(&url), Some(at));
77        let reopened = MemoryRoot::new(data.path());
78        assert_eq!(reopened.last_read(&url), Some(at));
79        // The file is not a memory, nor something a listing reports.
80        let listing = reopened.list(&super::super::store::ListScope::All).unwrap();
81        assert!(listing.ignored.is_empty(), "{:?}", listing.ignored);
82    }
