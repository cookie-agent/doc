131/// Failure returned by a producer recovery callback.
132#[derive(Debug, thiserror::Error)]

134pub struct RecoveryFailure {
135    message: SafeErrorMessage,
136}

138impl RecoveryFailure {
139    /// Creates a recovery failure from a bounded, control-free message.
140    pub fn new(message: impl Into<String>) -> Result<Self, PluginError> {
141        let message = SafeErrorMessage::new(message.into())
142            .map_err(|error| PluginError::Protocol(format!("invalid recovery failure: {error}")))?;
143        Ok(Self { message })
144    }
145}
146
147impl From<PluginError> for RecoveryFailure {
148    fn from(error: PluginError) -> Self {
149        let mut message = error.to_string().replace(char::is_control, " ");
150        let mut end = message.len().min(SafeErrorMessage::MAX_BYTES);
151        while !message.is_char_boundary(end) {
152            end -= 1;
153        }
154        message.truncate(end);
155        Self {
156            message: SafeErrorMessage::new(message)
157                .expect("sanitized plugin errors are valid recovery messages"),
158        }
159    }
160}

192    async fn run_io<R, W>(self, reader: R, writer: W) -> Result<(), PluginError>
193    where
194        R: AsyncRead + Unpin + Send + 'static,
195        W: AsyncWrite + Unpin + Send + 'static,
196    {
197        let pending_emits = Arc::new(Mutex::new(HashMap::new()));
198        let pending_requests = Arc::new(Mutex::new(HashMap::new()));
199        let (outbound, outbound_rx) = mpsc::channel(128);
200        let state = Arc::new(ContextState {
201            grants: Mutex::new(HashMap::new()),
202            pending_emits: Arc::clone(&pending_emits),
203            pending_requests: Arc::clone(&pending_requests),
204            outbound,
205            publishing: PublishingCapabilities {
206                bus: self.handlers.publish_bus,
207            },
208            producers: self.handlers.producers,
209            request_ids: AtomicI64::new(1),
210            pending_slots: Arc::new(Semaphore::new(MAX_PENDING_REQUESTS)),
211            connected: AtomicBool::new(true),
212        });
213        let context = PluginContext { state, grant: None };
214        let mut writer_task = tokio::spawn(writer_loop(
215            writer,
216            outbound_rx,
217            Arc::clone(&pending_emits),
218            Arc::clone(&pending_requests),
219        ));
220        let (inbound_tx, mut inbound_rx) = mpsc::channel(32);
221        let reader_task = tokio::spawn(reader_loop(reader, inbound_tx));
222        let handler_slots = Arc::new(Semaphore::new(MAX_CONCURRENT_HANDLERS));
223        let recovery_running = Arc::new(AtomicBool::new(false));
224
225        let mut dispatch_tasks: Vec<JoinHandle<()>> = Vec::new();
226        let (result, writer_finished) = loop {
227            tokio::select! {
228                inbound = inbound_rx.recv() => match inbound {
229                    Some(Ok(frame)) => {
230                        dispatch_tasks.retain(|task| !task.is_finished());
231                        match self.dispatch_message(
232                            &frame.message,
233                            frame.received_at,
234                            &context,
235                            &mut dispatch_tasks,
236                            &handler_slots,
237                            &recovery_running,
238                        ).await {
239                            Ok(Dispatch::Continue) => {}
240                            Ok(Dispatch::Shutdown) => break (Ok(()), false),
241                            Err(error) => break (Err(error), false),
242                        }
243                    }
244                    Some(Err(error)) => break (Err(error), false),
245                    None => break (Ok(()), false),
246                },
247                writer_result = writer_task_finished(&mut writer_task) => {
248                    let result = match writer_result {
249                        Ok(()) => Err(PluginError::TransportClosed),
250                        Err(error) => Err(error),
251                    };
252                    break (result, true);
253                }
254            }
255        };
256
257        context.state.connected.store(false, Ordering::Release);
258        reader_task.abort();
259        let _ = reader_task.await;
260        if !writer_finished {
261            writer_task.abort();
262            let _ = writer_task.await;
263        }
264        fail_pending_emits(&pending_emits);
265        fail_pending_requests(&pending_requests);
266        for task in dispatch_tasks {
267            task.abort();
268        }
269        result
270    }

272    async fn dispatch_message(
273        &self,
274        message: &str,
275        received_at: Instant,
276        context: &PluginContext,
277        tasks: &mut Vec<JoinHandle<()>>,
278        handler_slots: &Arc<Semaphore>,
279        recovery_running: &Arc<AtomicBool>,
280    ) -> Result<Dispatch, PluginError> {
281        let value: Value = serde_json::from_str(message)?;
282        if value.get("method").is_none() {
283            let response: Response = serde_json::from_value(value).map_err(|error| {
284                PluginError::Protocol(format!("malformed engine response: {error}"))
285            })?;
286            resolve_request(&context.state.pending_requests, response)?;
287            return Ok(Dispatch::Continue);
288        }
289        if value.get("id").is_some() {
290            let request: Request = serde_json::from_value(value).map_err(|error| {
291                PluginError::Protocol(format!("malformed engine request: {error}"))
292            })?;
293            self.dispatch_request(request, context, tasks, handler_slots)
294                .await?;
295            return Ok(Dispatch::Continue);
296        }
297
298        let notification: Notification = serde_json::from_value(value).map_err(|error| {
299            PluginError::Protocol(format!("malformed engine notification: {error}"))
300        })?;
301        self.dispatch_notification(
302            notification,
303            received_at,
304            context,
305            tasks,
306            handler_slots,
307            recovery_running,
308        )
309        .await
310    }

312    async fn dispatch_request(
313        &self,
314        request: Request,
315        context: &PluginContext,
316        tasks: &mut Vec<JoinHandle<()>>,
317        handler_slots: &Arc<Semaphore>,
318    ) -> Result<(), PluginError> {
319        match request.method.as_str() {
320            PLUGIN_INITIALIZE_METHOD => {
321                let _: ExtensionInitializeParams = parse_params(&request)?;
322                let result = ExtensionInitializeResult {
323                    protocol_version: ExtensionProtocolVersion::current(),
324                    name: self.name.clone(),
325                    version: self.version.clone(),
326                    capabilities: self.handlers.capabilities(),
327                    tools: self
328                        .handlers
329                        .tools
330                        .iter()
331                        .map(|tool| tool.declaration.clone())
332                        .collect(),
333                };
334                send_success(context, request.id, result).await?;
335            }
336            PLUGIN_PING_METHOD => {
337                let _: ExtensionPingParams = parse_params(&request)?;
338                send_success(context, request.id, ExtensionPingResult {}).await?;
339            }
340            PLUGIN_TOOLS_CALL_METHOD => {
341                let Some(permit) = try_handler_slot(context, &request, handler_slots).await? else {
342                    return Ok(());
343                };
344                let params: ExtensionToolCallParams = match parse_params(&request) {
345                    Ok(params) => params,
346                    Err(error) => {
347                        send_error(context, request.id, INVALID_PARAMS, error.to_string()).await?;
348                        return Ok(());
349                    }
350                };
351                let Some(tool) = self
352                    .handlers
353                    .tools
354                    .iter()
355                    .find(|tool| tool.declaration.name == params.tool)
356                    .cloned()
357                else {
358                    send_error(
359                        context,
360                        request.id,
361                        METHOD_NOT_FOUND,
362                        format!("unknown plugin tool `{}`", params.tool),
363                    )
364                    .await?;
365                    return Ok(());
366                };
367                let plugin_context =
368                    context.register_request(params.session_id, params.context_id.clone());
369                tasks.push(tokio::spawn(async move {
370                    let _permit = permit;
371                    let handler_context = plugin_context.clone();
372                    let handler_params = params.clone();
373                    let result =
374                        isolate(
375                            async move { (tool.handler)(handler_context, handler_params).await },
376                        )
377                        .await;
378                    plugin_context.revoke(&params.context_id);
379                    match result {
380                        Ok(Ok(output)) => {
381                            let _ = send_success(&plugin_context, request.id, output).await;
382                        }
383                        Ok(Err(error)) => {
384                            let _ =
385                                send_rpc_error(&plugin_context, request.id, error.into_rpc()).await;
386                        }
387                        Err(()) => {
388                            let _ = send_error(
389                                &plugin_context,
390                                request.id,
391                                INTERNAL_ERROR,
392                                "plugin tool handler panicked",
393                            )
394                            .await;
395                        }
396                    }
397                }));
398            }
399            method => match interception_hook(method) {
400                Some(hook) => {
401                    let handler = self.handlers.intercepts.get(&hook).cloned();
402                    dispatch_intercept(request, context, handler, tasks, handler_slots).await?;
403                }
404                None => {
405                    send_error(
406                        context,
407                        request.id,
408                        METHOD_NOT_FOUND,
409                        "plugin method is not supported",
410                    )
411                    .await?;
412                }
413            },
414        }
415        Ok(())
416    }

418    async fn dispatch_notification(
419        &self,
420        notification: Notification,
421        received_at: Instant,
422        context: &PluginContext,
423        tasks: &mut Vec<JoinHandle<()>>,
424        handler_slots: &Arc<Semaphore>,
425        recovery_running: &Arc<AtomicBool>,
426    ) -> Result<Dispatch, PluginError> {
427        match notification.method.as_str() {
428            PLUGIN_SHUTDOWN_METHOD => {
429                let _: ExtensionShutdownParams = parse_notification_params(&notification)?;
430                Ok(Dispatch::Shutdown)
431            }
432            PLUGIN_EVENT_METHOD => {
433                let params: ExtensionEventParams = parse_notification_params(&notification)?;
434                if let Some(handler) = self.handlers.event.clone() {
435                    let Ok(permit) = Arc::clone(handler_slots).try_acquire_owned() else {
436                        eprintln!("cookie-agent plugin event handler capacity exhausted");
437                        return Ok(Dispatch::Continue);
438                    };
439                    let plugin_context = context.register_notification(
440                        params.session_id,
441                        params.context_id.clone(),
442                        received_at,
443                    );
444                    tasks.push(tokio::spawn(async move {
445                        let _permit = permit;
446                        if isolate(async move { handler(plugin_context, params).await })
447                            .await
448                            .is_err()
449                        {
450                            eprintln!("cookie-agent plugin event handler panicked");
451                        }
452                    }));
453                }
454                Ok(Dispatch::Continue)
455            }
456            PLUGIN_BUS_EVENT_METHOD => {
457                let params: ExtensionBusEventParams = parse_notification_params(&notification)?;
458                if let Some(handler) = self.handlers.bus.clone() {
459                    let Ok(permit) = Arc::clone(handler_slots).try_acquire_owned() else {
460                        eprintln!("cookie-agent plugin bus handler capacity exhausted");
461                        return Ok(Dispatch::Continue);
462                    };
463                    let plugin_context = params.context_id.as_ref().map_or_else(
464                        || context.clone(),
465                        |context_id| {
466                            context.register_notification(
467                                params.session_id,
468                                context_id.clone(),
469                                received_at,
470                            )
471                        },
472                    );
473                    tasks.push(tokio::spawn(async move {
474                        let _permit = permit;
475                        if isolate(async move { handler(plugin_context, params).await })
476                            .await
477                            .is_err()
478                        {
479                            eprintln!("cookie-agent plugin bus handler panicked");
480                        }
481                    }));
482                }
483                Ok(Dispatch::Continue)
484            }
485            PLUGIN_EMIT_RESULT_METHOD => {
486                let result: ExtensionEmitResultParams = parse_notification_params(&notification)?;
487                resolve_emit(&context.state.pending_emits, result);
488                Ok(Dispatch::Continue)
489            }
490            PLUGIN_RECOVERY_START_METHOD => {
491                let _: ExtensionRecoveryStartParams = parse_notification_params(&notification)?;
492                let handler = self.handlers.recovery.clone();
493                if recovery_running.swap(true, Ordering::AcqRel) {
494                    return Err(PluginError::Protocol(
495                        "engine started recovery while recovery was already running".into(),
496                    ));
497                }
498                let plugin_context = context.clone();
499                let recovery_running = Arc::clone(recovery_running);
500                tasks.push(tokio::spawn(async move {
501                    let outcome = match handler {
502                        Some(handler) => {
503                            let handler_context = plugin_context.clone();
504                            match isolate(async move { handler(handler_context).await }).await {
505                                Ok(Ok(())) => ExtensionRecoveryOutcome::Ready,
506                                Ok(Err(error)) => ExtensionRecoveryOutcome::Failed {
507                                    message: error.message,
508                                },
509                                Err(()) => ExtensionRecoveryOutcome::Failed {
510                                    message: SafeErrorMessage::new(
511                                        "plugin recovery handler panicked",
512                                    )
513                                    .expect("static recovery message is valid"),
514                                },
515                            }
516                        }
517                        None => ExtensionRecoveryOutcome::Ready,
518                    };
519                    if let Err(error) = context_request::<_, ExtensionRecoveryCompleteResult>(
520                        &plugin_context,
521                        PLUGIN_RECOVERY_COMPLETE_METHOD,
522                        ExtensionRecoveryCompleteParams { outcome },
523                    )
524                    .await
525                    {
526                        eprintln!("cookie-agent plugin recovery completion failed: {error}");
527                    }
528                    recovery_running.store(false, Ordering::Release);
529                }));
530                Ok(Dispatch::Continue)
531            }
532            _ => Ok(Dispatch::Continue),
533        }
534    }
535}

545impl PluginServerBuilder {
546    /// Registers a declared tool and its asynchronous handler.
547    #[must_use]
548    pub fn tool<F, Fut>(mut self, declaration: ToolDecl, handler: F) -> Self
549    where
550        F: Fn(PluginContext, ExtensionToolCallParams) -> Fut + Send + Sync + 'static,
551        Fut: Future<Output = Result<ToolOutput, ToolFailure>> + Send + 'static,
552    {
553        if self.error.is_none()
554            && let Err(error) = validate_tool(
555                &declaration,
556                self.handlers
557                    .tools
558                    .iter()
559                    .any(|tool| tool.declaration.name == declaration.name),
560            )
561        {
562            self.error = Some(error);
563            return self;
564        }
565        self.handlers.tools.push(RegisteredTool {
566            declaration,
567            handler: Arc::new(move |context, request| Box::pin(handler(context, request))),
568        });
569        self
570    }
571
572    fn intercept<P, F, Fut, R>(mut self, handler: F) -> Self
573    where
574        P: InterceptParams,
575        F: Fn(PluginContext, P) -> Fut + Send + Sync + 'static,
576        Fut: Future<Output = R> + Send + 'static,
577        R: Into<P::Result> + Send + 'static,
578    {
579        let handler = Arc::new(handler);
580        let erased: InterceptHandler = Arc::new(move |params| {
581            let params: P = serde_json::from_value(params)?;
582            let handler = Arc::clone(&handler);
583            Ok(PreparedIntercept {
584                session_id: params.session_id(),
585                context_id: params.context_id().to_owned(),
586                call: Box::new(move |context| {
587                    let future = handler(context, params);
588                    Box::pin(async move {
589                        serde_json::to_value(future.await.into())
590                            .expect("interception results serialize")
591                    })
592                }),
593            })
594        });
595        self.handlers.intercepts.insert(P::HOOK, erased);
596        self
597    }

649    pub fn enable_bus_publishing(mut self) -> Self {
650        self.handlers.publish_bus = true;
651        self
652    }
653
654    /// Validates registrations and creates the server.
655    pub fn build(self) -> Result<PluginServer, PluginError> {
656        if let Some(error) = self.error {
657            return Err(error);
658        }
659        if !self.handlers.producers && self.handlers.recovery.is_some() {
660            return Err(PluginError::Protocol(
661                "on_recovery requires enable_producers".into(),
662            ));
663        }
664        Ok(PluginServer {
665            name: self.name,
666            version: self.version,
667            handlers: Arc::new(self.handlers),
668        })
669    }
670
671    /// Builds and runs this plugin on standard input and output.
672    pub async fn run_stdio(self) -> Result<(), PluginError> {
673        self.build()?.run_stdio().await
674    }

677/// Context passed to plugin handlers for publishing correlated events.
678#[derive(Clone)]
679pub struct PluginContext {
680    state: Arc<ContextState>,
681    grant: Option<ScopedGrant>,
682}
683
684impl PluginContext {
685    /// Publishes a non-durable bus event using this handler's one-shot context grant.
686    ///
687    /// The server must be configured with
688    /// [`PluginServerBuilder::enable_bus_publishing`]. Notification grants expire locally after
689    /// four seconds, one second before the engine's authoritative deadline.
690    pub async fn emit_bus(
691        &self,
692        session_id: SessionId,
693        name: impl Into<String>,
694        payload: Value,
695    ) -> Result<ExtensionEmitStatus, PluginError> {
696        self.emit(session_id, name.into(), payload).await
697    }
698
699    async fn emit(
700        &self,
701        session_id: SessionId,
702        name: String,
703        payload: Value,
704    ) -> Result<ExtensionEmitStatus, PluginError> {
705        if !self.state.publishing.bus {
706            return Err(PluginError::PublishingNotEnabled("bus"));
707        }
708        let context_id = self.consume(session_id)?;
709        let params = ExtensionEmitParams {
710            session_id,
711            context_id,
712            name: name.clone(),
713            payload,
714        };
715        let value = serde_json::to_value(Notification::new(
716            PLUGIN_EMIT_METHOD,
717            Some(serde_json::to_value(params)?),
718        ))?;
719        let (reply, receive) = oneshot::channel();
720        self.state
721            .outbound
722            .send(Outbound::Emit { value, name, reply })
723            .await
724            .map_err(|_| PluginError::TransportClosed)?;
725        let result = receive.await.map_err(|_| PluginError::TransportClosed)??;
726        Ok(result.bus)
727    }
728
729    /// Registers a long-lived producer for one session.
730    ///
731    /// The returned handle has no `Drop` cleanup. Call [`ProducerHandle::unregister`] explicitly,
732    /// including when no messages were sent.
733    pub async fn register_producer(
734        &self,
735        session_id: SessionId,
736    ) -> Result<ProducerHandle, PluginError> {
737        if !self.state.producers {
738            return Err(PluginError::ProducerMessagingNotEnabled);
739        }
740        let result: ExtensionProducerRegisterResult = context_request(
741            self,
742            PLUGIN_PRODUCER_REGISTER_METHOD,
743            ExtensionProducerRegisterParams { session_id },
744        )
745        .await?;
746        Ok(ProducerHandle {
747            context: self.clone(),
748            session_id,
749            producer_id: result.producer_id,
750        })
751    }
752
753    /// Discards this plugin's waiting message by session and durable receipt.
754    ///
755    /// No producer registration is required, including after reconnect. The engine
756    /// checks stable ownership and rejects consumed or currently claimed messages.
757    /// A durable actor claim removes a message from waiting before request preparation
758    /// and hooks. Discard rejects until release, even before network activity; the
759    /// claim does not prove provider receipt or execution. Release after failed
760    /// preparation or cancellation may return an unconsumed message to waiting;
761    /// consumed messages cannot return. Rejection makes no exactly-once execution
762    /// or external-effects guarantee. An already-discarded owned message succeeds.
763    /// This never unregisters a producer.
764    pub async fn discard_producer_message(
765        &self,
766        session_id: SessionId,
767        message_id: ProducerMessageId,
768    ) -> Result<(), PluginError> {
769        let _: ExtensionProducerDiscardResult = context_request(
770            self,
771            PLUGIN_PRODUCER_DISCARD_METHOD,
772            ExtensionProducerDiscardParams {
773                session_id,
774                message_id,
775            },
776        )
777        .await?;
778        Ok(())
779    }
780
781    fn register_notification(
782        &self,
783        session_id: SessionId,
784        context_id: String,
785        received_at: Instant,
786    ) -> Self {
787        self.register(
788            session_id,
789            context_id,
790            GrantExpiry::At(received_at + NOTIFICATION_CONTEXT_LIFETIME),
791        )
792    }
793
794    fn register_request(&self, session_id: SessionId, context_id: String) -> Self {
795        self.register(session_id, context_id, GrantExpiry::Request)
796    }
797
798    fn register(&self, session_id: SessionId, context_id: String, expiry: GrantExpiry) -> Self {
799        let now = Instant::now();
800        let mut grants = self
801            .state
802            .grants
803            .lock()
804            .unwrap_or_else(|poisoned| poisoned.into_inner());
805        grants.retain(
806            |_, grant| !matches!(grant.expiry, GrantExpiry::At(deadline) if deadline <= now),
807        );
808        grants.insert(context_id.clone(), Grant { session_id, expiry });
809        Self {
810            state: Arc::clone(&self.state),
811            grant: Some(ScopedGrant {
812                session_id,
813                context_id,
814            }),
815        }
816    }
817
818    fn consume(&self, session_id: SessionId) -> Result<String, PluginError> {
819        let Some(scope) = &self.grant else {
820            return Err(PluginError::ContextUnavailable(session_id));
821        };
822        if scope.session_id != session_id {
823            return Err(PluginError::ContextUnavailable(session_id));
824        }
825        let mut grants = self
826            .state
827            .grants
828            .lock()
829            .unwrap_or_else(|poisoned| poisoned.into_inner());
830        let Some(grant) = grants.get(&scope.context_id) else {
831            return Err(PluginError::ContextUnavailable(session_id));
832        };
833        if grant.session_id != session_id
834            || matches!(grant.expiry, GrantExpiry::At(deadline) if deadline <= Instant::now())
835        {
836            grants.remove(&scope.context_id);
837            return Err(PluginError::ContextUnavailable(session_id));
838        }
839        grants.remove(&scope.context_id);
840        Ok(scope.context_id.clone())
841    }
842
843    fn revoke(&self, context_id: &str) {
844        self.state
845            .grants
846            .lock()
847            .unwrap_or_else(|poisoned| poisoned.into_inner())
848            .remove(context_id);
849    }
850}

852/// A live producer registration bound to one session and plugin connection.
853///
854/// This handle intentionally performs no cleanup on drop. Explicit unregistration makes producer
855/// lifecycle completion observable and avoids implying that asynchronous cleanup was delivered.
856pub struct ProducerHandle {
857    context: PluginContext,
858    session_id: SessionId,
859    producer_id: ProducerId,
860}

862impl ProducerHandle {
863    /// Returns the runtime-only producer registration ID.
864    #[must_use]

866        self.producer_id
867    }
868
869    /// Returns the destination session fixed at registration.
870    #[must_use]
871    pub const fn session_id(&self) -> SessionId {
872        self.session_id

874
875    /// Durably sends a producer message using the selected per-send delivery mode.
876    pub async fn send(
877        &self,
878        message: impl Into<String>,
879        description: impl Into<String>,
880        mode: ProducerDeliveryMode,
881        key: ProducerIdempotencyKey,
882    ) -> Result<ProducerMessageId, PluginError> {
883        let description = cookie_agent_protocol::SafeDisplayText::new(description)
884            .map_err(|error| PluginError::Protocol(format!("invalid description: {error}")))?;
885        if description.as_str().trim().is_empty() {
886            return Err(PluginError::Protocol(
887                "description must not be blank".into(),
888            ));
889        }
890        let result: ExtensionProducerSendResult = context_request(
891            &self.context,
892            PLUGIN_PRODUCER_SEND_METHOD,
893            ExtensionProducerSendParams {
894                session_id: self.session_id,
895                producer_id: self.producer_id,
896                mode,
897                idempotency_key: key,
898                description,
899                body: message.into(),
900            },
901        )
902        .await?;
903        Ok(result.message_id)
904    }

906    /// Durably sends a message to the next safe model request.
907    pub async fn steer(
908        &self,
909        message: impl Into<String>,
910        description: impl Into<String>,
911        key: ProducerIdempotencyKey,
912    ) -> Result<ProducerMessageId, PluginError> {
913        self.send(message, description, ProducerDeliveryMode::Steer, key)
914            .await
915    }
916
917    /// Durably sends a message for a subsequent normal run.
918    pub async fn queue(
919        &self,
920        message: impl Into<String>,
921        description: impl Into<String>,
922        key: ProducerIdempotencyKey,
923    ) -> Result<ProducerMessageId, PluginError> {
924        self.send(message, description, ProducerDeliveryMode::Queue, key)
925            .await
926    }

928    /// Discards an owned waiting message in this handle's session.
929    ///
930    /// Still usable after unregister: message ownership is independent of this
931    /// registration. A durable actor claim removes the message from waiting before
932    /// request preparation and hooks, so discard rejects while claimed, not just
933    /// after network dispatch. See [`PluginContext::discard_producer_message`] for
934    /// claim release and consumption semantics.
935    pub async fn discard(&self, message_id: ProducerMessageId) -> Result<(), PluginError> {
936        self.context
937            .discard_producer_message(self.session_id, message_id)
938            .await
939    }
940
941    /// Explicitly unregisters this producer. Zero-send unregistration is valid.
942    ///
943    /// Failures leave the handle available for retry. After success, the engine remains
944    /// authoritative and rejects further sends on the closed registration. Discard
945    /// operates on message receipts and remains available after unregister.

947        let _: ExtensionProducerUnregisterResult = context_request(
948            &self.context,
949            PLUGIN_PRODUCER_UNREGISTER_METHOD,
950            ExtensionProducerUnregisterParams {
951                session_id: self.session_id,
952                producer_id: self.producer_id,
953            },
954        )
955        .await?;
956        Ok(())
957    }

1036/// Declares every interception hook once: its builder method, wire method, params, and result.
1037/// Rows are in capability-advertisement order.
1038macro_rules! interception_hooks {
1039    ($(
1040        $(#[doc = $doc:literal])+
1041        $builder:ident: $hook:ident, $method:ident, $params:ty => $result:ty;
1042    )+) => {
1043        const INTERCEPTION_HOOKS: &[ExtensionInterceptionHook] =
1044            &[$(ExtensionInterceptionHook::$hook),+];
1045
1046        fn interception_hook(method: &str) -> Option<ExtensionInterceptionHook> {
1047            match method {
1048                $($method => Some(ExtensionInterceptionHook::$hook),)+
1049                _ => None,
1050            }
1051        }
1052
1053        $(impl InterceptParams for $params {
1054            const HOOK: ExtensionInterceptionHook = ExtensionInterceptionHook::$hook;
1055            type Result = $result;
1056
1057            fn session_id(&self) -> SessionId {
1058                self.session_id
1059            }
1060
1061            fn context_id(&self) -> &str {
1062                &self.context_id
1063            }
1064        })+
1065
1066        impl PluginServerBuilder {
1067            $(
1068                $(#[doc = $doc])+
1069                #[must_use]
1070                pub fn $builder<F, Fut, R>(self, handler: F) -> Self
1071                where
1072                    F: Fn(PluginContext, $params) -> Fut + Send + Sync + 'static,
1073                    Fut: Future<Output = R> + Send + 'static,
1074                    R: Into<$result> + Send + 'static,
1075                {
1076                    self.intercept(handler)
1077                }
1078            )+
1079        }
1080    };
1081}

1125async fn dispatch_intercept(
1126    request: Request,
1127    context: &PluginContext,
1128    handler: Option<InterceptHandler>,
1129    tasks: &mut Vec<JoinHandle<()>>,
1130    handler_slots: &Arc<Semaphore>,
1131) -> Result<(), PluginError> {
1132    let Some(handler) = handler else {
1133        send_error(
1134            context,
1135            request.id,
1136            METHOD_NOT_FOUND,
1137            "interception hook is not registered",
1138        )
1139        .await?;
1140        return Ok(());
1141    };
1142    let Some(permit) = try_handler_slot(context, &request, handler_slots).await? else {
1143        return Ok(());
1144    };
1145    let prepared = match handler(request.params.clone().unwrap_or(Value::Null)) {
1146        Ok(prepared) => prepared,
1147        Err(error) => {
1148            send_error(context, request.id, INVALID_PARAMS, error.to_string()).await?;
1149            return Ok(());
1150        }
1151    };
1152    let PreparedIntercept {
1153        session_id,
1154        context_id,
1155        call,
1156    } = prepared;
1157    let plugin_context = context.register_request(session_id, context_id.clone());
1158    tasks.push(tokio::spawn(async move {
1159        let _permit = permit;
1160        let handler_context = plugin_context.clone();
1161        let result = isolate(async move { call(handler_context).await }).await;
1162        plugin_context.revoke(&context_id);
1163        match result {
1164            Ok(result) => {
1165                let _ = send_success(&plugin_context, request.id, result).await;
1166            }
1167            Err(()) => {
1168                let _ = send_error(
1169                    &plugin_context,
1170                    request.id,
1171                    INTERNAL_ERROR,
1172                    "plugin interception handler panicked",
1173                )
1174                .await;
1175            }
1176        }
1177    }));
1178    Ok(())
1179}

1181async fn try_handler_slot(
1182    context: &PluginContext,
1183    request: &Request,
1184    slots: &Arc<Semaphore>,
1185) -> Result<Option<OwnedSemaphorePermit>, PluginError> {
1186    match Arc::clone(slots).try_acquire_owned() {
1187        Ok(permit) => Ok(Some(permit)),
1188        Err(_) => {
1189            send_error(
1190                context,
1191                request.id.clone(),
1192                SERVER_BUSY,
1193                "plugin handler capacity exhausted",
1194            )
1195            .await?;
1196            Ok(None)
1197        }
1198    }
1199}

1271async fn context_request<P, R>(
1272    context: &PluginContext,
1273    method: &'static str,
1274    params: P,
1275) -> Result<R, PluginError>
1276where
1277    P: Serialize,
1278    R: serde::de::DeserializeOwned,
1279{
1280    if !context.state.producers {
1281        return Err(PluginError::ProducerMessagingNotEnabled);
1282    }
1283    if !context.state.connected.load(Ordering::Acquire) {
1284        return Err(PluginError::TransportClosed);
1285    }
1286    let permit = Arc::clone(&context.state.pending_slots)
1287        .try_acquire_owned()
1288        .map_err(|_| PluginError::TooManyPendingRequests)?;
1289    let id = context.state.request_ids.fetch_add(1, Ordering::Relaxed);
1290    let request = Request::new(
1291        JsonRpcId::Number(id),
1292        method,
1293        Some(serde_json::to_value(params)?),
1294    );
1295    let (reply, receive) = oneshot::channel();
1296    context
1297        .state
1298        .outbound
1299        .send(Outbound::Request {
1300            value: serde_json::to_value(request)?,
1301            id,
1302            pending: PendingRequest {
1303                reply,
1304                _permit: permit,
1305            },
1306        })
1307        .await
1308        .map_err(|_| PluginError::TransportClosed)?;
1309    let value = receive.await.map_err(|_| PluginError::TransportClosed)??;
1310    serde_json::from_value(value)
1311        .map_err(|error| PluginError::Protocol(format!("invalid result for `{method}`: {error}")))
1312}

1314async fn reader_loop<R>(reader: R, sender: mpsc::Sender<Result<InboundFrame, PluginError>>)
1315where
1316    R: AsyncRead + Unpin,
1317{
1318    let mut reader = BufReader::new(reader);
1319    loop {
1320        match read_frame(&mut reader).await {
1321            Ok(Some(frame)) => {
1322                let frame = InboundFrame {
1323                    message: frame,
1324                    received_at: Instant::now(),
1325                };
1326                if sender.send(Ok(frame)).await.is_err() {
1327                    return;
1328                }
1329            }
1330            Ok(None) => return,
1331            Err(error) => {
1332                let _ = sender.send(Err(error)).await;
1333                return;
1334            }
1335        }
1336    }
1337}

1339async fn writer_loop<W>(
1340    mut writer: W,
1341    mut receiver: mpsc::Receiver<Outbound>,
1342    pending_emits: PendingEmits,
1343    pending_requests: PendingRequests,
1344) -> Result<(), PluginError>
1345where
1346    W: AsyncWrite + Unpin,
1347{
1348    while let Some(outbound) = receiver.recv().await {
1349        let (value, emit, request) = match outbound {
1350            Outbound::Message(value) => (value, None, None),
1351            Outbound::Emit { value, name, reply } => (value, Some((name, reply)), None),
1352            Outbound::Request { value, id, pending } => (value, None, Some((id, pending))),
1353        };
1354        let bytes = serde_json::to_vec(&value)?;
1355        if bytes.len() > MAX_FRAME_BYTES {
1356            if let Some((_, reply)) = emit {
1357                let _ = reply.send(Err(PluginError::Protocol(format!(
1358                    "plugin frame exceeds {MAX_FRAME_BYTES} bytes"
1359                ))));
1360                continue;
1361            }
1362            if let Some((_, pending)) = request {
1363                let _ = pending.reply.send(Err(PluginError::Protocol(format!(
1364                    "plugin frame exceeds {MAX_FRAME_BYTES} bytes"
1365                ))));
1366                continue;
1367            }
1368            return Err(PluginError::Protocol(format!(
1369                "plugin frame exceeds {MAX_FRAME_BYTES} bytes"
1370            )));
1371        }
1372        if let Some((name, reply)) = emit {
1373            pending_emits
1374                .lock()
1375                .unwrap_or_else(|poisoned| poisoned.into_inner())
1376                .entry(name)
1377                .or_default()
1378                .push_back(reply);
1379        }
1380        if let Some((id, request)) = request {
1381            pending_requests
1382                .lock()
1383                .unwrap_or_else(|poisoned| poisoned.into_inner())
1384                .insert(id, request);
1385        }
1386        writer.write_all(&bytes).await?;
1387        writer.write_all(b"\n").await?;
1388        writer.flush().await?;
1389    }
1390    Ok(())
1391}

1394    task: &mut JoinHandle<Result<(), PluginError>>,
1395) -> Result<(), PluginError> {

1398        Err(_) => Err(PluginError::TransportClosed),
1399    }

1402fn resolve_emit(pending: &PendingEmits, result: ExtensionEmitResultParams) {
1403    let reply = {
1404        let mut pending = pending
1405            .lock()
1406            .unwrap_or_else(|poisoned| poisoned.into_inner());
1407        let reply = pending.get_mut(&result.name).and_then(VecDeque::pop_front);
1408        if pending.get(&result.name).is_some_and(VecDeque::is_empty) {
1409            pending.remove(&result.name);
1410        }
1411        reply
1412    };
1413    if let Some(reply) = reply {
1414        let _ = reply.send(Ok(result));
1415    }
1416}

1421        .unwrap_or_else(|poisoned| poisoned.into_inner())
1422        .drain()
1423        .flat_map(|(_, replies)| replies)

1430fn resolve_request(pending: &PendingRequests, response: Response) -> Result<(), PluginError> {
1431    let (id, result) = match response {
1432        Response::Success(response) => (response.id, Ok(response.result)),
1433        Response::Error(response) => (
1434            response.id,
1435            Err(PluginError::EngineRequest {
1436                code: response.error.code,
1437                message: response.error.message,
1438                data: response.error.data,
1439            }),
1440        ),
1441    };
1442    let JsonRpcId::Number(id) = id else {
1443        return Err(PluginError::Protocol(
1444            "engine response used a non-numeric plugin request ID".into(),
1445        ));
1446    };
1447    let request = pending
1448        .lock()
1449        .unwrap_or_else(|poisoned| poisoned.into_inner())
1450        .remove(&id)
1451        .ok_or_else(|| PluginError::Protocol(format!("engine response has unknown ID {id}")))?;
1452    let _ = request.reply.send(result);
1453    Ok(())
1454}

1456fn fail_pending_requests(pending: &PendingRequests) {
1457    let requests = pending
1458        .lock()
1459        .unwrap_or_else(|poisoned| poisoned.into_inner())
1460        .drain()
1461        .map(|(_, request)| request)
1462        .collect::<Vec<_>>();
1463    for request in requests {
1464        let _ = request.reply.send(Err(PluginError::TransportClosed));
1465    }
1466}

1468fn validate_tool(tool: &ToolDecl, duplicate: bool) -> Result<(), PluginError> {
1469    if duplicate {
1470        return Err(PluginError::InvalidTool(format!(
1471            "duplicate tool name `{}`",
1472            tool.name
1473        )));
1474    }
1475    if !is_snake_case(&tool.name) {
1476        return Err(PluginError::InvalidTool(format!(
1477            "tool name `{}` must be snake_case",
1478            tool.name
1479        )));
1480    }
1481    if tool.description.is_empty() {
1482        return Err(PluginError::InvalidTool(format!(
1483            "tool `{}` description must not be empty",
1484            tool.name
1485        )));
1486    }
1487    if !is_snake_case(&tool.permission_name) {
1488        return Err(PluginError::InvalidTool(format!(
1489            "tool `{}` permission_name must be snake_case",
1490            tool.name
1491        )));
1492    }
1493    if !tool.parameters.is_object() {
1494        return Err(PluginError::InvalidTool(format!(
1495            "tool `{}` parameters must be a JSON Schema object",
1496            tool.name
1497        )));
1498    }
1499    jsonschema::draft202012::meta::validate(&tool.parameters).map_err(|error| {
1500        PluginError::InvalidTool(format!(
1501            "tool `{}` has invalid JSON Schema: {error}",
1502            tool.name
1503        ))
1504    })?;
1505    if let Some(primary) = &tool.primary_resource_param
1506        && !tool
1507            .parameters
1508            .get("properties")
1509            .and_then(Value::as_object)
1510            .is_some_and(|properties| properties.contains_key(primary))
1511    {
1512        return Err(PluginError::InvalidTool(format!(
1513            "tool `{}` primary_resource_param `{primary}` is not declared in properties",
1514            tool.name
1515        )));
1516    }
1517    Ok(())
1518}

1520fn is_snake_case(value: &str) -> bool {
1521    !value.is_empty()
1522        && value.split('_').all(|word| {
1523            !word.is_empty()
1524                && word
1525                    .bytes()
1526                    .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit())
1527        })
1528        && value.as_bytes()[0].is_ascii_lowercase()
1529}
