1//! Delays between retries: one policy type for every retry loop.
2//!
3//! A [`Backoff`] doubles from `base` up to `cap`, optionally jittered. It
4//! only computes delays; callers sleep (on whichever executor they run) and
5//! decide how many attempts to make.
6
7use std::time::Duration;
8
9/// How a [`Backoff`] spreads its delays.
10#[derive(Clone, Copy, Debug, Eq, PartialEq)]
11pub enum Jitter {
12    /// Every attempt waits exactly its delay.
13    None,
14    /// Adds up to this much, in whole milliseconds, to each delay, which can
15    /// then exceed the cap by that much.
16    Add(Duration),
17    /// Scales each delay by a per-mille factor between `low` and `high`
18    /// inclusive, then keeps it between one millisecond and the cap.
19    Scale {
20        /// The smallest factor, in thousandths.
21        low: u32,
22        /// The largest factor, in thousandths.
23        high: u32,
24    },
25}
26
27/// Exponential backoff: attempt `n` (from 0) waits `base * 2^n`, at most
28/// `cap`, then [`Jitter`].
29#[derive(Clone, Copy, Debug, Eq, PartialEq)]
30pub struct Backoff {
31    /// The first delay.
32    pub base: Duration,
33    /// The longest delay before jitter.
34    pub cap: Duration,
35    /// How the delays are spread.
36    pub jitter: Jitter,
37}
38
39impl Backoff {
40    /// Doubles from `base` up to `cap`, without jitter.
41    #[must_use]
42    pub const fn new(base: Duration, cap: Duration) -> Self {
43        Self {
44            base,
45            cap,
46            jitter: Jitter::None,
47        }
48    }
49
50    /// Waits `delay` before every attempt.
51    #[must_use]
52    pub const fn fixed(delay: Duration) -> Self {
53        Self::new(delay, delay)
54    }
55
56    /// The same backoff with `jitter`.
57    #[must_use]
58    pub const fn with_jitter(self, jitter: Jitter) -> Self {
59        Self { jitter, ..self }
60    }
61
62    /// The delay before retry `attempt` (from 0), before jitter: `base`
63    /// doubled `attempt` times, at most `cap`.
64    #[must_use]
65    pub fn delay(&self, attempt: u32) -> Duration {
66        let multiplier = 1_u128.checked_shl(attempt).unwrap_or(u128::MAX);
67        let nanos = self
68            .base
69            .as_nanos()
70            .saturating_mul(multiplier)
71            .min(self.cap.as_nanos());
72        from_nanos(nanos)
73    }
74
75    /// The delay before retry `attempt` with jitter drawn from `entropy`,
76    /// any number the caller has (a clock reading, a random draw, a hash).
77    #[must_use]
78    pub fn jittered(&self, attempt: u32, entropy: u64) -> Duration {
79        let delay = self.delay(attempt);
80        match self.jitter {
81            Jitter::None => delay,
82            Jitter::Add(most) => {
83                let span = most.as_millis().saturating_add(1);
84                delay.saturating_add(from_millis(u128::from(entropy) % span))
85            }
86            Jitter::Scale { low, high } => {
87                let span = u64::from(high.saturating_sub(low)) + 1;
88                let per_mille = u64::from(low) + entropy % span;
89                let millis = delay.as_millis().saturating_mul(u128::from(per_mille)) / 1000;
90                from_millis(millis.clamp(1, self.cap.as_millis().max(1)))
91            }
92        }
93    }
94
95    /// The un-jittered delays of successive retries, from attempt 0.
96    #[must_use]
97    pub const fn delays(self) -> Delays {
98        Delays {
99            backoff: self,
100            attempt: 0,
101        }
102    }
103}
104
105/// The delays of a [`Backoff`], one per retry; see [`Backoff::delays`].
106#[derive(Clone, Debug)]
107pub struct Delays {
108    backoff: Backoff,
109    attempt: u32,
110}
111
112impl Delays {
113    /// The next delay; the delays never run out.
114    pub fn next_delay(&mut self) -> Duration {
115        let delay = self.backoff.delay(self.attempt);
116        self.attempt = self.attempt.saturating_add(1);
117        delay
118    }
119
120    /// Starts over from the first delay.
121    pub fn reset(&mut self) {
122        self.attempt = 0;
123    }
124}
125
126impl Iterator for Delays {
127    type Item = Duration;
128
129    fn next(&mut self) -> Option<Duration> {
130        Some(self.next_delay())
131    }
132}
133
134fn from_nanos(nanos: u128) -> Duration {
135    const PER_SECOND: u128 = 1_000_000_000;
136    let seconds = u64::try_from(nanos / PER_SECOND).unwrap_or(u64::MAX);
137    let subsecond = u32::try_from(nanos % PER_SECOND).expect("below one billion");
138    Duration::new(seconds, subsecond)
139}
140
141fn from_millis(millis: u128) -> Duration {
142    from_nanos(millis.saturating_mul(1_000_000))
143}

145#[cfg(test)]
146mod tests {
147    use std::time::Duration;
148
149    use super::{Backoff, Jitter};
150
151    const fn ms(value: u64) -> Duration {
152        Duration::from_millis(value)
153    }
154
155    #[test]
156    fn delays_double_from_the_base_up_to_the_cap() {
157        let backoff = Backoff::new(ms(100), ms(1_000));
158        let delays: Vec<_> = backoff.delays().take(6).collect();
159        assert_eq!(
160            delays,
161            [ms(100), ms(200), ms(400), ms(800), ms(1_000), ms(1_000)]
162        );
163    }
164
165    #[test]
166    fn a_huge_attempt_saturates_at_the_cap() {
167        let backoff = Backoff::new(ms(1_000), Duration::from_secs(60));
168        assert_eq!(backoff.delay(63), Duration::from_secs(60));
169        assert_eq!(backoff.delay(200), Duration::from_secs(60));
170        let unbounded = Backoff::new(Duration::from_secs(1), Duration::MAX);
171        assert_eq!(unbounded.delay(u32::MAX), Duration::MAX);
172    }
173
174    #[test]
175    fn a_fixed_backoff_never_grows() {
176        let delays: Vec<_> = Backoff::fixed(ms(25)).delays().take(3).collect();
177        assert_eq!(delays, [ms(25); 3]);
178    }
179
180    #[test]
181    fn reset_starts_from_the_base_again() {
182        let mut delays = Backoff::new(ms(10), ms(80)).delays();
183        assert_eq!(delays.nth(2), Some(ms(40)));
184        delays.reset();
185        assert_eq!(delays.next(), Some(ms(10)));
186    }
187
188    #[test]
189    fn additive_jitter_adds_at_most_its_bound() {
190        let backoff = Backoff::new(ms(2), ms(40)).with_jitter(Jitter::Add(ms(10)));
191        assert_eq!(backoff.jittered(0, 0), ms(2));
192        assert_eq!(backoff.jittered(0, 10), ms(12));
193        // The jitter wraps at its bound plus one.
194        assert_eq!(backoff.jittered(0, 11), ms(2));
195        assert_eq!(backoff.jittered(9, 10), ms(50));
196    }
197
198    #[test]
199    fn scaled_jitter_stays_between_one_millisecond_and_the_cap() {
200        let backoff = Backoff::new(ms(1_000), ms(60_000)).with_jitter(Jitter::Scale {
201            low: 750,
202            high: 1_250,
203        });
204        assert_eq!(backoff.jittered(0, 0), ms(750));
205        assert_eq!(backoff.jittered(0, 500), ms(1_250));
206        assert_eq!(backoff.jittered(0, 501), ms(750));
207        // Scaled above the cap, it is the cap.
208        assert_eq!(backoff.jittered(10, 500), ms(60_000));
209        let tiny = Backoff::new(Duration::from_micros(1), ms(5)).with_jitter(Jitter::Scale {
210            low: 750,
211            high: 1_250,
212        });
213        assert_eq!(tiny.jittered(0, 0), ms(1));
214    }
215
216    #[test]
217    fn no_jitter_is_the_plain_delay() {
218        let backoff = Backoff::new(ms(50), ms(5_000));
219        assert_eq!(backoff.jittered(3, u64::MAX), ms(400));
220    }
