31impl MessageToolProvider {
32    #[must_use]
33    pub fn new(engine: Engine) -> Self {
34        Self { engine }
35    }
36
37    fn parse(args: &serde_json::Value) -> Result<MessageArgs, ToolError> {
38        let value: MessageArgs = serde_json::from_value(args.clone())
39            .map_err(|error| ToolError::execution(error.to_string()))?;
40        if value.body.trim().is_empty() {
41            return Err(ToolError::execution("body must not be empty"));
42        }
43        Ok(value)
44    }
45}

47#[async_trait]
48impl ToolProvider for MessageToolProvider {
49    fn provider_id(&self) -> &'static str {
50        "builtin.message"
51    }
52
53    fn tools_for_session(&self, _ctx: &SessionToolContext) -> Result<Vec<ToolSpec>, ToolError> {
54        Ok(vec![ToolSpec {
55            output: Default::default(),
56            concurrency: cookie_agent_engine::ToolConcurrency::Parallel,
57            result_truncation: cookie_agent_engine::ToolResultTruncationPolicy::Bounded,
58            name: "send_message".into(),
59            permission_name: "message".into(),
60            description: "Send a message to an agent in the current delegation tree.".into(),
61            parameters: serde_json::json!({
62                "type":"object","additionalProperties":false,
63                "properties":{
64                    "recipient_session_id":{"type":"string"},
65                    "body":{"type":"string","minLength":1},
66                    "mode":{"type":"string","enum":["steer","queue"],"default":"steer"}
67                },"required":["recipient_session_id","body"]
68            }),
69        }])
70    }
71
72    fn get_permission_name(_name: &str) -> Result<&'static str, ToolError> {
73        Ok("message")
74    }
75
76    fn get_permission_resource(
77        &self,
78        _name: &str,
79        arguments: &serde_json::Value,
80    ) -> Result<(&'static str, Option<String>), ToolError> {
81        let _ = Self::parse(arguments)?;
82        Ok(("message", Some("*".into())))
83    }
84
85    fn get_display_argument(
86        &self,
87        _name: &str,
88        arguments: &serde_json::Value,
89    ) -> Result<String, ToolError> {
90        Ok(Self::parse(arguments)?.recipient_session_id.to_string())
91    }
92
93    async fn prepare(
94        &self,
95        ctx: ToolPreparationContext,
96        call: ToolCall,
97    ) -> Result<PreparedTool, ToolError> {
98        if call.name != "send_message" {
99            return Err(ToolError::execution(
100                "message provider received another tool",
101            ));
102        }
103        let args = Self::parse(&call.arguments)?;
104        let relationship = self
105            .engine
106            .message_relationship(ctx.session, args.recipient_session_id)
107            .map_err(|error| ToolError::execution(error.to_string()))?;
108        let cwd = fs_cap::cwd_context_bytes(&ctx.cwd)?;
109        let operation = prepared_operation(
110            "send_message",
111            &args,
112            vec![(PermissionAction::Message, relationship)],
113            vec![prepared_resource(
114                PermissionAction::Message,
115                relationship,
116                relationship.as_bytes(),
117                relationship.as_bytes(),
118                cookie_agent_protocol::PreparedBindingLifetime::RestartStable,
119                cookie_agent_protocol::ApprovalResourceSource::PrimaryOperation,
120            )?],
121            &cwd,
122        )?;
123        let normalized =
124            serde_json::to_value(&args).map_err(|error| ToolError::execution(error.to_string()))?;
125        PreparedTool::new(
126            operation,
127            normalized,
128            None,
129            Box::new(MessageExecutor {
130                engine: self.engine.clone(),
131                args,
132                call_id: call.id,
133            }),
134        )?
135        .with_policy_labels(vec![relationship.into()])
136    }
137}

139struct MessageExecutor {
140    engine: Engine,
141    args: MessageArgs,
142    call_id: cookie_agent_protocol::ToolCallId,
143}
144
145#[async_trait]
146impl PreparedExecutor for MessageExecutor {
147    async fn revalidate(&self) -> Result<(), ToolError> {
148        Ok(())
149    }
150    async fn execute(
151        self: Box<Self>,
152        context: ToolExecutionContext,
153    ) -> Result<ToolCompletion, ToolError> {
154        let handle = self
155            .engine
156            .send_agent_message(AgentMessageInvocation {
157                sender_session_id: context.session,
158                sender_run_id: context.run,
159                sender_tool_call_id: self.call_id,
160                recipient_session_id: self.args.recipient_session_id,
161                body: self.args.body,
162                mode: self.args.mode,
163            })
164            .await
165            .map_err(|error| ToolError::execution(error.to_string()))?;
166        let metadata = serde_json::json!({"message_id": handle.message_id, "mode": handle.mode, "recipient_state": handle.recipient_state.as_str()});
167        Ok(ToolCompletion::single(ToolResult {
168            display: None,
169            retained_output: None,
170            title: safe_title("Message sent"),
171            output: format!("Message accepted for {}.", handle.recipient_state.as_str()),
172            metadata,
173            truncation: None,
174            attachments: Vec::new(),
175            additional_messages: Vec::new(),
176        }))
177    }
178}
