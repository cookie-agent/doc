30impl WebfetchArgs {
31    fn validate(&self) -> Result<(), ToolError> {
32        if !self.url.starts_with("http://") && !self.url.starts_with("https://") {
33            return Err(ToolError::InvalidUrl(
34                "URL must start with http:// or https://".into(),
35            ));
36        }
37        Ok(())
38    }
39}

42impl ToolProvider for WebfetchTool {
43    fn provider_id(&self) -> &'static str {
44        "builtin.webfetch"
45    }

47    fn tools_for_session(&self, _ctx: &SessionToolContext) -> Result<Vec<ToolSpec>, ToolError> {
48        Ok(vec![ToolSpec {
49            name: "webfetch".into(),
50            permission_name: "webfetch".into(),
51            description: "Fetch an HTTP or HTTPS URL. Returns HTML as plaintext unless raw is true; other text passes through. Long results can be paged with read_tool_result.".into(),
52            parameters: schema::<WebfetchArgs>(),
53            concurrency: cookie_agent_engine::ToolConcurrency::Parallel,
54            result_truncation: Default::default(),

56    }

58    fn get_permission_name(name: &str) -> Result<&'static str, ToolError> {
59        if name != "webfetch" {
60            return Err(tool_error("webfetch provider received another tool"));
61        }
62        Ok("webfetch")
63    }
64
65    fn get_permission_resource(
66        &self,
67        name: &str,
68        arguments: &serde_json::Value,
69    ) -> Result<(&'static str, Option<String>), ToolError> {
70        let permission = Self::get_permission_name(name)?;
71        let args: WebfetchArgs = parse_args(name, arguments.clone())?;
72        args.validate()?;
73        Ok((permission, Some(args.url)))
74    }
75
76    fn get_display_argument(
77        &self,
78        name: &str,
79        arguments: &serde_json::Value,
80    ) -> Result<String, ToolError> {
81        let (_, resource) = self.get_permission_resource(name, arguments)?;
82        Ok(resource.expect("webfetch has a URL resource"))
83    }

85    async fn prepare(
86        &self,
87        _ctx: ToolPreparationContext,
88        call: ToolCall,
89    ) -> Result<PreparedTool, ToolError> {
90        Self::get_permission_name(&call.name)?;
91        let args: WebfetchArgs = parse_args(&call.name, call.arguments)?;
92        args.validate()?;
93        let resource = prepared_resource(
94            PermissionAction::Webfetch,
95            "url",
96            args.url.as_bytes(),
97            args.url.as_bytes(),
98            PreparedBindingLifetime::RestartStable,
99            ApprovalResourceSource::PrimaryOperation,
100        )?;
101        let operation = prepared_operation(
102            "webfetch",
103            &args,
104            vec![(PermissionAction::Webfetch, "get")],
105            vec![resource],
106            b"webfetch:v1",
107        )?;
108        let normalized = serde_json::to_value(&args).map_err(tool_error)?;
109        let label = args.url.clone();
110        PreparedTool::new(operation, normalized, None, Box::new(args))?
111            .with_policy_labels(vec![label])
112    }
113}

115fn request_error(error: reqwest::Error) -> ToolError {
116    if error.is_timeout() {
117        ToolError::Timeout(error.to_string())
118    } else if error.is_redirect() {
119        ToolError::RedirectError(error.to_string())
120    } else if error.is_builder() {
121        ToolError::InvalidUrl(error.to_string())
122    } else {
123        ToolError::TransportError(error.to_string())
124    }
125}

127impl WebfetchArgs {
128    async fn fetch(&self) -> Result<PersistedToolResult, ToolError> {
129        let client = reqwest::Client::builder()
130            .timeout(REQUEST_TIMEOUT)
131            .build()
132            .map_err(|error| ToolError::TransportError(error.to_string()))?;
133        let mut response = client.get(&self.url).send().await.map_err(request_error)?;
134        let final_url = response.url().to_string();
135        let status_code = response.status().as_u16();
136        let content_type = response
137            .headers()
138            .get(reqwest::header::CONTENT_TYPE)
139            .and_then(|value| value.to_str().ok())
140            .unwrap_or("application/octet-stream")
141            .to_owned();
142        let mime = content_type
143            .split(';')
144            .next()
145            .unwrap_or("")
146            .trim()
147            .to_ascii_lowercase();
148        let html = matches!(mime.as_str(), "text/html" | "application/xhtml+xml");
149        if !(mime.starts_with("text/")
150            || matches!(
151                mime.as_str(),
152                "application/json"
153                    | "application/xml"
154                    | "application/javascript"
155                    | "application/x-www-form-urlencoded"
156            )
157            || mime.ends_with("+json")
158            || mime.ends_with("+xml"))
159        {
160            return Err(ToolError::UnsupportedContentType(content_type));
161        }
162        let mut body = Vec::new();
163        let mut truncated = false;
164        while let Some(chunk) = response.chunk().await.map_err(request_error)? {
165            let remaining = DOWNLOAD_CAP - body.len();
166            body.extend_from_slice(&chunk[..chunk.len().min(remaining)]);
167            if chunk.len() > remaining {
168                truncated = true;
169                break;
170            }
171        }
172        let text = if html && !self.raw {
173            html2text::from_read(body.as_slice(), 80).map_err(tool_error)?
174        } else {
175            String::from_utf8_lossy(&body).into_owned()
176        };
177        let metadata = serde_json::json!({
178            "url": self.url,
179            "final_url": final_url,
180            "status_code": status_code,
181            "content_type": content_type,
182            "truncated": truncated,
183        });
184        let mut output = metadata.clone();
185        output["text"] = text.into();
186        Ok(PersistedToolResult {
187            title: safe_title(&self.url),
188            output: serde_json::to_string(&output).map_err(tool_error)?,
189            metadata,
190            truncation: None,
191            attachments: Vec::new(),
192            additional_messages: Vec::new(),
193        })
194    }
195}

219    fn server() -> String {
220        let listener = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
221        let address = listener.local_addr().unwrap();
222        listener.set_nonblocking(true).unwrap();
223        std::thread::spawn(move || {
224            let deadline = std::time::Instant::now() + Duration::from_secs(10);
225            while std::time::Instant::now() < deadline {
226                let (mut stream, _) = match listener.accept() {
227                    Ok(connection) => connection,
228                    Err(error) if error.kind() == std::io::ErrorKind::WouldBlock => {
229                        std::thread::sleep(Duration::from_millis(5));
230                        continue;
231                    }
232                    Err(error) => panic!("fixture accept: {error}"),
233                };
234                stream
235                    .set_read_timeout(Some(Duration::from_secs(2)))
236                    .unwrap();
237                let mut request = [0; 4096];
238                let count = stream.read(&mut request).unwrap();
239                let request = String::from_utf8_lossy(&request[..count]);
240                let path = request.split_whitespace().nth(1).unwrap();
241                let (status, headers, body) = match path {
242                    "/first" => ("302 Found", "Location: /second\r\n", Vec::new()),
243                    "/second" => ("301 Moved Permanently", "Location: /html\r\n", Vec::new()),
244                    "/loop" => ("302 Found", "Location: /loop\r\n", Vec::new()),
245                    "/html" => (
246                        "200 OK",
247                        "Content-Type: text/html; charset=utf-8\r\n",
248                        b"<h1>Hello</h1><p>World <b>wide</b> web.</p>".to_vec(),
249                    ),
250                    "/cap" => (
251                        "200 OK",
252                        "Content-Type: text/plain\r\n",
253                        vec![b'x'; DOWNLOAD_CAP + 1],
254                    ),
255                    "/exact" => (
256                        "200 OK",
257                        "Content-Type: text/plain\r\n",
258                        vec![b'x'; DOWNLOAD_CAP],
259                    ),
260                    "/binary" => ("200 OK", "Content-Type: image/png\r\n", vec![0, 255]),
261                    _ => (
262                        "404 Not Found",
263                        "Content-Type: application/json\r\n",
264                        b"{\"error\":\"missing\"}".to_vec(),
265                    ),
266                };
267                let _ = write!(
268                    stream,
269                    "HTTP/1.1 {status}\r\n{headers}Content-Length: {}\r\nConnection: close\r\n\r\n",
270                    body.len()
271                );
272                let _ = stream.write_all(&body);
273            }
274        });
275        format!("http://{address}")
276    }

378    #[tokio::test]
379    async fn one_initial_url_resource_includes_query_and_defaults_to_deny() {
380        use cookie_agent_protocol::{
381            AgentDocumentSource, AgentId, AgentMode, AgentSchemaVersion, AgentSnapshot,
382            PermissionRule, Sha256Digest, WildcardPattern,
383        };
384        let url = format!("{}/html?token=given", server());
385        let prepared = prepare(serde_json::json!({"url":url})).await.unwrap();
386        assert_eq!(prepared.operation().resources().len(), 1);
387        assert_eq!(prepared.policy_labels(), [Some(url.clone())]);
388        assert_eq!(
389            WebfetchTool
390                .get_display_argument("webfetch", prepared.normalized_arguments())
391                .unwrap(),
392            url
393        );
394        let mut policy = AgentSnapshot {
395            agent: AgentId::new("test").unwrap(),
396            schema: AgentSchemaVersion::current(),
397            mode: AgentMode::Primary,
398            description: "Test".into(),
399            document_source: AgentDocumentSource::Workspace,
400            document_fingerprint: Sha256Digest::of_bytes(b"test"),
401            composed_prompt: "test".into(),
402            prompt_fingerprint: Sha256Digest::of_bytes(b"test"),
403            max_output_tokens: 0,
404            permissions: Vec::new(),
405            delegation: None,
406            fallback_chain: Vec::new(),
407            selected_suffix_start: 0,
408        };
409        let pipeline = cookie_agent_engine::permissions::PermissionPipeline::default();
410        let decide = |policy: &AgentSnapshot| {
411            pipeline.decide_operation(
412                policy,
413                prepared.operation(),
414                prepared.policy_labels(),
415                std::path::Path::new("/tmp"),
416            )
417        };
418        assert_eq!(decide(&policy).effect, PermissionEffect::Deny);
419        policy.permissions.push(PermissionRule {
420            action: PermissionAction::Webfetch,
421            resource: WildcardPattern::new(url.split('?').next().unwrap()).unwrap(),
422            effect: PermissionEffect::Allow,
423        });
424        assert_eq!(decide(&policy).effect, PermissionEffect::Deny);
425        policy.permissions[0].resource = WildcardPattern::new(&url).unwrap();
426        for effect in [
427            PermissionEffect::Allow,
428            PermissionEffect::Ask,
429            PermissionEffect::Deny,
430        ] {
431            policy.permissions[0].effect = effect;
432            let decision = decide(&policy);
433            assert_eq!(decision.effect, effect);
434            assert_eq!(decision.evaluations.len(), 1);
435            assert_eq!(decision.evaluations[0].trace.normalized_resource, url);
436        }
437    }
