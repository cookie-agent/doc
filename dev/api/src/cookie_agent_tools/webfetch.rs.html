30impl WebfetchArgs {
31    fn validate(&self) -> Result<(), ToolError> {
32        if !self.url.starts_with("http://") && !self.url.starts_with("https://") {
33            return Err(ToolError::InvalidUrl(
34                "URL must start with http:// or https://".into(),
35            ));
36        }
37        Ok(())
38    }

42impl ToolProvider for WebfetchTool {

47    fn tools_for_session(&self, _ctx: &SessionToolContext) -> Result<Vec<ToolSpec>, ToolError> {
48        Ok(vec![ToolSpec {

Returns HTML as plaintext unless raw is true; other text passes through. Use read_tool_result to page retained output by line, including the body.".into(),

58    fn get_permission_name(name: &str) -> Result<&'static str, ToolError> {
59        if name != "webfetch" {
60            return Err(tool_error("webfetch provider received another tool"));
61        }
62        Ok("webfetch")

65    fn get_permission_resource(
66        &self,
67        name: &str,
68        arguments: &serde_json::Value,
69    ) -> Result<(&'static str, Option<String>), ToolError> {
70        let permission = Self::get_permission_name(name)?;
71        let args: WebfetchArgs = parse_args(name, arguments.clone())?;
72        args.validate()?;
73        Ok((permission, Some(args.url)))
74    }

76    fn get_display_argument(
77        &self,
78        name: &str,
79        arguments: &serde_json::Value,
80    ) -> Result<String, ToolError> {
81        let (_, resource) = self.get_permission_resource(name, arguments)?;
82        Ok(resource.expect("webfetch has a URL resource"))
83    }

85    async fn prepare(
86        &self,
87        _ctx: ToolPreparationContext,
88        call: ToolCall,
89    ) -> Result<PreparedTool, ToolError> {
90        Self::get_permission_name(&call.name)?;
91        let args: WebfetchArgs = parse_args(&call.name, call.arguments)?;
92        args.validate()?;
93        let resource = prepared_resource(
94            PermissionAction::Webfetch,
95            "url",
96            args.url.as_bytes(),
97            args.url.as_bytes(),
98            PreparedBindingLifetime::RestartStable,
99            ApprovalResourceSource::PrimaryOperation,
100        )?;
101        let operation = prepared_operation(
102            "webfetch",
103            &args,
104            vec![(PermissionAction::Webfetch, "get")],
105            vec![resource],
106            b"webfetch:v1",
107        )?;
108        let normalized = serde_json::to_value(&args).map_err(tool_error)?;
109        let label = args.url.clone();
110        PreparedTool::new(operation, normalized, None, Box::new(args))?
111            .with_policy_labels(vec![label])
112    }
113}

115fn request_error(error: reqwest::Error) -> ToolError {
116    if error.is_timeout() {
117        ToolError::Timeout(error.to_string())
118    } else if error.is_redirect() {
119        ToolError::RedirectError(error.to_string())
120    } else if error.is_builder() {
121        ToolError::InvalidUrl(error.to_string())
122    } else {
123        ToolError::TransportError(error.to_string())
124    }
125}

127impl WebfetchArgs {
128    async fn fetch(&self) -> Result<PersistedToolResult, ToolError> {
129        let client = reqwest::Client::builder()
130            .timeout(REQUEST_TIMEOUT)
131            .build()
132            .map_err(|error| ToolError::TransportError(error.to_string()))?;
133        let mut response = client.get(&self.url).send().await.map_err(request_error)?;
134        let final_url = response.url().to_string();
135        let status_code = response.status().as_u16();
136        let content_type = response
137            .headers()
138            .get(reqwest::header::CONTENT_TYPE)
139            .and_then(|value| value.to_str().ok())
140            .unwrap_or("application/octet-stream")
141            .to_owned();
142        let mime = content_type
143            .split(';')
144            .next()
145            .unwrap_or("")
146            .trim()
147            .to_ascii_lowercase();
148        let html = matches!(mime.as_str(), "text/html" | "application/xhtml+xml");
149        if !(mime.starts_with("text/")
150            || matches!(
151                mime.as_str(),
152                "application/json"
153                    | "application/xml"
154                    | "application/javascript"
155                    | "application/x-www-form-urlencoded"
156            )
157            || mime.ends_with("+json")
158            || mime.ends_with("+xml"))
159        {
160            return Err(ToolError::UnsupportedContentType(content_type));
161        }
162        let mut body = Vec::new();
163        let mut truncated = false;
164        while let Some(chunk) = response.chunk().await.map_err(request_error)? {
165            let remaining = DOWNLOAD_CAP - body.len();
166            body.extend_from_slice(&chunk[..chunk.len().min(remaining)]);
167            if chunk.len() > remaining {
168                truncated = true;
169                break;
170            }
171        }
172        let text = if html && !self.raw {
173            html2text::from_read(body.as_slice(), 80).map_err(tool_error)?
174        } else {
175            String::from_utf8_lossy(&body).into_owned()
176        };
177        let metadata = serde_json::json!({
178            "url": self.url,
179            "final_url": final_url,
180            "status_code": status_code,
181            "content_type": content_type,
182            "truncated": truncated,
183        });
184        Ok(PersistedToolResult {
185            title: safe_title(&self.url),
186            output: format!(
187                "final_url: {final_url}\nstatus_code: {status_code}\ncontent_type: {content_type}\ntruncated: {truncated}\n\n{text}"
188            ),
189            metadata,
190            truncation: None,
191            attachments: Vec::new(),
192            additional_messages: Vec::new(),
193        })
194    }
195}

233    impl HttpFixture {
234        fn start() -> Self {
235            let listener = TcpListener::bind("127.0.0.1:0").unwrap();
236            let address = listener.local_addr().unwrap();
237            let stop = Arc::new(AtomicBool::new(false));
238            let task_stop = Arc::clone(&stop);
239            let task = thread::spawn(move || {
240                while !task_stop.load(Ordering::Acquire) {
241                    let (stream, _) = match listener.accept() {
242                        Ok(connection) => connection,
243                        Err(error) if error.kind() == io::ErrorKind::Interrupted => continue,
244                        Err(error) => return Err(error),
245                    };
246                    if task_stop.load(Ordering::Acquire) {
247                        break;
248                    }
249                    // Early disconnects are normal when webfetch rejects or caps a body.
250                    let _ = serve_connection(stream);
251                }
252                Ok(())
253            });
254            Self {
255                address,
256                stop,
257                task: Some(task),
258            }
259        }
260
261        fn url(&self) -> String {
262            format!("http://{}", self.address)
263        }
264    }

266    impl Drop for HttpFixture {
267        fn drop(&mut self) {
268            self.stop.store(true, Ordering::Release);
269            // Wake accept even if shutdown races with the loop's stop check.
270            let _ = TcpStream::connect(self.address);
271            if let Some(task) = self.task.take() {
272                task.join()
273                    .expect("join HTTP fixture")
274                    .expect("HTTP fixture accept");
275            }
276        }

279    fn serve_connection(mut stream: TcpStream) -> io::Result<()> {
280        stream.set_read_timeout(Some(Duration::from_secs(10)))?;
281        stream.set_write_timeout(Some(Duration::from_secs(10)))?;
282        let mut reader = BufReader::new((&mut stream).take(16 * 1024));
283        let mut request = String::new();
284        reader.read_line(&mut request)?;
285        loop {
286            let mut header = String::new();
287            if reader.read_line(&mut header)? == 0 {
288                return Err(io::ErrorKind::UnexpectedEof.into());
289            }
290            if header == "\r\n" {
291                break;
292            }
293        }
294        let Some(path) = request.split_whitespace().nth(1) else {
295            return Ok(());
296        };
297        let (status, headers, body) = match path {
298            "/first" => ("302 Found", "Location: /second\r\n", Vec::new()),
299            "/second" => ("301 Moved Permanently", "Location: /html\r\n", Vec::new()),
300            "/loop" => ("302 Found", "Location: /loop\r\n", Vec::new()),
301            "/html" => (
302                "200 OK",
303                "Content-Type: text/html; charset=utf-8\r\n",
304                b"<h1>Hello</h1><p>World <b>wide</b> web.</p>".to_vec(),
305            ),
306            "/cap" => (
307                "200 OK",
308                "Content-Type: text/plain\r\n",
309                vec![b'x'; DOWNLOAD_CAP + 1],
310            ),
311            "/exact" => (
312                "200 OK",
313                "Content-Type: text/plain\r\n",
314                vec![b'x'; DOWNLOAD_CAP],
315            ),
316            "/binary" => ("200 OK", "Content-Type: image/png\r\n", vec![0, 255]),
317            "/lines" => (
318                "200 OK",
319                "Content-Type: text/plain\r\n",
320                b"first\nsecond\r\nlast".to_vec(),
321            ),
322            _ => (
323                "404 Not Found",
324                "Content-Type: application/json\r\n",
325                b"{\"error\":\"missing\"}".to_vec(),
326            ),
327        };
328        write!(
329            stream,
330            "HTTP/1.1 {status}\r\n{headers}Content-Length: {}\r\nConnection: close\r\n\r\n",
331            body.len()
332        )?;
333        stream.write_all(&body)
334    }

489    async fn one_initial_url_resource_includes_query_and_defaults_to_deny() {
490        use cookie_agent_protocol::{
491            AgentDocumentSource, AgentId, AgentMode, AgentSchemaVersion, AgentSnapshot,
492            PermissionRule, Sha256Digest, WildcardPattern,
493        };

497        assert_eq!(prepared.operation().resources().len(), 1);
498        assert_eq!(prepared.policy_labels(), [Some(url.clone())]);
499        assert_eq!(
500            WebfetchTool
501                .get_display_argument("webfetch", prepared.normalized_arguments())
502                .unwrap(),
503            url
504        );
505        let mut policy = AgentSnapshot {
506            agent: AgentId::new("test").unwrap(),
507            schema: AgentSchemaVersion::current(),
508            mode: AgentMode::Primary,
509            description: "Test".into(),
510            document_source: AgentDocumentSource::Workspace,
511            document_fingerprint: Sha256Digest::of_bytes(b"test"),
512            composed_prompt: "test".into(),
513            prompt_fingerprint: Sha256Digest::of_bytes(b"test"),
514            max_output_tokens: 0,
515            permissions: Vec::new(),
516            delegation: None,
517            fallback_chain: Vec::new(),
518            selected_suffix_start: 0,
519        };
520        let pipeline = cookie_agent_engine::permissions::PermissionPipeline::default();
521        let decide = |policy: &AgentSnapshot| {
522            pipeline.decide_operation(
523                policy,
524                prepared.operation(),
525                prepared.policy_labels(),
526                std::path::Path::new("/tmp"),
527            )
528        };
529        assert_eq!(decide(&policy).effect, PermissionEffect::Deny);
530        policy.permissions.push(PermissionRule {
531            action: PermissionAction::Webfetch,
532            resource: WildcardPattern::new(url.split('?').next().unwrap()).unwrap(),
533            effect: PermissionEffect::Allow,
534        });
535        assert_eq!(decide(&policy).effect, PermissionEffect::Deny);
536        policy.permissions[0].resource = WildcardPattern::new(&url).unwrap();
537        for effect in [
538            PermissionEffect::Allow,
539            PermissionEffect::Ask,
540            PermissionEffect::Deny,
541        ] {
542            policy.permissions[0].effect = effect;
543            let decision = decide(&policy);
544            assert_eq!(decision.effect, effect);
545            assert_eq!(decision.evaluations.len(), 1);
546            assert_eq!(decision.evaluations[0].trace.normalized_resource, url);
547        }
