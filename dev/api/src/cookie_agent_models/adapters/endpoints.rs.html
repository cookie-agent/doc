8#[derive(Clone, Copy, Debug, Eq, PartialEq)]
9pub enum BaseUrlOverridePolicy {
10    ManagedHttps,
11    Forbidden,
12    CustomHttpsOrReviewedLoopback { path: &'static str },
13}
14
15#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
16pub enum EndpointBuildError {
17    #[error("authored_base_url_forbidden")]
18    AuthoredOverrideForbidden,
19    #[error("invalid_endpoint_setup")]
20    Setup,
21    #[error("endpoint_override_policy_violation")]
22    Policy,
23}

25#[must_use]
26pub const fn managed_base_url_policy(policy: EndpointPolicy) -> BaseUrlOverridePolicy {
27    match policy {
28        EndpointPolicy::DefaultWithAuthoredHttpsOverride { .. } => {
29            BaseUrlOverridePolicy::ManagedHttps
30        }
31        EndpointPolicy::VertexPublisher
32        | EndpointPolicy::BedrockRegional
33        | EndpointPolicy::AzureOpenai => BaseUrlOverridePolicy::Forbidden,
34    }
35}

37#[must_use]
38pub const fn custom_endpoint_policy(family: OvenAdapterFamily) -> BaseUrlOverridePolicy {
39    BaseUrlOverridePolicy::CustomHttpsOrReviewedLoopback {
40        path: match family {
41            OvenAdapterFamily::Anthropic
42            | OvenAdapterFamily::AnthropicCompatible
43            | OvenAdapterFamily::OpenaiChat
44            | OvenAdapterFamily::OpenaiResponses
45            | OvenAdapterFamily::OpenaiCompatible
46            | OvenAdapterFamily::GoogleVertexGemini => "/v1",
47            OvenAdapterFamily::GoogleGemini => "/v1beta",
48            OvenAdapterFamily::AwsBedrockConverse
49            | OvenAdapterFamily::AzureOpenaiChat
50            | OvenAdapterFamily::AzureOpenaiResponses => "/",
51            OvenAdapterFamily::CohereV2Chat => "/v2",
52        },
53    }
54}

56pub fn validate_managed_base_url(
57    policy: EndpointPolicy,
58    authored: Option<&EndpointUrl>,
59) -> Result<(), EndpointBuildError> {
60    let Some(authored) = authored else {
61        return Ok(());
62    };
63    match managed_base_url_policy(policy) {
64        BaseUrlOverridePolicy::ManagedHttps => {
65            let parsed = Url::parse(authored.as_str()).map_err(|_| EndpointBuildError::Policy)?;
66            if parsed.scheme() == "https" || parsed.scheme() == "http" && loopback_host(&parsed) {
67                Ok(())
68            } else {
69                Err(EndpointBuildError::Policy)
70            }
71        }
72        BaseUrlOverridePolicy::Forbidden => Err(EndpointBuildError::AuthoredOverrideForbidden),
73        BaseUrlOverridePolicy::CustomHttpsOrReviewedLoopback { .. } => unreachable!(),
74    }
75}

77/// Whether `url` is an `http`/`https` URL on `localhost`, `127.0.0.1`, or
78/// `::1`: a local server, such as LM Studio, Ollama, or QVAC, that may be
79/// reached over plain HTTP and without credentials.
80#[must_use]
81pub fn is_loopback_url(url: &str) -> bool {
82    Url::parse(url)
83        .is_ok_and(|parsed| matches!(parsed.scheme(), "http" | "https") && loopback_host(&parsed))
84}
85
86fn loopback_host(url: &Url) -> bool {
87    match url.host() {
88        Some(url::Host::Domain(domain)) => domain == "localhost",
89        Some(url::Host::Ipv4(address)) => address == std::net::Ipv4Addr::LOCALHOST,
90        Some(url::Host::Ipv6(address)) => address == std::net::Ipv6Addr::LOCALHOST,
91        None => false,
92    }
93}

95pub fn validate_custom_endpoint(
96    family: OvenAdapterFamily,
97    endpoint: &EndpointUrl,
98) -> Result<(), EndpointBuildError> {
99    let parsed = Url::parse(endpoint.as_str()).map_err(|_| EndpointBuildError::Policy)?;
100    if parsed.scheme() == "https" {
101        return Ok(());
102    }
103    let BaseUrlOverridePolicy::CustomHttpsOrReviewedLoopback { path } =
104        custom_endpoint_policy(family)
105    else {
106        unreachable!()
107    };
108    if parsed.scheme() == "http"
109        && loopback_host(&parsed)
110        && parsed.port().is_some()
111        && parsed.path() == path
112    {
113        Ok(())
114    } else {
115        Err(EndpointBuildError::Policy)
116    }
117}

119pub fn build_endpoint(
120    policy: EndpointPolicy,
121    authored: Option<&EndpointUrl>,
122    setup: &ValidatedSetup,
123) -> Result<String, EndpointBuildError> {
124    validate_managed_base_url(policy, authored)?;
125    match policy {
126        EndpointPolicy::DefaultWithAuthoredHttpsOverride { default } => Ok(authored
127            .map(EndpointUrl::as_str)
128            .unwrap_or(default)
129            .trim_end_matches('/')
130            .to_owned()),
131        EndpointPolicy::VertexPublisher => {
132            if authored.is_some() {
133                return Err(EndpointBuildError::AuthoredOverrideForbidden);
134            }
135            let project = field(setup, "project")?;
136            let location = field(setup, "location")?;
137            let resource = field(setup, "resource")?;
138            Ok(format!(
139                "https://{location}-aiplatform.googleapis.com/v1/projects/{project}/locations/{location}/{resource}"
140            ))
141        }
142        EndpointPolicy::BedrockRegional => {
143            if authored.is_some() {
144                return Err(EndpointBuildError::AuthoredOverrideForbidden);
145            }
146            Ok(format!(
147                "https://bedrock-runtime.{}.amazonaws.com",
148                field(setup, "region")?
149            ))
150        }
151        EndpointPolicy::AzureOpenai => {
152            if authored.is_some() {
153                return Err(EndpointBuildError::AuthoredOverrideForbidden);
154            }
155            Ok(format!(
156                "https://{}.openai.azure.com",
157                field(setup, "resource_name")?
158            ))
159        }
160    }
161}

163fn field<'a>(setup: &'a ValidatedSetup, id: &str) -> Result<&'a str, EndpointBuildError> {
164    setup
165        .values
166        .get(id)
167        .map(String::as_str)
168        .ok_or(EndpointBuildError::Setup)
