4#[derive(Clone, Copy, Debug, Eq, PartialEq)]
5pub enum BaseUrlOverridePolicy {
6    ManagedHttps,
7    Forbidden,
8    CustomHttpsOrReviewedLoopback { path: &'static str },
9}

11#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
12pub enum EndpointBuildError {
13    #[error("authored_base_url_forbidden")]
14    AuthoredOverrideForbidden,
15    #[error("endpoint_override_policy_violation")]
16    Policy,
17}

19#[must_use]
20pub const fn managed_base_url_policy(policy: EndpointPolicy) -> BaseUrlOverridePolicy {
21    match policy {
22        EndpointPolicy::DefaultWithAuthoredHttpsOverride { .. } => {
23            BaseUrlOverridePolicy::ManagedHttps
24        }
25        EndpointPolicy::VertexPublisher
26        | EndpointPolicy::BedrockRegional
27        | EndpointPolicy::AzureOpenai => BaseUrlOverridePolicy::Forbidden,
28    }
29}

31#[must_use]
32pub const fn custom_endpoint_policy(family: OvenAdapterFamily) -> BaseUrlOverridePolicy {
33    BaseUrlOverridePolicy::CustomHttpsOrReviewedLoopback {
34        path: match family {
35            OvenAdapterFamily::Anthropic
36            | OvenAdapterFamily::AnthropicCompatible
37            | OvenAdapterFamily::OpenaiChat
38            | OvenAdapterFamily::OpenaiResponses
39            | OvenAdapterFamily::OpenaiCompatible
40            | OvenAdapterFamily::GoogleVertexGemini => "/v1",
41            OvenAdapterFamily::GoogleGemini => "/v1beta",
42            OvenAdapterFamily::AwsBedrockConverse
43            | OvenAdapterFamily::AzureOpenaiChat
44            | OvenAdapterFamily::AzureOpenaiResponses => "/",
45            OvenAdapterFamily::CohereV2Chat => "/v2",
46        },
47    }
48}

50pub fn validate_managed_base_url(
51    policy: EndpointPolicy,
52    authored: Option<&EndpointUrl>,
53) -> Result<(), EndpointBuildError> {
54    let Some(authored) = authored else {
55        return Ok(());
56    };
57    match managed_base_url_policy(policy) {
58        BaseUrlOverridePolicy::ManagedHttps => {
59            let parsed = Url::parse(authored.as_str()).map_err(|_| EndpointBuildError::Policy)?;
60            if parsed.scheme() == "https" || parsed.scheme() == "http" && loopback_host(&parsed) {
61                Ok(())
62            } else {
63                Err(EndpointBuildError::Policy)
64            }
65        }
66        BaseUrlOverridePolicy::Forbidden => Err(EndpointBuildError::AuthoredOverrideForbidden),
67        BaseUrlOverridePolicy::CustomHttpsOrReviewedLoopback { .. } => unreachable!(),
68    }
69}

71/// Whether `url` is an `http`/`https` URL on `localhost`, `127.0.0.1`, or
72/// `::1`: a local server, such as LM Studio, Ollama, or QVAC, that may be
73/// reached over plain HTTP and without credentials.
74#[must_use]
75pub fn is_loopback_url(url: &str) -> bool {
76    Url::parse(url)
77        .is_ok_and(|parsed| matches!(parsed.scheme(), "http" | "https") && loopback_host(&parsed))
78}
79
80fn loopback_host(url: &Url) -> bool {
81    match url.host() {
82        Some(url::Host::Domain(domain)) => domain == "localhost",
83        Some(url::Host::Ipv4(address)) => address == std::net::Ipv4Addr::LOCALHOST,
84        Some(url::Host::Ipv6(address)) => address == std::net::Ipv6Addr::LOCALHOST,
85        None => false,
86    }
87}

89pub fn validate_custom_endpoint(
90    family: OvenAdapterFamily,
91    endpoint: &EndpointUrl,
92) -> Result<(), EndpointBuildError> {
93    let parsed = Url::parse(endpoint.as_str()).map_err(|_| EndpointBuildError::Policy)?;
94    if parsed.scheme() == "https" {
95        return Ok(());
96    }
97    let BaseUrlOverridePolicy::CustomHttpsOrReviewedLoopback { path } =
98        custom_endpoint_policy(family)
99    else {
100        unreachable!()
101    };
102    if parsed.scheme() == "http"
103        && loopback_host(&parsed)
104        && parsed.port().is_some()
105        && parsed.path() == path
106    {
107        Ok(())
108    } else {
109        Err(EndpointBuildError::Policy)
110    }
