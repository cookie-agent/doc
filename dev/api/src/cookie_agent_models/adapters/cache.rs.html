70/// Whether the Bedrock model `wire_model_id` takes a one-hour cache TTL.
71/// Bedrock lists the TTLs each model supports
72/// (docs.aws.amazon.com/bedrock/latest/userguide/prompt-caching.html): the
73/// Claude models from Claude Sonnet 4.5, Haiku 4.5, and Opus 4.5 on take
74/// `5m` and `1h`, while older Claude models and other families take only
75/// `5m` and may reject a `ttl` with a `ValidationException`. Bedrock
76/// publishes no capability for it, so it is read from the model ID: a
77/// Claude ID of version 4.5 or later in any region or profile prefix
78/// (`us.anthropic.claude-sonnet-4-5-20250929-v1:0`,
79/// `anthropic.claude-opus-5`). Any other ID, an application inference
80/// profile ARN among them, counts as unsupported.
81#[must_use]
82pub fn bedrock_supports_one_hour_cache(wire_model_id: &str) -> bool {
83    let id = wire_model_id.to_ascii_lowercase();
84    let Some((_, rest)) = id.split_once("anthropic.claude-") else {
85        return false;
86    };
87    let mut parts = rest.split(['-', '.', ':']);
88    // `claude-3-7-sonnet-...` names the version before the family.
89    let Some(family) = parts
90        .next()
91        .filter(|family| !family.starts_with(|c: char| c.is_ascii_digit()))
92    else {
93        return false;
94    };
95    if family.is_empty() {
96        return false;
97    }
98    let Some(major) = parts.next().and_then(|part| part.parse::<u32>().ok()) else {
99        return false;
100    };
101    // A date (`20250929`) or a version suffix (`v1`) is not a minor version.
102    let minor = parts
103        .next()
104        .filter(|part| part.len() <= 2)
105        .and_then(|part| part.parse::<u32>().ok())
106        .unwrap_or(0);
107    (major, minor) >= (4, 5)
108}

119            "global.anthropic.claude-opus-4-5-20251101-v1:0",

121            "eu.anthropic.claude-sonnet-4-6",
122            "anthropic.claude-opus-5",
123            "anthropic.claude-sonnet-5-5",
124            "anthropic.claude-fable-5-1",
125            "Anthropic.Claude-Mythos-5",
126        ] {
127            assert!(bedrock_supports_one_hour_cache(supported), "{supported}");
128        }
129        for unsupported in [
130            "anthropic.claude-3-7-sonnet-20250219-v1:0",
131            "us.anthropic.claude-3-5-sonnet-20241022-v2:0",
132            "anthropic.claude-sonnet-4-20250514-v1:0",
133            "anthropic.claude-opus-4-1-20250805-v1:0",
134            "amazon.nova-pro-v1:0",
135            "openai.gpt-5.6-sol",
136            "arn:aws:bedrock:us-east-1:123456789012:application-inference-profile/abc123",
137            "claude-sonnet-4-5",
138            "anthropic.claude-",
139        ] {
140            assert!(
141                !bedrock_supports_one_hour_cache(unsupported),
