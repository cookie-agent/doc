94    pub(crate) fn selected(
95        &self,
96        options: &ProviderOptions,
97        model_id: Option<&crate::authoring::WireModelId>,
98    ) -> Result<Self, DynamicCompileError> {
99        let mut selected = self.clone();
100        if let Some(model_id) = model_id {
101            selected.wire_model_id = model_id.clone();
102        }
103        let compatible = self.adapter_id.starts_with("oven.openai-compatible.");
104        let family = if compatible {
105            OvenAdapterFamily::OpenaiCompatible
106        } else {
107            self.adapter
108        };
109        selected.adapter = if options.request_endpoint.is_none() {
110            self.adapter
111        } else {
112            family.with_endpoint(options.request_endpoint)?
113        };
114        selected.resolved_shape = if matches!(
115            selected.adapter,
116            OvenAdapterFamily::OpenaiResponses | OvenAdapterFamily::AzureOpenaiResponses
117        ) {
118            "responses"
119        } else {
120            "chat"
121        }
122        .into();
123        selected.adapter_id = if compatible {
124            self.adapter_id.replacen(
125                if self.adapter == OvenAdapterFamily::OpenaiResponses {
126                    ".responses"
127                } else {
128                    ".chat"
129                },
130                if selected.adapter == OvenAdapterFamily::OpenaiResponses {
131                    ".responses"
132                } else {
133                    ".chat"
134                },
135                1,
136            )
137        } else if selected.adapter == self.adapter {
138            self.adapter_id.clone()
139        } else {
140            selected.adapter.protocol_recipe().into()
141        };
142        selected.options = options.clone();
143        selected.capabilities.native_replay = self.replay_declaration.unwrap_or_else(|| {
144            automatic_replay(
145                selected.adapter,
146                selected.capabilities.reasoning,
147                &selected.wire_model_id,
148            )
149        });
150        Ok(selected)
151    }
152
153    fn validate_settings(&self) -> Result<(), DynamicCompileError> {
154        let compatible_accounts = self.adapter_id.starts_with("oven.openai-compatible.")
155            || self.auth.method == "no-auth-v1"
156                && self.adapter == OvenAdapterFamily::OpenaiResponses;
157        if compatible_accounts
158            && (self.options.organization.is_some() || self.options.project.is_some())
159        {
160            return Err(DynamicCompileError::EndpointSelection("organization/project adaptor_options require official OpenAI authentication, not compatible or unauthenticated Responses".into()));
161        }
162        if self.options.store == Some(true) {
163            return Err(DynamicCompileError::EndpointSelection("adaptor_options.store = true is unsupported: the integrated Responses encoder uses stateless store = false requests".into()));
164        }
165        let validate = |selected: &Self,
166                        defaults: &RequestDefaults,
167                        reasoning: Option<&crate::ReasoningBehavior>| {
168            validate_capability_shape(&selected.capabilities)
169                && validate_capability_ceiling(selected.adapter, &selected.capabilities).is_ok()
170                && validate_defaults(defaults, &selected.capabilities)
171                && validate_custom_options(&selected.options, selected.adapter)
172                && (reasoning.is_none() || selected.capabilities.reasoning)
173                && reasoning_supported(reasoning, selected.adapter, selected.thinking_toggle)
174        };
175        if !validate(self, &self.defaults, None) {
176            return Err(DynamicCompileError::CustomModel);
177        }
178        for variant in self.variants.values() {
179            let selected = self.selected(&variant.options, variant.model_id.as_ref())?;
180            if compatible_accounts
181                && (selected.options.organization.is_some() || selected.options.project.is_some())
182            {
183                return Err(DynamicCompileError::EndpointSelection(format!(
184                    "variant `{}`: organization/project adaptor_options require official OpenAI authentication",
185                    variant.id
186                )));
187            }
188            if selected.options.store == Some(true) {
189                return Err(DynamicCompileError::EndpointSelection(format!(
190                    "variant `{}`: adaptor_options.store = true is unsupported by the stateless Responses encoder",
191                    variant.id
192                )));
193            }
194            if !validate(&selected, &variant.defaults, variant.reasoning.as_ref()) {
195                return Err(DynamicCompileError::Variant);
196            }
197        }
198        Ok(())
199    }

202/// Claude thinking on Bedrock must replay its signed reasoning like the
203/// Anthropic families do; every other family keeps the adapter default.
204fn automatic_replay(
205    adapter: OvenAdapterFamily,
206    reasoning: bool,
207    wire_model_id: &crate::authoring::WireModelId,
208) -> crate::ReplayCapability {
209    if reasoning && bedrock_anthropic_thinking(adapter, wire_model_id.as_str()) {
210        crate::ReplayCapability::Required
211    } else {
212        adapter.automatic_replay(reasoning)
213    }

216pub(crate) fn provider_wire_model_id(
217    id: &ProviderModelId,
218    adapter: OvenAdapterFamily,
219    setup: Option<&ValidatedSetup>,
220) -> Result<crate::authoring::WireModelId, DynamicCompileError> {
221    let wire = if matches!(
222        adapter,
223        OvenAdapterFamily::AzureOpenaiChat | OvenAdapterFamily::AzureOpenaiResponses
224    ) {
225        setup
226            .and_then(|setup| setup.values.get("deployment"))
227            .map_or(id.as_str(), String::as_str)
228    } else {
229        id.as_str()
230    };
231    crate::authoring::WireModelId::new(wire)
232        .map_err(|reason| DynamicCompileError::EndpointSelection(reason.into()))
233}

324    pub fn compile_managed_with_headers(
325        &self,
326        catalog_revision: &str,
327        record: &CatalogProviderRecord,
328        authored: Option<&ModelsDevProvider>,
329        global_headers: &BTreeMap<HeaderName, SafeStaticHeaderValue>,
330    ) -> Result<CompiledDynamicProvider, DynamicCompileError> {
331        let family = self
332            .registry
333            .classify(record)
334            .ok_or(DynamicCompileError::UnsupportedProvider)?;
335        validate_managed_cache(record, authored, family)?;
336        if let Some(authored) = authored
337            && let Some(base_url) = authored.base_url.as_ref()
338        {
339            validate_managed_base_url(
340                crate::recipes::EndpointPolicy::DefaultWithAuthoredHttpsOverride {
341                    default: "https://invalid.example",
342                },
343                Some(base_url),
344            )
345            .map_err(|_| DynamicCompileError::Endpoint)?;
346        }
347        if authored.is_some_and(|value| {
348            value
349                .base_url
350                .as_ref()
351                .is_some_and(|url| !crate::adapters::is_loopback_url(url.as_str()))
352                && value.api_key.is_none()
353                && value.auth_override.is_none()
354        }) {
355            return Err(DynamicCompileError::BaseUrlWithoutAuth);
356        }
357        if let Some(authored) = authored {
358            for id in authored.model_overrides.keys() {
359                if !record.models.contains_key(id) {
360                    return Err(DynamicCompileError::UnknownModelOverride);
361                }
362            }
363        }
364        let mut models = BTreeMap::new();
365        let mut unsupported_models = Vec::new();
366        for (table_id, entry) in &record.models {
367            let Some(model) = entry.record.as_ref() else {
368                unsupported_models.push(UnsupportedModel {
369                    id: table_id.clone(),
370                    kind: UnsupportedModelKind::Quarantined,
371                    reason: entry
372                        .quarantine
373                        .as_ref()
374                        .map_or(
375                            "invalid_catalog_model_record",
376                            crate::catalog::CatalogQuarantineReason::code,
377                        )
378                        .to_owned(),
379                });
380                continue;
381            };
382            if model.status == crate::catalog::CatalogModelStatus::Deprecated
383                || !model.modalities.output.iter().any(|value| value == "text")
384            {
385                continue;
386            }
387            let override_ = authored.and_then(|value| value.model_overrides.get(table_id));
388            if override_.and_then(|value| value.enabled) == Some(false) {
389                continue;
390            }
391            if !model.tool_call {
392                unsupported_models.push(UnsupportedModel {
393                    id: table_id.clone(),
394                    kind: UnsupportedModelKind::Unsupported,
395                    reason: NO_TOOL_CALLING_REASON.to_owned(),
396                });
397                continue;
398            }
399            let mut resolved = match resolve_model(record, model, None, None) {
400                Ok(resolved) => resolved,
401                Err(error) => {
402                    unsupported_models.push(UnsupportedModel {
403                        id: table_id.clone(),
404                        kind: UnsupportedModelKind::Unsupported,
405                        reason: error.to_string(),
406                    });
407                    continue;
408                }
409            };
410            if let Some(base_url) = authored.and_then(|value| value.base_url.as_ref()) {
411                resolved.endpoint_template = Some(authored_model_endpoint(
412                    base_url.as_str(),
413                    record.api.as_deref(),
414                    model
415                        .provider
416                        .as_ref()
417                        .and_then(|value| value.api.as_deref()),
418                ));
419            }
420            match self.compile_managed_model(
421                catalog_revision,
422                &record.id,
423                model,
424                family,
425                &resolved,
426                authored,
427                override_,
428                global_headers,
429            ) {
430                Ok(compiled) => {
431                    models.insert(table_id.clone(), compiled);
432                }
433                Err(ModelLocalError::Unsupported(reason)) => {
434                    unsupported_models.push(UnsupportedModel {
435                        id: table_id.clone(),
436                        kind: UnsupportedModelKind::Unsupported,
437                        reason,
438                    });
439                }
440                Err(ModelLocalError::Provider(error)) => return Err(error),
441            }
442        }
443        unsupported_models.sort_by(|left, right| left.id.cmp(&right.id));
444        let provider_fingerprint = fingerprint(
445            "cookie-agent/dynamic-provider/v1",
446            &(
447                self.registry.revision(),
448                COMPILER_VERSION,
449                catalog_revision,
450                &record.id,
451                models
452                    .iter()
453                    .map(|(id, model)| (id, &model.behavior_fingerprint))
454                    .collect::<Vec<_>>(),
455                &unsupported_models,
456            ),
457        );
458        Ok(CompiledDynamicProvider {
459            id: record.id.clone(),
460            models,
461            unsupported_models,
462            fingerprint: provider_fingerprint,
463        })
464    }

466    #[allow(clippy::too_many_arguments)]
467    fn compile_managed_model(
468        &self,
469        catalog_revision: &str,
470        provider_id: &ProviderId,
471        model: &CatalogModelRecord,
472        provider_family: &'static FamilyRecipe,
473        resolved: &crate::recipes::ResolvedFamilyModel,
474        authored: Option<&ModelsDevProvider>,
475        override_: Option<&ManagedModelOverride>,
476        global_headers: &BTreeMap<HeaderName, SafeStaticHeaderValue>,
477    ) -> Result<CompiledDynamicModel, ModelLocalError> {
478        let options =
479            override_.map_or_else(ProviderOptions::default, |value| value.options.resolve());
480        let endpoint_family = if resolved.recipe.family == FamilyKind::OpenAiCompatibleChat {
481            OvenAdapterFamily::OpenaiCompatible
482        } else {
483            resolved.adapter
484        };
485        let adapter = if options.request_endpoint.is_none() {
486            resolved.adapter
487        } else {
488            endpoint_family
489                .with_endpoint(options.request_endpoint)
490                .map_err(ModelLocalError::Provider)?
491        };
492        let adapter_id = match adapter {
493            _ if resolved.recipe.family == FamilyKind::OpenAiCompatibleChat => {
494                format!(
495                    "oven.openai-compatible.{}.{}",
496                    if adapter == OvenAdapterFamily::OpenaiResponses {
497                        "responses"
498                    } else {
499                        "chat"
500                    },
501                    provider_id.as_str()
502                )
503            }
504            OvenAdapterFamily::AnthropicCompatible => {
505                format!(
506                    "oven.anthropic-compatible.messages.{}",
507                    provider_id.as_str()
508                )
509            }
510            _ => adapter.protocol_recipe().to_owned(),
511        };
512        let mut capabilities = capabilities_from_catalog(model, adapter).map_err(|_| {
513            ModelLocalError::Unsupported("unsupported_model_capabilities".to_owned())
514        })?;
515        apply_compaction_config(
516            &mut capabilities,
517            adapter,
518            provider_id,
519            override_.map_or(crate::NativeCompactionConfig::Unsupported, |value| {
520                value.compaction
521            }),
522        )
523        .map_err(ModelLocalError::Provider)?;
524        if !validate_capability_shape(&capabilities)
525            || validate_capability_ceiling(adapter, &capabilities).is_err()
526        {
527            return Err(if override_.is_some() {
528                ModelLocalError::Provider(DynamicCompileError::CustomModel)
529            } else {
530                ModelLocalError::Unsupported("unsupported_model_capabilities".to_owned())
531            });
532        }
533        let (setup, endpoint) = resolved_managed_setup_and_endpoint(
534            provider_family,
535            resolved.recipe.family,
536            resolved.endpoint_template.as_deref(),
537            authored,
538        )?;
539        let wire_model_id = override_
540            .and_then(|value| value.model_id.clone())
541            .map_or_else(
542                || provider_wire_model_id(&model.id, adapter, setup.as_ref()),
543                Ok,
544            )
545            .map_err(ModelLocalError::Provider)?;
546        capabilities.native_replay =
547            automatic_replay(adapter, capabilities.reasoning, &wire_model_id);
548        let required_auth_method = match adapter {
549            OvenAdapterFamily::AwsBedrockConverse => Some("aws-sigv4-credentials-v1"),
550            OvenAdapterFamily::OpenaiResponses if resolved.recipe.family == FamilyKind::Bedrock => {
551                Some("bearer-api-key-v1")
552            }
553            _ => None,
554        };
555        let auth = managed_auth(
556            provider_family,
557            resolved.recipe,
558            required_auth_method,
559            authored,
560        )?;
561        // Catalog models carry no output default: requests send the model's output limit
562        // unless an override or agent cap sets one.
563        let mut defaults = RequestDefaults::default();
564        if let Some(override_) = override_ {
565            override_.defaults.apply(&mut defaults);
566        }
567        if !validate_defaults(&defaults, &capabilities) {
568            return Err(if override_.is_some() {
569                ModelLocalError::Provider(DynamicCompileError::CustomModel)
570            } else {
571                ModelLocalError::Unsupported("unsupported_model_capabilities".to_owned())
572            });
573        }
574        let thinking_toggle = (adapter == OvenAdapterFamily::OpenaiCompatible)
575            .then(|| compatible_thinking_toggle(provider_id.as_str()))
576            .flatten();
577        let (mut variants, mut variant_order, default_variant) = managed_variants(
578            &model.reasoning_options,
579            model.limits.output,
580            |reasoning| reasoning_supported(Some(reasoning), adapter, thinking_toggle),
581            override_,
582            &defaults,
583            &options,
584        )
585        .map_err(|_| {
586            if override_.is_some() {
587                ModelLocalError::Provider(DynamicCompileError::Variant)
588            } else {
589                ModelLocalError::Unsupported("unsupported_protocol_feature".to_owned())
590            }
591        })?;
592        // Kimi Code documents `reasoning_effort: "none"` as its thinking switch,
593        // so its generated `off` is that effort level.
594        if thinking_toggle == Some(CompatibleThinkingToggle::ReasoningEffortNone) {
595            for variant in variants.values_mut() {
596                if variant.origin != CompiledVariantOrigin::Authored
597                    && variant.reasoning
598                        == Some(crate::ReasoningBehavior::Toggle { enabled: false })
599                {
600                    variant.reasoning = Some(crate::ReasoningBehavior::Effort {
601                        value: crate::ReasoningEffort::None,
602                    });
603                }
604            }
605        }
606        // Catalog reasoning controls can exceed the selected wire protocol.
607        // Keep supported generated choices without losing the whole model.
608        // Explicit model overrides remain strict and must disable or replace
609        // incompatible variants themselves.
610        if override_.is_none() {
611            variants.retain(|_, variant| {
612                reasoning_supported(variant.reasoning.as_ref(), adapter, thinking_toggle)
613            });
614            variant_order.retain(|id| variants.contains_key(id));
615        }
616        if variants
617            .values()
618            .any(|variant| !validate_defaults(&variant.defaults, &capabilities))
619        {
620            return Err(ModelLocalError::Provider(DynamicCompileError::Variant));
621        }
622        let headers = merge_headers([
623            (global_headers, "global".to_owned()),
624            (
625                authored.map_or(&EMPTY_HEADERS, |value| &value.headers),
626                format!("provider `{provider_id}`"),
627            ),
628            (
629                override_.map_or(&EMPTY_HEADERS, |value| &value.headers),
630                format!("provider `{provider_id}` model `{}`", model.id),
631            ),
632        ])
633        .map_err(ModelLocalError::Provider)?;
634        for variant in variants.values_mut() {
635            variant.headers = merge_headers([
636                (
637                    &headers,
638                    format!("provider `{provider_id}` model `{}`", model.id),
639                ),
640                (
641                    &variant.headers,
642                    format!(
643                        "provider `{provider_id}` model `{}` variant `{}`",
644                        model.id, variant.id
645                    ),
646                ),
647            ])
648            .map_err(ModelLocalError::Provider)?;
649        }
650        let display_name = override_
651            .and_then(|value| value.display_name.clone())
652            .unwrap_or_else(|| model.name.clone());
653        let status = if setup.is_none() {
654            CompiledModelStatus::SetupUnavailable
655        } else if auth.source == AuthSourceCategory::Unavailable {
656            CompiledModelStatus::CredentialsUnavailable
657        } else {
658            CompiledModelStatus::Available
659        };
660        let behavior_fingerprint = fingerprint(
661            "cookie-agent/dynamic-model-behavior/v1",
662            &(
663                (
664                    self.registry.revision(),
665                    COMPILER_VERSION,
666                    catalog_revision,
667                    provider_id,
668                    &model.id,
669                    &wire_model_id,
670                    resolved.recipe.family.id(),
671                    &adapter_id,
672                    adapter,
673                ),
674                &endpoint,
675                &setup,
676                &auth,
677                &capabilities,
678                &defaults,
679                &options,
680                &headers,
681                &variants,
682                &variant_order,
683                &default_variant,
684                "managed_catalog",
685            ),
686        );
687        let compiled = CompiledDynamicModel {
688            custom: false,
689            id: model.id.clone(),

746    pub fn compile_custom_with_headers(
747        &self,
748        provider_id: &ProviderId,
749        provider: &CustomProvider,
750        global_headers: &BTreeMap<HeaderName, SafeStaticHeaderValue>,
751    ) -> Result<CompiledDynamicProvider, DynamicCompileError> {
752        let adapter = OvenAdapterFamily::parse(provider.adaptor.as_str())
753            .ok_or(DynamicCompileError::UnsupportedAdapter)?;
754        crate::authoring::validate_provider_cache(provider.cache.as_ref(), adapter.id())
755            .map_err(DynamicCompileError::Cache)?;
756        validate_custom_endpoint(adapter, &provider.endpoint)
757            .map_err(|_| DynamicCompileError::Endpoint)?;
758        let setup_recipe = custom_setup_recipe(adapter);
759        let setup = validate_setup(setup_recipe, &provider.setup)
760            .map_err(|_| DynamicCompileError::Setup)?;
761        let auth_method = validate_auth_definition(&provider.auth, adapter.allowed_auth_methods())
762            .map_err(|_| DynamicCompileError::Auth)?;
763        let auth = custom_auth_shape(&provider.auth, auth_method);
764        let mut models = BTreeMap::new();
765        let mut unsupported_models = Vec::new();
766        for (id, model) in &provider.models {
767            let options = model.options.resolve();
768            let resolved_adapter = adapter.with_endpoint(options.request_endpoint)?;
769            let wire = wire_adapter_for_custom(resolved_adapter);
770            let mut capabilities = model.capabilities.resolve(resolved_adapter);
771            let defaults = model.defaults.resolve();
772            let wire_model_id = model.model_id.clone().map_or_else(
773                || provider_wire_model_id(id, resolved_adapter, Some(&setup)),
774                Ok,
775            )?;
776            if model.capabilities.native_replay.is_none() {
777                capabilities.native_replay =
778                    automatic_replay(resolved_adapter, capabilities.reasoning, &wire_model_id);
779            }
780            if !validate_capability_shape(&capabilities)
781                || validate_capability_ceiling(resolved_adapter, &capabilities).is_err()
782                || !validate_defaults(&defaults, &capabilities)
783                || !validate_custom_options(&options, resolved_adapter)
784            {
785                return Err(DynamicCompileError::CustomModel);
786            }
787            let (mut variants, variant_order, default_variant) = custom_variants(
788                &model.variants,
789                model.default_variant.as_ref(),
790                &defaults,
791                &options,
792            )
793            .map_err(|_| DynamicCompileError::Variant)?;
794            let headers = merge_headers([
795                (global_headers, "global".to_owned()),
796                (&provider.headers, format!("provider `{provider_id}`")),
797                (
798                    &model.headers,
799                    format!("provider `{provider_id}` model `{id}`"),
800                ),
801            ])?;
802            for variant in variants.values_mut() {
803                variant.headers = merge_headers([
804                    (&headers, format!("provider `{provider_id}` model `{id}`")),
805                    (
806                        &variant.headers,
807                        format!(
808                            "provider `{provider_id}` model `{id}` variant `{}`",
809                            variant.id
810                        ),
811                    ),
812                ])?;
813            }
814            if !model.enabled {
815                continue;
816            }
817            if !capabilities.tool_calling {
818                unsupported_models.push(UnsupportedModel {
819                    id: id.clone(),
820                    kind: UnsupportedModelKind::Unsupported,
821                    reason: NO_TOOL_CALLING_REASON.to_owned(),
822                });
823                continue;
824            }
825            let endpoint = provider.endpoint.as_str().trim_end_matches('/').to_owned();
826            let safe_headers = headers
827                .iter()
828                .map(|(name, value)| (name.as_str(), value.as_str()))
829                .collect::<Vec<_>>();
830            let behavior_fingerprint = fingerprint(
831                "cookie-agent/custom-model-behavior/v1",
832                &(
833                    self.registry.revision(),
834                    COMPILER_VERSION,
835                    provider_id,
836                    id,
837                    wire.adapter_id,
838                    adapter,
839                    &endpoint,
840                    &setup,
841                    &auth,
842                    &safe_headers,
843                    &capabilities,
844                    &model.defaults,
845                    &options,
846                    (&variants, &variant_order, &default_variant, &wire_model_id),
847                    "custom_authored",
848                ),
849            );
850            models.insert(
851                id.clone(),
852                CompiledDynamicModel {
853                    custom: true,
854                    id: id.clone(),
855                    wire_model_id,
856                    display_name: model.display_name.clone(),
857                    family_id: "custom".into(),
858                    effective_npm: "custom".into(),
859                    adapter_id: if adapter == OvenAdapterFamily::OpenaiCompatible {
860                        format!(
861                            "oven.openai-compatible.{}.{}",
862                            if resolved_adapter == OvenAdapterFamily::OpenaiResponses {
863                                "responses"
864                            } else {
865                                "chat"
866                            },
867                            provider_id
868                        )
869                    } else {
870                        wire.adapter_id.into()
871                    },
872                    resolved_shape: if resolved_adapter == OvenAdapterFamily::OpenaiResponses
873                        || resolved_adapter == OvenAdapterFamily::AzureOpenaiResponses
874                    {
875                        "responses"
876                    } else {
877                        "chat"
878                    }
879                    .into(),
880                    reasoning_field: "reasoning_content".into(),
881                    thinking_toggle: None,
882                    adapter: resolved_adapter,
883                    endpoint: Some(endpoint),
884                    setup: Some(setup.clone()),
885                    auth: auth.clone(),
886                    capabilities,
887                    replay_declaration: model.capabilities.native_replay,
888                    defaults,
889                    options,
890                    headers,
891                    cost: None,
892                    variants,
893                    variant_order,
894                    default_variant,
895                    status: CompiledModelStatus::Available,
896                    behavior_fingerprint,
897                },
898            );
899            models[id].validate_settings()?;
900        }
901        let safe_headers = provider
902            .headers
903            .iter()
904            .map(|(name, value)| (name.as_str(), value.as_str()))
905            .collect::<Vec<_>>();
906        let provider_fingerprint = fingerprint(
907            "cookie-agent/custom-provider/v1",
908            &(
909                self.registry.revision(),
910                COMPILER_VERSION,
911                provider_id,
912                provider.endpoint.as_str(),
913                adapter,
914                &setup,
915                &auth,
916                &safe_headers,
917                models
918                    .iter()
919                    .map(|(id, model)| (id, &model.behavior_fingerprint))
920                    .collect::<Vec<_>>(),
921                &unsupported_models,
922            ),
923        );
924        Ok(CompiledDynamicProvider {
925            id: provider_id.clone(),
926            models,
927            unsupported_models,
928            fingerprint: provider_fingerprint,
929        })
930    }

935/// Endpoint of one managed model when the provider has an authored `base_url`.
936///
937/// The authored URL replaces the provider-level catalog API URL. A model that
938/// pins its own catalog API URL on the same origin as the provider's (mixed
939/// gateways such as zenmux or ofox serve Anthropic under `/anthropic/v1`) is
940/// rebased: the path suffix where it differs from the provider API is swapped
941/// in for the provider's suffix at the end of `base_url`. When `base_url` does
942/// not end with that suffix it is a gateway with its own layout, and every
943/// model goes to `base_url` itself. A model API on another origin is a
944/// separate upstream and is kept.
945pub(crate) fn authored_model_endpoint(
946    base_url: &str,
947    provider_api: Option<&str>,
948    model_api: Option<&str>,
949) -> String {
950    let base_url = base_url.trim_end_matches('/');
951    let (Some(provider_api), Some(model_api)) = (provider_api, model_api) else {
952        return model_api.map_or(base_url, |api| api).to_owned();
953    };
954    let (provider_origin, provider_path) = split_origin(provider_api);
955    let (model_origin, model_path) = split_origin(model_api);
956    if provider_origin != model_origin {
957        return model_api.to_owned();
958    }
959    let common = provider_path
960        .iter()
961        .zip(&model_path)
962        .take_while(|(left, right)| left == right)
963        .count();
964    let (base_origin, base_path) = split_origin(base_url);
965    let Some(kept) = base_path
966        .len()
967        .checked_sub(provider_path.len() - common)
968        .filter(|kept| base_path[*kept..] == provider_path[common..])
969    else {
970        return base_url.to_owned();
971    };
972    let mut endpoint = base_origin.to_owned();
973    for segment in base_path[..kept].iter().chain(&model_path[common..]) {
974        endpoint.push('/');
975        endpoint.push_str(segment);
976    }
977    endpoint
978}

980/// Splits a URL (or `${PLACEHOLDER}` template) into its origin and non-empty
981/// path segments.
982fn split_origin(url: &str) -> (&str, Vec<&str>) {
983    let authority = url.find("://").map_or(0, |index| index + 3);
984    let end = url[authority..]
985        .find('/')
986        .map_or(url.len(), |index| authority + index);
987    (
988        &url[..end],
989        url[end..]
990            .split('/')
991            .filter(|segment| !segment.is_empty())
992            .collect(),
993    )
994}

996fn merge_headers<'a>(
997    layers: impl IntoIterator<Item = (&'a BTreeMap<HeaderName, SafeStaticHeaderValue>, String)>,
998) -> Result<BTreeMap<HeaderName, SafeStaticHeaderValue>, DynamicCompileError> {
999    let mut merged = BTreeMap::new();
1000    for (layer, scope) in layers {
1001        validate_header_ownership(layer, scope).map_err(DynamicCompileError::StaticHeaders)?;
1002        for (name, value) in layer {
1003            if value.as_str().is_empty() {
1004                merged.remove(name);
1005            } else {
1006                merged.insert(name.clone(), value.clone());
1007            }
1008        }
1009    }
1010    validate_header_limits(&merged).map_err(DynamicCompileError::StaticHeaders)?;
1011    Ok(merged)
1012}

1014pub(crate) fn validate_managed_cache(
1015    record: &CatalogProviderRecord,
1016    authored: Option<&ModelsDevProvider>,
1017    family: &'static FamilyRecipe,
1018) -> Result<(), DynamicCompileError> {
1019    let Some(authored) = authored else {
1020        return Ok(());
1021    };
1022    let Some(cache) = authored.cache.as_ref() else {
1023        return Ok(());
1024    };
1025    cache.validate_supported_shape().map_err(|error| {
1026        DynamicCompileError::Cache(format!("provider `{}`: {error}", record.id))
1027    })?;
1028
1029    let mut adapters = BTreeSet::from([managed_provider_adapter(
1030        family.family,
1031        None,
1032        record.shape.as_deref(),
1033    )]);
1034    for (model_id, entry) in &record.models {
1035        let Some(model) = entry.record.as_ref() else {
1036            continue;
1037        };
1038        let override_ = authored.model_overrides.get(model_id);
1039        if let Ok(resolved) = resolve_model(record, model, None, None) {
1040            let adapter = resolved
1041                .adapter
1042                .with_endpoint(override_.and_then(|value| value.options.request_endpoint))?;
1043            adapters.insert(
1044                if resolved.recipe.family == FamilyKind::OpenAiCompatibleChat {
1045                    OvenAdapterFamily::OpenaiCompatible
1046                } else {
1047                    adapter
1048                },
1049            );
1050        }
1051    }
1052    for adapter in adapters {
1053        crate::authoring::validate_provider_cache(Some(cache), adapter.id()).map_err(|error| {
1054            DynamicCompileError::Cache(format!("provider `{}`: {error}", record.id))
1055        })?;
1056    }
1057    Ok(())
1058}

1060pub(crate) fn managed_provider_adapter(
1061    family: FamilyKind,
1062    authored_shape: Option<crate::ManagedModelShape>,
1063    catalog_shape: Option<&str>,
1064) -> OvenAdapterFamily {
1065    let responses = matches!(authored_shape, Some(crate::ManagedModelShape::Responses))
1066        || authored_shape.is_none() && catalog_shape == Some("responses")
1067        || authored_shape.is_none() && catalog_shape.is_none() && family == FamilyKind::OpenAi;
1068    match family {
1069        FamilyKind::OpenAiCompatibleChat => OvenAdapterFamily::OpenaiCompatible,
1070        FamilyKind::Anthropic => OvenAdapterFamily::AnthropicCompatible,
1071        FamilyKind::OpenAi if responses => OvenAdapterFamily::OpenaiResponses,
1072        FamilyKind::OpenAi => OvenAdapterFamily::OpenaiChat,
1073        FamilyKind::Google => OvenAdapterFamily::GoogleGemini,
1074        FamilyKind::Vertex | FamilyKind::VertexAnthropic => OvenAdapterFamily::GoogleVertexGemini,
1075        FamilyKind::Bedrock if responses => OvenAdapterFamily::OpenaiResponses,
1076        FamilyKind::Bedrock => OvenAdapterFamily::AwsBedrockConverse,
1077        FamilyKind::Azure if responses => OvenAdapterFamily::AzureOpenaiResponses,
1078        FamilyKind::Azure => OvenAdapterFamily::AzureOpenaiChat,
1079        FamilyKind::Cohere => OvenAdapterFamily::CohereV2Chat,
1080    }

1109fn resolved_managed_setup_and_endpoint(
1110    _provider_family: &'static FamilyRecipe,
1111    family: FamilyKind,
1112    template: Option<&str>,
1113    authored: Option<&ModelsDevProvider>,
1114) -> Result<(Option<ValidatedSetup>, Option<String>), ModelLocalError> {
1115    if template.is_none() && family != FamilyKind::Vertex {
1116        return Ok((None, None));
1117    }
1118    let input = authored
1119        .map(|value| &value.setup)
1120        .cloned()
1121        .unwrap_or_default();
1122    let mut values = BTreeMap::new();
1123    for (id, value) in &input {
1124        let crate::authoring::SafeSetupValue::String(value) = value else {
1125            return Err(ModelLocalError::Provider(DynamicCompileError::Setup));
1126        };
1127        values.insert(id.as_str().to_owned(), value.as_str().to_owned());
1128    }
1129    let mut required = template
1130        .map(placeholders)
1131        .unwrap_or_default()
1132        .into_iter()
1133        .map(|name| crate::recipes::setup_field_name(&name))
1134        .collect::<Vec<_>>();
1135    match family {
1136        FamilyKind::Vertex => required.extend(["project".into(), "location".into()]),
1137        FamilyKind::Bedrock => required.push("region".into()),
1138        FamilyKind::Azure => required.push("resource_name".into()),
1139        _ => {}
1140    }
1141    required.sort();
1142    required.dedup();
1143    if required.iter().any(|field| !values.contains_key(field)) {
1144        return Ok((None, None));
1145    }
1146    let endpoint = template
1147        .and_then(|template| substitute_placeholders(template, &values))
1148        .map(|value| value.trim_end_matches('/').to_owned())
1149        .or_else(|| match family {
1150            FamilyKind::Vertex => Some(format!(
1151                "https://{}-aiplatform.googleapis.com/v1/projects/{}/locations/{}",
1152                values.get("location")?,
1153                values.get("project")?,
1154                values.get("location")?
1155            )),
1156            _ => None,
1157        });
1158    let setup = ValidatedSetup {
1159        recipe_id: "family-derived-setup-v1",
1160        values,
1161    };
1162    Ok((Some(setup), endpoint))
1163}

1165fn managed_auth(
1166    provider_family: &FamilyRecipe,
1167    effective_recipe: &FamilyRecipe,
1168    required_method: Option<&'static str>,
1169    authored: Option<&ModelsDevProvider>,
1170) -> Result<CompiledAuthShape, ModelLocalError> {
1171    if let Some(authored) = authored {
1172        if authored.api_key.is_some() {
1173            let source_method = provider_family.default_auth_method;
1174            let target_method = compatible_model_auth(
1175                source_method,
1176                provider_family,
1177                effective_recipe,
1178                required_method,
1179            );
1180            let Some(method) = target_method.and_then(auth_method) else {
1181                return Ok(auth_shape(
1182                    auth_method(effective_recipe.default_auth_method)
1183                        .ok_or(ModelLocalError::Provider(DynamicCompileError::Auth))?,
1184                    BTreeMap::new(),
1185                    AuthSourceCategory::Unavailable,
1186                ));
1187            };
1188            let required_api_key = method.credentials.len() == 1
1189                && method.credentials[0].required
1190                && method.credentials[0].name == "api_key";
1191            if !required_api_key {
1192                return Err(ModelLocalError::Provider(DynamicCompileError::Auth));
1193            }
1194            return Ok(auth_shape(
1195                method,
1196                BTreeMap::new(),
1197                AuthSourceCategory::AuthoredApiKey,
1198            ));
1199        }
1200        if let Some(auth) = &authored.auth_override {
1201            let target_method = compatible_model_auth(
1202                auth.method.as_str(),
1203                provider_family,
1204                effective_recipe,
1205                required_method,
1206            );
1207            let Some(method) = target_method.and_then(auth_method) else {
1208                return Ok(auth_shape(
1209                    auth_method(effective_recipe.default_auth_method)
1210                        .ok_or(ModelLocalError::Provider(DynamicCompileError::Auth))?,
1211                    BTreeMap::new(),
1212                    AuthSourceCategory::Unavailable,
1213                ));
1214            };
1215            return Ok(auth_shape(
1216                method,
1217                BTreeMap::new(),
1218                AuthSourceCategory::AuthoredOverride,
1219            ));
1220        }
1221    }
1222    let method = auth_method(required_method.unwrap_or(effective_recipe.default_auth_method))
1223        .ok_or(ModelLocalError::Provider(DynamicCompileError::Auth))?;
1224    Ok(auth_shape(
1225        method,
1226        BTreeMap::new(),
1227        AuthSourceCategory::Unavailable,
1228    ))
1229}

1282        credential_fields,
1283        owned_headers: owned_headers.into_iter().collect(),

1288fn validate_custom_options(options: &ProviderOptions, adapter: OvenAdapterFamily) -> bool {
1289    let has_openai =
1290        options.organization.is_some() || options.project.is_some() || options.store.is_some();
1291    let has_anthropic = !options.beta.is_empty();
1292    let has_compatible = options.request_endpoint.is_some()
1293        && !matches!(
1294            adapter,
1295            OvenAdapterFamily::OpenaiChat
1296                | OvenAdapterFamily::OpenaiResponses
1297                | OvenAdapterFamily::OpenaiCompatible
1298                | OvenAdapterFamily::AzureOpenaiChat
1299                | OvenAdapterFamily::AzureOpenaiResponses
1300        );
1301    let has_setup_leak = options.api_version.is_some()
1302        || options.location.is_some()
1303        || options.region.is_some()
1304        || options.deployment.is_some();
1305    !has_setup_leak
1306        && match adapter {
1307            OvenAdapterFamily::Anthropic | OvenAdapterFamily::AnthropicCompatible => {
1308                !has_openai && !has_compatible
1309            }
1310            OvenAdapterFamily::OpenaiChat => {
1311                !has_anthropic && !has_compatible && options.store.is_none()
1312            }
1313            OvenAdapterFamily::OpenaiResponses => !has_anthropic && !has_compatible,
1314            OvenAdapterFamily::OpenaiCompatible => !has_anthropic && !has_openai,
1315            OvenAdapterFamily::GoogleGemini
1316            | OvenAdapterFamily::GoogleVertexGemini
1317            | OvenAdapterFamily::AwsBedrockConverse
1318            | OvenAdapterFamily::AzureOpenaiChat
1319            | OvenAdapterFamily::AzureOpenaiResponses
1320            | OvenAdapterFamily::CohereV2Chat => !has_anthropic && !has_openai && !has_compatible,
1321        }
1322}

1325    reasoning: Option<&crate::ReasoningBehavior>,
1326    adapter: OvenAdapterFamily,
1327    thinking_toggle: Option<CompatibleThinkingToggle>,
1328) -> bool {
1329    match reasoning {
1330        None => true,
1331        Some(crate::ReasoningBehavior::Effort { .. }) => adapter != OvenAdapterFamily::CohereV2Chat,
1332        Some(crate::ReasoningBehavior::Toggle { .. })
1333            if adapter == OvenAdapterFamily::OpenaiCompatible =>
1334        {
1335            matches!(
1336                thinking_toggle,
1337                Some(
1338                    CompatibleThinkingToggle::ThinkingType
1339                        | CompatibleThinkingToggle::EnableThinking
1340                )
1341            )
1342        }
1343        Some(
1344            crate::ReasoningBehavior::Toggle { .. } | crate::ReasoningBehavior::BudgetTokens { .. },
1345        ) => matches!(
1346            adapter,
1347            OvenAdapterFamily::Anthropic
1348                | OvenAdapterFamily::AnthropicCompatible
1349                | OvenAdapterFamily::AwsBedrockConverse
1350                | OvenAdapterFamily::GoogleGemini
1351                | OvenAdapterFamily::GoogleVertexGemini
1352                | OvenAdapterFamily::CohereV2Chat
1353        ),
1354    }
1355}

1362    fn authored_base_url_rebases_same_origin_model_endpoints() {
1363        let rebase = |base, provider, model| authored_model_endpoint(base, provider, model);
1364        // Models without their own API URL, or with the provider's, use base_url.
1365        let zen = Some("https://zenmux.ai/api/v1");
1366        assert_eq!(
1367            rebase("https://gw.test/v1/", zen, None),
1368            "https://gw.test/v1"

1370        assert_eq!(rebase("https://gw.test/v1", zen, zen), "https://gw.test/v1");
1371        // Same-origin model endpoints keep their path difference.
1372        assert_eq!(
1373            rebase(
1374                "https://gw.test/zen/v1",
1375                zen,
1376                Some("https://zenmux.ai/api/anthropic/v1")
1377            ),
1378            "https://gw.test/zen/anthropic/v1"
1379        );
1380        assert_eq!(
1381            rebase(
1382                "https://gw.test/v1",
1383                Some("https://api.ofox.ai/v1"),
1384                Some("https://api.ofox.ai/gemini/v1beta")
1385            ),
1386            "https://gw.test/gemini/v1beta"
1387        );
1388        assert_eq!(
1389            rebase(
1390                "https://gw.test/v1",
1391                Some("${NEON_AI_GATEWAY_BASE_URL}/v1"),
1392                Some("${NEON_AI_GATEWAY_BASE_URL}/openai/v1")
1393            ),
1394            "https://gw.test/openai/v1"
1395        );
1396        // A gateway with its own layout serves every model at base_url.
1397        assert_eq!(
1398            rebase(
1399                "https://gw.test/proxy",
1400                zen,
1401                Some("https://zenmux.ai/api/anthropic/v1")

1406        assert_eq!(

1410                Some("https://api.freemodel.dev/v1")

1412            "https://api.freemodel.dev/v1"
1413        );
