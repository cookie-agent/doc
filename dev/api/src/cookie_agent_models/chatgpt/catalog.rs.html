20const CACHE: &str = "chatgpt-models.json";
21const CACHE_LOCK: &str = "chatgpt-models.lock";
22const CACHE_LIMIT: u64 = 4 * 1024 * 1024;
23/// Output limit for a model models.dev does not list: requests never send
24/// one, so it only sizes the input budget.
25const FALLBACK_OUTPUT_TOKENS: u64 = 128_000;
26/// Codex's multi-agent effort, which token sharing refuses (`multi_agent`).
27const UNSUPPORTED_EFFORTS: &[&str] = &["ultra"];
28
29/// One model the plan lists for use through the API.
30#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
31#[serde(deny_unknown_fields)]
32pub struct ChatGptModel {
33    pub slug: String,
34    pub display_name: String,
35    pub description: String,
36    /// The input window Codex reports.
37    pub context_window: Option<u64>,
38    pub input_modalities: Vec<String>,
39    /// Reasoning efforts, in the order the list gives them.
40    pub efforts: Vec<String>,
41    /// Codex's ordering; lower comes first.
42    pub priority: Option<i64>,
43}

45#[derive(Deserialize, Serialize)]
46#[serde(deny_unknown_fields)]
47struct ModelCache {
48    subject: String,
49    fetched_at: u64,
50    models: Vec<ChatGptModel>,
51}
52
53impl ChatGptCredentials {
54    /// Lists the signed-in plan's models and caches them. The list does not
55    /// prove access; a completed request does.
56    pub async fn fetch_models(&self) -> Result<Vec<ChatGptModel>, ChatGptError> {
57        let subject = self
58            .account()
59            .filter(|account| !account.needs_sign_in)
60            .ok_or(ChatGptError::SignInRequired)?
61            .subject;
62        let mut answer = self.list_models().await?;
63        if answer.status == 401 {
64            // A refused token is refreshed once, as a model request does.
65            let token = self.access_token().await?;
66            oven_sdk_openai::OpenAiTokenSource::rejected(self, &token);
67            answer = self.list_models().await?;
68        }
69        if !answer.success {
70            return Err(ChatGptError::Failed(format!(
71                "the model list answered HTTP {}",
72                answer.status
73            )));
74        }
75        let models = parse_models(&answer.body)?;
76        let cache = ModelCache {
77            subject,
78            fetched_at: now_unix(),
79            models: models.clone(),
80        };
81        let bytes = serde_json::to_vec_pretty(&cache)
82            .map_err(|_| ChatGptError::Failed("could not encode the model list".into()))?;
83        self.directory
84            .lock(CACHE_LOCK)?
85            .atomic_replace(CACHE, &bytes)?;
86        Ok(models)
87    }
88
89    /// The cached list of the signed-in account, or nothing when the cache
90    /// belongs to another account or is missing.
91    #[must_use]
92    pub fn cached_models(&self) -> Vec<ChatGptModel> {
93        let Some(account) = self.account() else {
94            return Vec::new();
95        };
96        self.directory
97            .read(CACHE, CACHE_LIMIT)
98            .ok()
99            .flatten()
100            .and_then(|bytes| serde_json::from_slice::<ModelCache>(&bytes).ok())
101            .filter(|cache| cache.subject == account.subject)
102            .map(|cache| cache.models)
103            .unwrap_or_default()
104    }
105
106    async fn list_models(&self) -> Result<HttpAnswer, ChatGptError> {
107        let token = self.access_token().await?;
108        let response = self
109            .http
110            .get(format!(
111                "{}/models",
112                self.endpoints.api.trim_end_matches('/')
113            ))
114            .header("user-agent", user_agent())
115            .bearer_auth(token.expose_secret())
116            .send()
117            .await
118            .map_err(|_| ChatGptError::Failed("could not reach api.openai.com".into()))?;
119        HttpAnswer::read(response).await
120    }
121}

123/// The models of a `GET /v1/models` answer that are listed and usable
124/// through the API.
125fn parse_models(body: &[u8]) -> Result<Vec<ChatGptModel>, ChatGptError> {
126    let value: Value = serde_json::from_slice(body)
127        .map_err(|_| ChatGptError::Failed("the model list is not JSON".into()))?;
128    let entries = value
129        .get("models")
130        .and_then(Value::as_array)
131        .ok_or_else(|| ChatGptError::Failed("the model list has no models".into()))?;
132    let text = |entry: &Value, key: &str| entry.get(key).and_then(Value::as_str).map(str::to_owned);
133    let mut models = entries
134        .iter()
135        .filter(|entry| entry.get("visibility").and_then(Value::as_str) == Some("list"))
136        .filter(|entry| entry.get("supported_in_api").and_then(Value::as_bool) != Some(false))
137        .filter_map(|entry| {
138            let slug = text(entry, "slug")?;
139            ProviderModelId::new(slug.clone()).ok()?;
140            Some(ChatGptModel {
141                display_name: text(entry, "display_name").unwrap_or_else(|| slug.clone()),
142                slug,
143                description: text(entry, "description").unwrap_or_default(),
144                context_window: entry.get("context_window").and_then(Value::as_u64),
145                input_modalities: entry
146                    .get("input_modalities")
147                    .and_then(Value::as_array)
148                    .map(|values| {
149                        values
150                            .iter()
151                            .filter_map(Value::as_str)
152                            .map(str::to_owned)
153                            .collect()
154                    })
155                    .unwrap_or_else(|| vec!["text".to_owned()]),
156                efforts: entry
157                    .get("supported_reasoning_levels")
158                    .and_then(Value::as_array)
159                    .map(|levels| {
160                        levels
161                            .iter()
162                            .filter_map(|level| level.get("effort").and_then(Value::as_str))
163                            .filter(|effort| !UNSUPPORTED_EFFORTS.contains(effort))
164                            .map(str::to_owned)
165                            .collect()
166                    })
167                    .unwrap_or_default(),
168                priority: entry.get("priority").and_then(Value::as_i64),
169            })
170        })
171        .collect::<Vec<_>>();
172    models.sort_by(|left, right| {
173        (left.priority.unwrap_or(i64::MAX), &left.slug)
174            .cmp(&(right.priority.unwrap_or(i64::MAX), &right.slug))
175    });
176    models.dedup_by(|left, right| left.slug == right.slug);
177    Ok(models)
178}

180/// The built-in provider's catalog row for `models`. Output limits and dates
181/// come from the models.dev `openai` record of the same model when it has
182/// one.
183#[must_use]
184pub fn provider_entry(
185    models: &[ChatGptModel],
186    openai: Option<&CatalogProviderRecord>,
187) -> CatalogProviderEntry {
188    let id = ProviderId::new(PROVIDER_ID).expect("the ChatGPT provider ID is valid");
189    let models = models
190        .iter()
191        .filter_map(|model| {
192            let model_id = ProviderModelId::new(model.slug.clone()).ok()?;
193            let known = openai
194                .and_then(|record| record.models.get(&model_id))
195                .and_then(|entry| entry.record.as_ref());
196            let output = known.map_or(FALLBACK_OUTPUT_TOKENS, |record| record.limits.output);
197            let input = model
198                .context_window
199                .or_else(|| known.and_then(|record| record.limits.input))
200                .unwrap_or(272_000);
201            let record = CatalogModelRecord {
202                id: model_id.clone(),
203                name: model.display_name.clone(),
204                description: model.description.clone(),
205                family: known.and_then(|record| record.family.clone()),
206                attachment: model.input_modalities.iter().any(|value| value == "image"),
207                reasoning: true,
208                tool_call: true,
209                structured_output: Some(true),
210                temperature: Some(false),
211                open_weights: false,
212                status: CatalogModelStatus::Stable,
213                release_date: known
214                    .map(|record| record.release_date.clone())
215                    .unwrap_or_default(),
216                last_updated: known
217                    .map(|record| record.last_updated.clone())
218                    .unwrap_or_default(),
219                modalities: CatalogModalities {
220                    input: model
221                        .input_modalities
222                        .iter()
223                        .filter(|value| matches!(value.as_str(), "text" | "image"))
224                        .cloned()
225                        .collect(),
226                    output: vec!["text".to_owned()],
227                },
228                limits: CatalogLimits {
229                    // Codex reports the input window; the context window holds
230                    // the output too.
231                    context: input + output,
232                    input: Some(input),
233                    output,
234                },
235                shape: Some("responses".to_owned()),
236                provider: None,
237                reasoning_options: if model.efforts.is_empty() {
238                    Vec::new()
239                } else {
240                    vec![CatalogReasoningOption::Effort {
241                        values: model.efforts.iter().cloned().map(Some).collect(),
242                    }]
243                },
244                cost: None,
245                interleaved: None,
246                canonical_provenance: None,
247                dynamically_loaded_tools: Some(true),
248            };
249            Some((
250                model_id.clone(),
251                CatalogModelEntry {
252                    id: model_id,
253                    record: Some(record),
254                    quarantine: None,
255                },
256            ))
257        })
258        .collect::<BTreeMap<_, _>>();
259    CatalogProviderEntry {
260        id: id.clone(),
261        record: Some(CatalogProviderRecord {
262            id,
263            name: DISPLAY_NAME.to_owned(),
264            environment: Vec::new(),
265            npm: crate::recipes::CHATGPT_NPM.to_owned(),
266            api: Some("https://api.openai.com/v1".to_owned()),
267            shape: Some("responses".to_owned()),
268            documentation_url: "https://developers.openai.com/siwc".to_owned(),
269            models,
270        }),
271        quarantine: None,
272    }
273}
