1//! Sign in with ChatGPT: OpenAI's token sharing, which runs models on a
2//! ChatGPT plan through the public Responses API.
3//!
4//! The sign-in is an OAuth authorization-code flow with PKCE against
5//! `auth.openai.com`. The first sign-in registers a client for the user with
6//! the shared `dynamic_agent_client` ID and keeps the client ID OpenAI issues
7//! on the callback. Tokens live in the provider-store directory, outside the
8//! provider store itself, so an hourly refresh never recompiles the model
9//! runtime; the store holds only the signed-in account.

43/// Why a sign-in, a refresh, or a model listing failed. Each code reaches
44/// clients unchanged.
45#[derive(Debug, thiserror::Error)]
46pub enum ChatGptError {
47    #[error("sign_in_denied: the ChatGPT sign-in was declined")]
48    Denied,
49    #[error("sign_in_not_eligible: this ChatGPT account cannot share its plan with other apps")]
50    NotEligible,
51    #[error("sign_in_scope_missing: the sign-in did not grant plan usage")]
52    ScopeMissing,
53    #[error("sign_in_state_mismatch: the callback does not belong to this sign-in")]
54    StateMismatch,
55    #[error("sign_in_expired: the sign-in waited more than 10 minutes; start it again")]
56    Expired,
57    #[error("sign_in_not_pending: no ChatGPT sign-in is in progress")]
58    NotPending,
59    #[error(
60        "sign_in_client_rejected: OpenAI refused cookie's saved client registration; sign in again to register a new one"
61    )]
62    ClientRejected,
63    #[error("invalid_redirect_uri: the redirect must be http://127.0.0.1:<port>/auth/callback")]
64    InvalidRedirectUri,
65    #[error("invalid_callback: {0}")]
66    InvalidCallback(&'static str),
67    #[error("sign_in_required: sign in with ChatGPT again")]
68    SignInRequired,
69    #[error("sign_in_failed: {0}")]
70    Failed(String),
71    #[error("chatgpt_storage: {0}")]
72    Storage(#[from] crate::secure_store::SecureStoreError),
73}
74
75impl ChatGptError {
76    /// The stable code before the message.
77    #[must_use]
78    pub fn code(&self) -> &'static str {
79        match self {
80            Self::Denied => "sign_in_denied",
81            Self::NotEligible => "sign_in_not_eligible",
82            Self::ScopeMissing => "sign_in_scope_missing",
83            Self::StateMismatch => "sign_in_state_mismatch",
84            Self::Expired => "sign_in_expired",
85            Self::NotPending => "sign_in_not_pending",
86            Self::ClientRejected => "sign_in_client_rejected",
87            Self::InvalidRedirectUri => "invalid_redirect_uri",
88            Self::InvalidCallback(_) => "invalid_callback",
89            Self::SignInRequired => "sign_in_required",
90            Self::Failed(_) => "sign_in_failed",
91            Self::Storage(_) => "chatgpt_storage",
92        }
93    }
94}

96/// Whether `uri` is the only redirect token sharing accepts:
97/// `http://127.0.0.1:<port>/auth/callback`, with nothing else.
98#[must_use]
99pub fn valid_redirect_uri(uri: &str) -> bool {
100    let Some(rest) = uri.strip_prefix("http://127.0.0.1:") else {
101        return false;
102    };
103    let Some(port) = rest.strip_suffix(REDIRECT_PATH) else {
104        return false;
105    };
106    !port.is_empty()
107        && port.len() <= 5
108        && port.bytes().all(|byte| byte.is_ascii_digit())
109        && !port.starts_with('0')
110        && port.parse::<u16>().is_ok()
111}

113fn now_unix() -> u64 {
114    std::time::SystemTime::now()
115        .duration_since(std::time::UNIX_EPOCH)
116        .map_or(0, |elapsed| elapsed.as_secs())
117}

119fn random_token(bytes: usize) -> Result<String, ChatGptError> {
120    let mut buffer = vec![0_u8; bytes];
121    getrandom::getrandom(&mut buffer)
122        .map_err(|_| ChatGptError::Failed("no system randomness".into()))?;
123    Ok(base64url::encode(&buffer))
124}
