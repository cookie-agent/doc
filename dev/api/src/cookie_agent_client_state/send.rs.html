1//! Sending a message the composer holds: it reaches the daemon, or it goes
2//! back to the draft it came from with why. A message never vanishes: one
3//! sent while the client has no connection, or whose answer a closing
4//! connection lost (the daemon restarting, say), is the client's to put
5//! back.
6
7use cookie_agent_protocol::{SessionSubmitInputParams, SessionSubmitInputResult};
8use cookie_agent_protocol_transport::{Client, ClientError};
9
10/// Why a message did not reach the daemon, or may not have.
11#[derive(Debug)]
12pub enum Unsent {
13    /// There was no connection to send it on: it was not sent.
14    NotConnected,
15    /// The daemon refused it, or the connection closed before it answered.
16    Failed(ClientError),
17}
18
19impl Unsent {
20    /// What the composer says of the message it holds again.
21    pub fn describe(&self, error: impl FnOnce(&ClientError) -> String) -> String {
22        match self {
23            Self::NotConnected => {
24                "not connected to the daemon: the message was not sent".to_owned()
25            }
26            Self::Failed(failure) => error(failure),
27        }
28    }
29}
30
31/// Submits `params` over `client`, `None` while the client has no
32/// connection. An `Err` hands the message back: it was not sent, or its
33/// answer never came.
34pub async fn submit_input(
35    client: Option<&Client>,
36    params: SessionSubmitInputParams,
37) -> Result<SessionSubmitInputResult, Unsent> {
38    let client = client.ok_or(Unsent::NotConnected)?;
39    client.submit_input(params).await.map_err(Unsent::Failed)
40}

42#[cfg(all(test, not(target_arch = "wasm32")))]
43mod tests {
44    use cookie_agent_protocol::{
45        ClientRunId, InputDelivery, MessageFrame, MessageStream as _, SessionId,
46        SessionSubmitInputParams, in_process_pair,
47    };
48    use cookie_agent_protocol_transport::Client;
49
50    use super::{Unsent, submit_input};
51
52    fn params() -> SessionSubmitInputParams {
53        SessionSubmitInputParams {
54            session_id: SessionId::new_v7(),
55            content: "say afterwards".into(),
56            delivery: InputDelivery::Steer,
57            selection: None,
58            client_run_id: ClientRunId::new("client-run").unwrap(),
59            reset_fallback: false,
60        }
61    }
62
63    #[tokio::test]
64    async fn a_message_sent_with_no_connection_comes_back() {
65        assert!(matches!(
66            submit_input(None, params()).await,
67            Err(Unsent::NotConnected)
68        ));
69    }
70
71    #[tokio::test]
72    async fn a_message_whose_connection_closes_before_the_answer_comes_back() {
73        let (client_end, mut daemon_end) = in_process_pair(8);
74        let client = Client::connect_stream(client_end);
75        // The daemon reads the message, then its connection closes without
76        // an answer, as a restarting daemon's does.
77        let daemon = tokio::spawn(async move {
78            let request = daemon_end.recv().await.unwrap().unwrap();
79            drop(daemon_end);
80            request
81        });
82        let unsent = tokio::time::timeout(
83            std::time::Duration::from_secs(30),
84            submit_input(Some(&client), params()),
85        )
86        .await
87        .expect("the send ends with the connection")
88        .expect_err("no answer came");
89        assert!(matches!(unsent, Unsent::Failed(_)), "{unsent:?}");
90        let MessageFrame::Text(request) = daemon.await.unwrap() else {
91            panic!("a text frame");
92        };
93        assert!(request.contains("say afterwards"), "{request}");
94    }
