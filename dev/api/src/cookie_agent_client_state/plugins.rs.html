14/// Whether `field` takes a secret.
15fn is_secret(field: &cookie_agent_protocol::ElicitationField) -> bool {
16    matches!(
17        field.input.known(),
18        Some(ElicitationInput::Text { secret: true, .. })
19    )
20}

22/// The form a client shows for a plugin's settings: a secret that is set
23/// is not required, since left empty it keeps its value, and says so.
24#[must_use]
25pub fn settings_form(settings: &PluginsSettingsGetResult) -> ElicitationForm {
26    let mut form = settings.form.clone();
27    for field in &mut form.fields {
28        if is_secret(field) && settings.secrets_set.contains(&field.name) {
29            field.required = false;
30            let note = "Set. Type a new value to replace it.";
31            field.description = Some(match field.description.take() {
32                Some(description) => format!("{description} {note}"),
33                None => note.to_owned(),
34            });
35        }
36    }
37    form
38}

40/// A draft of `form` holding the stored `values`, defaults elsewhere.
41#[must_use]
42pub fn settings_draft(form: &ElicitationForm, values: &ElicitationContent) -> FormDraft {
43    let mut draft = FormDraft::new(form);
44    for (field, slot) in form.fields.iter().zip(draft.values.iter_mut()) {
45        let Some(value) = values.get(&field.name) else {
46            continue;
47        };
48        let stored = match (field.input.known(), value) {
49            (Some(ElicitationInput::Text { .. }), ElicitationValue::Text(text)) => {
50                Some(DraftValue::Text(text.clone()))
51            }
52            (Some(ElicitationInput::Number { .. }), ElicitationValue::Number(number)) => {
53                Some(DraftValue::Text(number.to_string()))
54            }
55            (
56                Some(ElicitationInput::Number { .. } | ElicitationInput::Integer { .. }),
57                ElicitationValue::Integer(number),
58            ) => Some(DraftValue::Text(number.to_string())),
59            (Some(ElicitationInput::Choice { choices, .. }), ElicitationValue::Text(chosen)) => {
60                Some(DraftValue::Choice(
61                    choices.iter().position(|choice| &choice.value == chosen),
62                ))
63            }
64            (
65                Some(ElicitationInput::MultiChoice { choices, .. }),
66                ElicitationValue::List(chosen),
67            ) => Some(DraftValue::MultiChoice(
68                choices
69                    .iter()
70                    .map(|choice| chosen.contains(&choice.value))
71                    .collect(),
72            )),
73            (Some(ElicitationInput::Boolean { .. }), ElicitationValue::Boolean(value)) => {
74                Some(DraftValue::Boolean(*value))
75            }
76            _ => None,
77        };
78        if let Some(stored) = stored {
79            *slot = stored;
80        }
81    }
82    draft
83}

85/// The `plugins.settings.set` changes a filled-in form makes: every field
86/// that is not secret takes its value, or loses it when left empty; a
87/// secret changes only when one was typed.
88#[must_use]
89pub fn settings_changes(
90    form: &ElicitationForm,
91    content: &ElicitationContent,
92) -> BTreeMap<String, Option<ElicitationValue>> {
93    form.fields
94        .iter()
95        .filter_map(|field| {
96            let value = content.get(&field.name).cloned();
97            if is_secret(field) {
98                value.map(|value| (field.name.clone(), Some(value)))
99            } else {
100                Some((field.name.clone(), value))
101            }
102        })
103        .collect()
104}

110/// Whether the daemon at client endpoint `daemon_url` (`ws://host/ws`)
111/// serves the web page at origin `page_origin` (`http://host`): only then
112/// do its plugin pages open below the page, with the page's cookie.
113#[must_use]
114pub fn serves_page(daemon_url: &str, page_origin: &str) -> bool {
115    let Some((scheme, rest)) = daemon_url.split_once("://") else {
116        return false;
117    };
118    let scheme = match scheme.to_ascii_lowercase().as_str() {
119        "ws" => "http",
120        "wss" => "https",
121        _ => return false,
122    };
123    let authority = rest.split(['/', '?', '#']).next().unwrap_or_default();
124    format!("{scheme}://{authority}").eq_ignore_ascii_case(page_origin.trim_end_matches('/'))
125}

127/// Where a plugin's pages open, from the web page at `page_url` and the
128/// plugin's `ui_path` (`docs/site/specs/plugin-web-ui.md` section 7).
129#[must_use]
130pub fn plugin_page_url(page_url: &str, ui_path: &str) -> String {
131    let page = page_url.split('#').next().unwrap_or(page_url);
132    let separator = if page.ends_with('/') { "" } else { "/" };
133    format!("{page}{separator}{ui_path}")
134}

136pub const fn state_label(state: PluginState) -> &'static str {
137    match state {
138        PluginState::Disabled => "disabled",
139        PluginState::Starting => "starting",
140        PluginState::Connected => "connected",
141        PluginState::Reconnecting => "reconnecting",
142        PluginState::Failed => "failed",
143        PluginState::Stopped => "stopped",
144        PluginState::Unknown => "unknown",
145    }
146}

148pub fn plugin_detail(plugin: &PluginInfo, page_url: Option<&str>) -> String {
149    let mut text = format!("{}\nstate: {}", plugin.name, state_label(plugin.state));
150    if let Some(reason) = &plugin.reason {
151        text.push_str(&format!("\nreason: {reason}"));
152    }
153    if let Some(ui_path) = &plugin.ui_path {
154        let pages = match page_url {
155            Some(page_url) => plugin_page_url(page_url, ui_path),
156            None => format!("/{ui_path}"),
157        };
158        text.push_str(&format!("\npages: {pages}"));
159    }
160    if plugin.settings_missing {
161        text.push_str("\nsettings: required values missing");
162    }
163    let list = |items: &[String]| {
164        if items.is_empty() {
165            "none".to_owned()
166        } else {
167            items.join(", ")
168        }
169    };
170    text.push_str(&format!("\ntools: {}", list(&plugin.tools)));
171    let hooks = plugin
172        .hooks
173        .iter()
174        .map(|hook| hook.as_str().to_owned())
175        .collect::<Vec<_>>();
176    text.push_str(&format!("\nhooks: {}", list(&hooks)));
177    let mut streams = Vec::new();
178    if !plugin.subscribe_events.is_empty() {
179        let patterns = plugin
180            .subscribe_events
181            .iter()
182            .map(cookie_agent_protocol::EventPattern::as_str)
183            .collect::<Vec<_>>();
184        streams.push(format!("events ({})", patterns.join(", ")));
185    }
186    if plugin.subscribe_bus {
187        streams.push("bus".to_owned());
188    }
189    if plugin.publish_bus {
190        streams.push("publishes bus".to_owned());
191    }
192    if plugin.producer_messaging {
193        streams.push("producers".to_owned());
194    }
195    text.push_str(&format!("\nalso: {}", list(&streams)));
196    if plugin.dropped_events > 0 {
197        text.push_str(&format!(
198            "\ndropped notifications: {}",
199            plugin.dropped_events
200        ));
201    }
202    if let Some(stderr) = &plugin.stderr {
203        text.push_str("\n\nstderr:\n");
204        text.push_str(stderr);
205    }
206    text
207}

239    #[test]
240    fn a_set_secret_may_be_left_empty_and_only_typed_secrets_change() {
241        let settings = PluginsSettingsGetResult {
242            form: ElicitationForm {
243                title: None,
244                description: None,
245                fields: vec![
246                    text_field("client_id", true, false),
247                    text_field("client_secret", true, true),
248                    text_field("note", false, false),
249                ],
250            },
251            values: ElicitationContent::from([(
252                "client_id".to_owned(),
253                ElicitationValue::Text("abc".into()),
254            )]),
255            secrets_set: vec!["client_secret".into()],
256        };
257        let form = settings_form(&settings);
258        assert!(form.fields[0].required);
259        assert!(!form.fields[1].required);
260        assert!(form.fields[1].description.is_some());
261
262        let draft = settings_draft(&form, &settings.values);
263        assert_eq!(draft.values[0], DraftValue::Text("abc".into()));
264        assert_eq!(draft.values[1], DraftValue::Text(String::new()));
265        let content = draft.content(&form).unwrap();
266        let changes = settings_changes(&form, &content);
267        assert_eq!(
268            changes,
269            BTreeMap::from([
270                (
271                    "client_id".to_owned(),
272                    Some(ElicitationValue::Text("abc".into()))
273                ),
274                ("note".to_owned(), None),
275            ])
276        );
277
278        let typed = ElicitationContent::from([
279            ("client_id".to_owned(), ElicitationValue::Text("abc".into())),
280            (
281                "client_secret".to_owned(),
282                ElicitationValue::Text("new".into()),
283            ),
284        ]);
285        assert_eq!(
286            settings_changes(&form, &typed)["client_secret"],
287            Some(ElicitationValue::Text("new".into()))
288        );
289    }
