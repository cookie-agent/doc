16/// The client protocol minor from which the daemon keeps cron jobs
17/// (`crons.*`, `SessionMeta::cron`).
18pub const CRONS_MINOR: u32 = 15;

20/// Whether a daemon speaking `version` keeps cron jobs.
21pub fn daemon_has_crons(version: Option<SemVer>) -> bool {
22    version.is_some_and(|version| {
23        version
24            .shared_minor(cookie_agent_protocol::CLIENT_PROTOCOL_VERSION)
25            .is_some_and(|minor| minor >= CRONS_MINOR)
26    })
27}

29/// The most characters of a prompt a job without a name shows as its title.
30pub const TITLE_PROMPT_CHARS: usize = 48;
31
32/// What lists show for a job: its name, else the start of its prompt's first
33/// line.
34pub fn job_title(job: &CronJob) -> String {
35    match job.name.as_deref().map(str::trim) {
36        Some(name) if !name.is_empty() => name.to_owned(),
37        _ => prompt_start(&job.prompt),
38    }
39}

41fn prompt_start(prompt: &str) -> String {
42    let line = prompt
43        .lines()
44        .map(str::trim)
45        .find(|line| !line.is_empty())
46        .unwrap_or("");
47    let mut chars = line.chars();
48    let start: String = chars.by_ref().take(TITLE_PROMPT_CHARS).collect();
49    if chars.next().is_some() || prompt.trim() != line {
50        format!("{start}…")
51    } else {
52        start
53    }
54}

56/// The schedule as a user would type it again: the cron expression, `at
57/// <time>` in `zone`, or `in <delay>`.
58pub fn schedule_text(schedule: &CronSchedule, zone: &TimeZone) -> String {
59    match schedule {
60        CronSchedule::Cron(expression) => expression.clone(),
61        CronSchedule::In(delay) => format!("in {delay}"),
62        CronSchedule::At(at) => format!("at {}", local_minute(*at, zone)),
63    }
64}

66/// A plain reading of the schedule and its run limit, such as `every day at
67/// 03:00`, `every 15 minutes, 3 runs`, or `once at 2026-10-23 03:00`. Cron
68/// times are the daemon host's; `zone` shows a one-shot's time.
69pub fn schedule_reading(schedule: &CronSchedule, max_runs: CronMaxRuns, zone: &TimeZone) -> String {
70    match schedule {
71        CronSchedule::Cron(expression) => {
72            let reading = cron_reading(expression).unwrap_or_else(|| format!("cron {expression}"));
73            match max_runs {
74                CronMaxRuns::Unlimited => reading,
75                CronMaxRuns::Count(1) => format!("{reading}, once"),
76                CronMaxRuns::Count(count) => format!("{reading}, {count} runs"),
77            }
78        }
79        CronSchedule::In(delay) => format!("once in {delay}"),
80        CronSchedule::At(at) => format!("once at {}", local_minute(*at, zone)),
81    }
82}

84/// `at` in `zone` to the minute, such as `2026-10-23 03:00`. Formatted by
85/// hand: jiff's strftime would grow the web page.
86pub fn local_minute(at: Timestamp, zone: &TimeZone) -> String {
87    let time = zone.to_datetime(at);
88    format!(
89        "{}-{:02}-{:02} {:02}:{:02}",
90        time.year(),
91        time.month(),
92        time.day(),
93        time.hour(),
94        time.minute()
95    )
96}

98/// How long until `then` from `now`: `in 5 minutes`, `in 2 hours`, `in 3
99/// days`, or `now` once it is due.
100pub fn until(then: Timestamp, now: Timestamp) -> String {
101    let seconds = then.as_second() - now.as_second();
102    if seconds <= 0 {
103        return "now".to_owned();
104    }
105    let (count, unit) = if seconds < 60 {
106        (seconds, "second")
107    } else if seconds < 3600 {
108        (seconds / 60, "minute")
109    } else if seconds < 86_400 {
110        (seconds / 3600, "hour")
111    } else {
112        (seconds / 86_400, "day")
113    };
114    let plural = if count == 1 { "" } else { "s" };
115    format!("in {count} {unit}{plural}")
116}

142/// One field of an expression, in the shapes a reading names.
143#[derive(Debug, Eq, PartialEq)]
144enum Field {
145    Any,
146    /// `*/n`.
147    Every(u32),
148    /// Values and ranges, in the order written.
149    Values(Vec<(u32, u32)>),
150}
151
152fn field(text: &str, names: &[&str], sunday_seven: bool) -> Option<Field> {
153    if text == "*" {
154        return Some(Field::Any);
155    }
156    if let Some(step) = text.strip_prefix("*/") {
157        return step.parse().ok().filter(|step| *step > 0).map(Field::Every);
158    }
159    let value = |part: &str| -> Option<u32> {
160        let number = part.parse::<u32>().ok().or_else(|| {
161            names
162                .iter()
163                .position(|name| name[..3].eq_ignore_ascii_case(part))
164                .and_then(|index| u32::try_from(index).ok())
165                .map(|index| if names.len() == 12 { index + 1 } else { index })
166        })?;
167        Some(if sunday_seven && number == 7 {
168            0
169        } else {
170            number
171        })
172    };
173    let mut values = Vec::new();
174    for part in text.split(',') {
175        if part.contains('/') {
176            return None;
177        }
178        let range = match part.split_once('-') {
179            Some((low, high)) => (value(low)?, value(high)?),
180            None => {
181                let single = value(part)?;
182                (single, single)
183            }
184        };
185        values.push(range);
186    }
187    Some(Field::Values(values))
188}
189
190fn single(field: &Field) -> Option<u32> {
191    match field {
192        Field::Values(values) => match values.as_slice() {
193            [(low, high)] if low == high => Some(*low),
194            _ => None,
195        },
196        _ => None,
197    }
198}
199
200fn singles(field: &Field) -> Option<Vec<u32>> {
201    match field {
202        Field::Values(values) => values
203            .iter()
204            .map(|(low, high)| (low == high).then_some(*low))
205            .collect(),
206        _ => None,
207    }
208}

210fn listing(parts: &[String]) -> String {
211    match parts {
212        [] => String::new(),
213        [one] => one.clone(),
214        [init @ .., last] => format!("{} and {last}", init.join(", ")),
215    }
216}

218fn named(field: &Field, names: &[&str], offset: u32) -> Option<String> {
219    let Field::Values(values) = field else {
220        return None;
221    };
222    let name = |value: u32| -> Option<&str> {
223        names
224            .get(usize::try_from(value.checked_sub(offset)?).ok()?)
225            .copied()
226    };
227    let parts = values
228        .iter()
229        .map(|(low, high)| {
230            if low == high {
231                name(*low).map(str::to_owned)
232            } else {
233                Some(format!("{} to {}", name(*low)?, name(*high)?))
234            }
235        })
236        .collect::<Option<Vec<_>>>()?;
237    Some(listing(&parts))
238}

240fn ordinal(day: u32) -> String {
241    let suffix = match (day % 10, day % 100) {
242        (1, 11) | (2, 12) | (3, 13) => "th",
243        (1, _) => "st",
244        (2, _) => "nd",
245        (3, _) => "rd",
246        _ => "th",
247    };
248    format!("{day}{suffix}")
249}

251/// A plain reading of a five-field cron expression, for the shapes people
252/// write: `every day at 03:00`, `every Monday to Friday at 09:00 and 17:00`,
253/// `every 15 minutes`, `every hour at :05`, `on the 1st of every month at
254/// 08:00`. `None` for anything else, which clients show as the expression.
255pub fn cron_reading(expression: &str) -> Option<String> {
256    let fields: Vec<&str> = expression.split_whitespace().collect();
257    let [minute, hour, day, month, weekday] = fields.as_slice() else {
258        return None;
259    };
260    let minute = field(minute, &[], false)?;
261    let hour = field(hour, &[], false)?;
262    let day = field(day, &[], false)?;
263    let month = field(month, &MONTHS, false)?;
264    let weekday = field(weekday, &WEEKDAYS, true)?;
265
266    let time = match (&minute, &hour) {
267        (Field::Any, Field::Any) => "every minute".to_owned(),
268        (Field::Every(step), Field::Any) => format!("every {step} minutes"),
269        (minute, Field::Any) => format!("every hour at :{:02}", single(minute)?),
270        (minute, Field::Every(step)) => {
271            format!("every {step} hours at :{:02}", single(minute)?)
272        }
273        (minute, hours) => {
274            let minute = single(minute)?;
275            let hours = singles(hours)?;
276            let times: Vec<String> = hours
277                .iter()
278                .map(|hour| format!("{hour:02}:{minute:02}"))
279                .collect();
280            format!("at {}", listing(&times))
281        }
282    };
283    let daily = time.starts_with("at ");
284
285    let months = match &month {
286        Field::Any => None,
287        Field::Every(_) => return None,
288        month => Some(named(month, &MONTHS, 1)?),
289    };
290    let in_months = months
291        .as_deref()
292        .map(|months| format!(" in {months}"))
293        .unwrap_or_default();
294    let days = match (&day, &weekday) {
295        (Field::Any, Field::Any) => months.as_ref().map(|_| format!("every day{in_months}")),
296        (Field::Any, weekday) => Some(format!(
297            "every {}{in_months}",
298            named(weekday, &WEEKDAYS, 0)?
299        )),
300        (day, Field::Any) => {
301            let days: Vec<String> = singles(day)?.into_iter().map(ordinal).collect();
302            let of = months.as_deref().unwrap_or("every month");
303            Some(format!("on the {} of {of}", listing(&days)))
304        }
305        (day, weekday) => {
306            let days: Vec<String> = singles(day)?.into_iter().map(ordinal).collect();
307            Some(format!(
308                "on the {} and every {}{in_months}",
309                listing(&days),
310                named(weekday, &WEEKDAYS, 0)?
311            ))
312        }
313    };
314    let reading = match (days, daily) {
315        (None, true) => format!("every day {time}"),
316        (Some(days), true) => format!("{days} {time}"),
317        (None, false) => time,
318        (Some(days), false) => format!("{time}, {days}"),
319    };
320    Some(reading)
321}

323/// How lists name a job's target: `this session`, `session <title>`,
324/// `subagent <title>`, or `new session · <agent>`. `title` names a session
325/// the client knows.
326pub fn target_label(
327    target: &CronTarget,
328    current: Option<SessionId>,
329    title: impl Fn(SessionId) -> Option<String>,
330) -> String {
331    let session = |kind: &str, session: SessionId| {
332        if Some(session) == current {
333            format!("this {kind}")
334        } else {
335            let name = title(session).unwrap_or_else(|| short_session(session));
336            format!("{kind} {name}")
337        }
338    };
339    match target {
340        CronTarget::Session(id) => session("session", *id),
341        CronTarget::Subagent(id) => session("subagent", *id),
342        CronTarget::New(template) => format!("new session · {}", template.agent),
343    }
344}

350/// Whether the job is on, and why not: `enabled`, `finished`, `failed 3
351/// times in a row`, `disabled by you`, `disabled by agent builder`.
352pub fn state_label(job: &CronJob) -> String {
353    if job.enabled {
354        return "enabled".to_owned();
355    }
356    match &job.disabled_reason {
357        Some(Open::Known(reason)) => disabled_reason_label(reason),
358        Some(Open::Unknown { .. }) | None => "disabled".to_owned(),

362/// Why a job is disabled, as `state_label` says it.

364    match reason {
365        CronDisabledReason::Finished => "finished".to_owned(),
366        CronDisabledReason::Failed { failures } => {
367            let times = if *failures == 1 { "time" } else { "times" };
368            format!("failed {failures} {times} in a row")
369        }
370        CronDisabledReason::Disabled { by } => format!("disabled by {}", actor_label(by)),
371    }

396/// What a fire did, and how the run it led to ended: `delivered ·
397/// completed`, `skipped: daemon was down (3 missed)`, `failed: no model`,
398/// prefixed `run now · ` for a run now.
399pub fn fire_outcome(fire: &CronFire) -> String {
400    let outcome = match &fire.outcome {
401        Open::Known(CronFireOutcome::Pending) => "pending".to_owned(),
402        Open::Known(CronFireOutcome::Delivered) => match &fire.run_end {
403            Some(end) => format!("delivered · {}", run_status_label(end.status)),
404            None if fire.run_id.is_some() => "delivered · running".to_owned(),
405            None => "delivered".to_owned(),
406        },
407        Open::Known(CronFireOutcome::Skipped { reason }) => match fire.missed {
408            Some(missed) if missed > 1 => {
409                format!("skipped: {} ({missed} missed)", skip_reason_label(*reason))
410            }
411            _ => format!("skipped: {}", skip_reason_label(*reason)),
412        },
413        Open::Known(CronFireOutcome::Failed { error }) => format!("failed: {error}"),
414        Open::Unknown { tag, .. } => tag.replace('_', " "),
415    };
416    if fire.run_now {
417        format!("run now · {outcome}")
418    } else {
419        outcome
420    }
421}

423/// A short word for a fire's outcome, for a list column: `delivered`,
424/// `completed`, `failed`, `skipped`, `pending`, or `never` for a job that
425/// has not fired.
426pub fn last_outcome(job: &CronJob) -> &'static str {
427    let Some(fire) = job.history.last() else {
428        return "never";
429    };
430    match &fire.outcome {
431        Open::Known(CronFireOutcome::Pending) => "pending",
432        Open::Known(CronFireOutcome::Delivered) => match &fire.run_end {
433            Some(end) => run_status_label(end.status),
434            None => "delivered",
435        },
436        Open::Known(CronFireOutcome::Skipped { .. }) => "skipped",
437        Open::Known(CronFireOutcome::Failed { .. }) => "failed",
438        Open::Unknown { .. } => "unknown",
439    }
440}

453/// The detail lines of a job, each `label: value`, in the order clients
454/// show them.
455pub fn detail_lines(
456    job: &CronJob,
457    target: &str,
458    now: Timestamp,
459    zone: &TimeZone,
460) -> Vec<(&'static str, String)> {
461    let mut lines = vec![
462        ("id", job.id.to_string()),
463        ("schedule", schedule_text(&job.schedule, zone)),
464        ("reads", schedule_reading(&job.schedule, job.max_runs, zone)),
465        ("target", target.to_owned()),
466        ("state", state_label(job)),
467    ];
468    if let Some(next) = job.next_fire {
469        lines.push((
470            "next fire",
471            format!("{} ({})", local_minute(next, zone), until(next, now)),
472        ));
473    }
474    if matches!(job.schedule, CronSchedule::Cron(_)) {
475        lines.push(("runs", format!("{} of {}", job.runs, job.max_runs)));
476    }
477    if let Some(delivery) = job.delivery {
478        lines.push(("delivery", delivery_label(delivery).to_owned()));
479    }
480    if let CronTarget::New(template) = &job.target {
481        lines.push(("agent", template_agent(template)));
482        if let Some(model) = &template.model {
483            lines.push(("model", model.to_string()));
484        }
485        if let Some(project) = &template.project {
486            lines.push(("project", project.to_string()));
487        }
488        if !template.permission_rules.is_empty() {
489            lines.push((
490                "rules",
491                format!("{} added", template.permission_rules.len()),
492            ));
493        }
494    }
495    if let Some(mode) = job.permission_mode {
496        lines.push(("mode", permission_mode_label(mode).to_owned()));
497    }
498    if job.consecutive_failures > 0 {
499        lines.push(("failures", format!("{} in a row", job.consecutive_failures)));
500    }
501    if let Some(limit) = job.max_consecutive_failures {
502        lines.push(("failure limit", limit.to_string()));
503    }
504    lines.push((
505        "created",
506        format!(
507            "{} by {}",
508            local_minute(job.created_at, zone),
509            actor_label(&job.created_by)
510        ),
511    ));
512    lines
513}

530/// Reads a schedule as typed: `in <delay>` (such as `in 20m`), `at <RFC
531/// 3339 time>`, or a five-field cron expression.
532pub fn parse_schedule(text: &str) -> Result<CronSchedule, String> {
533    let text = text.trim();
534    if let Some(delay) = text.strip_prefix("in ") {
535        let delay = delay.trim();
536        if delay.is_empty() {
537            return Err("a delay needs a length, such as `in 20m`".to_owned());
538        }
539        return Ok(CronSchedule::In(delay.to_owned()));
540    }
541    if let Some(at) = text.strip_prefix("at ") {
542        return at
543            .trim()
544            .parse::<Timestamp>()
545            .map(CronSchedule::At)
546            .map_err(|_| "a time needs RFC 3339, such as `at 2026-10-23T03:00:00Z`".to_owned());
547    }
548    let fields: Vec<&str> = text.split_whitespace().collect();
549    if fields.len() == 5 {
550        return Ok(CronSchedule::Cron(fields.join(" ")));
551    }
552    Err("a schedule is five cron fields (`0 3 * * *`) or a delay (`in 20m`)".to_owned())
553}

555/// Reads a run limit as typed: empty for the daemon's default, a whole
556/// number from 1, or `unlimited`.
557pub fn parse_max_runs(text: &str) -> Result<Option<CronMaxRuns>, String> {
558    let text = text.trim();
559    if text.is_empty() {
560        return Ok(None);
561    }
562    if text.eq_ignore_ascii_case("unlimited") {
563        return Ok(Some(CronMaxRuns::Unlimited));
564    }
565    text.parse::<u32>()
566        .ok()
567        .filter(|count| *count >= 1)
568        .map(|count| Some(CronMaxRuns::Count(count)))
569        .ok_or_else(|| "runs are a whole number from 1, or `unlimited`".to_owned())
570}

572/// Reads a model pin as typed: empty for none, `provider/model`, or
573/// `provider/model[variant]`.
574pub fn parse_model_pin(text: &str) -> Result<Option<ModelSelection>, String> {
575    let text = text.trim();
576    if text.is_empty() {
577        return Ok(None);
578    }
579    let (model, variant) = match text.strip_suffix(']').and_then(|rest| rest.split_once('[')) {
580        Some((model, "base")) => (model, None),
581        Some((model, variant)) => (model, Some(variant)),
582        None => (text, None),
583    };
584    serde_json::from_value(serde_json::json!({"model": model, "variant": variant}))
585        .map(Some)
586        .map_err(|_| "a model is `provider/model` or `provider/model[variant]`".to_owned())
587}

589/// Where a form's job fires.
590#[derive(Clone, Debug, Eq, PartialEq)]
591pub enum CronDraftTarget {
592    /// An existing session: a root one, or a delegated one when
593    /// `subagent`.
594    Session { session: SessionId, subagent: bool },
595    /// A new session per fire, from `template`; the form edits its agent
596    /// and model pin and keeps the rest.
597    New { template: CronTemplate },
598}
599
600impl CronDraftTarget {
601    /// The target `crons.create` takes.
602    pub fn target(&self) -> CronTarget {
603        match self {
604            Self::Session {
605                session,
606                subagent: false,
607            } => CronTarget::Session(*session),
608            Self::Session {
609                session,
610                subagent: true,
611            } => CronTarget::Subagent(*session),
612            Self::New { template } => CronTarget::New(template.clone()),
613        }
614    }
615
616    pub const fn is_new(&self) -> bool {
617        matches!(self, Self::New { .. })
618    }
619}

638    pub name: String,

640    /// As [`parse_schedule`] reads it.
641    pub schedule: String,
642    /// As [`parse_max_runs`] reads it.
643    pub max_runs: String,
644    pub target: CronDraftTarget,
645    /// For an in-session job.
646    pub delivery: ProducerDeliveryMode,
647    /// For a new-session job; the daemon's default when `None`.
648    pub permission_mode: Option<PermissionMode>,
649    /// For a new-session job, as [`parse_model_pin`] reads it.
650    pub model: String,
651    /// For a new-session job: the template's agent.
652    pub agent: String,
653}
654
655impl CronDraft {
656    /// A form for a job that fires into `session`, a delegated one when
657    /// `subagent`.
658    pub fn for_session(session: SessionId, subagent: bool) -> Self {
659        Self {
660            name: String::new(),
661            prompt: String::new(),
662            schedule: String::new(),
663            max_runs: String::new(),
664            target: CronDraftTarget::Session { session, subagent },
665            delivery: ProducerDeliveryMode::Queue,
666            permission_mode: None,
667            model: String::new(),
668            agent: String::new(),
669        }
670    }
671
672    /// A form for a job that starts a session from `template` per fire.
673    pub fn for_new_session(template: CronTemplate) -> Self {
674        let agent = template.agent.to_string();
675        let model = template
676            .model
677            .as_ref()
678            .map(model_pin_text)
679            .unwrap_or_default();
680        Self {
681            name: String::new(),
682            prompt: String::new(),
683            schedule: String::new(),
684            max_runs: String::new(),
685            target: CronDraftTarget::New { template },
686            delivery: ProducerDeliveryMode::Queue,
687            permission_mode: None,
688            model,
689            agent,
690        }
691    }
692
693    /// The form filled from `job`, to edit it.
694    pub fn of_job(job: &CronJob, zone: &TimeZone) -> Self {
695        let target = match &job.target {
696            CronTarget::Session(session) => CronDraftTarget::Session {
697                session: *session,
698                subagent: false,
699            },
700            CronTarget::Subagent(session) => CronDraftTarget::Session {
701                session: *session,
702                subagent: true,
703            },
704            CronTarget::New(template) => CronDraftTarget::New {
705                template: template.clone(),
706            },
707        };
708        let (agent, model) = match &job.target {
709            CronTarget::New(template) => (
710                template.agent.to_string(),
711                template
712                    .model
713                    .as_ref()
714                    .map(model_pin_text)
715                    .unwrap_or_default(),
716            ),
717            _ => (String::new(), String::new()),
718        };
719        Self {
720            name: job.name.clone().unwrap_or_default(),
721            prompt: job.prompt.clone(),
722            schedule: match &job.schedule {
723                CronSchedule::At(at) => format!("at {at}"),
724                schedule => schedule_text(schedule, zone),
725            },
726            max_runs: match job.schedule {
727                CronSchedule::Cron(_) => job.max_runs.to_string(),
728                _ => String::new(),
729            },
730            target,
731            delivery: job.delivery.unwrap_or(ProducerDeliveryMode::Queue),
732            permission_mode: job.permission_mode,
733            model,
734            agent,
735        }
736    }
737
738    /// The target with the form's agent and model pin applied.
739    fn resolved_target(&self) -> Result<CronTarget, String> {
740        match &self.target {
741            CronDraftTarget::New { template } => {
742                let agent = self.agent.trim();
743                let agent = if agent.is_empty() {
744                    template.agent.clone()
745                } else {
746                    agent
747                        .parse::<AgentId>()
748                        .map_err(|_| format!("`{agent}` is not an agent id"))?
749                };
750                let mut template = template.clone();
751                template.agent = agent;
752                template.model = parse_model_pin(&self.model)?;
753                Ok(CronTarget::New(template))
754            }
755            target => Ok(target.target()),
756        }
757    }
758
759    fn checked_prompt(&self) -> Result<String, String> {
760        let prompt = self.prompt.trim();
761        if prompt.is_empty() {
762            return Err("a job needs a prompt".to_owned());
763        }
764        Ok(self.prompt.trim_end().to_owned())
765    }
766
767    fn checked_name(&self) -> Option<String> {
768        let name = self.name.trim();
769        (!name.is_empty()).then(|| name.to_owned())
770    }
771
772    /// `crons.create`'s params, or why the form cannot be sent.
773    pub fn create_params(&self) -> Result<CronCreateParams, String> {
774        let schedule = parse_schedule(&self.schedule)?;
775        let max_runs = match schedule {
776            CronSchedule::Cron(_) => parse_max_runs(&self.max_runs)?,
777            _ => None,
778        };
779        let new = self.target.is_new();
780        Ok(CronCreateParams {
781            name: self.checked_name(),
782            prompt: self.checked_prompt()?,
783            schedule,
784            max_runs,
785            target: self.resolved_target()?,
786            delivery: (!new).then_some(self.delivery),
787            permission_mode: if new { self.permission_mode } else { None },
788            max_consecutive_failures: None,
789            enabled: None,
790        })
791    }
792
793    /// `crons.update`'s params for what the form changed in `job`, or why
794    /// the form cannot be sent.
795    pub fn update_params(
796        &self,
797        job: &CronJob,
798        zone: &TimeZone,
799    ) -> Result<CronUpdateParams, String> {
800        let before = Self::of_job(job, zone);
801        let mut params = CronUpdateParams::new(job.id.clone());
802        if self.name.trim() != before.name.trim() {
803            params.name = Some(self.checked_name());
804        }
805        let prompt = self.checked_prompt()?;
806        if prompt != job.prompt {
807            params.prompt = Some(prompt);
808        }
809        let schedule = parse_schedule(&self.schedule)?;
810        let schedule_changed = self.schedule.trim() != before.schedule.trim();
811        if schedule_changed {
812            params.schedule = Some(schedule.clone());
813        }
814        if matches!(schedule, CronSchedule::Cron(_))
815            && (self.max_runs.trim() != before.max_runs.trim() || schedule_changed)
816        {
817            params.max_runs = parse_max_runs(&self.max_runs)?;
818            if params.max_runs == Some(job.max_runs) && !schedule_changed {
819                params.max_runs = None;
820            }
821        }
822        let target = self.resolved_target()?;
823        if target != job.target {
824            params.target = Some(target);
825        }
826        if !self.target.is_new() && Some(self.delivery) != job.delivery {
827            params.delivery = Some(self.delivery);
828        }
829        if self.target.is_new() && self.permission_mode != job.permission_mode {
830            params.permission_mode = Some(self.permission_mode);
831        }
832        Ok(params)
833    }
834}
835
836/// A model pin as [`parse_model_pin`] reads it back.
837pub fn model_pin_text(model: &ModelSelection) -> String {
838    match &model.variant {
839        Some(variant) => format!("{}[{variant}]", model.model),
840        None => model.model.to_string(),
841    }
842}
843
844/// Whether `update` changes anything.
845pub fn update_changes_anything(update: &CronUpdateParams) -> bool {
846    *update != CronUpdateParams::new(update.id.clone())
847}

849/// The job with `id` in `jobs`.
850pub fn job_by_id<'a>(jobs: &'a [CronJob], id: &str) -> Option<&'a CronJob> {
851    jobs.iter().find(|job| job.id.as_str() == id)
