1//! Where a session runs, as the badges a session row shows
2//! (`docs/site/specs/projects.md` section 9), and the stable colour of a
3//! machine or a runtime. The `global` attachment never counts. A session
4//! with no other attachment has a "no workspace" badge; one attachment
5//! shows its runtime and its workspace; several on one machine show the
6//! machine, then each runtime; several machines show each machine, in
7//! attachment order. The machine comes from `runtimes.list`; a runtime of
8//! no known machine stands for its own machine with its runtime badge.
9
10use cookie_agent_protocol::{RuntimeAttachment, RuntimeId, RuntimesListResult, SessionMeta};
11
12use crate::paths::RuntimeHomes;
13
14/// How many hues badges take.
15pub const BADGE_HUES: u8 = 8;
16
17/// The hue (`0..BADGE_HUES`) of the machine or runtime named `key`: a
18/// 64-bit FNV-1a hash of its bytes, so every client and every run picks the
19/// same one.
20#[must_use]
21pub fn badge_hue(key: &str) -> u8 {
22    let hash = key.bytes().fold(0xcbf2_9ce4_8422_2325_u64, |hash, byte| {
23        (hash ^ u64::from(byte)).wrapping_mul(0x0100_0000_01b3)
24    });
25    u8::try_from(hash % u64::from(BADGE_HUES)).expect("a hue below eight")
26}

28/// One badge of a session row.
29#[derive(Clone, Debug, Eq, PartialEq)]
30pub enum SessionBadge {
31    /// The session attaches nothing besides `global`.
32    NoWorkspace,
33    /// A machine, with the operating system its runtimes report.
34    Machine {
35        name: String,
36        platform: Option<String>,
37    },
38    /// A runtime; `capability` only when the runtime offers several.
39    Runtime {
40        runtime_id: String,
41        capability: Option<String>,
42    },
43    /// The one attachment's workspace, as clients show paths (`~/…`). From
44    /// a daemon that lists no attachments, the session's directory.
45    Workspace(String),
46}
47
48impl SessionBadge {
49    /// The badge's hue: a machine's by its name, a runtime's by its ID.
50    /// The others are grey.
51    #[must_use]
52    pub fn hue(&self) -> Option<u8> {
53        match self {
54            Self::Machine { name, .. } => Some(badge_hue(name)),
55            Self::Runtime { runtime_id, .. } => Some(badge_hue(runtime_id)),
56            Self::NoWorkspace | Self::Workspace(_) => None,
57        }
58    }
59
60    /// What the badge says.
61    #[must_use]
62    pub fn label(&self) -> String {
63        match self {
64            Self::NoWorkspace => "no workspace".to_owned(),
65            Self::Machine { name, .. } => name.clone(),
66            Self::Runtime {
67                runtime_id,
68                capability: Some(capability),
69            } => format!("{runtime_id}·{capability}"),
70            Self::Runtime { runtime_id, .. } => runtime_id.clone(),
71            Self::Workspace(path) => path.clone(),
72        }
73    }
74}

76/// What `runtimes.list` says of the runtime `runtime_id`.
77fn listing_of<'a>(
78    listing: Option<&'a RuntimesListResult>,
79    runtime_id: &RuntimeId,
80) -> Option<&'a cookie_agent_protocol::RuntimeListing> {
81    listing?
82        .runtimes
83        .iter()
84        .find(|runtime| runtime.runtime_id == *runtime_id)
85}
86
87/// The machine `runtime_id` runs on, if `runtimes.list` knows it.
88#[must_use]
89pub fn runtime_machine(
90    listing: Option<&RuntimesListResult>,
91    runtime_id: &RuntimeId,
92) -> Option<String> {
93    listing_of(listing, runtime_id)?
94        .machine
95        .as_ref()
96        .map(|machine| machine.as_str().to_owned())
97}
98
99/// The operating system of `machine`: the first of its runtimes that
100/// reported one.
101fn machine_platform(listing: Option<&RuntimesListResult>, machine: &str) -> Option<String> {
102    listing?
103        .runtimes
104        .iter()
105        .filter(|runtime| {
106            runtime
107                .machine
108                .as_ref()
109                .is_some_and(|name| name.as_str() == machine)
110        })
111        .find_map(|runtime| runtime.platform.clone())
112}

114/// The badge of `attachment`'s runtime: its capability named only when the
115/// runtime offers several.
116#[must_use]
117pub fn runtime_badge(
118    listing: Option<&RuntimesListResult>,
119    attachment: &RuntimeAttachment,
120) -> SessionBadge {
121    let several = listing_of(listing, &attachment.runtime_id)
122        .is_some_and(|runtime| runtime.capabilities.len() > 1);
123    SessionBadge::Runtime {
124        runtime_id: attachment.runtime_id.to_string(),
125        capability: several.then(|| attachment.capability.to_string()),
126    }
127}
128
129/// The attachments that count for badges: all but `global`.
130fn counted(session: &SessionMeta) -> impl Iterator<Item = &RuntimeAttachment> {
131    session
132        .attachments
133        .iter()
134        .filter(|attachment| attachment.runtime_id.as_str() != RuntimeId::GLOBAL)
135}

137/// Where an attachment runs: its machine, or the runtime itself when the
138/// machine is not known.
139#[derive(Clone, Debug, Eq, PartialEq)]
140enum Place {
141    Machine(String),
142    Runtime(RuntimeId),
143}
144
145fn place(listing: Option<&RuntimesListResult>, attachment: &RuntimeAttachment) -> Place {
146    runtime_machine(listing, &attachment.runtime_id).map_or_else(
147        || Place::Runtime(attachment.runtime_id.clone()),
148        Place::Machine,
149    )
150}

152/// The badges of `session`'s row, in order. `listed` says whether the
153/// daemon lists attachments (client protocol 1.7): from an older one the
154/// row shows the session's directory alone.
155#[must_use]
156pub fn session_badges(
157    session: &SessionMeta,
158    listing: Option<&RuntimesListResult>,
159    homes: &RuntimeHomes,
160    listed: bool,
161) -> Vec<SessionBadge> {
162    if session.attachments.is_empty() && !listed {
163        return vec![SessionBadge::Workspace(homes.session_directory(session))];
164    }
165    let attachments = counted(session).collect::<Vec<_>>();
166    let [first, rest @ ..] = attachments.as_slice() else {
167        return vec![SessionBadge::NoWorkspace];
168    };
169    if rest.is_empty() {
170        let mut badges = vec![runtime_badge(listing, first)];
171        if let Some(workspace) = &first.workspace {
172            badges.push(SessionBadge::Workspace(
173                homes.display(&first.runtime_id, workspace),
174            ));
175        }
176        return badges;
177    }
178    let mut places = Vec::<Place>::new();
179    for attachment in &attachments {
180        let place = place(listing, attachment);
181        if !places.contains(&place) {
182            places.push(place);
183        }
184    }
185    let machine_badge = |place: &Place| match place {
186        Place::Machine(name) => SessionBadge::Machine {
187            platform: machine_platform(listing, name),
188            name: name.clone(),
189        },
190        Place::Runtime(runtime_id) => {
191            let attachment = attachments
192                .iter()
193                .find(|attachment| attachment.runtime_id == *runtime_id)
194                .expect("a place comes from an attachment");
195            runtime_badge(listing, attachment)
196        }
197    };
198    let mut badges = Vec::new();
199    if let [only] = places.as_slice() {
200        // One machine: it, then each runtime once. A runtime of no known
201        // machine is that machine.
202        if matches!(only, Place::Machine(_)) {
203            badges.push(machine_badge(only));
204        }
205        for attachment in &attachments {
206            let badge = runtime_badge(listing, attachment);
207            if !badges.contains(&badge) {
208                badges.push(badge);
209            }
210        }
211    } else {
212        badges.extend(places.iter().map(machine_badge));
213    }
214    badges
215}

217/// The row's tooltip: every attachment but `global`, one a line, as
218/// `machine · runtime · ~/workspace`, the machine left out when it is not
219/// known.
220#[must_use]
221pub fn badges_tooltip(
222    session: &SessionMeta,
223    listing: Option<&RuntimesListResult>,
224    homes: &RuntimeHomes,
225    listed: bool,
226) -> String {
227    if session.attachments.is_empty() && !listed {
228        return homes.session_directory(session);
229    }
230    let lines = counted(session)
231        .map(|attachment| {
232            let mut parts = Vec::new();
233            if let Some(machine) = runtime_machine(listing, &attachment.runtime_id) {
234                parts.push(machine);
235            }
236            parts.push(runtime_badge(listing, attachment).label());
237            if let Some(workspace) = &attachment.workspace {
238                parts.push(homes.display(&attachment.runtime_id, workspace));
239            }
240            parts.join(" · ")
241        })
242        .collect::<Vec<_>>();
243    if lines.is_empty() {
244        return "no workspace".to_owned();
245    }
246    lines.join("\n")
247}

258    fn listing() -> RuntimesListResult {
259        let runtime = |id: &str, machine: Option<&str>, platform: Option<&str>, caps: &[&str]| {
260            RuntimeListing {
261                runtime_id: RuntimeId::new(id).unwrap(),
262                lifetime: RuntimeLifetime::Builtin,
263                connected: true,
264                capabilities: caps
265                    .iter()
266                    .map(|name| RuntimeCapabilityInfo {
267                        name: CapabilityName::new(*name).unwrap(),
268                        kind: RuntimeKind::new(*name).unwrap(),
269                        takes_workspace: true,
270                        description: String::new(),
271                    })
272                    .collect(),
273                platform: platform.map(str::to_owned),
274                machine: machine.map(|name| MachineName::new(name).unwrap()),
275                home: Some(format!("/home/{id}")),
276                message: None,
277            }
278        };
279        RuntimesListResult {
280            runtimes: vec![
281                runtime("global", Some("desk"), Some("linux"), &["global"]),
282                runtime("local", Some("desk"), Some("linux"), &["coding"]),
283                runtime("alt", Some("desk"), None, &["coding", "notes"]),
284                runtime("labpc", Some("lab"), Some("macos"), &["coding"]),
285                runtime("box", None, None, &[]),
286            ],
287            instances: Vec::new(),

347    #[test]
348    fn hues_are_stable_and_spread_over_eight() {
349        // Pinned: every client and every release colours these the same.
350        assert_eq!(badge_hue("local"), badge_hue("local"));
351        assert_eq!(
352            ["local", "alt", "box", "roamer", "labpc", "global"].map(badge_hue),
353            [0, 4, 2, 7, 5, 6]
354        );
355        let mut seen = [false; BADGE_HUES as usize];
356        for index in 0..64 {
357            let hue = badge_hue(&format!("runtime-{index}"));
358            assert!(hue < BADGE_HUES);
359            seen[usize::from(hue)] = true;
360        }
361        assert!(seen.iter().all(|seen| *seen), "{seen:?}");
362    }

427    #[test]
428    fn several_machines_are_each_machine_in_attachment_order() {
429        let spread = session(&[
430            ("labpc", "coding", Some("/home/labpc/api")),
431            ("local", "coding", Some("/home/local/web")),

435        let shown = badges(&spread);
436        assert_eq!(labels(&shown), ["lab", "desk", "box"]);
437        assert_eq!(
438            shown[0],

445        assert!(matches!(shown[2], SessionBadge::Runtime { .. }));
446        assert_eq!(shown[1].hue(), Some(badge_hue("desk")));
447        assert_eq!(shown[2].hue(), Some(badge_hue("box")));
448        assert_eq!(SessionBadge::NoWorkspace.hue(), None);
