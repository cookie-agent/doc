1//! Where a session runs, as the badges a session row shows
2//! (`docs/site/specs/projects.md` section 9), and the stable colour of a
3//! machine. Badges show machines, never runtime IDs, and the `global`
4//! attachment never counts. A session with no other attachment has no
5//! badge; one machine shows its badge and the workspaces, as `~/…`; several
6//! machines show each machine, in attachment order. A root's badges count
7//! its own attachments followed by the ones its subagents brought
8//! (`SessionMeta::delegated_attachments`, client protocol 1.10.0;
9//! `docs/site/specs/runtime-delegation.md` section 5.3). The machine comes
10//! from `runtimes.list`; a runtime of no known machine stands for its own
11//! machine, under its runtime ID. Under a project's header, a machine its
12//! defaults name shows as its icon alone, and the workspaces leave the row
13//! when its defaults name them all: the header already says both.
14

22/// How many hues badges take.
23pub const BADGE_HUES: u8 = 8;
24
25/// The hue (`0..BADGE_HUES`) of the machine named `key`: a 64-bit FNV-1a
26/// hash of its bytes, so every client and every run picks the same one.
27#[must_use]
28pub fn badge_hue(key: &str) -> u8 {
29    let hash = key.bytes().fold(0xcbf2_9ce4_8422_2325_u64, |hash, byte| {
30        (hash ^ u64::from(byte)).wrapping_mul(0x0100_0000_01b3)
31    });
32    u8::try_from(hash % u64::from(BADGE_HUES)).expect("a hue below eight")
33}

35/// What the colour of `runtime_id` is keyed on: its machine's name, or its
36/// own ID when `runtimes.list` knows no machine for it, so a runtime takes
37/// its machine's colour everywhere (badges, cards, the bottom bar).
38#[must_use]
39pub fn hue_key(listing: Option<&RuntimesListResult>, runtime_id: &str) -> String {
40    listing
41        .and_then(|listing| {
42            listing
43                .runtimes
44                .iter()
45                .find(|runtime| runtime.runtime_id.as_str() == runtime_id)
46        })
47        .and_then(|runtime| runtime.machine.as_ref())
48        .map_or_else(
49            || runtime_id.to_owned(),
50            |machine| machine.as_str().to_owned(),
51        )
52}

54/// One badge of a session row.
55#[derive(Clone, Debug, Eq, PartialEq)]
56pub enum SessionBadge {
57    /// A machine, with the operating system its runtimes report; a runtime
58    /// of no known machine is one under its runtime ID. `in_project` when
59    /// the session's project's defaults name it: its header shows it.
60    Machine {
61        name: String,
62        platform: Option<String>,
63        in_project: bool,
64    },
65    /// The workspaces on the one machine, as clients show paths (`~/…`),
66    /// comma-joined. From a daemon that lists no attachments, the session's
67    /// directory.
68    Workspace(String),
69}
70
71impl SessionBadge {
72    /// The badge's hue: a machine's by its name. A workspace has none.
73    #[must_use]
74    pub fn hue(&self) -> Option<u8> {
75        match self {
76            Self::Machine { name, .. } => Some(badge_hue(name)),
77            Self::Workspace(_) => None,
78        }
79    }
80
81    /// What the badge says.
82    #[must_use]
83    pub fn label(&self) -> String {
84        match self {
85            Self::Machine { name, .. } => name.clone(),
86            Self::Workspace(path) => path.clone(),
87        }
88    }
89}

91/// What `runtimes.list` says of the runtime `runtime_id`.
92fn listing_of<'a>(
93    listing: Option<&'a RuntimesListResult>,
94    runtime_id: &RuntimeId,
95) -> Option<&'a cookie_agent_protocol::RuntimeListing> {
96    listing?
97        .runtimes
98        .iter()
99        .find(|runtime| runtime.runtime_id == *runtime_id)
100}

102/// The machine `runtime_id` runs on, if `runtimes.list` knows it.
103#[must_use]
104pub fn runtime_machine(
105    listing: Option<&RuntimesListResult>,
106    runtime_id: &RuntimeId,
107) -> Option<String> {
108    listing_of(listing, runtime_id)?
109        .machine
110        .as_ref()
111        .map(|machine| machine.as_str().to_owned())
112}

114/// The operating system of `machine`: the first of its runtimes that
115/// reported one.
116fn machine_platform(listing: Option<&RuntimesListResult>, machine: &str) -> Option<String> {
117    listing?
118        .runtimes
119        .iter()
120        .filter(|runtime| {
121            runtime
122                .machine
123                .as_ref()
124                .is_some_and(|name| name.as_str() == machine)
125        })
126        .find_map(|runtime| runtime.platform.clone())
127}

129/// What the badges read of an attachment, whether a session has it or a
130/// project's defaults name it: the runtime, the capability when known, and
131/// the workspace.
132#[derive(Clone, Copy, Debug)]
133struct Placed<'a> {
134    runtime_id: &'a RuntimeId,
135    capability: Option<&'a CapabilityName>,
136    workspace: Option<&'a str>,
137}
138
139/// Where an attachment runs: its machine, or the runtime itself when the
140/// machine is not known.
141#[derive(Clone, Debug, Eq, PartialEq)]
142enum Place {
143    Machine(String),
144    Runtime(RuntimeId),
145}
146
147impl<'a> Placed<'a> {
148    fn of_attachment(attachment: &'a RuntimeAttachment) -> Self {
149        Self {
150            runtime_id: &attachment.runtime_id,
151            capability: Some(&attachment.capability),
152            workspace: attachment.workspace.as_deref(),
153        }
154    }
155
156    fn of_default(runtime: &'a ProjectRuntime) -> Self {
157        Self {
158            runtime_id: &runtime.runtime_id,
159            capability: runtime.capability.as_ref(),
160            workspace: runtime.workspace.as_deref(),
161        }
162    }
163
164    /// Whether `other` names the same runtime, capability, and workspace.
165    fn same_place(self, other: Self) -> bool {
166        self.runtime_id == other.runtime_id
167            && self.capability == other.capability
168            && self.workspace == other.workspace
169    }
170
171    /// The runtime as the tooltip names it: its ID, with the capability
172    /// when the runtime offers several (`alt·notes`).
173    fn runtime_label(self, listing: Option<&RuntimesListResult>) -> String {
174        let several = listing_of(listing, self.runtime_id)
175            .is_some_and(|runtime| runtime.capabilities.len() > 1);
176        match self.capability.filter(|_| several) {
177            Some(capability) => format!("{}·{capability}", self.runtime_id),
178            None => self.runtime_id.to_string(),
179        }
180    }
181
182    fn place(self, listing: Option<&RuntimesListResult>) -> Place {
183        runtime_machine(listing, self.runtime_id)
184            .map_or_else(|| Place::Runtime(self.runtime_id.clone()), Place::Machine)
185    }
186}
187
188impl Place {
189    fn badge(&self, listing: Option<&RuntimesListResult>, in_project: bool) -> SessionBadge {
190        match self {
191            Self::Machine(name) => SessionBadge::Machine {
192                platform: machine_platform(listing, name),
193                name: name.clone(),
194                in_project,
195            },
196            Self::Runtime(runtime_id) => SessionBadge::Machine {
197                name: runtime_id.to_string(),
198                platform: listing_of(listing, runtime_id)
199                    .and_then(|runtime| runtime.platform.clone()),
200                in_project,
201            },
202        }
203    }
204}

206/// The attachments that count for badges: the session's own, then the ones
207/// its subagents brought, all but `global`, each place once.
208fn counted(session: &SessionMeta) -> Vec<Placed<'_>> {
209    let mut counted = Vec::<Placed<'_>>::new();
210    for attachment in session
211        .attachments
212        .iter()
213        .chain(&session.delegated_attachments)
214        .filter(|attachment| attachment.runtime_id.as_str() != RuntimeId::GLOBAL)
215    {
216        let placed = Placed::of_attachment(attachment);
217        if !counted.iter().any(|known| known.same_place(placed)) {
218            counted.push(placed);
219        }
220    }
221    counted
222}
223
224/// The default attachments that count for badges: all but `global`.
225fn counted_defaults(defaults: Option<&ProjectDefaults>) -> Vec<Placed<'_>> {
226    defaults
227        .map(|defaults| {
228            defaults
229                .runtimes
230                .iter()
231                .filter(|runtime| runtime.runtime_id.as_str() != RuntimeId::GLOBAL)
232                .map(Placed::of_default)
233                .collect()
234        })
235        .unwrap_or_default()
236}

238/// The badges of `session`'s row, in order; none for a session that
239/// attaches nothing besides `global`. `listed` says whether the daemon
240/// lists attachments (client protocol 1.7): from an older one the row shows
241/// the session's directory alone. `project` is the defaults of the project
242/// whose header the row is under: what they name, the row shortens.
243#[must_use]
244pub fn session_badges(
245    session: &SessionMeta,
246    listing: Option<&RuntimesListResult>,
247    homes: &RuntimeHomes,
248    listed: bool,
249    project: Option<&ProjectDefaults>,
250) -> Vec<SessionBadge> {
251    if session.attachments.is_empty() && !listed {
252        return vec![SessionBadge::Workspace(homes.session_directory(session))];
253    }
254    let project = counted_defaults(project);
255    placed_badges(&counted(session), &project, listing, homes)
256}

258/// The badges of a project header: where its default attachments run, by
259/// the session rows' rules; none when the defaults name no runtime besides
260/// `global`, so the header has no second line.
261#[must_use]
262pub fn project_badges(
263    defaults: Option<&ProjectDefaults>,
264    listing: Option<&RuntimesListResult>,
265    homes: &RuntimeHomes,
266) -> Vec<SessionBadge> {
267    placed_badges(&counted_defaults(defaults), &[], listing, homes)

270/// Each place of `attachments` once, in order.
271fn places_of(attachments: &[Placed<'_>], listing: Option<&RuntimesListResult>) -> Vec<Place> {
272    let mut places = Vec::<Place>::new();
273    for attachment in attachments {
274        let place = attachment.place(listing);
275        if !places.contains(&place) {
276            places.push(place);
277        }
278    }
279    places
280}

282/// The workspaces of `attachments` as clients show them, each once.
283fn workspaces_of(attachments: &[Placed<'_>], homes: &RuntimeHomes) -> Vec<String> {
284    let mut workspaces = Vec::<String>::new();
285    for attachment in attachments {
286        if let Some(workspace) = attachment.workspace {
287            let shown = homes.display(attachment.runtime_id, workspace);
288            if !workspaces.contains(&shown) {
289                workspaces.push(shown);
290            }
291        }
292    }
293    workspaces
294}

296/// The badges of `attachments`; the places and workspaces of `project`,
297/// the defaults its header shows, shortened.
298fn placed_badges(
299    attachments: &[Placed<'_>],
300    project: &[Placed<'_>],
301    listing: Option<&RuntimesListResult>,
302    homes: &RuntimeHomes,
303) -> Vec<SessionBadge> {
304    let places = places_of(attachments, listing);
305    let project_places = places_of(project, listing);
306    let badge = |place: &Place| place.badge(listing, project_places.contains(place));
307    match places.as_slice() {
308        [] => Vec::new(),
309        // One machine: its badge, then its workspaces, unless the project
310        // names them all on that machine.
311        [only] => {
312            let workspaces = workspaces_of(attachments, homes);
313            let project_workspaces = workspaces_of(
314                &project
315                    .iter()
316                    .copied()
317                    .filter(|placed| placed.place(listing) == *only)
318                    .collect::<Vec<_>>(),
319                homes,
320            );
321            let mut badges = vec![badge(only)];
322            if !workspaces
323                .iter()
324                .all(|workspace| project_workspaces.contains(workspace))
325            {
326                badges.push(SessionBadge::Workspace(workspaces.join(", ")));
327            }
328            badges
329        }
330        several => several.iter().map(badge).collect(),
331    }
332}

334/// The row's tooltip: every attachment but `global`, one a line, as
335/// `machine · runtime · ~/workspace`, the machine left out when it is not
336/// known.
337#[must_use]
338pub fn badges_tooltip(
339    session: &SessionMeta,
340    listing: Option<&RuntimesListResult>,
341    homes: &RuntimeHomes,
342    listed: bool,
343) -> String {
344    if session.attachments.is_empty() && !listed {
345        return homes.session_directory(session);
346    }
347    placed_tooltip(&counted(session), listing, homes)
348}

350/// A project header's tooltip: its default attachments, as
351/// [`badges_tooltip`] lists a session's.
352#[must_use]
353pub fn project_badges_tooltip(
354    defaults: &ProjectDefaults,
355    listing: Option<&RuntimesListResult>,
356    homes: &RuntimeHomes,
357) -> String {
358    placed_tooltip(&counted_defaults(Some(defaults)), listing, homes)
359}
360
361fn placed_tooltip(
362    attachments: &[Placed<'_>],
363    listing: Option<&RuntimesListResult>,
364    homes: &RuntimeHomes,
365) -> String {
366    let lines = attachments
367        .iter()
368        .map(|attachment| {
369            let mut parts = Vec::new();
370            if let Some(machine) = runtime_machine(listing, attachment.runtime_id) {
371                parts.push(machine);
372            }
373            parts.push(attachment.runtime_label(listing));
374            if let Some(workspace) = attachment.workspace {
375                parts.push(homes.display(attachment.runtime_id, workspace));
376            }
377            parts.join(" · ")
378        })
379        .collect::<Vec<_>>();
380    if lines.is_empty() {
381        return "no workspace".to_owned();
382    }
383    lines.join("\n")
384}

647            ..ProjectDefaults::default()
648        };
649        let under = |session: &SessionMeta| {
650            session_badges(session, Some(&listing), &homes, true, Some(&defaults))
651        };
652        let lab = |in_project| SessionBadge::Machine {
653            name: "lab".into(),
654            platform: Some("macos".into()),
655            in_project,
656        };
657        // The project's machine and workspace: the machine's icon alone.
658        let same = session(&[
659            ("global", "global", None),
660            ("labpc", "coding", Some("/home/labpc/api")),
661        ]);
662        assert_eq!(under(&same), [lab(true)]);
663        // A workspace the project does not name: every workspace shows.
664        let more = session(&[
665            ("labpc", "coding", Some("/home/labpc/api")),
666            ("labpc", "coding", Some("/home/labpc/web")),
667        ]);
668        assert_eq!(
669            under(&more),
670            [lab(true), SessionBadge::Workspace("~/api, ~/web".into())]
671        );
672        // Several machines: the project's shortens, the other keeps its name.
673        let spread = session(&[
674            ("local", "coding", Some("/home/local/web")),
675            ("labpc", "coding", Some("/home/labpc/api")),
676        ]);
677        let shown = under(&spread);
678        assert!(
679            matches!(shown[0], SessionBadge::Machine { ref name, in_project: false, .. } if name == "desk")
680        );
681        assert_eq!(shown[1], lab(true));
682        // Outside a project nothing shortens.
683        assert_eq!(
684            badges(&same),
685            [lab(false), SessionBadge::Workspace("~/api".into())]
686        );
687    }
