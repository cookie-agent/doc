1//! The `/permissions` table: a session's effective rules as rows grouped by
2//! action, and the rules a new-session draft holds until `session.create`.
3
4use cookie_agent_protocol::{
5    EffectivePermissionAction, PermissionAction, PermissionEffect, PermissionRule,
6    PermissionRuleSource, WildcardPattern,
7};
8
9/// The effects a rule can have, in the order a segmented control shows them.
10pub const EFFECTS: [PermissionEffect; 3] = [
11    PermissionEffect::Allow,
12    PermissionEffect::Ask,
13    PermissionEffect::Deny,
14];
15
16/// One row of the table: an action's default (`*`) or one of its patterns.
17#[derive(Clone, Debug, Eq, PartialEq)]
18pub struct PermissionRow {
19    pub action: PermissionAction,
20    pub resource: String,
21    pub effect: PermissionEffect,
22    pub source: PermissionRuleSource,
23}
24
25/// The rows of `permissions`: each action's `*` row, then its patterns.
26pub fn permission_rows(permissions: &[EffectivePermissionAction]) -> Vec<PermissionRow> {
27    permissions
28        .iter()
29        .flat_map(|action| {
30            std::iter::once(PermissionRow {
31                action: action.action.clone(),
32                resource: "*".into(),
33                effect: action.effect,
34                source: action.source,
35            })
36            .chain(action.patterns.iter().map(|rule| PermissionRow {
37                action: action.action.clone(),
38                resource: rule.resource.as_str().to_owned(),
39                effect: rule.effect,
40                source: rule.source,
41            }))
42        })
43        .collect()
44}
45
46/// The actions a rule can name, in table order: the ones the daemon reports
47/// for the session, or the built-in actions until it has.
48pub fn permission_actions(
49    permissions: Option<&[EffectivePermissionAction]>,
50) -> Vec<PermissionAction> {
51    permissions.map_or_else(
52        || PermissionAction::BUILT_IN.to_vec(),
53        |permissions| {
54            permissions
55                .iter()
56                .map(|action| action.action.clone())
57                .collect()
58        },
59    )
60}
61
62pub fn effect_label(effect: PermissionEffect) -> &'static str {
63    match effect {
64        PermissionEffect::Allow => "allow",
65        PermissionEffect::Ask => "ask",
66        PermissionEffect::Deny => "deny",
67    }
68}
69
70pub fn source_label(source: PermissionRuleSource) -> &'static str {
71    match source {
72        PermissionRuleSource::SessionOverlay => "session",
73        PermissionRuleSource::AgentDocument => "agent",
74        PermissionRuleSource::Default => "default",
75    }
76}
77
78/// Sets a draft rule, replacing any rule for the same action and pattern;
79/// rules stay in edit order.
80pub fn set_draft_rule(rules: &mut Vec<PermissionRule>, rule: PermissionRule) {
81    remove_draft_rule(rules, &rule.action, &rule.resource);
82    rules.push(rule);
83}
84
85/// Removes the draft rule for `action` and `resource`.
86pub fn remove_draft_rule(
87    rules: &mut Vec<PermissionRule>,
88    action: &PermissionAction,
89    resource: &WildcardPattern,
90) {
91    rules.retain(|rule| rule.action != *action || &rule.resource != resource);
92}

94#[cfg(test)]
95mod tests {
96    use cookie_agent_protocol::{
97        EffectivePermissionAction, PermissionAction, PermissionEffect, PermissionRule,
98        PermissionRuleSource, WildcardPattern,
99    };
100
101    use super::{PermissionRow, permission_rows, remove_draft_rule, set_draft_rule};
102
103    #[test]
104    fn each_action_gives_its_default_row_then_its_patterns() {
105        let permissions: Vec<EffectivePermissionAction> = serde_json::from_value(serde_json::json!([
106            {
107                "action": "bash",
108                "effect": "ask",
109                "source": "agent_document",
110                "patterns": [
111                    {"runtime": null, "resource": "git *", "effect": "allow", "source": "session_overlay"}
112                ]
113            },
114            {"action": "read", "effect": "allow", "source": "default", "patterns": []}
115        ]))
116        .expect("permissions");
117        let row = |action: &str, resource: &str, effect, source| PermissionRow {
118            action: PermissionAction::new(action).expect("action"),
119            resource: resource.into(),
120            effect,
121            source,
122        };
123        assert_eq!(
124            permission_rows(&permissions),
125            vec![
126                row(
127                    "bash",
128                    "*",
129                    PermissionEffect::Ask,
130                    PermissionRuleSource::AgentDocument
131                ),
132                row(
133                    "bash",
134                    "git *",
135                    PermissionEffect::Allow,
136                    PermissionRuleSource::SessionOverlay
137                ),
138                row(
139                    "read",
140                    "*",
141                    PermissionEffect::Allow,
142                    PermissionRuleSource::Default
143                ),
144            ]
145        );
146    }
147
148    #[test]
149    fn a_draft_holds_one_rule_per_action_and_pattern() {
150        let rule = |resource: &str, effect| PermissionRule {
151            runtime: None,
152            action: PermissionAction::new("bash").expect("action"),
153            resource: WildcardPattern::new(resource).expect("pattern"),
154            effect,
155        };
156        let mut rules = Vec::new();
157        set_draft_rule(&mut rules, rule("git *", PermissionEffect::Allow));
158        set_draft_rule(&mut rules, rule("rm *", PermissionEffect::Deny));
159        set_draft_rule(&mut rules, rule("git *", PermissionEffect::Ask));
160        assert_eq!(
161            rules,
162            vec![
163                rule("rm *", PermissionEffect::Deny),
164                rule("git *", PermissionEffect::Ask)
165            ]
166        );
167        remove_draft_rule(
168            &mut rules,
169            &PermissionAction::new("bash").expect("action"),
170            &WildcardPattern::new("rm *").expect("pattern"),
171        );
172        assert_eq!(rules, vec![rule("git *", PermissionEffect::Ask)]);
173    }
