16/// One attachment row: what the session records and the instance it refers
17/// to, when the daemon reports one.
18#[derive(Clone, Debug, Eq, PartialEq)]
19pub struct RuntimeRow {
20    pub attachment: RuntimeAttachment,
21    pub instance: Option<RuntimeInstanceInfo>,
22    /// What the runtime says its capability offers, once it has connected.
23    pub description: Option<String>,
24}

26impl RuntimeRow {
27    /// The MCP servers the row's instance hosts, from `mcp.server.list`, in
28    /// name order: those of its runtime at the instance's workspace (the
29    /// attachment's until the instance reports one).
30    pub fn mcp_servers(&self, servers: &[McpServerInfo]) -> Vec<McpServerInfo> {
31        let workspace = match &self.instance {
32            Some(instance) => instance.workspace.as_deref(),
33            None => self.attachment.workspace.as_deref(),
34        };
35        crate::mcp::instance_servers(servers, &self.attachment.runtime_id, workspace)
36    }
37}

39/// The description `listing` has for `capability` of `runtime_id`.
40fn capability_description(
41    listing: Option<&RuntimesListResult>,
42    runtime_id: &RuntimeId,
43    capability: &CapabilityName,
44) -> Option<String> {
45    listing?
46        .runtimes
47        .iter()
48        .find(|runtime| runtime.runtime_id == *runtime_id)?
49        .capabilities
50        .iter()
51        .find(|offered| offered.name == *capability)
52        .map(|offered| offered.description.clone())
53        .filter(|description| !description.is_empty())
54}

56/// A runtime a new-session draft attaches. Its instance is opened as soon as
57/// it joins the draft, so its status, and why it failed, show before the
58/// session exists.
59#[derive(Clone, Debug, Eq, PartialEq)]
60pub struct DraftRuntime {
61    /// Matches the answer of the open request to this entry.
62    pub id: u64,
63    pub request: RuntimeAttachRequest,
64    /// The instance the open request answered, or why it failed.
65    pub opened: Option<Result<RuntimeInstanceInfo, String>>,
66}
67
68impl DraftRuntime {
69    pub fn is_global(&self) -> bool {
70        self.request.runtime_id.as_str() == RuntimeId::GLOBAL
71    }
72
73    /// The opened instance's canonical workspace, else the one requested.
74    pub fn workspace(&self) -> Option<&str> {
75        match &self.opened {
76            Some(Ok(instance)) => instance.workspace.as_deref(),
77            _ => self.request.workspace.as_deref(),
78        }
79    }
80
81    /// Whether `other` names the same instance as this entry.
82    pub fn same_instance(&self, other: &RuntimeAttachRequest) -> bool {
83        self.request.runtime_id == other.runtime_id
84            && self.request.capability == other.capability
85            && (self.request.workspace == other.workspace
86                || self.workspace() == other.workspace.as_deref())
87    }
88}

90/// One thing the attach step offers: a capability of a runtime, or a runtime
91/// that has not said what it offers yet, which attaching connects and takes
92/// its only capability of.
93#[derive(Clone, Debug, Eq, PartialEq)]
94pub struct RuntimeChoice {
95    pub runtime_id: RuntimeId,
96    pub capability: Option<RuntimeCapabilityInfo>,
97    pub connected: bool,
98    /// Why the runtime last failed to connect.
99    pub message: Option<String>,
100}
101
102impl RuntimeChoice {
103    /// Whether attaching asks for a workspace: the capability takes one, or
104    /// the runtime has not said what it offers yet (then it is optional).
105    pub fn asks_workspace(&self) -> bool {
106        self.capability
107            .as_ref()
108            .is_none_or(|capability| capability.takes_workspace)
109    }
110
111    pub fn label(&self) -> String {
112        match &self.capability {
113            Some(capability) => format!("{}/{}", self.runtime_id, capability.name),
114            None => self.runtime_id.to_string(),
115        }
116    }
117
118    /// Case-insensitive substring match over the runtime ID and the
119    /// capability's name, kind, and description.
120    pub fn matches(&self, query: &str) -> bool {
121        let query = query.trim().to_lowercase();
122        if query.is_empty() {
123            return true;
124        }
125        let mut haystack = self.runtime_id.to_string();
126        if let Some(capability) = &self.capability {
127            haystack.push(' ');
128            haystack.push_str(capability.name.as_str());
129            haystack.push(' ');
130            haystack.push_str(capability.kind.as_str());
131            haystack.push(' ');
132            haystack.push_str(&capability.description);
133        }
134        haystack.to_lowercase().contains(&query)
135    }
136}

138/// A change to the machine registry the machines view offers.
139#[derive(Clone, Copy, Debug, Eq, PartialEq)]
140pub enum MachineAction {
141    Rename,
142    Move,
143    Unpin,
144    Merge,
145    Forget,
146    /// Opens a runtime's MCP servers, to manage them.
147    Servers,
148    /// Opens a runtime's ACP agents, to manage them.
149    Agents,
150    /// Connects a runtime the daemon starts (`runtimes.connect`).
151    Connect,
152    /// Disconnects a runtime the daemon starts (`runtimes.disconnect`).
153    Disconnect,
154}
155
156impl MachineAction {
157    pub const fn key(self) -> char {
158        match self {
159            Self::Rename => 'n',
160            Self::Move => 'v',
161            Self::Unpin => 'u',
162            Self::Merge => 'g',
163            Self::Forget => 'f',
164            Self::Servers => 's',
165            Self::Agents => 'a',
166            Self::Connect => 'c',
167            Self::Disconnect => 'd',
168        }
169    }
170
171    pub const fn label(self) -> &'static str {
172        match self {
173            Self::Rename => "rename",
174            Self::Move => "move to machine",
175            Self::Unpin => "unpin",
176            Self::Merge => "merge into",
177            Self::Forget => "forget",
178            Self::Servers => "MCP servers",
179            Self::Agents => "ACP agents",
180            Self::Connect => "connect",
181            Self::Disconnect => "disconnect",
182        }
183    }
184
185    /// Whether the action opens a panel of the runtime rather than changing
186    /// the machine registry.
187    pub const fn opens_panel(self) -> bool {
188        matches!(self, Self::Servers | Self::Agents)
189    }
190
191    /// Whether the action asks for a machine name first.
192    pub const fn asks_name(self) -> bool {
193        matches!(self, Self::Rename | Self::Move | Self::Merge)
194    }
195
196    /// The action of the key `key`.
197    pub fn of_key(key: char) -> Option<Self> {
198        [
199            Self::Rename,
200            Self::Move,
201            Self::Unpin,
202            Self::Merge,
203            Self::Forget,
204            Self::Servers,
205            Self::Agents,
206            Self::Connect,
207            Self::Disconnect,
208        ]
209        .into_iter()
210        .find(|action| action.key() == key)
211    }
212}

214/// An ACP agent declaration a runtime offers, as the machines views list it
215/// (`docs/site/specs/runtime-management.md` section 4.2).
216#[derive(Clone, Debug, Eq, PartialEq)]
217pub struct RuntimeAcpAgent {
218    /// The declaration's key, which is also the agent of that ID.
219    pub agent: AgentId,
220    /// A session that attaches the runtime can run it now.
221    pub available: bool,
222    /// What the agent of the declaration's own ID says it is; empty when
223    /// only templates run the declaration.
224    pub description: String,
225    /// The agents that run the declaration on the runtime, in ID order:
226    /// the one of its own ID and every template over it (`acp.agent`).
227    pub agents: Vec<AgentId>,
228}
229
230/// The client protocol minor from which `runtimes.agents.*` take `global`
231/// for the ACP agents every runtime offers (`[acp.agents]`).
232pub const SHARED_ACP_AGENTS_MINOR: u32 = 43;
233
234/// Whether a daemon speaking `version` declares ACP agents for every
235/// runtime.
236pub fn daemon_shares_acp_agents(version: Option<cookie_agent_protocol::SemVer>) -> bool {
237    version.is_some_and(|version| {
238        version
239            .shared_minor(cookie_agent_protocol::CLIENT_PROTOCOL_VERSION)
240            .is_some_and(|minor| minor >= SHARED_ACP_AGENTS_MINOR)
241    })
242}

244/// The ACP agent declarations `runtime_id` offers, by key, from the
245/// catalog's `agents`.
246pub fn runtime_acp_agents(
247    agents: &[AgentDescriptor],
248    runtime_id: &RuntimeId,
249) -> Vec<RuntimeAcpAgent> {
250    let mut declarations = std::collections::BTreeMap::<AgentId, RuntimeAcpAgent>::new();
251    for descriptor in agents {
252        for offer in descriptor
253            .runtimes
254            .iter()
255            .filter(|offer| offer.runtime_id == *runtime_id)
256        {
257            let declaration =
258                declarations
259                    .entry(offer.agent.clone())
260                    .or_insert_with(|| RuntimeAcpAgent {
261                        agent: offer.agent.clone(),
262                        available: false,
263                        description: String::new(),
264                        agents: Vec::new(),
265                    });
266            declaration.available |= offer.available;
267            if descriptor.id == offer.agent {
268                declaration.description.clone_from(&descriptor.description);
269            }
270            if !declaration.agents.contains(&descriptor.id) {
271                declaration.agents.push(descriptor.id.clone());
272            }
273        }
274    }
275    declarations
276        .into_values()
277        .map(|mut declaration| {
278            declaration.agents.sort();
279            declaration
280        })
281        .collect()
282}

284/// The form of an ACP agent declaration,
285/// `[runtimes.<id>.acp.agents.<agent>]`.
286#[derive(Clone, Debug, Eq, PartialEq)]
287pub struct AcpAgentFields {
288    /// The declaration's key, which is also its agent's ID.
289    pub agent: String,
290    /// The program and its arguments, as a JSON array.
291    pub command: String,
292    /// The environment, as a JSON object.
293    pub environment: String,
294    pub description: String,
295    pub enabled: bool,
296}
297
298impl AcpAgentFields {
299    /// A new declaration's form.
300    #[must_use]
301    pub fn add() -> Self {
302        Self {
303            agent: String::new(),
304            command: "[]".into(),
305            environment: "{}".into(),
306            description: String::new(),
307            enabled: true,
308        }
309    }
310
311    /// `declaration`'s form.
312    #[must_use]
313    pub fn edit(declaration: &RuntimeAgentDeclaration) -> Self {
314        Self {
315            agent: declaration.agent.to_string(),
316            command: serde_json::to_string(&declaration.definition.command)
317                .unwrap_or_else(|_| "[]".into()),
318            environment: serde_json::to_string(&declaration.definition.env)
319                .unwrap_or_else(|_| "{}".into()),
320            description: declaration.definition.description.clone(),
321            enabled: declaration.definition.enabled,
322        }
323    }
324
325    /// The key and definition the form describes, or what is wrong.
326    pub fn definition(&self) -> Result<(AgentId, AcpAgentDefinition), String> {
327        let agent = AgentId::new(self.agent.trim()).map_err(|_| {
328            "the agent ID is lowercase letters, digits, and single inner hyphens".to_owned()
329        })?;
330        let command: Vec<String> = serde_json::from_str(self.command.trim())
331            .map_err(|_| "the command is a JSON array of strings".to_owned())?;
332        if command.first().is_none_or(String::is_empty) {
333            return Err("the command names a program first".into());
334        }
335        let env = if self.environment.trim().is_empty() {
336            BTreeMap::new()
337        } else {
338            serde_json::from_str(self.environment.trim())
339                .map_err(|_| "the environment is a JSON object of strings".to_owned())?
340        };
341        Ok((
342            agent,
343            AcpAgentDefinition {
344                command,
345                env,
346                description: self.description.trim().to_owned(),
347                enabled: self.enabled,
348            },
349        ))
350    }
351}

384/// One row of the machines view: a machine, or a runtime under it.
385#[derive(Clone, Debug, Eq, PartialEq)]
386pub enum MachineEntry {
387    Machine(MachineName),
388    Runtime {
389        machine: MachineName,
390        runtime: RuntimeId,
391        /// Why it belongs to the machine while it is connected.
392        assigned_by: Option<MachineAssignment>,
393        pinned: bool,
394    },
395}
396
397impl MachineEntry {
398    /// The actions the entry offers.
399    pub fn actions(&self) -> Vec<MachineAction> {
400        match self {
401            Self::Machine(_) => vec![
402                MachineAction::Rename,
403                MachineAction::Merge,
404                MachineAction::Forget,
405            ],
406            Self::Runtime {
407                runtime, pinned, ..
408            } => {
409                let mut actions = vec![MachineAction::Move];
410                if *pinned {
411                    actions.push(MachineAction::Unpin);
412                }
413                actions.push(MachineAction::Servers);
414                // `global` hosts no ACP agents.
415                if runtime.as_str() != RuntimeId::GLOBAL {
416                    actions.push(MachineAction::Agents);
417                }
418                actions
419            }
420        }
421    }
422}

480/// The rows for a new-session draft: each entry with its live instance from
481/// `listing`, else the one its open request answered. Until the instance
482/// reports its attachment name and kind, the row shows the runtime ID and the
483/// capability.
484pub fn draft_runtime_rows(
485    entries: &[DraftRuntime],
486    listing: Option<&RuntimesListResult>,
487) -> Vec<RuntimeRow> {
488    entries
489        .iter()
490        .filter_map(|entry| {
491            let capability = match (&entry.request.capability, &entry.opened) {
492                (Some(capability), _) => capability.clone(),
493                (None, Some(Ok(instance))) => instance.capability.clone(),
494                (None, _) => cookie_agent_protocol::default_capability(&entry.request.runtime_id),
495            };
496            let workspace = entry.workspace().map(str::to_owned);
497            let live = listing.and_then(|listing| {
498                listing
499                    .instances
500                    .iter()
501                    .find(|instance| {
502                        instance.runtime_id == entry.request.runtime_id
503                            && instance.capability == capability
504                            && instance.workspace == workspace
505                    })
506                    .cloned()
507            });
508            let instance = live.or_else(|| match &entry.opened {
509                Some(Ok(instance)) => Some(instance.clone()),
510                Some(Err(message)) => Some(RuntimeInstanceInfo {
511                    runtime_id: entry.request.runtime_id.clone(),
512                    capability: capability.clone(),
513                    attachment_name: None,
514                    kind: None,
515                    workspace: workspace.clone(),
516                    platform: None,
517                    status: RuntimeInstanceStatus::Failed,
518                    message: Some(message.clone()),
519                }),
520                None => None,
521            });
522            Some(RuntimeRow {
523                description: capability_description(
524                    listing,
525                    &entry.request.runtime_id,
526                    &capability,
527                ),
528                attachment: RuntimeAttachment {
529                    name: match instance
530                        .as_ref()
531                        .and_then(|instance| instance.attachment_name.clone())
532                    {
533                        Some(name) => name,
534                        None => AttachmentName::new(entry.request.runtime_id.as_str()).ok()?,
535                    },
536                    kind: match instance.as_ref().and_then(|instance| instance.kind.clone()) {
537                        Some(kind) => kind,
538                        None => RuntimeKind::new(capability.as_str()).ok()?,
539                    },
540                    runtime_id: entry.request.runtime_id.clone(),
541                    capability,
542                    workspace,
543                },
544                instance,
545            })
546        })
547        .collect()
548}

562/// Every runtime's capabilities, in listing order, that match `query`; a
563/// runtime that has not said what it offers is one choice.
564pub fn runtime_choices(runtimes: &[RuntimeListing], query: &str) -> Vec<RuntimeChoice> {
565    runtimes
566        .iter()
567        .flat_map(|runtime| {
568            let choice = |capability| RuntimeChoice {
569                runtime_id: runtime.runtime_id.clone(),
570                capability,
571                connected: runtime.connected,
572                message: runtime.message.clone(),
573            };
574            if runtime.capabilities.is_empty() {
575                vec![choice(None)]
576            } else {
577                runtime
578                    .capabilities
579                    .iter()
580                    .cloned()
581                    .map(|capability| choice(Some(capability)))
582                    .collect()
583            }
584        })
585        .filter(|choice| choice.matches(query))
586        .collect()
587}

589/// The attach request `choice` with the typed `workspace` describes.
590pub fn attach_parameters(
591    choice: &RuntimeChoice,
592    workspace: &str,
593) -> Result<(RuntimeId, Option<CapabilityName>, Option<String>), String> {
594    let workspace = workspace.trim();
595    let workspace = match (&choice.capability, workspace.is_empty()) {
596        (Some(capability), true) if capability.takes_workspace => {
597            return Err(format!(
598                "capability `{}` of runtime `{}` needs a workspace path",
599                capability.name, choice.runtime_id
600            ));
601        }
602        (_, true) => None,
603        (_, false) => Some(workspace.to_owned()),
604    };
605    Ok((
606        choice.runtime_id.clone(),
607        choice
608            .capability
609            .as_ref()
610            .map(|capability| capability.name.clone()),
611        workspace,
612    ))
613}

645/// What a client may do about a runtime's connection: only a runtime the
646/// daemon starts from a template is connected and disconnected by hand.
647#[derive(Clone, Debug, Eq, PartialEq)]
648pub enum ConnectionControl {
649    /// Not connected: connecting tries that it starts.
650    Connect,
651    /// Connected and idle.
652    Disconnect,
653    /// Connected, and a run or background call uses it: why it cannot be
654    /// disconnected now.
655    Busy(String),
656}
657
658impl ConnectionControl {
659    /// The action the control stands for.
660    #[must_use]
661    pub const fn action(&self) -> MachineAction {
662        match self {
663            Self::Connect => MachineAction::Connect,
664            Self::Disconnect | Self::Busy(_) => MachineAction::Disconnect,
665        }
666    }
667}
668
669/// The connection control of `listing`; `None` for a runtime that is built
670/// in or dials in, which connects on its own.
671#[must_use]
672pub fn connection_control(listing: &RuntimeListing) -> Option<ConnectionControl> {
673    if listing.lifetime != RuntimeLifetime::Spawned {
674        return None;
675    }
676    if !listing.connected {
677        return Some(ConnectionControl::Connect);
678    }
679    let count = |count: u32, one: &str, many: &str| match count {
680        0 => None,
681        1 => Some(format!("1 {one}")),
682        count => Some(format!("{count} {many}")),
683    };
684    let uses = [
685        count(listing.active_runs, "active run", "active runs"),
686        count(
687            listing.background_calls,
688            "background call",
689            "background calls",
690        ),
691    ]
692    .into_iter()
693    .flatten()
694    .collect::<Vec<_>>();
695    Some(if uses.is_empty() {
696        ConnectionControl::Disconnect
697    } else {
698        ConnectionControl::Busy(format!("in use by {}", uses.join(" and ")))
699    })
700}

702/// Whether `action` (connect or disconnect) applies to runtime `runtime_id`
703/// of `runtimes` now; the error is the status line that says why not.
704pub fn connection_request(
705    runtimes: &[RuntimeListing],
706    runtime_id: &RuntimeId,
707    action: MachineAction,
708) -> Result<(), String> {
709    let listing = runtimes
710        .iter()
711        .find(|runtime| runtime.runtime_id == *runtime_id)
712        .ok_or_else(|| format!("runtime `{runtime_id}` is not a known runtime"))?;
713    match (connection_control(listing), action) {
714        (None, _) => Err(format!(
715            "runtime `{runtime_id}` connects on its own; only runtimes the daemon starts connect by hand"
716        )),
717        (Some(ConnectionControl::Connect), MachineAction::Connect)
718        | (Some(ConnectionControl::Disconnect), MachineAction::Disconnect) => Ok(()),
719        (Some(ConnectionControl::Busy(reason)), MachineAction::Disconnect) => Err(format!(
720            "runtime `{runtime_id}` is {reason}; disconnect it once they end"
721        )),
722        (Some(ConnectionControl::Connect), _) => {
723            Err(format!("runtime `{runtime_id}` is not connected"))
724        }
725        (Some(_), _) => Err(format!("runtime `{runtime_id}` is already connected")),
726    }
727}

740/// The machines view's rows: each machine, then its connected runtimes, the
741/// runtimes pinned to it that are not connected, and the offline runtimes
742/// it remembers (client protocol 1.10.0), placed by `last_seen`.
743pub fn machine_entries(listing: &MachinesListResult) -> Vec<MachineEntry> {
744    listing
745        .machines
746        .iter()
747        .flat_map(|machine| {
748            let mut entries = vec![MachineEntry::Machine(machine.name.clone())];
749            for runtime in &machine.connected {
750                entries.push(MachineEntry::Runtime {
751                    machine: machine.name.clone(),
752                    runtime: runtime.runtime_id.clone(),
753                    assigned_by: Some(runtime.assigned_by),
754                    pinned: machine.pinned.contains(&runtime.runtime_id),
755                });
756            }
757            for runtime in machine.pinned.iter().filter(|pinned| {
758                !machine
759                    .connected
760                    .iter()
761                    .any(|connected| connected.runtime_id == **pinned)
762            }) {
763                entries.push(MachineEntry::Runtime {
764                    machine: machine.name.clone(),
765                    runtime: runtime.clone(),
766                    assigned_by: None,
767                    pinned: true,
768                });
769            }
770            for runtime in machine
771                .remembered
772                .iter()
773                .filter(|remembered| !machine.pinned.contains(&remembered.runtime_id))
774            {
775                entries.push(MachineEntry::Runtime {
776                    machine: machine.name.clone(),
777                    runtime: runtime.runtime_id.clone(),
778                    assigned_by: Some(runtime.assigned_by),
779                    pinned: false,
780                });
781            }
782            entries
783        })
784        .collect()
785}

787/// `action` on `entry`: a request ready to send, `None` when it first needs
788/// a machine name, or why the entry does not offer it.
789pub fn machine_action_request(
790    entry: &MachineEntry,
791    action: MachineAction,
792) -> Result<Option<MachineRequest>, String> {
793    if !entry.actions().contains(&action) {
794        return Err(match (entry, action) {
795            (MachineEntry::Machine(_), _) => format!("select a runtime to {}", action.label()),
796            (MachineEntry::Runtime { .. }, MachineAction::Unpin) => {
797                "only a pinned runtime can be unpinned".to_owned()
798            }
799            (MachineEntry::Runtime { .. }, _) => {
800                format!("select a machine to {}", action.label())
801            }
802        });
803    }
804    if action.asks_name() {
805        return Ok(None);
806    }
807    Ok(Some(match (action, entry) {
808        (MachineAction::Forget, MachineEntry::Machine(machine)) => MachineRequest::Forget {
809            machine: machine.clone(),
810        },
811        (MachineAction::Unpin, MachineEntry::Runtime { runtime, .. }) => MachineRequest::Unassign {
812            runtime: runtime.clone(),
813        },
814        _ => return Err("nothing to do".into()),
815    }))
816}

818/// The request `action` on `entry` makes with the typed machine name.
819pub fn machine_name_request(
820    action: MachineAction,
821    entry: &MachineEntry,
822    typed: &str,
823) -> Result<MachineRequest, String> {
824    let typed = typed.trim();
825    let name = MachineName::new(typed).map_err(|_| {
826        format!(
827            "`{typed}` is not a machine name: a lowercase letter, then up to 31 lowercase \
828             letters, digits, `_`, or `-`"
829        )
830    })?;
831    Ok(match (action, entry) {
832        (MachineAction::Rename, MachineEntry::Machine(machine)) => MachineRequest::Rename {
833            machine: machine.clone(),
834            name,
835        },
836        (MachineAction::Merge, MachineEntry::Machine(machine)) => MachineRequest::Merge {
837            from: machine.clone(),
838            into: name,
839        },
840        (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => MachineRequest::Assign {
841            runtime: runtime.clone(),
842            machine: name,
843        },
844        _ => return Err("nothing to do".into()),
845    })
846}

848/// What a machine form asks for `action` on `entry`.
849pub fn machine_form_title(action: MachineAction, entry: &MachineEntry) -> String {
850    match (action, entry) {
851        (MachineAction::Rename, MachineEntry::Machine(machine)) => {
852            format!("New name for {machine}")
853        }
854        (MachineAction::Merge, MachineEntry::Machine(machine)) => {
855            format!("Merge {machine} into (its memories move too)")
856        }
857        (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => {
858            format!("Machine for runtime {runtime} (a new name creates it)")
859        }
860        _ => "Machine name".to_owned(),
861    }
862}

864impl MachineRequest {
865    /// What the status line says once the change is made.
866    pub fn done(&self) -> String {
867        match self {
868            Self::Rename { machine, name } => format!("renamed machine {machine} to {name}"),
869            Self::Assign { runtime, machine } => {
870                format!("pinned runtime {runtime} to machine {machine}")
871            }
872            Self::Unassign { runtime } => format!("unpinned runtime {runtime}"),
873            Self::Merge { from, into } => format!("merged machine {from} into {into}"),
874            Self::Forget { machine } => format!("forgot machine {machine}"),
875            Self::ForgetRuntime { runtime } => format!("forgot where runtime {runtime} ran"),
876        }
877    }
878}
