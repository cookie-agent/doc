12/// One attachment row: what the session records and the instance it refers
13/// to, when the daemon reports one.
14#[derive(Clone, Debug, Eq, PartialEq)]
15pub struct RuntimeRow {
16    pub attachment: RuntimeAttachment,
17    pub instance: Option<RuntimeInstanceInfo>,
18    /// What the runtime says its capability offers, once it has connected.
19    pub description: Option<String>,
20}

22/// The description `listing` has for `capability` of `runtime_id`.
23fn capability_description(
24    listing: Option<&RuntimesListResult>,
25    runtime_id: &RuntimeId,
26    capability: &CapabilityName,
27) -> Option<String> {
28    listing?
29        .runtimes
30        .iter()
31        .find(|runtime| runtime.runtime_id == *runtime_id)?
32        .capabilities
33        .iter()
34        .find(|offered| offered.name == *capability)
35        .map(|offered| offered.description.clone())
36        .filter(|description| !description.is_empty())
37}

39/// A runtime a new-session draft attaches. Its instance is opened as soon as
40/// it joins the draft, so its status, and why it failed, show before the
41/// session exists.
42#[derive(Clone, Debug, Eq, PartialEq)]
43pub struct DraftRuntime {
44    /// Matches the answer of the open request to this entry.
45    pub id: u64,
46    pub request: RuntimeAttachRequest,
47    /// The instance the open request answered, or why it failed.
48    pub opened: Option<Result<RuntimeInstanceInfo, String>>,
49}
50
51impl DraftRuntime {
52    pub fn is_global(&self) -> bool {
53        self.request.runtime_id.as_str() == RuntimeId::GLOBAL
54    }
55
56    /// The opened instance's canonical workspace, else the one requested.
57    pub fn workspace(&self) -> Option<&str> {
58        match &self.opened {
59            Some(Ok(instance)) => instance.workspace.as_deref(),
60            _ => self.request.workspace.as_deref(),
61        }
62    }
63
64    /// Whether `other` names the same instance as this entry.
65    pub fn same_instance(&self, other: &RuntimeAttachRequest) -> bool {
66        self.request.runtime_id == other.runtime_id
67            && self.request.capability == other.capability
68            && (self.request.workspace == other.workspace
69                || self.workspace() == other.workspace.as_deref())
70    }
71}

73/// One thing the attach step offers: a capability of a runtime, or a runtime
74/// that has not said what it offers yet, which attaching connects and takes
75/// its only capability of.
76#[derive(Clone, Debug, Eq, PartialEq)]
77pub struct RuntimeChoice {
78    pub runtime_id: RuntimeId,
79    pub capability: Option<RuntimeCapabilityInfo>,
80    pub connected: bool,
81    /// Why the runtime last failed to connect.
82    pub message: Option<String>,
83}
84
85impl RuntimeChoice {
86    /// Whether attaching asks for a workspace: the capability takes one, or
87    /// the runtime has not said what it offers yet (then it is optional).
88    pub fn asks_workspace(&self) -> bool {
89        self.capability
90            .as_ref()
91            .is_none_or(|capability| capability.takes_workspace)
92    }
93
94    pub fn label(&self) -> String {
95        match &self.capability {
96            Some(capability) => format!("{}/{}", self.runtime_id, capability.name),
97            None => self.runtime_id.to_string(),
98        }
99    }
100
101    /// Case-insensitive substring match over the runtime ID and the
102    /// capability's name, kind, and description.
103    pub fn matches(&self, query: &str) -> bool {
104        let query = query.trim().to_lowercase();
105        if query.is_empty() {
106            return true;
107        }
108        let mut haystack = self.runtime_id.to_string();
109        if let Some(capability) = &self.capability {
110            haystack.push(' ');
111            haystack.push_str(capability.name.as_str());
112            haystack.push(' ');
113            haystack.push_str(capability.kind.as_str());
114            haystack.push(' ');
115            haystack.push_str(&capability.description);
116        }
117        haystack.to_lowercase().contains(&query)
118    }
119}

121/// A change to the machine registry the machines view offers.
122#[derive(Clone, Copy, Debug, Eq, PartialEq)]
123pub enum MachineAction {
124    Rename,
125    Move,
126    Unpin,
127    Merge,
128    Forget,
129}
130
131impl MachineAction {
132    pub const fn key(self) -> char {
133        match self {
134            Self::Rename => 'n',
135            Self::Move => 'v',
136            Self::Unpin => 'u',
137            Self::Merge => 'g',
138            Self::Forget => 'f',
139        }
140    }
141
142    pub const fn label(self) -> &'static str {
143        match self {
144            Self::Rename => "rename",
145            Self::Move => "move to machine",
146            Self::Unpin => "unpin",
147            Self::Merge => "merge into",
148            Self::Forget => "forget",
149        }
150    }
151
152    /// Whether the action asks for a machine name first.
153    pub const fn asks_name(self) -> bool {
154        matches!(self, Self::Rename | Self::Move | Self::Merge)
155    }
156
157    /// The action of the key `key`.
158    pub fn of_key(key: char) -> Option<Self> {
159        [
160            Self::Rename,
161            Self::Move,
162            Self::Unpin,
163            Self::Merge,
164            Self::Forget,
165        ]
166        .into_iter()
167        .find(|action| action.key() == key)
168    }
169}

171/// One row of the machines view: a machine, or a runtime under it.
172#[derive(Clone, Debug, Eq, PartialEq)]
173pub enum MachineEntry {
174    Machine(MachineName),
175    Runtime {
176        machine: MachineName,
177        runtime: RuntimeId,
178        /// Why it belongs to the machine while it is connected.
179        assigned_by: Option<MachineAssignment>,
180        pinned: bool,
181    },
182}
183
184impl MachineEntry {
185    /// The actions the entry offers.
186    pub fn actions(&self) -> Vec<MachineAction> {
187        match self {
188            Self::Machine(_) => vec![
189                MachineAction::Rename,
190                MachineAction::Merge,
191                MachineAction::Forget,
192            ],
193            Self::Runtime { pinned: true, .. } => vec![MachineAction::Move, MachineAction::Unpin],
194            Self::Runtime { pinned: false, .. } => vec![MachineAction::Move],
195        }
196    }
197}

199/// A machine change as the daemon takes it.
200#[derive(Clone, Debug, Eq, PartialEq)]
201pub enum MachineRequest {

208        machine: MachineName,
209    },

211        runtime: RuntimeId,

222/// The rows for `attachments`, each matched to its instance in `listing`.
223pub fn runtime_rows(
224    attachments: &[RuntimeAttachment],
225    listing: Option<&RuntimesListResult>,
226) -> Vec<RuntimeRow> {
227    attachments
228        .iter()
229        .map(|attachment| RuntimeRow {
230            description: capability_description(
231                listing,
232                &attachment.runtime_id,
233                &attachment.capability,
234            ),
235            attachment: attachment.clone(),
236            instance: listing.and_then(|listing| {
237                listing
238                    .instances
239                    .iter()
240                    .find(|instance| {
241                        instance.runtime_id == attachment.runtime_id
242                            && instance.capability == attachment.capability
243                            && instance.workspace == attachment.workspace
244                    })
245                    .cloned()
246            }),
247        })
248        .collect()
249}

251/// The rows for a new-session draft: each entry with its live instance from
252/// `listing`, else the one its open request answered. Until the instance
253/// reports its attachment name and kind, the row shows the runtime ID and the
254/// capability.
255pub fn draft_runtime_rows(
256    entries: &[DraftRuntime],
257    listing: Option<&RuntimesListResult>,
258) -> Vec<RuntimeRow> {
259    entries
260        .iter()
261        .filter_map(|entry| {
262            let capability = match (&entry.request.capability, &entry.opened) {
263                (Some(capability), _) => capability.clone(),
264                (None, Some(Ok(instance))) => instance.capability.clone(),
265                (None, _) => cookie_agent_protocol::default_capability(&entry.request.runtime_id),
266            };
267            let workspace = entry.workspace().map(str::to_owned);
268            let live = listing.and_then(|listing| {
269                listing
270                    .instances
271                    .iter()
272                    .find(|instance| {
273                        instance.runtime_id == entry.request.runtime_id
274                            && instance.capability == capability
275                            && instance.workspace == workspace
276                    })
277                    .cloned()
278            });
279            let instance = live.or_else(|| match &entry.opened {
280                Some(Ok(instance)) => Some(instance.clone()),
281                Some(Err(message)) => Some(RuntimeInstanceInfo {
282                    runtime_id: entry.request.runtime_id.clone(),
283                    capability: capability.clone(),
284                    attachment_name: None,
285                    kind: None,
286                    workspace: workspace.clone(),
287                    platform: None,
288                    status: RuntimeInstanceStatus::Failed,
289                    message: Some(message.clone()),
290                }),
291                None => None,
292            });
293            Some(RuntimeRow {
294                description: capability_description(
295                    listing,
296                    &entry.request.runtime_id,
297                    &capability,
298                ),
299                attachment: RuntimeAttachment {
300                    name: match instance
301                        .as_ref()
302                        .and_then(|instance| instance.attachment_name.clone())
303                    {
304                        Some(name) => name,
305                        None => AttachmentName::new(entry.request.runtime_id.as_str()).ok()?,
306                    },
307                    kind: match instance.as_ref().and_then(|instance| instance.kind.clone()) {
308                        Some(kind) => kind,
309                        None => RuntimeKind::new(capability.as_str()).ok()?,
310                    },
311                    runtime_id: entry.request.runtime_id.clone(),
312                    capability,
313                    workspace,
314                },
315                instance,
316            })
317        })
318        .collect()
319}

321pub const fn status_label(status: Option<RuntimeInstanceStatus>) -> &'static str {

323        Some(RuntimeInstanceStatus::Starting) => "starting",

333/// Every runtime's capabilities, in listing order, that match `query`; a
334/// runtime that has not said what it offers is one choice.
335pub fn runtime_choices(runtimes: &[RuntimeListing], query: &str) -> Vec<RuntimeChoice> {
336    runtimes
337        .iter()
338        .flat_map(|runtime| {
339            let choice = |capability| RuntimeChoice {
340                runtime_id: runtime.runtime_id.clone(),
341                capability,
342                connected: runtime.connected,
343                message: runtime.message.clone(),
344            };
345            if runtime.capabilities.is_empty() {
346                vec![choice(None)]
347            } else {
348                runtime
349                    .capabilities
350                    .iter()
351                    .cloned()
352                    .map(|capability| choice(Some(capability)))
353                    .collect()
354            }
355        })
356        .filter(|choice| choice.matches(query))
357        .collect()
358}

360/// The attach request `choice` with the typed `workspace` describes.
361pub fn attach_parameters(
362    choice: &RuntimeChoice,
363    workspace: &str,
364) -> Result<(RuntimeId, Option<CapabilityName>, Option<String>), String> {
365    let workspace = workspace.trim();
366    let workspace = match (&choice.capability, workspace.is_empty()) {
367        (Some(capability), true) if capability.takes_workspace => {
368            return Err(format!(
369                "capability `{}` of runtime `{}` needs a workspace path",
370                capability.name, choice.runtime_id
371            ));
372        }
373        (_, true) => None,
374        (_, false) => Some(workspace.to_owned()),
375    };
376    Ok((
377        choice.runtime_id.clone(),
378        choice
379            .capability
380            .as_ref()
381            .map(|capability| capability.name.clone()),
382        workspace,
383    ))
384}

386/// The `runtimes.reconnect` request that retries `row`'s instance: its
387/// workspace goes along exactly when it has one.
388pub fn reconnect_request(
389    runtimes: &[RuntimeListing],
390    row: &RuntimeRow,
391) -> Result<(RuntimeId, CapabilityName, Option<String>), String> {
392    if !runtimes
393        .iter()
394        .any(|runtime| runtime.runtime_id == row.attachment.runtime_id)
395    {
396        return Err(format!(
397            "runtime `{}` is not a known runtime",
398            row.attachment.runtime_id
399        ));
400    }
401    Ok((
402        row.attachment.runtime_id.clone(),
403        row.attachment.capability.clone(),
404        row.attachment.workspace.clone(),
405    ))
406}

416/// The machines view's rows: each machine, then its connected runtimes and
417/// the runtimes pinned to it that are not connected.
418pub fn machine_entries(listing: &MachinesListResult) -> Vec<MachineEntry> {
419    listing
420        .machines
421        .iter()
422        .flat_map(|machine| {
423            let mut entries = vec![MachineEntry::Machine(machine.name.clone())];
424            for runtime in &machine.connected {
425                entries.push(MachineEntry::Runtime {
426                    machine: machine.name.clone(),
427                    runtime: runtime.runtime_id.clone(),
428                    assigned_by: Some(runtime.assigned_by),
429                    pinned: machine.pinned.contains(&runtime.runtime_id),
430                });
431            }
432            for runtime in machine.pinned.iter().filter(|pinned| {
433                !machine
434                    .connected
435                    .iter()
436                    .any(|connected| connected.runtime_id == **pinned)
437            }) {
438                entries.push(MachineEntry::Runtime {
439                    machine: machine.name.clone(),
440                    runtime: runtime.clone(),
441                    assigned_by: None,
442                    pinned: true,
443                });
444            }
445            entries
446        })
447        .collect()
448}

450/// `action` on `entry`: a request ready to send, `None` when it first needs
451/// a machine name, or why the entry does not offer it.
452pub fn machine_action_request(
453    entry: &MachineEntry,
454    action: MachineAction,
455) -> Result<Option<MachineRequest>, String> {
456    if !entry.actions().contains(&action) {
457        return Err(match (entry, action) {
458            (MachineEntry::Machine(_), _) => format!("select a runtime to {}", action.label()),
459            (MachineEntry::Runtime { .. }, MachineAction::Unpin) => {
460                "only a pinned runtime can be unpinned".to_owned()
461            }
462            (MachineEntry::Runtime { .. }, _) => {
463                format!("select a machine to {}", action.label())
464            }
465        });
466    }
467    if action.asks_name() {
468        return Ok(None);
469    }
470    Ok(Some(match (action, entry) {
471        (MachineAction::Forget, MachineEntry::Machine(machine)) => MachineRequest::Forget {
472            machine: machine.clone(),
473        },
474        (MachineAction::Unpin, MachineEntry::Runtime { runtime, .. }) => MachineRequest::Unassign {
475            runtime: runtime.clone(),
476        },
477        _ => return Err("nothing to do".into()),
478    }))
479}

481/// The request `action` on `entry` makes with the typed machine name.
482pub fn machine_name_request(
483    action: MachineAction,
484    entry: &MachineEntry,
485    typed: &str,
486) -> Result<MachineRequest, String> {
487    let typed = typed.trim();
488    let name = MachineName::new(typed).map_err(|_| {
489        format!(
490            "`{typed}` is not a machine name: a lowercase letter, then up to 31 lowercase \
491             letters, digits, `_`, or `-`"
492        )
493    })?;
494    Ok(match (action, entry) {
495        (MachineAction::Rename, MachineEntry::Machine(machine)) => MachineRequest::Rename {
496            machine: machine.clone(),
497            name,
498        },
499        (MachineAction::Merge, MachineEntry::Machine(machine)) => MachineRequest::Merge {
500            from: machine.clone(),
501            into: name,
502        },
503        (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => MachineRequest::Assign {
504            runtime: runtime.clone(),
505            machine: name,
506        },
507        _ => return Err("nothing to do".into()),
508    })
509}

511/// What a machine form asks for `action` on `entry`.
512pub fn machine_form_title(action: MachineAction, entry: &MachineEntry) -> String {
513    match (action, entry) {
514        (MachineAction::Rename, MachineEntry::Machine(machine)) => {
515            format!("New name for {machine}")
516        }
517        (MachineAction::Merge, MachineEntry::Machine(machine)) => {
518            format!("Merge {machine} into (its memories move too)")
519        }
520        (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => {
521            format!("Machine for runtime {runtime} (a new name creates it)")
522        }
523        _ => "Machine name".to_owned(),
524    }
525}

527impl MachineRequest {
528    /// What the status line says once the change is made.
529    pub fn done(&self) -> String {
530        match self {
531            Self::Rename { machine, name } => format!("renamed machine {machine} to {name}"),
532            Self::Assign { runtime, machine } => {
533                format!("pinned runtime {runtime} to machine {machine}")
534            }
535            Self::Unassign { runtime } => format!("unpinned runtime {runtime}"),
536            Self::Merge { from, into } => format!("merged machine {from} into {into}"),
537            Self::Forget { machine } => format!("forgot machine {machine}"),
538        }
539    }
