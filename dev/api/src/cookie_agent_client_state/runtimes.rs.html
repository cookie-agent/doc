14/// One attachment row: what the session records and the instance it refers
15/// to, when the daemon reports one.
16#[derive(Clone, Debug, Eq, PartialEq)]
17pub struct RuntimeRow {
18    pub attachment: RuntimeAttachment,
19    pub instance: Option<RuntimeInstanceInfo>,
20    /// What the runtime says its capability offers, once it has connected.
21    pub description: Option<String>,
22}
23
24impl RuntimeRow {
25    /// The MCP servers the row's instance hosts, from `mcp.server.list`, in
26    /// name order: those of its runtime at the instance's workspace (the
27    /// attachment's until the instance reports one).
28    pub fn mcp_servers(&self, servers: &[McpServerInfo]) -> Vec<McpServerInfo> {
29        let workspace = match &self.instance {
30            Some(instance) => instance.workspace.as_deref(),
31            None => self.attachment.workspace.as_deref(),
32        };
33        crate::mcp::instance_servers(servers, &self.attachment.runtime_id, workspace)
34    }
35}

37/// The description `listing` has for `capability` of `runtime_id`.
38fn capability_description(
39    listing: Option<&RuntimesListResult>,
40    runtime_id: &RuntimeId,
41    capability: &CapabilityName,
42) -> Option<String> {
43    listing?
44        .runtimes
45        .iter()
46        .find(|runtime| runtime.runtime_id == *runtime_id)?
47        .capabilities
48        .iter()
49        .find(|offered| offered.name == *capability)
50        .map(|offered| offered.description.clone())
51        .filter(|description| !description.is_empty())
52}

54/// A runtime a new-session draft attaches. Its instance is opened as soon as
55/// it joins the draft, so its status, and why it failed, show before the
56/// session exists.
57#[derive(Clone, Debug, Eq, PartialEq)]
58pub struct DraftRuntime {
59    /// Matches the answer of the open request to this entry.
60    pub id: u64,
61    pub request: RuntimeAttachRequest,
62    /// The instance the open request answered, or why it failed.
63    pub opened: Option<Result<RuntimeInstanceInfo, String>>,
64}
65
66impl DraftRuntime {
67    pub fn is_global(&self) -> bool {
68        self.request.runtime_id.as_str() == RuntimeId::GLOBAL
69    }
70
71    /// The opened instance's canonical workspace, else the one requested.
72    pub fn workspace(&self) -> Option<&str> {
73        match &self.opened {
74            Some(Ok(instance)) => instance.workspace.as_deref(),
75            _ => self.request.workspace.as_deref(),
76        }
77    }
78
79    /// Whether `other` names the same instance as this entry.
80    pub fn same_instance(&self, other: &RuntimeAttachRequest) -> bool {
81        self.request.runtime_id == other.runtime_id
82            && self.request.capability == other.capability
83            && (self.request.workspace == other.workspace
84                || self.workspace() == other.workspace.as_deref())
85    }
86}

88/// One thing the attach step offers: a capability of a runtime, or a runtime
89/// that has not said what it offers yet, which attaching connects and takes
90/// its only capability of.
91#[derive(Clone, Debug, Eq, PartialEq)]
92pub struct RuntimeChoice {
93    pub runtime_id: RuntimeId,
94    pub capability: Option<RuntimeCapabilityInfo>,
95    pub connected: bool,
96    /// Why the runtime last failed to connect.
97    pub message: Option<String>,
98}
99
100impl RuntimeChoice {
101    /// Whether attaching asks for a workspace: the capability takes one, or
102    /// the runtime has not said what it offers yet (then it is optional).
103    pub fn asks_workspace(&self) -> bool {
104        self.capability
105            .as_ref()
106            .is_none_or(|capability| capability.takes_workspace)
107    }
108
109    pub fn label(&self) -> String {
110        match &self.capability {
111            Some(capability) => format!("{}/{}", self.runtime_id, capability.name),
112            None => self.runtime_id.to_string(),
113        }
114    }
115
116    /// Case-insensitive substring match over the runtime ID and the
117    /// capability's name, kind, and description.
118    pub fn matches(&self, query: &str) -> bool {
119        let query = query.trim().to_lowercase();
120        if query.is_empty() {
121            return true;
122        }
123        let mut haystack = self.runtime_id.to_string();
124        if let Some(capability) = &self.capability {
125            haystack.push(' ');
126            haystack.push_str(capability.name.as_str());
127            haystack.push(' ');
128            haystack.push_str(capability.kind.as_str());
129            haystack.push(' ');
130            haystack.push_str(&capability.description);
131        }
132        haystack.to_lowercase().contains(&query)
133    }
134}

136/// A change to the machine registry the machines view offers.
137#[derive(Clone, Copy, Debug, Eq, PartialEq)]
138pub enum MachineAction {
139    Rename,
140    Move,
141    Unpin,
142    Merge,
143    Forget,
144}
145
146impl MachineAction {
147    pub const fn key(self) -> char {
148        match self {
149            Self::Rename => 'n',
150            Self::Move => 'v',
151            Self::Unpin => 'u',
152            Self::Merge => 'g',
153            Self::Forget => 'f',
154        }
155    }
156
157    pub const fn label(self) -> &'static str {
158        match self {
159            Self::Rename => "rename",
160            Self::Move => "move to machine",
161            Self::Unpin => "unpin",
162            Self::Merge => "merge into",
163            Self::Forget => "forget",
164        }
165    }
166
167    /// Whether the action asks for a machine name first.
168    pub const fn asks_name(self) -> bool {
169        matches!(self, Self::Rename | Self::Move | Self::Merge)
170    }
171
172    /// The action of the key `key`.
173    pub fn of_key(key: char) -> Option<Self> {
174        [
175            Self::Rename,
176            Self::Move,
177            Self::Unpin,
178            Self::Merge,
179            Self::Forget,
180        ]
181        .into_iter()
182        .find(|action| action.key() == key)
183    }
184}

186/// An ACP agent declaration a runtime offers, as the machines views list it
187/// (`docs/site/specs/runtime-management.md` section 4.2).
188#[derive(Clone, Debug, Eq, PartialEq)]
189pub struct RuntimeAcpAgent {
190    /// The declaration's key, which is also the agent of that ID.
191    pub agent: AgentId,
192    /// A session that attaches the runtime can run it now.
193    pub available: bool,
194    /// What the agent of the declaration's own ID says it is; empty when
195    /// only templates run the declaration.
196    pub description: String,
197    /// The agents that run the declaration on the runtime, in ID order:
198    /// the one of its own ID and every template over it (`acp.agent`).
199    pub agents: Vec<AgentId>,
200}
201
202/// The ACP agent declarations `runtime_id` offers, by key, from the
203/// catalog's `agents`.
204pub fn runtime_acp_agents(
205    agents: &[AgentDescriptor],
206    runtime_id: &RuntimeId,
207) -> Vec<RuntimeAcpAgent> {
208    let mut declarations = std::collections::BTreeMap::<AgentId, RuntimeAcpAgent>::new();
209    for descriptor in agents {
210        for offer in descriptor
211            .runtimes
212            .iter()
213            .filter(|offer| offer.runtime_id == *runtime_id)
214        {
215            let declaration =
216                declarations
217                    .entry(offer.agent.clone())
218                    .or_insert_with(|| RuntimeAcpAgent {
219                        agent: offer.agent.clone(),
220                        available: false,
221                        description: String::new(),
222                        agents: Vec::new(),
223                    });
224            declaration.available |= offer.available;
225            if descriptor.id == offer.agent {
226                declaration.description.clone_from(&descriptor.description);
227            }
228            if !declaration.agents.contains(&descriptor.id) {
229                declaration.agents.push(descriptor.id.clone());
230            }
231        }
232    }
233    declarations
234        .into_values()
235        .map(|mut declaration| {
236            declaration.agents.sort();
237            declaration
238        })
239        .collect()
240}

242/// One row of the machines view: a machine, or a runtime under it.
243#[derive(Clone, Debug, Eq, PartialEq)]
244pub enum MachineEntry {
245    Machine(MachineName),
246    Runtime {
247        machine: MachineName,
248        runtime: RuntimeId,
249        /// Why it belongs to the machine while it is connected.
250        assigned_by: Option<MachineAssignment>,
251        pinned: bool,
252    },
253}
254
255impl MachineEntry {
256    /// The actions the entry offers.
257    pub fn actions(&self) -> Vec<MachineAction> {
258        match self {
259            Self::Machine(_) => vec![
260                MachineAction::Rename,
261                MachineAction::Merge,
262                MachineAction::Forget,
263            ],
264            Self::Runtime { pinned: true, .. } => vec![MachineAction::Move, MachineAction::Unpin],
265            Self::Runtime { pinned: false, .. } => vec![MachineAction::Move],
266        }
267    }
268}

297/// The rows for `attachments`, each matched to its instance in `listing`.
298pub fn runtime_rows(
299    attachments: &[RuntimeAttachment],
300    listing: Option<&RuntimesListResult>,
301) -> Vec<RuntimeRow> {
302    attachments
303        .iter()
304        .map(|attachment| RuntimeRow {
305            description: capability_description(
306                listing,
307                &attachment.runtime_id,
308                &attachment.capability,
309            ),
310            attachment: attachment.clone(),
311            instance: listing.and_then(|listing| {
312                listing
313                    .instances
314                    .iter()
315                    .find(|instance| {
316                        instance.runtime_id == attachment.runtime_id
317                            && instance.capability == attachment.capability
318                            && instance.workspace == attachment.workspace
319                    })
320                    .cloned()
321            }),
322        })
323        .collect()
324}

326/// The rows for a new-session draft: each entry with its live instance from
327/// `listing`, else the one its open request answered. Until the instance
328/// reports its attachment name and kind, the row shows the runtime ID and the
329/// capability.
330pub fn draft_runtime_rows(
331    entries: &[DraftRuntime],
332    listing: Option<&RuntimesListResult>,
333) -> Vec<RuntimeRow> {
334    entries
335        .iter()
336        .filter_map(|entry| {
337            let capability = match (&entry.request.capability, &entry.opened) {
338                (Some(capability), _) => capability.clone(),
339                (None, Some(Ok(instance))) => instance.capability.clone(),
340                (None, _) => cookie_agent_protocol::default_capability(&entry.request.runtime_id),
341            };
342            let workspace = entry.workspace().map(str::to_owned);
343            let live = listing.and_then(|listing| {
344                listing
345                    .instances
346                    .iter()
347                    .find(|instance| {
348                        instance.runtime_id == entry.request.runtime_id
349                            && instance.capability == capability
350                            && instance.workspace == workspace
351                    })
352                    .cloned()
353            });
354            let instance = live.or_else(|| match &entry.opened {
355                Some(Ok(instance)) => Some(instance.clone()),
356                Some(Err(message)) => Some(RuntimeInstanceInfo {
357                    runtime_id: entry.request.runtime_id.clone(),
358                    capability: capability.clone(),
359                    attachment_name: None,
360                    kind: None,
361                    workspace: workspace.clone(),
362                    platform: None,
363                    status: RuntimeInstanceStatus::Failed,
364                    message: Some(message.clone()),
365                }),
366                None => None,
367            });
368            Some(RuntimeRow {
369                description: capability_description(
370                    listing,
371                    &entry.request.runtime_id,
372                    &capability,
373                ),
374                attachment: RuntimeAttachment {
375                    name: match instance
376                        .as_ref()
377                        .and_then(|instance| instance.attachment_name.clone())
378                    {
379                        Some(name) => name,
380                        None => AttachmentName::new(entry.request.runtime_id.as_str()).ok()?,
381                    },
382                    kind: match instance.as_ref().and_then(|instance| instance.kind.clone()) {
383                        Some(kind) => kind,
384                        None => RuntimeKind::new(capability.as_str()).ok()?,
385                    },
386                    runtime_id: entry.request.runtime_id.clone(),
387                    capability,
388                    workspace,
389                },
390                instance,
391            })
392        })
393        .collect()
394}

408/// Every runtime's capabilities, in listing order, that match `query`; a
409/// runtime that has not said what it offers is one choice.
410pub fn runtime_choices(runtimes: &[RuntimeListing], query: &str) -> Vec<RuntimeChoice> {
411    runtimes
412        .iter()
413        .flat_map(|runtime| {
414            let choice = |capability| RuntimeChoice {
415                runtime_id: runtime.runtime_id.clone(),
416                capability,
417                connected: runtime.connected,
418                message: runtime.message.clone(),
419            };
420            if runtime.capabilities.is_empty() {
421                vec![choice(None)]
422            } else {
423                runtime
424                    .capabilities
425                    .iter()
426                    .cloned()
427                    .map(|capability| choice(Some(capability)))
428                    .collect()
429            }
430        })
431        .filter(|choice| choice.matches(query))
432        .collect()
433}

435/// The attach request `choice` with the typed `workspace` describes.
436pub fn attach_parameters(
437    choice: &RuntimeChoice,
438    workspace: &str,
439) -> Result<(RuntimeId, Option<CapabilityName>, Option<String>), String> {
440    let workspace = workspace.trim();
441    let workspace = match (&choice.capability, workspace.is_empty()) {
442        (Some(capability), true) if capability.takes_workspace => {
443            return Err(format!(
444                "capability `{}` of runtime `{}` needs a workspace path",
445                capability.name, choice.runtime_id
446            ));
447        }
448        (_, true) => None,
449        (_, false) => Some(workspace.to_owned()),
450    };
451    Ok((
452        choice.runtime_id.clone(),
453        choice
454            .capability
455            .as_ref()
456            .map(|capability| capability.name.clone()),
457        workspace,
458    ))
459}

461/// The `runtimes.reconnect` request that retries `row`'s instance: its
462/// workspace goes along exactly when it has one.
463pub fn reconnect_request(
464    runtimes: &[RuntimeListing],
465    row: &RuntimeRow,
466) -> Result<(RuntimeId, CapabilityName, Option<String>), String> {
467    if !runtimes
468        .iter()
469        .any(|runtime| runtime.runtime_id == row.attachment.runtime_id)
470    {
471        return Err(format!(
472            "runtime `{}` is not a known runtime",
473            row.attachment.runtime_id
474        ));
475    }
476    Ok((
477        row.attachment.runtime_id.clone(),
478        row.attachment.capability.clone(),
479        row.attachment.workspace.clone(),
480    ))
481}

491/// The machines view's rows: each machine, then its connected runtimes, the
492/// runtimes pinned to it that are not connected, and the offline runtimes
493/// it remembers (client protocol 1.10.0), placed by `last_seen`.
494pub fn machine_entries(listing: &MachinesListResult) -> Vec<MachineEntry> {
495    listing
496        .machines
497        .iter()
498        .flat_map(|machine| {
499            let mut entries = vec![MachineEntry::Machine(machine.name.clone())];
500            for runtime in &machine.connected {
501                entries.push(MachineEntry::Runtime {
502                    machine: machine.name.clone(),
503                    runtime: runtime.runtime_id.clone(),
504                    assigned_by: Some(runtime.assigned_by),
505                    pinned: machine.pinned.contains(&runtime.runtime_id),
506                });
507            }
508            for runtime in machine.pinned.iter().filter(|pinned| {
509                !machine
510                    .connected
511                    .iter()
512                    .any(|connected| connected.runtime_id == **pinned)
513            }) {
514                entries.push(MachineEntry::Runtime {
515                    machine: machine.name.clone(),
516                    runtime: runtime.clone(),
517                    assigned_by: None,
518                    pinned: true,
519                });
520            }
521            for runtime in machine
522                .remembered
523                .iter()
524                .filter(|remembered| !machine.pinned.contains(&remembered.runtime_id))
525            {
526                entries.push(MachineEntry::Runtime {
527                    machine: machine.name.clone(),
528                    runtime: runtime.runtime_id.clone(),
529                    assigned_by: Some(runtime.assigned_by),
530                    pinned: false,
531                });
532            }
533            entries
534        })
535        .collect()
536}

538/// `action` on `entry`: a request ready to send, `None` when it first needs
539/// a machine name, or why the entry does not offer it.
540pub fn machine_action_request(
541    entry: &MachineEntry,
542    action: MachineAction,
543) -> Result<Option<MachineRequest>, String> {
544    if !entry.actions().contains(&action) {
545        return Err(match (entry, action) {
546            (MachineEntry::Machine(_), _) => format!("select a runtime to {}", action.label()),
547            (MachineEntry::Runtime { .. }, MachineAction::Unpin) => {
548                "only a pinned runtime can be unpinned".to_owned()
549            }
550            (MachineEntry::Runtime { .. }, _) => {
551                format!("select a machine to {}", action.label())
552            }
553        });
554    }
555    if action.asks_name() {
556        return Ok(None);
557    }
558    Ok(Some(match (action, entry) {
559        (MachineAction::Forget, MachineEntry::Machine(machine)) => MachineRequest::Forget {
560            machine: machine.clone(),
561        },
562        (MachineAction::Unpin, MachineEntry::Runtime { runtime, .. }) => MachineRequest::Unassign {
563            runtime: runtime.clone(),
564        },
565        _ => return Err("nothing to do".into()),
566    }))
567}

569/// The request `action` on `entry` makes with the typed machine name.
570pub fn machine_name_request(
571    action: MachineAction,
572    entry: &MachineEntry,
573    typed: &str,
574) -> Result<MachineRequest, String> {
575    let typed = typed.trim();
576    let name = MachineName::new(typed).map_err(|_| {
577        format!(
578            "`{typed}` is not a machine name: a lowercase letter, then up to 31 lowercase \
579             letters, digits, `_`, or `-`"
580        )
581    })?;
582    Ok(match (action, entry) {
583        (MachineAction::Rename, MachineEntry::Machine(machine)) => MachineRequest::Rename {
584            machine: machine.clone(),
585            name,
586        },
587        (MachineAction::Merge, MachineEntry::Machine(machine)) => MachineRequest::Merge {
588            from: machine.clone(),
589            into: name,
590        },
591        (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => MachineRequest::Assign {
592            runtime: runtime.clone(),
593            machine: name,
594        },
595        _ => return Err("nothing to do".into()),
596    })
597}

599/// What a machine form asks for `action` on `entry`.
600pub fn machine_form_title(action: MachineAction, entry: &MachineEntry) -> String {
601    match (action, entry) {
602        (MachineAction::Rename, MachineEntry::Machine(machine)) => {
603            format!("New name for {machine}")
604        }
605        (MachineAction::Merge, MachineEntry::Machine(machine)) => {
606            format!("Merge {machine} into (its memories move too)")
607        }
608        (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => {
609            format!("Machine for runtime {runtime} (a new name creates it)")
610        }
611        _ => "Machine name".to_owned(),
612    }
613}

615impl MachineRequest {
616    /// What the status line says once the change is made.
617    pub fn done(&self) -> String {
618        match self {
619            Self::Rename { machine, name } => format!("renamed machine {machine} to {name}"),
620            Self::Assign { runtime, machine } => {
621                format!("pinned runtime {runtime} to machine {machine}")
622            }
623            Self::Unassign { runtime } => format!("unpinned runtime {runtime}"),
624            Self::Merge { from, into } => format!("merged machine {from} into {into}"),
625            Self::Forget { machine } => format!("forgot machine {machine}"),
626            Self::ForgetRuntime { runtime } => format!("forgot where runtime {runtime} ran"),
627        }
628    }

793    #[test]
794    fn a_runtimes_acp_agents_are_its_declarations_with_the_agents_over_them() {
795        use cookie_agent_protocol::{AgentDescriptor, AgentId, AgentMode, AgentRuntime};
796        let offer = |runtime: &str, declaration: &str, available| AgentRuntime {
797            runtime_id: RuntimeId::new(runtime).unwrap(),
798            agent: AgentId::new(declaration).unwrap(),
799            available,
800            options: Vec::new(),
801            input_media: Vec::new(),
802        };
803        let agent = |id: &str, description: &str, runtimes: Vec<AgentRuntime>| AgentDescriptor {
804            id: AgentId::new(id).unwrap(),
805            preset: None,
806            description: description.into(),
807            mode: AgentMode::All,
808            enabled: true,
809            runnable_as_root: true,
810            resolved_fallback: Vec::new(),
811            delegation_targets: Vec::new(),
812            runtime: None,
813            runtimes,
814            unset_models: Vec::new(),
815        };
816        let agents = [
817            agent("build", "Builds", Vec::new()),
818            agent(
819                "claude",
820                "Claude Code",
821                vec![
822                    offer("labpc", "claude", true),
823                    offer("local", "claude", false),
824                ],
825            ),
826            // A template over `claude`, and one over `codex`, which no
827            // document of its own ID describes.
828            agent(
829                "claude-review",
830                "Reviews",
831                vec![
832                    offer("labpc", "claude", true),
833                    offer("local", "claude", false),
834                ],
835            ),
836            agent(
837                "codex-fast",
838                "Fast Codex",
839                vec![offer("labpc", "codex", false)],
840            ),
841        ];
842        let listed = super::runtime_acp_agents(&agents, &RuntimeId::new("labpc").unwrap());
843        assert_eq!(
844            listed
845                .iter()
846                .map(|declaration| (
847                    declaration.agent.as_str(),
848                    declaration.available,
849                    declaration.description.as_str(),
850                    declaration
851                        .agents
852                        .iter()
853                        .map(AgentId::as_str)
854                        .collect::<Vec<_>>()
855                ))
856                .collect::<Vec<_>>(),
857            [
858                (
859                    "claude",
860                    true,
861                    "Claude Code",
862                    vec!["claude", "claude-review"]
863                ),
864                ("codex", false, "", vec!["codex-fast"]),
865            ]
866        );
867        let local = super::runtime_acp_agents(&agents, &RuntimeId::new("local").unwrap());
868        assert_eq!(local.len(), 1);
869        assert!(!local[0].available);
870        assert!(super::runtime_acp_agents(&agents, &RuntimeId::new("global").unwrap()).is_empty());
871    }
