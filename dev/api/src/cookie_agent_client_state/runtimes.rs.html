16/// One attachment row: what the session records and the instance it refers
17/// to, when the daemon reports one.
18#[derive(Clone, Debug, Eq, PartialEq)]
19pub struct RuntimeRow {
20    pub attachment: RuntimeAttachment,
21    pub instance: Option<RuntimeInstanceInfo>,
22    /// What the runtime says its capability offers, once it has connected.
23    pub description: Option<String>,
24}
25
26impl RuntimeRow {
27    /// The MCP servers the row's instance hosts, from `mcp.server.list`, in
28    /// name order: those of its runtime at the instance's workspace (the
29    /// attachment's until the instance reports one).
30    pub fn mcp_servers(&self, servers: &[McpServerInfo]) -> Vec<McpServerInfo> {
31        let workspace = match &self.instance {
32            Some(instance) => instance.workspace.as_deref(),
33            None => self.attachment.workspace.as_deref(),
34        };
35        crate::mcp::instance_servers(servers, &self.attachment.runtime_id, workspace)
36    }
37}

39/// The description `listing` has for `capability` of `runtime_id`.
40fn capability_description(
41    listing: Option<&RuntimesListResult>,
42    runtime_id: &RuntimeId,
43    capability: &CapabilityName,
44) -> Option<String> {
45    listing?
46        .runtimes
47        .iter()
48        .find(|runtime| runtime.runtime_id == *runtime_id)?
49        .capabilities
50        .iter()
51        .find(|offered| offered.name == *capability)
52        .map(|offered| offered.description.clone())
53        .filter(|description| !description.is_empty())
54}

56/// A runtime a new-session draft attaches. Its instance is opened as soon as
57/// it joins the draft, so its status, and why it failed, show before the
58/// session exists.
59#[derive(Clone, Debug, Eq, PartialEq)]
60pub struct DraftRuntime {
61    /// Matches the answer of the open request to this entry.
62    pub id: u64,
63    pub request: RuntimeAttachRequest,
64    /// The instance the open request answered, or why it failed.
65    pub opened: Option<Result<RuntimeInstanceInfo, String>>,
66}
67
68impl DraftRuntime {
69    pub fn is_global(&self) -> bool {
70        self.request.runtime_id.as_str() == RuntimeId::GLOBAL
71    }
72
73    /// The opened instance's canonical workspace, else the one requested.
74    pub fn workspace(&self) -> Option<&str> {
75        match &self.opened {
76            Some(Ok(instance)) => instance.workspace.as_deref(),
77            _ => self.request.workspace.as_deref(),
78        }
79    }
80
81    /// Whether `other` names the same instance as this entry.
82    pub fn same_instance(&self, other: &RuntimeAttachRequest) -> bool {
83        self.request.runtime_id == other.runtime_id
84            && self.request.capability == other.capability
85            && (self.request.workspace == other.workspace
86                || self.workspace() == other.workspace.as_deref())
87    }
88}

90/// One thing the attach step offers: a capability of a runtime, or a runtime
91/// that has not said what it offers yet, which attaching connects and takes
92/// its only capability of.
93#[derive(Clone, Debug, Eq, PartialEq)]
94pub struct RuntimeChoice {
95    pub runtime_id: RuntimeId,
96    pub capability: Option<RuntimeCapabilityInfo>,
97    pub connected: bool,
98    /// Why the runtime last failed to connect.
99    pub message: Option<String>,
100}
101
102impl RuntimeChoice {
103    /// Whether attaching asks for a workspace: the capability takes one, or
104    /// the runtime has not said what it offers yet (then it is optional).
105    pub fn asks_workspace(&self) -> bool {
106        self.capability
107            .as_ref()
108            .is_none_or(|capability| capability.takes_workspace)
109    }
110
111    pub fn label(&self) -> String {
112        match &self.capability {
113            Some(capability) => format!("{}/{}", self.runtime_id, capability.name),
114            None => self.runtime_id.to_string(),
115        }
116    }
117
118    /// Case-insensitive substring match over the runtime ID and the
119    /// capability's name, kind, and description.
120    pub fn matches(&self, query: &str) -> bool {
121        let query = query.trim().to_lowercase();
122        if query.is_empty() {
123            return true;
124        }
125        let mut haystack = self.runtime_id.to_string();
126        if let Some(capability) = &self.capability {
127            haystack.push(' ');
128            haystack.push_str(capability.name.as_str());
129            haystack.push(' ');
130            haystack.push_str(capability.kind.as_str());
131            haystack.push(' ');
132            haystack.push_str(&capability.description);
133        }
134        haystack.to_lowercase().contains(&query)
135    }
136}

138/// A change to the machine registry the machines view offers.
139#[derive(Clone, Copy, Debug, Eq, PartialEq)]
140pub enum MachineAction {
141    Rename,
142    Move,
143    Unpin,
144    Merge,
145    Forget,
146    /// Opens a runtime's MCP servers, to manage them.
147    Servers,
148    /// Opens a runtime's ACP agents, to manage them.
149    Agents,
150}
151
152impl MachineAction {
153    pub const fn key(self) -> char {
154        match self {
155            Self::Rename => 'n',
156            Self::Move => 'v',
157            Self::Unpin => 'u',
158            Self::Merge => 'g',
159            Self::Forget => 'f',
160            Self::Servers => 's',
161            Self::Agents => 'a',
162        }
163    }
164
165    pub const fn label(self) -> &'static str {
166        match self {
167            Self::Rename => "rename",
168            Self::Move => "move to machine",
169            Self::Unpin => "unpin",
170            Self::Merge => "merge into",
171            Self::Forget => "forget",
172            Self::Servers => "MCP servers",
173            Self::Agents => "ACP agents",
174        }
175    }
176
177    /// Whether the action opens a panel of the runtime rather than changing
178    /// the machine registry.
179    pub const fn opens_panel(self) -> bool {
180        matches!(self, Self::Servers | Self::Agents)
181    }
182
183    /// Whether the action asks for a machine name first.
184    pub const fn asks_name(self) -> bool {
185        matches!(self, Self::Rename | Self::Move | Self::Merge)
186    }
187
188    /// The action of the key `key`.
189    pub fn of_key(key: char) -> Option<Self> {
190        [
191            Self::Rename,
192            Self::Move,
193            Self::Unpin,
194            Self::Merge,
195            Self::Forget,
196            Self::Servers,
197            Self::Agents,
198        ]
199        .into_iter()
200        .find(|action| action.key() == key)
201    }
202}

204/// An ACP agent declaration a runtime offers, as the machines views list it
205/// (`docs/site/specs/runtime-management.md` section 4.2).
206#[derive(Clone, Debug, Eq, PartialEq)]
207pub struct RuntimeAcpAgent {
208    /// The declaration's key, which is also the agent of that ID.
209    pub agent: AgentId,
210    /// A session that attaches the runtime can run it now.
211    pub available: bool,
212    /// What the agent of the declaration's own ID says it is; empty when
213    /// only templates run the declaration.
214    pub description: String,
215    /// The agents that run the declaration on the runtime, in ID order:
216    /// the one of its own ID and every template over it (`acp.agent`).
217    pub agents: Vec<AgentId>,
218}
219
220/// The ACP agent declarations `runtime_id` offers, by key, from the
221/// catalog's `agents`.
222pub fn runtime_acp_agents(
223    agents: &[AgentDescriptor],
224    runtime_id: &RuntimeId,
225) -> Vec<RuntimeAcpAgent> {
226    let mut declarations = std::collections::BTreeMap::<AgentId, RuntimeAcpAgent>::new();
227    for descriptor in agents {
228        for offer in descriptor
229            .runtimes
230            .iter()
231            .filter(|offer| offer.runtime_id == *runtime_id)
232        {
233            let declaration =
234                declarations
235                    .entry(offer.agent.clone())
236                    .or_insert_with(|| RuntimeAcpAgent {
237                        agent: offer.agent.clone(),
238                        available: false,
239                        description: String::new(),
240                        agents: Vec::new(),
241                    });
242            declaration.available |= offer.available;
243            if descriptor.id == offer.agent {
244                declaration.description.clone_from(&descriptor.description);
245            }
246            if !declaration.agents.contains(&descriptor.id) {
247                declaration.agents.push(descriptor.id.clone());
248            }
249        }
250    }
251    declarations
252        .into_values()
253        .map(|mut declaration| {
254            declaration.agents.sort();
255            declaration
256        })
257        .collect()
258}

260/// The form of an ACP agent declaration,
261/// `[runtimes.<id>.acp.agents.<agent>]`.
262#[derive(Clone, Debug, Eq, PartialEq)]
263pub struct AcpAgentFields {
264    /// The declaration's key, which is also its agent's ID.
265    pub agent: String,
266    /// The program and its arguments, as a JSON array.
267    pub command: String,
268    /// The environment, as a JSON object.
269    pub environment: String,
270    pub description: String,
271    pub enabled: bool,
272}
273
274impl AcpAgentFields {
275    /// A new declaration's form.
276    #[must_use]
277    pub fn add() -> Self {
278        Self {
279            agent: String::new(),
280            command: "[]".into(),
281            environment: "{}".into(),
282            description: String::new(),
283            enabled: true,
284        }
285    }
286
287    /// `declaration`'s form.
288    #[must_use]
289    pub fn edit(declaration: &RuntimeAgentDeclaration) -> Self {
290        Self {
291            agent: declaration.agent.to_string(),
292            command: serde_json::to_string(&declaration.definition.command)
293                .unwrap_or_else(|_| "[]".into()),
294            environment: serde_json::to_string(&declaration.definition.env)
295                .unwrap_or_else(|_| "{}".into()),
296            description: declaration.definition.description.clone(),
297            enabled: declaration.definition.enabled,
298        }
299    }
300
301    /// The key and definition the form describes, or what is wrong.
302    pub fn definition(&self) -> Result<(AgentId, AcpAgentDefinition), String> {
303        let agent = AgentId::new(self.agent.trim()).map_err(|_| {
304            "the agent ID is lowercase letters, digits, and single inner hyphens".to_owned()
305        })?;
306        let command: Vec<String> = serde_json::from_str(self.command.trim())
307            .map_err(|_| "the command is a JSON array of strings".to_owned())?;
308        if command.first().is_none_or(String::is_empty) {
309            return Err("the command names a program first".into());
310        }
311        let env = if self.environment.trim().is_empty() {
312            BTreeMap::new()
313        } else {
314            serde_json::from_str(self.environment.trim())
315                .map_err(|_| "the environment is a JSON object of strings".to_owned())?
316        };
317        Ok((
318            agent,
319            AcpAgentDefinition {
320                command,
321                env,
322                description: self.description.trim().to_owned(),
323                enabled: self.enabled,
324            },
325        ))
326    }
327}

360/// One row of the machines view: a machine, or a runtime under it.
361#[derive(Clone, Debug, Eq, PartialEq)]
362pub enum MachineEntry {
363    Machine(MachineName),
364    Runtime {
365        machine: MachineName,
366        runtime: RuntimeId,
367        /// Why it belongs to the machine while it is connected.
368        assigned_by: Option<MachineAssignment>,
369        pinned: bool,
370    },
371}
372
373impl MachineEntry {
374    /// The actions the entry offers.
375    pub fn actions(&self) -> Vec<MachineAction> {
376        match self {
377            Self::Machine(_) => vec![
378                MachineAction::Rename,
379                MachineAction::Merge,
380                MachineAction::Forget,
381            ],
382            Self::Runtime {
383                runtime, pinned, ..
384            } => {
385                let mut actions = vec![MachineAction::Move];
386                if *pinned {
387                    actions.push(MachineAction::Unpin);
388                }
389                actions.push(MachineAction::Servers);
390                // `global` hosts no ACP agents.
391                if runtime.as_str() != RuntimeId::GLOBAL {
392                    actions.push(MachineAction::Agents);
393                }
394                actions
395            }
396        }
397    }
398}

456/// The rows for a new-session draft: each entry with its live instance from
457/// `listing`, else the one its open request answered. Until the instance
458/// reports its attachment name and kind, the row shows the runtime ID and the
459/// capability.
460pub fn draft_runtime_rows(
461    entries: &[DraftRuntime],
462    listing: Option<&RuntimesListResult>,
463) -> Vec<RuntimeRow> {
464    entries
465        .iter()
466        .filter_map(|entry| {
467            let capability = match (&entry.request.capability, &entry.opened) {
468                (Some(capability), _) => capability.clone(),
469                (None, Some(Ok(instance))) => instance.capability.clone(),
470                (None, _) => cookie_agent_protocol::default_capability(&entry.request.runtime_id),
471            };
472            let workspace = entry.workspace().map(str::to_owned);
473            let live = listing.and_then(|listing| {
474                listing
475                    .instances
476                    .iter()
477                    .find(|instance| {
478                        instance.runtime_id == entry.request.runtime_id
479                            && instance.capability == capability
480                            && instance.workspace == workspace
481                    })
482                    .cloned()
483            });
484            let instance = live.or_else(|| match &entry.opened {
485                Some(Ok(instance)) => Some(instance.clone()),
486                Some(Err(message)) => Some(RuntimeInstanceInfo {
487                    runtime_id: entry.request.runtime_id.clone(),
488                    capability: capability.clone(),
489                    attachment_name: None,
490                    kind: None,
491                    workspace: workspace.clone(),
492                    platform: None,
493                    status: RuntimeInstanceStatus::Failed,
494                    message: Some(message.clone()),
495                }),
496                None => None,
497            });
498            Some(RuntimeRow {
499                description: capability_description(
500                    listing,
501                    &entry.request.runtime_id,
502                    &capability,
503                ),
504                attachment: RuntimeAttachment {
505                    name: match instance
506                        .as_ref()
507                        .and_then(|instance| instance.attachment_name.clone())
508                    {
509                        Some(name) => name,
510                        None => AttachmentName::new(entry.request.runtime_id.as_str()).ok()?,
511                    },
512                    kind: match instance.as_ref().and_then(|instance| instance.kind.clone()) {
513                        Some(kind) => kind,
514                        None => RuntimeKind::new(capability.as_str()).ok()?,
515                    },
516                    runtime_id: entry.request.runtime_id.clone(),
517                    capability,
518                    workspace,
519                },
520                instance,
521            })
522        })
523        .collect()
524}

538/// Every runtime's capabilities, in listing order, that match `query`; a
539/// runtime that has not said what it offers is one choice.
540pub fn runtime_choices(runtimes: &[RuntimeListing], query: &str) -> Vec<RuntimeChoice> {
541    runtimes
542        .iter()
543        .flat_map(|runtime| {
544            let choice = |capability| RuntimeChoice {
545                runtime_id: runtime.runtime_id.clone(),
546                capability,
547                connected: runtime.connected,
548                message: runtime.message.clone(),
549            };
550            if runtime.capabilities.is_empty() {
551                vec![choice(None)]
552            } else {
553                runtime
554                    .capabilities
555                    .iter()
556                    .cloned()
557                    .map(|capability| choice(Some(capability)))
558                    .collect()
559            }
560        })
561        .filter(|choice| choice.matches(query))
562        .collect()
563}

565/// The attach request `choice` with the typed `workspace` describes.
566pub fn attach_parameters(
567    choice: &RuntimeChoice,
568    workspace: &str,
569) -> Result<(RuntimeId, Option<CapabilityName>, Option<String>), String> {
570    let workspace = workspace.trim();
571    let workspace = match (&choice.capability, workspace.is_empty()) {
572        (Some(capability), true) if capability.takes_workspace => {
573            return Err(format!(
574                "capability `{}` of runtime `{}` needs a workspace path",
575                capability.name, choice.runtime_id
576            ));
577        }
578        (_, true) => None,
579        (_, false) => Some(workspace.to_owned()),
580    };
581    Ok((
582        choice.runtime_id.clone(),
583        choice
584            .capability
585            .as_ref()
586            .map(|capability| capability.name.clone()),
587        workspace,
588    ))
589}

591/// The `runtimes.reconnect` request that retries `row`'s instance: its
592/// workspace goes along exactly when it has one.
593pub fn reconnect_request(
594    runtimes: &[RuntimeListing],
595    row: &RuntimeRow,
596) -> Result<(RuntimeId, CapabilityName, Option<String>), String> {
597    if !runtimes
598        .iter()
599        .any(|runtime| runtime.runtime_id == row.attachment.runtime_id)
600    {
601        return Err(format!(
602            "runtime `{}` is not a known runtime",
603            row.attachment.runtime_id
604        ));
605    }
606    Ok((
607        row.attachment.runtime_id.clone(),
608        row.attachment.capability.clone(),
609        row.attachment.workspace.clone(),
610    ))
611}

621/// The machines view's rows: each machine, then its connected runtimes, the
622/// runtimes pinned to it that are not connected, and the offline runtimes
623/// it remembers (client protocol 1.10.0), placed by `last_seen`.
624pub fn machine_entries(listing: &MachinesListResult) -> Vec<MachineEntry> {
625    listing
626        .machines
627        .iter()
628        .flat_map(|machine| {
629            let mut entries = vec![MachineEntry::Machine(machine.name.clone())];
630            for runtime in &machine.connected {
631                entries.push(MachineEntry::Runtime {
632                    machine: machine.name.clone(),
633                    runtime: runtime.runtime_id.clone(),
634                    assigned_by: Some(runtime.assigned_by),
635                    pinned: machine.pinned.contains(&runtime.runtime_id),
636                });
637            }
638            for runtime in machine.pinned.iter().filter(|pinned| {
639                !machine
640                    .connected
641                    .iter()
642                    .any(|connected| connected.runtime_id == **pinned)
643            }) {
644                entries.push(MachineEntry::Runtime {
645                    machine: machine.name.clone(),
646                    runtime: runtime.clone(),
647                    assigned_by: None,
648                    pinned: true,
649                });
650            }
651            for runtime in machine
652                .remembered
653                .iter()
654                .filter(|remembered| !machine.pinned.contains(&remembered.runtime_id))
655            {
656                entries.push(MachineEntry::Runtime {
657                    machine: machine.name.clone(),
658                    runtime: runtime.runtime_id.clone(),
659                    assigned_by: Some(runtime.assigned_by),
660                    pinned: false,
661                });
662            }
663            entries
664        })
665        .collect()
666}

668/// `action` on `entry`: a request ready to send, `None` when it first needs
669/// a machine name, or why the entry does not offer it.
670pub fn machine_action_request(
671    entry: &MachineEntry,
672    action: MachineAction,
673) -> Result<Option<MachineRequest>, String> {
674    if !entry.actions().contains(&action) {
675        return Err(match (entry, action) {
676            (MachineEntry::Machine(_), _) => format!("select a runtime to {}", action.label()),
677            (MachineEntry::Runtime { .. }, MachineAction::Unpin) => {
678                "only a pinned runtime can be unpinned".to_owned()
679            }
680            (MachineEntry::Runtime { .. }, _) => {
681                format!("select a machine to {}", action.label())
682            }
683        });
684    }
685    if action.asks_name() {
686        return Ok(None);
687    }
688    Ok(Some(match (action, entry) {
689        (MachineAction::Forget, MachineEntry::Machine(machine)) => MachineRequest::Forget {
690            machine: machine.clone(),
691        },
692        (MachineAction::Unpin, MachineEntry::Runtime { runtime, .. }) => MachineRequest::Unassign {
693            runtime: runtime.clone(),
694        },
695        _ => return Err("nothing to do".into()),
696    }))
697}

699/// The request `action` on `entry` makes with the typed machine name.
700pub fn machine_name_request(
701    action: MachineAction,
702    entry: &MachineEntry,
703    typed: &str,
704) -> Result<MachineRequest, String> {
705    let typed = typed.trim();
706    let name = MachineName::new(typed).map_err(|_| {
707        format!(
708            "`{typed}` is not a machine name: a lowercase letter, then up to 31 lowercase \
709             letters, digits, `_`, or `-`"
710        )
711    })?;
712    Ok(match (action, entry) {
713        (MachineAction::Rename, MachineEntry::Machine(machine)) => MachineRequest::Rename {
714            machine: machine.clone(),
715            name,
716        },
717        (MachineAction::Merge, MachineEntry::Machine(machine)) => MachineRequest::Merge {
718            from: machine.clone(),
719            into: name,
720        },
721        (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => MachineRequest::Assign {
722            runtime: runtime.clone(),
723            machine: name,
724        },
725        _ => return Err("nothing to do".into()),
726    })
727}
728
729/// What a machine form asks for `action` on `entry`.
730pub fn machine_form_title(action: MachineAction, entry: &MachineEntry) -> String {
731    match (action, entry) {
732        (MachineAction::Rename, MachineEntry::Machine(machine)) => {
733            format!("New name for {machine}")
734        }
735        (MachineAction::Merge, MachineEntry::Machine(machine)) => {
736            format!("Merge {machine} into (its memories move too)")
737        }
738        (MachineAction::Move, MachineEntry::Runtime { runtime, .. }) => {
739            format!("Machine for runtime {runtime} (a new name creates it)")
740        }
741        _ => "Machine name".to_owned(),
742    }
743}
744
745impl MachineRequest {
746    /// What the status line says once the change is made.
747    pub fn done(&self) -> String {
748        match self {
749            Self::Rename { machine, name } => format!("renamed machine {machine} to {name}"),
750            Self::Assign { runtime, machine } => {
751                format!("pinned runtime {runtime} to machine {machine}")
752            }
753            Self::Unassign { runtime } => format!("unpinned runtime {runtime}"),
754            Self::Merge { from, into } => format!("merged machine {from} into {into}"),
755            Self::Forget { machine } => format!("forgot machine {machine}"),
756            Self::ForgetRuntime { runtime } => format!("forgot where runtime {runtime} ran"),
757        }
758    }
759}
