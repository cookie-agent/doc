5/// Whether a model matches `query`: every word of it, ignoring case, is in
6/// one of `fields` (its name, provider, ID, description). An empty query
7/// matches everything.
8pub fn matches(query: &str, fields: &[&str]) -> bool {
9    let fields: Vec<String> = fields.iter().map(|field| field.to_lowercase()).collect();
10    query
11        .split_whitespace()
12        .map(str::to_lowercase)
13        .all(|word| fields.iter().any(|field| field.contains(&word)))
14}

16/// Where the first word of `query` falls in `name`, ignoring case, as a
17/// byte range of `name` on character boundaries: the part a row marks.
18pub fn match_range(name: &str, query: &str) -> Option<(usize, usize)> {
19    let word: Vec<char> = query.split_whitespace().next()?.chars().collect();
20    let chars: Vec<(usize, char)> = name.char_indices().collect();
21    (0..chars.len()).find_map(|start| {
22        let fits = word.iter().enumerate().all(|(offset, wanted)| {
23            chars
24                .get(start + offset)
25                .is_some_and(|(_, found)| found.to_lowercase().eq(wanted.to_lowercase()))
26        });
27        fits.then(|| {
28            let from = chars[start].0;
29            let to = chars
30                .get(start + word.len())
31                .map_or(name.len(), |(index, _)| *index);
32            (from, to)
33        })
34    })
35}

37/// `items` in groups by `provider`, each provider where it first appears;
38/// one group without a heading when they all share one provider.
39pub fn provider_groups<T>(
40    items: Vec<T>,
41    provider: impl Fn(&T) -> String,
42) -> Vec<(Option<String>, Vec<T>)> {
43    let mut groups: Vec<(String, Vec<T>)> = Vec::new();
44    for item in items {
45        let name = provider(&item);
46        match groups.iter_mut().find(|(known, _)| *known == name) {
47            Some((_, members)) => members.push(item),
48            None => groups.push((name, vec![item])),
49        }
50    }
51    if groups.len() <= 1 {
52        return groups
53            .into_iter()
54            .map(|(_, members)| (None, members))
55            .collect();
56    }
57    groups
58        .into_iter()
59        .map(|(name, members)| (Some(name), members))
60        .collect()
61}

63#[cfg(test)]
64mod tests {
65    use super::{match_range, matches, provider_groups};
66
67    #[test]
68    fn every_word_must_match_some_field_ignoring_case() {
69        let fields = ["Claude Opus 5.5", "anthropic", "anthropic/claude-opus-5-5"];
70        assert!(matches("", &fields));
71        assert!(matches("opus", &fields));
72        assert!(matches("OPUS anthropic", &fields));
73        assert!(matches("claude-opus", &fields));
74        assert!(!matches("opus openai", &fields));
75        assert!(!matches("sonnet", &fields));
76    }
77
78    #[test]
79    fn a_match_is_found_on_character_boundaries() {
80        assert_eq!(match_range("Claude Opus 5.5", "opus"), Some((7, 11)));
81        assert_eq!(match_range("Claude Opus 5.5", "  OPUS  x"), Some((7, 11)));
82        assert_eq!(match_range("Claude Opus", "sonnet"), None);
83        assert_eq!(match_range("Claude Opus", ""), None);
84        // Multi-byte characters before and in the match.
85        assert_eq!(match_range("Ünïcode Öpus", "öpus"), Some((10, 15)));
86        assert_eq!(match_range("ab", "abc"), None);
87    }
88
89    #[test]
90    fn groups_keep_first_appearance_order_and_one_provider_has_no_heading() {
91        let items = vec!["anthropic/a", "openai/b", "anthropic/c"];
92        let provider = |item: &&str| item.split('/').next().unwrap().to_owned();
93        let groups = provider_groups(items, provider);
94        assert_eq!(groups.len(), 2);
95        assert_eq!(groups[0].0.as_deref(), Some("anthropic"));
96        assert_eq!(groups[0].1, ["anthropic/a", "anthropic/c"]);
97        assert_eq!(groups[1].0.as_deref(), Some("openai"));
98        let one = provider_groups(vec!["anthropic/a", "anthropic/b"], provider);
99        assert_eq!(one.len(), 1);
100        assert_eq!(one[0].0, None);
101        assert!(provider_groups(Vec::<&str>::new(), provider).is_empty());
102    }
