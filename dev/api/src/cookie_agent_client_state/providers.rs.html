47pub fn row_state(
48    provider: &ProviderDescriptor,
49    _models: &[AvailableModelDescriptor],
50    operation: Option<&ProviderOperation>,
51) -> ProviderRowState {
52    if matches!(operation, Some(ProviderOperation::Error { .. })) {
53        return ProviderRowState::ErrorRetry;
54    }
55    if provider.support.state != ProviderSupportState::Supported {
56        return ProviderRowState::Unsupported;
57    }
58    if provider.presence == ProviderPresence::Removed {
59        return ProviderRowState::Removed;
60    }
61    if provider.durable_connection.is_some() {
62        return ProviderRowState::ConnectedReconnect;
63    }
64    ProviderRowState::Disconnected
65}

67pub fn authored_override_effective(
68    provider: &ProviderDescriptor,
69    models: &[AvailableModelDescriptor],
70) -> bool {
71    provider.configuration == ProviderConfigurationState::Authored
72        && provider.effective_auth_state == EffectiveAuthState::AuthoredOverride
73        && provider.setup_fields.iter().all(|field| {
74            !field.required
75                || field.default.is_some()
76                || models
77                    .iter()
78                    .any(|model| model.key.provider_id() == provider.id)
79        })
80}

122/// Short verb phrase for one unavailable kind, read after a model count.
123pub const fn unavailable_kind_label(kind: ModelUnavailableKind) -> &'static str {
124    match kind {
125        ModelUnavailableKind::Quarantined => "quarantined",
126        ModelUnavailableKind::Unsupported => "unsupported",
127        ModelUnavailableKind::NeedsSetup => "need setup",
128        ModelUnavailableKind::NeedsCredentials => "need credentials",

132/// One unavailable model's reason, e.g. `quarantined: invalid_catalog_model_record`.
133pub fn unavailable_model_reason(model: &UnavailableModelDescriptor) -> String {
134    let label = match model.kind {
135        ModelUnavailableKind::NeedsSetup => "needs setup",
136        ModelUnavailableKind::NeedsCredentials => "needs credentials",
137        kind => unavailable_kind_label(kind),
138    };
139    model
140        .reason
141        .as_ref()
142        .map_or_else(|| label.to_owned(), |reason| format!("{label}: {reason}"))
143}

145/// The unavailable kinds with nonzero counts, e.g.
146/// `4 quarantined (invalid_catalog_model_record) · 2 need credentials`. Up to
147/// two distinct listed reasons follow quarantined and unsupported counts.
148fn unavailable_parts(provider: &ProviderDescriptor) -> Vec<String> {
149    UNAVAILABLE_KINDS
150        .into_iter()
151        .filter_map(|kind| {
152            let count = provider.model_counts.count(kind);
153            if count == 0 {
154                return None;
155            }
156            let mut reasons = Vec::<&str>::new();
157            for model in &provider.unavailable_models {
158                if model.kind == kind
159                    && let Some(reason) = &model.reason
160                    && !reasons.contains(&reason.as_str())
161                {
162                    reasons.push(reason.as_str());
163                }
164            }
165            let label = unavailable_kind_label(kind);
166            Some(match reasons.len() {
167                0 => format!("{count} {label}"),
168                1 | 2 => format!("{count} {label} ({})", reasons.join(", ")),
169                _ => format!("{count} {label} ({}, …)", reasons[..2].join(", ")),
170            })
171        })
172        .collect()
173}

175/// Full model availability for the details view, e.g.
176/// `3 usable · 4 quarantined (invalid_catalog_model_record)`.
177pub fn model_counts_summary(provider: &ProviderDescriptor) -> String {
178    let counts = &provider.model_counts;
179    if counts.available == 0 && counts.unavailable() == 0 {
180        return "no catalog models".to_owned();
181    }
182    let mut parts = vec![format!("{} usable", counts.available)];
183    parts.extend(unavailable_parts(provider));
184    parts.join(" · ")
185}

187/// Why a provider row yields no (or fewer) usable models, when that is news
188/// to the user: configured providers always explain unavailable models, and
189/// unconfigured providers only when connecting could not help because every
190/// model is quarantined or unsupported.
191pub fn unusable_models_note(provider: &ProviderDescriptor) -> Option<String> {
192    let counts = &provider.model_counts;
193    let unavailable = counts.unavailable();
194    if unavailable == 0 {
195        return None;
196    }
197    let broken = counts.quarantined.saturating_add(counts.unsupported);
198    if provider_configured(provider) {
199        if counts.available == 0 {
200            return Some(format!(
201                "no usable models: {}",
202                unavailable_parts(provider).join(" · ")
203            ));
204        }
205        return Some(format!(
206            "{unavailable} of {} models unavailable",
207            counts.available.saturating_add(unavailable)
208        ));
209    }
210    (counts.available == 0 && broken == unavailable).then(|| {
211        format!(
212            "no usable models: {}",
213            unavailable_parts(provider).join(" · ")
214        )
215    })
216}

218/// A config.toml line that supplies the provider's single API key from the
219/// first catalog environment variable, offered only when the selected method
220/// is exactly one API key. Display-only: cookie never reads the variable
221/// unless the user authors this line.
222pub fn env_config_hint(
223    provider: &ProviderDescriptor,
224    auth: Option<&cookie_agent_protocol::AuthMethodDescriptor>,
225) -> Option<String> {
226    let single_key = auth.is_some_and(|method| {
227        method.credentials.len() == 1
228            && method.credentials[0].credential_type
229                == cookie_agent_protocol::CredentialFieldType::ApiKey
230    });
231    if !single_key {
232        return None;
233    }
234    let name = provider
235        .environment
236        .iter()
237        .find(|name| name.as_str().contains("KEY"))
238        .or_else(|| provider.environment.first())?;
239    Some(format!(
240        "Or in config.toml: [providers.{}] source = \"models_dev\", api_key = \"${{env:{name}}}\"",
241        provider.id
242    ))
243}

245/// One-line catalog provenance: source, age when stale, quarantine counts,
246/// and the last refresh error when there is one.
247pub fn catalog_status_line(snapshot: &ModelCatalogView) -> String {
248    let state = &snapshot.catalog;
249    let mut parts = vec![format!(
250        "Catalog: {}",
251        match snapshot.catalog.source {
252            CatalogSource::Network => "models.dev (network)",
253            CatalogSource::Cache => "models.dev (cache)",
254            CatalogSource::Bootstrap => "bundled bootstrap",
255        }
256    )];
257    match state.age {
258        CatalogAge::Current => {}
259        CatalogAge::OlderThanSevenDays => parts.push("older than 7 days".to_owned()),
260        CatalogAge::OlderThanThirtyDays => parts.push("older than 30 days".to_owned()),
261    }
262    if state.stale && snapshot.catalog.source == CatalogSource::Cache {
263        parts.push("stale".to_owned());
264    }
265    if state.model_quarantine_count > 0 {
266        parts.push(format!(
267            "{} quarantined model record{}",
268            state.model_quarantine_count,
269            if state.model_quarantine_count == 1 {
270                ""
271            } else {
272                "s"
273            }
274        ));
275    }
276    if state.provider_quarantine_count > 0 {
277        parts.push(format!(
278            "{} quarantined provider record{}",
279            state.provider_quarantine_count,
280            if state.provider_quarantine_count == 1 {
281                ""
282            } else {
283                "s"
284            }
285        ));
286    }
287    if let Some(error) = &state.last_error {
288        parts.push(format!(
289            "last refresh failed ({}): {}",
290            error.code, error.message
291        ));
292    }
293    parts.join(" · ")
294}

311/// The auth method a provider's form starts with: its stored connection's,
312/// else its first.
313pub fn initial_auth_method(provider: &ProviderDescriptor) -> Option<&AuthMethodDescriptor> {
314    let id = provider
315        .durable_connection
316        .as_ref()
317        .map(|connection| &connection.auth_method)
318        .or_else(|| provider.auth_methods.first().map(|method| &method.id))?;
319    provider.auth_methods.iter().find(|method| method.id == *id)
320}

322/// The text a setup field starts with: the stored connection's value, else

334/// The public setup values typed into `fields`, parsed, or what is wrong.
335pub fn setup_values(
336    fields: &[(&SetupFieldDescriptor, &str)],
337) -> Result<BTreeMap<SetupFieldId, SafeSetupValue>, String> {
338    let mut values = BTreeMap::new();
339    for (descriptor, raw) in fields {
340        let raw = raw.trim();
341        if raw.is_empty() {
342            if descriptor.required {
343                return Err(format!("{} is required", descriptor.display_name));
344            }
345            continue;
346        }
347        values.insert(
348            descriptor.id.clone(),
349            parse_setup_value(descriptor, raw)
350                .map_err(|error| format!("{}: {error}", descriptor.display_name))?,
351        );
352    }
353    Ok(values)
354}

356/// The credentials typed into `fields`, or what is wrong. They are
357/// serialized once into a buffer wiped on drop; empty optional fields are
358/// left out.
359pub fn credential_values(
360    fields: &[(&AuthCredentialDescriptor, &str)],
361) -> Result<ProviderCredentialValues, String> {
362    for (descriptor, value) in fields {
363        if value.is_empty() {
364            if descriptor.required {
365                return Err(format!("{} is required", descriptor.display_name));
366            }
367            continue;
368        }
369        if value.len() > 16 * 1024 {
370            return Err(format!("{} is too long", descriptor.display_name));
371        }
372    }
373    let capacity = fields.iter().fold(2usize, |capacity, (descriptor, value)| {
374        capacity
375            .saturating_add(descriptor.id.as_str().len().saturating_mul(6))
376            .saturating_add(value.len().saturating_mul(6))
377            .saturating_add(8)
378    });
379    let mut serialized = Zeroizing::new(Vec::with_capacity(capacity));
380    serde_json::to_writer(&mut *serialized, &CredentialProjection(fields))
381        .map_err(|error| error.to_string())?;
382    serde_json::from_slice(&serialized).map_err(|error| error.to_string())
383}

385struct CredentialProjection<'a>(&'a [(&'a AuthCredentialDescriptor, &'a str)]);
386
387impl Serialize for CredentialProjection<'_> {
388    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
389    where
390        S: serde::Serializer,
391    {
392        let populated = self.0.iter().filter(|(_, value)| !value.is_empty()).count();
393        let mut map = serializer.serialize_map(Some(populated))?;
394        for (descriptor, value) in self.0 {
395            if !value.is_empty() {
396                map.serialize_entry(descriptor.id.as_str(), value)?;
397            }
398        }
399        map.end()
400    }
401}

403/// Whether the daemon reported retryable store contention
404/// (`lock_contention`), covering both the bounded-lock budget and a lost
405/// commit-time compare-and-swap.
406pub fn is_store_contention(error: &ClientError) -> bool {
407    error.is(&cookie_agent_protocol::errors::LOCK_CONTENTION)
408}
409
410/// Provider disconnect resubmit policy: the first contention-class failure
411/// gets exactly one automatic resubmission of the identical request (same
412/// client request id, so the daemon's replay path answers a commit whose
413/// response was lost); a second consecutive contention surfaces to the user.
414/// Connect follows the same policy through `Client::connect_provider_retrying`.
415pub fn retry_store_contention_once(attempt: usize, error: &ClientError) -> bool {
416    attempt == 0 && is_store_contention(error)
417}
418
419/// The spec's store-contention copy when the daemon reports
420/// `lock_contention`; `store` is the human noun for the error site.
421pub fn store_contention_message(error: &ClientError, store: &str) -> String {
422    if is_store_contention(error) {
423        format!("Another cookie-agent process is writing the {store} — try again.")
424    } else {
425        error.to_string()
426    }
427}
