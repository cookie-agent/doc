1//! How far back a client's session list reaches
2//! (`docs/site/specs/session-storage-v3.md` section 4.3): the last 30 days
3//! at first, or as many as the client's date filter says
4//! (`docs/site/specs/session-pins-and-filters.md`), then a month further
5//! each time the user asks for older sessions (the TUI). Unread, pinned,
6//! and busy sessions come with every answer whatever their age. A window
7//! of recent days that finds none of the rest may roll back to the days up
8//! to the newest session's last activity ([`SessionWindow::rolled_back`]).
9
10use cookie_agent_protocol::{SemVer, SessionListParams, SessionMeta};
11use jiff::{SignedDuration, Timestamp};
12
13/// The first client protocol whose daemons list sessions by activity.
14pub const LIST_WINDOW_SINCE: SemVer = SemVer::new(1, 57, 0);
15
16/// How many days a session list shows before the user asks for more,
17/// unless the client says otherwise ([`SessionWindow::recent`]).
18pub const RECENT_DAYS: i64 = 30;
19
20/// Whether the daemon, speaking `version`, takes `session.list`'s window.
21#[must_use]
22pub fn daemon_lists_by_activity(version: Option<SemVer>) -> bool {
23    version.is_some_and(|version| version >= LIST_WINDOW_SINCE)
24}
25
26/// How far back the list reaches.
27#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
28enum Reach {
29    /// The window's recent days, counted when the list is read.
30    #[default]
31    Recent,
32    /// Back to this time, after older pages came in.
33    Since(Timestamp),
34    /// Every session.
35    All,
36    /// The sessions last active from the first time to before the second:
37    /// a range of days the user chose, with nothing older to offer.
38    Between(Timestamp, Timestamp),
39    /// The window's days up to this time, the newest session's last
40    /// activity, then on to now: the recent days held none.
41    RolledBack(Timestamp),
42}
43
44/// The session list's window, and whether the daemon has older sessions.
45#[derive(Clone, Copy, Debug, Eq, PartialEq)]
46pub struct SessionWindow {
47    reach: Reach,
48    /// How many days [`Reach::Recent`] reaches back; `None`: every
49    /// session.
50    days: Option<i64>,
51    /// The `older` cursor of the newest answer.
52    older: Option<Timestamp>,
53}
54
55impl Default for SessionWindow {
56    /// The last [`RECENT_DAYS`] days.
57    fn default() -> Self {
58        Self::recent(Some(RECENT_DAYS))
59    }
60}
61
62impl SessionWindow {
63    /// A window of the last `days` days, or of every session for `None`.
64    #[must_use]
65    pub fn recent(days: Option<i64>) -> Self {
66        Self {
67            reach: Reach::Recent,
68            days,
69            older: None,
70        }
71    }
72
73    /// The `days` days up to `newest`, the last activity of the newest
74    /// session, and on to now: for a window of recent days that found
75    /// none of its own.
76    #[must_use]
77    pub fn rolled_back(newest: Timestamp, days: i64) -> Self {
78        Self {
79            reach: Reach::RolledBack(newest),
80            days: Some(days),
81            older: None,
82        }
83    }
84
85    /// The newest session's last activity a rolled-back window reaches up
86    /// to; `None` for any other window.
87    #[must_use]
88    pub fn rolled_back_to(&self) -> Option<Timestamp> {
89        match self.reach {
90            Reach::RolledBack(newest) => Some(newest),
91            _ => None,
92        }
93    }
94
95    /// Whether the window is the plain recent days, counted from now.
96    #[must_use]
97    pub fn is_recent(&self) -> bool {
98        self.reach == Reach::Recent && self.days.is_some()
99    }
100
101    /// The sessions last active from `low` to before `high`, and every
102    /// pinned, unread, and busy one; it offers nothing older.
103    #[must_use]
104    pub fn between(low: Timestamp, high: Timestamp) -> Self {
105        Self {
106            reach: Reach::Between(low, high),
107            days: None,
108            older: None,
109        }
110    }
111
112    /// How many days the window reaches back before older pages came in;
113    /// `None`: every session, or a range.
114    #[must_use]
115    pub fn days(&self) -> Option<i64> {
116        self.days
117    }
118    /// `params` with the window a full read of the list asks for at `now`.
119    /// A daemon older than [`LIST_WINDOW_SINCE`] gets none, and lists
120    /// everything.
121    #[must_use]
122    pub fn read_params(
123        &self,
124        mut params: SessionListParams,
125        version: Option<SemVer>,
126        now: Timestamp,
127    ) -> SessionListParams {
128        if daemon_lists_by_activity(version) {
129            params.active_since = match self.reach {
130                Reach::Recent => self
131                    .days
132                    .and_then(|days| now.checked_sub(SignedDuration::from_hours(24 * days)).ok()),
133                Reach::Since(since) => Some(since),
134                Reach::All => None,
135                Reach::Between(low, high) => {
136                    params.before = Some(high);
137                    Some(low)
138                }
139                Reach::RolledBack(newest) => self.days.and_then(|days| {
140                    newest
141                        .checked_sub(SignedDuration::from_hours(24 * days))
142                        .ok()
143                }),
144            };
145        }
146        params
147    }
148
149    /// Records a full read's `older` cursor; a range has none to offer.
150    pub fn read(&mut self, older: Option<Timestamp>) {
151        self.older = older.filter(|_| !matches!(self.reach, Reach::Between(..)));
152    }
153
154    /// The parameters of the next older page, when the daemon has one.
155    #[must_use]
156    pub fn older_params(&self, mut params: SessionListParams) -> Option<SessionListParams> {
157        params.before = Some(self.older?);
158        params.active_since = None;
159        Some(params)
160    }
161
162    /// Whether the daemon has sessions older than the list shows.
163    #[must_use]
164    pub fn has_older(&self) -> bool {
165        self.older.is_some()
166    }
167
168    /// Records an older page's `older` cursor: the list now reaches back to
169    /// it, or to the first session once nothing is older.
170    pub fn paged(&mut self, older: Option<Timestamp>) {
171        self.reach = older.map_or(Reach::All, Reach::Since);
172        self.older = older;
173    }
174
175    /// Back to the window's recent days, or its range, as for another
176    /// daemon: a rolled-back window rolls back again only if the recent
177    /// days hold nothing there.
178    pub fn reset(&mut self) {
179        *self = match self.reach {
180            Reach::Between(low, high) => Self::between(low, high),
181            _ => Self::recent(self.days),
182        };
183    }
184}
185
186/// Whether a session counts toward a window by its own activity: not one
187/// listed whatever its age (pinned, unread, or busy).
188fn counts_by_activity(session: &SessionMeta) -> bool {
189    !(crate::rows::listed_first(session) || session.unread)
190}
191
192/// Whether an answer for the days since `since` holds no session of its
193/// own, only those listed whatever their age: then the window may roll
194/// back.
195#[must_use]
196pub fn holds_nothing_recent(sessions: &[SessionMeta], since: Timestamp) -> bool {
197    !sessions
198        .iter()
199        .any(|session| counts_by_activity(session) && session.last_activity >= since)
200}
201
202/// The last activity of the newest session in `sessions` that counts by
203/// its activity, which a window rolls back to.
204#[must_use]
205pub fn newest_activity(sessions: &[SessionMeta]) -> Option<Timestamp> {
206    sessions
207        .iter()
208        .filter(|session| counts_by_activity(session))
209        .map(|session| session.last_activity)
210        .max()
211}
212
213/// Adds the sessions of an older page to `sessions`, keeping the copy the
214/// list already holds of any session in both: it may be newer.
215pub fn merge_older(sessions: &mut Vec<SessionMeta>, page: Vec<SessionMeta>) {
216    for session in page {
217        if !sessions
218            .iter()
219            .any(|listed| listed.session_id == session.session_id)
220        {
221            sessions.push(session);
222        }
223    }
224}

236    #[test]
237    fn a_daemon_before_the_window_lists_everything() {
238        let window = SessionWindow::default();
239        let params = window.read_params(
240            SessionListParams::default(),
241            Some(SemVer::new(1, 56, 0)),
242            at("2026-10-31T00:00:00Z"),
243        );
244        assert_eq!(params, SessionListParams::default());
245        assert!(!daemon_lists_by_activity(None));
246    }

248    #[test]
249    fn the_list_reaches_back_a_page_at_a_time_and_then_to_everything() {
250        let now = at("2026-10-31T12:00:00Z");
251        let mut window = SessionWindow::default();
252        assert_eq!(
253            window
254                .read_params(SessionListParams::default(), WINDOWED, now)
255                .active_since,
256            Some(at("2026-10-01T12:00:00Z")),
257            "thirty days of twenty-four hours"
258        );
259        assert_eq!(window.older_params(SessionListParams::default()), None);
260
261        window.read(Some(at("2026-10-01T12:00:00Z")));
262        assert!(window.has_older());
263        let page = window
264            .older_params(SessionListParams::default())
265            .expect("an older page");
266        assert_eq!(page.before, Some(at("2026-10-01T12:00:00Z")));
267        assert_eq!(page.active_since, None);
268
269        // The page reached back to September's start: a later read keeps
270        // what it showed.
271        window.paged(Some(at("2026-09-01T00:00:00Z")));
272        assert_eq!(
273            window
274                .read_params(SessionListParams::default(), WINDOWED, now)
275                .active_since,
276            Some(at("2026-09-01T00:00:00Z"))
277        );
278
279        // Nothing is older: every later read lists everything.
280        window.paged(None);
281        assert!(!window.has_older());
282        let params = window.read_params(SessionListParams::default(), WINDOWED, now);
283        assert_eq!(params.active_since, None);
284        assert_eq!(params.before, None);
285
286        window.reset();
287        assert_eq!(window, SessionWindow::default());
288    }

290    #[test]
291    fn a_window_rolls_back_to_the_days_before_the_newest_session() {
292        let newest = at("2026-09-20T15:00:00Z");
293        let now = at("2026-10-31T12:00:00Z");
294        let mut window = SessionWindow::rolled_back(newest, 7);
295        assert_eq!(window.rolled_back_to(), Some(newest));
296        assert!(!window.is_recent());
297        let params = window.read_params(SessionListParams::default(), WINDOWED, now);
298        assert_eq!(params.active_since, Some(at("2026-09-13T15:00:00Z")));
299        assert_eq!(params.before, None, "on to now");
300        window.reset();
301        assert_eq!(window, SessionWindow::recent(Some(7)));
302        assert!(window.is_recent());
303        assert!(
304            !SessionWindow::recent(None).is_recent(),
305            "every session has nothing to roll back"
306        );
307    }

309    #[test]
310    fn only_sessions_listed_by_their_activity_count_for_a_roll_back() {
311        let since = at("2026-10-24T12:00:00Z");
312        let mut old = session("old");
313        old.last_activity = at("2026-09-20T15:00:00Z");
314        let mut pinned = session("pinned");
315        pinned.last_activity = at("2026-10-01T00:00:00Z");
316        pinned.pinned = true;
317        let mut unread_now = session("unread");
318        unread_now.last_activity = at("2026-10-30T00:00:00Z");
319        unread_now.unread = true;
320        let listed = [old.clone(), pinned.clone(), unread_now.clone()];
321        assert!(
322            holds_nothing_recent(&listed, since),
323            "only the unread one is recent"
324        );
325        assert_eq!(
326            newest_activity(&listed),
327            Some(old.last_activity),
328            "the pinned one is newer"
329        );
330        let mut recent = session("recent");
331        recent.last_activity = at("2026-10-25T00:00:00Z");
332        assert!(!holds_nothing_recent(&[recent], since));
333        assert_eq!(newest_activity(&[pinned, unread_now]), None);
334    }

337    fn a_range_asks_for_its_two_ends_and_offers_nothing_older() {
338        let (low, high) = (at("2026-09-03T00:00:00Z"), at("2026-09-11T00:00:00Z"));
339        let mut range = SessionWindow::between(low, high);

345        assert_eq!(

352        assert_eq!(range.older_params(SessionListParams::default()), None);

366    #[test]
367    fn a_window_reaches_back_its_own_days_or_to_everything() {
368        let now = at("2026-10-31T12:00:00Z");
369        let mut week = SessionWindow::recent(Some(7));
370        assert_eq!(
371            week.read_params(SessionListParams::default(), WINDOWED, now)
372                .active_since,
373            Some(at("2026-10-24T12:00:00Z"))
374        );
375        // Older pages, then a reset: the week again, not thirty days.
376        week.paged(Some(at("2026-09-01T00:00:00Z")));
377        week.reset();
378        assert_eq!(week, SessionWindow::recent(Some(7)));
379
380        let every = SessionWindow::recent(None);
381        let params = every.read_params(SessionListParams::default(), WINDOWED, now);
382        assert_eq!(params.active_since, None);
383        assert_eq!(params.before, None);
384        assert_eq!(every.days(), None);
385    }
