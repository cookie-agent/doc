1//! The skills a client lists (`skills.list`): grouped by where they come
2//! from, searched by name and description, and what a refresh of the
3//! official skills (`skills.update`) did, in words.
4
5use cookie_agent_protocol::{SkillDescriptor, SkillSource, SkillsUpdateResult};
6
7/// The order the groups show in: the user's own skills, the workspace's,
8/// then the official ones, which either of the others overrides by name.
9pub const SOURCES: [SkillSource; 3] = [
10    SkillSource::User,
11    SkillSource::Project,
12    SkillSource::Official,
13];
14
15/// A group's heading.
16#[must_use]
17pub const fn source_label(source: SkillSource) -> &'static str {
18    match source {
19        SkillSource::User => "Your skills",
20        SkillSource::Project => "This workspace's skills",
21        SkillSource::Official => "Official skills",
22    }
23}
24
25/// Whether `skill`'s name or description holds `query` (any case); every
26/// skill for a blank query.
27#[must_use]
28pub fn skill_matches(skill: &SkillDescriptor, query: &str) -> bool {
29    let query = query.trim().to_lowercase();
30    query.is_empty()
31        || skill.name.to_lowercase().contains(&query)
32        || skill.description.to_lowercase().contains(&query)
33}
34
35/// The skills matching `query`, grouped in [`SOURCES`] order, each group by
36/// name; groups with none are left out.
37#[must_use]
38pub fn skill_groups<'a>(
39    skills: &'a [SkillDescriptor],
40    query: &str,
41) -> Vec<(SkillSource, Vec<&'a SkillDescriptor>)> {
42    SOURCES
43        .into_iter()
44        .filter_map(|source| {
45            let mut group = skills
46                .iter()
47                .filter(|skill| skill.source == source && skill_matches(skill, query))
48                .collect::<Vec<_>>();
49            group.sort_by(|left, right| left.name.cmp(&right.name));
50            (!group.is_empty()).then_some((source, group))
51        })
52        .collect()
53}

55/// What a refresh of the official skills did, for a status line, and
56/// whether it failed.
57#[must_use]
58pub fn update_summary(result: &SkillsUpdateResult) -> (String, bool) {
59    if !result.enabled {
60        return (
61            "Official skills are off ([skills] official = false).".to_owned(),
62            false,
63        );
64    }
65    let mut parts = Vec::new();
66    if !result.updated.is_empty() {
67        parts.push(format!("updated {}", result.updated.join(", ")));
68    }
69    if !result.removed.is_empty() {
70        parts.push(format!("removed {}", result.removed.join(", ")));
71    }
72    let done = if parts.is_empty() {
73        "The official skills are up to date.".to_owned()
74    } else {
75        let mut text = parts.join("; ");
76        if let Some(first) = text.get(..1) {
77            text = first.to_uppercase() + &text[1..];
78        }
79        format!("{text}.")
80    };
81    match &result.error {
82        Some(error) => (format!("{done} The refresh failed: {error}"), true),
83        None => (done, false),
84    }
85}

91    fn skill(name: &str, description: &str, source: SkillSource) -> SkillDescriptor {
92        SkillDescriptor {
93            name: name.to_owned(),
94            description: description.to_owned(),
95            when_to_use: None,
96            location: format!("/skills/{name}/SKILL.md"),
97            source,
98            precedence_winner: true,
99            permission_effect: cookie_agent_protocol::PermissionEffect::Allow,
100            visible: true,
101            user_invocable: true,
102            argument_hint: None,
103            version: None,
104        }
105    }
106
107    fn names(groups: &[(SkillSource, Vec<&SkillDescriptor>)]) -> Vec<(SkillSource, Vec<String>)> {
108        groups
109            .iter()
110            .map(|(source, skills)| {
111                (
112                    *source,
113                    skills.iter().map(|skill| skill.name.clone()).collect(),
114                )
115            })
116            .collect()
117    }

119    #[test]
120    fn skills_group_by_source_in_order_and_by_name() {
121        let skills = [
122            skill("zeta", "", SkillSource::Official),
123            skill("deploy", "Ship it", SkillSource::User),
124            skill("alpha", "", SkillSource::Official),
125            skill("review", "Read a diff", SkillSource::Project),
126        ];
127        assert_eq!(
128            names(&skill_groups(&skills, "")),
129            [
130                (SkillSource::User, vec!["deploy".to_owned()]),
131                (SkillSource::Project, vec!["review".to_owned()]),
132                (
133                    SkillSource::Official,
134                    vec!["alpha".to_owned(), "zeta".to_owned()]
135                ),
136            ]
137        );
138        // The search looks in descriptions too, in any case, and leaves
139        // out the groups it empties.
140        assert_eq!(
141            names(&skill_groups(&skills, "  DIFF ")),
142            [(SkillSource::Project, vec!["review".to_owned()])]
143        );
144        assert!(skill_groups(&skills, "nothing").is_empty());
145    }

147    fn result(updated: &[&str], removed: &[&str], error: Option<&str>) -> SkillsUpdateResult {
148        SkillsUpdateResult {
149            enabled: true,
150            skills: Vec::new(),
151            updated: updated.iter().map(|name| (*name).to_owned()).collect(),
152            removed: removed.iter().map(|name| (*name).to_owned()).collect(),
153            error: error.map(str::to_owned),
154        }
155    }

157    #[test]
158    fn an_update_says_what_changed_and_whether_it_failed() {
159        let (quiet, failed) = update_summary(&result(&[], &[], None));
160        assert!(!failed);
161        let (changed, _) = update_summary(&result(&["a", "b"], &["c"], None));
162        assert!(
163            changed.contains("a, b") && changed.contains('c'),
164            "{changed}"
165        );
166        assert_ne!(quiet, changed);
167        let (text, failed) = update_summary(&result(&[], &[], Some("offline")));
168        assert!(failed && text.contains("offline"), "{text}");
169        // With official skills off, nothing was refreshed, whatever it lists.
170        let mut off = result(&["listed"], &[], None);
171        off.enabled = false;
172        let (text, failed) = update_summary(&off);
173        assert!(
174            !failed && !text.contains("listed") && text != quiet,
175            "{text}"
176        );
177    }
