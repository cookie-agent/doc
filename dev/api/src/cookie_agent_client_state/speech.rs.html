1//! Voice input on the client side (`docs/site/specs/voice-input.md`
2//! sections 3, 4, and 7): sending a recording to be transcribed, the
3//! recorder's format, and the words a dictation adds to a draft.
4
5use cookie_agent_protocol::{
6    ARTIFACT_CHUNK_BYTES, SessionId, Sha256Digest, SpeechStatus, SpeechTranscribeBeginParams,
7    SpeechTranscribeCancelParams, SpeechTranscribeChunkParams, SpeechTranscribeFinishParams,
8    SpeechTranscribeFinishResult, UploadId, errors,
9};
10use cookie_agent_protocol_transport::{Client, ClientError};

12/// The containers a recorder may make, best first: Opus in WebM (Chrome,
13/// Firefox), Opus in Ogg, then MP4 audio (Safari).
14pub const RECORDER_TYPES: &[&str] = &[
15    "audio/webm;codecs=opus",
16    "audio/webm",
17    "audio/ogg;codecs=opus",
18    "audio/mp4",
19];
20
21/// The first of [`RECORDER_TYPES`] the browser `supports`; `None` leaves
22/// the choice to it.
23pub fn recorder_type(supports: impl Fn(&str) -> bool) -> Option<&'static str> {
24    RECORDER_TYPES.iter().copied().find(|kind| supports(kind))
25}
26
27/// Sends `bytes`, a recording of `media_type`, and waits for its words.
28/// `begun` gets the transcription's ID as soon as it has one, so the
29/// caller can cancel it.
30pub async fn transcribe(
31    client: &Client,
32    bytes: &[u8],
33    media_type: &str,
34    session_id: Option<SessionId>,
35    begun: impl FnOnce(UploadId),
36) -> Result<SpeechTranscribeFinishResult, ClientError> {
37    let started = client
38        .begin_transcription(SpeechTranscribeBeginParams {
39            byte_length: bytes.len() as u64,
40            media_type: media_type.to_owned(),
41            session_id,
42        })
43        .await?;
44    let id = started.transcription_id;
45    begun(id.clone());
46    let chunk =
47        usize::try_from(started.chunk_bytes.clamp(1, ARTIFACT_CHUNK_BYTES)).unwrap_or(usize::MAX);
48    let mut offset = 0;
49    for piece in bytes.chunks(chunk) {
50        use base64::Engine as _;
51        let sent = client
52            .transcription_chunk(SpeechTranscribeChunkParams {
53                transcription_id: id.clone(),
54                offset: offset as u64,
55                data: base64::engine::general_purpose::STANDARD.encode(piece),
56            })
57            .await;
58        if let Err(error) = sent {
59            cancel(client, id).await;
60            return Err(error);
61        }
62        offset += piece.len();
63    }
64    client
65        .finish_transcription(SpeechTranscribeFinishParams {
66            transcription_id: id,
67            sha256: Sha256Digest::of_bytes(bytes).to_string(),
68        })
69        .await
70}

72/// Drops a recording, or abandons its transcription; a recording already
73/// gone is no error.
74pub async fn cancel(client: &Client, transcription_id: UploadId) {
75    let _ = client
76        .cancel_transcription(SpeechTranscribeCancelParams { transcription_id })
77        .await;
78}

80/// Whether `error` is a cancellation the user asked for, which says
81/// nothing.
82#[must_use]
83pub fn cancelled(error: &ClientError) -> bool {
84    error.is(&errors::SPEECH_CANCELLED)
85}

87/// What the recording bar says when a transcription fails.
88#[must_use]
89pub fn failure_text(error: &ClientError) -> String {
90    if error.is(&errors::SPEECH_FAILED) {
91        let reason = error.error_data().and_then(|data| {
92            data.fields
93                .get("reason")
94                .and_then(|value| value.as_str())
95                .map(str::to_owned)
96        });
97        if let Some(reason) = reason {
98            return reason;
99        }
100    }
101    if error.is(&errors::SPEECH_UNAVAILABLE) {
102        return "voice input was turned off on the daemon".to_owned();
103    }
104    error.to_string()
105}

107/// `draft` with `words` dictated at its end, a space between them.
108#[must_use]
109pub fn add_words(draft: &str, words: &str) -> String {
110    let words = words.trim();
111    if words.is_empty() {
112        return draft.to_owned();
113    }
114    if draft.is_empty() || draft.ends_with(char::is_whitespace) {
115        format!("{draft}{words}")
116    } else {
117        format!("{draft} {words}")
118    }
119}

121/// A recording's length as the bar shows it, `m:ss`.
122#[must_use]
123pub fn clock(seconds: u64) -> String {
124    format!("{}:{:02}", seconds / 60, seconds % 60)
125}
126
127/// The recording limit, if any, the bar shows after the time.
128#[must_use]
129pub fn limit_text(status: &SpeechStatus) -> Option<String> {
130    status
131        .max_seconds
132        .map(|seconds| format!("/ {}", clock(seconds)))
133}

135#[cfg(test)]
136mod tests {
137    use super::*;
138
139    #[test]
140    fn dictated_words_join_the_draft_with_one_space() {
141        assert_eq!(add_words("", "  add a test "), "add a test");
142        assert_eq!(
143            add_words("Fix it", "and run the check."),
144            "Fix it and run the check."
145        );
146        assert_eq!(add_words("Fix it\n", "then push"), "Fix it\nthen push");
147        assert_eq!(add_words("Fix it ", "then push"), "Fix it then push");
148        assert_eq!(add_words("Fix it", "   "), "Fix it");
149    }
150
151    #[test]
152    fn the_clock_shows_minutes_and_seconds() {
153        assert_eq!(clock(0), "0:00");
154        assert_eq!(clock(59), "0:59");
155        assert_eq!(clock(60), "1:00");
156        assert_eq!(clock(605), "10:05");
157        assert_eq!(
158            limit_text(&SpeechStatus {
159                max_seconds: Some(300),
160                max_bytes: None
161            })
162            .as_deref(),
163            Some("/ 5:00")
164        );
165        assert_eq!(limit_text(&SpeechStatus::default()), None);
166    }
167
168    #[test]
169    fn the_recorder_takes_the_first_type_the_browser_supports() {
170        assert_eq!(recorder_type(|_| true), Some("audio/webm;codecs=opus"));
171        assert_eq!(
172            recorder_type(|kind| kind.starts_with("audio/mp4")),
173            Some("audio/mp4")
174        );
175        assert_eq!(recorder_type(|_| false), None);
176    }
