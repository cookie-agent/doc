12/// The containers a recorder may make, best first: Opus in WebM (Chrome,
13/// Firefox), Opus in Ogg, then MP4 audio (Safari).
14pub const RECORDER_TYPES: &[&str] = &[
15    "audio/webm;codecs=opus",
16    "audio/webm",
17    "audio/ogg;codecs=opus",
18    "audio/mp4",
19];

21/// The first of [`RECORDER_TYPES`] the browser `supports`; `None` leaves
22/// the choice to it.
23pub fn recorder_type(supports: impl Fn(&str) -> bool) -> Option<&'static str> {
24    RECORDER_TYPES.iter().copied().find(|kind| supports(kind))
25}

30/// `samples` at `from` Hz, at `to` Hz: each output sample averages the
31/// input it covers when going down (a crude low-pass, enough for speech),
32/// and interpolates between neighbours when going up.
33#[must_use]
34pub fn resample(samples: &[f32], from: f64, to: u32) -> Vec<f32> {
35    if samples.is_empty() || from <= 0.0 || to == 0 {
36        return Vec::new();
37    }
38    let ratio = from / f64::from(to);
39    if (ratio - 1.0).abs() < 1e-9 {
40        return samples.to_vec();
41    }
42    let count = (samples.len() as f64 / ratio).floor() as usize;
43    (0..count)
44        .map(|index| {
45            let start = index as f64 * ratio;
46            if ratio > 1.0 {
47                let first = start as usize;
48                let last = (((index + 1) as f64 * ratio) as usize).clamp(first + 1, samples.len());
49                let window = &samples[first..last];
50                window.iter().sum::<f32>() / window.len() as f32
51            } else {
52                let first = start as usize;
53                let fraction = (start - first as f64) as f32;
54                let a = samples[first];
55                let b = samples.get(first + 1).copied().unwrap_or(a);
56                a + (b - a) * fraction
57            }
58        })
59        .collect()
60}

62/// `samples` (-1 to 1, mono, at `rate` Hz) as a 16-bit PCM WAV file.
63#[must_use]
64pub fn encode_wav(samples: &[f32], rate: u32) -> Vec<u8> {
65    let data_bytes = u32::try_from(samples.len() * 2).unwrap_or(u32::MAX - 36);
66    let mut wav = Vec::with_capacity(44 + samples.len() * 2);
67    wav.extend_from_slice(b"RIFF");
68    wav.extend_from_slice(&(36 + data_bytes).to_le_bytes());
69    wav.extend_from_slice(b"WAVEfmt ");
70    wav.extend_from_slice(&16u32.to_le_bytes());
71    wav.extend_from_slice(&1u16.to_le_bytes()); // PCM
72    wav.extend_from_slice(&1u16.to_le_bytes()); // mono
73    wav.extend_from_slice(&rate.to_le_bytes());
74    wav.extend_from_slice(&(rate * 2).to_le_bytes()); // bytes a second
75    wav.extend_from_slice(&2u16.to_le_bytes()); // bytes a frame
76    wav.extend_from_slice(&16u16.to_le_bytes()); // bits a sample
77    wav.extend_from_slice(b"data");
78    wav.extend_from_slice(&data_bytes.to_le_bytes());
79    for sample in samples {
80        let value = (sample.clamp(-1.0, 1.0) * f32::from(i16::MAX)).round() as i16;
81        wav.extend_from_slice(&value.to_le_bytes());
82    }
83    wav
84}

86/// Sends `bytes`, a recording of `media_type`, and waits for its words.
87/// `begun` gets the transcription's ID as soon as it has one, so the
88/// caller can cancel it.
89pub async fn transcribe(
90    client: &Client,
91    bytes: &[u8],
92    media_type: &str,
93    session_id: Option<SessionId>,
94    begun: impl FnOnce(UploadId),
95) -> Result<SpeechTranscribeFinishResult, ClientError> {
96    let started = client
97        .begin_transcription(SpeechTranscribeBeginParams {
98            byte_length: bytes.len() as u64,
99            media_type: media_type.to_owned(),
100            session_id,
101        })
102        .await?;
103    let id = started.transcription_id;
104    begun(id.clone());
105    let chunk =
106        usize::try_from(started.chunk_bytes.clamp(1, ARTIFACT_CHUNK_BYTES)).unwrap_or(usize::MAX);
107    let mut offset = 0;
108    for piece in bytes.chunks(chunk) {
109        use base64::Engine as _;
110        let sent = client
111            .transcription_chunk(SpeechTranscribeChunkParams {
112                transcription_id: id.clone(),
113                offset: offset as u64,
114                data: base64::engine::general_purpose::STANDARD.encode(piece),
115            })
116            .await;
117        if let Err(error) = sent {
118            cancel(client, id).await;
119            return Err(error);
120        }
121        offset += piece.len();
122    }
123    client
124        .finish_transcription(SpeechTranscribeFinishParams {
125            transcription_id: id,
126            sha256: Sha256Digest::of_bytes(bytes).to_string(),
127        })
128        .await
129}

131/// Drops a recording, or abandons its transcription; a recording already
132/// gone is no error.
133pub async fn cancel(client: &Client, transcription_id: UploadId) {
134    let _ = client
135        .cancel_transcription(SpeechTranscribeCancelParams { transcription_id })
136        .await;
137}

139/// Whether `error` is a cancellation the user asked for, which says
140/// nothing.
141#[must_use]
142pub fn cancelled(error: &ClientError) -> bool {
143    error.is(&errors::SPEECH_CANCELLED)
144}

146/// What the recording bar says when a transcription fails.
147#[must_use]
148pub fn failure_text(error: &ClientError) -> String {
149    if error.is(&errors::SPEECH_FAILED) {
150        let reason = error.error_data().and_then(|data| {
151            data.fields
152                .get("reason")
153                .and_then(|value| value.as_str())
154                .map(str::to_owned)
155        });
156        if let Some(reason) = reason {
157            return reason;
158        }
159    }
160    if error.is(&errors::SPEECH_UNAVAILABLE) {
161        return "voice input was turned off on the daemon".to_owned();
162    }
163    error.to_string()
164}

166/// `draft` with `words` dictated at its end, a space between them.
167#[must_use]
168pub fn add_words(draft: &str, words: &str) -> String {
169    let words = words.trim();
170    if words.is_empty() {
171        return draft.to_owned();
172    }
173    if draft.is_empty() || draft.ends_with(char::is_whitespace) {
174        format!("{draft}{words}")
175    } else {
176        format!("{draft} {words}")
177    }
178}

180/// A recording's length as the bar shows it, `m:ss`.
181#[must_use]
182pub fn clock(seconds: u64) -> String {
183    format!("{}:{:02}", seconds / 60, seconds % 60)
184}
185
186/// The recording limit, if any, the bar shows after the time.
187#[must_use]
188pub fn limit_text(status: &SpeechStatus) -> Option<String> {
189    status
190        .max_seconds
191        .map(|seconds| format!("/ {}", clock(seconds)))
192}
