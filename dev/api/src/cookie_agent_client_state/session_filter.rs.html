1//! The session list's filter (`docs/site/specs/session-pins-and-filters.md`):
2//! how far back the list reaches, and which agents and machines its
3//! sessions run on. The date is the list's window
4//! ([`crate::session_window`]), which the daemon answers with every pinned,
5//! unread, and busy session too; the agents and the machines narrow what
6//! the list holds, those sessions included.
7

15/// How far back the list reaches: a number of days back from now, every
16/// session, or the days between two dates.
17#[derive(Clone, Copy, Debug, Default, Eq, Hash, PartialEq)]
18pub enum ActiveWithin {
19    Day,
20    #[default]
21    Week,
22    Month,
23    Quarter,
24    /// Every session.
25    Ever,
26    /// The sessions last active on the days of a range.
27    Between(DateRange),
28}
29
30impl ActiveWithin {
31    /// The choices that reach back from now, in the order the filter
32    /// offers them; a range comes after them.
33    pub const ALL: [Self; 5] = [
34        Self::Day,
35        Self::Week,
36        Self::Month,
37        Self::Quarter,
38        Self::Ever,
39    ];
40
41    /// The days it reaches back from now; `None`: every session, or a
42    /// range.
43    #[must_use]
44    pub const fn days(self) -> Option<i64> {
45        match self {
46            Self::Day => Some(1),
47            Self::Week => Some(7),
48            Self::Month => Some(30),
49            Self::Quarter => Some(90),
50            Self::Ever | Self::Between(_) => None,
51        }
52    }
53
54    /// Its name in the filter.
55    #[must_use]
56    pub const fn label(self) -> &'static str {
57        match self {
58            Self::Day => "24 hours",
59            Self::Week => "7 days",
60            Self::Month => "30 days",
61            Self::Quarter => "90 days",
62            Self::Ever => "All time",
63            Self::Between(_) => "Dates…",
64        }
65    }
66
67    /// What the list's summary says of it, `today` being the browser's
68    /// date: "Last 7 days", or a range's dates.
69    #[must_use]
70    pub fn summary(self, today: Date) -> String {
71        match self {
72            Self::Day => "Last 24 hours".to_owned(),
73            Self::Week => "Last 7 days".to_owned(),
74            Self::Month => "Last 30 days".to_owned(),
75            Self::Quarter => "Last 90 days".to_owned(),
76            Self::Ever => "All time".to_owned(),
77            Self::Between(range) => range.label(today),
78        }
79    }
80
81    /// The summary short enough to share a line with more: "7 days", or a
82    /// range's dates.
83    #[must_use]
84    pub fn short(self, today: Date) -> String {
85        match self {
86            Self::Between(range) => range.label(today),
87            preset => preset.label().to_owned(),
88        }
89    }
90
91    /// A stable name for it, as a page's `data-` attribute says it.
92    #[must_use]
93    pub const fn id(self) -> &'static str {
94        match self {
95            Self::Day => "day",
96            Self::Week => "week",
97            Self::Month => "month",
98            Self::Quarter => "quarter",
99            Self::Ever => "ever",
100            Self::Between(_) => "between",
101        }
102    }
103
104    /// The list's window for it, a range's days counted in `zone`.
105    #[must_use]
106    pub fn window(self, zone: &TimeZone) -> SessionWindow {
107        match self {
108            Self::Between(range) => match range.bounds(zone) {
109                Some((low, high)) => SessionWindow::between(low, high),
110                None => SessionWindow::recent(None),
111            },
112            preset => SessionWindow::recent(preset.days()),
113        }
114    }
115}

117/// The days from one date to another, both included.
118#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
119pub struct DateRange {
120    pub from: Date,
121    pub to: Date,
122}
123
124impl DateRange {
125    /// The days between `one` and `other`, whichever comes first.
126    #[must_use]
127    pub fn new(one: Date, other: Date) -> Self {
128        Self {
129            from: one.min(other),
130            to: one.max(other),
131        }
132    }
133
134    /// The last `days` days up to `today`, both included.
135    #[must_use]
136    pub fn ending(today: Date, days: i64) -> Self {
137        let from = today
138            .checked_sub(jiff::Span::new().days(days.max(1) - 1))
139            .unwrap_or(today);
140        Self::new(from, today)
141    }
142
143    /// From the start of the first day to the start of the day after the
144    /// last, in `zone`; `None` past the ends of time.
145    #[must_use]
146    pub fn bounds(self, zone: &TimeZone) -> Option<(Timestamp, Timestamp)> {
147        let low = self.from.to_zoned(zone.clone()).ok()?.timestamp();
148        let high = self
149            .to
150            .tomorrow()
151            .ok()?
152            .to_zoned(zone.clone())
153            .ok()?
154            .timestamp();
155        Some((low, high))
156    }
157
158    /// "Oct 3 – Oct 10", "Oct 3" for one day, with the year when either
159    /// date is not in `today`'s.
160    #[must_use]
161    pub fn label(self, today: Date) -> String {
162        let format = if self.from.year() == today.year() && self.to.year() == today.year() {
163            "%b %-d"
164        } else {
165            "%b %-d, %Y"
166        };
167        let from = self.from.strftime(format).to_string();
168        if self.from == self.to {
169            return from;
170        }
171        format!("{from} – {}", self.to.strftime(format))
172    }
173}

175/// Whether the list keeps `session` for a range ending at `high` and
176/// starting at `low`: pinned, unread, and busy sessions whatever their age,
177/// as the daemon lists them, and the rest only when last active in it, so
178/// that a session announced since (one just created, say) stays out of a
179/// range in the past.
180#[must_use]
181pub fn in_range(session: &SessionMeta, (low, high): (Timestamp, Timestamp)) -> bool {
182    crate::rows::listed_first(session)
183        || session.unread
184        || (low <= session.last_activity && session.last_activity < high)
185}

187/// What the list shows: sessions active within [`Self::within`], run by
188/// one of [`Self::agents`] (any while empty), on one of
189/// [`Self::machines`] (any while empty).
190#[derive(Clone, Debug, Default, Eq, PartialEq)]
191pub struct SessionFilter {
192    pub within: ActiveWithin,
193    /// Agent IDs, as `creation_selection.agent` names them.
194    pub agents: BTreeSet<String>,
195    /// Machine names, as [`crate::badges::session_machines`] gives them.
196    pub machines: BTreeSet<String>,
197}
198
199impl SessionFilter {
200    /// Whether `session`, working on `machines`, passes the agents and the
201    /// machines. The date is the daemon's to apply.
202    #[must_use]
203    pub fn matches(&self, session: &SessionMeta, machines: &[String]) -> bool {
204        (self.agents.is_empty()
205            || self
206                .agents
207                .contains(session.creation_selection.agent.as_str()))
208            && (self.machines.is_empty()
209                || machines
210                    .iter()
211                    .any(|machine| self.machines.contains(machine)))
212    }
213
214    /// Whether only the date narrows the list: no agent or machine is
215    /// chosen.
216    #[must_use]
217    pub fn by_date_only(&self) -> bool {
218        self.agents.is_empty() && self.machines.is_empty()
219    }
220
221    /// How many of its parts differ from the default filter's.
222    #[must_use]
223    pub fn changed(&self) -> usize {
224        usize::from(self.within != ActiveWithin::default())
225            + usize::from(!self.agents.is_empty())
226            + usize::from(!self.machines.is_empty())
227    }
228
229    /// Chooses `agent`, or lets it go when chosen.
230    pub fn toggle_agent(&mut self, agent: &str) {
231        if !self.agents.remove(agent) {
232            self.agents.insert(agent.to_owned());
233        }
234    }
235
236    /// Chooses `machine`, or lets it go when chosen.
237    pub fn toggle_machine(&mut self, machine: &str) {
238        if !self.machines.remove(machine) {
239            self.machines.insert(machine.to_owned());
240        }
241    }
242}

244/// The agents a filter offers: those that run a listed root session, and
245/// those `chosen` already, so each can be let go; sorted, each once.
246#[must_use]
247pub fn agent_options<'a>(
248    sessions: impl IntoIterator<Item = &'a SessionMeta>,
249    chosen: &BTreeSet<String>,
250) -> Vec<String> {
251    let mut options = chosen.clone();
252    options.extend(
253        sessions
254            .into_iter()
255            .filter(|session| matches!(session.origin, SessionOrigin::Root))
256            .map(|session| session.creation_selection.agent.to_string()),
257    );
258    options.into_iter().collect()
259}
260
261/// The machines a filter offers: those `machines` names for the listed
262/// sessions, and those `chosen` already; sorted, each once.
263#[must_use]
264pub fn machine_options<I, M>(machines: I, chosen: &BTreeSet<String>) -> Vec<String>
265where
266    I: IntoIterator<Item = M>,
267    M: IntoIterator<Item = String>,
268{
269    let mut options = chosen.clone();
270    options.extend(machines.into_iter().flatten());
271    options.into_iter().collect()
272}

330    #[test]
331    fn a_range_runs_from_its_first_days_start_to_its_last_days_end() {
332        let range = DateRange::new(date("2026-09-10"), date("2026-09-03"));
333        assert_eq!(
334            (range.from, range.to),
335            (date("2026-09-03"), date("2026-09-10"))
336        );
337        let zone = TimeZone::fixed(jiff::tz::offset(2));
338        assert_eq!(
339            range.bounds(&zone),
340            Some((at("2026-09-02T22:00:00Z"), at("2026-09-10T22:00:00Z"))),
341            "local midnights, the last day whole"
342        );
343        assert_eq!(
344            DateRange::ending(date("2026-09-10"), 7),
345            DateRange::new(date("2026-09-04"), date("2026-09-10"))
346        );
347        assert_eq!(
348            ActiveWithin::Between(range).window(&TimeZone::UTC),
349            SessionWindow::between(at("2026-09-03T00:00:00Z"), at("2026-09-11T00:00:00Z"))
350        );
351        assert_eq!(
352            ActiveWithin::Week.window(&TimeZone::UTC),
353            SessionWindow::recent(Some(7))
354        );
355    }

405    #[test]
406    fn agents_and_machines_each_narrow_and_any_of_them_passes() {
407        let claude = session("claude", false);
408        let primary = session("primary", false);
409        let mut filter = SessionFilter::default();
410        assert!(filter.by_date_only());
411        assert!(filter.matches(&claude, &[]), "nothing chosen passes all");
412
413        filter.toggle_agent("claude");
414        assert!(filter.matches(&claude, &[]));
415        assert!(!filter.matches(&primary, &[]));
416        filter.toggle_agent("primary");
417        assert!(filter.matches(&primary, &[]), "either agent passes");
418
419        filter.toggle_machine("labpc");
420        assert!(filter.matches(&claude, &names(&["laptop", "labpc"])));
421        assert!(!filter.matches(&claude, &names(&["laptop"])));
422        assert!(!filter.matches(&claude, &[]), "no machine is not labpc");
423        assert_eq!(filter.changed(), 2);
424
425        filter.toggle_machine("labpc");
426        filter.toggle_agent("claude");
427        filter.toggle_agent("primary");
428        filter.within = ActiveWithin::Ever;
429        assert!(filter.by_date_only());
430        assert_eq!(filter.changed(), 1);
431    }
