1//! Pinned root sessions (`docs/site/specs/session-pins-and-filters.md`):
2//! the daemon keeps them; clients show `SessionMeta::pinned`, list them
3//! with the sessions at work, and pin with `session.pin.set`.
4
5use cookie_agent_protocol::SemVer;
6
7/// The first client protocol whose daemons keep pinned sessions.
8pub const PINS_SINCE: SemVer = SemVer::new(1, 66, 0);
9
10/// Whether the daemon, speaking `version`, keeps pinned sessions.
11#[must_use]
12pub fn daemon_keeps_pins(version: Option<SemVer>) -> bool {
13    version.is_some_and(|version| version >= PINS_SINCE)
14}
15
16/// What a status line says after pinning or unpinning `count` sessions.
17#[must_use]
18pub fn pinned_message(count: usize, pinned: bool) -> String {
19    let (state, done) = if pinned {
20        ("pinned", "Pinned")
21    } else {
22        ("unpinned", "Unpinned")
23    };
24    match count {
25        0 => format!("Nothing changed: already {state}."),
26        1 => format!("{done} 1 session."),
27        count => format!("{done} {count} sessions."),
28    }
29}
