1//! Questions an agent asks the user (client protocol 1.25.0; ACP
2//! elicitation, `docs/site/specs/acp-agents.md` section 15): what the
3//! clients keep of a question still waiting, the user's draft answer to a
4//! form, and the words both clients show.
5//!
6//! A session's waiting questions come from its `elicitation_requested`
7//! events; `elicitation_resolved` or the run's end removes one.
8
9use cookie_agent_protocol::{
10    ElicitationAction, ElicitationAnsweredBy, ElicitationContent, ElicitationField,
11    ElicitationForm, ElicitationInput, ElicitationRequest, ElicitationResolution, ElicitationValue,
12    RunId, SessionId,
13};

15/// A question waiting for the user's answer.
16#[derive(Clone, Debug, PartialEq)]
17pub struct QuestionState {
18    pub session_id: SessionId,
19    pub run_id: Option<RunId>,
20    pub request: ElicitationRequest,
21}
22
23impl QuestionState {
24    /// What kind of question it is.
25    #[must_use]
26    pub fn kind(&self) -> QuestionKind<'_> {
27        match (&self.request.form, &self.request.url) {
28            (Some(form), _) => QuestionKind::Form(form),
29            (None, Some(url)) => QuestionKind::Url(url),
30            (None, None) => QuestionKind::Unsupported,
31        }
32    }
33
34    /// The heading both clients show: who asks.
35    #[must_use]
36    pub fn heading(&self) -> String {
37        format!("{} asks", self.request.agent)
38    }
39}

41/// A question's kind.
42#[derive(Clone, Copy, Debug, PartialEq)]
43pub enum QuestionKind<'a> {
44    Form(&'a ElicitationForm),
45    Url(&'a str),
46    /// A kind a newer daemon asks: it can only be declined or cancelled.
47    Unsupported,
48}

50/// One field's draft value.
51#[derive(Clone, Debug, PartialEq)]
52pub enum DraftValue {
53    /// Text, a number, or a whole number, as typed.
54    Text(String),
55    /// The chosen choice's index.
56    Choice(Option<usize>),
57    /// Whether each choice is chosen.
58    MultiChoice(Vec<bool>),
59    Boolean(bool),
60    /// A field of a kind this build cannot show.
61    Unsupported,
62}

64/// The user's draft answer to a form: one value per field, in the form's
65/// order, starting from the fields' defaults.
66#[derive(Clone, Debug, PartialEq)]
67pub struct FormDraft {
68    pub values: Vec<DraftValue>,
69}

71/// A field whose draft value does not fit it, by its index.
72#[derive(Clone, Debug, Eq, PartialEq)]
73pub struct FieldProblem {
74    pub index: usize,
75    pub message: String,
76}
77
78impl FormDraft {
79    /// A draft of `form` with its fields' defaults.
80    #[must_use]
81    pub fn new(form: &ElicitationForm) -> Self {
82        Self {
83            values: form.fields.iter().map(default_value).collect(),
84        }
85    }
86
87    /// The value field `index` of `form` would answer with: `None` for a
88    /// field left empty, an error for a value that does not fit.
89    pub fn value(
90        &self,
91        form: &ElicitationForm,
92        index: usize,
93    ) -> Result<Option<ElicitationValue>, String> {
94        let (Some(field), Some(draft)) = (form.fields.get(index), self.values.get(index)) else {
95            return Ok(None);
96        };
97        let value = match (field.input.known(), draft) {
98            (_, DraftValue::Unsupported) | (None, _) => None,
99            (Some(ElicitationInput::Text { .. }), DraftValue::Text(text)) => {
100                (!text.is_empty()).then(|| ElicitationValue::Text(text.clone()))
101            }
102            (Some(ElicitationInput::Number { .. }), DraftValue::Text(text)) => match text.trim() {
103                "" => None,
104                text => Some(ElicitationValue::Number(
105                    text.parse::<f64>()
106                        .ok()
107                        .filter(|number| number.is_finite())
108                        .ok_or_else(|| format!("{} must be a number", field.label()))?,
109                )),
110            },
111            (Some(ElicitationInput::Integer { .. }), DraftValue::Text(text)) => match text.trim() {
112                "" => None,
113                text => Some(ElicitationValue::Integer(
114                    text.parse::<i64>()
115                        .map_err(|_| format!("{} must be a whole number", field.label()))?,
116                )),
117            },
118            (Some(ElicitationInput::Choice { choices, .. }), DraftValue::Choice(chosen)) => chosen
119                .and_then(|index| choices.get(index))
120                .map(|choice| ElicitationValue::Text(choice.value.clone())),
121            (
122                Some(ElicitationInput::MultiChoice { choices, .. }),
123                DraftValue::MultiChoice(chosen),
124            ) => {
125                let values = choices
126                    .iter()
127                    .zip(chosen)
128                    .filter(|(_, chosen)| **chosen)
129                    .map(|(choice, _)| choice.value.clone())
130                    .collect::<Vec<_>>();
131                (!values.is_empty()).then_some(ElicitationValue::List(values))
132            }
133            (Some(ElicitationInput::Boolean { .. }), DraftValue::Boolean(value)) => {
134                Some(ElicitationValue::Boolean(*value))
135            }
136            _ => None,
137        };
138        field.check(value.as_ref())?;
139        Ok(value)
140    }
141
142    /// The answer the draft makes, or the first field that does not fit.
143    pub fn content(&self, form: &ElicitationForm) -> Result<ElicitationContent, FieldProblem> {
144        let mut content = ElicitationContent::new();
145        for (index, field) in form.fields.iter().enumerate() {
146            match self.value(form, index) {
147                Ok(Some(value)) => {
148                    content.insert(field.name.clone(), value);
149                }
150                Ok(None) => {}
151                Err(message) => return Err(FieldProblem { index, message }),
152            }
153        }
154        Ok(content)
155    }
156}

158fn default_value(field: &ElicitationField) -> DraftValue {
159    match field.input.known() {
160        None => DraftValue::Unsupported,
161        Some(ElicitationInput::Text { default, .. }) => {
162            DraftValue::Text(default.clone().unwrap_or_default())
163        }
164        Some(ElicitationInput::Number { default, .. }) => {
165            DraftValue::Text(default.map(|number| number.to_string()).unwrap_or_default())
166        }
167        Some(ElicitationInput::Integer { default, .. }) => {
168            DraftValue::Text(default.map(|number| number.to_string()).unwrap_or_default())
169        }
170        Some(ElicitationInput::Choice { choices, default }) => DraftValue::Choice(
171            default
172                .as_ref()
173                .and_then(|default| choices.iter().position(|choice| &choice.value == default)),
174        ),
175        Some(ElicitationInput::MultiChoice {
176            choices, default, ..
177        }) => DraftValue::MultiChoice(
178            choices
179                .iter()
180                .map(|choice| default.contains(&choice.value))
181                .collect(),
182        ),
183        Some(ElicitationInput::Boolean { default }) => {
184            DraftValue::Boolean(default.unwrap_or(false))
185        }
186    }
187}

189/// What a field asks for beyond its description, in a few words: its
190/// bounds, format, and pattern, and whether it is required.
191#[must_use]
192pub fn field_hint(field: &ElicitationField) -> String {
193    let mut parts = Vec::new();
194    if field.required {
195        parts.push("required".to_owned());
196    }
197    match field.input.known() {
198        None => parts.push("this client cannot fill it in".to_owned()),
199        Some(ElicitationInput::Text {
200            min_length,
201            max_length,
202            pattern,
203            format,
204            ..
205        }) => {
206            if let Some(format) = format {
207                parts.push(cookie_agent_protocol::format_description(format).to_owned());
208            }
209            match (min_length, max_length) {
210                (Some(min), Some(max)) => parts.push(format!("{min} to {max} characters")),
211                (Some(min), None) => parts.push(format!("at least {min} characters")),
212                (None, Some(max)) => parts.push(format!("at most {max} characters")),
213                (None, None) => {}
214            }
215            if let Some(pattern) = pattern {
216                parts.push(format!("matching {pattern}"));
217            }
218        }
219        Some(ElicitationInput::Number {
220            minimum, maximum, ..
221        }) => parts.extend(bounds("a number", *minimum, *maximum)),
222        Some(ElicitationInput::Integer {
223            minimum, maximum, ..
224        }) => parts.extend(bounds("a whole number", *minimum, *maximum)),
225        Some(ElicitationInput::MultiChoice {
226            min_items,
227            max_items,
228            ..
229        }) => match (min_items, max_items) {
230            (Some(min), Some(max)) => parts.push(format!("choose {min} to {max}")),
231            (Some(min), None) => parts.push(format!("choose at least {min}")),
232            (None, Some(max)) => parts.push(format!("choose at most {max}")),
233            (None, None) => parts.push("choose any".to_owned()),
234        },
235        Some(ElicitationInput::Choice { .. } | ElicitationInput::Boolean { .. }) => {}
236    }
237    parts.join(" · ")
238}

240fn bounds<T: std::fmt::Display>(what: &str, minimum: Option<T>, maximum: Option<T>) -> Vec<String> {
241    vec![match (minimum, maximum) {
242        (Some(min), Some(max)) => format!("{what} from {min} to {max}"),
243        (Some(min), None) => format!("{what}, at least {min}"),
244        (None, Some(max)) => format!("{what}, at most {max}"),
245        (None, None) => what.to_owned(),
246    }]
247}

249/// The host of `url` (without any user name or port), shown before the
250/// user agrees to open it.
251#[must_use]
252pub fn url_host(url: &str) -> Option<&str> {
253    let (_, rest) = url.split_once("://")?;
254    let authority = rest.split(['/', '?', '#']).next()?;
255    let host = authority
256        .rsplit_once('@')
257        .map_or(authority, |(_, host)| host);
258    let host = if host.starts_with('[') {
259        host.split_once(']').map_or(host, |(inner, _)| &inner[1..])
260    } else {
261        host.split(':').next().unwrap_or(host)
262    };
263    (!host.is_empty()).then_some(host)
264}

266/// What looks off about a URL the user is asked to open.
267#[must_use]
268pub fn url_warnings(url: &str) -> Vec<&'static str> {
269    let mut warnings = Vec::new();
270    let scheme = url
271        .split_once("://")
272        .map(|(scheme, _)| scheme.to_ascii_lowercase());
273    match scheme.as_deref() {
274        Some("https") => {}
275        Some("http") => warnings.push("it is not HTTPS"),
276        _ => warnings.push("it is not a web address"),
277    }
278    let host = url_host(url).unwrap_or_default();
279    if host
280        .split('.')
281        .any(|label| label.to_ascii_lowercase().starts_with("xn--"))
282        || !host.is_ascii()
283    {
284        warnings.push("its host has non-Latin letters (punycode): check it is the site you expect");
285    }
286    if url
287        .split_once("://")
288        .and_then(|(_, rest)| rest.split(['/', '?', '#']).next())
289        .is_some_and(|authority| authority.contains('@'))
290    {
291        warnings.push("it carries a user name before the host");
292    }
293    warnings
294}

296/// The words for how a question was answered: `accepted`, `declined`, or
297/// `cancelled`, by whom when the daemon answered, and its reason.
298#[must_use]
299pub fn resolution_text(resolution: &ElicitationResolution) -> String {
300    let action = match resolution.action {
301        ElicitationAction::Accept => "accepted",
302        ElicitationAction::Decline => "declined",
303        ElicitationAction::Cancel => "cancelled",
304        ElicitationAction::Unknown => "answered",
305    };
306    let by = match resolution.answered_by {
307        ElicitationAnsweredBy::User => "",
308        ElicitationAnsweredBy::Unattended => " (no one attends the session)",
309        ElicitationAnsweredBy::RunEnded => " (the run ended)",
310        ElicitationAnsweredBy::Completed => " (the agent finished it)",
311        ElicitationAnsweredBy::Unsupported => " (it cannot be shown)",
312        ElicitationAnsweredBy::Unknown => " (by the daemon)",
313    };
314    match &resolution.reason {
315        Some(reason) => format!("{action}{by}: {reason}"),
316        None => format!("{action}{by}"),
317    }
318}

320/// What the user answered `question` with: each answered field's title
321/// (else its name) and value, one per line; empty for an answer with no
322/// values.
323pub fn answer_text(
324    question: &QuestionState,
325    content: &cookie_agent_protocol::ElicitationContent,
326) -> String {
327    use cookie_agent_protocol::ElicitationValue;
328    let title = |name: &str| {
329        question
330            .request
331            .form
332            .as_ref()
333            .and_then(|form| form.fields.iter().find(|field| field.name == name))
334            .and_then(|field| field.title.clone())
335            .unwrap_or_else(|| name.to_owned())
336    };
337    content
338        .iter()
339        .map(|(name, value)| {
340            let value = match value {
341                ElicitationValue::Boolean(value) => if *value { "yes" } else { "no" }.to_owned(),
342                ElicitationValue::Integer(value) => value.to_string(),
343                ElicitationValue::Number(value) => value.to_string(),
344                ElicitationValue::Text(value) => value.clone(),
345                ElicitationValue::List(values) => values.join(", "),
346            };
347            format!("{}: {value}", title(name))
348        })
349        .collect::<Vec<_>>()
350        .join("\n")
351}
