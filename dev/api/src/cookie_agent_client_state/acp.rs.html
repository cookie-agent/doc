1//! ACP agents in the clients (`docs/site/specs/acp-agents.md` section 13,
2//! phase 4): the agents a runtime hosts in the agent pickers, with their
3//! runtime and availability, and the synthetic model they run under.
4
5use cookie_agent_protocol::{
6    ACP_MODEL_PROVIDER, AgentDescriptor, AgentMode, ModelSelection, ToolSource,
7};
8
9/// Whether an agent picker for root sessions of `preset` lists `agent`:
10/// every agent that can run as a root, and an ACP agent that could but is
11/// unavailable now (its runtime is away, or it is disabled), which the
12/// picker shows dimmed and does not let the user pick.
13pub fn listed_in_root_picker(agent: &AgentDescriptor, preset: Option<&str>) -> bool {
14    agent.mode != AgentMode::Internal
15        && agent.preset.as_deref() == preset
16        && (agent.runnable_as_root
17            || (agent.runtime.is_some()
18                && matches!(agent.mode, AgentMode::Primary | AgentMode::All)))
19}
20
21/// Whether the user can pick `agent` in a root picker.
22pub fn pickable(agent: &AgentDescriptor) -> bool {
23    agent.runnable_as_root
24}
25
26/// The model a session of ACP agent `agent` records, `acp/<agent>`; `None`
27/// for the daemon's own agents.
28pub fn acp_model(agent: &AgentDescriptor) -> Option<ModelSelection> {
29    agent.runtime.as_ref()?;
30    agent
31        .resolved_fallback
32        .first()
33        .cloned()
34        .or_else(|| Some(cookie_agent_protocol::acp_model_selection(&agent.id)))
35}
36
37/// Whether `selection` is an ACP agent's synthetic model, which no model
38/// picker lists.
39pub fn is_acp_model(selection: &ModelSelection) -> bool {
40    selection.model.provider_id().as_str() == ACP_MODEL_PROVIDER
41}
42
43/// What a picker shows after an ACP agent's description: its runtime, and
44/// why it cannot be picked when it cannot. `None` for the daemon's own
45/// agents.
46pub fn runtime_note(agent: &AgentDescriptor) -> Option<String> {
47    let runtime = agent.runtime.as_ref()?;
48    Some(if !runtime.available {
49        format!("on {} · unavailable", runtime.runtime_id)
50    } else if !agent.runnable_as_root {
51        format!("on {} · disabled", runtime.runtime_id)
52    } else {
53        format!("on {}", runtime.runtime_id)
54    })
55}
56
57/// Why picking `agent` is refused, for a status line.
58pub fn unpickable_reason(agent: &AgentDescriptor) -> String {
59    match &agent.runtime {
60        Some(runtime) if !runtime.available => format!(
61            "`{}` is unavailable: runtime `{}` is not connected",
62            agent.id, runtime.runtime_id
63        ),
64        Some(runtime) => format!(
65            "`{}` is disabled on runtime `{}`",
66            agent.id, runtime.runtime_id
67        ),
68        None => format!("`{}` cannot run as a root agent", agent.id),
69    }
70}
71
72/// The label of a tool call's source when it is an ACP agent's own tool:
73/// the agent and its runtime.
74pub fn acp_source_label(source: &ToolSource) -> Option<String> {
75    match source {
76        ToolSource::Acp { agent, runtime_id } => Some(format!("{agent} on {runtime_id}")),
77        _ => None,
78    }
79}

81#[cfg(test)]
82mod tests {
83    use cookie_agent_protocol::{AgentDescriptor, AgentId, AgentMode, AgentRuntime, RuntimeId};
84
85    use super::*;
86
87    fn agent(
88        id: &str,
89        runtime: Option<(&str, bool)>,
90        runnable: bool,
91        mode: AgentMode,
92    ) -> AgentDescriptor {
93        let id = AgentId::new(id).unwrap();
94        AgentDescriptor {
95            resolved_fallback: if runtime.is_some() {
96                vec![cookie_agent_protocol::acp_model_selection(&id)]
97            } else {
98                Vec::new()
99            },
100            runtime: runtime.map(|(runtime_id, available)| AgentRuntime {
101                runtime_id: RuntimeId::new(runtime_id).unwrap(),
102                agent: AgentId::new("claude").unwrap(),
103                available,
104            }),
105            id,
106            preset: None,
107            description: "An agent".into(),
108            mode,
109            enabled: true,
110            runnable_as_root: runnable,
111            delegation_targets: Vec::new(),
112        }
113    }
114
115    #[test]
116    fn an_unavailable_acp_agent_is_listed_but_not_pickable() {
117        let away = agent("lab-claude", Some(("lab", false)), false, AgentMode::All);
118        assert!(listed_in_root_picker(&away, None));
119        assert!(!pickable(&away));
120        assert_eq!(runtime_note(&away).as_deref(), Some("on lab · unavailable"));
121        assert!(unpickable_reason(&away).contains("runtime `lab` is not connected"));
122        // A subagent-only ACP agent is never a root choice.
123        let sub = agent("lab-codex", Some(("lab", true)), false, AgentMode::Subagent);
124        assert!(!listed_in_root_picker(&sub, None));
125        // A daemon agent that cannot run as a root is not listed.
126        let daemon = agent("reviewer", None, false, AgentMode::All);
127        assert!(!listed_in_root_picker(&daemon, None));
128        assert_eq!(runtime_note(&daemon), None);
129        // Another preset's agents are not listed.
130        assert!(!listed_in_root_picker(&away, Some("python")));
131    }
132
133    #[test]
134    fn an_acp_agent_runs_its_synthetic_model() {
135        let local = agent("local-claude", Some(("local", true)), true, AgentMode::All);
136        let model = acp_model(&local).unwrap();
137        assert_eq!(model.model.as_str(), "acp/local-claude");
138        assert!(is_acp_model(&model));
139        assert_eq!(runtime_note(&local).as_deref(), Some("on local"));
140        assert_eq!(
141            acp_model(&agent("primary", None, true, AgentMode::Primary)),
142            None
143        );
144    }
