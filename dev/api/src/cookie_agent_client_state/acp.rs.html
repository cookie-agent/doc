14/// Whether `agent` is an ACP agent: some runtime offers it.
15pub fn is_acp(agent: &AgentDescriptor) -> bool {
16    agent.runtime.is_some() || !agent.runtimes.is_empty()
17}

19/// The runtimes that offer ACP agent `agent`: `runtimes`, else the one
20/// `runtime` a daemon older than client protocol 1.21.0 sends.
21pub fn offering_runtimes(agent: &AgentDescriptor) -> Vec<&AgentRuntime> {
22    if agent.runtimes.is_empty() {
23        agent.runtime.iter().collect()
24    } else {
25        agent.runtimes.iter().collect()
26    }
27}

29/// Where a session that attaches `attached` would run an ACP agent
30/// (`docs/site/specs/acp-several-runtimes.md` section 3.1).
31#[derive(Clone, Debug, Eq, PartialEq)]
32pub enum AcpPlacement {
33    /// On this runtime.
34    Runs(RuntimeId),
35    /// The runtime it must run on is away, or has it disabled.
36    Unavailable(RuntimeId),
37    /// No attached runtime offers it, and none was chosen; these do.
38    NeedsOneOf(Vec<RuntimeId>),
39}

41/// The declaration ACP agent `agent` runs, when its document names
42/// another one than its own ID (`acp.agent`, addendum of 2026-10-25).
43pub fn declaration(agent: &AgentDescriptor) -> Option<&AgentId> {
44    offering_runtimes(agent)
45        .first()
46        .map(|offer| &offer.agent)
47        .filter(|declaration| **declaration != agent.id)
48}

50/// Where a session or draft that attaches `attached` (in attach order)
51/// would run `agent`; `None` for the daemon's own agents. `host` is the
52/// runtime it must run on: the one its session started it on, else the one
53/// the user chose, attached or not (the daemon attaches a chosen one).
54/// Without one, the first attached runtime that offers it and can run it.
55pub fn placement(
56    agent: &AgentDescriptor,
57    attached: &[RuntimeId],
58    host: Option<&RuntimeId>,
59) -> Option<AcpPlacement> {
60    let offering = offering_runtimes(agent);
61    if offering.is_empty() {
62        return None;
63    }
64    let on = |offer: &AgentRuntime| {
65        if offer.available {
66            AcpPlacement::Runs(offer.runtime_id.clone())
67        } else {
68            AcpPlacement::Unavailable(offer.runtime_id.clone())
69        }
70    };
71    if let Some(host) = host {
72        return Some(
73            offering
74                .iter()
75                .find(|offer| &offer.runtime_id == host)
76                .map_or_else(
77                    || AcpPlacement::Unavailable(host.clone()),
78                    |offer| on(offer),
79                ),
80        );
81    }
82    let here = attached
83        .iter()
84        .filter_map(|runtime| offering.iter().find(|offer| &offer.runtime_id == runtime))
85        .collect::<Vec<_>>();
86    Some(
87        match here
88            .iter()
89            .find(|offer| offer.available)
90            .or_else(|| here.first())
91        {
92            Some(offer) => on(offer),
93            None => AcpPlacement::NeedsOneOf(
94                offering
95                    .iter()
96                    .map(|offer| offer.runtime_id.clone())
97                    .collect(),
98            ),
99        },
100    )
101}

103/// The runtime a draft that attaches `attached` runs `agent` on when the
104/// user chooses none: the first attached runtime that offers it and can
105/// run it, else the first that offers it and can run it at all.
106pub fn default_runtime(agent: &AgentDescriptor, attached: &[RuntimeId]) -> Option<RuntimeId> {
107    let offering = offering_runtimes(agent);
108    attached
109        .iter()
110        .find(|runtime| {
111            offering
112                .iter()
113                .any(|offer| &offer.runtime_id == *runtime && offer.available)
114        })
115        .cloned()
116        .or_else(|| {
117            offering
118                .iter()
119                .find(|offer| offer.available)
120                .map(|offer| offer.runtime_id.clone())
121        })
122}

141/// Whether an agent picker for root sessions of `preset` lists `agent`:
142/// every agent that can run as a root, and a root agent that cannot run
143/// now (no model, an ACP agent no runtime can run), which the picker shows
144/// dimmed, below the others, with the reason, and does not let the user
145/// pick. An agent the user turned off is not listed
146/// (`docs/site/specs/acp-onboarding.md` section 8.2, D6).
147pub fn listed_in_root_picker(agent: &AgentDescriptor, preset: Option<&str>) -> bool {
148    agent.mode != AgentMode::Internal
149        && agent.preset.as_deref() == preset
150        && !turned_off(agent)
151        && (agent.runnable_as_root
152            || (matches!(agent.mode, AgentMode::Primary | AgentMode::All)
153                && (is_acp(agent) || agent.unavailable.is_some())))
154}

156/// Whether the user turned `agent` off: its document is disabled, or its
157/// ACP agent is turned off on every runtime.
158#[must_use]
159pub fn turned_off(agent: &AgentDescriptor) -> bool {
160    !agent.enabled
161        || agent.unavailable.as_ref().is_some_and(|unavailable| {
162            matches!(
163                unavailable.reason,
164                cookie_agent_protocol::AgentUnavailableReason::Disabled
165                    | cookie_agent_protocol::AgentUnavailableReason::AcpDeclarationDisabled
166            )
167        })
168}

170/// `agents` in picker order: the ones that can run as a root, then the
171/// others, each part in the order given; and where the second part starts,
172/// where pickers draw their divider.
173#[must_use]
174pub fn picker_order(agents: Vec<&AgentDescriptor>) -> (Vec<&AgentDescriptor>, usize) {
175    let (mut runnable, unavailable): (Vec<_>, Vec<_>) =
176        agents.into_iter().partition(|agent| agent.runnable_as_root);
177    let divider = runnable.len();
178    runnable.extend(unavailable);
179    (runnable, divider)
180}

191/// Whether the user can pick `agent` for a session or draft that attaches
192/// `attached`, with `host` as [`placement`] takes it: it runs as a root,
193/// and an ACP agent can run where it would, or, with no runtime chosen and
194/// none attached offering it, on [`default_runtime`].
195pub fn pickable(agent: &AgentDescriptor, attached: &[RuntimeId], host: Option<&RuntimeId>) -> bool {
196    agent.runnable_as_root
197        && match placement(agent, attached, host) {
198            None | Some(AcpPlacement::Runs(_)) => true,
199            Some(AcpPlacement::NeedsOneOf(_)) => default_runtime(agent, attached).is_some(),
200            Some(AcpPlacement::Unavailable(_)) => false,
201        }
202}

204/// What a picker shows after an ACP agent's description, for a session or
205/// draft that attaches `attached`: where it runs, or why it cannot, after
206/// the declaration it runs when that is not its own ID. `None` for the
207/// daemon's own agents.
208pub fn runtime_note(
209    agent: &AgentDescriptor,
210    attached: &[RuntimeId],
211    host: Option<&RuntimeId>,
212) -> Option<String> {
213    // An agent the daemon says cannot run shows why, ACP or not.
214    if !agent.runnable_as_root
215        && let Some(note) = unavailable_note(agent)
216    {
217        return Some(note);
218    }
219    let note = placement_note(agent, attached, host)?;
220    Some(match declaration(agent) {
221        Some(declaration) if note.starts_with("on ") => format!("{declaration} {note}"),
222        Some(declaration) => format!("{declaration} · {note}"),
223        None => note,
224    })
225}
226
227fn placement_note(
228    agent: &AgentDescriptor,
229    attached: &[RuntimeId],
230    host: Option<&RuntimeId>,
231) -> Option<String> {
232    Some(match placement(agent, attached, host)? {
233        AcpPlacement::Runs(runtime) if agent.runnable_as_root => format!("on {runtime}"),
234        AcpPlacement::Runs(runtime) => format!("on {runtime} · disabled"),
235        AcpPlacement::Unavailable(runtime) => {
236            let away = offering_runtimes(agent)
237                .iter()
238                .any(|offer| offer.runtime_id == runtime && !offer.available);
239            if away && agent.enabled {
240                format!("on {runtime} · unavailable")
241            } else {
242                format!("on {runtime} · disabled")
243            }
244        }
245        AcpPlacement::NeedsOneOf(runtimes) => match default_runtime(agent, attached) {
246            Some(runtime) => format!("on {runtime}"),
247            None => format!("needs one of {} connected", joined(&runtimes)),
248        },
249    })
250}

252/// Why picking `agent` for a session or draft that attaches `attached` is
253/// refused, for a status line.
254pub fn unpickable_reason(
255    agent: &AgentDescriptor,
256    attached: &[RuntimeId],
257    host: Option<&RuntimeId>,
258) -> String {
259    if !agent.runnable_as_root
260        && let Some(note) = unavailable_note(agent)
261    {
262        return format!("`{}` cannot run: {note}", agent.id);
263    }
264    match placement(agent, attached, host) {
265        Some(AcpPlacement::NeedsOneOf(runtimes)) => format!(
266            "`{}` needs one of these runtimes connected: {}",
267            agent.id,
268            joined(&runtimes)
269        ),
270        Some(AcpPlacement::Unavailable(runtime)) => format!(
271            "`{}` is unavailable on runtime `{runtime}`: the runtime is not connected, or the \
272             agent is disabled there",
273            agent.id
274        ),
275        Some(AcpPlacement::Runs(runtime)) => {
276            format!("`{}` is disabled on runtime `{runtime}`", agent.id)
277        }
278        None => format!("`{}` cannot run as a root agent", agent.id),
279    }
280}

307/// Whether agent command `name` is listed for the search `query` (without
308/// its leading `/`) once the query matches it: the agent's skills only for
309/// a query that starts with `$`, as Codex offers them, so a few dozen of
310/// them do not bury the agent's commands.

312    !is_skill_command(name) || query.trim_start().starts_with('$')

366/// The value after `option`'s current one, wrapping, for a key that cycles
367/// the agent's mode; `None` when it has fewer than two values.
368pub fn next_value(option: &AcpConfigOption) -> Option<&cookie_agent_protocol::AcpConfigValue> {
369    if option.values.len() < 2 {
370        return None;
371    }
372    let current = option
373        .values
374        .iter()
375        .position(|value| value.value == option.current_value);
376    let next = current.map_or(0, |index| (index + 1) % option.values.len());
377    option.values.get(next)
378}

421/// The runtime a draft or session that attaches `attached` runs ACP agent
422/// `agent` on, whatever its availability ([`placement`], else
423/// [`default_runtime`]).
424pub fn placed_runtime<'a>(
425    agent: &'a AgentDescriptor,
426    attached: &[RuntimeId],
427    host: Option<&RuntimeId>,
428) -> Option<&'a AgentRuntime> {
429    let runtime = match placement(agent, attached, host)? {
430        AcpPlacement::Runs(runtime) | AcpPlacement::Unavailable(runtime) => runtime,
431        AcpPlacement::NeedsOneOf(_) => default_runtime(agent, attached)?,
432    };
433    offering_runtimes(agent)
434        .into_iter()
435        .find(|offer| offer.runtime_id == runtime)
436}

452/// `options` with `choices` made their current values, where an option
453/// takes the value chosen.
454pub fn with_choices(
455    options: &[AcpConfigOption],
456    choices: &[AcpOptionChoice],
457) -> Vec<AcpConfigOption> {
458    let mut options = options.to_vec();
459    for choice in choices {
460        if let Some(option) = options.iter_mut().find(|option| {
461            option.id == choice.option_id
462                && option
463                    .values
464                    .iter()
465                    .any(|value| value.value == choice.value)
466        }) {
467            option.current_value.clone_from(&choice.value);
468        }
469    }
470    options
471}

505/// The runtime a session's ACP agent must run on, as [`placement`] takes
506/// `host`: the one its ACP session started on (`view`), while it runs the
507/// same agent, else the one its `selection` chose.
508pub fn session_host<'a>(
509    view: &'a AcpSessionView,
510    selection: Option<&'a cookie_agent_protocol::RunSelection>,
511) -> Option<&'a RuntimeId> {
512    let selection = selection?;
513    if view.agent.as_ref() == Some(&selection.agent)

516        return Some(runtime);
517    }
518    selection.acp_runtime.as_ref()

521/// The choices a draft makes for its agent's options, which it sends as
522/// `session.create`'s `agent_options`. Kept with the agent they were made
523/// for, so choosing another agent drops them.
524#[derive(Clone, Debug, Default, Eq, PartialEq)]
525pub struct DraftAgentOptions {
526    agent: Option<AgentId>,
527    choices: Vec<AcpOptionChoice>,
528}
529
530impl DraftAgentOptions {
531    /// The choices made for `agent`.
532    pub fn for_agent(&self, agent: &AgentId) -> &[AcpOptionChoice] {
533        if self.agent.as_ref() == Some(agent) {
534            &self.choices
535        } else {
536            &[]
537        }
538    }
539
540    /// Chooses `value` for option `option_id` of `agent`, replacing an
541    /// earlier choice of that option; choices made for another agent are
542    /// dropped.
543    pub fn choose(&mut self, agent: &AgentId, option_id: &str, value: &str) {
544        if self.agent.as_ref() != Some(agent) {
545            self.agent = Some(agent.clone());
546            self.choices.clear();
547        }
548        match self
549            .choices
550            .iter_mut()
551            .find(|choice| choice.option_id == option_id)
552        {
553            Some(choice) => value.clone_into(&mut choice.value),
554            None => self.choices.push(AcpOptionChoice {
555                option_id: option_id.to_owned(),
556                value: value.to_owned(),
557            }),
558        }
559    }
560
561    /// Forgets every choice, as a draft becomes a session or is dropped.
562    pub fn clear(&mut self) {
563        *self = Self::default();
564    }
565}

651        assert!(pickable(&claude, &local, None));
652        // Two attached runtimes offer it: the first in attach order runs it,

655        assert_eq!(
656            placement(&claude, &both, None),
657            Some(AcpPlacement::Runs(runtime("labpc")))

660            placement(&claude, &both, Some(&runtime("local"))),
661            Some(AcpPlacement::Runs(runtime("local")))
662        );
663        // A chosen runtime runs it even before it is attached.
664        assert_eq!(
665            placement(&claude, &local, Some(&runtime("labpc"))),
666            Some(AcpPlacement::Runs(runtime("labpc")))
667        );

670            placement(&claude, &both, Some(&runtime("gpu"))),
671            Some(AcpPlacement::Unavailable(runtime("gpu")))
672        );
673        assert!(!pickable(&claude, &both, Some(&runtime("gpu"))));
674        // Without a host, an attached runtime that cannot run it is passed
675        // over for one that can.
676        let gpu_first = [runtime("gpu"), runtime("local")];
677        assert_eq!(
678            placement(&claude, &gpu_first, None),
679            Some(AcpPlacement::Runs(runtime("local")))
680        );
681        // No attached runtime offers it: it can still be picked, and runs on
682        // the first that can run it, which the daemon attaches.
683        let elsewhere = [runtime("global")];
684        assert!(matches!(
685            placement(&claude, &elsewhere, None),
686            Some(AcpPlacement::NeedsOneOf(_))
687        ));
688        assert!(pickable(&claude, &elsewhere, None));
689        assert_eq!(default_runtime(&claude, &elsewhere), Some(runtime("labpc")));
690        assert_eq!(default_runtime(&claude, &local), Some(runtime("local")));
691        assert_eq!(
692            placed_runtime(&claude, &elsewhere, None).map(|offer| offer.runtime_id.clone()),
693            Some(runtime("labpc"))
694        );
695        assert_eq!(runs_on_choices(&claude).len(), 3);

938/// The server and tool of an ACP agent's call titled as Claude Code titles
939/// an MCP tool, `mcp__<server>__<tool>`.
940fn mcp_title(title: &str) -> Option<(&str, &str)> {
941    title
942        .strip_prefix("mcp__")?
943        .split_once("__")
944        .filter(|(server, tool)| !server.is_empty() && !tool.is_empty())
945}
946
947/// What a tool call of `session` belongs to, shown beside it where the
948/// user answers for it (`docs/site/specs/acp-permissions.md`, section 4):
949/// in a session an ACP agent runs, the server of an MCP tool the agent
950/// connected itself, whose name its title leaves out. `None` for anything else.
951#[must_use]
952pub fn tool_source_label(
953    session: &crate::SessionState,
954    tool: &crate::ToolCallState,
955) -> Option<String> {
956    session.acp.agent.as_ref()?;
957    match tool
958        .presentation
959        .tool
960        .source
961        .as_ref()
962        .and_then(cookie_agent_protocol::Open::known)
963    {
964        Some(cookie_agent_protocol::ToolSource::Acp { .. }) | None => {
965            mcp_title(tool.presentation.title.as_str()).map(|(server, _)| server.to_owned())
966        }
967        Some(_) => None,
968    }
969}
970
971/// The title a tool call shows where the user answers for it: an ACP
972/// agent's call of an MCP tool by the tool's own name, without the
973/// `mcp__<server>__` prefix its harness gives it (its server is shown
974/// beside it, [`tool_source_label`]).
975#[must_use]
976pub fn approval_title(title: &str) -> &str {
977    mcp_title(title).map_or(title, |(_, tool)| tool)
978}
