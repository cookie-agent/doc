14/// Whether `agent` is an ACP agent: some runtime offers it.
15pub fn is_acp(agent: &AgentDescriptor) -> bool {
16    agent.runtime.is_some() || !agent.runtimes.is_empty()
17}

19/// The runtimes that offer ACP agent `agent`: `runtimes`, else the one
20/// `runtime` a daemon older than client protocol 1.21.0 sends.
21pub fn offering_runtimes(agent: &AgentDescriptor) -> Vec<&AgentRuntime> {
22    if agent.runtimes.is_empty() {
23        agent.runtime.iter().collect()
24    } else {
25        agent.runtimes.iter().collect()
26    }
27}

29/// Where a session that attaches `attached` would run an ACP agent
30/// (`docs/site/specs/acp-several-runtimes.md` section 3.1).
31#[derive(Clone, Debug, Eq, PartialEq)]
32pub enum AcpPlacement {
33    /// On this runtime.
34    Runs(RuntimeId),
35    /// The runtime it must run on is away, or has it disabled.
36    Unavailable(RuntimeId),
37    /// No attached runtime offers it, and none was chosen; these do.
38    NeedsOneOf(Vec<RuntimeId>),
39}

41/// The declaration ACP agent `agent` runs, when its document names
42/// another one than its own ID (`acp.agent`, addendum of 2026-10-25).
43pub fn declaration(agent: &AgentDescriptor) -> Option<&AgentId> {
44    offering_runtimes(agent)
45        .first()
46        .map(|offer| &offer.agent)
47        .filter(|declaration| **declaration != agent.id)
48}

50/// Where a session or draft that attaches `attached` (in attach order)
51/// would run `agent`; `None` for the daemon's own agents. `host` is the
52/// runtime it must run on: the one its session started it on, else the one
53/// the user chose, attached or not (the daemon attaches a chosen one).
54/// Without one, the first attached runtime that offers it and can run it.
55pub fn placement(
56    agent: &AgentDescriptor,
57    attached: &[RuntimeId],
58    host: Option<&RuntimeId>,
59) -> Option<AcpPlacement> {
60    let offering = offering_runtimes(agent);
61    if offering.is_empty() {
62        return None;
63    }
64    let on = |offer: &AgentRuntime| {
65        if offer.available {
66            AcpPlacement::Runs(offer.runtime_id.clone())
67        } else {
68            AcpPlacement::Unavailable(offer.runtime_id.clone())
69        }
70    };
71    if let Some(host) = host {
72        return Some(
73            offering
74                .iter()
75                .find(|offer| &offer.runtime_id == host)
76                .map_or_else(
77                    || AcpPlacement::Unavailable(host.clone()),
78                    |offer| on(offer),
79                ),
80        );
81    }
82    let here = attached
83        .iter()
84        .filter_map(|runtime| offering.iter().find(|offer| &offer.runtime_id == runtime))
85        .collect::<Vec<_>>();
86    Some(
87        match here
88            .iter()
89            .find(|offer| offer.available)
90            .or_else(|| here.first())
91        {
92            Some(offer) => on(offer),
93            None => AcpPlacement::NeedsOneOf(
94                offering
95                    .iter()
96                    .map(|offer| offer.runtime_id.clone())
97                    .collect(),
98            ),
99        },
100    )
101}

103/// The runtime a draft that attaches `attached` runs `agent` on when the
104/// user chooses none: the first attached runtime that offers it and can
105/// run it, else the first that offers it and can run it at all.
106pub fn default_runtime(agent: &AgentDescriptor, attached: &[RuntimeId]) -> Option<RuntimeId> {
107    let offering = offering_runtimes(agent);
108    attached
109        .iter()
110        .find(|runtime| {
111            offering
112                .iter()
113                .any(|offer| &offer.runtime_id == *runtime && offer.available)
114        })
115        .cloned()
116        .or_else(|| {
117            offering
118                .iter()
119                .find(|offer| offer.available)
120                .map(|offer| offer.runtime_id.clone())
121        })
122}

124/// The runtimes that offer `agent`, for the Runs on choice: each with
125/// whether it can run the agent now. Empty for the daemon's own agents.
126pub fn runs_on_choices(agent: &AgentDescriptor) -> Vec<(RuntimeId, bool)> {
127    offering_runtimes(agent)
128        .into_iter()
129        .map(|offer| (offer.runtime_id.clone(), offer.available))

141/// Whether an agent picker for root sessions of `preset` lists `agent`:
142/// every agent that can run as a root, and an ACP agent that could but no
143/// runtime can run now, which the picker shows dimmed and does not let the
144/// user pick.
145pub fn listed_in_root_picker(agent: &AgentDescriptor, preset: Option<&str>) -> bool {
146    agent.mode != AgentMode::Internal
147        && agent.preset.as_deref() == preset
148        && (agent.runnable_as_root
149            || (is_acp(agent) && matches!(agent.mode, AgentMode::Primary | AgentMode::All)))
150}

152/// Whether the user can pick `agent` for a session or draft that attaches
153/// `attached`, with `host` as [`placement`] takes it: it runs as a root,
154/// and an ACP agent can run where it would, or, with no runtime chosen and
155/// none attached offering it, on [`default_runtime`].
156pub fn pickable(agent: &AgentDescriptor, attached: &[RuntimeId], host: Option<&RuntimeId>) -> bool {
157    agent.runnable_as_root
158        && match placement(agent, attached, host) {
159            None | Some(AcpPlacement::Runs(_)) => true,
160            Some(AcpPlacement::NeedsOneOf(_)) => default_runtime(agent, attached).is_some(),
161            Some(AcpPlacement::Unavailable(_)) => false,
162        }
163}

165/// What a picker shows after an ACP agent's description, for a session or
166/// draft that attaches `attached`: where it runs, or why it cannot, after
167/// the declaration it runs when that is not its own ID. `None` for the
168/// daemon's own agents.
169pub fn runtime_note(
170    agent: &AgentDescriptor,
171    attached: &[RuntimeId],
172    host: Option<&RuntimeId>,
173) -> Option<String> {
174    let note = placement_note(agent, attached, host)?;
175    Some(match declaration(agent) {
176        Some(declaration) if note.starts_with("on ") => format!("{declaration} {note}"),
177        Some(declaration) => format!("{declaration} · {note}"),
178        None => note,
179    })
180}

182fn placement_note(
183    agent: &AgentDescriptor,
184    attached: &[RuntimeId],
185    host: Option<&RuntimeId>,
186) -> Option<String> {
187    Some(match placement(agent, attached, host)? {
188        AcpPlacement::Runs(runtime) if agent.runnable_as_root => format!("on {runtime}"),
189        AcpPlacement::Runs(runtime) => format!("on {runtime} · disabled"),
190        AcpPlacement::Unavailable(runtime) => {
191            let away = offering_runtimes(agent)
192                .iter()
193                .any(|offer| offer.runtime_id == runtime && !offer.available);
194            if away && agent.enabled {
195                format!("on {runtime} · unavailable")
196            } else {
197                format!("on {runtime} · disabled")
198            }
199        }
200        AcpPlacement::NeedsOneOf(runtimes) => match default_runtime(agent, attached) {
201            Some(runtime) => format!("on {runtime}"),
202            None => format!("needs one of {} connected", joined(&runtimes)),
203        },
204    })
205}

207/// Why picking `agent` for a session or draft that attaches `attached` is
208/// refused, for a status line.
209pub fn unpickable_reason(
210    agent: &AgentDescriptor,
211    attached: &[RuntimeId],
212    host: Option<&RuntimeId>,
213) -> String {
214    match placement(agent, attached, host) {
215        Some(AcpPlacement::NeedsOneOf(runtimes)) => format!(
216            "`{}` needs one of these runtimes connected: {}",
217            agent.id,
218            joined(&runtimes)
219        ),
220        Some(AcpPlacement::Unavailable(runtime)) => format!(
221            "`{}` is unavailable on runtime `{runtime}`: the runtime is not connected, or the \
222             agent is disabled there",
223            agent.id
224        ),
225        Some(AcpPlacement::Runs(runtime)) => {
226            format!("`{}` is disabled on runtime `{runtime}`", agent.id)
227        }
228        None => format!("`{}` cannot run as a root agent", agent.id),
229    }
230}

251/// Whether agent command `name` is one of the agent's skills, which Codex
252/// lists as commands named `$<skill>` and runs from a `$<skill>` mention.
253pub fn is_skill_command(name: &str) -> bool {
254    name.starts_with('$')
255}

257/// Whether agent command `name` is listed for the search `query` (without
258/// its leading `/`) once the query matches it: the agent's skills only for
259/// a query that starts with `$`, as Codex offers them, so a few dozen of
260/// them do not bury the agent's commands.
261pub fn command_listed(name: &str, query: &str) -> bool {
262    !is_skill_command(name) || query.trim_start().starts_with('$')
263}

265/// How agent command `name` is written: `/<name>`, or the skill `$<skill>`
266/// as it is.
267pub fn command_label(name: &str) -> String {
268    if is_skill_command(name) {
269        name.to_owned()
270    } else {
271        format!("/{name}")
272    }

275/// The prompt that runs agent command `name` with `input`: `/<name>
276/// <input>`, as ACP specifies, or for a skill the mention `$<skill>
277/// <input>`, as the agent takes it.
278pub fn command_prompt(name: &str, input: &str) -> String {
279    let input = input.trim();
280    let label = command_label(name);

316/// The value after `option`'s current one, wrapping, for a key that cycles
317/// the agent's mode; `None` when it has fewer than two values.
318pub fn next_value(option: &AcpConfigOption) -> Option<&cookie_agent_protocol::AcpConfigValue> {
319    if option.values.len() < 2 {
320        return None;
321    }
322    let current = option
323        .values
324        .iter()
325        .position(|value| value.value == option.current_value);
326    let next = current.map_or(0, |index| (index + 1) % option.values.len());
327    option.values.get(next)
328}

330/// The name of `option`'s current value, else the value itself.
331pub fn current_name(option: &AcpConfigOption) -> String {
332    option
333        .values
334        .iter()
335        .find(|value| value.value == option.current_value)
336        .map_or_else(|| option.current_value.clone(), |value| value.name.clone())
337}

339/// What the model chip shows for an agent with `options`: its model's
340/// current value and, when it has one, its effort's, as the model and the
341/// variant of cookie's own models. `None` without a model option.
342pub fn model_and_effort(options: &[AcpConfigOption]) -> Option<(String, Option<String>)> {
343    let model = model_option(options)?;
344    Some((
345        current_name(model),
346        effort_option(options).map(current_name),
347    ))

350/// The model chip's label for an agent with `options`, in the form cookie

352pub fn model_label(options: &[AcpConfigOption]) -> Option<String> {
353    let (model, effort) = model_and_effort(options)?;

355        Some(effort) => format!("{model} • {effort}"),
356        None => model,

360/// The runtime a draft or session that attaches `attached` runs ACP agent
361/// `agent` on, whatever its availability ([`placement`], else
362/// [`default_runtime`]).
363pub fn placed_runtime<'a>(
364    agent: &'a AgentDescriptor,
365    attached: &[RuntimeId],
366    host: Option<&RuntimeId>,
367) -> Option<&'a AgentRuntime> {
368    let runtime = match placement(agent, attached, host)? {
369        AcpPlacement::Runs(runtime) | AcpPlacement::Unavailable(runtime) => runtime,
370        AcpPlacement::NeedsOneOf(_) => default_runtime(agent, attached)?,
371    };
372    offering_runtimes(agent)
373        .into_iter()
374        .find(|offer| offer.runtime_id == runtime)
375}

391/// `options` with `choices` made their current values, where an option
392/// takes the value chosen.
393pub fn with_choices(
394    options: &[AcpConfigOption],
395    choices: &[AcpOptionChoice],
396) -> Vec<AcpConfigOption> {
397    let mut options = options.to_vec();
398    for choice in choices {
399        if let Some(option) = options.iter_mut().find(|option| {
400            option.id == choice.option_id
401                && option
402                    .values
403                    .iter()
404                    .any(|value| value.value == choice.value)
405        }) {
406            option.current_value.clone_from(&choice.value);
407        }
408    }
409    options
410}

444/// The runtime a session's ACP agent must run on, as [`placement`] takes
445/// `host`: the one its ACP session started on (`view`), while it runs the
446/// same agent, else the one its `selection` chose.
447pub fn session_host<'a>(
448    view: &'a AcpSessionView,
449    selection: Option<&'a cookie_agent_protocol::RunSelection>,
450) -> Option<&'a RuntimeId> {
451    let selection = selection?;
452    if view.agent.as_ref() == Some(&selection.agent)
453        && let Some(runtime) = &view.runtime_id
454    {
455        return Some(runtime);
456    }
457    selection.acp_runtime.as_ref()
458}

460/// The choices a draft makes for its agent's options, which it sends as
461/// `session.create`'s `agent_options`. Kept with the agent they were made
462/// for, so choosing another agent drops them.
463#[derive(Clone, Debug, Default, Eq, PartialEq)]
464pub struct DraftAgentOptions {
465    agent: Option<AgentId>,
466    choices: Vec<AcpOptionChoice>,
467}
468
469impl DraftAgentOptions {
470    /// The choices made for `agent`.
471    pub fn for_agent(&self, agent: &AgentId) -> &[AcpOptionChoice] {
472        if self.agent.as_ref() == Some(agent) {
473            &self.choices
474        } else {
475            &[]
476        }
477    }
478
479    /// Chooses `value` for option `option_id` of `agent`, replacing an
480    /// earlier choice of that option; choices made for another agent are
481    /// dropped.
482    pub fn choose(&mut self, agent: &AgentId, option_id: &str, value: &str) {
483        if self.agent.as_ref() != Some(agent) {
484            self.agent = Some(agent.clone());
485            self.choices.clear();
486        }
487        match self
488            .choices
489            .iter_mut()
490            .find(|choice| choice.option_id == option_id)
491        {
492            Some(choice) => value.clone_into(&mut choice.value),
493            None => self.choices.push(AcpOptionChoice {
494                option_id: option_id.to_owned(),
495                value: value.to_owned(),
496            }),
497        }
498    }
499
500    /// Forgets every choice, as a draft becomes a session or is dropped.
501    pub fn clear(&mut self) {
502        *self = Self::default();
503    }
504}

839/// The server and tool of an ACP agent's call titled as Claude Code titles
840/// an MCP tool, `mcp__<server>__<tool>`.
841fn mcp_title(title: &str) -> Option<(&str, &str)> {
842    title
843        .strip_prefix("mcp__")?
844        .split_once("__")
845        .filter(|(server, tool)| !server.is_empty() && !tool.is_empty())
846}
847
848/// What a tool call of `session` belongs to, shown beside it where the
849/// user answers for it (`docs/site/specs/acp-permissions.md`, section 4):
850/// in a session an ACP agent runs, `cookie` for one of cookie's own tools,
851/// and the server of an MCP tool the agent connected itself. `None`
852/// elsewhere.
853#[must_use]
854pub fn tool_source_label(
855    session: &crate::SessionState,
856    tool: &crate::ToolCallState,
857) -> Option<String> {
858    session.acp.agent.as_ref()?;
859    match tool
860        .presentation
861        .tool
862        .source
863        .as_ref()
864        .and_then(cookie_agent_protocol::Open::known)
865    {
866        Some(cookie_agent_protocol::ToolSource::Acp { .. }) | None => {
867            mcp_title(tool.presentation.title.as_str()).map(|(server, _)| server.to_owned())
868        }
869        Some(_) => Some("cookie".to_owned()),
870    }
871}
872
873/// The title a tool call shows where the user answers for it: an ACP
874/// agent's call of an MCP tool by the tool's own name, without the
875/// `mcp__<server>__` prefix its harness gives it (its server is shown
876/// beside it, [`tool_source_label`]).
877#[must_use]
878pub fn approval_title(title: &str) -> &str {
879    mcp_title(title).map_or(title, |(_, tool)| tool)
880}
