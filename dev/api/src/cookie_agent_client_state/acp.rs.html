14/// Whether `agent` is an ACP agent: some runtime offers it.
15pub fn is_acp(agent: &AgentDescriptor) -> bool {
16    agent.runtime.is_some() || !agent.runtimes.is_empty()
17}

19/// The runtimes that offer ACP agent `agent`: `runtimes`, else the one
20/// `runtime` a daemon older than client protocol 1.21.0 sends.
21pub fn offering_runtimes(agent: &AgentDescriptor) -> Vec<&AgentRuntime> {
22    if agent.runtimes.is_empty() {
23        agent.runtime.iter().collect()
24    } else {
25        agent.runtimes.iter().collect()
26    }
27}

29/// Where a session that attaches `attached` would run an ACP agent
30/// (`docs/site/specs/acp-agents.md` section 15, naming of 2026-10-23).
31#[derive(Clone, Debug, Eq, PartialEq)]
32pub enum AcpPlacement {
33    /// On this runtime, the one attached that offers it.
34    Runs(RuntimeId),
35    /// The one attached runtime that offers it is away, or has it
36    /// disabled.
37    Unavailable(RuntimeId),
38    /// No attached runtime offers it; these do.
39    NeedsOneOf(Vec<RuntimeId>),
40    /// Several attached runtimes offer it, and a session runs it on one.
41    Ambiguous(Vec<RuntimeId>),
42}
43
44/// The declaration ACP agent `agent` runs, when its document names
45/// another one than its own ID (`acp.agent`, addendum of 2026-10-25).
46pub fn declaration(agent: &AgentDescriptor) -> Option<&AgentId> {
47    offering_runtimes(agent)
48        .first()
49        .map(|offer| &offer.agent)
50        .filter(|declaration| **declaration != agent.id)
51}

53/// Where a session or draft that attaches `attached` would run `agent`;
54/// `None` for the daemon's own agents.
55pub fn placement(agent: &AgentDescriptor, attached: &[RuntimeId]) -> Option<AcpPlacement> {
56    let offering = offering_runtimes(agent);
57    if offering.is_empty() {
58        return None;
59    }
60    let here = offering
61        .iter()
62        .filter(|offer| attached.contains(&offer.runtime_id))
63        .collect::<Vec<_>>();
64    Some(match here[..] {
65        [] => AcpPlacement::NeedsOneOf(
66            offering
67                .iter()
68                .map(|offer| offer.runtime_id.clone())
69                .collect(),
70        ),
71        [offer] if offer.available => AcpPlacement::Runs(offer.runtime_id.clone()),
72        [offer] => AcpPlacement::Unavailable(offer.runtime_id.clone()),
73        _ => AcpPlacement::Ambiguous(here.iter().map(|offer| offer.runtime_id.clone()).collect()),
74    })
75}

77fn joined(runtimes: &[RuntimeId]) -> String {
78    runtimes
79        .iter()
80        .map(RuntimeId::as_str)
81        .collect::<Vec<_>>()
82        .join(", ")
83}

85/// Whether an agent picker for root sessions of `preset` lists `agent`:
86/// every agent that can run as a root, and an ACP agent that could but no
87/// runtime can run now, which the picker shows dimmed and does not let the
88/// user pick.
89pub fn listed_in_root_picker(agent: &AgentDescriptor, preset: Option<&str>) -> bool {
90    agent.mode != AgentMode::Internal
91        && agent.preset.as_deref() == preset
92        && (agent.runnable_as_root
93            || (is_acp(agent) && matches!(agent.mode, AgentMode::Primary | AgentMode::All)))
94}

96/// Whether the user can pick `agent` for a session or draft that attaches
97/// `attached`: it runs as a root, and an ACP agent has exactly one of
98/// those runtimes offering it, available.
99pub fn pickable(agent: &AgentDescriptor, attached: &[RuntimeId]) -> bool {
100    agent.runnable_as_root
101        && placement(agent, attached)
102            .is_none_or(|placement| matches!(placement, AcpPlacement::Runs(_)))
103}

105/// What a picker shows after an ACP agent's description, for a session or
106/// draft that attaches `attached`: where it runs, or why it cannot, after
107/// the declaration it runs when that is not its own ID. `None` for the
108/// daemon's own agents.
109pub fn runtime_note(agent: &AgentDescriptor, attached: &[RuntimeId]) -> Option<String> {
110    let note = placement_note(agent, attached)?;
111    Some(match declaration(agent) {
112        Some(declaration) if note.starts_with("on ") => format!("{declaration} {note}"),
113        Some(declaration) => format!("{declaration} · {note}"),
114        None => note,
115    })
116}
117
118fn placement_note(agent: &AgentDescriptor, attached: &[RuntimeId]) -> Option<String> {
119    Some(match placement(agent, attached)? {
120        AcpPlacement::Runs(runtime) if agent.runnable_as_root => format!("on {runtime}"),
121        AcpPlacement::Runs(runtime) => format!("on {runtime} · disabled"),
122        AcpPlacement::Unavailable(runtime) => {
123            let away = offering_runtimes(agent)
124                .iter()
125                .any(|offer| offer.runtime_id == runtime && !offer.available);
126            if away && agent.enabled {
127                format!("on {runtime} · unavailable")
128            } else {
129                format!("on {runtime} · disabled")
130            }
131        }
132        AcpPlacement::NeedsOneOf(runtimes) => format!("needs {} attached", joined(&runtimes)),
133        AcpPlacement::Ambiguous(runtimes) => {
134            format!("offered by {}: attach one", joined(&runtimes))
135        }
136    })
137}

139/// Why picking `agent` for a session or draft that attaches `attached` is
140/// refused, for a status line.
141pub fn unpickable_reason(agent: &AgentDescriptor, attached: &[RuntimeId]) -> String {
142    match placement(agent, attached) {
143        Some(AcpPlacement::NeedsOneOf(runtimes)) => format!(
144            "`{}` needs a runtime that offers it attached: {}",
145            agent.id,
146            joined(&runtimes)
147        ),
148        Some(AcpPlacement::Ambiguous(runtimes)) => format!(
149            "`{}` is offered by more than one attached runtime ({}); attach only one of them",
150            agent.id,
151            joined(&runtimes)
152        ),
153        Some(AcpPlacement::Unavailable(runtime)) => format!(
154            "`{}` is unavailable on runtime `{runtime}`: the runtime is not connected, or the \
155             agent is disabled there",
156            agent.id
157        ),
158        Some(AcpPlacement::Runs(runtime)) => {
159            format!("`{}` is disabled on runtime `{runtime}`", agent.id)
160        }
161        None => format!("`{}` cannot run as a root agent", agent.id),
162    }
163}

165/// The model a session of ACP agent `agent` records, `acp/<agent>`; `None`
166/// for the daemon's own agents.
167pub fn acp_model(agent: &AgentDescriptor) -> Option<ModelSelection> {
168    if !is_acp(agent) {
169        return None;
170    }
171    agent
172        .resolved_fallback
173        .first()
174        .cloned()
175        .or_else(|| Some(cookie_agent_protocol::acp_model_selection(&agent.id)))
176}

184/// Whether agent command `name` is one of the agent's skills, which Codex
185/// lists as commands named `$<skill>` and runs from a `$<skill>` mention.
186pub fn is_skill_command(name: &str) -> bool {
187    name.starts_with('$')
188}
189
190/// Whether agent command `name` is listed for the search `query` (without
191/// its leading `/`) once the query matches it: the agent's skills only for
192/// a query that starts with `$`, as Codex offers them, so a few dozen of
193/// them do not bury the agent's commands.
194pub fn command_listed(name: &str, query: &str) -> bool {
195    !is_skill_command(name) || query.trim_start().starts_with('$')
196}

198/// How agent command `name` is written: `/<name>`, or the skill `$<skill>`
199/// as it is.
200pub fn command_label(name: &str) -> String {
201    if is_skill_command(name) {
202        name.to_owned()
203    } else {
204        format!("/{name}")
205    }
206}

208/// The prompt that runs agent command `name` with `input`: `/<name>
209/// <input>`, as ACP specifies, or for a skill the mention `$<skill>
210/// <input>`, as the agent takes it.
211pub fn command_prompt(name: &str, input: &str) -> String {
212    let input = input.trim();
213    let label = command_label(name);
214    if input.is_empty() {
215        label
216    } else {
217        format!("{label} {input}")
218    }

224    options

238/// The name of `option`'s current value, else the value itself.
239pub fn current_name(option: &AcpConfigOption) -> String {
240    option
241        .values
242        .iter()
243        .find(|value| value.value == option.current_value)
244        .map_or_else(|| option.current_value.clone(), |value| value.name.clone())
245}

247/// What the model chip shows for an agent with `options`: its model's
248/// current value and, when it has one, its effort's, as the model and the
249/// variant of cookie's own models. `None` without a model option.
250pub fn model_and_effort(options: &[AcpConfigOption]) -> Option<(String, Option<String>)> {
251    let model = model_option(options)?;
252    Some((
253        current_name(model),
254        effort_option(options).map(current_name),
255    ))
256}

258/// The model chip's label for an agent with `options`, in the form cookie
259/// labels its own model and variant: `Large • High`.
260pub fn model_label(options: &[AcpConfigOption]) -> Option<String> {
261    let (model, effort) = model_and_effort(options)?;
262    Some(match effort {
263        Some(effort) => format!("{model} • {effort}"),
264        None => model,
265    })

268/// The runtime a draft or session that attaches `attached` runs ACP agent
269/// `agent` on, whatever its availability: the one attached runtime that
270/// offers it.
271pub fn placed_runtime<'a>(
272    agent: &'a AgentDescriptor,
273    attached: &[RuntimeId],
274) -> Option<&'a AgentRuntime> {
275    match placement(agent, attached)? {
276        AcpPlacement::Runs(runtime) | AcpPlacement::Unavailable(runtime) => {
277            offering_runtimes(agent)
278                .into_iter()
279                .find(|offer| offer.runtime_id == runtime)
280        }
281        AcpPlacement::NeedsOneOf(_) | AcpPlacement::Ambiguous(_) => None,
282    }
283}

285/// The files ACP agent `agent` takes in a prompt on the runtime a draft or
286/// session that attaches `attached` runs it on, as the daemon last saw it
287/// start there: what the clients check attachments against, its synthetic

295/// `options` with `choices` made their current values, where an option
296/// takes the value chosen.
297pub fn with_choices(
298    options: &[AcpConfigOption],
299    choices: &[AcpOptionChoice],
300) -> Vec<AcpConfigOption> {
301    let mut options = options.to_vec();
302    for choice in choices {
303        if let Some(option) = options.iter_mut().find(|option| {
304            option.id == choice.option_id
305                && option
306                    .values
307                    .iter()
308                    .any(|value| value.value == choice.value)
309        }) {
310            option.current_value.clone_from(&choice.value);
311        }
312    }
313    options
314}

316/// The options a draft of `agent` that attaches `attached` offers: the
317/// ones a new session of it starts with on the runtime it would run on,
318/// with the draft's `choices` applied. Empty until the agent has started a
319/// session there.
320pub fn draft_options(
321    agent: &AgentDescriptor,
322    attached: &[RuntimeId],
323    choices: &[AcpOptionChoice],
324) -> Vec<AcpConfigOption> {
325    placed_runtime(agent, attached)
326        .map(|runtime| with_choices(&runtime.options, choices))
327        .unwrap_or_default()
328}

330/// The options of a session whose ACP view is `view`: the agent's, once
331/// its ACP session started; before that, what a draft of `agent` would
332/// offer with the values the session was created with.
333pub fn session_options(
334    view: &AcpSessionView,
335    agent: Option<&AgentDescriptor>,
336    attached: &[RuntimeId],
337) -> Vec<AcpConfigOption> {
338    if !view.options.is_empty() {
339        return view.options.clone();
340    }
341    agent
342        .map(|agent| draft_options(agent, attached, &view.starting_options))
343        .unwrap_or_default()
344}

346/// The choices a draft makes for its agent's options, which it sends as
347/// `session.create`'s `agent_options`. Kept with the agent they were made
348/// for, so choosing another agent drops them.
349#[derive(Clone, Debug, Default, Eq, PartialEq)]
350pub struct DraftAgentOptions {
351    agent: Option<AgentId>,
352    choices: Vec<AcpOptionChoice>,
353}
354
355impl DraftAgentOptions {
356    /// The choices made for `agent`.
357    pub fn for_agent(&self, agent: &AgentId) -> &[AcpOptionChoice] {
358        if self.agent.as_ref() == Some(agent) {
359            &self.choices
360        } else {
361            &[]
362        }
363    }
364
365    /// Chooses `value` for option `option_id` of `agent`, replacing an
366    /// earlier choice of that option; choices made for another agent are
367    /// dropped.
368    pub fn choose(&mut self, agent: &AgentId, option_id: &str, value: &str) {
369        if self.agent.as_ref() != Some(agent) {
370            self.agent = Some(agent.clone());
371            self.choices.clear();
372        }
373        match self
374            .choices
375            .iter_mut()
376            .find(|choice| choice.option_id == option_id)
377        {
378            Some(choice) => value.clone_into(&mut choice.value),
379            None => self.choices.push(AcpOptionChoice {
380                option_id: option_id.to_owned(),
381                value: value.to_owned(),
382            }),
383        }
384    }
385
386    /// Forgets every choice, as a draft becomes a session or is dropped.
387    pub fn clear(&mut self) {
388        *self = Self::default();
389    }
390}

421    fn agent(
422        id: &str,
423        offers: &[(&str, bool)],
424        runnable: bool,
425        mode: AgentMode,
426    ) -> AgentDescriptor {
427        let id = AgentId::new(id).unwrap();
428        let runtimes = offers
429            .iter()
430            .map(|(runtime_id, available)| AgentRuntime {
431                runtime_id: runtime(runtime_id),
432                agent: id.clone(),
433                available: *available,
434                options: Vec::new(),
435                input_media: Vec::new(),
436            })
437            .collect::<Vec<_>>();
438        AgentDescriptor {
439            official: false,
440            official_version: None,
441            resolved_fallback: if offers.is_empty() {
442                Vec::new()
443            } else {
444                vec![cookie_agent_protocol::acp_model_selection(&id)]
445            },
446            runtime: match &runtimes[..] {
447                [only] => Some(only.clone()),
448                _ => None,
449            },
450            runtimes,
451            id,
452            preset: None,
453            description: "An agent".into(),
454            mode,
455            enabled: true,
456            runnable_as_root: runnable,
457            delegation_targets: Vec::new(),
458            unset_models: Vec::new(),
459        }
460    }

629        assert_eq!(
630            draft.for_agent(&claude.id),
631            [
632                AcpOptionChoice {
633                    option_id: "model".into(),
634                    value: "opus".into()
635                },
636                AcpOptionChoice {
637                    option_id: "effort".into(),
638                    value: "extreme".into()
639                },
640            ]
641        );
642        let options = draft_options(&claude, &local, draft.for_agent(&claude.id));
643        assert_eq!(model_label(&options).as_deref(), Some("Opus • Low"));
644        // labpc has never started the agent: nothing to offer there.
645        assert!(draft_options(&claude, &[runtime("labpc")], &[]).is_empty());
646        // Choosing for another agent drops the first agent's choices.
647        let codex = AgentId::new("codex").unwrap();
648        draft.choose(&codex, "model", "gpt");
649        assert!(draft.for_agent(&claude.id).is_empty());
650        assert_eq!(draft.for_agent(&codex).len(), 1);
651    }
