1//! Markdown kept in a renderer-neutral form: the source text and its
2//! pulldown-cmark events, split into top-level blocks and parsed
3//! incrementally while text streams in. Each client renders the blocks its
4//! own way and may keep what it derives from a settled block in that block's
5//! [`RendererCache`].
6

8    any::Any,

12};

14use pulldown_cmark::{BrokenLink, Event, LinkType, Options, Parser, Tag};

16#[derive(Clone, Debug, Default)]
17pub struct MarkdownDocument {
18    source: String,
19    stable_prefix_len: usize,
20    stable_blocks: Vec<MarkdownBlock>,
21    tail_blocks: Vec<MarkdownBlock>,
22    stable_reference_definitions: BTreeMap<String, ReferenceDefinition>,
23    reference_definitions: BTreeMap<String, ReferenceDefinition>,
24    stable_reference_dependencies: BTreeSet<String>,
25    /// Render every source newline as a line break instead of joining soft
26    /// breaks with a space: typed text means its newlines.
27    hard_breaks: bool,
28    /// No more text follows: the open tail renders and caches like the
29    /// stable prefix. Cleared by [`Self::append`].
30    finished: bool,
31    parse_passes: u64,
32    parsed_bytes: u64,
33    reference_reparses: u64,
34}

36/// One top-level block of a [`MarkdownDocument`]: its kind, its source, and
37/// its pulldown-cmark events.
38#[derive(Clone, Debug)]
39pub struct MarkdownBlock {
40    kind: MarkdownBlockKind,
41    events: Vec<Event<'static>>,
42    source: Arc<str>,

80/// A slot where a client keeps what it derives from one block, such as its
81/// rendered lines. It holds one value of the type the client first asks
82/// for; clones of the block share it.
83#[derive(Clone, Default)]
84pub struct RendererCache(Arc<OnceLock<Arc<dyn Any + Send + Sync>>>);
85
86impl RendererCache {
87    /// The cache, created empty on first use. A slot already holding
88    /// another type answers a fresh, unshared value.
89    pub fn get_or_default<T: Any + Default + Send + Sync>(&self) -> Arc<T> {
90        Arc::clone(
91            self.0
92                .get_or_init(|| Arc::new(T::default()) as Arc<dyn Any + Send + Sync>),
93        )
94        .downcast::<T>()
95        .unwrap_or_default()
96    }
97}

99impl std::fmt::Debug for RendererCache {
100    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
101        formatter.write_str("RendererCache")
102    }

118impl MarkdownDocument {
119    pub fn new(source: String) -> Self {
120        let mut document = Self {
121            source,
122            ..Self::default()
123        };
124        document.reparse_open_tail();
125        document
126    }
127
128    /// A document whose soft line breaks render as hard breaks, for text a
129    /// person typed (user messages) rather than prose a model reflowed.
130    pub fn with_hard_breaks(source: String) -> Self {
131        let mut document = Self {
132            source,
133            hard_breaks: true,
134            ..Self::default()
135        };
136        document.reparse_open_tail();
137        document
138    }
139
140    pub fn append(&mut self, delta: &str) {
141        self.source.push_str(delta);
142        self.finished = false;
143        self.reparse_open_tail();
144    }
145
146    /// Mark the document complete, so its last block — never promoted to
147    /// the stable prefix, since more text could still extend it — renders
148    /// through the stable block and highlight caches instead of being
149    /// re-rendered and re-highlighted on every layout.
150    #[must_use]
151    pub fn finished(mut self) -> Self {
152        self.finished = true;
153        self
154    }
155
156    pub fn as_str(&self) -> &str {
157        &self.source
158    }
159
160    pub fn stable_prefix_len(&self) -> usize {
161        self.stable_prefix_len
162    }
163
164    pub fn parse_passes(&self) -> u64 {
165        self.parse_passes
166    }
167
168    pub fn parsed_bytes(&self) -> u64 {
169        self.parsed_bytes
170    }
171
172    pub fn reference_reparses(&self) -> u64 {
173        self.reference_reparses
174    }
175
176    pub fn blocks(&self) -> impl Iterator<Item = &MarkdownBlock> {
177        self.stable_blocks.iter().chain(&self.tail_blocks)
178    }
179
180    fn reparse_open_tail(&mut self) {
181        let tail = self.source[self.stable_prefix_len..].to_owned();
182        let parsed = self.parse(&tail);
183        let effective_definitions = self.effective_definitions(&parsed.definitions);
184        let references_changed = self
185            .stable_reference_dependencies
186            .iter()
187            .any(|label| self.reference_definitions.get(label) != effective_definitions.get(label));
188        if references_changed {
189            self.reference_reparses = self.reference_reparses.wrapping_add(1);
190            self.stable_prefix_len = 0;
191            self.stable_blocks.clear();
192            self.tail_blocks.clear();
193            self.stable_reference_definitions.clear();
194            self.reference_definitions.clear();
195            self.stable_reference_dependencies.clear();
196            let source = self.source.clone();
197            let parsed = self.parse(&source);
198            self.reference_definitions = self.effective_definitions(&parsed.definitions);
199            self.apply_parsed(parsed);
200            return;
201        }
202        self.reference_definitions = effective_definitions;
203        self.apply_parsed(parsed);
204    }
205
206    fn parse(&mut self, source: &str) -> ParsedDocument {
207        self.parse_passes = self.parse_passes.wrapping_add(1);
208        self.parsed_bytes = self.parsed_bytes.wrapping_add(source.len() as u64);
209        parse_document(source, self.hard_breaks)
210    }
211
212    fn effective_definitions(
213        &self,
214        tail_definitions: &[ParsedReferenceDefinition],
215    ) -> BTreeMap<String, ReferenceDefinition> {
216        let mut definitions = self.stable_reference_definitions.clone();
217        for definition in tail_definitions {
218            definitions
219                .entry(definition.label.clone())
220                .or_insert_with(|| definition.definition.clone());
221        }
222        definitions
223    }
224
225    fn apply_parsed(&mut self, parsed: ParsedDocument) {
226        let split = parsed.blocks.last().map_or(0, |block| block.range.start);
227        self.stable_reference_dependencies.extend(
228            parsed
229                .blocks
230                .iter()
231                .filter(|block| block.range.end <= split)
232                .flat_map(|block| block.reference_dependencies.iter().cloned()),
233        );
234        self.stable_blocks.extend(
235            parsed
236                .blocks
237                .iter()
238                .filter(|block| block.range.end <= split)
239                .map(|block| block.block.clone()),
240        );
241        self.tail_blocks = parsed
242            .blocks
243            .into_iter()
244            .filter(|block| block.range.start >= split)
245            .map(|block| block.block)
246            .collect();
247        for definition in parsed.definitions {
248            if definition.span.end <= split {
249                self.stable_reference_definitions
250                    .entry(definition.label)
251                    .or_insert(definition.definition);
252            }
253        }
254        self.stable_prefix_len += split;
255    }
256
257    /// The settled blocks: no text that follows can change them.
258    pub fn stable_blocks(&self) -> &[MarkdownBlock] {
259        &self.stable_blocks
260    }
261
262    /// The blocks after the stable prefix, which text that follows may still
263    /// change unless the document is finished.
264    pub fn tail_blocks(&self) -> &[MarkdownBlock] {
265        &self.tail_blocks
266    }
267
268    /// Whether no more text follows.
269    pub const fn is_finished(&self) -> bool {
270        self.finished
271    }
272}

281struct ParsedBlock {
282    block: MarkdownBlock,
283    range: std::ops::Range<usize>,
284    reference_dependencies: BTreeSet<String>,

287struct ParsedDocument {
288    blocks: Vec<ParsedBlock>,
289    definitions: Vec<ParsedReferenceDefinition>,
290}

292#[derive(Clone, Debug, Eq, PartialEq)]
293struct ReferenceDefinition {
294    destination: String,
295    title: Option<String>,
296}
297
298struct ParsedReferenceDefinition {
299    label: String,
300    definition: ReferenceDefinition,
301    span: std::ops::Range<usize>,
302}

304fn parse_document(source: &str, hard_breaks: bool) -> ParsedDocument {
305    let (definitions, events, broken_references) = {
306        let mut broken_references = Vec::new();
307        let mut broken_link_callback = |broken: BrokenLink<'_>| {
308            broken_references.push((
309                broken.span,
310                normalize_reference_label(broken.reference.as_ref()),
311            ));
312            None
313        };
314        let parser = Parser::new_with_broken_link_callback(
315            source,
316            markdown_options(),
317            Some(&mut broken_link_callback),
318        );
319        let definitions = parser
320            .reference_definitions()
321            .iter()
322            .map(|(label, definition)| ParsedReferenceDefinition {
323                label: normalize_reference_label(label),
324                definition: ReferenceDefinition {
325                    destination: definition.dest.to_string(),
326                    title: definition.title.as_ref().map(ToString::to_string),
327                },
328                span: definition.span.clone(),
329            })
330            .collect::<Vec<_>>();
331        let events = parser.into_offset_iter().collect::<Vec<_>>();
332        (definitions, events, broken_references)
333    };
334    let mut blocks = Vec::new();
335    let mut current = Vec::new();
336    let mut current_kind = MarkdownBlockKind::Other;
337    let mut current_start = 0;
338    let mut depth = 0usize;
339
340    for (event, range) in events {
341        if depth == 0 && current.is_empty() {
342            current_start = range.start;
343            current_kind = block_kind(&event);
344        }
345        let current_end = range.end;
346        match &event {
347            Event::Start(_) => depth += 1,
348            Event::End(_) => depth = depth.saturating_sub(1),
349            _ => {}
350        }
351        current.push(match event {
352            Event::SoftBreak if hard_breaks => Event::HardBreak,
353            event => event.into_static(),
354        });
355        if depth == 0 {
356            let range = current_start..current_end;
357            blocks.push(ParsedBlock {
358                block: MarkdownBlock {
359                    kind: current_kind,
360                    events: std::mem::take(&mut current),
361                    source: Arc::from(&source[range.clone()]),
362                    content_hash: foldhash::fast::FixedState::default()
363                        .hash_one(&source[range.clone()]),
364                    content_len: range.len(),
365                    renderer_cache: RendererCache::default(),
366                },
367                range,
368                reference_dependencies: BTreeSet::new(),
369            });
370        }
371    }
372    for block in &mut blocks {
373        for event in &block.block.events {
374            let Event::Start(Tag::Link { link_type, id, .. } | Tag::Image { link_type, id, .. }) =
375                event
376            else {
377                continue;
378            };
379            if matches!(
380                link_type,
381                LinkType::Reference
382                    | LinkType::ReferenceUnknown
383                    | LinkType::Collapsed
384                    | LinkType::CollapsedUnknown
385                    | LinkType::Shortcut
386                    | LinkType::ShortcutUnknown
387            ) {
388                block
389                    .reference_dependencies
390                    .insert(normalize_reference_label(id.as_ref()));
391            }
392        }
393        block.reference_dependencies.extend(
394            broken_references
395                .iter()
396                .filter(|(span, _)| span.start >= block.range.start && span.end <= block.range.end)
397                .map(|(_, label)| label.clone()),
398        );
399    }
400    ParsedDocument {
401        blocks,
402        definitions,
403    }
404}

406fn normalize_reference_label(label: &str) -> String {
407    label
408        .split_whitespace()
409        .collect::<Vec<_>>()
410        .join(" ")
411        .to_lowercase()
412}
413
414fn block_kind(event: &Event<'_>) -> MarkdownBlockKind {
415    match event {
416        Event::Start(Tag::Paragraph) => MarkdownBlockKind::Paragraph,
417        Event::Start(Tag::Heading { .. }) => MarkdownBlockKind::Heading,
418        Event::Start(Tag::BlockQuote(_)) => MarkdownBlockKind::Quote,
419        Event::Start(Tag::CodeBlock(_)) => MarkdownBlockKind::Code,
420        Event::Start(Tag::List(_)) => MarkdownBlockKind::List,
421        Event::Start(Tag::Table(_)) => MarkdownBlockKind::Table,
422        Event::Start(Tag::HtmlBlock) | Event::Html(_) => MarkdownBlockKind::Html,
423        Event::Rule => MarkdownBlockKind::ThematicBreak,
424        _ => MarkdownBlockKind::Other,
425    }
426}

428#[cfg(test)]
429mod tests {
430    use std::sync::{Arc, Mutex};
431
432    use super::{MarkdownBlockKind, MarkdownDocument};
433
434    #[test]
435    fn streamed_text_settles_blocks_and_keeps_the_open_tail_apart() {
436        let mut document = MarkdownDocument::new("# Title\n\nfirst".to_owned());
437        assert_eq!(document.stable_blocks().len(), 1);
438        assert_eq!(
439            document.stable_blocks()[0].kind(),
440            MarkdownBlockKind::Heading
441        );
442        assert_eq!(document.tail_blocks().len(), 1);
443        document.append(" paragraph\n\n```rust\nfn a() {}\n```\n");
444        assert_eq!(
445            document
446                .blocks()
447                .map(super::MarkdownBlock::kind)
448                .collect::<Vec<_>>(),
449            [
450                MarkdownBlockKind::Heading,
451                MarkdownBlockKind::Paragraph,
452                MarkdownBlockKind::Code,
453            ]
454        );
455        assert!(!document.is_finished());
456        assert!(document.clone().finished().is_finished());
457    }
458
459    #[test]
460    fn clones_of_a_block_share_the_renderer_cache() {
461        let document = MarkdownDocument::new("first\n\nsecond".to_owned());
462        let block = document.stable_blocks()[0].clone();
463        block
464            .renderer_cache()
465            .get_or_default::<Mutex<Vec<String>>>()
466            .lock()
467            .unwrap()
468            .push("rendered".to_owned());
469        let shared = document.stable_blocks()[0]
470            .renderer_cache()
471            .get_or_default::<Mutex<Vec<String>>>();
472        assert_eq!(*shared.lock().unwrap(), ["rendered"]);
473        // A renderer asking for another type gets a fresh, unshared value.
474        let other = block.renderer_cache().get_or_default::<Mutex<u8>>();
475        *other.lock().unwrap() = 7;
476        assert_eq!(
477            *block
478                .renderer_cache()
479                .get_or_default::<Mutex<u8>>()
480                .lock()
481                .unwrap(),
482            0
483        );
484        assert!(Arc::ptr_eq(
485            &shared,
486            &block
487                .renderer_cache()
488                .get_or_default::<Mutex<Vec<String>>>()
489        ));
490    }
