1//! What a session list shows about each session's activity: running, waiting
2//! on a producer, a problem with its last run, or nothing.
3
4use cookie_agent_protocol::{SemVer, SessionMeta, SessionStatus, WaitingOn};
5
6/// The client protocol minor from which `SessionMeta` reports `waiting_on`.
7pub const WAITING_ON_MINOR: u32 = 9;
8
9/// Whether a daemon whose handshake reported `version` reports what sessions
10/// wait on. Before the handshake nothing is known; from an older daemon a
11/// session never appears to wait.
12#[must_use]
13pub fn daemon_reports_waiting(version: Option<SemVer>) -> bool {
14    version.is_some_and(|version| {
15        version
16            .shared_minor(cookie_agent_protocol::CLIENT_PROTOCOL_VERSION)
17            .is_some_and(|minor| minor >= WAITING_ON_MINOR)
18    })
19}
20
21/// A session's activity as a session row shows it.
22#[derive(Clone, Debug, Eq, PartialEq)]
23pub enum SessionActivity {
24    /// A run is going. It takes precedence over everything else: a running
25    /// session shows as running whatever it waits on.
26    Running,
27    /// No run is going, and these producers will start the next one without
28    /// the user, in `SessionMeta::waiting_on` order. Waiting takes
29    /// precedence over a problem with the last run, since the session will
30    /// run again.
31    Waiting(Vec<WaitingOn>),
32    /// The last run failed, was cancelled, or was interrupted, and nothing
33    /// will wake the session.
34    Problem(SessionStatus),
35    /// Idle or completed with nothing to wait on: a row shows no activity.
36    None,
37}
38
39/// What a session row shows for `meta`: [`SessionActivity::Running`] over
40/// [`SessionActivity::Waiting`] over [`SessionActivity::Problem`]. A status
41/// this build does not know shows as no activity.
42#[must_use]
43pub fn session_activity(meta: &SessionMeta) -> SessionActivity {
44    if meta.status == SessionStatus::Running {
45        return SessionActivity::Running;
46    }
47    if !meta.waiting_on.is_empty() {
48        return SessionActivity::Waiting(meta.waiting_on.clone());
49    }
50    match meta.status {
51        SessionStatus::Failed | SessionStatus::Cancelled | SessionStatus::Interrupted => {
52            SessionActivity::Problem(meta.status)
53        }
54        SessionStatus::Idle
55        | SessionStatus::Running
56        | SessionStatus::Completed
57        | SessionStatus::Unknown => SessionActivity::None,
58    }
59}

98    #[test]
99    fn running_beats_waiting_beats_a_problem() {
100        let goal = [WaitingOn::Goal];
101        assert_eq!(
102            session_activity(&meta(SessionStatus::Running, &goal)),
103            SessionActivity::Running
104        );
105        assert_eq!(
106            session_activity(&meta(SessionStatus::Running, &[])),
107            SessionActivity::Running
108        );
109        for status in [
110            SessionStatus::Idle,
111            SessionStatus::Completed,
112            SessionStatus::Failed,
113            SessionStatus::Cancelled,
114            SessionStatus::Interrupted,
115        ] {
116            assert_eq!(
117                session_activity(&meta(
118                    status,
119                    &[WaitingOn::BackgroundCall, WaitingOn::Subagent]
120                )),
121                SessionActivity::Waiting(vec![WaitingOn::BackgroundCall, WaitingOn::Subagent]),
122                "{status:?}"
123            );
124        }
125        // A kind from a newer daemon still waits.
126        assert_eq!(
127            session_activity(&meta(SessionStatus::Completed, &[WaitingOn::Unknown])),
128            SessionActivity::Waiting(vec![WaitingOn::Unknown])
129        );
130    }

132    #[test]
133    fn only_a_failed_cancelled_or_interrupted_run_is_a_problem() {
134        for status in [
135            SessionStatus::Failed,

139            assert_eq!(
140                session_activity(&meta(status, &[])),

145            SessionStatus::Idle,
146            SessionStatus::Completed,
147            SessionStatus::Unknown,
148        ] {
149            assert_eq!(
150                session_activity(&meta(status, &[])),
151                SessionActivity::None,
152                "{status:?}"
153            );
154        }
155    }

157    #[test]
158    fn waiting_comes_from_a_daemon_of_minor_nine() {
159        let reports = |version| daemon_reports_waiting(Some(version));
160        assert!(!daemon_reports_waiting(None));
161        assert!(!reports(SemVer::new(1, 8, 0)));
162        assert!(reports(SemVer::new(1, 9, 0)));
163        assert!(reports(SemVer::new(1, 12, 3)));
164        assert!(!reports(SemVer::new(2, 9, 0)));
165    }
