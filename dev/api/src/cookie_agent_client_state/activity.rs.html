1//! What a session list shows about each session's activity: running, waiting
2//! on a producer, a problem with its last run, or nothing.
3
4use cookie_agent_protocol::{HeldReason, Open, SemVer, SessionMeta, SessionStatus, WaitingOn};
5
6/// The client protocol minor from which `SessionMeta` reports `waiting_on`.
7pub const WAITING_ON_MINOR: u32 = 9;
8
9/// Whether a daemon whose handshake reported `version` reports what sessions
10/// wait on. Before the handshake nothing is known; from an older daemon a
11/// session never appears to wait.
12#[must_use]
13pub fn daemon_reports_waiting(version: Option<SemVer>) -> bool {
14    version.is_some_and(|version| {
15        version
16            .shared_minor(cookie_agent_protocol::CLIENT_PROTOCOL_VERSION)
17            .is_some_and(|minor| minor >= WAITING_ON_MINOR)
18    })
19}
20
21/// A session's activity as a session row shows it.
22#[derive(Clone, Debug, Eq, PartialEq)]
23pub enum SessionActivity {
24    /// A run is going. It takes precedence over everything else: a running
25    /// session shows as running whatever it waits on.
26    Running,
27    /// No run is going and the session's automatic runs are held
28    /// (`docs/site/specs/held-sessions.md`): it waits for the user to resume
29    /// it. It takes precedence over waiting, which the hold stops.
30    Held(Open<HeldReason>),
31    /// No run is going, and these producers will start the next one without
32    /// the user, in `SessionMeta::waiting_on` order. Waiting takes
33    /// precedence over a problem with the last run, since the session will
34    /// run again.
35    Waiting(Vec<WaitingOn>),
36    /// The last run failed, was cancelled, or was interrupted, and nothing
37    /// will wake the session.
38    Problem(SessionStatus),
39    /// Idle or completed with nothing to wait on: a row shows no activity.
40    None,
41}
42
43/// What a session row shows for `meta`: [`SessionActivity::Running`] over
44/// [`SessionActivity::Waiting`] over [`SessionActivity::Problem`]. A status
45/// this build does not know shows as no activity.
46#[must_use]
47pub fn session_activity(meta: &SessionMeta) -> SessionActivity {
48    if meta.status == SessionStatus::Running {
49        return SessionActivity::Running;
50    }
51    if let Some(reason) = &meta.held {
52        return SessionActivity::Held(reason.clone());
53    }
54    if !meta.waiting_on.is_empty() {
55        return SessionActivity::Waiting(meta.waiting_on.clone());
56    }
57    match meta.status {
58        SessionStatus::Failed | SessionStatus::Cancelled | SessionStatus::Interrupted => {
59            SessionActivity::Problem(meta.status)
60        }
61        SessionStatus::Idle
62        | SessionStatus::Running
63        | SessionStatus::Completed
64        | SessionStatus::Unknown => SessionActivity::None,
65    }
66}

68/// What a waiting row's tooltip says: `waiting on: goal, background call`,
69/// each kind once, in `kinds` order.
70#[must_use]
71pub fn waiting_label(kinds: &[WaitingOn]) -> String {
72    let mut names = Vec::<&str>::new();
73    for kind in kinds {
74        let name = match kind {
75            WaitingOn::Goal => "goal",
76            WaitingOn::BackgroundCall => "background call",
77            WaitingOn::Subagent => "subagent",
78            WaitingOn::ExternalProducer => "external producer",
79            WaitingOn::ScheduledJob => "scheduled job",
80            WaitingOn::Unknown => "another producer",
81        };
82        if !names.contains(&name) {
83            names.push(name);
84        }
85    }
86    format!("waiting on: {}", names.join(", "))
87}

105                "preset": null

107            "runtime_revision": revision,
108            "catalog_revision": revision,
109            "provider_state_revision": revision,
110            "model_revision": revision,
111            "agent_revision": revision,
112            "recipe_registry_revision": revision,
113            "manifest_revision": revision,
114            "title": null,
115            "title_updated_seq": 0,
116            "last_event_seq": 1,
117            "last_activity": "2026-10-21T12:00:00Z",

122        meta.waiting_on = waiting_on.to_vec();

126    #[test]
127    fn held_beats_waiting_and_running_beats_held() {
128        let stopped = Open::Known(HeldReason::Stopped {});
129        let mut held = meta(SessionStatus::Cancelled, &[WaitingOn::Goal]);
130        held.held = Some(stopped.clone());
131        assert_eq!(session_activity(&held), SessionActivity::Held(stopped));
132        held.status = SessionStatus::Running;
133        assert_eq!(session_activity(&held), SessionActivity::Running);
134    }
135
136    #[test]
137    fn running_beats_waiting_beats_a_problem() {
138        let goal = [WaitingOn::Goal];
139        assert_eq!(
140            session_activity(&meta(SessionStatus::Running, &goal)),
141            SessionActivity::Running
142        );
143        assert_eq!(
144            session_activity(&meta(SessionStatus::Running, &[])),
145            SessionActivity::Running
146        );
147        for status in [
148            SessionStatus::Idle,
149            SessionStatus::Completed,
150            SessionStatus::Failed,
151            SessionStatus::Cancelled,
152            SessionStatus::Interrupted,
153        ] {
154            assert_eq!(
155                session_activity(&meta(
156                    status,
157                    &[WaitingOn::BackgroundCall, WaitingOn::Subagent]
158                )),
159                SessionActivity::Waiting(vec![WaitingOn::BackgroundCall, WaitingOn::Subagent]),
160                "{status:?}"
161            );
162        }
163        // A kind from a newer daemon still waits.
164        assert_eq!(
165            session_activity(&meta(SessionStatus::Completed, &[WaitingOn::Unknown])),
166            SessionActivity::Waiting(vec![WaitingOn::Unknown])
167        );
168    }

170    #[test]

195    #[test]
196    fn waiting_comes_from_a_daemon_of_minor_nine() {
197        let reports = |version| daemon_reports_waiting(Some(version));
198        assert!(!daemon_reports_waiting(None));
199        assert!(!reports(SemVer::new(1, 8, 0)));
200        assert!(reports(SemVer::new(1, 9, 0)));
201        assert!(reports(SemVer::new(1, 12, 3)));
202        assert!(!reports(SemVer::new(2, 9, 0)));
