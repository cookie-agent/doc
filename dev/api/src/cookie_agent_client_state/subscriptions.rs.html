1//! Which sessions a client keeps subscribed, and when it tries again.
2//!
3//! A session with a live tail cannot be evicted by the daemon, so a client
4//! that browses many sessions keeps only the trees it viewed most recently
5//! subscribed ([`RECENT_SUBSCRIBED_TREES`]) and releases the settled
6//! sessions of the others: a session with a run, or with input whose start
7//! or return the view must still see, is released once it settles. The
8//! client owns the clock and the requests; this only decides.
9
10use std::{collections::VecDeque, time::Duration};
11
12use cookie_agent_protocol::SessionId;
13
14use crate::SessionState;
15
16/// Trees whose sessions stay subscribed after the view moves on.
17pub const RECENT_SUBSCRIBED_TREES: usize = 8;
18
19/// Wait before resubscribing a viewed session whose recovery gave up.
20pub const SUBSCRIPTION_RETRY_DELAY: Duration = Duration::from_secs(2);
21
22/// First wait before redialing a dropped daemon; doubles per failure up to
23/// [`RECONNECT_MAX_DELAY`].
24pub const RECONNECT_FIRST_DELAY: Duration = Duration::from_millis(500);
25
26/// The longest wait between redials of a dropped daemon.
27pub const RECONNECT_MAX_DELAY: Duration = Duration::from_secs(10);
28
29/// The wait before the redial after one that waited `delay` and failed.
30pub fn next_reconnect_delay(delay: Duration) -> Duration {
31    (delay * 2).min(RECONNECT_MAX_DELAY)
32}
33
34/// The roots of the trees viewed most recently, newest first.
35#[derive(Clone, Debug, Default)]
36pub struct RecentTrees(VecDeque<SessionId>);
37
38impl RecentTrees {
39    /// Whether the tree rooted at `root` is one of the recent ones.
40    pub fn contains(&self, root: SessionId) -> bool {
41        self.0.contains(&root)
42    }
43
44    /// Records `root` as the most recently viewed tree and returns the
45    /// roots this pushed out of the recent set, oldest last: their settled
46    /// sessions are released.
47    pub fn note_viewed(&mut self, root: SessionId) -> Vec<SessionId> {
48        self.0.retain(|recent| *recent != root);
49        self.0.push_front(root);
50        let mut evicted = Vec::new();
51        while self.0.len() > RECENT_SUBSCRIBED_TREES {
52            let Some(oldest) = self.0.pop_back() else {
53                break;
54            };
55            evicted.push(oldest);
56        }
57        evicted
58    }
59}
60
61impl SessionState {
62    /// Whether nothing in the session is still owed to its view: no run,
63    /// and no pending or queued input whose start or return it must see.
64    pub fn is_settled(&self) -> bool {
65        self.active_run.is_none() && self.pending_inputs.is_empty() && self.queued_inputs.is_empty()
66    }
67}

69#[cfg(test)]
70mod tests {
71    use std::time::Duration;
72
73    use cookie_agent_protocol::SessionId;
74
75    use super::{RECENT_SUBSCRIBED_TREES, RECONNECT_MAX_DELAY, RecentTrees, next_reconnect_delay};
76
77    #[test]
78    fn viewing_more_trees_than_are_kept_evicts_the_least_recent() {
79        let mut recent = RecentTrees::default();
80        let roots = (0..=RECENT_SUBSCRIBED_TREES)
81            .map(|_| SessionId::new_v7())
82            .collect::<Vec<_>>();
83        for root in &roots[..RECENT_SUBSCRIBED_TREES] {
84            assert!(recent.note_viewed(*root).is_empty());
85        }
86        // Viewing a kept tree again makes it the newest.
87        assert!(recent.note_viewed(roots[0]).is_empty());
88        assert_eq!(
89            recent.note_viewed(roots[RECENT_SUBSCRIBED_TREES]),
90            vec![roots[1]]
91        );
92        assert!(!recent.contains(roots[1]));
93        assert!(recent.contains(roots[0]));
94    }
95
96    #[test]
97    fn redials_back_off_up_to_the_longest_wait() {
98        assert_eq!(
99            next_reconnect_delay(Duration::from_millis(700)),
100            Duration::from_millis(1400)
101        );
102        assert_eq!(
103            next_reconnect_delay(RECONNECT_MAX_DELAY),
104            RECONNECT_MAX_DELAY
105        );
106    }
