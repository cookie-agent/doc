16/// What `memory.write` answered.
17#[derive(Debug, Eq, PartialEq)]
18pub enum MemoryWriteOutcome {
19    /// Written; `notice` says what the daemon kept in place of what the
20    /// text gave (an `agent://` file's description and first keywords).
21    Saved {
22        notice: Option<String>,
23    },
24    /// The text is not a valid memory: why.
25    Invalid(String),
26    /// The memory changed, appeared, or went away since it was read.
27    Changed,
28    Failed(String),
29}
30
31impl MemoryWriteOutcome {
32    pub fn from_result(result: Result<MemoryWriteResult, ClientError>) -> Self {
33        match result {
34            Ok(written) => Self::Saved {
35                notice: written.notice,
36            },
37            Err(error) if error.is(&errors::MEMORY_INVALID) => Self::Invalid(message_of(&error)),
38            Err(error)
39                if error.is(&errors::MEMORY_REVISION_MISMATCH)
40                    || error.is(&errors::MEMORY_EXISTS)
41                    || error.is(&errors::MEMORY_NOT_FOUND) =>
42            {
43                Self::Changed
44            }
45            Err(error) => Self::Failed(message_of(&error)),
46        }
47    }
48}

50/// What the status line says once `url` is saved: `Created` for a new
51/// memory, else `Saved`, then the daemon's notice when it gave one.
52pub fn saved_status(url: &str, created: bool, notice: Option<&str>) -> String {
53    let verb = if created { "Created" } else { "Saved" };
54    match notice {
55        Some(notice) => format!("{verb} {url}. {notice}"),
56        None => format!("{verb} {url}."),
57    }
58}

60/// The message of a daemon error, without its code.
61pub fn message_of(error: &ClientError) -> String {
62    match error {
63        ClientError::Rpc(rpc) => rpc.message.clone(),
64        error => error.to_string(),
65    }
66}

68/// What a refused `memory.delete` of `url` says.
69pub fn delete_refusal(url: &str, error: &ClientError) -> String {
70    if error.is(&errors::MEMORY_REVISION_MISMATCH) {
71        format!("{url} changed since it was listed; it was not deleted")
72    } else {
73        format!("{url} was not deleted: {}", message_of(error))
74    }

77/// One row of the panel.
78#[derive(Clone, Debug, Eq, PartialEq)]
79pub enum MemoryRow {
80    /// A layer's heading.
81    Layer(MemoryLayer),
82    /// A machine or project of the layer.
83    Group {
84        layer: MemoryLayer,
85        name: String,
86    },
87    Memory(MemoryInfo),
88    /// The heading of what is not a memory.
89    IgnoredHeading,
90    Ignored(MemoryIgnoredEntry),
91}
92
93impl MemoryRow {
94    /// The URL prefix a new memory started from this row gets.
95    pub fn new_prefix(&self) -> String {
96        match self {
97            Self::Layer(layer) => format!("{}://", layer.as_str()),
98            Self::Group { layer, name } => format!("{}://{name}/", layer.as_str()),
99            Self::Memory(memory) => match &memory.group {
100                Some(group) => format!("{}://{group}/", memory.layer.as_str()),
101                None => format!("{}://", memory.layer.as_str()),
102            },
103            Self::IgnoredHeading | Self::Ignored(_) => "project://".to_owned(),
104        }
105    }
106}

108/// The rows: each layer's heading, its memories grouped by machine or
109/// project, then what is not a memory.
110pub fn memory_rows(listing: &MemoryListResult) -> Vec<MemoryRow> {
111    let mut rows = Vec::new();
112    for layer in MemoryLayer::ALL {
113        rows.push(MemoryRow::Layer(layer));
114        let mut group: Option<&str> = None;
115        for memory in listing
116            .memories
117            .iter()
118            .filter(|memory| memory.layer == layer)
119        {
120            if let Some(name) = memory.group.as_deref()
121                && group != Some(name)
122            {
123                rows.push(MemoryRow::Group {
124                    layer,
125                    name: name.to_owned(),
126                });
127                group = Some(name);
128            }
129            rows.push(MemoryRow::Memory(memory.clone()));
130        }
131    }
132    if !listing.ignored.is_empty() {
133        rows.push(MemoryRow::IgnoredHeading);
134        rows.extend(listing.ignored.iter().cloned().map(MemoryRow::Ignored));
135    }
136    rows
137}

139/// Why `url` cannot be a new memory of `listing`, if it cannot. The daemon
140/// checks the URL itself when the memory is saved.
141pub fn new_memory_refusal(listing: &MemoryListResult, url: &str) -> Option<String> {
142    if listing.memories.iter().any(|memory| memory.url == url) {
143        return Some(format!("{url} exists; edit it instead"));
144    }
145    let complete = url
146        .split_once("://")
147        .is_some_and(|(_, rest)| !rest.is_empty() && !rest.ends_with('/'));
148    (!complete).then(|| {
149        "a memory URL is user://<category>, agent://<agent>, machine://<machine>/<name>, \
150         or project://<project>/<name>"
151            .to_owned()
152    })
153}

155/// The memory's own name: what its URL says after its layer and group.
156pub fn memory_name(memory: &MemoryInfo) -> &str {
157    let rest = memory
158        .url
159        .split_once("://")
160        .map_or(memory.url.as_str(), |(_, rest)| rest);
161    rest.rsplit_once('/').map_or(rest, |(_, name)| name)
162}

164/// The keywords an agent's mode gives its memory, after its ID.
165const MODE_KEYWORDS: [&str; 3] = ["primary", "subagent", "all"];
166
167/// An `agent://` file as an editor shows it: the daemon owns its
168/// description and its first keywords (the agent's ID, then its mode) and
169/// keeps them whatever a write gives; the other keywords and the body are
170/// the user's.
171#[derive(Clone, Debug, Eq, PartialEq)]
172pub struct AgentMemoryParts {
173    pub description: String,
174    /// The daemon's keywords: the agent's ID, then its mode.
175    pub fixed: Vec<String>,
176    pub extra: Vec<String>,
177    pub body: String,
178}
179
180/// The parts of `memory`, an `agent://` file whose whole text is `text`,
181/// as `memory.list` describes it; `None` for any other memory, an invalid
182/// one, or a text without frontmatter, which an editor shows whole.
183pub fn agent_memory_parts(memory: &MemoryInfo, text: &str) -> Option<AgentMemoryParts> {
184    if memory.layer != MemoryLayer::Agent || memory.invalid.is_some() {
185        return None;
186    }
187    let agent = memory.url.strip_prefix("agent://")?;
188    let body = frontmatter_body(text)?;
189    let mut keywords = memory.keywords.iter();
190    let mut fixed = Vec::new();
191    let mut extra = Vec::new();
192    for keyword in keywords.by_ref() {
193        let owned = match fixed.len() {
194            0 => keyword == agent,
195            1 => agent != keyword && MODE_KEYWORDS.contains(&keyword.as_str()),
196            _ => false,
197        };
198        if owned {
199            fixed.push(keyword.clone());
200        } else {
201            extra.push(keyword.clone());
202            break;
203        }
204    }
205    extra.extend(keywords.cloned());
206    Some(AgentMemoryParts {
207        description: memory.description.clone().unwrap_or_default(),
208        fixed,
209        extra,
210        body: body.to_owned(),
211    })
212}

214/// What follows the frontmatter of `text`: the lines after the `---` that
215/// closes the one that opens it.
216fn frontmatter_body(text: &str) -> Option<&str> {
217    let rest = text
218        .strip_prefix("---\n")
219        .or_else(|| text.strip_prefix("---\r\n"))?;
220    let mut offset = 0;
221    for line in rest.split_inclusive('\n') {
222        offset += line.len();
223        if line.trim_end() == "---" {
224            return Some(&rest[offset..]);
225        }
226    }
227    None
228}

230impl AgentMemoryParts {
231    /// The extra keywords of a comma-separated field.
232    pub fn typed_keywords(typed: &str) -> Vec<String> {
233        typed
234            .split(',')
235            .map(str::trim)
236            .filter(|keyword| !keyword.is_empty())
237            .map(str::to_owned)
238            .collect()
239    }
240
241    /// The whole file to write: the frontmatter with the daemon's
242    /// description and keywords (it keeps them anyway), the extra
243    /// keywords, then the body. Scalars are double-quoted JSON strings,
244    /// which YAML reads as the same text.
245    pub fn text(&self) -> String {
246        let quote = |value: &str| serde_json::to_string(value).unwrap_or_default();
247        let mut text = format!("---\ndescription: {}\n", quote(&self.description));
248        let keywords = self
249            .fixed
250            .iter()
251            .chain(&self.extra)
252            .map(|keyword| quote(keyword))
253            .collect::<Vec<_>>();
254        if !keywords.is_empty() {
255            text.push_str(&format!("keywords: [{}]\n", keywords.join(", ")));
256        }
257        text.push_str("---\n");
258        text.push_str(&self.body);
259        text
260    }
261}

263/// The client protocol minor from which a daemon stamps every memory with
264/// `created` and `modified`, reports them with `last_read` and the size
265/// limits in `memory.list` and `memory.read`, and owns the stamp lines of
266/// a memory's frontmatter, rewriting them on every save.
267pub const MEMORY_STAMPS_MINOR: u32 = 14;
268
269/// Whether a daemon whose handshake reported `version` owns the stamp lines
270/// of a memory's frontmatter, so an editor leaves them out. An older
271/// daemon's files have none of its own, and whatever a user wrote there is
272/// theirs to edit.
273#[must_use]
274pub fn daemon_stamps_memory(version: Option<SemVer>) -> bool {
275    version.is_some_and(|version| {
276        version
277            .shared_minor(cookie_agent_protocol::CLIENT_PROTOCOL_VERSION)
278            .is_some_and(|minor| minor >= MEMORY_STAMPS_MINOR)
279    })
280}
281
282/// The frontmatter keys the daemon owns on every memory.
283const STAMP_KEYS: [&str; 2] = ["created", "modified"];
284
285/// `text` without the stamp lines of its frontmatter (`created:` and
286/// `modified:` at the top level), the rest kept byte for byte. The daemon
287/// writes its own stamps on every save whatever the text says, so an editor
288/// shows the text without them and saves it as it is left. A text without
289/// frontmatter is returned whole.
290pub fn without_stamps(text: &str) -> String {
291    let Some(opening) = ["---\n", "---\r\n"]
292        .into_iter()
293        .find(|opening| text.starts_with(opening))
294    else {
295        return text.to_owned();
296    };
297    let mut kept = String::with_capacity(text.len());
298    kept.push_str(opening);
299    let mut lines = text[opening.len()..].split_inclusive('\n');
300    for line in lines.by_ref() {
301        if line.trim_end() == "---" {
302            kept.push_str(line);
303            break;
304        }
305        let stamp = STAMP_KEYS.iter().any(|key| {
306            line.strip_prefix(key)
307                .is_some_and(|rest| rest.starts_with(':'))
308        });
309        if !stamp {
310            kept.push_str(line);
311        }
312    }
313    for line in lines {
314        kept.push_str(line);
315    }
316    kept
317}
318
319/// The text an editor opens for a memory read from a daemon of `version`:
320/// without the stamp lines when the daemon owns them.
321pub fn editor_text(text: &str, version: Option<SemVer>) -> String {
322    if daemon_stamps_memory(version) {
323        without_stamps(text)
324    } else {
325        text.to_owned()
326    }
327}

336/// A memory's stamps and size as `memory.list` or `memory.read` report
337/// them; each is absent from a daemon older than client protocol 1.14.0.
338#[derive(Clone, Debug, Default, Eq, PartialEq)]
339pub struct MemoryStamps {
340    pub created: Option<Timestamp>,
341    pub modified: Option<Timestamp>,
342    /// When a model last read it, if the daemon knows.
343    pub last_read: Option<Timestamp>,
344    /// The file's size in bytes, frontmatter included.
345    pub size: Option<u64>,
346    /// The `[memory]` limit of its kind, in bytes.
347    pub max_size: Option<u64>,
348}

350/// One of a memory's times, for a line of its own.
351#[derive(Clone, Copy, Debug, Eq, PartialEq)]
352pub struct MemoryTime {
353    /// `created`, `modified`, or `last read`.
354    pub label: &'static str,
355    pub at: Timestamp,
356}
357
358impl MemoryTime {
359    /// `created: 2026-10-01 09:12:44 UTC (3 weeks ago)`.
360    pub fn line(&self, now: Timestamp) -> String {
361        format!(
362            "{}: {} ({})",
363            self.label,
364            exact_time(self.at),
365            ago(self.at, now)
366        )
367    }
368}
369
370impl MemoryStamps {
371    pub fn of_info(memory: &MemoryInfo) -> Self {
372        Self {
373            created: memory.created,
374            modified: memory.modified,
375            last_read: memory.last_read,
376            size: Some(memory.size),
377            max_size: memory.max_size,
378        }
379    }
380
381    /// The stamps of a read; `last_read` comes from the listing, which
382    /// alone reports it.
383    pub fn of_read(read: &MemoryReadResult, last_read: Option<Timestamp>) -> Self {
384        Self {
385            created: read.created,
386            modified: read.modified,
387            last_read,
388            size: read.size,
389            max_size: read.max_size,
390        }
391    }
392
393    /// Whether the memory is over its limit.
394    pub fn over_limit(&self) -> bool {
395        matches!((self.size, self.max_size), (Some(size), Some(max)) if size > max)
396    }
397
398    /// The times known, in the order created, modified, last read.
399    pub fn times(&self) -> Vec<MemoryTime> {
400        [
401            ("created", self.created),
402            ("modified", self.modified),
403            ("last read", self.last_read),
404        ]
405        .into_iter()
406        .filter_map(|(label, at)| at.map(|at| MemoryTime { label, at }))
407        .collect()
408    }
409
410    /// `812 bytes of 2 KB`, `2.5 KB of 2 KB, over its limit`, or only the
411    /// size without a limit; `None` without a size.
412    pub fn size_text(&self) -> Option<String> {
413        let size = memory_size(self.size?);
414        Some(match self.max_size {
415            Some(max) if self.over_limit() => {
416                format!("{size} of {}, {OVER_LIMIT}", memory_size(max))
417            }
418            Some(max) => format!("{size} of {}", memory_size(max)),
419            None => size,
420        })
421    }
422
423    /// The lines a detail view shows: each time known with its exact time
424    /// and age, then the size and limit.
425    pub fn detail_lines(&self, now: Timestamp) -> Vec<String> {
426        let mut lines = self
427            .times()
428            .iter()
429            .map(|time| time.line(now))
430            .collect::<Vec<_>>();
431        if let Some(size) = self.size_text() {
432            lines.push(format!("size: {size}"));
433        }
434        lines
435    }
436
437    /// Every time known as an age, then the size and limit, such as
438    /// `created 3 weeks ago · modified 1 day ago · last read 2 minutes ago
439    /// · 812 bytes of 2 KB`.
440    pub fn ages(&self, now: Timestamp) -> String {
441        let mut parts = self
442            .times()
443            .iter()
444            .map(|time| format!("{} {}", time.label, ago(time.at, now)))
445            .collect::<Vec<_>>();
446        parts.extend(self.size_text());
447        parts.join(" · ")
448    }
449
450    /// One short line for a row: when it was modified, then its size and
451    /// limit, such as `modified 2 days ago · 812 bytes of 2 KB`.
452    pub fn summary(&self, now: Timestamp) -> String {
453        let mut parts = Vec::new();
454        if let Some(modified) = self.modified {
455            parts.push(format!("modified {}", ago(modified, now)));
456        }
457        if let Some(size) = self.size_text() {
458            parts.push(size);
459        }
460        parts.join(" · ")
461    }
462
463    /// The exact times, one per line, for a hover: `created: 2026-10-01
464    /// 09:12:44 UTC`.
465    pub fn exact_times(&self) -> String {
466        self.times()
467            .iter()
468            .map(|time| format!("{}: {}", time.label, exact_time(time.at)))
469            .collect::<Vec<_>>()
470            .join("\n")
471    }
472}
473
474/// How long before `now` `then` was: `3 weeks ago`, `less than a minute
475/// ago`, in the units memory reads ages in.
476pub fn ago(then: Timestamp, now: Timestamp) -> String {
477    format!("{} ago", memory_age(then, now))
478}
479
480/// `at` in UTC to the second, such as `2026-10-01 09:12:44 UTC`. Formatted
481/// by hand: jiff's strftime would grow the web page.
482pub fn exact_time(at: Timestamp) -> String {
483    let time = jiff::tz::TimeZone::UTC.to_datetime(at);
484    format!(
485        "{}-{:02}-{:02} {:02}:{:02}:{:02} UTC",
486        time.year(),
487        time.month(),
488        time.day(),
489        time.hour(),
490        time.minute(),
491        time.second()
492    )
493}

633    #[test]
634    fn stamps_read_as_ages_exact_times_and_a_size_against_the_limit() {
635        let now = at("2026-10-22T12:00:00Z");
636        let stamps = MemoryStamps {
637            created: Some(at("2026-10-01T09:12:44Z")),
638            modified: Some(at("2026-10-20T16:03:10Z")),
639            last_read: Some(at("2026-10-22T11:58:00Z")),
640            size: Some(812),
641            max_size: Some(2048),
642        };
643        assert!(!stamps.over_limit());
644        assert_eq!(
645            stamps.detail_lines(now),
646            [
647                "created: 2026-10-01 09:12:44 UTC (3 weeks ago)",
648                "modified: 2026-10-20 16:03:10 UTC (1 day ago)",
649                "last read: 2026-10-22 11:58:00 UTC (2 minutes ago)",
650                "size: 812 bytes of 2 KB",
651            ]
652        );
653        assert_eq!(
654            stamps.summary(now),
655            "modified 1 day ago · 812 bytes of 2 KB"
656        );
657        assert_eq!(
658            stamps.ages(now),
659            "created 3 weeks ago · modified 1 day ago · last read 2 minutes ago · 812 bytes of 2 KB"
660        );
661        assert_eq!(
662            stamps.exact_times(),
663            "created: 2026-10-01 09:12:44 UTC\nmodified: 2026-10-20 16:03:10 UTC\n\
664             last read: 2026-10-22 11:58:00 UTC"
665        );
666
667        let over = MemoryStamps {
668            size: Some(2611),
669            max_size: Some(2048),
670            ..MemoryStamps::default()
671        };
672        assert!(over.over_limit());
673        assert_eq!(
674            over.size_text().as_deref(),
675            Some("2.5 KB of 2 KB, over its limit")
676        );
677        assert_eq!(over.summary(now), "2.5 KB of 2 KB, over its limit");
678        assert!(over.times().is_empty());
679
680        // An older daemon reports a size and nothing else.
681        let older = MemoryStamps {
682            size: Some(64),
683            ..MemoryStamps::default()
684        };
685        assert!(!older.over_limit());
686        assert_eq!(older.detail_lines(now), ["size: 64 bytes"]);
687    }

690    fn a_listing_and_a_read_give_their_stamps() {
691        let info: cookie_agent_protocol::MemoryInfo = serde_json::from_value(serde_json::json!({
692            "url": "user://style", "layer": "user", "revision": "r1", "size": 3000,

695        }))
696        .expect("memory");
697        let listed = MemoryStamps::of_info(&info);
698        assert!(listed.over_limit());
699        assert_eq!(listed.last_read, Some(at("2026-10-21T08:00:00Z")));

704                "size": 3000, "max_size": 2048
705            }))
706            .expect("read");
707        assert_eq!(MemoryStamps::of_read(&read, info.last_read), listed);
708    }
