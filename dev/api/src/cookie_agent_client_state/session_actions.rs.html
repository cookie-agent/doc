16/// The first client protocol whose daemons let any client take back a
17/// waiting (pending or queued) input, not only the client that sent it.
18pub const RECALL_ANY_SINCE: SemVer = SemVer::new(1, 31, 0);
19
20/// Whether the daemon, speaking `version`, lets this client take back
21/// another client's waiting input. An older daemon answers `not_owner`.
22#[must_use]
23pub fn daemon_recalls_any_input(version: Option<SemVer>) -> bool {
24    version.is_some_and(|version| version >= RECALL_ANY_SINCE)
25}

27/// How clients name a permission mode.
28pub const fn permission_mode_label(mode: PermissionMode) -> &'static str {
29    match mode {
30        PermissionMode::AutoApprove => "auto-approve",
31        PermissionMode::AutoApproveN => "auto-n",
32        PermissionMode::AutoApproveY => "auto-y",
33        PermissionMode::Ask => "ask",
34        PermissionMode::Yolo => "yolo",
35    }
36}
37
38/// The mode after `mode` when a client cycles through them.
39pub const fn next_permission_mode(mode: PermissionMode) -> PermissionMode {
40    match mode {
41        PermissionMode::AutoApprove => PermissionMode::AutoApproveN,
42        PermissionMode::AutoApproveN => PermissionMode::AutoApproveY,
43        PermissionMode::AutoApproveY => PermissionMode::Ask,
44        PermissionMode::Ask => PermissionMode::Yolo,
45        PermissionMode::Yolo => PermissionMode::AutoApprove,
46    }
47}

49impl StateStore {
50    /// Installs an `approval.list` of escalated approvals for a tree whose
51    /// sessions are `tree_sessions`: the list replaces only their
52    /// user-visible queues, keeping event-projected internal requests so a
53    /// later, strictly ordered `approval_escalated` can still reveal them.
54    pub fn apply_approval_list(&mut self, tree_sessions: &[SessionId], result: ApprovalListResult) {
55        for session_id in tree_sessions {
56            if let Some(state) = self.sessions.get_mut(session_id) {
57                state.approvals.retain(|approval| !approval.escalated);
58            }
59        }
60        for record in result.approvals {
61            if let Some(approval) = approval_state_from_record(record) {
62                let state = self.sessions.entry(approval.session_id).or_default();
63                state
64                    .approvals
65                    .retain(|candidate| candidate.approval_id != approval.approval_id);
66                state.approvals.push(approval);
67            }
68        }
69    }
70
71    /// The approvals waiting for the user in `sessions`, session by
72    /// session in that order: what a client shows while it shows the first
73    /// of them, with the sessions below it ([`crate::tree::subtree`]), so a
74    /// subagent's approval reaches the session that started it.
75    #[must_use]
76    pub fn waiting_approvals(&self, sessions: &[SessionId]) -> Vec<ApprovalState> {
77        sessions
78            .iter()
79            .filter_map(|session_id| self.sessions.get(session_id))
80            .flat_map(|state| &state.approvals)
81            .filter(|approval| approval.is_visible_user_escalation())
82            .cloned()
83            .collect()
84    }
85
86    /// The agents' questions waiting for the user in `sessions`, as
87    /// [`Self::waiting_approvals`] orders them: each session's oldest
88    /// first.
89    #[must_use]
90    pub fn waiting_questions(&self, sessions: &[SessionId]) -> Vec<QuestionState> {
91        sessions
92            .iter()
93            .filter_map(|session_id| self.sessions.get(session_id))
94            .flat_map(|state| state.questions.iter().cloned())
95            .collect()
96    }
97}

99/// The `full` artifact of display block `index` in `blocks`: where a cut
100/// block's complete content lives.
101pub fn display_full(blocks: &[Open<DisplayBlock>], index: usize) -> Option<&ArtifactRef> {
102    blocks.get(index).and_then(Open::known).and_then(block_full)
103}
104
105/// The `full` artifact of a cut block.
106pub fn block_full(block: &DisplayBlock) -> Option<&ArtifactRef> {
107    match block {
108        DisplayBlock::FileExcerpt { full, .. } | DisplayBlock::Terminal { full, .. } => {
109            full.as_ref()
110        }
111        DisplayBlock::Diff(diff) => diff.full.as_ref(),
112        _ => None,
113    }
114}

116/// What a client must read for the complete content of a cut block: the
117/// artifact and how many bytes of it.
118#[derive(Clone, Debug, Eq, PartialEq)]
119pub struct FullContentRead {
120    pub artifact: ArtifactRef,
121    pub length: u64,
122}
123
124impl SessionState {
125    /// Toggles the complete content of display block `block` of `call_id`:
126    /// loaded content is put away, a read in flight stays, and otherwise
127    /// the block is marked loading and the read to make is returned.
128    pub fn toggle_full_content(
129        &mut self,
130        call_id: ToolCallId,
131        block: usize,
132    ) -> Option<FullContentRead> {
133        let tool = self.tools.get(&call_id)?;
134        match tool.full_content(block) {
135            Some(FullContentState::Loading) => return None,
136            Some(FullContentState::Loaded { .. }) => {
137                self.update_tool(call_id, |tool| tool.full.retain(|full| full.block != block));
138                return None;
139            }
140            Some(FullContentState::Failed(_)) | None => {}
141        }
142        let artifact = display_full(tool.shown_display(), block)?.clone();
143        self.update_tool(call_id, |tool| {
144            tool.full.retain(|full| full.block != block);
145            tool.full.push(FullContent {
146                block,
147                state: FullContentState::Loading,
148            });
149        });
150        Some(FullContentRead {
151            length: artifact.byte_length.clamp(1, MAX_FULL_CONTENT_BYTES),
152            artifact,
153        })
154    }
155
156    /// The read for block `block` of `call_id` ended. A call that ended or
157    /// was put away since keeps what it shows now.
158    pub fn finish_full_content(
159        &mut self,
160        call_id: ToolCallId,
161        block: usize,
162        state: FullContentState,
163    ) {
164        let pending = self
165            .tools
166            .get(&call_id)
167            .is_some_and(|tool| tool.full_content(block) == Some(&FullContentState::Loading));
168        if pending {
169            self.update_tool(call_id, |tool| {
170                if let Some(full) = tool.full.iter_mut().find(|full| full.block == block) {
171                    full.state = state;
172                }
173            });
174        }
175    }
176}
177
178impl FullContentState {
179    /// The state an `artifact.read` of a block's complete content leaves.
180    pub fn of_read(read: Result<ArtifactReadResult, String>) -> Self {
181        read.and_then(|read| {
182            use base64::Engine as _;
183            let bytes = base64::engine::general_purpose::STANDARD
184                .decode(read.data)
185                .map_err(|error| format!("the daemon sent malformed data: {error}"))?;
186            Ok(Self::Loaded {
187                text: String::from_utf8_lossy(&bytes).into_owned(),
188                truncated: read.total_bytes > bytes.len() as u64,
189            })
190        })
191        .unwrap_or_else(Self::Failed)
192    }
193}

195#[cfg(test)]
196mod tests {
197    use cookie_agent_protocol::PermissionMode;
198
199    use super::{next_permission_mode, permission_mode_label};
200
201    #[test]
202    fn the_mode_cycle_visits_every_mode() {
203        let mut mode = PermissionMode::AutoApprove;
204        let mut seen = Vec::new();
205        for _ in 0..5 {
206            seen.push(permission_mode_label(mode));
207            mode = next_permission_mode(mode);
208        }
209        assert_eq!(mode, PermissionMode::AutoApprove);
210        assert_eq!(seen, ["auto-approve", "auto-n", "auto-y", "ask", "yolo"]);
211    }
