15/// Where a webhook's events go, in words: the session it delivers into,
16/// linked, then the rest, such as `queued after the current turn`, or for
17/// new sessions `new sessions · primary · one per issue.number`.
18#[derive(Clone, Debug, Eq, PartialEq)]
19pub struct TargetWords {
20    /// The session events go into, and its name.
21    pub session: Option<(SessionId, String)>,
22    pub rest: String,
23}
24
25/// [`TargetWords`] for `webhook`. `title` names a session the client knows;
26/// `projects` is the client's `projects.list`.
27pub fn target_words(
28    webhook: &Webhook,
29    projects: &[ProjectInfo],
30    title: impl Fn(SessionId) -> Option<String>,
31) -> TargetWords {
32    let name = |session: SessionId| {
33        title(session).unwrap_or_else(|| session.to_string().chars().take(8).collect())
34    };
35    let delivery = match webhook.delivery.unwrap_or(ProducerDeliveryMode::Queue) {
36        ProducerDeliveryMode::Steer => "into the running turn",
37        _ => "queued after the current turn",
38    };
39    match &webhook.target {
40        WebhookTarget::Session(session) => TargetWords {
41            session: Some((*session, name(*session))),
42            rest: delivery.to_owned(),
43        },
44        WebhookTarget::Subagent(session) => TargetWords {
45            session: Some((*session, name(*session))),
46            rest: format!("a subagent · {delivery}"),
47        },
48        WebhookTarget::New(template) => {
49            let mut rest = format!("new sessions · {}", template.agent);
50            if template.project.is_some() {
51                rest.push_str(&format!(
52                    " · {}",
53                    crate::webhook::template_project(template, projects)
54                ));
55            }
56            rest.push_str(&match &template.thread_key {
57                Some(source) => format!(" · one per {}", source_words(source)),
58                None => " · one per event".to_owned(),
59            });
60            TargetWords {
61                session: None,
62                rest,
63            }
64        }
65    }
66}

68/// A thread key's source as a person reads it: a JSON Pointer as a dotted
69/// path (`/issue/number` is `issue.number`), a header as `X-Thread header`.
70pub fn source_words(source: &WebhookValueSource) -> String {
71    match (&source.header, &source.json) {
72        (Some(header), _) => format!("{header} header"),
73        (None, Some(pointer)) => {
74            let path = pointer
75                .trim_start_matches('/')
76                .split('/')
77                .map(|part| part.replace("~1", "/").replace("~0", "~"))
78                .collect::<Vec<_>>()
79                .join(".");
80            if path.is_empty() {
81                "body".to_owned()
82            } else {
83                path
84            }
85        }
86        (None, None) => "event".to_owned(),
87    }
88}

90/// The tags a webhook's row carries, for what its switch cannot say: off
91/// after failures, and no check (W4).
92pub fn row_tags(webhook: &Webhook) -> Vec<String> {
93    let mut tags = Vec::new();
94    if !webhook.enabled
95        && let Some(Open::Known(WebhookDisabledReason::Failed { failures })) =
96            &webhook.disabled_reason
97    {
98        let noun = if *failures == 1 {
99            "failure"
100        } else {
101            "failures"
102        };
103        tags.push(format!("off after {failures} {noun}"));
104    }
105    if webhook.verify.is_none() {
106        tags.push("no check".to_owned());
107    }
108    tags
109}

111/// How a webhook is doing, for its row's dot.
112#[derive(Clone, Copy, Debug, Eq, PartialEq)]
113pub enum Tone {
114    /// Its last event was delivered, or waits to be.
115    Good,
116    /// A request was refused since its last event.
117    Warn,
118    /// Its last event failed, or failures turned it off.
119    Bad,
120    /// No events yet.
121    Quiet,
122}

124/// A request turned away: the newest of a kind that means something may be
125/// wrong (rejected, rate-limited, or the backlog full).
126#[derive(Clone, Debug, Eq, PartialEq)]
127pub struct Refusal {
128    pub at: Timestamp,
129    /// How many of its kind.
130    pub count: u64,
131    /// `rejected`, `rate-limited`, or `turned away`.
132    pub what: &'static str,
133    /// What it means for this webhook, as [`rejection_reason`] says it.
134    pub reason: String,
135}

137/// A row's health line (H1): its tone, its last event (tests included,
138/// W2), and the last refusal (W3).
139#[derive(Clone, Debug, Eq, PartialEq)]
140pub struct Health {
141    pub tone: Tone,
142    /// When the newest event arrived, and what it did.
143    pub last_event: Option<(Timestamp, String)>,
144    pub refusal: Option<Refusal>,
145}
146
147/// [`Health`] for `webhook`.
148pub fn health(webhook: &Webhook) -> Health {
149    let newest = webhook.history.iter().max_by_key(|event| event.received_at);
150    let last_event = newest.map(|event| (event.received_at, event_outcome(event)));
151    let counts = &webhook.counts;
152    let refusal = [
153        (counts.last_rejected, counts.rejected, "rejected"),
154        (
155            counts.last_rate_limited,
156            counts.rate_limited,
157            "rate-limited",
158        ),
159        (counts.last_backlog_full, counts.backlog_full, "turned away"),
160    ]
161    .into_iter()
162    .filter_map(|(at, count, what)| Some((at?, count, what)))
163    .max_by_key(|(at, _, _)| *at)
164    .map(|(at, count, what)| Refusal {
165        at,
166        count,
167        what,
168        reason: match what {
169            "rejected" => rejection_reason(webhook.verify.as_ref()).to_owned(),
170            "rate-limited" => rate_reason(webhook.max_per_hour),
171            _ => "too many events were waiting".to_owned(),
172        },
173    });
174    let failed_off = !webhook.enabled
175        && matches!(
176            webhook.disabled_reason,
177            Some(Open::Known(WebhookDisabledReason::Failed { .. }))
178        );
179    let tone =
180        if failed_off || newest.is_some_and(|event| outcome_kind(event) == OutcomeKind::Failed) {
181            Tone::Bad
182        } else if refusal
183            .as_ref()
184            .is_some_and(|refusal| last_event.as_ref().is_none_or(|(at, _)| refusal.at > *at))
185        {
186            Tone::Warn
187        } else if last_event.is_some() {
188            Tone::Good
189        } else {
190            Tone::Quiet
191        };
192    Health {
193        tone,
194        last_event,
195        refusal,
196    }
197}

199/// What a rejected request (`401`) means for a webhook's check.
200pub fn rejection_reason(verify: Option<&WebhookVerify>) -> &'static str {
201    match verify.map(|verify| verify.kind) {
202        Some(WebhookVerifyKind::Github | WebhookVerifyKind::HmacSha256) => {
203            "the signature didn't match the secret"
204        }
205        Some(WebhookVerifyKind::Standard) => "the signature didn't match, or it was too old",
206        Some(WebhookVerifyKind::Bearer) => "the bearer token was missing or wrong",
207        None => "it was refused",
208    }
209}
210
211fn rate_reason(max_per_hour: Option<i64>) -> String {
212    match max_per_hour {
213        Some(limit) if limit > 0 => format!("over {limit} events an hour"),
214        _ => "over the hourly limit".to_owned(),
215    }
216}

218/// One of an open webhook's counters (H3).
219#[derive(Clone, Debug, Eq, PartialEq)]
220pub struct Counter {
221    /// `Delivered`, `Rejected`, `Filtered`, `Rate-limited`, `Duplicates`,
222    /// `Backlog full`.
223    pub what: &'static str,
224    /// The status the sender was answered.
225    pub status: u16,
226    pub count: u64,
227    pub last: Option<Timestamp>,
228    pub meaning: String,
229    /// Whether a nonzero count means something may be wrong.
230    pub warn: bool,
231}
232
233/// The counters an open webhook shows: delivered, rejected, filtered, and
234/// rate-limited always; duplicates and a full backlog once they happen.
235pub fn counters(webhook: &Webhook) -> Vec<Counter> {
236    let counts = &webhook.counts;
237    let counter = |what, status, count: u64, last, meaning: String, warn: bool| Counter {
238        what,
239        status,
240        count,
241        last,
242        meaning,
243        warn: warn && count > 0,
244    };
245    let mut counters = vec![
246        counter(
247            "Delivered",
248            202,
249            counts.accepted,
250            counts.last_accepted,
251            "accepted and passed on; test events are not counted".to_owned(),
252            false,
253        ),
254        counter(
255            "Rejected",
256            401,
257            counts.rejected,
258            counts.last_rejected,
259            rejection_reason(webhook.verify.as_ref()).to_owned(),
260            true,
261        ),
262        counter(
263            "Filtered",
264            204,
265            counts.filtered,
266            counts.last_filtered,
267            "didn't match the filter".to_owned(),
268            false,
269        ),
270        counter(
271            "Rate-limited",
272            429,
273            counts.rate_limited,
274            counts.last_rate_limited,
275            rate_reason(webhook.max_per_hour),
276            true,
277        ),
278    ];
279    if counts.duplicate > 0 {
280        counters.push(counter(
281            "Duplicates",
282            200,
283            counts.duplicate,
284            counts.last_duplicate,
285            "a delivery ID seen before".to_owned(),
286            false,
287        ));
288    }
289    if counts.backlog_full > 0 {
290        counters.push(counter(
291            "Backlog full",
292            503,
293            counts.backlog_full,
294            counts.last_backlog_full,
295            "too many events were waiting".to_owned(),
296            true,
297        ));
298    }
299    counters
300}

302/// One line of an open webhook's "What happens" (H2).
303#[derive(Clone, Debug, Eq, PartialEq)]
304pub struct Happening {
305    pub heading: String,
306    pub detail: Option<String>,
307}
308
309/// What happens to an event, in a few lines: where it goes, what a new
310/// session is told first, what comes with every event, and what the
311/// sessions are called. `title` names a session the client knows.
312pub fn what_happens(
313    webhook: &Webhook,
314    title: impl Fn(SessionId) -> Option<String>,
315) -> Vec<Happening> {
316    let line = |heading: String, detail: Option<String>| Happening { heading, detail };
317    let mut lines = Vec::new();
318    let at_once = match webhook.max_concurrent {
319        Some(limit) if limit > 0 => format!("up to {limit} at once"),
320        Some(_) => "any number at once".to_owned(),
321        None => "as many at once as the daemon allows".to_owned(),
322    };
323    match &webhook.target {
324        WebhookTarget::Session(session) | WebhookTarget::Subagent(session) => {
325            let name = title(*session).unwrap_or_else(|| "its session".to_owned());
326            let when = match webhook.delivery.unwrap_or(ProducerDeliveryMode::Queue) {
327                ProducerDeliveryMode::Steer => "into the turn running then",
328                _ => "after the turn running then",
329            };
330            lines.push(line(
331                format!("Each event goes into {name}"),
332                Some(when.to_owned()),
333            ));
334        }
335        WebhookTarget::New(template) => {
336            match &template.thread_key {
337                Some(source) => {
338                    let key = source_words(source);
339                    lines.push(line(
340                        format!("Each {key} gets its own session, with {}", template.agent),
341                        Some(format!(
342                            "Later events with the same {key} go into it · {at_once}"
343                        )),
344                    ));
345                }
346                None => lines.push(line(
347                    format!("Each event starts a new session, with {}", template.agent),
348                    Some(at_once),
349                )),
350            }
351            if let Some(prompt) = &template.thread_prompt {
352                lines.push(line("Its first message".to_owned(), Some(prompt.clone())));
353            }
354            lines.push(line(
355                "Sessions are called".to_owned(),
356                Some(session_title_example(webhook)),
357            ));
358        }
359    }
360    lines.push(line(
361        "With every event".to_owned(),
362        Some(
363            webhook
364                .prompt
365                .clone()
366                .unwrap_or_else(|| "The request alone, marked as a webhook event".to_owned()),
367        ),
368    ));
369    lines
370}

372/// How a new-session webhook's sessions are called, as an example: the
373/// title or the webhook's name, then the thread key or the time.
374pub fn session_title_example(webhook: &Webhook) -> String {
375    let WebhookTarget::New(template) = &webhook.target else {
376        return String::new();
377    };
378    if template.title_auto {
379        return "named by the model after the first turn".to_owned();
380    }
381    let name = template
382        .title
383        .clone()
384        .unwrap_or_else(|| crate::webhook::webhook_title(webhook));
385    let suffix = match &template.thread_key {
386        Some(source) => format!("the {}", source_words(source)),
387        None => "the time".to_owned(),
388    };
389    format!("{name} · {suffix}")
390}

501/// What the form's "one session per" reads (H4).
502#[derive(Clone, Copy, Debug, Eq, PartialEq)]
503pub enum ThreadBy {
504    /// A JSON field of the body.
505    Json,
506    Header,
507    /// No thread key: each event starts its own session.
508    Event,
509}
510
511/// Reads the form's thread key into its choice and the field's text.
512pub fn thread_by(thread_key: &str) -> (ThreadBy, String) {
513    let text = thread_key.trim();
514    if let Some(name) = text.strip_prefix("header:") {
515        (ThreadBy::Header, name.trim().to_owned())
516    } else if let Some(pointer) = text.strip_prefix("json:") {
517        (ThreadBy::Json, pointer.trim().to_owned())
518    } else if text.is_empty() {
519        (ThreadBy::Event, String::new())
520    } else {
521        // Typed without its prefix: a pointer is JSON, else a header.
522        if text.starts_with('/') {
523            (ThreadBy::Json, text.to_owned())
524        } else {
525            (ThreadBy::Header, text.to_owned())
526        }
527    }
528}
529
530/// The thread key the form writes for a choice and its field.
531pub fn thread_key_text(by: ThreadBy, value: &str) -> String {
532    let value = value.trim();
533    match by {
534        ThreadBy::Event => String::new(),
535        ThreadBy::Json => format!("json:{value}"),
536        ThreadBy::Header => format!("header:{value}"),
537    }
538}

540/// One condition of the form's filter, as its builder edits it (H4).
541#[derive(Clone, Debug, Eq, PartialEq)]
542pub struct Condition {
543    /// A header, else a JSON field.
544    pub header: bool,
545    /// The header's name or the JSON Pointer.
546    pub name: String,
547    pub values: Vec<String>,
548}
549
550/// The form's filter lines as conditions, or what is wrong with one.
551pub fn conditions(filter: &str) -> Result<Vec<Condition>, String> {
552    filter
553        .lines()
554        .map(str::trim)
555        .filter(|line| !line.is_empty())
556        .map(|line| {
557            let condition = parse_condition(line)?;
558            Ok(match (condition.header, condition.json) {
559                (Some(header), _) => Condition {
560                    header: true,
561                    name: header,
562                    values: condition.one_of,
563                },
564                (None, json) => Condition {
565                    header: false,
566                    name: json.unwrap_or_default(),
567                    values: condition.one_of,
568                },
569            })
570        })
571        .collect()
572}
573
574/// Conditions back as the form's filter lines; a condition with neither a
575/// name nor values is left out.
576pub fn conditions_text(conditions: &[Condition]) -> String {
577    conditions
578        .iter()
579        .filter(|condition| !condition.name.trim().is_empty() || !condition.values.is_empty())
580        .map(|condition| {
581            let kind = if condition.header { "header" } else { "json" };
582            format!(
583                "{kind}:{}={}",
584                condition.name.trim(),
585                condition.values.join(",")
586            )
587        })
588        .collect::<Vec<_>>()
589        .join("\n")
590}
