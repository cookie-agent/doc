1//! How much of an agent's work the transcript shows in every client
2//! (`docs/site/specs/reduced-transcript.md`): every thinking block and tool
3//! call, or each assistant block's work folded into one `Working…` /
4//! `Worked for` row above its reply.

12/// How the transcript shows agent work.
13#[derive(Clone, Copy, Debug, Default, Eq, Hash, PartialEq)]
14pub enum TranscriptMode {
15    /// Each block's work folds into one row above its reply.
16    #[default]
17    Reduced,
18    /// Every thinking block and tool call shows.
19    Full,
20}
21
22impl TranscriptMode {
23    /// The mode's name in configuration and storage.
24    #[must_use]
25    pub const fn name(self) -> &'static str {
26        match self {
27            Self::Reduced => "reduced",
28            Self::Full => "full",
29        }
30    }
31
32    /// The mode a stored name names; `None` for any other text.
33    #[must_use]
34    pub fn parse(name: &str) -> Option<Self> {
35        match name {
36            "reduced" => Some(Self::Reduced),
37            "full" => Some(Self::Full),
38            _ => None,
39        }
40    }
41
42    /// The other mode.
43    #[must_use]
44    pub const fn toggled(self) -> Self {
45        match self {
46            Self::Reduced => Self::Full,
47            Self::Full => Self::Reduced,
48        }
49    }
50}

52/// Where a block's reply starts among its `children`: after its last
53/// thinking part or tool call. Everything before is its work, the text
54/// written between calls included. `None` when the block has no work, so
55/// it shows as in the full mode.
56#[must_use]
57pub fn reply_start(children: &[AssistantChild]) -> Option<usize> {
58    children
59        .iter()
60        .rposition(|child| {
61            matches!(
62                child,
63                AssistantChild::Thinking { .. }
64                    | AssistantChild::Tool { .. }
65                    | AssistantChild::CommittedTool { .. }
66            )
67        })
68        .map(|index| index + 1)
69}

71/// Whether the call `call_id` waits on the user's approval, so it shows
72/// outside its block's fold until answered: an approval escalated to the
73/// user that has not expired.
74#[must_use]
75pub fn needs_user(state: &SessionState, call_id: &ToolCallId) -> bool {
76    state
77        .approvals
78        .iter()
79        .any(|approval| &approval.tool_call_id == call_id && approval.is_visible_user_escalation())
80}

82/// What a block's work row says.
83#[derive(Clone, Debug, Eq, PartialEq)]
84pub struct WorkSummary {
85    /// The block still takes output: `Working…`, its time ticking.
86    pub live: bool,
87    /// Its run was cancelled or interrupted: `Stopped after`.
88    pub stopped: bool,
89    pub elapsed: Duration,
90    pub tool_calls: usize,
91    pub failed: usize,
92}
93
94impl WorkSummary {
95    /// The row's label: `Working… 42s · 3 tool calls`, `Worked for 1m 23s`,
96    /// or `Stopped after 12s`, the count left out at none.
97    #[must_use]
98    pub fn label(&self) -> String {
99        let elapsed = format_elapsed(self.elapsed);
100        let mut label = if self.live {
101            format!("Working… {elapsed}")
102        } else if self.stopped {
103            format!("Stopped after {elapsed}")
104        } else {
105            format!("Worked for {elapsed}")
106        };
107        match self.tool_calls {
108            0 => {}
109            1 => label.push_str(" · 1 tool call"),
110            count => label.push_str(&format!(" · {count} tool calls")),
111        }
112        label
113    }
114
115    /// The failed calls, shown after the label in the warning colour:
116    /// `· 1 failed`; `None` at none.
117    #[must_use]
118    pub fn failed_label(&self) -> Option<String> {
119        (self.failed > 0).then(|| format!("· {} failed", self.failed))
120    }
121}

123/// The work row of the assistant block `item_id`, whose work is `work`
124/// (its children before [`reply_start`]), at `now`. The time runs from the
125/// block's durable start to its durable end, or to `now` while it is live.
126#[must_use]
127pub fn work_summary(
128    state: &SessionState,
129    item_id: u64,
130    work: &[AssistantChild],
131    now: jiff::Timestamp,
132) -> WorkSummary {
133    let end = state.assistant_ends.get(&item_id).copied();
134    let live = end.is_none() && state.assistant_live(item_id);
135    let started = state.item_time(item_id);
136    let ended = match end {
137        Some(end) => Some(end.at),
138        None if live => Some(now),
139        // A block that never sealed and is not open (a run with no end in
140        // the log): its last part's arrival.
141        None => state.assistant_last_child_time(item_id),
142    };
143    let elapsed = started
144        .zip(ended)
145        .and_then(|(started, ended)| Duration::try_from(ended.duration_since(started)).ok())
146        .unwrap_or_default();
147    let mut tool_calls = 0;
148    let mut failed = 0;
149    for child in work {
150        match child {
151            AssistantChild::Tool { call_id } => {
152                tool_calls += 1;
153                if state
154                    .tools
155                    .get(call_id)
156                    .is_some_and(|tool| tool.status == ToolStatus::Failed)
157                {
158                    failed += 1;
159                }
160            }
161            AssistantChild::CommittedTool { .. } => tool_calls += 1,
162            _ => {}
163        }
164    }
165    WorkSummary {
166        live,
167        stopped: end.is_some_and(|end| end.stopped),
168        elapsed,
169        tool_calls,
170        failed,
171    }
172}

174/// The line under a live work row: what its newest part is doing. A
175/// thinking part's newest step or line, a tool call's header (`Waiting for
176/// approval: …` while it waits on the user), or the newest line of text
177/// between calls; `Starting…` before any says anything.
178#[must_use]
179pub fn working_subtext(state: &SessionState, work: &[AssistantChild]) -> String {
180    work.iter()
181        .rev()
182        .find_map(|child| match child {
183            AssistantChild::Thinking { text, .. } => {
184                Some(thinking_step(text).unwrap_or_else(|| "Thinking…".to_owned()))
185            }
186            AssistantChild::Tool { call_id } => {
187                let tool = state.tools.get(call_id)?;
188                let header = tool.header_text();
189                Some(if needs_user(state, call_id) {
190                    format!("Waiting for approval: {header}")
191                } else {
192                    header
193                })
194            }
195            AssistantChild::CommittedTool { name, .. } => Some(name.as_str().to_owned()),
196            AssistantChild::Text { markdown, .. } => newest_line(markdown.as_str()),
197            AssistantChild::Attribution { .. }
198            | AssistantChild::MediaFile { .. }
199            | AssistantChild::Notice { .. } => None,
200        })
201        .unwrap_or_else(|| "Starting…".to_owned())
202}

204/// The step a thinking trace is on: its newest Markdown heading or line
205/// that is bold alone (`**Inspecting the reducer**`, as OpenAI's reasoning
206/// summaries start each step), without its markers; else its newest line
207/// with text. `None` while it has none.
208#[must_use]
209pub fn thinking_step(text: &str) -> Option<String> {
210    let step = text.lines().rev().find_map(|line| {
211        let line = line.trim();
212        let heading = line.trim_start_matches('#');
213        if heading.len() < line.len() && heading.starts_with(' ') {
214            return Some(heading.trim());
215        }
216        line.strip_prefix("**")
217            .and_then(|rest| rest.strip_suffix("**"))
218            .filter(|inner| !inner.is_empty() && !inner.contains("**"))
219            .map(str::trim)
220    });
221    step.filter(|step| !step.is_empty())
222        .map(crate::tool_header::flatten_header_argument)
223        .or_else(|| newest_line(text))
224}
225
226/// The newest line of `text` that has any, whitespace collapsed.
227fn newest_line(text: &str) -> Option<String> {
228    text.lines()
229        .rev()
230        .map(crate::tool_header::flatten_header_argument)
231        .find(|line| !line.is_empty())
232}
