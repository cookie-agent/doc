10/// The icon for a tool hint (spec 12.2). Hints are open labels: one this
11/// build does not document, or none, gets the generic icon. The tool's name
12/// is never consulted.
13pub fn hint_icon(hint: Option<&str>) -> &'static str {
14    match hint.unwrap_or(ToolHint::GENERIC) {
15        ToolHint::SHELL => "💻",
16        ToolHint::FILE_READ => "📖",
17        ToolHint::FILE_WRITE | ToolHint::FILE_EDIT => "📝",
18        ToolHint::FETCH => "🌐",
19        ToolHint::SEARCH => "🔍",
20        ToolHint::DELEGATE => "🤖",
21        ToolHint::MESSAGE => "📨",
22        ToolHint::SKILL => "✨",
23        ToolHint::GOAL => "🎯",
24        _ => "🔨",
25    }
26}

28/// Flatten an on-wire primary argument for single-line header display. The
29/// wire type is byte-capped only, so control characters reach the client:
30/// map each to a space before collapsing whitespace runs.
31pub fn flatten_header_argument(argument: &str) -> String {
32    argument
33        .chars()
34        .map(|character| {
35            if character.is_control() {
36                ' '
37            } else {
38                character
39            }
40        })
41        .collect::<String>()
42        .split_whitespace()
43        .collect::<Vec<_>>()
44        .join(" ")
45}

47/// The schemes a memory URL starts with, one per memory layer.
48const MEMORY_SCHEMES: [&str; 4] = ["user://", "agent://", "machine://", "project://"];
49
50/// Whether `text` is a memory URL, such as `project://cookie-agent/release`:
51/// a layer's scheme, then a name, and no whitespace. A client opens one in
52/// its Memory view, from a call's header or from a display `link` block.
53pub fn is_memory_url(text: &str) -> bool {
54    MEMORY_SCHEMES.iter().any(|scheme| {
55        text.strip_prefix(scheme)
56            .is_some_and(|rest| !rest.is_empty() && !rest.contains(char::is_whitespace))
57    })
58}
59
60/// The byte range of the first memory URL in `text`, a header as a client
61/// drew it: a word that starts with a layer's scheme, up to the next
62/// whitespace. A URL the header cut short ends at its `…`, which the range
63/// covers.
64pub fn memory_url_span(text: &str) -> Option<std::ops::Range<usize>> {
65    let mut start = 0;
66    for word in text.split_inclusive(char::is_whitespace) {
67        let trimmed = word.trim_end_matches(char::is_whitespace);
68        if MEMORY_SCHEMES
69            .iter()
70            .any(|scheme| trimmed.len() > scheme.len() && trimmed.starts_with(scheme))
71        {
72            return Some(start..start + trimmed.len());
73        }
74        start += word.len();
75    }
76    None
77}

79/// Characters of an argument a title must show before its `…` for the
80/// title to count as naming the argument it cut short.
81const MIN_CUT_ARGUMENT_CHARS: usize = 8;
82
83/// Whether `title` already shows `argument`, whole or cut short before a
84/// `…`, so the header does not repeat it.
85fn title_names_argument(title: &str, argument: &str) -> bool {
86    if argument.is_empty() || title.contains(argument) {
87        return true;
88    }
89    // A path the title names by its trailing parts, as an agent's
90    // `Read src/main.rs` names `/work/src/main.rs`.
91    if !argument.contains(char::is_whitespace)
92        && argument
93            .match_indices('/')
94            .map(|(slash, _)| &argument[slash + 1..])
95            .any(|tail| !tail.is_empty() && title_has_path(title, tail))
96    {
97        return true;
98    }
99    title.match_indices('…').any(|(end, _)| {
100        let shown = &title[..end];
101        shown.char_indices().any(|(start, _)| {
102            let prefix = &shown[start..];
103            prefix.chars().count() >= MIN_CUT_ARGUMENT_CHARS && argument.starts_with(prefix)
104        })
105    })
106}

108/// Whether `title` holds `path` as a whole path: not inside a longer word
109/// or path (`a.rs` is not named by `data.rs`).
110fn title_has_path(title: &str, path: &str) -> bool {
111    title.match_indices(path).any(|(start, _)| {
112        let before = title[..start].chars().next_back();
113        let after = title[start + path.len()..].chars().next();
114        before.is_none_or(|character| character.is_whitespace() || "`'\"(".contains(character))
115            && after
116                .is_none_or(|character| character.is_whitespace() || "`'\"):,".contains(character))
117    })
118}

120/// The texts a header already shows of a call: its primary argument, and
121/// what its title quotes between backticks, each trimmed.
122fn header_texts(presentation: &ToolPresentation) -> Vec<String> {
123    let mut texts = Vec::new();
124    if let Some(argument) = &presentation.primary_argument {
125        texts.push(argument.as_str().trim().to_owned());
126    }
127    let title = presentation.title.as_str();
128    let mut quoted = title.split('`').skip(1).step_by(2);
129    // Only a closed quote counts: `split` yields an unclosed one too.
130    if title.matches('`').count() >= 2 {
131        texts.extend(quoted.by_ref().map(|text| text.trim().to_owned()));
132    }
133    texts.retain(|text| !text.is_empty());
134    texts
135}

137/// The argument blocks an expanded call shows: a block whose text is what
138/// the header already shows (the primary argument, or the text its title
139/// quotes) is left out, and so is a key/value entry whose value is the
140/// primary argument, and a key/value block left with no entries. A block
141/// that differs, such as a long command the header shortened, stays. The
142/// decision reads the presentation only, never the tool's name.
143pub fn shown_arguments(presentation: &ToolPresentation) -> Vec<Open<DisplayBlock>> {
144    let primary = presentation
145        .primary_argument
146        .as_ref()
147        .map(|argument| argument.as_str());
148    let shown = header_texts(presentation);
149    let repeats = |text: &str| {
150        let text = text.trim();
151        shown.iter().any(|shown| shown == text)
152    };
153    presentation
154        .arguments
155        .iter()
156        .filter_map(|block| match block {
157            Open::Known(
158                DisplayBlock::Text { text, .. }
159                | DisplayBlock::Code { text, .. }
160                | DisplayBlock::Markdown { text, .. },
161            ) if repeats(text) => None,
162            Open::Known(DisplayBlock::KeyValue { entries, .. }) if primary.is_some() => {
163                let kept = entries
164                    .iter()
165                    .filter(|entry| Some(entry.value.as_str()) != primary)
166                    .cloned()
167                    .collect::<Vec<_>>();
168                if kept.is_empty() {
169                    return None;
170                }
171                let mut block = block.clone();
172                if let Open::Known(DisplayBlock::KeyValue { entries, .. }) = &mut block {
173                    *entries = kept;
174                }
175                Some(block)
176            }
177            _ => Some(block.clone()),
178        })
179        .collect()
180}

182impl ToolCallState {
183    /// The argument blocks its expanded body shows ([`shown_arguments`]).
184    pub fn shown_arguments(&self) -> Vec<Open<DisplayBlock>> {
185        shown_arguments(&self.presentation)
186    }
187
188    /// The primary argument as the header shows it after the title,
189    /// flattened to one line; `None` when there is none or the title
190    /// already names it.
191    pub fn header_argument(&self) -> Option<String> {
192        let argument =
193            flatten_header_argument(self.presentation.primary_argument.as_ref()?.as_str());
194        let title = flatten_header_argument(self.presentation.title.as_str());
195        (!title_names_argument(&title, &argument)).then_some(argument)
196    }
197
198    /// The memory URL the call names as its primary argument, which its
199    /// header links to the Memory view; `None` for any other call. The
200    /// decision reads the argument only, never the tool's name.
201    pub fn memory_url(&self) -> Option<&str> {
202        let argument = self.presentation.primary_argument.as_ref()?.as_str().trim();
203        is_memory_url(argument).then_some(argument)
204    }
205
206    /// The header's text: the title, then the argument it does not name.
207    pub fn header_text(&self) -> String {
208        let title = flatten_header_argument(self.presentation.title.as_str());
209        match self.header_argument() {
210            Some(argument) => format!("{title} {argument}"),
211            None => title,
212        }
213    }
214}

228    fn presentation(
229        title: &str,
230        primary: Option<&str>,
231        arguments: Vec<DisplayBlock>,
232    ) -> ToolPresentation {
233        let prepared = PreparedPresentation::new(
234            SafeDisplayText::new(title).expect("title"),
235            primary.map(|primary| BoundedDisplayText::new(primary).expect("argument")),
236        )
237        .with_hint("shell")
238        .with_arguments(arguments);
239        ToolPresentation::new(
240            prepared,
241            ToolIdentity {
242                name: "x".into(),
243                published_name: "x".into(),
244                source: None,
245            },
246        )
247    }
248
249    fn code(text: &str) -> DisplayBlock {
250        DisplayBlock::Code {
251            text: text.into(),
252            language: Some("bash".into()),
253            start_line: None,
254            omitted_lines: None,
255        }
256    }

258    #[test]
259    fn a_block_the_header_already_shows_is_left_out() {
260        // The title quotes it.
261        let quoted = presentation(
262            "Run `python3 test_greet.py`",
263            None,
264            vec![code(" python3 test_greet.py\n")],
265        );
266        assert!(shown_arguments(&quoted).is_empty());
267        // The primary argument is it.
268        let primary = presentation(
269            "Run a command",
270            Some("cargo test"),
271            vec![code("cargo test")],
272        );
273        assert!(shown_arguments(&primary).is_empty());
274        // A key/value entry of the primary argument, beside one kept.
275        let entries = presentation(
276            "Fetch a page",
277            Some("https://example.com"),
278            vec![DisplayBlock::KeyValue {
279                entries: vec![
280                    KeyValueEntry {
281                        key: "url".into(),
282                        value: "https://example.com".into(),
283                    },
284                    KeyValueEntry {
285                        key: "format".into(),
286                        value: "markdown".into(),
287                    },
288                ],
289                omitted_entries: None,
290            }],
291        );
292        let shown = shown_arguments(&entries);
293        let [Open::Known(DisplayBlock::KeyValue { entries, .. })] = shown.as_slice() else {
294            panic!("one key/value block: {shown:?}")
295        };
296        assert_eq!(entries.len(), 1);
297        assert_eq!(entries[0].key, "format");
298    }

300    #[test]
301    fn a_block_that_says_more_than_the_header_stays() {
302        // The header shortened a long command: the block has all of it.
303        let long = "for f in *.rs; do\n  rustfmt --check \"$f\"\ndone";
304        let shortened = presentation("Run `for f in *.rs; do…`", None, vec![code(long)]);
305        assert_eq!(shown_arguments(&shortened).len(), 1);
306        let other = presentation("Run `ls`", Some("ls"), vec![code("ls -la")]);
307        assert_eq!(shown_arguments(&other).len(), 1);
308        // An unclosed backtick quotes nothing.
309        let unclosed = presentation("Run `ls", None, vec![code("ls")]);
310        assert_eq!(shown_arguments(&unclosed).len(), 1);
311    }

313    #[test]
314    fn a_title_that_names_its_argument_is_not_repeated() {
315        assert!(title_names_argument("Run `sleep 8`", "sleep 8"));
316        assert!(title_names_argument(
317            "Run `printf 'a long…`",
318            "printf 'a long command'"
319        ));
320        assert!(!title_names_argument("Run a command", "sleep 8"));
321        assert!(!title_names_argument("Run `pr…`", "printf 'x'"));
322        // A path the title names by its trailing parts.
323        assert!(title_names_argument(
324            "Read src/main.rs (10 - 20)",
325            "/work/app/src/main.rs"
326        ));
327        assert!(title_names_argument(
328            "Edit main.rs",
329            "/work/app/src/main.rs"
330        ));
331        assert!(!title_names_argument("Edit data.rs", "/work/a.rs"));
332        assert!(!title_names_argument(
333            "Read other/main.rs",
334            "/work/src/main.rs"

336        assert_eq!(flatten_header_argument("a\n\tb  c"), "a b c");

347    fn memory_urls_are_found_by_their_layer_scheme() {
348        for url in [
349            "user://preferences",
350            "agent://builder",
351            "machine://labpc/gpu",
352            "project://cookie-agent/release",

354            assert!(is_memory_url(url), "{url}");

357            "user://",

365        assert_eq!(

367            Some("agent://builder")

369        let cut = "Edit memory project://cookie-agent/rel…";
370        assert_eq!(
371            memory_url_span(cut).map(|span| &cut[span]),
372            Some("project://cookie-agent/rel…")
373        );
374        assert_eq!(memory_url_span("Search memory build"), None);
375        assert_eq!(memory_url_span("Fetch xuser://a"), None);
