1//! What a client tells the daemon it shows the user, so the daemon pushes
2//! no notification for it (`docs/site/specs/push-notifications.md` section
3//! 8). Every client decides it the same way.
4
5use std::time::Duration;
6
7use cookie_agent_protocol::SemVer;
8
9/// The first client protocol whose daemons push notifications and take
10/// `client.presence`.
11pub const PUSH_SINCE: SemVer = SemVer::new(1, 30, 0);
12
13/// How long without input before a client says the user is not there.
14pub const IDLE_AFTER: Duration = Duration::from_secs(120);
15
16/// Whether the daemon, speaking `version`, pushes notifications.
17#[must_use]
18pub fn daemon_has_push(version: Option<SemVer>) -> bool {
19    version.is_some_and(|version| version >= PUSH_SINCE)
20}
21
22/// What a client shows the user: the `viewed` session while its window
23/// is `attended` (visible and focused) and was used in the last
24/// [`IDLE_AFTER`]; `None` when the user is not there.
25#[must_use]
26pub fn shown<T>(viewed: Option<T>, attended: bool, idle: Duration) -> Option<T> {
27    viewed.filter(|_| attended && idle < IDLE_AFTER)
28}
29
30#[cfg(test)]
31mod tests {
32    use super::*;
33
34    #[test]
35    fn a_session_is_shown_only_to_an_attended_recently_used_client() {
36        assert_eq!(shown(Some(1), true, Duration::ZERO), Some(1));
37        assert_eq!(shown(Some(1), false, Duration::ZERO), None);
38        assert_eq!(shown(Some(1), true, IDLE_AFTER), None);
39        assert_eq!(shown(None::<u8>, true, Duration::ZERO), None);
40    }
41
42    #[test]
43    fn push_needs_protocol_1_30() {
44        assert!(daemon_has_push(Some(SemVer::new(1, 30, 0))));
45        assert!(!daemon_has_push(Some(SemVer::new(1, 29, 0))));
46        assert!(!daemon_has_push(None));
47    }
48}
