25/// The one-line summary of a producer message: its description when it has
26/// one, else who produced it and how it is delivered.
27pub fn producer_summary(
28    owner: &ProducerOwner,
29    mode: ProducerDeliveryMode,
30    description: Option<&str>,
31) -> String {
32    if let Some(description) = description.filter(|text| !text.trim().is_empty()) {
33        return description.split_whitespace().collect::<Vec<_>>().join(" ");
34    }
35    format!(
36        "{} · {}",
37        producer_owner_label(owner),
38        producer_mode_label(mode)
39    )
40}
41
42/// Who produced a message, as a person reads it.
43pub fn producer_owner_label(owner: &ProducerOwner) -> String {
44    match owner {
45        ProducerOwner::External { source, name } => format!("{source} · {name}"),
46        ProducerOwner::Delegation { invocation_id } => format!("delegation {invocation_id}"),
47        ProducerOwner::Goal { .. } => "goal controller".to_owned(),
48        ProducerOwner::GoalControl { .. } => "goal control".to_owned(),
49        ProducerOwner::Agent { session_id } => format!("agent {session_id}"),
50        ProducerOwner::Tool { tool_call_id } => format!("background call {tool_call_id}"),
51        ProducerOwner::Subagent { session_id } => format!("subagent {session_id}"),
52    }
53}

55pub fn producer_mode_label(mode: ProducerDeliveryMode) -> &'static str {
56    match mode {
57        ProducerDeliveryMode::Steer => "steer",
58        ProducerDeliveryMode::Queue => "queue",
59    }
60}

62pub fn goal_status_label(status: GoalStatus) -> &'static str {

64        GoalStatus::Active => "active",
65        GoalStatus::Paused => "paused",
66        GoalStatus::Completed => "completed",
67        GoalStatus::Cancelled => "cancelled",

71/// Whether the bar above the composer shows a goal in `status`: only while
72/// it is still being pursued. A completed or cancelled goal stays reachable
73/// through the goal details, not the bar.
74pub fn goal_bar_shows(status: GoalStatus) -> bool {
75    matches!(status, GoalStatus::Active | GoalStatus::Paused)

78/// How many lines `text` has, as the collapsed rows count them.
79pub fn display_line_count(text: &str) -> usize {

83/// Whether a session counts as active in a session list: it runs, or it
84/// waits on a producer that will start its next run (an active goal, a
85/// background call, a subagent, an external producer).
86pub fn session_is_active(session: &SessionMeta) -> bool {
87    matches!(
88        crate::activity::session_activity(session),
89        crate::activity::SessionActivity::Running | crate::activity::SessionActivity::Waiting(_)
90    )
91}

93/// Orders a session list: the active sessions first, then the rest, each
94/// group newest activity first; sessions that tie keep their order. It
95/// knows nothing of how a list is grouped, so a client orders each group
96/// (each project's sessions, say) on its own.
97pub fn order_sessions(sessions: &mut [SessionMeta]) {
98    sessions.sort_by_key(|session| {
99        (
100            !session_is_active(session),
101            std::cmp::Reverse(session.last_activity),
102        )
103    });
104}

106/// The agent snapshot the system prompt row shows: the latest run's, else
107/// the one the session was created with; and whether it is the run's.
108pub fn system_prompt_source(state: &SessionState) -> Option<(&AgentSnapshot, bool)> {
109    state
110        .run_snapshot
111        .as_deref()
112        .map(|snapshot| (snapshot, true))
113        .or_else(|| {
114            state
115                .creation_agent
116                .as_deref()
117                .map(|snapshot| (snapshot, false))
118        })
119}

121/// The system prompt row's title, after its icon and chevron:
122/// `system prompt · primary (last run) · next: fast (42 lines)`. The next
123/// agent shows only when the draft's differs from the snapshot's.
124pub fn system_prompt_summary(
125    snapshot: &AgentSnapshot,
126    from_run: bool,
127    draft_agent: Option<&AgentId>,
128) -> String {
129    let when = if from_run { "last run" } else { "at creation" };
130    let next = draft_agent
131        .filter(|draft_agent| **draft_agent != snapshot.agent)
132        .map(|draft_agent| format!(" · next: {draft_agent}"))
133        .unwrap_or_default();
134    format!(
135        "system prompt · {} ({when}){next} ({} lines)",
136        snapshot.agent,
137        display_line_count(&snapshot.composed_prompt)
138    )
139}

141/// The tools row's title, after its icon and chevron: the tools of the
142/// latest model request, or for an ACP agent the cookie tools its MCP
143/// bridge offers beside the agent's own.
144pub fn model_tools_summary(state: &SessionState, tools: &[String]) -> String {
145    if bridged_tools(state) {
146        format!("cookie tools · {} offered to the agent", tools.len())
147    } else {
148        format!("tools · {} available (last request)", tools.len())
149    }
150}

152/// Whether the session's tools row lists what cookie offers an ACP agent
153/// over its MCP bridge: its latest run is of the agent of its ACP session.
154fn bridged_tools(state: &SessionState) -> bool {
155    state.acp.agent.is_some()
156        && state.run_snapshot.as_ref().map(|snapshot| &snapshot.agent) == state.acp.agent.as_ref()
157}

159/// A body cut to the bounds an opened row shows: its lines, control
160/// characters but tabs shown as U+FFFD, and how many lines did not fit.
161#[derive(Clone, Debug, Default, Eq, PartialEq)]
162pub struct BoundedText {
163    pub lines: Vec<String>,
164    pub omitted_lines: usize,
165}
166
167/// `text` within `max_lines` lines and `max_bytes` bytes: whole lines,
168/// then the part of the line the bytes run out in. The lines past the
169/// bounds, and a line cut short, count as omitted.
170pub fn bounded_text(text: &str, max_lines: usize, max_bytes: usize) -> BoundedText {
171    let mut bounded = BoundedText::default();
172    let mut bytes = 0usize;
173    let mut whole = 0usize;
174    let mut total = 0usize;
175    for line in text.split('\n') {
176        total += 1;
177        if bounded.lines.len() >= max_lines || bytes >= max_bytes {
178            continue;
179        }
180        let mut shown = String::with_capacity(line.len().min(max_bytes - bytes));
181        let mut complete = true;
182        for character in line.chars() {
183            let character = if character.is_control() && character != '\t' {
184                '\u{FFFD}'
185            } else {
186                character
187            };
188            if bytes + shown.len() + character.len_utf8() > max_bytes {
189                complete = false;
190                break;
191            }
192            shown.push(character);
193        }
194        if complete {
195            bytes += shown.len();
196            whole += 1;
197            bounded.lines.push(shown);
198        } else {
199            if !shown.is_empty() {
200                bounded.lines.push(shown);
201            }
202            bytes = max_bytes;
203        }
204    }
205    bounded.omitted_lines = total - whole;
206    bounded
207}

209/// The tools the opened tools row lists: as many as fit
210/// [`MAX_EXPANDED_BODY_BYTES`] joined with `, `, as the TUI's line does,
211/// and how many more there are.
212pub fn bounded_tools(tools: &[String]) -> (&[String], usize) {
213    let mut bytes = 0usize;
214    let mut shown = 0usize;
215    for tool in tools {
216        let next = bytes + tool.len() + if shown == 0 { 0 } else { 2 };
217        if next > MAX_EXPANDED_BODY_BYTES {
218            break;
219        }
220        bytes = next;
221        shown += 1;
222    }
223    (&tools[..shown], tools.len() - shown)
224}

233    fn a_producer_summary_prefers_the_description() {

251    #[test]
252    fn bodies_keep_within_their_line_and_byte_bounds() {
253        assert_eq!(
254            bounded_text("a\nb\nc", 2, 100),
255            BoundedText {
256                lines: vec!["a".into(), "b".into()],
257                omitted_lines: 1
258            }
259        );
260        // The bytes run out inside the second line: its start shows, and
261        // it counts as omitted.
262        assert_eq!(
263            bounded_text("abc\ndefg\nh", 10, 5),
264            BoundedText {
265                lines: vec!["abc".into(), "de".into()],
266                omitted_lines: 2
267            }
268        );
269        assert_eq!(
270            bounded_text("tab\there\u{7}", 10, 100).lines,
271            ["tab\there\u{FFFD}"]
272        );
273        assert_eq!(bounded_text("", 1, 1).lines, [""]);
274    }

276    #[test]
277    fn the_tools_row_lists_what_fits_its_bytes() {
278        let tools = vec!["read".to_owned(), "bash".to_owned()];
279        assert_eq!(bounded_tools(&tools), (&tools[..], 0));
280        let many = (0..2000)
281            .map(|index| format!("tool_{index:04}"))
282            .collect::<Vec<_>>();
283        let (shown, omitted) = bounded_tools(&many);
284        // Each name is 9 bytes, then 2 for the separator.
285        assert_eq!(shown.len(), (MAX_EXPANDED_BODY_BYTES + 2) / 11);
286        assert_eq!(shown.len() + omitted, many.len());
287        let mut state = SessionState::default();
288        assert_eq!(
289            model_tools_summary(&state, &tools),
290            "tools · 2 available (last request)"
291        );
292        // An ACP agent's run lists the tools cookie's bridge offers it.
293        let claude = AgentId::new("claude").unwrap();
294        state.acp.agent = Some(claude.clone());
295        state.run_snapshot = Some(Box::new(snapshot_of(claude)));
296        assert_eq!(
297            model_tools_summary(&state, &tools),
298            "cookie tools · 2 offered to the agent"
299        );
300    }
