13/// The most lines an opened row shows of its body (the tools row's list
14/// included).
15pub const MAX_EXPANDED_BODY_LINES: usize = 64;
16
17/// The most bytes an opened row shows of its body.
18pub const MAX_EXPANDED_BODY_BYTES: usize = 8 * 1024;
19
20/// The most lines the opened system prompt row shows.
21pub const MAX_SYSTEM_PROMPT_BODY_LINES: usize = 256;

24pub const MAX_SYSTEM_PROMPT_BODY_BYTES: usize = 32 * 1024;

26/// The one-line summary of a producer message: its description when it has
27/// one, else who produced it and how it is delivered.
28pub fn producer_summary(
29    owner: &ProducerOwner,
30    mode: ProducerDeliveryMode,
31    description: Option<&str>,
32) -> String {
33    if let Some(description) = description.filter(|text| !text.trim().is_empty()) {
34        return description.split_whitespace().collect::<Vec<_>>().join(" ");
35    }
36    format!(
37        "{} · {}",
38        producer_owner_label(owner),
39        producer_mode_label(mode)
40    )
41}
42
43/// Who produced a message, as a person reads it.
44pub fn producer_owner_label(owner: &ProducerOwner) -> String {
45    match owner {
46        ProducerOwner::External { source, name } => format!("{source} · {name}"),
47        ProducerOwner::Delegation { invocation_id } => format!("delegation {invocation_id}"),
48        ProducerOwner::Goal { .. } => "goal controller".to_owned(),
49        ProducerOwner::GoalControl { .. } => "goal control".to_owned(),
50        ProducerOwner::Agent { session_id } => format!("agent {session_id}"),
51        ProducerOwner::Tool { tool_call_id } => format!("background call {tool_call_id}"),
52        ProducerOwner::Subagent { session_id } => format!("subagent {session_id}"),
53    }
54}

56pub fn producer_mode_label(mode: ProducerDeliveryMode) -> &'static str {
57    match mode {
58        ProducerDeliveryMode::Steer => "steer",
59        ProducerDeliveryMode::Queue => "queue",
60    }
61}

63pub fn goal_status_label(status: GoalStatus) -> &'static str {
64    match status {
65        GoalStatus::Active => "active",
66        GoalStatus::Paused => "paused",
67        GoalStatus::Completed => "completed",
68        GoalStatus::Cancelled => "cancelled",
69    }

72/// Whether the bar above the composer shows a goal in `status`: only while
73/// it is still being pursued. A completed or cancelled goal stays reachable
74/// through the goal details, not the bar.
75pub fn goal_bar_shows(status: GoalStatus) -> bool {
76    matches!(status, GoalStatus::Active | GoalStatus::Paused)
77}

79/// How many lines `text` has, as the collapsed rows count them.
80pub fn display_line_count(text: &str) -> usize {
81    text.split('\n').count()
82}

84/// Whether a session counts as active in a session list: for now, while
85/// it runs.
86pub fn session_is_active(session: &SessionMeta) -> bool {
87    session.status == SessionStatus::Running
88}
89
90/// Orders a session list: the active sessions first, then the rest, each
91/// group newest activity first; sessions that tie keep their order. It
92/// knows nothing of how a list is grouped, so a client orders each group
93/// (each project's sessions, say) on its own.
94pub fn order_sessions(sessions: &mut [SessionMeta]) {
95    sessions.sort_by_key(|session| {
96        (
97            !session_is_active(session),
98            std::cmp::Reverse(session.last_activity),
99        )
100    });
101}

103/// The agent snapshot the system prompt row shows: the latest run's, else
104/// the one the session was created with; and whether it is the run's.
105pub fn system_prompt_source(state: &SessionState) -> Option<(&AgentSnapshot, bool)> {
106    state
107        .run_snapshot
108        .as_deref()
109        .map(|snapshot| (snapshot, true))
110        .or_else(|| {
111            state
112                .creation_agent
113                .as_deref()
114                .map(|snapshot| (snapshot, false))
115        })
116}

118/// The system prompt row's title, after its icon and chevron:
119/// `system prompt · primary (last run) · next: fast (42 lines)`. The next
120/// agent shows only when the draft's differs from the snapshot's.
121pub fn system_prompt_summary(
122    snapshot: &AgentSnapshot,
123    from_run: bool,
124    draft_agent: Option<&AgentId>,
125) -> String {
126    let when = if from_run { "last run" } else { "at creation" };
127    let next = draft_agent
128        .filter(|draft_agent| **draft_agent != snapshot.agent)
129        .map(|draft_agent| format!(" · next: {draft_agent}"))
130        .unwrap_or_default();
131    format!(
132        "system prompt · {} ({when}){next} ({} lines)",
133        snapshot.agent,
134        display_line_count(&snapshot.composed_prompt)
135    )
136}

143/// A body cut to the bounds an opened row shows: its lines, control
144/// characters but tabs shown as U+FFFD, and how many lines did not fit.
145#[derive(Clone, Debug, Default, Eq, PartialEq)]
146pub struct BoundedText {
147    pub lines: Vec<String>,
148    pub omitted_lines: usize,
149}
150
151/// `text` within `max_lines` lines and `max_bytes` bytes: whole lines,
152/// then the part of the line the bytes run out in. The lines past the
153/// bounds, and a line cut short, count as omitted.
154pub fn bounded_text(text: &str, max_lines: usize, max_bytes: usize) -> BoundedText {
155    let mut bounded = BoundedText::default();
156    let mut bytes = 0usize;
157    let mut whole = 0usize;
158    let mut total = 0usize;
159    for line in text.split('\n') {
160        total += 1;
161        if bounded.lines.len() >= max_lines || bytes >= max_bytes {
162            continue;
163        }
164        let mut shown = String::with_capacity(line.len().min(max_bytes - bytes));
165        let mut complete = true;
166        for character in line.chars() {
167            let character = if character.is_control() && character != '\t' {
168                '\u{FFFD}'
169            } else {
170                character
171            };
172            if bytes + shown.len() + character.len_utf8() > max_bytes {
173                complete = false;
174                break;
175            }
176            shown.push(character);
177        }
178        if complete {
179            bytes += shown.len();
180            whole += 1;
181            bounded.lines.push(shown);
182        } else {
183            if !shown.is_empty() {
184                bounded.lines.push(shown);
185            }
186            bytes = max_bytes;
187        }
188    }
189    bounded.omitted_lines = total - whole;
190    bounded
191}

193/// The tools the opened tools row lists: as many as fit
194/// [`MAX_EXPANDED_BODY_BYTES`] joined with `, `, as the TUI's line does,
195/// and how many more there are.
196pub fn bounded_tools(tools: &[String]) -> (&[String], usize) {
197    let mut bytes = 0usize;
198    let mut shown = 0usize;
199    for tool in tools {
200        let next = bytes + tool.len() + if shown == 0 { 0 } else { 2 };
201        if next > MAX_EXPANDED_BODY_BYTES {
202            break;
203        }
204        bytes = next;
205        shown += 1;
206    }
207    (&tools[..shown], tools.len() - shown)
208}

216    #[test]
217    fn a_producer_summary_prefers_the_description() {
218        let owner = ProducerOwner::GoalControl {
219            goal_id: cookie_agent_protocol::GoalId::new_v7(),
220        };
221        assert_eq!(
222            producer_summary(
223                &owner,
224                ProducerDeliveryMode::Queue,
225                Some("  build\n failed ")
226            ),
227            "build failed"
228        );
229        assert_eq!(
230            producer_summary(&owner, ProducerDeliveryMode::Steer, Some(" ")),
231            "goal control · steer"
232        );
233    }

235    #[test]
236    fn bodies_keep_within_their_line_and_byte_bounds() {
237        assert_eq!(
238            bounded_text("a\nb\nc", 2, 100),
239            BoundedText {
240                lines: vec!["a".into(), "b".into()],
241                omitted_lines: 1
242            }
243        );
244        // The bytes run out inside the second line: its start shows, and
245        // it counts as omitted.
246        assert_eq!(
247            bounded_text("abc\ndefg\nh", 10, 5),
248            BoundedText {
249                lines: vec!["abc".into(), "de".into()],
250                omitted_lines: 2
251            }
252        );
253        assert_eq!(
254            bounded_text("tab\there\u{7}", 10, 100).lines,
255            ["tab\there\u{FFFD}"]
256        );
257        assert_eq!(bounded_text("", 1, 1).lines, [""]);
258    }

260    #[test]
261    fn the_tools_row_lists_what_fits_its_bytes() {
262        let tools = vec!["read".to_owned(), "bash".to_owned()];
263        assert_eq!(bounded_tools(&tools), (&tools[..], 0));
264        let many = (0..2000)
265            .map(|index| format!("tool_{index:04}"))
266            .collect::<Vec<_>>();
267        let (shown, omitted) = bounded_tools(&many);
268        // Each name is 9 bytes, then 2 for the separator.
269        assert_eq!(shown.len(), (MAX_EXPANDED_BODY_BYTES + 2) / 11);
270        assert_eq!(shown.len() + omitted, many.len());
271        assert_eq!(
272            model_tools_summary(&tools),
273            "tools · 2 available (last request)"
274        );
275    }
