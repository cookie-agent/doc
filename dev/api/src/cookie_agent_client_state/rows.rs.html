25/// The one-line summary of a producer message: its description when it has
26/// one, else who produced it and how it is delivered.
27pub fn producer_summary(
28    owner: &ProducerOwner,
29    mode: ProducerDeliveryMode,
30    description: Option<&str>,
31) -> String {
32    if let Some(description) = description.filter(|text| !text.trim().is_empty()) {
33        return description.split_whitespace().collect::<Vec<_>>().join(" ");
34    }
35    format!(
36        "{} · {}",
37        producer_owner_label(owner),
38        producer_mode_label(mode)
39    )
40}

42/// Who produced a message, as a person reads it.
43pub fn producer_owner_label(owner: &ProducerOwner) -> String {
44    match owner {
45        ProducerOwner::External { source, name } => format!("{source} · {name}"),
46        ProducerOwner::Delegation { invocation_id } => format!("delegation {invocation_id}"),
47        ProducerOwner::Goal { .. } => "goal controller".to_owned(),
48        ProducerOwner::GoalControl { .. } => "goal control".to_owned(),
49        ProducerOwner::Agent { session_id } => format!("agent {session_id}"),
50        ProducerOwner::Tool { tool_call_id } => format!("background call {tool_call_id}"),
51        ProducerOwner::Subagent { session_id } => format!("subagent {session_id}"),
52    }
53}

55pub fn producer_mode_label(mode: ProducerDeliveryMode) -> &'static str {
56    match mode {
57        ProducerDeliveryMode::Steer => "steer",
58        ProducerDeliveryMode::Queue => "queue",
59    }

83/// Whether a session counts as active in a session list: it runs, it is
84/// held and waits for the user to resume it, or it waits on a producer that
85/// will start its next run (an active goal, a background call, a subagent,
86/// an external producer).
87pub fn session_is_active(session: &SessionMeta) -> bool {
88    matches!(
89        crate::activity::session_activity(session),
90        crate::activity::SessionActivity::Running
91            | crate::activity::SessionActivity::Held(_)
92            | crate::activity::SessionActivity::Waiting(_)
93    )
94}

96/// Orders a session list: the active sessions first, then the rest, each
97/// group newest [`ordering_activity`] first; sessions that tie keep their
98/// order. It knows nothing of how a list is grouped, so a client orders each
99/// group (each project's sessions, say) on its own.
100pub fn order_sessions(sessions: &mut [SessionMeta]) {
101    sessions.sort_by_key(|session| {
102        (
103            !session_is_active(session),
104            std::cmp::Reverse(ordering_activity(session)),
105        )
106    });
107}
108
109/// When a session last had activity that orders it: the user's last input
110/// or delegation (`last_agent_activity`), which its own output and tool
111/// results never move, so a busy session keeps its place while it works;
112/// else, from an older daemon or before any such activity, its last event.
113pub fn ordering_activity(session: &SessionMeta) -> jiff::Timestamp {
114    session.last_agent_activity.unwrap_or(session.last_activity)
115}
116
117/// Orders a session tree's subagents as [`order_sessions`] orders a list,
118/// under each session at every depth: the active ones first (running, held,
119/// or waiting on a producer), then the rest, each group newest
120/// [`ordering_activity`] first. The root keeps its place.
121pub fn order_tree(tree: &mut cookie_agent_protocol::SessionTree) {
122    tree.children.sort_by_key(|child| {
123        (
124            !session_is_active(&child.session),
125            std::cmp::Reverse(ordering_activity(&child.session)),
126        )
127    });
128    for child in &mut tree.children {
129        order_tree(child);
130    }
131}

133/// The agent snapshot the system prompt row shows: the latest run's, else
134/// the one the session was created with; and whether it is the run's.
135pub fn system_prompt_source(state: &SessionState) -> Option<(&AgentSnapshot, bool)> {
136    state
137        .run_snapshot
138        .as_deref()
139        .map(|snapshot| (snapshot, true))
140        .or_else(|| {
141            state
142                .creation_agent
143                .as_deref()
144                .map(|snapshot| (snapshot, false))
145        })
146}

148/// The system prompt row's title, after its icon and chevron:
149/// `system prompt · primary (last run) · next: fast (42 lines)`. The next
150/// agent shows only when the draft's differs from the snapshot's.
151pub fn system_prompt_summary(
152    snapshot: &AgentSnapshot,
153    from_run: bool,
154    draft_agent: Option<&AgentId>,
155) -> String {
156    let when = if from_run { "last run" } else { "at creation" };
157    let next = draft_agent
158        .filter(|draft_agent| **draft_agent != snapshot.agent)
159        .map(|draft_agent| format!(" · next: {draft_agent}"))
160        .unwrap_or_default();
161    format!(
162        "system prompt · {} ({when}){next} ({} lines)",
163        snapshot.agent,
164        display_line_count(&snapshot.composed_prompt)
165    )
166}

168/// The tools row's title, after its icon and chevron: the tools of the
169/// latest model request, or for an ACP agent the cookie tools its MCP
170/// bridge offers beside the agent's own.
171pub fn model_tools_summary(state: &SessionState, tools: &[String]) -> String {
172    if bridged_tools(state) {
173        format!("cookie tools · {} offered to the agent", tools.len())
174    } else {
175        format!("tools · {} available (last request)", tools.len())
176    }
177}

179/// Whether the session's tools row lists what cookie offers an ACP agent
180/// over its MCP bridge: its latest run is of the agent of its ACP session.
181fn bridged_tools(state: &SessionState) -> bool {
182    state.acp.agent.is_some()
183        && state.run_snapshot.as_ref().map(|snapshot| &snapshot.agent) == state.acp.agent.as_ref()
184}

186/// A body cut to the bounds an opened row shows: its lines, control
187/// characters but tabs shown as U+FFFD, and how many lines did not fit.
188#[derive(Clone, Debug, Default, Eq, PartialEq)]
189pub struct BoundedText {
190    pub lines: Vec<String>,
191    pub omitted_lines: usize,
192}
193
194/// `text` within `max_lines` lines and `max_bytes` bytes: whole lines,
195/// then the part of the line the bytes run out in. The lines past the
196/// bounds, and a line cut short, count as omitted.
197pub fn bounded_text(text: &str, max_lines: usize, max_bytes: usize) -> BoundedText {
198    let mut bounded = BoundedText::default();
199    let mut bytes = 0usize;
200    let mut whole = 0usize;
201    let mut total = 0usize;
202    for line in text.split('\n') {
203        total += 1;
204        if bounded.lines.len() >= max_lines || bytes >= max_bytes {
205            continue;
206        }
207        let mut shown = String::with_capacity(line.len().min(max_bytes - bytes));
208        let mut complete = true;
209        for character in line.chars() {
210            let character = if character.is_control() && character != '\t' {
211                '\u{FFFD}'
212            } else {
213                character
214            };
215            if bytes + shown.len() + character.len_utf8() > max_bytes {
216                complete = false;
217                break;
218            }
219            shown.push(character);
220        }
221        if complete {
222            bytes += shown.len();
223            whole += 1;
224            bounded.lines.push(shown);
225        } else {
226            if !shown.is_empty() {
227                bounded.lines.push(shown);
228            }
229            bytes = max_bytes;
230        }
231    }
232    bounded.omitted_lines = total - whole;
233    bounded
234}

236/// The tools the opened tools row lists: as many as fit
237/// [`MAX_EXPANDED_BODY_BYTES`] joined with `, `, as the TUI's line does,
238/// and how many more there are.
239pub fn bounded_tools(tools: &[String]) -> (&[String], usize) {
240    let mut bytes = 0usize;
241    let mut shown = 0usize;
242    for tool in tools {
243        let next = bytes + tool.len() + if shown == 0 { 0 } else { 2 };
244        if next > MAX_EXPANDED_BODY_BYTES {
245            break;
246        }
247        bytes = next;
248        shown += 1;
249    }
250    (&tools[..shown], tools.len() - shown)
251}

278    #[test]
279    fn bodies_keep_within_their_line_and_byte_bounds() {
280        assert_eq!(
281            bounded_text("a\nb\nc", 2, 100),
282            BoundedText {
283                lines: vec!["a".into(), "b".into()],
284                omitted_lines: 1
285            }
286        );
287        // The bytes run out inside the second line: its start shows, and
288        // it counts as omitted.
289        assert_eq!(
290            bounded_text("abc\ndefg\nh", 10, 5),
291            BoundedText {
292                lines: vec!["abc".into(), "de".into()],
293                omitted_lines: 2
294            }
295        );
296        assert_eq!(
297            bounded_text("tab\there\u{7}", 10, 100).lines,
298            ["tab\there\u{FFFD}"]
299        );
300        assert_eq!(bounded_text("", 1, 1).lines, [""]);
301    }

303    #[test]
304    fn the_tools_row_lists_what_fits_its_bytes() {
305        let tools = vec!["read".to_owned(), "bash".to_owned()];
306        assert_eq!(bounded_tools(&tools), (&tools[..], 0));
307        let many = (0..2000)
308            .map(|index| format!("tool_{index:04}"))
309            .collect::<Vec<_>>();
310        let (shown, omitted) = bounded_tools(&many);
311        // Each name is 9 bytes, then 2 for the separator.
312        assert_eq!(shown.len(), (MAX_EXPANDED_BODY_BYTES + 2) / 11);
313        assert_eq!(shown.len() + omitted, many.len());
314        let mut state = SessionState::default();
315        assert_eq!(
316            model_tools_summary(&state, &tools),
317            "tools · 2 available (last request)"
318        );
319        // An ACP agent's run lists the tools cookie's bridge offers it.
320        let claude = AgentId::new("claude").unwrap();
321        state.acp.agent = Some(claude.clone());
322        state.run_snapshot = Some(Box::new(snapshot_of(claude)));
323        assert_eq!(
324            model_tools_summary(&state, &tools),
325            "cookie tools · 2 offered to the agent"
326        );

388    #[test]
389    fn a_busy_session_keeps_its_place_by_the_users_last_activity() {
390        // Both run; `busy` produced output since, but the user wrote to
391        // `asked` last, so `asked` stays first.
392        let mut busy = session("busy", "running", "2026-10-20T12:00:00Z");
393        busy.last_agent_activity = Some("2026-10-20T09:00:00Z".parse().unwrap());
394        let mut asked = session("asked", "running", "2026-10-20T11:00:00Z");
395        asked.last_agent_activity = Some("2026-10-20T10:00:00Z".parse().unwrap());
396        // Without the user's activity, the last event orders it.
397        let unknown = session("unknown", "running", "2026-10-20T09:30:00Z");
398        let mut sessions = vec![busy, unknown, asked];
399        order_sessions(&mut sessions);
400        let names = sessions
401            .iter()
402            .map(|session| session.short_id.as_deref().unwrap_or_default())
403            .collect::<Vec<_>>();
404        assert_eq!(names, ["asked", "unknown", "busy"]);
405    }

409        let node = |meta: SessionMeta, children: Vec<cookie_agent_protocol::SessionTree>| {

453    fn active_sessions_come_first_then_the_rest_newest_first() {
454        let mut sessions = vec![
455            session("old-idle", "idle", "2026-10-20T08:00:00Z"),
456            session("old-running", "running", "2026-10-20T07:00:00Z"),
457            session("new-failed", "failed", "2026-10-20T12:00:00Z"),
458            session("tie-a", "completed", "2026-10-20T10:00:00Z"),

460            session("tie-b", "cancelled", "2026-10-20T10:00:00Z"),
461            {
462                // Waiting on a producer counts as active, under the newer
463                // running sessions but above every idle one.
464                let mut waiting = session("old-waiting", "idle", "2026-10-20T06:00:00Z");
465                waiting.waiting_on = vec![cookie_agent_protocol::WaitingOn::Goal];

469        order_sessions(&mut sessions);
470        let names = sessions
471            .iter()
472            .map(|session| session.short_id.as_deref().unwrap_or_default())
473            .collect::<Vec<_>>();
474        assert_eq!(
475            names,
476            [
477                "new-running",
478                "old-running",
479                "old-waiting",
480                "new-failed",
481                "tie-a",
482                "tie-b",
483                "old-idle"
