25/// The one-line summary of a producer message: its description when it has
26/// one, else who produced it and how it is delivered.
27pub fn producer_summary(
28    owner: &ProducerOwner,
29    mode: ProducerDeliveryMode,
30    description: Option<&str>,
31) -> String {
32    if let Some(description) = description.filter(|text| !text.trim().is_empty()) {
33        return description.split_whitespace().collect::<Vec<_>>().join(" ");
34    }
35    format!(
36        "{} · {}",
37        producer_owner_label(owner),
38        producer_mode_label(mode)
39    )
40}

42/// The one-line summary of a message between an ACP agent's sessions:
43/// which way it went, and the other session.
44pub fn agent_session_message_summary(message: &cookie_agent_protocol::AcpSessionMessage) -> String {
45    let peer = message
46        .peer
47        .as_deref()
48        .map(|peer| peer.split_whitespace().collect::<Vec<_>>().join(" "))
49        .filter(|peer| !peer.is_empty());
50    let received = message.direction != cookie_agent_protocol::AcpMessageDirection::Sent;
51    match (received, peer.as_deref()) {
52        (true, Some("parent")) => "message from the parent session".to_owned(),
53        (false, Some("parent")) => "message to the parent session".to_owned(),
54        (true, Some(peer)) => format!("message from subagent {peer}"),
55        (false, Some(peer)) => format!("message to subagent {peer}"),
56        (true, None) => "message from another of the agent's sessions".to_owned(),
57        (false, None) => "message to another of the agent's sessions".to_owned(),
58    }
59}

61/// The one-line summary of an ACP agent's command that ended after running
62/// in the background, as a background call's report reads: what it ran,
63/// else its tool.
64pub fn agent_background_ended_summary(ended: &cookie_agent_protocol::AcpBackgroundEnded) -> String {
65    let what = ended
66        .display
67        .as_deref()
68        .map(|display| display.split_whitespace().collect::<Vec<_>>().join(" "))
69        .filter(|display| !display.is_empty())
70        .unwrap_or_else(|| ended.tool.clone());
71    format!("Background: {what} ended")
72}

74/// What an opened row of an ACP agent's ended background command says
75/// before the command itself: its tool, how long it ran until `ended_at`,
76/// and how it exited when the agent said so.
77pub fn agent_background_ended_details(
78    ended: &cookie_agent_protocol::AcpBackgroundEnded,
79    ended_at: jiff::Timestamp,
80) -> String {
81    let ran = ended_at
82        .duration_since(ended.started_at)
83        .unsigned_abs()
84        .as_secs();
85    let ran = match ran {
86        0..60 => format!("{ran}s"),
87        60..3600 => format!("{}m {}s", ran / 60, ran % 60),
88        _ => format!("{}h {}m", ran / 3600, ran % 3600 / 60),
89    };
90    let exit = match (&ended.signal, ended.exit_code) {
91        (Some(signal), _) => format!("ended by {signal}"),
92        (None, Some(code)) => format!("exit code {code}"),
93        (None, None) => "no exit code reported".to_owned(),
94    };
95    format!(
96        "the agent's {} call · ran in the background for {ran} · {exit}",
97        ended.tool
98    )
99}

104/// Who produced a message, as a person reads it.
105pub fn producer_owner_label(owner: &ProducerOwner) -> String {
106    match owner {
107        ProducerOwner::External { source, name } => format!("{source} · {name}"),
108        ProducerOwner::Delegation { invocation_id } => format!("delegation {invocation_id}"),
109        ProducerOwner::Goal { .. } => "goal controller".to_owned(),
110        ProducerOwner::GoalControl { .. } => "goal control".to_owned(),
111        ProducerOwner::Agent { session_id } => format!("agent {session_id}"),
112        ProducerOwner::Tool { tool_call_id } => format!("background call {tool_call_id}"),
113        ProducerOwner::Subagent { session_id } => format!("subagent {session_id}"),
114        ProducerOwner::AcpAgent { agent } => format!("{agent}, on its own"),
115    }
116}

146/// Whether a session counts as active in a session list: it runs, it is
147/// held and waits for the user to resume it, or it waits on a producer that
148/// will start its next run (an active goal, a background call, a subagent,
149/// an external producer).
150pub fn session_is_active(session: &SessionMeta) -> bool {
151    matches!(
152        crate::activity::session_activity(session),
153        crate::activity::SessionActivity::Running
154            | crate::activity::SessionActivity::Held(_)
155            | crate::activity::SessionActivity::Waiting(_)
156    )
157}

159/// Orders a session list: the active sessions first, then the rest, each
160/// group newest [`ordering_activity`] first; sessions that tie keep their
161/// order. It knows nothing of how a list is grouped, so a client orders each
162/// group (each project's sessions, say) on its own.
163pub fn order_sessions(sessions: &mut [SessionMeta]) {
164    sessions.sort_by_key(|session| {
165        (
166            !session_is_active(session),
167            std::cmp::Reverse(ordering_activity(session)),
168        )
169    });
170}
171
172/// When a session last had activity that orders it: the user's last input
173/// or delegation (`last_agent_activity`), which its own output and tool
174/// results never move, so a busy session keeps its place while it works;
175/// else, from an older daemon or before any such activity, its last event.
176pub fn ordering_activity(session: &SessionMeta) -> jiff::Timestamp {
177    session.last_agent_activity.unwrap_or(session.last_activity)
178}
179
180/// Orders a session tree's subagents as [`order_sessions`] orders a list,
181/// under each session at every depth: the active ones first (running, held,
182/// or waiting on a producer), then the rest, each group newest
183/// [`ordering_activity`] first. The root keeps its place.
184pub fn order_tree(tree: &mut cookie_agent_protocol::SessionTree) {
185    tree.children.sort_by_key(|child| {
186        (
187            !session_is_active(&child.session),
188            std::cmp::Reverse(ordering_activity(&child.session)),
189        )
190    });
191    for child in &mut tree.children {
192        order_tree(child);
193    }
194}

219/// The system prompt row's title, after its icon and chevron:
220/// `system prompt · primary (last run) · next: fast (42 lines)`. The next
221/// agent shows only when the draft's differs from the snapshot's.
222pub fn system_prompt_summary(
223    snapshot: &AgentSnapshot,
224    from_run: bool,
225    draft_agent: Option<&AgentId>,
226) -> String {
227    let when = if from_run { "last run" } else { "at creation" };
228    let next = draft_agent
229        .filter(|draft_agent| **draft_agent != snapshot.agent)
230        .map(|draft_agent| format!(" · next: {draft_agent}"))
231        .unwrap_or_default();
232    format!(
233        "system prompt · {} ({when}){next} ({} lines)",
234        snapshot.agent,
235        display_line_count(&snapshot.composed_prompt)
236    )
237}

239/// The tools row's title, after its icon and chevron: the tools of the
240/// latest model request, or for an ACP agent the cookie tools its MCP
241/// bridge offers beside the agent's own.
242pub fn model_tools_summary(state: &SessionState, tools: &[String]) -> String {
243    if bridged_tools(state) {
244        format!("cookie tools · {} offered to the agent", tools.len())
245    } else if state.model_deferred_tools.is_empty() {
246        format!("tools · {} available (last request)", tools.len())
247    } else {
248        let (provided, searchable) = model_tool_groups(state, tools);
249        format!(
250            "tools · {} provided, {} searchable (last request)",
251            provided.len(),
252            searchable.len()
253        )
254    }
255}

263/// Each of `tools`' group slot, by the order its group first appears among
264/// them, so the clients colour a group's tools alike; a tool of no known
265/// group shares the slot of the others like it.
266pub fn tool_group_slots(state: &SessionState, tools: &[&str]) -> Vec<usize> {
267    let mut groups = Vec::<Option<&str>>::new();
268    tools
269        .iter()
270        .map(|tool| {
271            let group = state.deferred_tool_groups.get(*tool).map(String::as_str);
272            groups
273                .iter()
274                .position(|known| *known == group)
275                .unwrap_or_else(|| {
276                    groups.push(group);
277                    groups.len() - 1
278                })
279        })
280        .collect()
281}

283/// The latest request's tools split in two: the ones it provided (sent up
284/// front, or deferred and loaded) and the ones it left searchable
285/// (deferred, not loaded), each in the request's order.
286pub fn model_tool_groups<'a>(
287    state: &SessionState,
288    tools: &'a [String],
289) -> (Vec<&'a str>, Vec<&'a str>) {
290    let deferred = state
291        .model_deferred_tools
292        .iter()
293        .map(String::as_str)
294        .collect::<std::collections::HashSet<_>>();
295    tools
296        .iter()
297        .map(String::as_str)
298        .partition(|tool| !deferred.contains(tool))
299}

308/// A body cut to the bounds an opened row shows: its lines, control
309/// characters but tabs shown as U+FFFD, and how many lines did not fit.
310#[derive(Clone, Debug, Default, Eq, PartialEq)]
311pub struct BoundedText {
312    pub lines: Vec<String>,
313    pub omitted_lines: usize,
314}
315
316/// `text` within `max_lines` lines and `max_bytes` bytes: whole lines,
317/// then the part of the line the bytes run out in. The lines past the
318/// bounds, and a line cut short, count as omitted.
319pub fn bounded_text(text: &str, max_lines: usize, max_bytes: usize) -> BoundedText {
320    let mut bounded = BoundedText::default();
321    let mut bytes = 0usize;
322    let mut whole = 0usize;
323    let mut total = 0usize;
324    for line in text.split('\n') {
325        total += 1;
326        if bounded.lines.len() >= max_lines || bytes >= max_bytes {
327            continue;
328        }
329        let mut shown = String::with_capacity(line.len().min(max_bytes - bytes));
330        let mut complete = true;
331        for character in line.chars() {
332            let character = if character.is_control() && character != '\t' {
333                '\u{FFFD}'
334            } else {
335                character
336            };
337            if bytes + shown.len() + character.len_utf8() > max_bytes {
338                complete = false;
339                break;
340            }
341            shown.push(character);
342        }
343        if complete {
344            bytes += shown.len();
345            whole += 1;
346            bounded.lines.push(shown);
347        } else {
348            if !shown.is_empty() {
349                bounded.lines.push(shown);
350            }
351            bytes = max_bytes;
352        }
353    }
354    bounded.omitted_lines = total - whole;
355    bounded
356}

358/// The tools the opened tools row lists: as many as fit
359/// [`MAX_EXPANDED_BODY_BYTES`] joined with `, `, as the TUI's line does,
360/// and how many more there are.
361pub fn bounded_tools<T: AsRef<str>>(tools: &[T]) -> (&[T], usize) {
362    let mut bytes = 0usize;
363    let mut shown = 0usize;
364    for tool in tools {
365        let next = bytes + tool.as_ref().len() + if shown == 0 { 0 } else { 2 };
366        if next > MAX_EXPANDED_BODY_BYTES {
367            break;
368        }
369        bytes = next;
370        shown += 1;
371    }
372    (&tools[..shown], tools.len() - shown)
373}
