1//! How hot a session's prompt cache is, for the cache badge both clients
2//! show (`docs/site/specs/cache-aware-idle.md` section 4.5). The daemon
3//! stamps each request's TTL and plans the compaction before expiry; this
4//! only counts down.
5
6use std::time::Duration;
7
8use cookie_agent_protocol::{CacheTtlSource, CacheTtlStamp, PlannedCacheCompaction};
9use jiff::Timestamp;
10
11/// The prompt cache of a session's latest request.
12#[derive(Clone, Copy, Debug, Eq, PartialEq)]
13pub struct CacheClock {
14    /// When the request went out.
15    pub sent_at: Timestamp,
16    pub stamp: CacheTtlStamp,
17}
18
19/// How much of its TTL a cache has left.
20#[derive(Clone, Copy, Debug, Eq, PartialEq)]
21pub enum CacheHeat {
22    /// More than half.
23    Hot,
24    /// A fifth to a half.
25    Warm,
26    /// Less than a fifth.
27    Cooling,
28    /// None: the next request writes the cache again.
29    Expired,
30}
31
32/// What the cache badge shows.
33#[derive(Clone, Debug, Eq, PartialEq)]
34pub enum CacheBadge {
35    /// A compaction before expiry is running.
36    Compacting,
37    /// A run is in progress: the cache is in use, with no countdown.
38    InUse { assumed: bool },
39    /// The cache never expires.
40    Forever { assumed: bool },
41    /// Counting down.
42    Counting {
43        heat: CacheHeat,
44        /// Time left; zero once expired.
45        left: Duration,
46        /// The TTL is assumed or declared, not set on the request.
47        assumed: bool,
48        /// A compaction before expiry is planned.
49        planned: bool,
50    },
51}
52
53impl CacheClock {
54    /// When the cache expires; `None` when it never does.
55    #[must_use]
56    pub fn expires_at(&self) -> Option<Timestamp> {
57        self.stamp.expires_at(self.sent_at)
58    }
59
60    /// The badge at `now`, with the daemon's `plan`, while a run is
61    /// `running` or not.
62    #[must_use]
63    pub fn badge(
64        &self,
65        now: Timestamp,
66        running: bool,
67        plan: Option<&PlannedCacheCompaction>,
68    ) -> CacheBadge {
69        let assumed = self.stamp.source != CacheTtlSource::Set;
70        if plan.is_some_and(|plan| plan.running) {
71            return CacheBadge::Compacting;
72        }
73        if running {
74            return CacheBadge::InUse { assumed };
75        }
76        let Some(expires_at) = self.expires_at() else {
77            return CacheBadge::Forever { assumed };
78        };
79        let left = Duration::try_from(expires_at.duration_since(now)).unwrap_or_default();
80        let ttl = Duration::from_secs(self.stamp.ttl_secs);
81        CacheBadge::Counting {
82            heat: heat(left, ttl),
83            left,
84            assumed,
85            planned: plan.is_some(),
86        }
87    }
88}
89
90/// The heat of a cache with `left` of its `ttl` to go.
91#[must_use]
92pub fn heat(left: Duration, ttl: Duration) -> CacheHeat {
93    if left.is_zero() || ttl.is_zero() {
94        return CacheHeat::Expired;
95    }
96    // Compared in whole milliseconds: `left / ttl` against a half and a
97    // fifth.
98    let left = left.as_millis();
99    let ttl = ttl.as_millis();
100    if left * 2 > ttl {
101        CacheHeat::Hot
102    } else if left * 5 > ttl {
103        CacheHeat::Warm
104    } else {
105        CacheHeat::Cooling
106    }
107}

109/// The countdown's text: minutes, rounded up (`41m`); whole hours above 90
110/// minutes, rounded to the nearest (`11h`); seconds in the last minute
111/// (`45s`).
112#[must_use]
113pub fn countdown(left: Duration) -> String {
114    let seconds = left.as_secs();
115    if seconds < 60 {
116        return format!("{seconds}s");
117    }
118    if seconds > 90 * 60 {
119        return format!("{}h", (seconds + 30 * 60) / 3_600);
120    }
121    format!("{}m", seconds.div_ceil(60))
122}
123
124/// The badge's text: the countdown with `~` before an assumed TTL,
125/// `compacting`, `∞`, or `cache expired`; empty while in use.
126#[must_use]
127pub fn badge_text(badge: &CacheBadge) -> String {
128    match badge {
129        CacheBadge::Compacting => "compacting".into(),
130        CacheBadge::InUse { .. } => String::new(),
131        CacheBadge::Forever { .. } => "∞".into(),
132        CacheBadge::Counting {
133            heat: CacheHeat::Expired,
134            ..
135        } => "cache expired".into(),
136        CacheBadge::Counting { left, assumed, .. } => {
137            format!("{}{}", if *assumed { "~" } else { "" }, countdown(*left))
138        }
139    }
140}

142/// Where a TTL came from, for the tooltip and the usage card.
143#[must_use]
144pub fn source_label(source: CacheTtlSource) -> &'static str {
145    match source {
146        CacheTtlSource::Set => "set on the request",
147        CacheTtlSource::Declared => "declared in the configuration",
148        CacheTtlSource::Assumed => "assumed",
149    }
150}
151
152/// A TTL's text: `1h`, `30m`, `90s`, or `never expires`.
153#[must_use]
154pub fn ttl_label(ttl_secs: u64) -> String {
155    match ttl_secs {
156        0 => "never expires".into(),
157        seconds if seconds % 3_600 == 0 => format!("{}h", seconds / 3_600),
158        seconds if seconds % 60 == 0 => format!("{}m", seconds / 60),
159        seconds => format!("{seconds}s"),
160    }
161}
162
163/// `at` in `zone` as `HH:MM`, the date left out for today's.
164fn clock_time(at: Timestamp, now: Timestamp, zone: &jiff::tz::TimeZone) -> String {
165    let time = zone.to_datetime(at);
166    if time.date() == zone.to_datetime(now).date() {
167        format!("{:02}:{:02}", time.hour(), time.minute())
168    } else {
169        crate::cron::local_minute(at, zone)
170    }
171}

173/// The lines of the badge's tooltip and of the usage panel's cache card:
174/// its state; the request, its TTL and where it came from, and expiry;
175/// and the planned compaction, with the scheduled prompt that decided it.
176#[must_use]
177pub fn detail_lines(
178    clock: &CacheClock,
179    badge: &CacheBadge,
180    plan: Option<&PlannedCacheCompaction>,
181    now: Timestamp,
182    zone: &jiff::tz::TimeZone,
183) -> Vec<String> {
184    let mut lines = vec![match badge {
185        CacheBadge::Compacting => "Compacting before the prompt cache expires".to_owned(),
186        CacheBadge::InUse { .. } => "Prompt cache in use".to_owned(),
187        CacheBadge::Forever { .. } => "Prompt cache warm; it never expires".to_owned(),
188        CacheBadge::Counting {
189            heat: CacheHeat::Expired,
190            ..
191        } => "Prompt cache expired: the next prompt caches the context again".to_owned(),
192        CacheBadge::Counting { heat, left, .. } => format!(
193            "Prompt cache {} · {} left",
194            match heat {
195                CacheHeat::Hot => "hot",
196                CacheHeat::Warm => "warm",
197                _ => "cooling",
198            },
199            countdown(*left)
200        ),
201    }];
202    let mut request = format!(
203        "Last request {} · {} TTL ({})",
204        clock_time(clock.sent_at, now, zone),
205        ttl_label(clock.stamp.ttl_secs),
206        source_label(clock.stamp.source)
207    );
208    if let Some(expires_at) = clock.expires_at() {
209        request.push_str(&format!(" · expires {}", clock_time(expires_at, now, zone)));
210    }
211    lines.push(request);
212    if let Some(plan) = plan.filter(|plan| !plan.running) {
213        lines.push(match plan.next_fire {
214            Some(fire) => format!(
215                "Next scheduled prompt at {}, after the cache expires: compacts at {}",
216                clock_time(fire, now, zone),
217                clock_time(plan.at, now, zone)
218            ),
219            None => format!(
220                "Compacts at {}, before the cache expires",
221                clock_time(plan.at, now, zone)
222            ),
223        });
224    }
225    lines
226}

229mod tests {
230    use super::*;

232    fn clock(ttl_secs: u64, source: CacheTtlSource) -> CacheClock {
233        CacheClock {
234            sent_at: Timestamp::from_second(1_800_000_000).unwrap(),
235            stamp: CacheTtlStamp { ttl_secs, source },
236        }
237    }
238
239    fn after(clock: &CacheClock, seconds: i64) -> Timestamp {
240        clock
241            .sent_at
242            .checked_add(jiff::SignedDuration::from_secs(seconds))
243            .unwrap()
244    }

246    #[test]
247    fn heat_changes_at_a_half_and_a_fifth_of_the_ttl_and_at_expiry() {
248        let ttl = Duration::from_secs(1_000);
249        assert_eq!(heat(Duration::from_secs(501), ttl), CacheHeat::Hot);
250        assert_eq!(heat(Duration::from_secs(500), ttl), CacheHeat::Warm);
251        assert_eq!(heat(Duration::from_secs(201), ttl), CacheHeat::Warm);
252        assert_eq!(heat(Duration::from_secs(200), ttl), CacheHeat::Cooling);
253        assert_eq!(heat(Duration::from_millis(1), ttl), CacheHeat::Cooling);
254        assert_eq!(heat(Duration::ZERO, ttl), CacheHeat::Expired);
255    }
256
257    #[test]
258    fn the_countdown_counts_minutes_hours_and_its_last_seconds() {
259        assert_eq!(countdown(Duration::from_secs(45)), "45s");
260        assert_eq!(countdown(Duration::from_secs(60)), "1m");
261        assert_eq!(countdown(Duration::from_secs(40 * 60 + 1)), "41m");
262        assert_eq!(countdown(Duration::from_secs(90 * 60)), "90m");
263        assert_eq!(countdown(Duration::from_secs(90 * 60 + 1)), "2h");
264        assert_eq!(countdown(Duration::from_secs(11 * 3_600 + 20 * 60)), "11h");
265    }

267    #[test]
268    fn the_badge_follows_the_run_the_plan_and_the_clock() {
269        let set = clock(3_600, CacheTtlSource::Set);
270        let plan = PlannedCacheCompaction {
271            at: after(&set, 3_300),
272            expires_at: after(&set, 3_600),
273            next_fire: None,
274            job_id: None,
275            running: false,
276        };
277        let counting = set.badge(after(&set, 1_140), false, Some(&plan));
278        assert_eq!(
279            counting,
280            CacheBadge::Counting {
281                heat: CacheHeat::Hot,
282                left: Duration::from_secs(2_460),
283                assumed: false,
284                planned: true,
285            }
286        );
287        assert_eq!(badge_text(&counting), "41m");
288        assert_eq!(
289            set.badge(after(&set, 10), true, None),
290            CacheBadge::InUse { assumed: false }
291        );
292        let running = PlannedCacheCompaction {
293            running: true,
294            ..plan
295        };
296        assert_eq!(
297            set.badge(after(&set, 3_400), false, Some(&running)),
298            CacheBadge::Compacting
299        );
300        assert!(matches!(
301            set.badge(after(&set, 3_601), false, None),
302            CacheBadge::Counting {
303                heat: CacheHeat::Expired,
304                ..
305            }
306        ));
307        let assumed = clock(1_800, CacheTtlSource::Assumed);
308        assert!(matches!(
309            assumed.badge(after(&assumed, 360), false, None),
310            CacheBadge::Counting { assumed: true, .. }
311        ));
312        let forever = clock(0, CacheTtlSource::Declared);
313        assert_eq!(
314            forever.badge(after(&forever, 99_999), false, None),
315            CacheBadge::Forever { assumed: true }
316        );
317    }

319    #[test]
320    fn the_details_name_the_plan_only_while_it_waits() {
321        let set = clock(3_600, CacheTtlSource::Set);
322        let now = after(&set, 60);
323        let zone = jiff::tz::TimeZone::UTC;
324        let plan = PlannedCacheCompaction {
325            at: after(&set, 3_300),
326            expires_at: after(&set, 3_600),
327            next_fire: Some(after(&set, 7_200)),
328            job_id: None,
329            running: false,
330        };
331        let badge = set.badge(now, false, Some(&plan));
332        assert_eq!(detail_lines(&set, &badge, Some(&plan), now, &zone).len(), 3);
333        assert_eq!(detail_lines(&set, &badge, None, now, &zone).len(), 2);
334        let running = PlannedCacheCompaction {
335            running: true,
336            ..plan
337        };
338        let compacting = set.badge(now, false, Some(&running));
339        assert_eq!(
340            detail_lines(&set, &compacting, Some(&running), now, &zone).len(),
341            2
342        );
343    }
