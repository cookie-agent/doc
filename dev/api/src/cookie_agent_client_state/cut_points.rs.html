1//! Where a client's revert and fork cut a session's history: before a
2//! user message, at the end of a reply, and before a compaction. The
3//! daemon cuts at any sequence (`session.revert`, `session.fork`); these
4//! are the points the transcript offers.
5//!
6//! A cut before a user message keeps the reply before it: the daemon
7//! leaves out the run the message starts along with it. So reverting or
8//! forking at a user message keeps what a cut at the end of the reply
9//! before it keeps, apart from the settings changed in between, and the
10//! message's text goes back to a composer.
11
12use cookie_agent_protocol::ContextCheckpointCommit;
13
14use crate::{SessionState, TranscriptItem};
15
16/// A point of history a client's menu reverts or forks at.
17#[derive(Clone, Debug, PartialEq, Eq)]
18pub enum CutPoint {
19    /// A past user message: a revert or fork cuts to just before it, and
20    /// its text goes back to a composer (for a fork, the fork's).
21    UserMessage {
22        /// The sequence of its `user_input_submitted`.
23        seq: u64,
24        text: String,
25    },
26    /// The end of a reply: a revert or fork keeps it and drops what came
27    /// after ([`SessionState::reply_cut`]). `at_tip` when nothing came
28    /// after, so there is nothing to revert.
29    ReplyEnd { through_seq: u64, at_tip: bool },
30    /// A compaction: a revert through `through_seq`
31    /// ([`before_compaction`]) drops it.
32    Compaction { through_seq: u64 },
33}
34
35impl CutPoint {
36    /// The sequence a revert or fork at the point cuts through.
37    #[must_use]
38    pub fn through_seq(&self) -> u64 {
39        match self {
40            Self::UserMessage { seq, .. } => before_user_message(*seq),
41            Self::ReplyEnd { through_seq, .. } => *through_seq,
42            Self::Compaction { through_seq } => *through_seq,
43        }
44    }
45
46    /// The user message's text, which a revert or fork gives back.
47    #[must_use]
48    pub fn text(&self) -> Option<&str> {
49        match self {
50            Self::UserMessage { text, .. } => Some(text),
51            Self::ReplyEnd { .. } | Self::Compaction { .. } => None,
52        }
53    }
54
55    /// Whether a revert at the point would remove anything.
56    #[must_use]
57    pub fn revertible(&self) -> bool {
58        !matches!(self, Self::ReplyEnd { at_tip: true, .. })
59    }
60
61    /// Whether the menu offers a fork at the point: a compaction's offers
62    /// only the revert.
63    #[must_use]
64    pub fn forkable(&self) -> bool {
65        !matches!(self, Self::Compaction { .. })
66    }
67}
68
69/// The sequence a revert or fork to just before the user message whose
70/// `user_input_submitted` is `seq` cuts through. A user message always
71/// follows `session_created` (sequence 1).
72#[must_use]
73pub fn before_user_message(seq: u64) -> u64 {
74    seq.saturating_sub(1).max(1)
75}
76
77/// The sequence a revert to before the compaction `commit`, committed at
78/// `seq`, cuts through: the end of the input it read, so the summary goes
79/// with what made it, and the context it replaced comes back.
80#[must_use]
81pub fn before_compaction(seq: u64, commit: &ContextCheckpointCommit) -> u64 {
82    commit
83        .boundaries
84        .input_through_seq
85        .min(seq.saturating_sub(1))
86        .max(1)
87}
88
89impl SessionState {
90    /// The [`CutPoint::Compaction`] of the compaction row committed at
91    /// `seq`.
92    #[must_use]
93    pub fn compaction_point(&self, seq: u64) -> Option<CutPoint> {
94        self.transcript.iter().rev().find_map(|item| match item {
95            TranscriptItem::Compaction {
96                seq: at, commit, ..
97            } if *at == seq => Some(CutPoint::Compaction {
98                through_seq: before_compaction(seq, commit),
99            }),
100            _ => None,
101        })
102    }
103
104    /// The text of the reply that ends with the assistant block `item_id`:
105    /// the markdown of every text part of its run's blocks, in transcript
106    /// order, a blank line between them. `None` when it has none.
107    #[must_use]
108    pub fn reply_text(&self, item_id: u64) -> Option<String> {
109        let run = self.assistant_runs.get(&item_id);
110        let texts = self
111            .transcript
112            .iter()
113            .filter(|item| match item {
114                TranscriptItem::Assistant { id, .. } => {
115                    *id == item_id || (run.is_some() && self.assistant_runs.get(id) == run)
116                }
117                _ => false,
118            })
119            .flat_map(|item| match item {
120                TranscriptItem::Assistant { children, .. } => children.as_slice(),
121                _ => &[],
122            })
123            .filter_map(|child| match child {
124                crate::AssistantChild::Text { markdown, .. } => {
125                    Some(markdown.as_str().trim().to_owned())
126                }
127                _ => None,
128            })
129            .filter(|text| !text.is_empty())
130            .collect::<Vec<_>>();
131        (!texts.is_empty()).then(|| texts.join("\n\n"))
132    }
133
134    /// The [`CutPoint::ReplyEnd`] of the reply that ends with the
135    /// assistant block `item_id`, by [`Self::reply_cut`].
136    #[must_use]
137    pub fn reply_point(&self, item_id: u64) -> Option<CutPoint> {
138        self.reply_cut(item_id)
139            .map(|through_seq| CutPoint::ReplyEnd {
140                through_seq,
141                at_tip: through_seq >= self.last_seq,
142            })
143    }
144
145    /// The last block of the finished reply the assistant block `item_id`
146    /// is part of, which [`Self::reply_point`] takes. `None` while its run
147    /// goes on.
148    #[must_use]
149    pub fn reply_end(&self, item_id: u64) -> Option<u64> {
150        let run = self.assistant_runs.get(&item_id)?;
151        let last = *self.run_last_blocks.get(run)?;
152        self.reply_cut(last).is_some().then_some(last)
153    }
154
155    /// The sequence a revert or fork that keeps the reply ending with the
156    /// assistant block `item_id` cuts through: just before what came next
157    /// ([`Self::reply_boundaries`]), else the session's last event, so the
158    /// settings changed since stay. `None` unless the block is the last of its
159    /// run and the run finished.
160    #[must_use]
161    pub fn reply_cut(&self, item_id: u64) -> Option<u64> {
162        let run = *self.assistant_runs.get(&item_id)?;
163        let finished = *self.run_finishes.get(&run)?;
164        if self.run_last_blocks.get(&run) != Some(&item_id) {
165            return None;
166        }
167        Some(
168            self.reply_boundaries
169                .range(finished + 1..)
170                .next()
171                .map_or(self.last_seq, |next| next - 1)
172                .max(finished),
173        )
174    }
175}

177#[cfg(test)]
178mod tests {
179    use super::*;
180
181    #[test]
182    fn a_user_message_cuts_just_before_it() {
183        let point = CutPoint::UserMessage {
184            seq: 8,
185            text: "again".into(),
186        };
187        assert_eq!(point.through_seq(), 7);
188        assert_eq!(point.text(), Some("again"));
189        assert!(point.revertible() && point.forkable());
190        assert_eq!(before_user_message(1), 1);
191        let tip = CutPoint::ReplyEnd {
192            through_seq: 9,
193            at_tip: true,
194        };
195        assert!(!tip.revertible() && tip.forkable());
196        assert!(!CutPoint::Compaction { through_seq: 3 }.forkable());
197    }
198}
