1//! What the engine adds to a model's input beyond the conversation, as the
2//! transcript shows it, so nothing reaches the model unseen: the memory
3//! snapshot turn, a deferred tools announcement, a changed instructions
4//! announcement, the context a subagent
5//! inherits from its parent, and the context an ACP agent's first prompt
6//! carries, each a row whose body is the text the model reads; and the
7//! notes a tool row gets for what its result sent beyond the tool's own
8//! output (a loop warning, tool-emitted messages, an elision marker).
9
10use cookie_agent_protocol::{
11    AnnouncedPromptBlock, AnnouncedToolGroup, DelegatedContextRole, DelegatedContextTurn,
12    DisplayBlock, Open, ToolEmittedContent, ToolEmittedMessage, ToolEmittedMessageRole, model_text,
13};
14
15/// Which added input a [`crate::TranscriptItem::ModelInput`] row is.
16#[derive(Clone, Copy, Debug, Eq, PartialEq)]
17pub enum ModelInputKind {
18    /// The memory snapshot turn, shown when it differs from the last one.
19    Memory,
20    /// A deferred tools announcement.
21    DeferredTools,
22    /// Instructions that changed since the system prompt carried them.
23    PromptBlocks,
24    /// The parent's conversation a subagent starts from.
25    InheritedContext,
26    /// What an ACP agent's first prompt carries before the input.
27    AgentContext,
28}
29
30impl ModelInputKind {
31    /// A stable label for keys and test IDs.
32    pub fn as_str(self) -> &'static str {
33        match self {
34            Self::Memory => "memory",
35            Self::DeferredTools => "deferred-tools",
36            Self::PromptBlocks => "prompt-blocks",
37            Self::InheritedContext => "inherited-context",
38            Self::AgentContext => "agent-context",
39        }
40    }
41}

43/// The memory row's heading: `memory · 42 lines`, or that the model has
44/// none from here on.
45pub fn memory_header(text: &str) -> String {
46    if text.is_empty() {
47        "memory · none in the model's context from here on".to_owned()
48    } else {
49        format!("memory · {}", lines(text))
50    }
51}

53/// The deferred tools row's heading and body: `Deferred tools prompt`, and
54/// the announcement the model reads, which opens `[cookie: deferred tools]`.
55pub fn deferred_tools(
56    added: &[AnnouncedToolGroup],
57    removed: &[AnnouncedToolGroup],
58) -> (String, String) {
59    (
60        "Deferred tools prompt".to_owned(),
61        model_text::deferred_tools_announcement(added, removed),
62    )
63}
64
65/// A changed instructions row's heading and body: the blocks that changed,
66/// then the turn the model reads.
67pub fn prompt_blocks(blocks: &[AnnouncedPromptBlock]) -> (String, String) {
68    let keys = blocks
69        .iter()
70        .map(|block| match block.text {
71            Some(_) => block.key.clone(),
72            None => format!("{} removed", block.key),
73        })
74        .collect::<Vec<_>>();
75    (
76        format!("instructions changed · {}", keys.join(", ")),
77        model_text::prompt_blocks_announcement(blocks),
78    )
79}

81/// The inherited context row's heading and body: each turn under the
82/// role it has in the model's history.
83pub fn inherited_context(turns: &[DelegatedContextTurn]) -> (String, String) {
84    let body = turns
85        .iter()
86        .map(|turn| {
87            let role = match turn.role {
88                DelegatedContextRole::User => "user",
89                DelegatedContextRole::Assistant => "assistant",
90            };
91            format!("── {role}\n{}", turn.text)
92        })
93        .collect::<Vec<_>>()
94        .join("\n");
95    let count = if turns.len() == 1 {
96        "1 turn".to_owned()
97    } else {
98        format!("{} turns", turns.len())
99    };
100    (format!("context from the parent · {count}"), body)
101}

103/// Every file of the `AGENTS.md` turn `sources` make, in the turn's order
104/// (a project's first), and the turn as the model reads it.
105pub fn agent_md_turn(
106    sources: &[crate::AgentMdSourceState],
107) -> (Vec<cookie_agent_protocol::AgentMdEntry>, String) {
108    let entries = sources
109        .iter()
110        .filter(|source| source.project)
111        .chain(sources.iter().filter(|source| !source.project))
112        .flat_map(|source| source.entries.iter().cloned())
113        .collect();
114    let sources = sources
115        .iter()
116        .map(|source| model_text::AgentMdSource {
117            attachment: &source.attachment,
118            project: source.project,
119            entries: &source.entries,
120        })
121        .collect::<Vec<_>>();
122    (entries, model_text::agent_md_turn(&sources))
123}

125/// The note in a reply an interrupt cut off: the marker the model reads
126/// after it from the next request on, when the reply kept content, else
127/// only that it was interrupted.
128pub fn interrupted_note(marked: bool) -> String {
129    if marked {
130        format!(
131            "model interrupted · later requests follow this reply with: {}",
132            model_text::INTERRUPTED_TURN_MARKER
133        )
134    } else {
135        "model interrupted".to_owned()
136    }
137}

139/// The ACP context row's heading: `context sent to claude · 120 lines`.
140pub fn agent_context_header(agent: &str, text: &str) -> String {
141    format!("context sent to {agent} · {}", lines(text))

144fn lines(text: &str) -> String {
145    match crate::rows::display_line_count(text) {
146        1 => "1 line".to_owned(),
147        count => format!("{count} lines"),
148    }
149}

151/// The note a tool row shows for the loop warning its output carries.
152pub fn loop_warning_note(output: &str) -> Option<Open<DisplayBlock>> {
153    model_text::loop_warning(output)
154        .map(|warning| note(format!("Sent to the model with this result:\n{warning}")))
155}

157/// The notes a tool row shows for the messages its result emitted, each a
158/// user turn after the tool's result in the model's history.
159pub fn emitted_message_notes(messages: &[ToolEmittedMessage]) -> Vec<Open<DisplayBlock>> {
160    messages
161        .iter()
162        .map(|message| {
163            let mut text =
164                String::from("Sent to the model after this result, as a user message:\n");
165            if message.role == ToolEmittedMessageRole::System {
166                text.push_str(model_text::TOOL_EMITTED_SYSTEM_USER_MARKER);
167                text.push('\n');
168            }
169            let parts = message
170                .content
171                .iter()
172                .map(|part| match part {
173                    ToolEmittedContent::Text(text) => text.clone(),
174                    ToolEmittedContent::File(file) => format!(
175                        "[file: {} ({})]",
176                        file.filename.as_deref().unwrap_or("unnamed"),
177                        file.mime_type.as_str()
178                    ),
179                })
180                .collect::<Vec<_>>();
181            text.push_str(&parts.join("\n"));
182            note(text)
183        })
184        .collect()
185}

187/// The note a tool row shows once its output left the model's context:
188/// the marker the model reads in its place.
189pub fn elision_note(
190    retained_uri: &str,
191    original_bytes: u64,
192    emitted_messages: usize,
193) -> Open<DisplayBlock> {
194    let artifact = retained_uri
195        .strip_prefix("artifact://sha256/")
196        .unwrap_or(retained_uri);
197    note(format!(
198        "Left the model's context; the model reads instead:\n{}",
199        model_text::tool_output_elision_marker(artifact, original_bytes, emitted_messages)
200    ))
201}
202
203fn note(text: String) -> Open<DisplayBlock> {
204    Open::Known(DisplayBlock::Text {
205        text,
206        omitted_lines: None,
207    })
208}

210#[cfg(test)]
211mod tests {
212    use cookie_agent_protocol::{

221    fn text(block: &Open<DisplayBlock>) -> &str {
222        match block {
223            Open::Known(DisplayBlock::Text { text, .. }) => text,
224            other => panic!("not a text block: {other:?}"),
225        }
226    }
227
228    #[test]
229    fn a_tool_rows_notes_quote_what_the_model_reads() {
230        let output = format!("ls{LOOP_WARNING_MARKER}: you have made …\n</system-reminder>");
231        let warning = loop_warning_note(&output).expect("a warning");
232        assert!(text(&warning).ends_with(
233            "<system-reminder>\nRepeated tool calls detected: you have made …\n</system-reminder>"
234        ));
235        assert!(loop_warning_note("ls").is_none());
236
237        let system = ToolEmittedMessage {
238            role: ToolEmittedMessageRole::System,
239            content: vec![ToolEmittedContent::Text("Mind the cwd.".into())],
240        };
241        let notes = emitted_message_notes(&[system]);
242        assert_eq!(
243            text(&notes[0]),
244            "Sent to the model after this result, as a user message:\n\
245             [tool-emitted system message; materialized as user history]\nMind the cwd."
246        );
247
248        let elided = elision_note("artifact://sha256/abc", 12, 1);
249        assert_eq!(
250            text(&elided),
251            "Left the model's context; the model reads instead:\n[tool output elided; 12 bytes \
252             retained; read it with read_artifact(artifact=\"abc\")] 1 tool-emitted message(s) \
253             were elided with this result and are not recoverable."
254        );
255    }

257    #[test]
258    fn inherited_context_lists_each_turn_under_its_role() {
259        let turns = [
260            DelegatedContextTurn {
261                role: DelegatedContextRole::User,
262                text: "Fix the build.".into(),
263            },
264            DelegatedContextTurn {
265                role: DelegatedContextRole::Assistant,
266                text: "On it.".into(),
267            },
268        ];
269        let (header, body) = inherited_context(&turns);
270        assert_eq!(header, "context from the parent · 2 turns");
271        assert_eq!(body, "── user\nFix the build.\n── assistant\nOn it.");
272        assert_eq!(
273            memory_header(""),
274            "memory · none in the model's context from here on"
275        );
276        assert_eq!(memory_header("a\nb"), "memory · 2 lines");
277    }
