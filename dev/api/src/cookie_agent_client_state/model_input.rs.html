1//! What the engine adds to a model's input beyond the conversation, as the
2//! transcript shows it, so nothing reaches the model unseen: the memory
3//! snapshot turn, a changed instructions announcement, and the context a
4//! subagent inherits from its parent, each a row whose body is the text the
5//! model reads; and the notes a tool row gets for what its result sent
6//! beyond the tool's own output (a loop warning, tool-emitted messages, an
7//! elision marker). What an ACP agent's session is given shows as its
8//! system prompt ([`crate::rows::system_prompt_summary`]).
9
10use cookie_agent_protocol::{
11    AnnouncedPromptBlock, DelegatedContextRole, DelegatedContextTurn, DisplayBlock, Open,
12    ToolEmittedContent, ToolEmittedMessage, ToolEmittedMessageRole, model_text,
13};

15/// Which added input a [`crate::TranscriptItem::ModelInput`] row is.
16#[derive(Clone, Copy, Debug, Eq, PartialEq)]
17pub enum ModelInputKind {
18    /// The memory snapshot turn, shown when it differs from the last one.
19    Memory,
20    /// Instructions that changed since the system prompt carried them.
21    PromptBlocks,
22    /// The parent's conversation a subagent starts from.
23    InheritedContext,
24    /// The context before the conversation composed again between
25    /// compactions (frozen context spec, section 7.2).
26    Refresh,
27    /// Earlier reasoning the model is no longer sent (frozen context spec,
28    /// section 7.1).
29    ReasoningCut,
30    /// Tools declared up front that left mid-round, or came back (session
31    /// tool groups spec, section 8.2).
32    ToolsAvailability,
33}
34
35impl ModelInputKind {
36    /// A stable label for keys and test IDs.
37    pub fn as_str(self) -> &'static str {
38        match self {
39            Self::Memory => "memory",
40            Self::PromptBlocks => "prompt-blocks",
41            Self::InheritedContext => "inherited-context",
42            Self::Refresh => "refresh",
43            Self::ReasoningCut => "reasoning-cut",
44            Self::ToolsAvailability => "tools-availability",
45        }
46    }
47}

49/// The memory row's heading: `memory · 42 lines`, or that the model has
50/// none from here on.
51pub fn memory_header(text: &str) -> String {
52    if text.is_empty() {
53        "memory · none in the model's context from here on".to_owned()
54    } else {
55        format!("memory · {}", lines(text))
56    }
57}

59/// A changed instructions row's heading and body: the blocks that changed,
60/// then the turn the model reads.
61/// The tools availability row's heading and text: `tools no longer
62/// available · coding on labpc, exa`, and the notice as the model reads it.
63pub fn tools_availability(
64    unavailable: &[cookie_agent_protocol::UnavailableToolGroup],
65    available_again: &[cookie_agent_protocol::UnavailableToolGroup],
66) -> (String, String) {
67    let name = |group: &cookie_agent_protocol::UnavailableToolGroup| match &group.machine {
68        Some(machine) => format!("{} on {machine}", group.name),
69        None => group.name.clone(),
70    };
71    let mut parts = Vec::new();
72    if !unavailable.is_empty() {
73        parts.push(format!(
74            "tools no longer available · {}",
75            unavailable.iter().map(name).collect::<Vec<_>>().join(", ")
76        ));
77    }
78    if !available_again.is_empty() {
79        parts.push(format!(
80            "tools available again · {}",
81            available_again
82                .iter()
83                .map(name)
84                .collect::<Vec<_>>()
85                .join(", ")
86        ));
87    }
88    (
89        parts.join("; "),
90        model_text::tools_availability_notice(unavailable, available_again),
91    )
92}

94pub fn prompt_blocks(blocks: &[AnnouncedPromptBlock]) -> (String, String) {
95    let keys = blocks
96        .iter()
97        .map(|block| match block.text {
98            Some(_) => block.key.clone(),
99            None => format!("{} removed", block.key),
100        })
101        .collect::<Vec<_>>();
102    (
103        format!("instructions changed · {}", keys.join(", ")),
104        model_text::prompt_blocks_announcement(blocks),
105    )
106}

108/// The inherited context row's heading and body: each turn under the
109/// role it has in the model's history.
110pub fn inherited_context(turns: &[DelegatedContextTurn]) -> (String, String) {
111    let body = turns
112        .iter()
113        .map(|turn| {
114            let role = match turn.role {
115                DelegatedContextRole::User => "user",
116                DelegatedContextRole::Assistant => "assistant",
117            };
118            format!("── {role}\n{}", turn.text)
119        })
120        .collect::<Vec<_>>()
121        .join("\n");
122    let count = if turns.len() == 1 {
123        "1 turn".to_owned()
124    } else {
125        format!("{} turns", turns.len())
126    };
127    (format!("context from the parent · {count}"), body)
128}

140/// Every file of the `AGENTS.md` turn `sources` make, in the turn's order
141/// (a project's first), and the turn as the model reads it.
142pub fn agent_md_turn(
143    sources: &[crate::AgentMdSourceState],
144) -> (Vec<cookie_agent_protocol::AgentMdEntry>, String) {
145    let entries = sources
146        .iter()
147        .filter(|source| source.project)
148        .chain(sources.iter().filter(|source| !source.project))
149        .flat_map(|source| source.entries.iter().cloned())
150        .collect();
151    let sources = sources
152        .iter()
153        .map(|source| model_text::AgentMdSource {
154            attachment: &source.attachment,
155            project: source.project,
156            entries: &source.entries,
157        })
158        .collect::<Vec<_>>();
159    (entries, model_text::agent_md_turn(&sources))
160}

162/// The refresh row: its heading, with the reason, and its body, the parts
163/// composed again and what that costs (frozen context spec, section 7.2).
164pub fn refresh_row(
165    reason: cookie_agent_protocol::RefreshReason,
166    changed: &cookie_agent_protocol::RefreshedParts,
167) -> (String, String) {
168    use cookie_agent_protocol::RefreshReason;
169    let reason = match reason {
170        RefreshReason::EngineUpdated => "cookie was updated",
171        RefreshReason::ToolsChanged => "tools changed",
172        RefreshReason::AgentChanged => "the agent changed",
173        RefreshReason::Unknown => "the context changed",
174    };
175    let mut parts = Vec::new();
176    if changed.system_prompt {
177        parts.push("system prompt".to_owned());
178    }
179    let tools = &changed.tools;
180    let tool_count = tools.added.len() + tools.removed.len() + tools.changed.len();
181    if tool_count > 0 {
182        parts.push(format!(
183            "tools ({tool_count} {})",
184            if tool_count == 1 { "change" } else { "changes" }
185        ));
186    }
187    if changed.agent_md {
188        parts.push("AGENTS.md".to_owned());
189    }
190    if changed.memory {
191        parts.push("memory".to_owned());
192    }
193    let header = if parts.is_empty() {
194        format!("context refreshed · {reason}")
195    } else {
196        format!("context refreshed · {reason} · {}", parts.join(" · "))
197    };
198    let mut body = Vec::new();
199    for (label, names) in [
200        ("added", &tools.added),
201        ("removed", &tools.removed),
202        ("changed", &tools.changed),
203    ] {
204        if !names.is_empty() {
205            body.push(format!("tools {label}: {}", names.join(", ")));
206        }
207    }
208    body.push(
209        "The context before the conversation was composed again: the prompt cache is written \
210         again from the start, and earlier thinking is no longer sent."
211            .to_owned(),
212    );
213    (header, body.join("\n"))
214}

216/// The row of a `native_replay_cut` the clients show, its heading and its
217/// body: earlier reasoning is no longer sent. A cut after a compaction or a
218/// refresh shows no row of its own, since that row stands for it (frozen
219/// context spec, section 7.1).
220pub fn reasoning_cut_row(
221    cause: cookie_agent_protocol::ReplayCutCause,
222) -> Option<(&'static str, &'static str)> {
223    use cookie_agent_protocol::ReplayCutCause;
224    let why = match cause {
225        ReplayCutCause::Compaction | ReplayCutCause::Refreshed => return None,
226        ReplayCutCause::Refused => {
227            "The provider refused the earlier thinking, because the conversation before it changed \
228             since it was made; the request was sent again without it."
229        }
230        ReplayCutCause::PrefixChanged | ReplayCutCause::Unknown => {
231            "The conversation before the earlier thinking changed since it was made, so the \
232             provider would refuse it."
233        }
234        ReplayCutCause::ProviderDropped => {
235            "The provider dropped the earlier thinking, because the conversation before it changed \
236             since it was made."
237        }
238    };
239    Some(("earlier thinking no longer sent", why))
240}

242/// The note in a reply an interrupt cut off: the marker the model reads
243/// after it from the next request on, when the reply kept content, else
244/// only that it was interrupted.
245pub fn interrupted_note(marked: bool) -> String {
246    if marked {
247        format!(
248            "model interrupted · later requests follow this reply with: {}",
249            model_text::INTERRUPTED_TURN_MARKER
250        )
251    } else {
252        "model interrupted".to_owned()
253    }
254}

256fn lines(text: &str) -> String {
257    match crate::rows::display_line_count(text) {
258        1 => "1 line".to_owned(),
259        count => format!("{count} lines"),
260    }

263/// The note a tool row shows for the loop warning its output carries.
264pub fn loop_warning_note(output: &str) -> Option<Open<DisplayBlock>> {
265    model_text::loop_warning(output)
266        .map(|warning| note(format!("Sent to the model with this result:\n{warning}")))
267}

269/// The notes a tool row shows for the messages its result emitted, each a
270/// user turn after the tool's result in the model's history.
271pub fn emitted_message_notes(messages: &[ToolEmittedMessage]) -> Vec<Open<DisplayBlock>> {
272    messages
273        .iter()
274        .map(|message| {
275            let mut text =
276                String::from("Sent to the model after this result, as a user message:\n");
277            if message.role == ToolEmittedMessageRole::System {
278                text.push_str(model_text::TOOL_EMITTED_SYSTEM_USER_MARKER);
279                text.push('\n');
280            }
281            let parts = message
282                .content
283                .iter()
284                .map(|part| match part {
285                    ToolEmittedContent::Text(text) => text.clone(),
286                    ToolEmittedContent::File(file) => format!(
287                        "[file: {} ({})]",
288                        file.filename.as_deref().unwrap_or("unnamed"),
289                        file.mime_type.as_str()
290                    ),
291                })
292                .collect::<Vec<_>>();
293            text.push_str(&parts.join("\n"));
294            note(text)
295        })
296        .collect()
297}

299/// The note a tool row shows once its output left the model's context:
300/// the marker the model reads in its place.
301pub fn elision_note(
302    retained_uri: &str,
303    original_bytes: u64,
304    emitted_messages: usize,
305) -> Open<DisplayBlock> {
306    let artifact = retained_uri
307        .strip_prefix("artifact://sha256/")
308        .unwrap_or(retained_uri);
309    note(format!(
310        "Left the model's context; the model reads instead:\n{}",
311        model_text::tool_output_elision_marker(artifact, original_bytes, emitted_messages)
312    ))
313}
314
315fn note(text: String) -> Open<DisplayBlock> {
316    Open::Known(DisplayBlock::Text {
317        text,
318        omitted_lines: None,
319    })
320}

340    #[test]
341    fn a_tool_rows_notes_quote_what_the_model_reads() {
342        let output = format!("ls{LOOP_WARNING_MARKER}: you have made …\n</system-reminder>");
343        let warning = loop_warning_note(&output).expect("a warning");
344        assert!(text(&warning).ends_with(
345            "<system-reminder>\nRepeated tool calls detected: you have made …\n</system-reminder>"
346        ));
347        assert!(loop_warning_note("ls").is_none());
348
349        let system = ToolEmittedMessage {
350            role: ToolEmittedMessageRole::System,
351            content: vec![ToolEmittedContent::Text("Mind the cwd.".into())],
352        };
353        let notes = emitted_message_notes(&[system]);
354        assert_eq!(
355            text(&notes[0]),
356            "Sent to the model after this result, as a user message:\n\
357             [tool-emitted system message; materialized as user history]\nMind the cwd."
358        );
359
360        let elided = elision_note("artifact://sha256/abc", 12, 1);
361        assert_eq!(
362            text(&elided),
363            "Left the model's context; the model reads instead:\n[tool output elided; 12 bytes \
364             retained; read it with `read_artifact(artifact=\"abc\")`] 1 tool-emitted message(s) \
365             were elided with this result and are not recoverable."
366        );
367    }
