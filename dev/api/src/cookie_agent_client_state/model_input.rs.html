1//! What the engine adds to a model's input beyond the conversation, as the
2//! transcript shows it, so nothing reaches the model unseen: the memory
3//! snapshot turn, a deferred tools announcement, the context a subagent
4//! inherits from its parent, and the context an ACP agent's first prompt
5//! carries, each a row whose body is the text the model reads; and the
6//! notes a tool row gets for what its result sent beyond the tool's own
7//! output (a loop warning, tool-emitted messages, an elision marker).
8
9use cookie_agent_protocol::{
10    AnnouncedToolGroup, DelegatedContextRole, DelegatedContextTurn, DisplayBlock, Open,
11    ToolEmittedContent, ToolEmittedMessage, ToolEmittedMessageRole, model_text,
12};

14/// Which added input a [`crate::TranscriptItem::ModelInput`] row is.
15#[derive(Clone, Copy, Debug, Eq, PartialEq)]
16pub enum ModelInputKind {
17    /// The memory snapshot turn, shown when it differs from the last one.
18    Memory,
19    /// A deferred tools announcement.
20    DeferredTools,
21    /// The parent's conversation a subagent starts from.
22    InheritedContext,
23    /// What an ACP agent's first prompt carries before the input.
24    AgentContext,
25}
26
27impl ModelInputKind {
28    /// A stable label for keys and test IDs.
29    pub fn as_str(self) -> &'static str {
30        match self {
31            Self::Memory => "memory",
32            Self::DeferredTools => "deferred-tools",
33            Self::InheritedContext => "inherited-context",
34            Self::AgentContext => "agent-context",
35        }
36    }
37}

39/// The memory row's heading: `memory · 42 lines`, or that the model has
40/// none from here on.
41pub fn memory_header(text: &str) -> String {
42    if text.is_empty() {
43        "memory · none in the model's context from here on".to_owned()
44    } else {
45        format!("memory · {}", lines(text))
46    }
47}

49/// The deferred tools row's heading and body: `deferred tools · 3 added,
50/// 1 removed`, and the announcement the model reads.
51pub fn deferred_tools(
52    added: &[AnnouncedToolGroup],
53    removed: &[AnnouncedToolGroup],
54) -> (String, String) {
55    let count = |groups: &[AnnouncedToolGroup]| -> usize {
56        groups.iter().map(|group| group.tools.len()).sum()
57    };
58    let mut changes = Vec::new();
59    if !added.is_empty() {
60        changes.push(format!("{} added", count(added)));
61    }
62    if !removed.is_empty() {
63        changes.push(format!("{} removed", count(removed)));
64    }
65    (
66        format!("deferred tools · {}", changes.join(", ")),
67        model_text::deferred_tools_announcement(added, removed),
68    )
69}

71/// The inherited context row's heading and body: each turn under the
72/// role it has in the model's history.
73pub fn inherited_context(turns: &[DelegatedContextTurn]) -> (String, String) {
74    let body = turns
75        .iter()
76        .map(|turn| {
77            let role = match turn.role {
78                DelegatedContextRole::User => "user",
79                DelegatedContextRole::Assistant => "assistant",
80            };
81            format!("── {role}\n{}", turn.text)
82        })
83        .collect::<Vec<_>>()
84        .join("\n");
85    let count = if turns.len() == 1 {
86        "1 turn".to_owned()
87    } else {
88        format!("{} turns", turns.len())
89    };
90    (format!("context from the parent · {count}"), body)
91}

93/// Every file of the `AGENTS.md` turn `sources` make, in the turn's order
94/// (a project's first), and the turn as the model reads it.
95pub fn agent_md_turn(
96    sources: &[crate::AgentMdSourceState],
97) -> (Vec<cookie_agent_protocol::AgentMdEntry>, String) {
98    let entries = sources
99        .iter()
100        .filter(|source| source.project)
101        .chain(sources.iter().filter(|source| !source.project))
102        .flat_map(|source| source.entries.iter().cloned())
103        .collect();
104    let sources = sources
105        .iter()
106        .map(|source| model_text::AgentMdSource {
107            attachment: &source.attachment,
108            project: source.project,
109            entries: &source.entries,
110        })
111        .collect::<Vec<_>>();
112    (entries, model_text::agent_md_turn(&sources))
113}

115/// The note in a reply an interrupt cut off: the marker the model reads
116/// after it from the next request on, when the reply kept content, else
117/// only that it was interrupted.
118pub fn interrupted_note(marked: bool) -> String {
119    if marked {
120        format!(
121            "model interrupted · later requests follow this reply with: {}",
122            model_text::INTERRUPTED_TURN_MARKER
123        )
124    } else {
125        "model interrupted".to_owned()
126    }
127}

129/// The ACP context row's heading: `context sent to claude · 120 lines`.
130pub fn agent_context_header(agent: &str, text: &str) -> String {
131    format!("context sent to {agent} · {}", lines(text))
132}

134fn lines(text: &str) -> String {
135    match crate::rows::display_line_count(text) {
136        1 => "1 line".to_owned(),
137        count => format!("{count} lines"),
138    }
139}

141/// The note a tool row shows for the loop warning its output carries.
142pub fn loop_warning_note(output: &str) -> Option<Open<DisplayBlock>> {
143    model_text::loop_warning(output)
144        .map(|warning| note(format!("Sent to the model with this result:\n{warning}")))
145}

147/// The notes a tool row shows for the messages its result emitted, each a
148/// user turn after the tool's result in the model's history.
149pub fn emitted_message_notes(messages: &[ToolEmittedMessage]) -> Vec<Open<DisplayBlock>> {
150    messages
151        .iter()
152        .map(|message| {
153            let mut text =
154                String::from("Sent to the model after this result, as a user message:\n");
155            if message.role == ToolEmittedMessageRole::System {
156                text.push_str(model_text::TOOL_EMITTED_SYSTEM_USER_MARKER);
157                text.push('\n');
158            }
159            let parts = message
160                .content
161                .iter()
162                .map(|part| match part {
163                    ToolEmittedContent::Text(text) => text.clone(),
164                    ToolEmittedContent::File(file) => format!(
165                        "[file: {} ({})]",
166                        file.filename.as_deref().unwrap_or("unnamed"),
167                        file.mime_type.as_str()
168                    ),
169                })
170                .collect::<Vec<_>>();
171            text.push_str(&parts.join("\n"));
172            note(text)
173        })
174        .collect()
175}

177/// The note a tool row shows once its output left the model's context:
178/// the marker the model reads in its place.
179pub fn elision_note(
180    retained_uri: &str,
181    original_bytes: u64,
182    emitted_messages: usize,
183) -> Open<DisplayBlock> {
184    let artifact = retained_uri
185        .strip_prefix("artifact://sha256/")
186        .unwrap_or(retained_uri);
187    note(format!(
188        "Left the model's context; the model reads instead:\n{}",
189        model_text::tool_output_elision_marker(artifact, original_bytes, emitted_messages)
190    ))
191}
192
193fn note(text: String) -> Open<DisplayBlock> {
194    Open::Known(DisplayBlock::Text {
195        text,
196        omitted_lines: None,
197    })
198}

211    fn text(block: &Open<DisplayBlock>) -> &str {
212        match block {
213            Open::Known(DisplayBlock::Text { text, .. }) => text,
214            other => panic!("not a text block: {other:?}"),
215        }
216    }

218    #[test]
219    fn a_tool_rows_notes_quote_what_the_model_reads() {
220        let output = format!("ls{LOOP_WARNING_MARKER}: you have made …\n</system-reminder>");
221        let warning = loop_warning_note(&output).expect("a warning");
222        assert!(text(&warning).ends_with(
223            "<system-reminder>\nRepeated tool calls detected: you have made …\n</system-reminder>"
224        ));
225        assert!(loop_warning_note("ls").is_none());
226
227        let system = ToolEmittedMessage {
228            role: ToolEmittedMessageRole::System,
229            content: vec![ToolEmittedContent::Text("Mind the cwd.".into())],
230        };
231        let notes = emitted_message_notes(&[system]);
232        assert_eq!(
233            text(&notes[0]),
234            "Sent to the model after this result, as a user message:\n\
235             [tool-emitted system message; materialized as user history]\nMind the cwd."
236        );
237
238        let elided = elision_note("artifact://sha256/abc", 12, 1);
239        assert_eq!(
240            text(&elided),
241            "Left the model's context; the model reads instead:\n[tool output elided; 12 bytes \
242             retained; read it with read_artifact(artifact=\"abc\")] 1 tool-emitted message(s) \
243             were elided with this result and are not recoverable."
244        );
245    }

248    fn inherited_context_lists_each_turn_under_its_role() {
249        let turns = [
250            DelegatedContextTurn {
251                role: DelegatedContextRole::User,
252                text: "Fix the build.".into(),
253            },
254            DelegatedContextTurn {
255                role: DelegatedContextRole::Assistant,

259        let (header, body) = inherited_context(&turns);

261        assert_eq!(body, "── user\nFix the build.\n── assistant\nOn it.");
262        assert_eq!(
263            memory_header(""),
264            "memory · none in the model's context from here on"
265        );
266        assert_eq!(memory_header("a\nb"), "memory · 2 lines");
267    }
