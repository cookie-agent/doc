1//! Unread root sessions (`docs/site/specs/unread-sessions.md`): the daemon
2//! keeps them; clients show `SessionMeta::unread` and mark sessions with
3//! `session.unread.set`.
4
5use cookie_agent_protocol::SemVer;
6
7/// The first client protocol whose daemons keep unread sessions.
8pub const UNREAD_SINCE: SemVer = SemVer::new(1, 51, 0);
9
10/// Whether the daemon, speaking `version`, keeps unread sessions.
11#[must_use]
12pub fn daemon_keeps_unread(version: Option<SemVer>) -> bool {
13    version.is_some_and(|version| version >= UNREAD_SINCE)
14}
15
16/// What a status line says after marking `count` sessions `unread` or read.
17#[must_use]
18pub fn marked_message(count: usize, unread: bool) -> String {
19    let state = if unread { "unread" } else { "read" };
20    match count {
21        0 => format!("Nothing changed: already {state}."),
22        1 => format!("Marked 1 session {state}."),
23        count => format!("Marked {count} sessions {state}."),
24    }
25}

27#[cfg(test)]
28mod tests {
29    use super::*;
30
31    #[test]
32    fn unread_needs_protocol_1_51() {
33        assert!(daemon_keeps_unread(Some(SemVer::new(1, 51, 0))));
34        assert!(!daemon_keeps_unread(Some(SemVer::new(1, 50, 0))));
35        assert!(!daemon_keeps_unread(None));
36    }
37
38    #[test]
39    fn the_marked_message_counts_what_changed() {
40        assert_eq!(marked_message(0, false), "Nothing changed: already read.");
41        assert_eq!(marked_message(1, true), "Marked 1 session unread.");
42        assert_eq!(marked_message(3, false), "Marked 3 sessions read.");
43    }
44}
