1//! A session's model-selection draft, shared with the daemon's other
2//! clients: a pick is published as the session's selection once it settles,
3//! [`DRAFT_PUBLISH_DELAY`] after the last change, so cycling through agents
4//! or variants publishes only where it stops. A pick another client makes
5//! meanwhile wins: the unsent pick yields, and the draft follows the
6//! session's latest selection.
7//!
8//! The client owns the clock: it waits the delay itself and then asks
9//! whether its publish is still the latest.
10
11use std::time::Duration;
12
13use cookie_agent_protocol::{EventOrigin, EventPayload, EventSubscriptionMessage, SessionId};
14use cookie_agent_protocol_transport::ClientDelivery;
15
16/// How long a draft must stay unchanged before it is published.
17pub const DRAFT_PUBLISH_DELAY: Duration = Duration::from_millis(300);
18
19/// Which pick of the draft may still be published.
20#[derive(Clone, Copy, Debug, Default)]
21pub struct DraftPublish {
22    generation: u64,
23}
24
25impl DraftPublish {
26    /// The draft changed: returns the generation its publish, due after
27    /// [`DRAFT_PUBLISH_DELAY`], carries. An earlier pending publish no
28    /// longer goes out.
29    pub fn schedule(&mut self) -> u64 {
30        self.generation = self.generation.wrapping_add(1);
31        self.generation
32    }
33
34    /// Whether the publish of `generation` is still the latest pick.
35    pub fn is_current(&self, generation: u64) -> bool {
36        generation == self.generation
37    }
38
39    /// Another client picked: the pending publish, if any, yields.
40    pub fn yield_to_foreign_pick(&mut self) {
41        self.generation = self.generation.wrapping_add(1);
42    }
43}
44
45/// Whether `delivery` is a live model pick in the `viewed` session by a
46/// client other than the one whose origin is `own_origin`.
47pub fn is_foreign_model_pick(
48    delivery: &ClientDelivery,
49    viewed: Option<SessionId>,
50    own_origin: Option<&EventOrigin>,
51) -> bool {
52    let ClientDelivery::Live { message, .. } = delivery else {
53        return false;
54    };
55    let EventSubscriptionMessage::Event { event } = message.as_ref() else {
56        return false;
57    };
58    matches!(&event.payload, EventPayload::SessionModelSelected { .. })
59        && Some(event.session_id) == viewed
60        && event.origin.as_ref() != own_origin
61}

63#[cfg(test)]
64mod tests {
65    use super::DraftPublish;
66
67    #[test]
68    fn only_the_last_pick_publishes_and_a_foreign_pick_cancels_it() {
69        let mut publish = DraftPublish::default();
70        let first = publish.schedule();
71        let second = publish.schedule();
72        assert!(!publish.is_current(first));
73        assert!(publish.is_current(second));
74        publish.yield_to_foreign_pick();
75        assert!(!publish.is_current(second));
76        let third = publish.schedule();
77        assert!(publish.is_current(third));
78    }
79}
