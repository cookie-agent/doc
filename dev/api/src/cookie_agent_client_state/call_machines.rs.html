1//! Which machine a tool call ran on, as every client labels it
2//! (`docs/site/specs/tool-call-machines.md`). A session labels its calls once
3//! its machines come to two or more: those attached now and those its calls
4//! ran on, so detaching back down to one keeps the labels. Machines come from
5//! the listing a client already reads (`runtimes.list`, or
6//! `session.runtimes.list` for a guest); until one answers, nothing is labelled.
7

16/// The columns a label's name may take; a longer one is cut.
17const MAX_NAME_COLUMNS: usize = 24;
18
19/// How many machines a work row names before it counts the rest.
20const WORK_ROW_MACHINES: usize = 3;
21
22/// Each runtime's machine, as a listing says.
23#[derive(Clone, Debug, Default, Eq, PartialEq)]
24pub struct RuntimeMachines(BTreeMap<RuntimeId, String>);
25
26impl RuntimeMachines {
27    #[must_use]
28    pub fn from_listing(listing: &RuntimesListResult) -> Self {
29        Self(
30            listing
31                .runtimes
32                .iter()
33                .filter_map(|runtime| {
34                    let machine = runtime.machine.as_ref()?;
35                    Some((runtime.runtime_id.clone(), machine.as_str().to_owned()))
36                })
37                .collect(),
38        )
39    }
40
41    /// The name `runtime` goes by: its machine's, or its own ID when the
42    /// listing knows no machine for it.
43    fn name(&self, runtime: &RuntimeId) -> String {
44        self.0
45            .get(runtime)
46            .map_or_else(|| runtime.to_string(), Clone::clone)
47    }
48}

50/// The runtime a call ran on, when it ran on a machine: `global`'s tools, the
51/// base layer, and plugins act on none.
52#[must_use]
53pub fn call_runtime(tool: &ToolCallState) -> Option<&RuntimeId> {
54    let runtime = match tool.presentation.tool.source.as_ref()?.known()? {
55        ToolSource::Runtime { runtime_id, .. } | ToolSource::Acp { runtime_id, .. } => runtime_id,
56        ToolSource::Mcp {
57            runtime_id: Some(runtime_id),
58            ..
59        } => runtime_id,
60        _ => return None,
61    };
62    (runtime.as_str() != RuntimeId::GLOBAL).then_some(runtime)
63}

65/// One machine as its label shows it.
66#[derive(Clone, Debug, Eq, Hash, PartialEq)]
67pub struct MachineLabel {
68    pub name: String,
69    /// The name on a narrow row: without the prefix it shares with the
70    /// session's other machines.
71    pub short: String,
72    /// Its colour, the one its badges have (`badges::badge_hue`).
73    pub hue: u8,
74}
75
76/// The labels of one session's calls; empty while it does not label them.
77#[derive(Clone, Debug, Default, Eq, PartialEq)]
78pub struct MachineLabels {
79    runtimes: BTreeMap<RuntimeId, MachineLabel>,
80    wide_columns: usize,
81    narrow_columns: usize,
82    key: u64,
83}
84
85impl MachineLabels {
86    /// The labels of `state`'s calls, given the listing's `machines`.
87    #[must_use]
88    pub fn of_session(state: &SessionState, machines: Option<&RuntimeMachines>) -> Self {
89        let Some(machines) = machines else {
90            return Self::default();
91        };
92        let runtimes = state
93            .runtime_attachments
94            .attachments()
95            .iter()
96            .map(|attachment| &attachment.runtime_id)
97            .chain(state.acp.runtime_id.as_ref())
98            .chain(&state.call_runtimes)
99            .filter(|runtime| runtime.as_str() != RuntimeId::GLOBAL)
100            .map(|runtime| (runtime.clone(), machines.name(runtime)))
101            .collect::<BTreeMap<_, _>>();
102        let names = runtimes
103            .values()
104            .map(String::as_str)
105            .collect::<BTreeSet<_>>();
106        if names.len() < 2 {
107            return Self::default();
108        }
109        let shorts = short_names(&names);
110        let columns = |name: &str| name.chars().count().min(MAX_NAME_COLUMNS);
111        let wide_columns = names.iter().map(|name| columns(name)).max().unwrap_or(0);
112        let narrow_columns = shorts.values().map(|name| columns(name)).max().unwrap_or(0);
113        let labels = runtimes
114            .iter()
115            .map(|(runtime, name)| {
116                let label = MachineLabel {
117                    name: name.clone(),
118                    short: shorts[name.as_str()].to_owned(),
119                    hue: badge_hue(name),
120                };
121                (runtime.clone(), label)
122            })
123            .collect::<BTreeMap<_, _>>();
124        // FNV-1a, as `badge_hue`: stable, and a client's caches compare it only
125        // within one run.
126        let mut hasher = Fnv(0xcbf2_9ce4_8422_2325);
127        labels.hash(&mut hasher);
128        Self {
129            runtimes: labels,
130            wide_columns,
131            narrow_columns,
132            key: hasher.0,
133        }
134    }

136    /// `tool`'s label, while the session labels its calls.
137    #[must_use]
138    pub fn call(&self, tool: &ToolCallState) -> Option<&MachineLabel> {
139        self.runtime(call_runtime(tool)?)
140    }
141
142    #[must_use]
143    pub fn runtime(&self, runtime: &RuntimeId) -> Option<&MachineLabel> {
144        self.runtimes.get(runtime)
145    }
146
147    /// What a work row names of the machines of `runtimes`, each once, and how
148    /// many more it leaves to a count.
149    #[must_use]
150    pub fn work_row(&self, runtimes: &[RuntimeId]) -> (Vec<&MachineLabel>, usize) {
151        let mut labels = Vec::<&MachineLabel>::new();
152        for label in runtimes.iter().filter_map(|runtime| self.runtime(runtime)) {
153            if !labels.iter().any(|known| known.name == label.name) {
154                labels.push(label);
155            }
156        }
157        let more = labels.len().saturating_sub(WORK_ROW_MACHINES);
158        labels.truncate(WORK_ROW_MACHINES);
159        (labels, more)
160    }
161
162    /// The columns the widest name takes, so labels line up down a transcript.
163    #[must_use]
164    pub fn wide_columns(&self) -> usize {
165        self.wide_columns
166    }
167
168    /// The columns the widest short name takes.
169    #[must_use]
170    pub fn narrow_columns(&self) -> usize {
171        self.narrow_columns
172    }
173
174    /// A fingerprint of what the labels show, which a cached row depends on.
175    #[must_use]
176    pub fn key(&self) -> u64 {
177        self.key
178    }
179}

181struct Fnv(u64);
182
183impl Hasher for Fnv {
184    fn write(&mut self, bytes: &[u8]) {
185        for byte in bytes {
186            self.0 = (self.0 ^ u64::from(*byte)).wrapping_mul(0x0100_0000_01b3);
187        }
188    }
189
190    fn finish(&self) -> u64 {
191        self.0
192    }
193}

195/// Each of `names` without the longest prefix, up to a separator, it shares
196/// with another: `yunhao-labpc` beside `yunhao-labserver` is `labpc`. A name
197/// the cut would leave empty or like another's stays whole.
198fn short_names<'a>(names: &BTreeSet<&'a str>) -> BTreeMap<&'a str, &'a str> {
199    let cut = |name: &'a str| -> &'a str {
200        let shared = names
201            .iter()
202            .filter(|other| **other != name)
203            .map(|other| {
204                let common = name
205                    .char_indices()
206                    .zip(other.chars())
207                    .find(|((_, ours), theirs)| ours != theirs)
208                    .map_or(name.len().min(other.len()), |((at, _), _)| at);
209                name[..common]
210                    .rfind(['-', '_', '.'])
211                    .map_or(0, |separator| separator + 1)
212            })
213            .max()
214            .unwrap_or(0);
215        &name[shared..]
216    };
217    let cuts = names.iter().map(|name| cut(name)).collect::<Vec<_>>();
218    names
219        .iter()
220        .zip(&cuts)
221        .map(|(name, short)| {
222            let unique = cuts.iter().filter(|other| *other == short).count() == 1;
223            (
224                *name,
225                if unique && !short.is_empty() {
226                    *short
227                } else {
228                    *name
229                },
230            )
231        })
232        .collect()
233}

457    #[test]
458    fn short_names_drop_only_what_is_shared() {
459        let shorts = |names: &[&'static str]| {
460            let set = names.iter().copied().collect::<BTreeSet<_>>();
461            names
462                .iter()
463                .map(|name| short_names(&set)[name])
464                .collect::<Vec<_>>()
465        };
466        assert_eq!(
467            shorts(&["yunhao-labpc", "yunhao-labserver", "unicorn"]),
468            ["labpc", "labserver", "unicorn"]
469        );
470        // No separator inside what they share: nothing is cut mid-word.
471        assert_eq!(shorts(&["labpc", "labserver"]), ["labpc", "labserver"]);
472        // Two machines the cut would make alike stay whole.
473        assert_eq!(
474            shorts(&["a-foo", "a-bar", "b-foo", "b-bar"]),
475            ["a-foo", "a-bar", "b-foo", "b-bar"]
476        );
477        // A name that is the shared prefix itself stays whole.
478        assert_eq!(shorts(&["a-", "a-b"]), ["a-", "b"]);
479    }
