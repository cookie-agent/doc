14/// The client protocol minor from which the daemon keeps projects:
15/// `projects.list`, the `project.*` methods, `SessionMeta.project`, and
16/// `session.create.project`.
17pub const PROJECTS_MINOR: u32 = 8;
18
19/// Whether a daemon whose handshake reported `version` keeps projects.
20/// Before the handshake nothing is known.
21#[must_use]
22pub fn daemon_has_projects(version: Option<cookie_agent_protocol::SemVer>) -> bool {
23    version.is_some_and(|version| {
24        version
25            .shared_minor(cookie_agent_protocol::CLIENT_PROTOCOL_VERSION)
26            .is_some_and(|minor| minor >= PROJECTS_MINOR)
27    })
28}

30/// What a client calls a session outside every project.
31pub const NO_PROJECT: &str = "no project";

33/// A project as clients name it: its emoji, when it has one, then its name;
34/// [`NO_PROJECT`] for none.
35#[must_use]
36pub fn project_label(project: Option<&ProjectInfo>) -> String {
37    match project {
38        Some(ProjectInfo {
39            emoji: Some(emoji),
40            name,
41            ..
42        }) => format!("{emoji} {name}"),
43        Some(project) => project.name.to_string(),
44        None => NO_PROJECT.to_owned(),
45    }
46}

48/// The project `id` names among `projects`; `None` for no ID and for a
49/// project that no longer exists, which lists as no project.
50#[must_use]
51pub fn find_project<'a>(
52    projects: &'a [ProjectInfo],
53    id: Option<&ProjectId>,
54) -> Option<&'a ProjectInfo> {
55    let id = id?;
56    projects.iter().find(|project| project.id == *id)
57}

59/// One section of the session list: a project's root sessions, or (with no
60/// project) those of no project.
61#[derive(Clone, Debug, PartialEq)]
62pub struct ProjectSection {
63    pub project: Option<ProjectInfo>,
64    /// Active sessions first, then the rest, each newest first.
65    pub sessions: Vec<SessionMeta>,
66}
67
68impl ProjectSection {
69    /// The key a client remembers the section's folding under.
70    #[must_use]
71    pub fn key(&self) -> String {
72        section_key(self.project.as_ref().map(|project| &project.id))
73    }
74}

76/// The key of the section of `project` (`None`: no project).
77#[must_use]
78pub fn section_key(project: Option<&ProjectId>) -> String {
79    project.map_or_else(|| "none".to_owned(), |id| format!("project:{id}"))
80}

82/// The session list as project sections. A section per project, ordered by
83/// its latest session's activity; projects without sessions follow in name
84/// order; "No project" comes last and holds every root session of no
85/// project or of a project that no longer exists. Each section orders its
86/// sessions by [`order_sessions`]. Delegated sessions are left out.
87#[must_use]
88pub fn project_sections(projects: &[ProjectInfo], sessions: &[SessionMeta]) -> Vec<ProjectSection> {
89    let mut sections = projects
90        .iter()
91        .map(|project| ProjectSection {
92            project: Some(project.clone()),
93            sessions: Vec::new(),
94        })
95        .collect::<Vec<_>>();
96    let mut none = ProjectSection {
97        project: None,
98        sessions: Vec::new(),
99    };
100    for session in sessions
101        .iter()
102        .filter(|session| matches!(session.origin, SessionOrigin::Root))
103    {
104        let section = session.project.as_ref().and_then(|id| {
105            sections.iter_mut().find(|section| {
106                section
107                    .project
108                    .as_ref()
109                    .is_some_and(|project| project.id == *id)
110            })
111        });
112        section.unwrap_or(&mut none).sessions.push(session.clone());
113    }
114    for section in &mut sections {
115        order_sessions(&mut section.sessions);
116    }
117    order_sessions(&mut none.sessions);
118    let latest = |section: &ProjectSection| {
119        section
120            .sessions
121            .iter()
122            .map(|session| session.last_activity)
123            .max()
124    };
125    // The stable sort keeps `projects.list`'s name order among the projects
126    // without sessions.
127    sections.sort_by_key(|section| std::cmp::Reverse(latest(section)));
128    sections.push(none);
129    sections
130}

132/// One line of a project picker.
133#[derive(Clone, Debug, Eq, PartialEq)]
134pub enum ProjectChoice {
135    Project(ProjectId),
136    NoProject,
137    /// Creates a project, then picks it.
138    NewProject,
139}
140
141/// The lines of a project picker filtered by `query` (case-insensitive,
142/// over the name and the emoji): the projects in `projects`' order, "No
143/// project", then "New project…" (offered whatever the query says, when
144/// `creates`).
145#[must_use]
146pub fn project_choices(projects: &[ProjectInfo], query: &str, creates: bool) -> Vec<ProjectChoice> {
147    let query = query.trim().to_lowercase();
148    let mut choices = projects
149        .iter()
150        .filter(|project| query.is_empty() || project_label(Some(project)).contains(&query))
151        .map(|project| ProjectChoice::Project(project.id.clone()))
152        .collect::<Vec<_>>();
153    if query.is_empty() || NO_PROJECT.contains(&query) {
154        choices.push(ProjectChoice::NoProject);
155    }
156    if creates {
157        choices.push(ProjectChoice::NewProject);
158    }
159    choices
160}

162/// The longest emoji the daemon takes, in bytes.
163pub const MAX_EMOJI_BYTES: usize = 32;
164
165/// Checks `emoji` as the daemon does (`docs/site/specs/projects.md`
166/// section 3): one extended grapheme cluster of at most 32 bytes, holding a
167/// character Unicode marks as an emoji, made only of emoji characters and
168/// emoji components, and not ASCII, whitespace, or a control character.
169/// The error says why, for the field to show at once.
170pub fn validate_emoji(emoji: &str) -> Result<(), String> {
171    use unicode_properties::UnicodeEmoji as _;
172    use unicode_segmentation::UnicodeSegmentation as _;
173    if emoji.is_empty() {
174        return Err("it is empty".to_owned());
175    }
176    if emoji.len() > MAX_EMOJI_BYTES {
177        return Err(format!(
178            "it is {} bytes, over the {MAX_EMOJI_BYTES}-byte limit",
179            emoji.len()
180        ));
181    }
182    if emoji.is_ascii() {
183        return Err("it is not an emoji".to_owned());
184    }
185    if emoji
186        .chars()
187        .any(|character| character.is_whitespace() || character.is_control())
188    {
189        return Err("it holds a space or a control character".to_owned());
190    }
191    if emoji.graphemes(true).count() != 1 {
192        return Err("it is more than one emoji".to_owned());
193    }
194    if !emoji.chars().any(|character| character.is_emoji_char())
195        || !emoji
196            .chars()
197            .all(|character| character.is_emoji_char_or_emoji_component())
198    {
199        return Err("it is not an emoji".to_owned());
200    }
201    Ok(())
202}

204/// Whether a new session in a project with `defaults` takes their
205/// attachments: they name runtimes or leave out `global`, as the daemon
206/// decides. Defaults that only name an agent or a preset leave the usual
207/// attachments.
208#[must_use]
209pub fn defaults_attach(defaults: &ProjectDefaults) -> bool {
210    !defaults.runtimes.is_empty() || !defaults.global
211}

213/// The attachments `defaults` name: whether `global` is attached, and the
214/// other runtimes as `session.create` takes them, in order.
215#[must_use]
216pub fn defaults_requests(defaults: &ProjectDefaults) -> (bool, Vec<RuntimeAttachRequest>) {
217    (
218        defaults.global,
219        defaults
220            .runtimes
221            .iter()
222            .map(|runtime| RuntimeAttachRequest {
223                runtime_id: runtime.runtime_id.clone(),
224                capability: runtime.capability.clone(),
225                workspace: runtime.workspace.clone(),
226            })
227            .collect(),
228    )
229}

231/// The defaults a project editor saves: the attachments (`global` and the
232/// other runtimes, a runtime listed twice kept once), the agent, and the
233/// preset. `None` when that is all the daemon's usual defaults, so the
234/// project keeps none.
235#[must_use]
236pub fn editor_defaults(
237    global: bool,
238    requests: &[RuntimeAttachRequest],
239    agent: Option<&cookie_agent_protocol::AgentId>,
240    preset: Option<&str>,
241) -> Option<ProjectDefaults> {
242    let mut runtimes = Vec::<ProjectRuntime>::new();
243    for request in requests
244        .iter()
245        .filter(|request| request.runtime_id.as_str() != RuntimeId::GLOBAL)
246    {
247        let runtime = ProjectRuntime {
248            runtime_id: request.runtime_id.clone(),
249            capability: request.capability.clone(),
250            workspace: request
251                .workspace
252                .as_deref()
253                .map(str::trim)
254                .filter(|workspace| !workspace.is_empty())
255                .map(str::to_owned),
256        };
257        if !runtimes.contains(&runtime) {
258            runtimes.push(runtime);
259        }
260    }
261    let defaults = ProjectDefaults {
262        preset: preset.map(str::to_owned),
263        agent: agent.cloned(),
264        global,
265        runtimes,
266    };
267    (defaults != ProjectDefaults::default()).then_some(defaults)
268}

270/// What a project's default agent and preset make of a new session's
271/// selection.
272#[derive(Clone, Debug, Eq, PartialEq)]
273pub enum ProjectAgent {
274    /// The project names neither: the draft keeps its selection.
275    Unset,
276    /// The agent, its preset, and the agent's own default model.
277    Selected(RunSelection),
278    /// The agent is gone or cannot run as root: the draft keeps its usual
279    /// selection, and the message says so.
280    Unavailable(String),
281}
282
283/// The selection `project`'s default agent and preset give a new session
284/// (`docs/site/specs/projects.md` section 4): the agent among `agents` that
285/// runs as root in that preset (with no agent named, the preset's `primary`,
286/// else its first), and that agent's first model `live` accepts, else
287/// `fallback_model`.
288pub fn project_agent(
289    project: &ProjectInfo,
290    agents: &[AgentDescriptor],
291    live: impl Fn(&ModelSelection) -> bool,
292    fallback_model: Option<ModelSelection>,
293) -> ProjectAgent {
294    let Some(defaults) = project.defaults.as_ref() else {
295        return ProjectAgent::Unset;
296    };
297    if defaults.agent.is_none() && defaults.preset.is_none() {
298        return ProjectAgent::Unset;
299    }
300    let candidates = agents
301        .iter()
302        .filter(|agent| {
303            agent.runnable_as_root
304                && agent.mode != AgentMode::Internal
305                && agent.preset == defaults.preset
306        })
307        .collect::<Vec<_>>();
308    let agent = match &defaults.agent {
309        Some(wanted) => candidates.iter().find(|agent| agent.id == *wanted).copied(),
310        None => candidates
311            .iter()
312            .find(|agent| agent.id.as_str() == "primary")
313            .or_else(|| candidates.first())
314            .copied(),
315    };
316    let unavailable = || {
317        let named = match (&defaults.agent, &defaults.preset) {
318            (Some(agent), Some(preset)) => format!("agent {agent} of preset {preset}"),
319            (Some(agent), None) => format!("agent {agent}"),
320            (None, Some(preset)) => format!("preset {preset}"),
321            (None, None) => unreachable!("checked above"),
322        };
323        ProjectAgent::Unavailable(format!(
324            "Project {} names {named}, which cannot start a session here; using the usual agent.",
325            project.name
326        ))
327    };
328    let Some(agent) = agent else {
329        return unavailable();
330    };
331    let Some(model) = agent
332        .resolved_fallback
333        .iter()
334        .find(|selection| live(selection))
335        .cloned()
336        .or(fallback_model)
337    else {
338        return unavailable();
339    };
340    ProjectAgent::Selected(RunSelection {
341        agent: agent.id.clone(),
342        model,
343        preset: defaults.preset.clone(),
344    })
345}

347/// The question a project's delete asks, with what goes.
348#[must_use]
349pub fn delete_question(project: &ProjectInfo) -> String {
350    let sessions = match project.sessions {
351        1 => "1 session".to_owned(),
352        count => format!("{count} sessions"),
353    };
354    let memories = match project.memories {
355        1 => "1 memory".to_owned(),
356        count => format!("{count} memories"),
357    };
358    format!(
359        "Delete project {} with its {sessions}, {memories}, and AGENTS.md? This cannot be undone.",
360        project.name
361    )
362}

364/// The sessions a refused delete names as running
365/// (`project_sessions_running`); empty for any other error.
366#[must_use]
367pub fn running_sessions(error: &ClientError) -> Vec<SessionId> {
368    if !error.is(&cookie_agent_protocol::errors::PROJECT_SESSIONS_RUNNING) {
369        return Vec::new();
370    }
371    error
372        .error_data()
373        .and_then(|data| data.fields.get("session_ids").cloned())
374        .and_then(|ids| serde_json::from_value::<Vec<SessionId>>(ids).ok())
375        .unwrap_or_default()
376}

378/// What a client says when a delete was refused because `running` sessions
379/// run: their titles (else short IDs) from `sessions`.
380#[must_use]
381pub fn running_sessions_message(running: &[SessionId], sessions: &[SessionMeta]) -> String {
382    let names = running
383        .iter()
384        .map(|id| {
385            sessions
386                .iter()
387                .find(|session| session.session_id == *id)
388                .and_then(|session| {
389                    session
390                        .title
391                        .as_ref()
392                        .map(ToString::to_string)
393                        .or_else(|| session.short_id.clone())
394                })
395                .map_or_else(|| id.to_string(), |name| format!("“{name}”"))
396        })
397        .collect::<Vec<_>>();
398    match names.as_slice() {
399        [] => "Sessions of this project are running; stop them, then delete it.".to_owned(),
400        [one] => format!("{one} is running; stop it, then delete the project."),
401        several => format!(
402            "{} are running; stop them, then delete the project.",
403            several.join(", ")
404        ),
405    }
406}
