1//! The cookie agent web client's built page, embedded at compile time for
2//! `cookie daemon` to serve: `scripts/build-web.sh` writes it to `bundle/`
3//! beside this crate. Without a built bundle the crate holds a placeholder
4//! `index.html` that says how to build one.
5
6/// One file of the page.
7#[derive(Clone, Copy, Debug)]
8pub struct Asset {
9    /// Its path below the page's root, `/`-separated, without a leading `/`.
10    pub path: &'static str,
11    /// Its `Content-Type`.
12    pub content_type: &'static str,
13    /// Whether its name carries a hash of its content, so a client may
14    /// keep it for good; `index.html` and other plain names may change.
15    pub immutable: bool,
16    /// A strong `ETag` of its bytes, quoted, for revalidating a file that
17    /// is not immutable.
18    pub etag: &'static str,
19    pub bytes: &'static [u8],
20    /// The file gzip-compressed, when that makes it smaller.
21    pub gzip: Option<&'static [u8]>,
22}
23
24include!(concat!(env!("OUT_DIR"), "/assets.rs"));
25
26/// The file at `path` below the page's root (`index.html` for the root).
27#[must_use]
28pub fn get(path: &str) -> Option<&'static Asset> {
29    ASSETS.iter().find(|asset| asset.path == path)
30}
31
32/// Every file of the page.
33#[must_use]
34pub fn assets() -> &'static [Asset] {
35    ASSETS
36}
37
38/// The Content-Security-Policy the page is served with: its own origin,
39/// its inline scripts by hash, WebAssembly, `blob:` media, and WebSocket
40/// connections to any daemon.
41#[must_use]
42pub const fn content_security_policy() -> &'static str {
43    POLICY
44}
45
46/// Whether the real page was built in, rather than the placeholder.
47#[must_use]
48pub const fn is_built() -> bool {
49    BUILT
50}

52#[cfg(test)]
53mod tests {
54    #[test]
55    fn the_page_always_has_an_index() {
56        let index = super::get("index.html").expect("index.html");
57        assert_eq!(index.content_type, "text/html; charset=utf-8");
58        assert!(!index.bytes.is_empty());
59        assert!(!index.immutable);
60        assert!(super::content_security_policy().contains("default-src 'self'"));
61        assert!(super::get("missing.js").is_none());
62    }
