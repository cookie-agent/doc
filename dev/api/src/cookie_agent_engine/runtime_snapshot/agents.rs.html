36/// (`docs/site/specs/acp-agents.md` section 3): what a session that

81    /// The agent on the one runtime of `attached` that offers it. Refused,
82    /// saying why: no attached runtime offers it, two or more do, or the
83    /// one that does is away or has it disabled.
84    pub(crate) fn place(
85        &self,
86        attached: &[cookie_agent_protocol::RuntimeId],
87    ) -> Result<RuntimeAgent, String> {
88        let offering = self
89            .offers
90            .iter()
91            .filter(|offer| attached.contains(&offer.runtime))
92            .collect::<Vec<_>>();
93        match offering[..] {
94            [] => Err(format!(
95                "`{}` needs a runtime that offers it attached: {}",
96                self.id,
97                join_runtimes(&self.runtimes())
98            )),
99            [offer] if !offer.enabled => Err(format!(
100                "ACP agent `{}` is disabled on runtime `{}`",
101                self.id, offer.runtime
102            )),
103            [offer] if !offer.connected => Err(format!(
104                "ACP agent `{}` is unavailable: runtime `{}` is not connected",
105                self.id, offer.runtime
106            )),
107            [offer] => Ok(offer.clone()),
108            _ => Err(format!(
109                "`{}` is offered by more than one attached runtime ({}); a session runs it on \
110                 one, so detach the others",
111                self.id,
112                join_runtimes(
113                    &offering
114                        .iter()
115                        .map(|offer| offer.runtime.clone())
116                        .collect::<Vec<_>>()
117                )
118            )),
119        }
120    }
121}

125    runtimes

170/// The variant `variant` selects of `model`: its default variant when
171/// absent.
172fn resolve_variant(
173    models: &CompiledModelRuntime,
174    model: &ModelKey,
175    variant: Option<&ConfiguredVariantRef>,
176) -> Option<cookie_agent_identity::VariantId> {
177    match variant {
178        None => models
179            .model(model)
180            .and_then(|model| model.model.default_variant.clone()),
181        Some(ConfiguredVariantRef::Base) => None,
182        Some(ConfiguredVariantRef::Named(id)) => Some(id.clone()),
183    }
184}

200    }

215    pub(crate) fn resolve(
216        authored: &ConfigAgentRegistry,
217        models: &CompiledModelRuntime,
218        preset: Option<String>,
219        runtime_agents: &[RuntimeAgent],
220        variables: ModelVariables<'_>,
221    ) -> Result<Self, EngineError> {
222        let mut documents = built_in_internal_documents()?;
223        // An ACP agent's document is an agent only while a runtime declares
224        // the agent.
225        documents.extend(
226            authored
227                .documents()
228                .iter()
229                .filter(|(_, document)| document.frontmatter.acp.is_none())
230                .map(|(id, document)| (id.clone(), document.clone())),
231        );
232        let mut agents = BTreeMap::new();
233        // Every runtime that declares an ID offers the one agent of that ID.
234        let mut offered = BTreeMap::<AgentId, Vec<RuntimeAgent>>::new();
235        for runtime_agent in runtime_agents {
236            offered
237                .entry(runtime_agent.id.clone())
238                .or_default()
239                .push(runtime_agent.clone());
240        }
241        for (id, mut offers) in offered {
242            if documents.contains_key(&id) {
243                // A daemon agent of the same ID wins; the configuration
244                // status names the ignored declarations.
245                continue;
246            }
247            offers.sort_by(|left, right| left.runtime.cmp(&right.runtime));
248            let offers = AcpOffers { id, offers };
249            let document = match authored.get(&offers.id) {
250                Some(document) => document.clone(),
251                None => runtime_agent_default_document(&offers)?,
252            };
253            // Whether it can run depends on the session's runtimes: it is
254            // a root choice while some runtime could run it.
255            let runnable_as_root = offers.any_available()
256                && document.frontmatter.enabled
257                && matches!(
258                    document.frontmatter.mode,
259                    AgentMode::Primary | AgentMode::All
260                );
261            agents.insert(
262                offers.id.clone(),
263                ResolvedAgent {
264                    document,
265                    resolved_fallback: vec![ResolvedAgentFallback::Selection {
266                        selection: cookie_agent_protocol::acp_model_selection(&offers.id),
267                        cache: None,
268                    }],
269                    unset_models: Vec::new(),
270                    runnable_as_root,
271                    acp: Some(offers),
272                },
273            );
274        }
275        for document in documents.into_values() {
276            let id = document.id.clone();
277            if id.as_str() == BUILT_IN_DEFAULT_AGENT_ID {
278                return Err(EngineError::RuntimeCompileFailed);
279            }
280            let mut resolved_fallback = Vec::new();
281            let mut unset_models = Vec::new();
282            for fallback in &document.frontmatter.models {
283                match &fallback.model {
284                    AgentModelRef::ParentModel => {
285                        resolved_fallback.push(ResolvedAgentFallback::ParentModel {
286                            variant: fallback.variant.clone(),
287                            cache: fallback.cache.clone(),
288                        });
289                    }
290                    AgentModelRef::Model(model_key) => {
291                        resolved_fallback.push(ResolvedAgentFallback::Selection {
292                            selection: ModelSelection {
293                                model: model_key.clone(),
294                                variant: resolve_variant(
295                                    models,
296                                    model_key,
297                                    fallback.variant.as_ref(),
298                                ),
299                            },
300                            cache: fallback.cache.clone(),
301                        });
302                    }
303                    variable @ (AgentModelRef::DefaultModel | AgentModelRef::SmallModel) => {
304                        match variables.get(variable) {
305                            Some(configured) => {
306                                resolved_fallback.push(ResolvedAgentFallback::Selection {
307                                    selection: configured_selection(
308                                        models,
309                                        configured,
310                                        fallback.variant.as_ref(),
311                                    ),
312                                    cache: fallback.cache.clone(),
313                                });
314                            }
315                            // The entry is skipped; the agent reports the
316                            // field to set.
317                            None => unset_models.extend(variable.config_field()),
318                        }
319                    }
320                }
321            }
322            unset_models.sort_unstable();
323            unset_models.dedup();
324            // A chain names each model once: an entry resolving to a model an
325            // earlier one already did (`${default_model}` and the key it
326            // names) adds nothing.
327            let mut seen = std::collections::BTreeSet::new();
328            resolved_fallback.retain(|fallback| match fallback {
329                ResolvedAgentFallback::Selection { selection, .. } => {
330                    seen.insert(selection.model.clone())
331                }
332                ResolvedAgentFallback::ParentModel { .. } => true,
333            });
334            let available = resolved_fallback
335                .iter()
336                .filter_map(|fallback| match fallback {
337                    ResolvedAgentFallback::Selection { selection, .. } => Some(selection),
338                    ResolvedAgentFallback::ParentModel { .. } => None,
339                })
340                .any(|selection| selection_available(models, selection));
341            let runnable_as_root = document.frontmatter.enabled
342                && matches!(
343                    document.frontmatter.mode,
344                    AgentMode::Primary | AgentMode::All
345                )
346                && available;
347            agents.insert(
348                id,
349                ResolvedAgent {
350                    document,
351                    resolved_fallback,
352                    unset_models,
353                    runnable_as_root,
354                    acp: None,
355                },
356            );
357        }
358        if !agents.values().any(|agent| agent.runnable_as_root) {
359            // `${default_model}` first, then the first available model.
360            let configured = variables
361                .default_model
362                .map(|configured| configured_selection(models, configured, None))
363                .filter(|selection| selection_available(models, selection));
364            let first = first_available_selection(models)
365                .filter(|first| configured.as_ref().is_none_or(|c| c.model != first.model));
366            let resolved_fallback = configured
367                .iter()
368                .chain(first.iter())
369                .map(|selection| ResolvedAgentFallback::Selection {
370                    selection: selection.clone(),
371                    cache: None,
372                })
373                .collect::<Vec<_>>();
374            if !resolved_fallback.is_empty() {
375                let document = built_in_default_document(first.as_ref())?;
376                agents.insert(
377                    document.id.clone(),
378                    ResolvedAgent {
379                        document,
380                        resolved_fallback,
381                        unset_models: if variables.default_model.is_none() {
382                            vec!["default_model"]
383                        } else {
384                            Vec::new()
385                        },
386                        runnable_as_root: true,
387                        acp: None,
388                    },
389                );
390            }
391        }
392        let descriptors = agents
393            .iter()
394            .map(|(id, agent)| AgentDescriptor {
395                id: id.clone(),
396                preset: preset.clone(),
397                description: agent.document.frontmatter.description.clone(),
398                mode: agent.document.frontmatter.mode,
399                enabled: agent.document.frontmatter.enabled,
400                runnable_as_root: agent.runnable_as_root,
401                resolved_fallback: agent
402                    .resolved_fallback
403                    .iter()
404                    .filter_map(|fallback| match fallback {
405                        ResolvedAgentFallback::Selection { selection, .. } => {
406                            Some(selection.clone())
407                        }
408                        ResolvedAgentFallback::ParentModel { .. } => None,
409                    })
410                    .collect(),
411                delegation_targets: delegation_targets(
412                    &agent.document.frontmatter.permissions,
413                    &agents,
414                ),
415                unset_models: agent
416                    .unset_models
417                    .iter()
418                    .map(|field| (*field).to_owned())
419                    .collect(),
420                runtime: agent.acp.as_ref().and_then(|acp| match &acp.offers[..] {
421                    [offer] => Some(agent_runtime(offer)),
422                    _ => None,
423                }),
424                runtimes: agent
425                    .acp
426                    .iter()
427                    .flat_map(|acp| acp.offers.iter().map(agent_runtime))
428                    .collect(),
429            })
430            .collect();
431        Ok(Self {
432            preset,
433            agents,
434            descriptors,
435        })
436    }

450    /// The delegation rules of its agents that reach no agent yet: a named
451    /// target no agent has, and a pattern that matches no eligible agent.
452    /// Either may be an ACP agent a runtime has not reported (or a typo);
453    /// the configuration loads, and the diagnostic clears once a runtime
454    /// reports a matching agent.
455    /// The diagnostics of the agents and of their `models` entries: the
456    /// delegation ones, then an authored agent's entries naming a field that
457    /// is not set.
458    pub(crate) fn diagnostics(&self) -> Vec<String> {
459        let mut diagnostics = self.delegation_diagnostics();
460        for (id, agent) in &self.agents {
461            if agent.unset_models.is_empty()
462                || agent.document.source == AgentDocumentSource::BuiltIn
463            {
464                continue;
465            }
466            let fields = agent
467                .unset_models
468                .iter()
469                .map(|field| format!("`{field}`"))
470                .collect::<Vec<_>>()
471                .join(" and ");
472            let verb = if agent.unset_models.len() == 1 {
473                "is"
474            } else {
475                "are"
476            };
477            diagnostics.push(if agent.resolved_fallback.is_empty() {
478                format!("agent `{id}` has no model: {fields} {verb} not set")
479            } else {
480                format!("agent `{id}` skips a `models` entry: {fields} {verb} not set")
481            });
482        }
483        diagnostics
484    }
485
486    pub(crate) fn delegation_diagnostics(&self) -> Vec<String> {
487        let mut diagnostics = Vec::new();
488        for (id, agent) in &self.agents {
489            let Some(PermissionValue::Resources(resources)) =
490                agent.document.frontmatter.permissions.get(
491                    &cookie_agent_config::PermissionKey::any(PermissionAction::DELEGATE),
492                )
493            else {
494                continue;
495            };
496            for (resource, effect) in resources {
497                if *effect == PermissionEffect::Deny {
498                    continue;
499                }
500                let pattern = resource.as_str();
501                if pattern.contains('*') || pattern.contains('?') {
502                    let single = IndexMap::from([(
503                        cookie_agent_config::PermissionKey::any(PermissionAction::DELEGATE),
504                        PermissionValue::Resources(IndexMap::from([(resource.clone(), *effect)])),
505                    )]);
506                    if delegation_targets(&single, &self.agents).is_empty() {
507                        diagnostics.push(format!(
508                            "agent `{id}` delegates to `{pattern}`, which matches no agent yet: \
509                             no enabled subagent document matches it and no connected runtime \
510                             reports an agent it matches"
511                        ));
512                    }
513                } else if AgentId::new(pattern)
514                    .is_ok_and(|target| !self.agents.contains_key(&target))
515                {
516                    diagnostics.push(format!(
517                        "agent `{id}` delegates to `{pattern}`, which is not known yet: no \
518                         agent document has that ID and no connected runtime reports it"
519                    ));
520                }
521            }
522        }
523        diagnostics
524    }
525}

536/// The built-in internal agent documents.
537///
538/// Each declares `max_output_tokens: 0`, so every built-in internal agent
539/// inherits the owner run's document cap at execution time.
540fn built_in_internal_documents() -> Result<BTreeMap<AgentId, AgentDocument>, EngineError> {
541    // Approval and titles run on the small model when one is set, then on
542    // the run's own; compaction stays on the run's model, whose prompt
543    // cache the session keeps warm.
544    let small_then_parent = [AgentModelRef::SmallModel, AgentModelRef::ParentModel];
545    [
546        (
547            BUILT_IN_APPROVAL_AGENT_ID,
548            "Built-in approval evaluator",
549            &small_then_parent[..],
550            AgentLimits {
551                timeout_ms: 30_000,
552                max_output_tokens: 0,
553            },
554        ),
555        (
556            BUILT_IN_COMPACTION_AGENT_ID,
557            "Built-in context compaction agent",
558            &[AgentModelRef::ParentModel][..],
559            AgentLimits {
560                timeout_ms: 180_000,
561                max_output_tokens: 0,
562            },
563        ),
564        (
565            BUILT_IN_TITLE_AGENT_ID,
566            "Built-in session title agent",
567            &small_then_parent[..],
568            AgentLimits {
569                timeout_ms: 30_000,
570                max_output_tokens: 0,
571            },
572        ),
573    ]
574    .into_iter()
575    .map(|(id, description, models, limits)| {
576        let document =
577            built_in_internal_document(id, description, "{{default_prompt}}\n", models, limits)?;
578        Ok((document.id.clone(), document))
579    })
580    .collect()
581}

583fn built_in_internal_document(
584    id: &str,
585    description: &str,
586    body: &str,
587    models: &[AgentModelRef],
588    limits: AgentLimits,
589) -> Result<AgentDocument, EngineError> {
590    let id = IdentityAgentId::new(id).map_err(|_| EngineError::RuntimeCompileFailed)?;
591    let body = body.to_owned();
592    let frontmatter = AgentFrontmatter {
593        description: description.to_owned(),
594        mode: AgentMode::Internal,
595        enabled: true,
596        models: models
597            .iter()
598            .map(|model| AgentModelFallback {
599                model: model.clone(),
600                variant: None,
601                cache: None,
602            })
603            .collect(),
604        limits,
605        permissions: IndexMap::new(),
606        agent_md: None,
607        acp: None,
608    };
609    let document_fingerprint = fingerprint(
610        "cookie-agent/built-in-internal-agent-document/v2",
611        &(id.as_str(), &frontmatter, &body),
612    )?;
613    let prompt_fingerprint = fingerprint("cookie-agent/system-prompt/v1", &body)?;
614    Ok(AgentDocument {
615        id,
616        frontmatter,
617        body,
618        source: AgentDocumentSource::BuiltIn,
619        document_fingerprint,
620        prompt_fingerprint,
621    })
622}

632/// The document of an ACP agent the daemon has none for
633/// (`docs/site/specs/acp-agents.md` section 4): mode `all`, the first
634/// offering runtime's description, no rules of its own, and an empty body.
635fn runtime_agent_default_document(agent: &AcpOffers) -> Result<AgentDocument, EngineError> {
636    let description = agent
637        .offers
638        .iter()
639        .map(|offer| offer.description.trim())
640        .find(|description| !description.is_empty())
641        .map_or_else(
642            || {
643                format!(
644                    "ACP agent `{}` on {}",
645                    agent.id,
646                    join_runtimes(&agent.runtimes())
647                )
648            },
649            str::to_owned,
650        );
651    let frontmatter = AgentFrontmatter {
652        description,
653        mode: AgentMode::All,
654        enabled: true,
655        models: Vec::new(),
656        limits: AgentLimits::default(),
657        permissions: IndexMap::new(),
658        agent_md: None,
659        acp: Some(cookie_agent_config::AcpDocumentSettings::default()),
660    };
661    let body = String::new();
662    let document_fingerprint = fingerprint(
663        "cookie-agent/runtime-agent-document/v1",
664        &(agent.id.as_str(), &frontmatter, &body),
665    )?;
666    let prompt_fingerprint = fingerprint("cookie-agent/system-prompt/v1", &body)?;
667    Ok(AgentDocument {
668        id: agent.id.clone(),
669        frontmatter,
670        body,
671        source: AgentDocumentSource::BuiltIn,
672        document_fingerprint,
673        prompt_fingerprint,
674    })
675}

677/// The built-in `default` agent: `${default_model}`, then `first`, the
678/// first available model, when it is another model.
679fn built_in_default_document(first: Option<&ModelSelection>) -> Result<AgentDocument, EngineError> {
680    let id = IdentityAgentId::new(BUILT_IN_DEFAULT_AGENT_ID)
681        .map_err(|_| EngineError::RuntimeCompileFailed)?;
682    let body = "You are Cookie Agent's built-in default coding agent. Help the user inspect, understand, and modify software safely and precisely.\n".to_owned();
683    let models = std::iter::once(AgentModelFallback {
684        model: AgentModelRef::DefaultModel,
685        variant: None,
686        cache: None,
687    })
688    .chain(first.map(|selection| AgentModelFallback {
689        model: AgentModelRef::Model(selection.model.clone()),
690        variant: selection.variant.clone().map(ConfiguredVariantRef::Named),
691        cache: None,
692    }))
693    .collect();
694    let frontmatter = AgentFrontmatter {
695        description: "Built-in default coding agent".to_owned(),
696        mode: AgentMode::Primary,
697        enabled: true,
698        models,
699        limits: AgentLimits::default(),
700        permissions: built_in_default_permissions()?,
701        agent_md: None,
702        acp: None,
703    };
704    let document_fingerprint = fingerprint(
705        "cookie-agent/built-in-default-document/v3",
706        &(id.as_str(), &frontmatter, &body),
707    )?;
708    let prompt_fingerprint = fingerprint("cookie-agent/system-prompt/v1", &body)?;
709    Ok(AgentDocument {
710        id,
711        frontmatter,
712        body,
713        source: AgentDocumentSource::BuiltIn,
714        document_fingerprint,
715        prompt_fingerprint,
716    })
717}

719fn fingerprint(
720    domain: &str,
721    value: &impl Serialize,
722) -> Result<cookie_agent_models::Sha256Digest, EngineError> {
723    let mut hasher = Sha256::new();
724    hasher.update(domain.as_bytes());
725    hasher.update([0]);
726    hasher.update(serde_json::to_vec(value).map_err(|_| EngineError::RuntimeCompileFailed)?);
727    cookie_agent_models::Sha256Digest::new(format!("{:x}", hasher.finalize()))
728        .map_err(|_| EngineError::RuntimeCompileFailed)
729}

731fn built_in_default_permissions()
732-> Result<IndexMap<cookie_agent_config::PermissionKey, PermissionValue>, EngineError> {
733    let definitions = [
734        (
735            PermissionAction::READ,
736            "tool_result:*",
737            PermissionEffect::Allow,
738        ),
739        (PermissionAction::READ, "*?*", PermissionEffect::Allow),
740        (PermissionAction::WRITE, "*", PermissionEffect::Ask),
741        (PermissionAction::BASH, "*", PermissionEffect::Ask),
742        (PermissionAction::DELEGATE, "*", PermissionEffect::Ask),
743        (PermissionAction::READ, "/*", PermissionEffect::Ask),
744        (PermissionAction::READ, ".env", PermissionEffect::Deny),
745        (PermissionAction::READ, "*/.env", PermissionEffect::Deny),
746        (PermissionAction::READ, ".env.*", PermissionEffect::Deny),
747        (PermissionAction::READ, "*/.env.*", PermissionEffect::Deny),
748        (
749            PermissionAction::READ,
750            ".env.example",
751            PermissionEffect::Allow,
752        ),
753        (
754            PermissionAction::READ,
755            "*/.env.example",
756            PermissionEffect::Allow,
757        ),
758        (
759            PermissionAction::READ,
760            "store-v3.json",
761            PermissionEffect::Deny,
762        ),
763        (
764            PermissionAction::READ,
765            "*/store-v3.json",
766            PermissionEffect::Deny,
767        ),
768        (PermissionAction::READ, "id_*", PermissionEffect::Deny),
769        (PermissionAction::READ, "*/id_*", PermissionEffect::Deny),
770        (PermissionAction::READ, ".netrc", PermissionEffect::Deny),
771        (PermissionAction::READ, "*/.netrc", PermissionEffect::Deny),
772        (
773            PermissionAction::READ,
774            "application_default_credentials.json",
775            PermissionEffect::Deny,
776        ),
777        (
778            PermissionAction::READ,
779            "*/application_default_credentials.json",
780            PermissionEffect::Deny,
781        ),
782    ];
783    let mut permissions = IndexMap::new();
784    for (action, resource, effect) in definitions {
785        let entry = permissions
786            .entry(action.into())
787            .or_insert_with(|| PermissionValue::Resources(IndexMap::new()));
788        let PermissionValue::Resources(resources) = entry else {
789            unreachable!("synthetic permissions use map form")
790        };
791        resources.insert(
792            WildcardPattern::new(resource).map_err(|_| EngineError::RuntimeCompileFailed)?,
793            effect,
794        );
795    }
796    Ok(permissions)
797}

799pub(crate) fn delegation_targets(
800    permissions: &IndexMap<cookie_agent_config::PermissionKey, PermissionValue>,
801    known_agents: &BTreeMap<AgentId, ResolvedAgent>,
802) -> Vec<AgentId> {
803    let eligible_matches = |pattern: &str| -> Vec<AgentId> {
804        known_agents
805            .iter()
806            .filter(move |(id, agent)| {
807                // Only eligible subagents become targets; runtime
808                // spawn enforces the same eligibility again.
809                agent.document.frontmatter.enabled
810                    && matches!(
811                        agent.document.frontmatter.mode,
812                        cookie_agent_protocol::AgentMode::Subagent
813                            | cookie_agent_protocol::AgentMode::All
814                    )
815                    && cookie_agent_config::simple_wildcard_match(pattern, id.as_str())
816            })
817            .map(|(id, _)| id.clone())
818            .collect()
819    };
820    let mut targets = Vec::new();
821    match permissions.get(&cookie_agent_config::PermissionKey::any(
822        PermissionAction::DELEGATE,
823    )) {
824        // Scalar effects are documented as equivalent to a `"*"` mapping:
825        // expand them over every eligible subagent like any wildcard rule.
826        Some(PermissionValue::Effect(effect)) if *effect != PermissionEffect::Deny => {
827            targets.extend(eligible_matches("*"));
828        }
829        Some(PermissionValue::Resources(resources)) => {
830            for (resource, effect) in resources {
831                if *effect == PermissionEffect::Deny {
832                    continue;
833                }
834                let pattern = resource.as_str();
835                if pattern.contains('*') || pattern.contains('?') {
836                    // Wildcard delegation uses the same matching as every
837                    // other permission: expand against the known agent set.
838                    targets.extend(eligible_matches(pattern));
839                } else if let Ok(target) = AgentId::new(pattern)
840                    && known_agents.contains_key(&target)
841                {
842                    // A name no agent has yet (an ACP agent whose runtime
843                    // has not reported it) becomes a target once one does;
844                    // until then it is a configuration diagnostic.
845                    targets.push(target);
846                }
847            }
848        }
849        _ => {}
850    }
851    targets.sort();
852    targets.dedup();
853    targets
854}

856pub(crate) fn selection_available(
857    models: &CompiledModelRuntime,
858    selection: &ModelSelection,
859) -> bool {
860    models.model(&selection.model).is_some_and(|model| {
861        model.model.status == CompiledModelStatus::Available
862            && selection
863                .variant
864                .as_ref()
865                .is_none_or(|variant| model.model.variants.contains_key(variant))
866    })

1078        // expands over every eligible subagent.
