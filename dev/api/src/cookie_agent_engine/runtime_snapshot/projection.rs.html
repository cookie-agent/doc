17pub(crate) fn build_runtime_snapshot(
18    models: &CompiledModelRuntime,
19    agents: &AgentRegistry,
20    agent_presets: &BTreeMap<String, std::sync::Arc<AgentRegistry>>,
21) -> Result<protocol::RuntimeSnapshotV1, EngineError> {
22    let mut compiled_by_provider =
23        BTreeMap::<protocol::ProviderId, Vec<&cookie_agent_models::CompiledRuntimeModel>>::new();
24    for model in models.models().values() {
25        compiled_by_provider
26            .entry(model.key.provider_id())
27            .or_default()
28            .push(model);
29    }
30    let providers = models
31        .providers()
32        .iter()
33        .map(|provider| {
34            provider_descriptor(
35                models,
36                provider,
37                compiled_by_provider
38                    .get(&provider.id)
39                    .map_or(&[][..], Vec::as_slice),
40            )
41        })
42        .collect::<Result<Vec<_>, _>>()?;
43    let available_models = models
44        .models()
45        .values()
46        .filter(|model| model.model.status == CompiledModelStatus::Available)
47        .map(model_descriptor)
48        .collect::<Result<Vec<_>, _>>()?;
49    let agent_descriptors = agents
50        .descriptors()
51        .iter()
52        .cloned()
53        .chain(
54            agent_presets
55                .values()
56                .flat_map(|registry| registry.descriptors().iter().cloned()),
57        )
58        .collect::<Vec<_>>();
59    let agent_revision = revision::<protocol::AgentRevision, _>(
60        "cookie-agent/agent-runtime/v1",
61        &agent_descriptors,
62        protocol::AgentRevision::new,
63    )?;
64    let runtime_revision = runtime_revision(
65        &family_registry().revision(),
66        &models.catalog().revision,
67        &models.provider_state_revision(),
68        models.model_revision(),
69        &agent_revision,
70    )?;
71    let catalog = models.catalog();
72    let quarantine = quarantine_summary(models)?;
73    let last_error = catalog
74        .state
75        .last_error
76        .as_ref()
77        .map(|error| {
78            Ok::<_, EngineError>(protocol::CatalogSafeErrorMeta {
79                code: protocol::SafeCode::new(error.code.clone())
80                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
81                message: protocol::SafeErrorMessage::new(error.safe_message.clone())
82                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
83                time: error.occurred_at,
84            })
85        })
86        .transpose()?;
87    let snapshot = protocol::RuntimeSnapshotV1 {
88        snapshot_schema_version: protocol::RuntimeSnapshotSchemaVersion::current(),
89        recipe_registry_revision: family_registry().revision(),
90        catalog_revision: catalog.revision.clone(),
91        catalog_source: match catalog.source {
92            cookie_agent_models::catalog::CatalogSource::Network => {
93                protocol::CatalogSource::Network
94            }
95            cookie_agent_models::catalog::CatalogSource::Cache => protocol::CatalogSource::Cache,
96            cookie_agent_models::catalog::CatalogSource::Bootstrap => {
97                protocol::CatalogSource::Bootstrap
98            }
99        },
100        catalog_state: protocol::CatalogRuntimeState {
101            stale: !matches!(
102                catalog.state.availability,
103                cookie_agent_models::catalog::CatalogAvailability::Ready
104            ),
105            age: match catalog.state.age {
106                cookie_agent_models::catalog::CatalogAgeState::Current => {
107                    protocol::CatalogAge::Current
108                }
109                cookie_agent_models::catalog::CatalogAgeState::OlderThanSevenDays => {
110                    protocol::CatalogAge::OlderThanSevenDays
111                }
112                cookie_agent_models::catalog::CatalogAgeState::OlderThanThirtyDays => {
113                    protocol::CatalogAge::OlderThanThirtyDays
114                }
115            },
116            provider_quarantine_count: quarantine.provider_count,
117            model_quarantine_count: quarantine.model_count,
118            quarantine_digest: quarantine.digest,
119            last_error,
120        },
121        provider_state_revision: models.provider_state_revision(),
122        provider_store_generation: protocol::ProviderStoreGeneration::new(
123            models.store().generation().get(),
124        )
125        .map_err(|_| EngineError::RuntimeCompileFailed)?,
126        model_revision: models.model_revision().clone(),
127        agent_revision,
128        runtime_revision,
129        providers,
130        models: available_models,
131        agents: agent_descriptors,
132    };
133    snapshot
134        .validate()
135        .map_err(|_| EngineError::RuntimeCompileFailed)?;
136    Ok(snapshot)
137}

159fn quarantine_summary(
160    runtime: &CompiledModelRuntime,
161) -> Result<RuntimeQuarantineSummary, EngineError> {
162    let catalog = runtime.catalog();
163    let entries = catalog
164        .quarantine
165        .iter()
166        .map(|entry| RuntimeQuarantineEntry {
167            provider_id: entry.provider_id.clone(),
168            model_id: entry.model_id.clone(),
169            canonical_model_id: entry.canonical_model_id.clone(),
170            reason: RuntimeQuarantineReason::Parser(entry.reason.clone()),
171        })
172        .collect::<BTreeSet<_>>();
173    let provider_count = entries
174        .iter()
175        .filter(|entry| entry.model_id.is_none() && entry.canonical_model_id.is_none())
176        .count() as u32;
177    let model_count = entries.len() as u32 - provider_count;
178    let digest = protocol::Sha256Digest::new(hash_bytes(
179        "cookie-agent/runtime-quarantine/v1",
180        &serde_json::to_vec(&entries).map_err(|_| EngineError::RuntimeCompileFailed)?,
181    ))
182    .map_err(|_| EngineError::RuntimeCompileFailed)?;
183    Ok(RuntimeQuarantineSummary {
184        provider_count,
185        model_count,
186        digest,
187    })
188}

210/// Model availability for one provider: counts over every compiled or
211/// rejected catalog row, plus per-model reasons for configured providers.
212fn model_availability(
213    provider: &cookie_agent_models::CompiledProviderState,
214    record: Option<&cookie_agent_models::catalog::CatalogProviderRecord>,
215    compiled: &[&cookie_agent_models::CompiledRuntimeModel],
216) -> (
217    protocol::ProviderModelCounts,
218    Vec<protocol::UnavailableModelDescriptor>,
219) {
220    let listed = provider.authored || provider.stored;
221    let mut counts = protocol::ProviderModelCounts::default();
222    let mut unavailable = Vec::new();
223    let mut push = |id: &protocol::ProviderModelId,
224                    display_name: &str,
225                    kind: protocol::ModelUnavailableKind,
226                    reason: Option<&str>| {
227        let count = match kind {
228            protocol::ModelUnavailableKind::Quarantined => &mut counts.quarantined,
229            protocol::ModelUnavailableKind::Unsupported => &mut counts.unsupported,
230            protocol::ModelUnavailableKind::NeedsSetup => &mut counts.needs_setup,
231            protocol::ModelUnavailableKind::NeedsCredentials => &mut counts.needs_credentials,
232        };
233        *count = count.saturating_add(1);
234        if listed {
235            unavailable.push(protocol::UnavailableModelDescriptor {
236                id: id.clone(),
237                display_name: protocol::SafeDisplayText::new(display_name)
238                    .or_else(|_| protocol::SafeDisplayText::new(id.as_str()))
239                    .expect("provider model IDs are safe display text"),
240                kind,
241                reason: reason.and_then(safe_message),
242            });
243        }
244    };
245    for model in compiled {
246        let kind = match model.model.status {
247            CompiledModelStatus::Available => {
248                counts.available = counts.available.saturating_add(1);
249                continue;
250            }
251            CompiledModelStatus::SetupUnavailable => protocol::ModelUnavailableKind::NeedsSetup,
252            CompiledModelStatus::CredentialsUnavailable => {
253                protocol::ModelUnavailableKind::NeedsCredentials
254            }
255        };
256        push(&model.key.model_id(), &model.model.display_name, kind, None);
257    }
258    for model in &provider.unsupported_models {
259        let display_name = record
260            .and_then(|record| record.models.get(&model.id))
261            .and_then(|entry| entry.record.as_ref())
262            .map_or(model.id.as_str(), |record| record.name.as_str());
263        let kind = match model.kind {
264            UnsupportedModelKind::Quarantined => protocol::ModelUnavailableKind::Quarantined,
265            UnsupportedModelKind::Unsupported => protocol::ModelUnavailableKind::Unsupported,
266        };
267        push(&model.id, display_name, kind, Some(&model.reason));
268    }
269    unavailable.sort_by(|left, right| left.id.cmp(&right.id));
270    unavailable.truncate(4096);
271    (counts, unavailable)
272}

274/// Control-free, byte-bounded display text for a free-form compiler reason.
275fn safe_message(value: &str) -> Option<protocol::SafeErrorMessage> {
276    let mut text = value
277        .chars()
278        .filter(|character| !character.is_control())
279        .collect::<String>();
280    if text.len() > protocol::SafeErrorMessage::MAX_BYTES {
281        let mut end = protocol::SafeErrorMessage::MAX_BYTES;
282        while !text.is_char_boundary(end) {
283            end -= 1;
284        }
285        text.truncate(end);
286    }
287    protocol::SafeErrorMessage::new(text).ok()
288}

290fn provider_descriptor(
291    runtime: &CompiledModelRuntime,
292    provider: &cookie_agent_models::CompiledProviderState,
293    compiled: &[&cookie_agent_models::CompiledRuntimeModel],
294) -> Result<protocol::ProviderDescriptor, EngineError> {
295    let catalog_entry = runtime.catalog().provider(&provider.id);
296    let quarantined = catalog_entry.is_some_and(|entry| entry.quarantine.is_some());
297    let quarantine_code = catalog_entry
298        .and_then(|entry| entry.quarantine.as_ref())
299        .map_or_else(
300            || "invalid_catalog_provider_record".to_owned(),
301            |reason| reason.code().to_owned(),
302        );
303    let support_reason = provider
304        .support_reason
305        .as_deref()
306        .map(safe_code)
307        .transpose()?;
308    let support = if quarantined {
309        protocol::ProviderSupport {
310            state: protocol::ProviderSupportState::Quarantined,
311            reason: Some(safe_code(&quarantine_code)?),
312        }
313    } else if provider.retained_family_match == Some(RetainedFamilyMatch::SupportedRemoved) {
314        protocol::ProviderSupport {
315            state: protocol::ProviderSupportState::Supported,
316            reason: None,
317        }
318    } else if provider.retained_family_match
319        == Some(RetainedFamilyMatch::RemovedWithoutRetainedFamilyMatch)
320    {
321        protocol::ProviderSupport {
322            state: protocol::ProviderSupportState::Unsupported,
323            reason: Some(safe_code("removed_without_retained_recipe_match")?),
324        }
325    } else if let Some(reason) = support_reason {
326        protocol::ProviderSupport {
327            state: protocol::ProviderSupportState::Unsupported,
328            reason: Some(reason),
329        }
330    } else {
331        protocol::ProviderSupport {
332            state: protocol::ProviderSupportState::Supported,
333            reason: None,
334        }
335    };
336    let record = catalog_entry.and_then(|entry| entry.record.as_ref());
337    let recipe = record
338        .and_then(|record| family_registry().classify(record))
339        .or_else(|| {
340            runtime
341                .store()
342                .provider(&provider.id)
343                .and_then(|connection| {
344                    family_registry().by_npm(connection.policy.package_claim.as_str())
345                })
346        });
347    let mut setup_ids = BTreeMap::<String, bool>::new();
348    let mut add_template = |template: &str| {
349        for name in placeholders(template) {
350            let secret = placeholder_is_secret(&name);
351            setup_ids
352                .entry(setup_field_name(&name))
353                .and_modify(|value| *value |= secret)
354                .or_insert(secret);
355        }
356    };
357    if let Some(record) = record {
358        if let Some(api) = record.api.as_deref() {
359            add_template(api);
360        }
361        for model in record
362            .models
363            .values()
364            .filter_map(|entry| entry.record.as_ref())
365        {
366            if let Some(api) = model
367                .provider
368                .as_ref()
369                .and_then(|provider| provider.api.as_deref())
370            {
371                add_template(api);
372            }
373        }
374    }
375    if let Some(recipe) = recipe {
376        match recipe.family {
377            cookie_agent_models::recipes::FamilyKind::Vertex
378            | cookie_agent_models::recipes::FamilyKind::VertexAnthropic => {
379                setup_ids.insert("project".to_owned(), false);
380                setup_ids.insert("location".to_owned(), false);
381            }
382            cookie_agent_models::recipes::FamilyKind::Bedrock => {
383                setup_ids.insert("region".to_owned(), false);
384            }
385            cookie_agent_models::recipes::FamilyKind::Azure => {
386                setup_ids.insert("resource_name".to_owned(), false);
387            }
388            _ => {}
389        }
390    }
391    let mut setup_fields = setup_ids
392        .iter()
393        .map(|(id, secret)| setup_descriptor(id, *secret))
394        .collect::<Result<Vec<_>, EngineError>>()?;
395    setup_fields.sort_by(|left, right| left.id.cmp(&right.id));
396    let mut auth_methods = if let Some(recipe) = recipe {
397        recipe
398            .allowed_auth_methods
399            .iter()
400            .filter_map(|id| auth_method(id))
401            .map(auth_descriptor)
402            .collect::<Result<Vec<_>, EngineError>>()?
403    } else {
404        Vec::new()
405    };
406    auth_methods.sort_by(|left, right| left.id.cmp(&right.id));
407    let (model_counts, unavailable_models) = model_availability(provider, record, compiled);
408    let documentation_url = record
409        .map(|record| record.documentation_url.trim())
410        .filter(|url| !url.is_empty())
411        .and_then(|url| protocol::SafeDisplayText::new(url).ok());
412    let environment = record
413        .map(|record| {
414            record
415                .environment
416                .iter()
417                .filter_map(|name| protocol::CredentialFieldName::new(name.as_str()).ok())
418                .take(32)
419                .collect()
420        })
421        .unwrap_or_default();
422    Ok(protocol::ProviderDescriptor {
423        id: provider.id.clone(),
424        display_name: protocol::SafeDisplayText::new(provider.display_name.clone())
425            .map_err(|_| EngineError::RuntimeCompileFailed)?,
426        presence: match provider.presence {
427            ModelProviderPresence::Current => protocol::ProviderPresence::Current,
428            ModelProviderPresence::Removed => protocol::ProviderPresence::Removed,
429        },
430        support,
431        setup_fields,
432        auth_methods,
433        configuration: match (provider.authored, provider.stored) {
434            (false, false) => protocol::ProviderConfigurationState::Unconfigured,
435            (true, false) => protocol::ProviderConfigurationState::Authored,
436            (false, true) => protocol::ProviderConfigurationState::Stored,
437            (true, true) => protocol::ProviderConfigurationState::AuthoredAndStored,
438        },
439        effective_auth_state: effective_auth_state(provider.effective_auth),
440        durable_connection: provider
441            .durable_connection
442            .as_ref()
443            .map(project_durable_connection)
444            .transpose()?,
445        quarantine: quarantined.then(|| protocol::QuarantineDiagnostic {
446            code: safe_code(&quarantine_code).expect("validated quarantine code is valid"),
447            message: protocol::SafeErrorMessage::new("catalog provider record is quarantined")
448                .expect("static quarantine message is valid"),
449        }),
450        documentation_url,
451        environment,
452        model_counts,
453        unavailable_models,
454    })
455}

457fn placeholder_is_secret(name: &str) -> bool {
458    let name = name.to_ascii_uppercase();
459    name.contains("KEY") || name.contains("TOKEN") || name.contains("SECRET")
460}

462fn setup_descriptor(
463    field: &str,
464    secret: bool,
465) -> Result<protocol::SetupFieldDescriptor, EngineError> {

486fn auth_descriptor(
487    method: &cookie_agent_models::recipes::AuthMethodRecipe,
488) -> Result<protocol::AuthMethodDescriptor, EngineError> {
489    let mut credentials = method
490        .credentials
491        .iter()
492        .map(|field| {
493            Ok(protocol::AuthCredentialDescriptor {
494                id: protocol::AuthFieldName::new(field.name)
495                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
496                display_name: protocol::SafeDisplayText::new(field.name.replace('_', " "))
497                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
498                help: protocol::SafeDisplayText::new(format!("Secret credential `{}`", field.name))
499                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
500                required: field.required,
501                credential_type: match field.kind {
502                    CredentialKind::ApiKey => protocol::CredentialFieldType::ApiKey,
503                    CredentialKind::AccessToken => protocol::CredentialFieldType::AccessToken,
504                    CredentialKind::AccessKeyId => protocol::CredentialFieldType::AccessKeyId,
505                    CredentialKind::SecretAccessKey => {
506                        protocol::CredentialFieldType::SecretAccessKey
507                    }
508                    CredentialKind::SessionToken => protocol::CredentialFieldType::SessionToken,
509                },
510            })
511        })
512        .collect::<Result<Vec<_>, EngineError>>()?;
513    credentials.sort_by(|left, right| left.id.cmp(&right.id));
514    Ok(protocol::AuthMethodDescriptor {
515        id: protocol::AuthMethodId::new(method.id)
516            .map_err(|_| EngineError::RuntimeCompileFailed)?,
517        display_name: protocol::SafeDisplayText::new(method.id.replace('-', " "))
518            .map_err(|_| EngineError::RuntimeCompileFailed)?,
519        credentials,
520    })
521}

523fn model_descriptor(
524    model: &cookie_agent_models::CompiledRuntimeModel,
525) -> Result<protocol::AvailableModelDescriptor, EngineError> {
526    let mut capability_value = serde_json::to_value(&model.model.capabilities)
527        .map_err(|_| EngineError::RuntimeCompileFailed)?;
528    // Compaction support and the input limit are engine-internal; the engine reads the input
529    // limit from the frozen binding descriptor, so the client-facing capabilities omit both.
530    let capability_object = capability_value
531        .as_object_mut()
532        .ok_or(EngineError::RuntimeCompileFailed)?;
533    capability_object.remove("compaction");
534    capability_object.remove("input_tokens");
535    let capabilities =
536        serde_json::from_value(capability_value).map_err(|_| EngineError::RuntimeCompileFailed)?;
537    let variants = model
538        .model
539        .variants
540        .values()
541        .map(|variant| {
542            let fingerprint = protocol::Sha256Digest::new(hash(
543                "cookie-agent/model-variant/v1",
544                &(
545                    variant.id.clone(),
546                    &variant.defaults,
547                    &variant.options,
548                    &variant.reasoning,
549                ),
550            )?)
551            .map_err(|_| EngineError::RuntimeCompileFailed)?;
552            Ok(protocol::AvailableVariantDescriptor {
553                id: variant.id.clone(),
554                display_name: variant.display_name.clone(),
555                origin: match variant.origin {
556                    CompiledVariantOrigin::ModelsDevEffort => {
557                        protocol::VariantOrigin::ModelsDevEffort
558                    }
559                    CompiledVariantOrigin::ModelsDevToggle => {
560                        protocol::VariantOrigin::ModelsDevToggle
561                    }
562                    CompiledVariantOrigin::ModelsDevBudgetTokens => {
563                        protocol::VariantOrigin::ModelsDevBudgetTokens
564                    }
565                    CompiledVariantOrigin::Authored => protocol::VariantOrigin::Explicit,
566                },
567                behavior_fingerprint: fingerprint,
568            })
569        })
570        .collect::<Result<Vec<_>, EngineError>>()?;
571    Ok(protocol::AvailableModelDescriptor {
572        key: model.key.clone(),
573        display_name: model.model.display_name.clone(),
574        capabilities,
575        variants,
576        variant_order: model.model.variant_order.clone(),
577        default_variant: model.model.default_variant.clone(),
578        behavior_fingerprint: protocol::Sha256Digest::new(
579            model.model.behavior_fingerprint.as_str(),
580        )
581        .map_err(|_| EngineError::RuntimeCompileFailed)?,
582    })
583}

585pub(crate) fn project_durable_connection(
586    value: &cookie_agent_models::provider_store::DurableConnectionDescriptor,
587) -> Result<protocol::DurableConnectionDescriptor, EngineError> {
588    Ok(protocol::DurableConnectionDescriptor {
589        provider_id: value.provider_id.clone(),
590        setup_values: value
591            .setup_values
592            .iter()
593            .map(|(id, value)| {
594                let value = serde_json::from_value(
595                    serde_json::to_value(value).map_err(|_| EngineError::RuntimeCompileFailed)?,
596                )
597                .map_err(|_| EngineError::RuntimeCompileFailed)?;
598                Ok((id.clone(), value))
599            })
600            .collect::<Result<BTreeMap<_, _>, EngineError>>()?,
601        setup_fingerprint: protocol::Sha256Digest::new(value.setup_fingerprint.as_str())
602            .map_err(|_| EngineError::RuntimeCompileFailed)?,
603        recipe_fingerprint: protocol::Sha256Digest::new(value.recipe_fingerprint.as_str())
604            .map_err(|_| EngineError::RuntimeCompileFailed)?,
605        auth_method: value.auth_method.clone(),
606        credential_fields: value.credential_fields.clone(),
607        connection_generation: protocol::ProviderConnectionGeneration::new(
608            value.connection_generation.get(),
609        )
610        .map_err(|_| EngineError::RuntimeCompileFailed)?,
611        connected_at: value.connected_at,
612    })
613}

629pub(crate) fn effective_auth_source(
630    value: EffectiveCredentialSource,
631) -> Result<protocol::EffectiveAuthSource, EngineError> {
632    match value {
633        EffectiveCredentialSource::AuthoredApiKey => {
634            Ok(protocol::EffectiveAuthSource::AuthoredApiKey)
635        }
636        EffectiveCredentialSource::AuthoredOverride => {
637            Ok(protocol::EffectiveAuthSource::AuthoredOverride)
638        }
639        EffectiveCredentialSource::ProviderStore => {
640            Ok(protocol::EffectiveAuthSource::ProviderStore)
641        }
642        EffectiveCredentialSource::NoAuth => Ok(protocol::EffectiveAuthSource::NoAuth),
643        EffectiveCredentialSource::Unavailable => Err(EngineError::RuntimeCompileFailed),
644    }
645}
646
647fn safe_code(value: &str) -> Result<protocol::SafeCode, EngineError> {
648    let normalized = value
649        .chars()
650        .map(|character| {
651            if character.is_ascii_alphanumeric() || character == '_' || character == '-' {
652                character.to_ascii_lowercase()
653            } else {
654                '_'
655            }
656        })
657        .collect::<String>();
658    protocol::SafeCode::new(normalized).map_err(|_| EngineError::RuntimeCompileFailed)
659}
660
661fn revision<T, E>(
662    domain: &str,
663    value: &impl Serialize,
664    constructor: impl FnOnce(String) -> Result<T, E>,
665) -> Result<T, EngineError> {
666    constructor(format!("sha256:{}", hash(domain, value)?))
667        .map_err(|_| EngineError::RuntimeCompileFailed)
668}
669
670fn hash(domain: &str, value: &impl Serialize) -> Result<String, EngineError> {
671    let bytes = serde_json::to_vec(value).map_err(|_| EngineError::RuntimeCompileFailed)?;
672    Ok(hash_bytes(domain, &bytes))
673}
674
675fn hash_bytes(domain: &str, bytes: &[u8]) -> String {
676    let mut hasher = Sha256::new();
677    hasher.update(domain.as_bytes());
678    hasher.update([0]);
679    hasher.update(bytes);
680    format!("{:x}", hasher.finalize())
681}
