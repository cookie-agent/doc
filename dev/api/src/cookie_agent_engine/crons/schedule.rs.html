1//! Cron schedules (`docs/site/specs/cron.md` section 3): five-field cron
2//! expressions evaluated in a time zone, and one-shot delays.
3//!
4//! A field is `*`, a number, a range `a-b`, any of these with a step
5//! (`*/n`, `a-b/n`, `a/n`, which runs from `a` to the field's end), or a
6//! comma-separated list of them. Months and days of the week may also be
7//! named by their first three letters (`jan`, `mon`), in any case, and a day
8//! of the week of 7 is Sunday. Day of month and day of week combine as in
9//! Vixie cron: when both fields are restricted (neither starts with `*`), a
10//! day matches when either matches; otherwise both must.
11//!
12//! Local times that do not exist (a spring-forward gap) fire at the
13//! instant the gap jumps to, as `jiff`'s compatible disambiguation maps
14//! them; local times that occur twice (a fall-back fold) fire once, at the
15//! first occurrence.
16
17use jiff::{
18    Span, Timestamp, ToSpan,
19    civil::{Date, DateTime, Time},
20    tz::TimeZone,
21};
22
23/// The longest a one-shot delay may be, in days.
24pub(crate) const MAX_DELAY_DAYS: i64 = 365;
25/// How many days ahead a next match is searched for: past the longest gap
26/// between two February 29ths (eight years, around 2100).
27const SEARCH_DAYS: i64 = 366 * 9;
28
29/// A parsed five-field cron expression.
30#[derive(Clone, Debug, Eq, PartialEq)]
31pub(crate) struct CronExpr {
32    minutes: u64,
33    hours: u32,
34    days_of_month: u32,
35    months: u16,
36    days_of_week: u8,
37    /// Whether day of month and day of week combine with OR (both
38    /// restricted).
39    either_day: bool,
40}
41
42const MONTH_NAMES: [&str; 12] = [
43    "jan", "feb", "mar", "apr", "may", "jun", "jul", "aug", "sep", "oct", "nov", "dec",
44];
45const DAY_NAMES: [&str; 7] = ["sun", "mon", "tue", "wed", "thu", "fri", "sat"];
46
47struct Field {
48    name: &'static str,
49    min: u32,
50    max: u32,
51    names: &'static [&'static str],
52    /// The first value `names` stands for.
53    names_from: u32,
54}
55
56const FIELDS: [Field; 5] = [
57    Field {
58        name: "minute",
59        min: 0,
60        max: 59,
61        names: &[],
62        names_from: 0,
63    },
64    Field {
65        name: "hour",
66        min: 0,
67        max: 23,
68        names: &[],
69        names_from: 0,
70    },
71    Field {
72        name: "day of month",
73        min: 1,
74        max: 31,
75        names: &[],
76        names_from: 0,
77    },
78    Field {
79        name: "month",
80        min: 1,
81        max: 12,
82        names: &MONTH_NAMES,
83        names_from: 1,
84    },
85    Field {
86        name: "day of week",
87        min: 0,
88        max: 7,
89        names: &DAY_NAMES,
90        names_from: 0,
91    },
92];
93
94impl Field {
95    fn value(&self, text: &str) -> Result<u32, String> {
96        if let Some(index) = self
97            .names
98            .iter()
99            .position(|name| name.eq_ignore_ascii_case(text))
100        {
101            return Ok(self.names_from + u32::try_from(index).expect("few names"));
102        }
103        let value = text
104            .parse::<u32>()
105            .ok()
106            .filter(|_| !text.is_empty() && text.bytes().all(|byte| byte.is_ascii_digit()))
107            .ok_or_else(|| format!("`{text}` is not a valid {}", self.name))?;
108        if value < self.min || value > self.max {
109            return Err(format!(
110                "{} {value} is outside {}-{}",
111                self.name, self.min, self.max
112            ));
113        }
114        Ok(value)
115    }
116
117    /// The bit set of the values `text` names.
118    fn parse(&self, text: &str) -> Result<u64, String> {
119        let mut bits = 0_u64;
120        for item in text.split(',') {
121            let (range, step) = match item.split_once('/') {
122                Some((range, step)) => {
123                    let step = Some(step)
124                        .filter(|step| {
125                            !step.is_empty() && step.bytes().all(|byte| byte.is_ascii_digit())
126                        })
127                        .and_then(|step| step.parse::<u32>().ok())
128                        .filter(|step| *step >= 1)
129                        .ok_or_else(|| {
130                            format!("`{item}`: the step must be a whole number from 1")
131                        })?;
132                    (range, Some(step))
133                }
134                None => (item, None),
135            };
136            let (from, to) = if range == "*" {
137                (self.min, self.max)
138            } else if let Some((from, to)) = range.split_once('-') {
139                let (from, to) = (self.value(from)?, self.value(to)?);
140                if from > to {
141                    return Err(format!("`{item}`: the range runs backwards"));
142                }
143                (from, to)
144            } else {
145                let from = self.value(range)?;
146                // `a/n` runs from `a` to the field's end.
147                (from, if step.is_some() { self.max } else { from })
148            };
149            let step = step.unwrap_or(1);
150            let mut value = from;
151            while value <= to {
152                bits |= 1 << value;
153                value += step;
154            }
155        }
156        Ok(bits)
157    }
158}

160impl CronExpr {
161    /// Parses `text`, refusing an expression that never matches.
162    pub(crate) fn parse(text: &str) -> Result<Self, String> {
163        let parts = text.split_whitespace().collect::<Vec<_>>();
164        let [minute, hour, day_of_month, month, day_of_week] = parts[..] else {
165            return Err(format!(
166                "a cron expression has five fields (minute, hour, day of month, month, day of week); `{text}` has {}",
167                parts.len()
168            ));
169        };
170        let mut days_of_week = FIELDS[4].parse(day_of_week)?;
171        // 7 is Sunday too.
172        if days_of_week & (1 << 7) != 0 {
173            days_of_week = (days_of_week | 1) & !(1 << 7);
174        }
175        let expr = Self {
176            minutes: FIELDS[0].parse(minute)?,
177            hours: u32::try_from(FIELDS[1].parse(hour)?).expect("24 hour bits"),
178            days_of_month: u32::try_from(FIELDS[2].parse(day_of_month)? >> 1).expect("31 day bits"),
179            months: u16::try_from(FIELDS[3].parse(month)? >> 1).expect("12 month bits"),
180            days_of_week: u8::try_from(days_of_week).expect("7 weekday bits"),
181            either_day: !day_of_month.starts_with('*') && !day_of_week.starts_with('*'),
182        };
183        if !expr.can_match() {
184            return Err(format!("`{text}` has no next match"));
185        }
186        Ok(expr)
187    }
188
189    /// Whether some day of some month matches: always when the days of the
190    /// week count, else when a day of month exists in a month.
191    fn can_match(&self) -> bool {
192        const LONGEST: [u32; 12] = [31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
193        if self.either_day || self.days_of_month == (1 << 31) - 1 {
194            return true;
195        }
196        (0..12).any(|month| {
197            self.months & (1 << month) != 0
198                && self.days_of_month & ((1_u32 << LONGEST[month]) - 1) != 0
199        })
200    }
201
202    fn day_matches(&self, date: Date) -> bool {
203        let month = u8::try_from(date.month()).expect("month") - 1;
204        if self.months & (1 << month) == 0 {
205            return false;
206        }
207        let day_of_month = self.days_of_month & (1 << (date.day() - 1)) != 0;
208        let day_of_week = self.days_of_week & (1 << date.weekday().to_sunday_zero_offset()) != 0;
209        if self.either_day {
210            day_of_month || day_of_week
211        } else {
212            day_of_month && day_of_week
213        }
214    }
215
216    /// The first instant after `after` that matches, in `zone`.
217    pub(crate) fn next_after(&self, after: Timestamp, zone: &TimeZone) -> Option<Timestamp> {
218        let local = zone.to_datetime(after);
219        // The next whole minute after `after`, as a local time.
220        let start = local
221            .with()
222            .second(0)
223            .subsec_nanosecond(0)
224            .build()
225            .ok()?
226            .checked_add(1.minute())
227            .ok()?;
228        let mut date = start.date();
229        for day in 0..SEARCH_DAYS {
230            if self.day_matches(date) {
231                for hour in 0..24_i8 {
232                    if self.hours & (1 << hour) == 0 {
233                        continue;
234                    }
235                    for minute in 0..60_i8 {
236                        if self.minutes & (1 << minute) == 0 {
237                            continue;
238                        }
239                        let candidate =
240                            DateTime::from_parts(date, Time::constant(hour, minute, 0, 0));
241                        if day == 0 && candidate < start {
242                            continue;
243                        }
244                        let Ok(zoned) = zone.to_ambiguous_zoned(candidate).compatible() else {
245                            continue;
246                        };
247                        let instant = zoned.timestamp();
248                        if instant > after {
249                            return Some(instant);
250                        }
251                    }
252                }
253            }
254            date = date.tomorrow().ok()?;
255        }
256        None
257    }
258
259    /// Every match after `after` up to and including `until`, counted, up
260    /// to `limit`.
261    pub(crate) fn count_between(
262        &self,
263        after: Timestamp,
264        until: Timestamp,
265        zone: &TimeZone,
266        limit: u64,
267    ) -> u64 {
268        let mut count = 0;
269        let mut cursor = after;
270        while count < limit {
271            match self.next_after(cursor, zone) {
272                Some(next) if next <= until => {
273                    count += 1;
274                    cursor = next;
275                }
276                _ => break,
277            }
278        }
279        count
280    }
281}

283/// Parses a one-shot delay: a whole number with `s`, `m`, `h`, or `d`, from
284/// `1s` to `365d`.
285pub(crate) fn parse_delay(text: &str) -> Result<Span, String> {
286    let refused = || {
287        format!(
288            "`{text}` is not a delay: write a whole number with s, m, h, or d, from 1s to {MAX_DELAY_DAYS}d"
289        )
290    };
291    let unit = text.chars().last().ok_or_else(refused)?;
292    let number = &text[..text.len() - unit.len_utf8()];
293    if number.is_empty() || !number.bytes().all(|byte| byte.is_ascii_digit()) {
294        return Err(refused());
295    }
296    let value = number.parse::<i64>().map_err(|_| refused())?;
297    let seconds = match unit {
298        's' => Some(value),
299        'm' => value.checked_mul(60),
300        'h' => value.checked_mul(3600),
301        'd' => value.checked_mul(86_400),
302        _ => None,
303    }
304    .ok_or_else(refused)?;
305    if !(1..=MAX_DELAY_DAYS * 86_400).contains(&seconds) {
306        return Err(refused());
307    }
308    Ok(seconds.seconds())
309}

311/// The instant `delay` after `now`.
312pub(crate) fn delay_after(now: Timestamp, delay: Span) -> Result<Timestamp, String> {
313    now.checked_add(delay).map_err(|error| error.to_string())
