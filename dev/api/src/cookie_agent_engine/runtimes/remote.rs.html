1//! The daemon's end of the runtime protocol: runtimes that run in another
2//! task or process and are reached over a frame transport.
3//!
4//! A [`ProtocolTemplate`] connects its runtime with its
5//! [`RuntimeConnector`] and sends `runtime/initialize`, once: the
6//! [`RuntimeClient`] it gets back keeps the connection, and every capability
7//! instance opens over it (`capability/open`). Each instance's
8//! [`RemoteToolProvider`] publishes the tools the runtime declared for it and
9//! proxies preparation and execution, it routes `tool/progress`
10//! into the call's capture, and it hands `tool/detachable` to the call's
11//! detach point. When the connection is lost, calls in flight, backgrounded
12//! ones included, fail with `runtime_unavailable`.
13//! The client also keeps what the runtime reports about the MCP servers it
14//! hosts (`mcp/changed`), forwards their changes and failures to the
15//! [`McpHub`], and asks the runtime to reconnect them (`mcp/reconnect`).
16//!
17//! A runtime in a child process plugs in as another [`RuntimeConnector`]
18//! (`super::process`).
19

66/// A runtime that speaks the runtime protocol: one connection, which every
67/// capability instance of the runtime shares.
68pub(crate) struct ProtocolTemplate {
69    runtime_id: String,
70    /// The workspace a `coding` instance gets when a session names none,
71    /// such as the daemon's working directory for `local`.
72    default_workspace: Option<PathBuf>,
73    connector: Arc<dyn RuntimeConnector>,
74    settings: RuntimeSettings,
75    /// Bound on connecting and initializing, and on opening an instance;
76    /// `None` for an in-process runtime.
77    startup_timeout: Option<Duration>,
78    /// How long a stopping runtime gets to answer `runtime/shutdown`.
79    shutdown_grace: Duration,
80    /// Where the MCP servers instances host report to.

84    connecting: tokio::sync::Mutex<()>,
85    /// The current connection and what it reported.
86    connection: Mutex<Option<RuntimeClient>>,
87    /// What the runtime offered when it last connected.
88    offered: Mutex<RuntimeListingState>,

149    /// The runtime's live connection, connecting and initializing it when
150    /// there is none or it ended.
151    async fn connected(&self) -> Result<RuntimeClient, String> {
152        let current = || {
153            lock(&self.connection)
154                .clone()
155                .filter(|client| !client.is_closed())
156        };
157        if let Some(client) = current() {
158            return Ok(client);
159        }
160        let _connecting = self.connecting.lock().await;
161        if let Some(client) = current() {
162            return Ok(client);
163        }
164        let diagnostics = Mutex::new(None);
165        let attempt = async {
166            let connection = self.connector.connect().await?;
167            lock(&diagnostics).clone_from(&connection.diagnostics);
168            RuntimeClient::initialize(
169                connection,
170                &self.runtime_id,
171                &self.settings,
172                self.shutdown_grace,
173                self.mcp_hub.clone(),
174            )
175            .await
176        };
177        let result = match self.startup_timeout {
178            Some(timeout) => tokio::time::timeout(timeout, attempt)
179                .await
180                .unwrap_or_else(|_| {
181                    Err(format!(
182                        "the runtime did not start within {} ms",
183                        timeout.as_millis()
184                    ))
185                }),
186            None => attempt.await,
187        };
188        let (client, result) = result
189            .map_err(|reason| match lock(&diagnostics).as_ref() {
190                Some(tail) => tail.annotate(&reason),
191                None => reason,
192            })
193            .inspect_err(|reason| lock(&self.offered).message = Some(reason.clone()))?;
194        *lock(&self.offered) = RuntimeListingState {
195            connected: true,
196            capabilities: client.capabilities(),
197            platform: result.platform,
198            message: None,
199        };
200        *lock(&self.connection) = Some(client.clone());
201        Ok(client)
202    }
203}

206impl RuntimeTemplate for ProtocolTemplate {
207    fn runtime_id(&self) -> &str {
208        &self.runtime_id
209    }
210
211    fn lifetime(&self) -> cookie_agent_protocol::RuntimeLifetime {
212        self.lifetime
213    }
214
215    async fn capabilities(&self) -> Result<Vec<RuntimeCapabilityInfo>, String> {
216        Ok(self.connected().await?.capabilities())
217    }
218
219    fn listing(&self) -> RuntimeListingState {
220        let mut listing = lock(&self.offered).clone();
221        listing.connected = lock(&self.connection)
222            .as_ref()
223            .is_some_and(|client| !client.is_closed());
224        if listing.connected {
225            listing.message = None;
226        }
227        listing
228    }
229
230    fn default_workspace(&self, capability: &CapabilityName) -> Option<PathBuf> {
231        (capability.as_str() == CapabilityName::CODING)
232            .then(|| self.default_workspace.clone())
233            .flatten()
234    }
235
236    async fn start(
237        &self,
238        capability: &CapabilityName,
239        workspace: Option<&Path>,
240    ) -> Result<StartedRuntime, String> {
241        let client = self.connected().await?;
242        let open = client.open(capability, workspace);
243        let result = match self.startup_timeout {
244            Some(timeout) => tokio::time::timeout(timeout, open)
245                .await
246                .unwrap_or_else(|_| {
247                    Err(format!(
248                        "the runtime did not open `{capability}` within {} ms",
249                        timeout.as_millis()
250                    ))
251                }),
252            None => open.await,
253        };
254        result.map_err(|reason| client.inner.annotate(&reason))
255    }
256
257    fn disconnect(&self) {
258        lock(&self.connection).take();
259    }

262/// The settings instances of `runtime_id` get from the daemon's
263/// configuration, its template's MCP servers included.
264pub(crate) fn runtime_settings(
265    config: &cookie_agent_config::LoadedConfiguration,
266    runtime_id: &str,
267) -> RuntimeSettings {
268    let limits = crate::preview::PreviewLimits::from(&config.runtime.tool_output);
269    RuntimeSettings {
270        tool_output: protocol::ToolOutputLimits {
271            max_lines: limits.max_lines as u64,
272            max_bytes: limits.max_bytes as u64,
273            head_lines: limits.head_lines as u64,
274            tail_lines: limits.tail_lines as u64,
275        },
276        agent_md: protocol::AgentMdLimits {
277            max_bytes: cookie_agent_runtime_api::context::AGENT_MD_MAX_BYTES,
278        },
279        mcp_servers: config
280            .runtimes
281            .iter()
282            .find(|(id, _)| id.as_str() == runtime_id)
283            .map(|(_, template)| {
284                template
285                    .mcp_servers
286                    .iter()
287                    .map(|(name, server)| protocol::McpServerSettings {
288                        name: name.clone(),
289                        config: serde_json::to_value(server)
290                            .expect("MCP server configuration serializes"),
291                    })
292                    .collect()
293            })
294            .unwrap_or_default(),
295    }
296}

298/// Registers a protocol template per configured `[runtimes.<id>]` other
299/// than `local`, each starting its instances with a [`ProcessConnector`].
300pub(crate) fn register_configured_templates(
301    registry: &RuntimeRegistry,
302    config: &cookie_agent_config::LoadedConfiguration,
303    mcp_hub: &Arc<McpHub>,
304) -> Result<(), EngineError> {
305    for (runtime_id, template) in &config.runtimes {
306        if runtime_id.is_reserved() {
307            continue;
308        }
309        registry.register_template(Arc::new(
310            ProtocolTemplate::new(
311                runtime_id.as_str(),
312                Arc::new(ProcessConnector::new(
313                    template.command.clone(),
314                    template.env.clone(),
315                    template.shutdown_grace(),
316                )),
317                runtime_settings(config, runtime_id.as_str()),
318                template.startup_timeout(),
319                template.shutdown_grace(),
320            )
321            .with_mcp_hub(Arc::clone(mcp_hub)),
322        ))?;
323    }
324    Ok(())
325}

451struct ClientInner {
452    peer: Peer,
453    runtime_id: String,
454    /// This connection's producer epoch; see [`NEXT_CONNECTION_EPOCH`].
455    epoch: u64,
456    /// The producer source of the connection, once initialized:
457    /// `runtime:<runtime ID>`, or `plugin:<name>` for a plugin.
458    producer_source: OnceLock<cookie_agent_protocol::ProducerSource>,
459    /// Handles what the runtime protocol does not define, for a plugin.
460    extension: OnceLock<Arc<dyn ClientExtension>>,
461    producers: Mutex<ProducerState>,
462    /// What the runtime writes outside the protocol, such as its stderr.
463    diagnostics: Option<DiagnosticTail>,
464    /// How long [`RuntimeClient::shutdown`] waits for the runtime.
465    shutdown_grace: Duration,
466    /// What the runtime offers, from its initialize reply.
467    capabilities: OnceLock<Vec<RuntimeCapabilityInfo>>,
468    /// The runtime's operating system, from its initialize reply.
469    platform: Mutex<Option<String>>,
470    /// Each instance's declared tools, by instance handle (`None` for a
471    /// plugin's implicit instance): from its open reply or its latest
472    /// `tools_changed`.
473    tools: Mutex<HashMap<Option<String>, Arc<DeclaredTools>>>,
474    /// Executing calls by prepared ID: where their progress and detach
475    /// point reach the host.
476    executions: Mutex<HashMap<u64, ExecutionHost>>,
477    /// Presentations of calls that failed to prepare, by call.
478    presentations: Mutex<VecDeque<(ToolCallId, ToolCallPresentation)>>,
479    /// The MCP servers the runtime hosts.
480    mcp: Mutex<HostedMcp>,
481    mcp_hub: Option<Arc<McpHub>>,
482    /// `mcp/reconnect` retries in flight; a run waits for them.
483    mcp_retries: AtomicUsize,
484    mcp_settled: Notify,

566impl RuntimeClient {
567    /// Connects over `transport` and initializes the runtime. The error is
568    /// a human-readable reason.
569    pub(crate) async fn initialize(
570        connection: RuntimeConnection,
571        runtime_id: &str,
572        settings: &RuntimeSettings,
573        shutdown_grace: Duration,
574        mcp_hub: Option<Arc<McpHub>>,
575    ) -> Result<(Self, InitializeResult), String> {
576        let (peer, incoming) = Peer::start(connection.transport);
577        let definitions = settings
578            .mcp_servers
579            .iter()
580            .filter_map(|server| {
581                serde_json::from_value(server.config.clone())
582                    .ok()
583                    .map(|config| (server.name.clone(), config))
584            })
585            .collect();
586        let inner = Arc::new(ClientInner {
587            peer,
588            runtime_id: runtime_id.to_owned(),
589            epoch: NEXT_CONNECTION_EPOCH.fetch_add(1, Ordering::Relaxed),
590            producer_source: OnceLock::new(),
591            extension: OnceLock::new(),
592            producers: Mutex::default(),
593            diagnostics: connection.diagnostics,
594            shutdown_grace,
595            capabilities: OnceLock::new(),
596            platform: Mutex::default(),
597            tools: Mutex::default(),
598            executions: Mutex::default(),
599            presentations: Mutex::default(),
600            mcp: Mutex::new(HostedMcp {
601                definitions,
602                ..HostedMcp::default()
603            }),
604            mcp_hub,
605            mcp_retries: AtomicUsize::new(0),
606            mcp_settled: Notify::new(),
607        });
608        tokio::spawn(handle_incoming(Arc::downgrade(&inner), incoming));
609        let result: InitializeResult = inner
610            .peer
611            .request(
612                method::INITIALIZE,
613                &InitializeParams {
614                    version: RUNTIME_PROTOCOL_VERSION,
615                    runtime_id: runtime_id.to_owned(),
616                    settings: settings.clone(),
617                },
618            )
619            .await
620            .map_err(|error| error.to_string())?;
621        if result.version != RUNTIME_PROTOCOL_VERSION {
622            return Err(format!(
623                "runtime protocol version mismatch: the daemon speaks version \
624                 {RUNTIME_PROTOCOL_VERSION}, the runtime speaks version {}",
625                result.version
626            ));
627        }
628        if let Ok(source) = cookie_agent_protocol::ProducerSource::runtime(runtime_id) {
629            let _ = inner.producer_source.set(source);
630        }
631        lock(&inner.producers).declared = result.producer_messaging;
632        let mut capabilities = Vec::with_capacity(result.capabilities.len());
633        for declared in &result.capabilities {
634            if capabilities
635                .iter()
636                .any(|known: &RuntimeCapabilityInfo| known.name == declared.name)
637            {
638                return Err(format!(
639                    "the runtime offers capability `{}` twice",
640                    declared.name
641                ));
642            }
643            capabilities.push(RuntimeCapabilityInfo {
644                name: declared.name.clone(),
645                kind: declared.kind.clone(),
646                takes_workspace: declared.takes_workspace,
647                description: declared.description.clone(),
648            });
649        }
650        let _ = inner.capabilities.set(capabilities);
651        lock(&inner.platform).clone_from(&result.platform);
652        Ok((Self { inner }, result))
653    }
654
655    /// Connects a plugin over `connection` and runs its `plugin/initialize`
656    /// handshake. The plugin's tools then run over the runtime protocol's
657    /// `tool/*` messages and its producers as `plugin:<name>`; what only
658    /// plugins send goes to `extension`. `producer_messaging` is whether its
659    /// configuration lets it produce.
660    pub(crate) async fn initialize_plugin(
661        connection: RuntimeConnection,
662        name: &str,
663        producer_messaging: bool,
664        shutdown_grace: Duration,
665        extension: Arc<dyn ClientExtension>,
666    ) -> Result<(Self, cookie_agent_protocol::ExtensionInitializeResult), String> {
667        let (peer, incoming) = Peer::start(connection.transport);
668        let source = cookie_agent_protocol::ProducerSource::plugin(name)
669            .map_err(|error| format!("invalid plugin name: {error}"))?;
670        let inner = Arc::new(ClientInner {
671            peer,
672            runtime_id: name.to_owned(),
673            epoch: NEXT_CONNECTION_EPOCH.fetch_add(1, Ordering::Relaxed),
674            producer_source: OnceLock::from(source),
675            extension: OnceLock::from(extension),
676            producers: Mutex::default(),
677            diagnostics: connection.diagnostics,
678            shutdown_grace,
679            capabilities: OnceLock::from(Vec::new()),
680            platform: Mutex::default(),
681            tools: Mutex::default(),
682            executions: Mutex::default(),
683            presentations: Mutex::default(),
684            mcp: Mutex::default(),
685            mcp_hub: None,
686            mcp_retries: AtomicUsize::new(0),
687            mcp_settled: Notify::new(),
688        });
689        tokio::spawn(handle_incoming(Arc::downgrade(&inner), incoming));
690        let client = Self { inner };
691        let result: cookie_agent_protocol::ExtensionInitializeResult = client
692            .inner
693            .peer
694            .request(
695                cookie_agent_protocol::PLUGIN_INITIALIZE_METHOD,
696                &cookie_agent_protocol::ExtensionInitializeParams {
697                    protocol_version: cookie_agent_protocol::ExtensionProtocolVersion::current(),
698                    engine_version: env!("CARGO_PKG_VERSION").to_owned(),
699                    capabilities: cookie_agent_protocol::ExtensionEngineCapabilities {
700                        producer_messaging,
701                    },
702                },
703            )
704            .await
705            .map_err(|error| client.inner.annotate(&format!("handshake failed: {error}")))?;
706        if result.name != name {
707            return Err(format!(
708                "plugin name mismatch: configured `{name}`, reported `{}`",
709                result.name
710            ));
711        }
712        lock(&client.inner.tools)
713            .entry(None)
714            .or_insert_with(|| Arc::new(result.tools.clone().into()));
715        lock(&client.inner.producers).declared =
716            producer_messaging && result.capabilities.producer_messaging;
717        Ok((client, result))
718    }

751    /// Opens `capability` for `workspace` and returns the instance. The
752    /// error is a human-readable reason.
753    pub(crate) async fn open(
754        &self,
755        capability: &CapabilityName,
756        workspace: Option<&Path>,
757    ) -> Result<StartedRuntime, String> {
758        let offered = self
759            .capabilities()
760            .into_iter()
761            .find(|offered| &offered.name == capability)
762            .ok_or_else(|| format!("the runtime offers no capability `{capability}`"))?;
763        let result: CapabilityOpenResult = self
764            .inner
765            .peer
766            .request(
767                method::CAPABILITY_OPEN,
768                &CapabilityOpenParams {
769                    capability: capability.clone(),
770                    workspace: workspace.map(|path| path.to_string_lossy().into_owned()),
771                },
772            )
773            .await
774            .map_err(|error| error.to_string())?;
775        let attachment_name = cookie_agent_protocol::AttachmentName::new(
776            result.attachment_name.as_str(),
777        )
778        .map_err(|_| {
779            format!(
780                "the runtime reported invalid attachment name `{}`",
781                result.attachment_name
782            )
783        })?;
784        let skills = skill_registry(result.skills)?;
785        let platform = self.platform();
786        let workspace = result
787            .workspace
788            .clone()
789            .map(|root| WorkspaceContext::reported(PathBuf::from(root), platform));
790        // A notification the runtime sent after its reply may already have
791        // been handled; it is newer than the reply.
792        lock(&self.inner.tools)
793            .entry(Some(result.instance.clone()))
794            .or_insert_with(|| Arc::new(result.tools.into()));
795        let failed = {
796            let mut mcp = lock(&self.inner.mcp);
797            let instance = mcp.instances.entry(result.instance.clone()).or_default();
798            instance.workspace.clone_from(&result.workspace);
799            instance
800                .servers
801                .get_or_insert_with(|| result.mcp_servers.clone())
802                .iter()
803                .filter(|server| server.state == protocol::McpServerState::Failed)
804                .map(|server| {
805                    (
806                        server.name.clone(),
807                        server.message.clone().unwrap_or_default(),
808                    )
809                })
810                .collect::<Vec<_>>()
811        };
812        if let Some(hub) = &self.inner.mcp_hub {
813            for (server, reason) in failed {
814                hub.failed(self.inner.mcp_failure(&result.instance, server, reason));
815            }
816            if !result.mcp_servers.is_empty() {
817                hub.changed();
818            }
819        }
820        let info = RuntimeInfo::new(
821            &self.inner.runtime_id,
822            capability.clone(),
823            offered.kind,
824            attachment_name.as_str(),
825            workspace,
826        );
827        Ok(StartedRuntime {
828            info,
829            providers: vec![Arc::new(RemoteToolProvider {
830                client: self.clone(),
831                instance: Some(result.instance.clone()),
832            })],
833            skills,
834            connection: Some(InstanceConnection {
835                client: self.clone(),
836                instance: Some(result.instance),
837            }),
838        })
839    }

877    /// The runtime's producer recovery state; `None` when it does not
878    /// produce. A closed connection is `starting`: its work is unknown
879    /// until the instance reconnects and recovers.
880    pub(crate) fn producer_status(&self) -> Option<ProducerSourceStatus> {
881        let state = lock(&self.inner.producers);
882        if !state.declared {
883            return None;
884        }
885        if self.inner.peer.is_closed() {
886            return Some(ProducerSourceStatus::Starting);
887        }

889    }

892    /// its requests go to `host` from now on, and the runtime is told to

953    /// Asks the runtime to retry the failed MCP servers of instance
954    /// `instance`, as when a session that uses it is created or opened;
955    /// [`Self::mcp_settled`] waits for the attempts.
956    fn retry_failed_mcp(&self, instance: &str) {
957        let any_failed = lock(&self.inner.mcp)
958            .instances
959            .get(instance)
960            .and_then(|hosted| hosted.servers.as_ref())
961            .into_iter()
962            .flatten()
963            .any(|server| server.state == protocol::McpServerState::Failed);
964        if !any_failed {
965            return;
966        }
967        let Ok(handle) = tokio::runtime::Handle::try_current() else {
968            return;
969        };
970        self.inner.mcp_retries.fetch_add(1, Ordering::AcqRel);
971        let retry = McpRetry(Arc::clone(&self.inner));
972        let params = McpReconnectParams {
973            instance: instance.to_owned(),
974            server: None,
975        };
976        handle.spawn(async move {
977            let _ = retry
978                .0
979                .peer
980                .request::<_, Outcome>(method::MCP_RECONNECT, &params)
981                .await;
982            drop(retry);
983        });
984    }

986    /// Waits until every retry [`Self::retry_failed_mcp`] started has
987    /// finished, or the connection ended.
988    pub(crate) async fn mcp_settled(&self) {
989        while self.inner.mcp_retries.load(Ordering::Acquire) != 0 {
990            let notified = self.inner.mcp_settled.notified();
991            if self.inner.mcp_retries.load(Ordering::Acquire) == 0 {
992                break;
993            }
994            notified.await;
995        }
996    }

1033        &self,
1034        instance: &str,
1035        preset: Option<&str>,

1067                }),
1068                None => Err(EngineError::Runtime(message)),
1069            },
1070            Err(error) => Err(EngineError::Runtime(format!(
1071                "runtime `{}` context: {error}",
1072                self.inner.runtime_id
1073            ))),
1074        }
1075    }

1111async fn handle_incoming(
1112    client: Weak<ClientInner>,
1113    mut incoming: tokio::sync::mpsc::Receiver<Incoming>,
1114) {
1115    while let Some(message) = incoming.recv().await {
1116        let Some(client) = client.upgrade() else {
1117            return;
1118        };
1119        match message {
1120            Incoming::Request(request) => {
1121                tokio::spawn(async move { client.handle_request(request).await });
1122            }
1123            Incoming::Notification(notification) => {
1124                if !client.handle_notification(&notification)
1125                    && let Some(extension) = client.extension.get().cloned()
1126                {
1127                    extension.notification(
1128                        RuntimeClient {
1129                            inner: Arc::clone(&client),
1130                        },
1131                        notification,
1132                    );
1133                }
1134            }
1135        }
1136    }
1137    if let Some(client) = client.upgrade() {
1138        client.connection_lost();
1139    }
1140}

1173    async fn handle_request(self: Arc<Self>, request: IncomingRequest) {
1174        let result = match request.method.as_str() {
1175            method::TOOL_PROGRESS => self.call_progress(&request).await.and_then(|outcome| {
1176                serde_json::to_value(outcome)
1177                    .map_err(|error| RpcError::remote(-32603, error.to_string()))
1178            }),
1179            method::PRODUCER_REGISTER
1180            | method::PRODUCER_SEND
1181            | method::PRODUCER_UNREGISTER
1182            | method::PRODUCER_DISCARD => self.call_producer(&request).await,
1183            method::PRODUCER_RECOVERY_COMPLETE => self
1184                .complete_producer_recovery(&request)
1185                .map(|()| serde_json::json!({})),
1186            other => Err(RpcError::remote(
1187                error_code::METHOD_NOT_FOUND,
1188                format!("unknown method `{other}`"),
1189            )),
1190        };
1191        let _ = match result {
1192            Ok(outcome) => request.respond(&outcome).await,
1193            Err(error) => request.fail(error).await,
1194        };
1195    }

1197    /// Handles a notification the runtime protocol defines; returns whether
1198    /// it was one.
1199    fn handle_notification(&self, notification: &IncomingNotification) -> bool {
1200        match notification.method.as_str() {
1201            // A plugin's implicit instance.
1202            method::TOOLS_CHANGED => {
1203                if let Ok(tools) = notification.params::<RuntimeTools>() {
1204                    lock(&self.tools).insert(None, Arc::new(tools.into()));
1205                }
1206            }
1207            method::CAPABILITY_TOOLS_CHANGED => {
1208                if let Ok(CapabilityToolsChangedParams { instance, tools }) = notification.params()
1209                {
1210                    lock(&self.tools).insert(Some(instance), Arc::new(tools.into()));
1211                }
1212            }
1213            method::MCP_CHANGED => {
1214                if let Ok(McpChangedParams {
1215                    instance,
1216                    servers,
1217                    failures,
1218                }) = notification.params()
1219                {
1220                    lock(&self.mcp)
1221                        .instances
1222                        .entry(instance.clone())
1223                        .or_default()
1224                        .servers = Some(servers);
1225                    if let Some(hub) = &self.mcp_hub {
1226                        hub.changed();
1227                        for failure in failures {
1228                            hub.failed(self.mcp_failure(&instance, failure.server, failure.reason));
1229                        }
1230                    }
1231                }
1232            }
1233            method::TOOL_DETACHABLE => {
1234                if let Ok(DetachableParams {
1235                    prepared_id,
1236                    note,
1237                    reports_itself,
1238                }) = notification.params()
1239                    && let Some(host) = lock(&self.executions).get(&prepared_id)
1240                {
1241                    host.detach
1242                        .reached_with(cookie_agent_runtime_api::Detachable {
1243                            note,
1244                            reports_itself,
1245                        });
1246                }
1247            }
1248            _ => return false,
1249        }
1250        true
1251    }

1253    /// Hands a producer operation to the engine once it opened producer
1254    /// messaging on this connection.
1255    async fn call_producer(
1256        &self,
1257        request: &IncomingRequest,
1258    ) -> Result<serde_json::Value, RpcError> {
1259        let host = lock(&self.producers).host.clone().ok_or_else(|| {
1260            RpcError::remote(
1261                error_code::PRODUCER_REJECTED,
1262                "producer messaging has not started on this connection",
1263            )
1264        })?;
1265        let source = self.producer_source.get().cloned().ok_or_else(|| {
1266            RpcError::remote(
1267                error_code::NOT_INITIALIZED,
1268                "the runtime is not initialized",
1269            )
1270        })?;
1271        (host.call)(RuntimeProducerCall {
1272            source,
1273            epoch: self.epoch,
1274            method: request.method.clone(),
1275            params: request.params()?,
1276        })
1277        .await
1278        .map_err(|message| RpcError::remote(error_code::PRODUCER_REJECTED, message))
1279    }
1280
1281    /// Records the runtime's recovery outcome. A repeat of the same outcome
1282    /// is accepted; a different one is not.
1283    fn complete_producer_recovery(&self, request: &IncomingRequest) -> Result<(), RpcError> {
1284        let RecoveryCompleteParams { outcome } = request.params()?;
1285        let host = {
1286            let mut state = lock(&self.producers);
1287            let Some(host) = state.host.clone() else {
1288                return Err(RpcError::remote(
1289                    error_code::PRODUCER_REJECTED,
1290                    "producer recovery has not started on this connection",
1291                ));
1292            };
1293            if let Some(previous) = &state.outcome {
1294                return if *previous == outcome {
1295                    Ok(())
1296                } else {
1297                    Err(RpcError::remote(
1298                        error_code::PRODUCER_REJECTED,
1299                        "recovery outcome already reported",
1300                    ))
1301                };
1302            }
1303            state.status = Some(match outcome {
1304                RecoveryOutcome::Ready => ProducerSourceStatus::Ready,
1305                RecoveryOutcome::Failed { .. } => ProducerSourceStatus::Failed,
1306            });
1307            state.outcome = Some(outcome);
1308            host
1309        };
1310        (host.changed)();
1311        Ok(())
1312    }

1314    async fn call_progress(&self, request: &IncomingRequest) -> Result<Outcome, RpcError> {
1315        let ToolProgressParams {
1316            prepared_id,
1317            progress,
1318        } = request.params()?;
1319        let sink = lock(&self.executions)
1320            .get(&prepared_id)
1321            .map(|host| host.progress.clone());
1322        let Some(sink) = sink else {
1323            return Ok(Outcome::of(Err(ToolError::ProgressSinkClosed)));
1324        };
1325        Ok(Outcome::of(sink.send(progress.into()).await))
1326    }

1328    /// The connection ended: calls fail through their pending replies, and
1329    /// its producer registrations no longer pass for live.
1330    fn connection_lost(&self) {
1331        lock(&self.executions).clear();
1332        let host = lock(&self.producers).host.clone();
1333        if let Some(host) = host {
1334            (host.changed)();
1335        }
1336    }
1337
1338    fn remember_presentation(&self, call: ToolCallId, presentation: ToolCallPresentation) {
1339        let mut presentations = lock(&self.presentations);
1340        if presentations.len() == MAX_PENDING_PRESENTATIONS {
1341            presentations.pop_front();
1342        }
1343        presentations.push_back((call, presentation));
1344    }
1345
1346    fn take_presentation(&self, call: ToolCallId) -> Option<ToolCallPresentation> {
1347        let mut presentations = lock(&self.presentations);
1348        let index = presentations.iter().position(|(id, _)| *id == call)?;
1349        presentations
1350            .remove(index)
1351            .map(|(_, presentation)| presentation)
1352    }
1353
1354    async fn execute(
1355        self: &Arc<Self>,
1356        prepared_id: u64,
1357        context: ToolExecutionContext,
1358    ) -> Result<ToolCompletion, ToolError> {
1359        let ToolExecutionContext {
1360            session,
1361            run,
1362            progress,
1363            cancellation,
1364            turn_context,
1365            artifacts,
1366            detach,
1367            background,
1368            producers: _,
1369        } = context;
1370        let params = ExecuteParams {
1371            prepared_id,
1372            session,
1373            run,
1374            turn: turn_context.as_ref().into(),
1375            background,
1376        };
1377        lock(&self.executions).insert(prepared_id, ExecutionHost { progress, detach });
1378        let mut guard = ExecutionGuard {
1379            client: self,
1380            prepared_id,
1381            finished: false,
1382        };
1383        let request = self
1384            .peer
1385            .request::<_, ExecuteResult>(method::TOOL_EXECUTE, &params);
1386        tokio::pin!(request);
1387        let mut cancel_sent = false;
1388        let reply = loop {
1389            tokio::select! {
1390                reply = &mut request => break reply,
1391                () = cancellation.cancelled(), if !cancel_sent => {
1392                    cancel_sent = true;
1393                    let _ = self.peer.notify(method::TOOL_CANCEL, &PreparedRef { prepared_id }).await;
1394                }
1395            }
1396        };
1397        guard.finished = true;
1398        drop(guard);
1399        let reply = reply.map_err(|error| self.unavailable(&error))?;
1400        let completion = match reply.outcome {
1401            ExecuteOutcome::Completed {
1402                output,
1403                result,
1404                failed,
1405            } => ToolCompletion {
1406                output,
1407                result,
1408                failed,
1409            },
1410            ExecuteOutcome::Failed { error } => return Err(error.into()),
1411        };
1412        retain_media(artifacts.as_ref(), session, reply.media, &completion)?;
1413        Ok(completion)
1414    }
1415}

1417/// Unregisters an execution; one dropped before its reply also cancels the
1418/// call in the runtime, unless the call reported itself at its detach point.
1419struct ExecutionGuard<'a> {
1420    client: &'a Arc<ClientInner>,
1421    prepared_id: u64,
1422    finished: bool,
1423}
1424
1425impl Drop for ExecutionGuard<'_> {
1426    fn drop(&mut self) {
1427        let host = lock(&self.client.executions).remove(&self.prepared_id);
1428        // Work that reports itself outlives the call that waited for it.
1429        let reports_itself = host
1430            .and_then(|host| host.detach.current())
1431            .is_some_and(|detachable| detachable.reports_itself);
1432        if !self.finished && !reports_itself {
1433            self.client.peer.notify_detached(
1434                method::TOOL_CANCEL,
1435                &PreparedRef {
1436                    prepared_id: self.prepared_id,
1437                },
1438            );
1439        }
1440    }
1441}

1443/// Stores the media a reply carried in the session's artifact store; the
1444/// result's attachments must all be among it.
1445fn retain_media(
1446    artifacts: &dyn ToolArtifacts,
1447    session: SessionId,
1448    media: Vec<InlineMedia>,
1449    completion: &ToolCompletion,
1450) -> Result<(), ToolError> {
1451    let mut total = 0_u64;
1452    let mut retained = Vec::with_capacity(media.len());
1453    for blob in media {
1454        let bytes = base64::engine::general_purpose::STANDARD
1455            .decode(&blob.data)
1456            .map_err(|error| {
1457                ToolError::execution(format!("invalid media from runtime: {error}"))
1458            })?;
1459        total += bytes.len() as u64;
1460        if total > protocol::MAX_INLINE_MEDIA_BYTES {
1461            return Err(ToolError::resource_limit(format!(
1462                "a runtime result carries at most {} bytes of media",
1463                protocol::MAX_INLINE_MEDIA_BYTES
1464            )));
1465        }
1466        if cookie_agent_protocol::Sha256Digest::of_bytes(&bytes).as_str() != blob.sha256 {
1467            return Err(ToolError::execution(
1468                "media from the runtime does not match its digest",
1469            ));
1470        }
1471        let (reference, _) = artifacts.retain(session, &bytes)?;
1472        if reference != protocol::artifact_reference(&blob.sha256) {
1473            return Err(ToolError::execution(
1474                "the artifact store gave runtime media an unexpected reference",
1475            ));
1476        }
1477        retained.push(reference);
1478    }
1479    if let Some(attachment) = completion
1480        .result
1481        .all_attachments()
1482        .find(|attachment| !retained.contains(&attachment.reference))
1483    {
1484        return Err(ToolError::execution(format!(
1485            "the runtime's result references media it did not send ({})",
1486            attachment.reference.uri
1487        )));
1488    }
1489    Ok(())
1490}

1492/// The tools a protocol runtime declared, as one provider. The sync
1493/// metadata methods are answered from the declarations and from what the
1494/// runtime reports when it prepares a call.
1495pub(crate) struct RemoteToolProvider {
1496    client: RuntimeClient,
1497    /// The instance whose tools these are; `None` for a plugin.
1498    instance: Option<String>,
1499}
1500
1501impl RemoteToolProvider {
1502    fn declared(&self) -> Option<Arc<DeclaredTools>> {
1503        lock(&self.client.inner.tools).get(&self.instance).cloned()
1504    }
1505}
1506
1507impl RemoteToolProvider {
1508    fn prepared_tool(&self, call: PreparedCall) -> Result<PreparedTool, ToolError> {
1509        let serialization_key = call
1510            .serialization_key
1511            .as_deref()
1512            .map(|key| {
1513                protocol::hex_decode(key)
1514                    .map(PreparedSerializationKey::new)
1515                    .ok_or_else(|| ToolError::execution("invalid serialization key from runtime"))
1516            })
1517            .transpose()?;
1518        let executor = Box::new(RemoteExecutor {
1519            client: Arc::clone(&self.client.inner),
1520            prepared_id: call.prepared_id,
1521            executed: false,
1522        });
1523        Ok(PreparedTool::new(
1524            call.operation,
1525            call.normalized_arguments,
1526            serialization_key,
1527            executor,
1528        )?
1529        .with_reported_policy_labels(call.policy_labels)?
1530        .with_declaration(PreparedDeclaration {
1531            permission_name: call
1532                .permission
1533                .map(|permission| permission.as_str().to_owned()),
1534            permission_resource: call.permission_resource,
1535            presentation: call.presentation,
1536        }))
1537    }
1538}

1596    async fn prepare(
1597        &self,
1598        ctx: ToolPreparationContext,
1599        call: ToolCall,
1600    ) -> Result<PreparedTool, ToolError> {
1601        self.prepare_parallel(ctx, vec![call])
1602            .await
1603            .pop()
1604            .expect("one result per call")
1605    }
1606
1607    async fn prepare_parallel(
1608        &self,
1609        ctx: ToolPreparationContext,
1610        calls: Vec<ToolCall>,
1611    ) -> Vec<Result<PreparedTool, ToolError>> {
1612        let client = &self.client.inner;
1613        let mut prepared = Vec::with_capacity(calls.len());
1614        let turn: protocol::WireTurnContext = ctx.turn_context.as_ref().into();
1615        for batch in calls.chunks(protocol::MAX_PREPARE_BATCH) {
1616            let reply: Result<PrepareResult, RpcError> = client
1617                .peer
1618                .request(
1619                    method::TOOL_PREPARE,
1620                    &PrepareParams {
1621                        instance: self.instance.clone(),
1622                        session: ctx.session,
1623                        run: ctx.run,
1624                        turn: turn.clone(),
1625                        calls: batch
1626                            .iter()
1627                            .map(|call| WireToolCall {
1628                                id: call.id,
1629                                name: call.name.clone(),
1630                                arguments: call.arguments.clone(),
1631                            })
1632                            .collect(),
1633                    },
1634                )
1635                .await;
1636            let outcomes = match reply {
1637                Ok(reply) if reply.results.len() == batch.len() => reply.results,
1638                Ok(reply) => {
1639                    // Release whatever the runtime prepared before failing.
1640                    for outcome in reply.results {
1641                        if let PrepareOutcome::Prepared(call) = outcome {
1642                            client.peer.notify_detached(
1643                                method::TOOL_RELEASE,
1644                                &PreparedRef {
1645                                    prepared_id: call.prepared_id,
1646                                },
1647                            );
1648                        }
1649                    }
1650                    prepared.extend(batch.iter().map(|_| {
1651                        Err(ToolError::execution(
1652                            "the runtime answered a different number of calls",
1653                        ))
1654                    }));
1655                    continue;
1656                }
1657                Err(error) => {
1658                    prepared.extend(batch.iter().map(|_| Err(client.unavailable(&error))));
1659                    continue;
1660                }
1661            };
1662            for (call, outcome) in batch.iter().zip(outcomes) {
1663                prepared.push(match outcome {
1664                    PrepareOutcome::Prepared(declared) => self.prepared_tool(declared),
1665                    PrepareOutcome::Failed {
1666                        error,
1667                        presentation,
1668                    } => {
1669                        client.remember_presentation(call.id, presentation);
1670                        Err(error.into())
1671                    }
1672                });
1673            }
1674        }
1675        prepared
1676    }
1677}

1679/// Executes, revalidates, or releases one call the runtime prepared.
1680struct RemoteExecutor {
1681    client: Arc<ClientInner>,
1682    prepared_id: u64,
1683    executed: bool,
1684}
1685
1686impl Drop for RemoteExecutor {
1687    fn drop(&mut self) {
1688        if !self.executed {
1689            self.client.peer.notify_detached(
1690                method::TOOL_RELEASE,
1691                &PreparedRef {
1692                    prepared_id: self.prepared_id,
1693                },
1694            );
1695        }
1696    }
1697}
1698
1699#[async_trait]
1700impl PreparedExecutor for RemoteExecutor {
1701    async fn revalidate(&self) -> Result<(), ToolError> {
1702        let outcome: Outcome = self
1703            .client
1704            .peer
1705            .request(
1706                method::TOOL_REVALIDATE,
1707                &PreparedRef {
1708                    prepared_id: self.prepared_id,
1709                },
1710            )
1711            .await
1712            .map_err(|error| self.client.unavailable(&error))?;
1713        outcome.into_result()
1714    }
1715
1716    async fn execute(
1717        mut self: Box<Self>,
1718        context: ToolExecutionContext,
1719    ) -> Result<ToolCompletion, ToolError> {
1720        self.executed = true;
1721        let client = Arc::clone(&self.client);
1722        client.execute(self.prepared_id, context).await
1723    }
1724}
