1//! Dial-in runtimes (`docs/site/specs/runtime-capabilities.md` section 5):
2//! runtimes that connect to the daemon on their own, persistently or for a
3//! client's lifetime, instead of the daemon starting them.
4//!
5//! A [`DialInTemplate`] exists from a runtime's first connection on and
6//! keeps its ID. It has a connection only while the runtime is connected; a
7//! runtime that dials in again under the same ID takes over the template, and
8//! the daemon reopens the instances its sessions use over the new
9//! connection.
10

29/// How long a dial-in runtime gets to answer `runtime/initialize` and each
30/// `capability/open`.
31pub(crate) const DIAL_IN_TIMEOUT: Duration = Duration::from_secs(30);
32
33/// How long a stopping dial-in runtime gets to answer `runtime/shutdown`.
34const DIAL_IN_SHUTDOWN_GRACE: Duration = Duration::from_secs(3);
35
36/// Why a session cannot use dial-in runtime `runtime_id` now: it is not
37/// connected, and was last heard from at `last_seen` when that is known.
38pub(crate) fn not_connected(runtime_id: &str, last_seen: Option<jiff::Timestamp>) -> String {
39    let last_seen = last_seen.map(|seen| seen.round(jiff::Unit::Second).unwrap_or(seen));
40    match last_seen {
41        Some(seen) => format!("runtime `{runtime_id}` is not connected (last seen {seen})"),
42        None => format!("runtime `{runtime_id}` is not connected"),
43    }
44}

46/// Why the daemon refused a runtime that dialed in.
47#[derive(Clone, Debug, Eq, PartialEq, thiserror::Error)]
48pub enum DialInRefusal {
49    /// The ID is not a runtime ID.
50    #[error("invalid runtime ID `{0}`: it must match [a-z][a-z0-9_]{{0,31}}")]
51    InvalidId(String),
52    /// The ID is a built-in runtime's.
53    #[error("runtime ID `{0}` is reserved for the daemon's built-in runtime")]
54    Reserved(String),
55    /// The daemon configures a runtime of the ID itself.
56    #[error("runtime `{0}` is configured on the daemon ([runtimes.{0}]) and cannot dial in")]
57    Configured(String),
58    /// A runtime is connected under the ID now.
59    #[error("runtime `{0}` is already connected")]
60    Connected(String),
61    /// The runtime did not complete `runtime/initialize`.
62    #[error("runtime `{0}` failed to start: {1}")]
63    Initialize(String, String),
64}
65
66impl DialInRefusal {
67    /// Whether dialing in again under the ID cannot succeed until the
68    /// runtime's ID or the daemon's configuration changes.
69    #[must_use]
70    pub fn is_permanent(&self) -> bool {
71        matches!(
72            self,
73            Self::InvalidId(_) | Self::Reserved(_) | Self::Configured(_)
74        )
75    }
76}

78/// A runtime that dials in.
79pub(crate) struct DialInTemplate {
80    runtime_id: String,
81    /// The connection, while the runtime is connected.
82    connection: Mutex<Option<RuntimeClient>>,
83    /// Whether a runtime dialing in under the ID is being accepted now.
84    claimed: AtomicBool,
85    /// What the runtime offered when it last connected.
86    offered: Mutex<RuntimeListingState>,
87    /// When the runtime was last heard from, once a connection ended.
88    last_seen: Mutex<Option<jiff::Timestamp>>,
89    /// How its connections' liveness is checked.
90    liveness: super::Liveness,
91    /// The ACP agents the runtime reported last, kept while it is away.
92    acp_agents: Mutex<Vec<cookie_agent_runtime_api::protocol::AcpAgentReport>>,
93    /// The runtime's settings as the daemon's configuration last had them,
94    /// once a connection or a configuration change set them.
95    settings: Mutex<Option<RuntimeSettings>>,
96}
97
98fn lock<T>(mutex: &Mutex<T>) -> std::sync::MutexGuard<'_, T> {
99    mutex
100        .lock()
101        .unwrap_or_else(|poisoned| poisoned.into_inner())
102}
103
104impl DialInTemplate {
105    pub(crate) fn new(runtime_id: &str) -> Self {
106        Self {
107            runtime_id: runtime_id.to_owned(),
108            connection: Mutex::new(None),
109            claimed: AtomicBool::new(false),
110            offered: Mutex::default(),
111            last_seen: Mutex::default(),
112            liveness: super::Liveness::DEFAULT,
113            acp_agents: Mutex::default(),
114            settings: Mutex::default(),
115        }
116    }
117
118    /// Reserves the ID for one runtime dialing in, until
119    /// [`Self::release`]: refused while a runtime is connected under it or
120    /// another is being accepted. A runtime that dials in while its old
121    /// connection still looks open has most likely lost that connection
122    /// without a close, so the old one is checked at once, and the runtime
123    /// dialing in again gets in once it is ended.
124    pub(crate) fn claim(&self) -> Result<(), DialInRefusal> {
125        // Checked under the connection's lock: an accepted runtime's
126        // connection is set before its claim is released.
127        let connection = lock(&self.connection);
128        let connected = connection.as_ref().filter(|client| !client.is_closed());
129        if let Some(client) = connected {
130            client.suspect();
131            return Err(DialInRefusal::Connected(self.runtime_id.clone()));
132        }
133        if self.claimed.swap(true, Ordering::AcqRel) {
134            return Err(DialInRefusal::Connected(self.runtime_id.clone()));
135        }
136        Ok(())
137    }
138
139    /// Ends a claim [`Self::claim`] made.
140    pub(crate) fn release(&self) {
141        self.claimed.store(false, Ordering::Release);
142    }
143
144    /// The live connection, if the runtime is connected.
145    fn current(&self) -> Option<RuntimeClient> {
146        lock(&self.connection)
147            .clone()
148            .filter(|client| !client.is_closed())
149    }
150
151    /// Whether the runtime is connected now.
152    pub(crate) fn is_connected(&self) -> bool {
153        self.current().is_some()
154    }
155
156    /// Initializes the runtime over `connection` with `settings`, its
157    /// `[runtimes.<id>]` table, and makes it the template's connection.
158    pub(crate) async fn connect(
159        &self,
160        connection: cookie_agent_runtime_api::RuntimeConnection,
161        settings: &RuntimeSettings,
162        mcp_hub: Option<Arc<McpHub>>,
163    ) -> Result<RuntimeClient, String> {
164        let initialize = RuntimeClient::initialize(
165            connection,
166            &self.runtime_id,
167            settings,
168            DIAL_IN_SHUTDOWN_GRACE,
169            mcp_hub,
170        );
171        let (client, result) = tokio::time::timeout(DIAL_IN_TIMEOUT, initialize)
172            .await
173            .unwrap_or_else(|_| {
174                Err(format!(
175                    "the runtime did not initialize within {} s",
176                    DIAL_IN_TIMEOUT.as_secs()
177                ))
178            })
179            .inspect_err(|reason| lock(&self.offered).message = Some(reason.clone()))?;
180        client.keep_alive(self.liveness);
181        *lock(&self.offered) = RuntimeListingState {
182            connected: true,
183            capabilities: client.capabilities(),
184            platform: result.platform,
185            machine: result
186                .machine
187                .filter(cookie_agent_protocol::runtime_protocol::RuntimeMachine::is_valid),
188            home: client.home(),
189            message: None,
190        };
191        *lock(&self.connection) = Some(client.clone());
192        // Settings that changed while the runtime initialized reach it now.
193        let latest = lock(&self.settings)
194            .get_or_insert_with(|| settings.clone())
195            .clone();
196        if latest != *settings {
197            client.update_settings(&latest);
198        }
199        Ok(client)
200    }
201
202    /// Records that connection `epoch`, last heard from at `last_seen`,
203    /// ended for `reason`, unless a newer connection took over meanwhile.
204    pub(crate) fn disconnected(&self, epoch: u64, reason: &str, last_seen: jiff::Timestamp) {
205        let connection = lock(&self.connection);
206        if connection
207            .as_ref()
208            .is_some_and(|client| client.epoch() != epoch)
209        {
210            return;
211        }
212        *lock(&self.last_seen) = Some(last_seen);
213        if let Some(client) = connection.as_ref() {
214            *lock(&self.acp_agents) = client.acp_agents();
215        }
216        lock(&self.offered).message = Some(format!(
217            "runtime `{}` disconnected: {reason}",
218            self.runtime_id
219        ));
220    }
221
222    fn not_connected(&self) -> String {
223        let recorded = *lock(&self.last_seen);
224        // A connection that has just ended may not be recorded yet.
225        let last_seen = recorded.or_else(|| {
226            lock(&self.connection)
227                .as_ref()
228                .map(RuntimeClient::last_heard_at)
229        });
230        not_connected(&self.runtime_id, last_seen)
231    }
232}

234#[async_trait]
235impl RuntimeTemplate for DialInTemplate {
236    fn runtime_id(&self) -> &str {
237        &self.runtime_id
238    }
239
240    fn lifetime(&self) -> RuntimeLifetime {
241        RuntimeLifetime::DialIn
242    }
243
244    async fn capabilities(&self) -> Result<Vec<RuntimeCapabilityInfo>, String> {
245        self.current()
246            .map(|client| client.capabilities())
247            .ok_or_else(|| self.not_connected())
248    }
249
250    fn listing(&self) -> RuntimeListingState {
251        let mut listing = lock(&self.offered).clone();
252        listing.connected = self.is_connected();
253        if listing.connected {
254            listing.message = None;
255        }
256        listing
257    }
258
259    fn offline(&self) -> Option<String> {
260        (!self.is_connected()).then(|| self.not_connected())
261    }
262
263    /// Opens `capability` over the live connection: refused at once when
264    /// the runtime is not connected, or its connection no longer answers.
265    async fn start(
266        &self,
267        capability: &CapabilityName,
268        workspace: Option<&Path>,
269    ) -> Result<StartedRuntime, StartError> {
270        let client = self
271            .current()
272            .ok_or_else(|| StartError::Connect(self.not_connected()))?;
273        if let Err(reason) = client.verify_live().await {
274            return Err(StartError::Connect(format!(
275                "runtime `{}` is not connected: {reason}",
276                self.runtime_id
277            )));
278        }
279        tokio::time::timeout(DIAL_IN_TIMEOUT, client.open(capability, workspace))
280            .await
281            .unwrap_or_else(|_| {
282                client.suspect();
283                Err(StartError::Open(format!(
284                    "the runtime did not open `{capability}` within {} s",
285                    DIAL_IN_TIMEOUT.as_secs()
286                )))
287            })
288    }
289
290    fn disconnect(&self) {
291        lock(&self.connection).take();
292    }
293
294    fn update_settings(&self, settings: &RuntimeSettings) {
295        {
296            let mut current = lock(&self.settings);
297            if current.as_ref() == Some(settings) {
298                return;
299            }
300            *current = Some(settings.clone());
301        }
302        if let Some(client) = self.current() {
303            client.update_settings(settings);
304        }
305    }
306
307    fn runtime_mcp_servers(&self) -> Vec<cookie_agent_protocol::McpServerInfo> {
308        self.current()
309            .map(|client| client.runtime_mcp_servers())
310            .unwrap_or_default()
311    }
312
313    async fn reconnect_runtime_mcp(&self, server: &str) -> Result<(), crate::EngineError> {
314        match self.current() {
315            Some(client) => client.reconnect_runtime_mcp(server).await,
316            None => Err(crate::EngineError::Mcp(self.not_connected())),
317        }
318    }
319
320    /// Its agents are available while it is connected, and listed as
321    /// unavailable, as it last reported them, while it is away.
322    fn acp_agents(&self) -> super::registry::RuntimeAcpAgents {
323        match self.current() {
324            Some(client) => super::registry::RuntimeAcpAgents {
325                agents: client.acp_agents(),
326                available: true,
327            },
328            // The connection that ended, until another takes over, else
329            // what was kept when one ended.
330            None => super::registry::RuntimeAcpAgents {
331                agents: lock(&self.connection)
332                    .as_ref()
333                    .map(RuntimeClient::acp_agents)
334                    .unwrap_or_else(|| lock(&self.acp_agents).clone()),
335                available: false,
336            },
337        }
338    }
339
340    async fn agent_connection(&self) -> Result<super::registry::AgentConnection, String> {
341        let client = self.current().ok_or_else(|| self.not_connected())?;
342        client.verify_live().await.map_err(|reason| {
343            format!("runtime `{}` is not connected: {reason}", self.runtime_id)
344        })?;
345        Ok(super::registry::AgentConnection(client))
346    }
