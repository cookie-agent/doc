1//! Tool layers and runtimes: where a tool comes from and which workspace it
2//! sees.
3//!
4//! Not to be confused with `crate::runtime` (the engine internals: sessions,
5//! runs, the model loop) or [`crate::PublishedRuntime`] (the compiled model
6//! runtime published to clients). Here a *runtime* is a source of tools and
7//! workspace context, as specified in `docs/site/specs/runtimes.md`.
8//!
9//! The engine composes a session's tool list from:
10//!
11//! 1. The **base layer** ([`ToolLayer::Base`]): tools the daemon implements
12//!    for its own orchestration (delegation, messaging, skills, goals,
13//!    `read_artifact`). It is not a runtime and has no workspace.
14//! 2. The session's **attachments**, in attach order: capability instances
15//!    that the [`RuntimeRegistry`] opens from [`RuntimeTemplate`]s, one per
16//!    (runtime ID, capability, canonical workspace), shared by every session
17//!    attached to them. The built-in templates are the **global runtime** (ID `global`):
18//!    top-level MCP servers, plugins, and a `webfetch` declared
19//!    `ToolCollision::Fallback`, no workspace, called in-process; and the
20//!    **local coding runtime** (ID `local`): `read`, `write`, `edit`, `bash`,
21//!    and `webfetch` in the daemon's working directory, reached over the
22//!    runtime protocol (`remote`) in production.
23//!
24//! [`compose`] turns the sources' tools into published names, prefixing
25//! collisions with the attachment name. Every tool is traced to its
26//! [`ToolOrigin`]. Preparation, permission decisions, and visibility read the
27//! workspace of that origin, never the daemon's working directory, which only
28//! locates the session store.
29//!
30//! The runtime identity types ([`ToolOrigin`], `RuntimeInfo`,
31//! `WorkspaceContext`) live in `cookie_agent_runtime_api`, which tool
32//! providers depend on; this module composes them.

64/// One-line description of where a tool runs, for the approval reviewer.
65pub(crate) fn describe_origin(origin: &ToolOrigin) -> String {
66    match origin {
67        ToolOrigin::Base => "the daemon's base layer (no workspace)".to_owned(),
68        ToolOrigin::Runtime(runtime) => match runtime.workspace() {
69            Some(workspace) => format!(
70                "the `{}` {} runtime with workspace {}",
71                runtime.id(),
72                runtime.kind().as_str(),
73                workspace.root().display()
74            ),
75            None => format!(
76                "the `{}` {} runtime (no workspace)",
77                runtime.id(),
78                runtime.kind().as_str()
79            ),
80        },
81    }
82}

84/// Where [`crate::Engine::try_register_tool_provider`] adds a provider.
85#[derive(Clone, Copy, Debug, Eq, PartialEq)]
86pub enum ToolLayer {
87    /// The daemon's base layer.
88    Base,
89    /// The global runtime instance.
90    Global,
91    /// The local coding runtime instance.
92    Local,
93}

95/// A provider together with the layer or runtime attachment that owns it.
96#[derive(Clone)]
97pub(crate) struct SourcedProvider {
98    /// The session's attachment name; `None` for the base layer.
99    pub(crate) attachment: Option<cookie_agent_protocol::AttachmentName>,
100    /// The instance generation; `0` for the base layer.
101    pub(crate) generation: u64,
102    pub(crate) origin: ToolOrigin,
103    pub(crate) provider: Arc<dyn ToolProvider>,
104}
