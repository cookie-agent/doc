24/// Where a permission is evaluated: the runtime that offers the tool, and
25/// that runtime's workspace.
26///
27/// A rule with a runtime selector (`bash@labpc`) only applies when `runtime`
28/// is that runtime ID; the base layer (`runtime: None`) only sees unqualified
29/// rules. `${workspace_dir}` and relative resources resolve against
30/// `workspace`; without one they never match.
31#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
32pub struct PermissionScope<'a> {
33    pub runtime: Option<&'a str>,
34    pub workspace: Option<&'a Path>,
35}
36
37impl<'a> PermissionScope<'a> {
38    /// The base layer: no runtime, no workspace.
39    pub const BASE: Self = Self {
40        runtime: None,
41        workspace: None,
42    };
43
44    /// The scope of a tool from `origin`.
45    #[must_use]
46    pub fn of(origin: &'a crate::ToolOrigin) -> Self {
47        Self {
48            runtime: origin
49                .runtime()
50                .map(cookie_agent_runtime_api::RuntimeInfo::id),
51            workspace: origin.workspace_root(),
52        }
53    }
54
55    /// The scope of the runtime with ID `runtime` working in `workspace`.
56    #[must_use]
57    pub const fn runtime(runtime: &'a str, workspace: Option<&'a Path>) -> Self {
58        Self {
59            runtime: Some(runtime),
60            workspace,
61        }
62    }
63}
64
65/// Only unqualified rules, resolved against `workspace`.
66impl<'a> From<Option<&'a Path>> for PermissionScope<'a> {
67    fn from(workspace: Option<&'a Path>) -> Self {
68        Self {
69            runtime: None,
70            workspace,
71        }
72    }

75/// Visibility is action-level opt-in, not a resource authorization decision.
76/// `runtime` is the runtime ID of the tool (`None` for the base layer).
77#[must_use]
78pub fn tool_visible(
79    rules: &[PermissionRule],
80    overlay: Option<&SessionPermissionOverlay>,
81    action: &PermissionAction,
82    runtime: Option<&str>,
83) -> bool {
84    rules
85        .iter()
86        .chain(overlay.into_iter().flat_map(|overlay| &overlay.rules))
87        .any(|rule| {
88            rule.action == *action
89                && rule.applies_to(runtime)
90                && matches!(rule.effect, PermissionEffect::Allow | PermissionEffect::Ask)
91        })
92}

108#[derive(Debug, Default)]
109pub struct ApprovalStore {

111}

113impl ApprovalStore {
114    pub fn replace(&self, grants: impl IntoIterator<Item = TreeApprovalGrant>) {
115        let mut stored = self
116            .grants
117            .lock()
118            .unwrap_or_else(|poisoned| poisoned.into_inner());
119        stored.clear();
120        for grant in grants {
121            stored.insert(
122                ApprovalKey {
123                    root: grant.root_session_id,
124                    fingerprint: grant.operation_fingerprint.clone(),
125                    runtime: grant.runtime_id.clone(),
126                },
127                grant,
128            );
129        }
130    }
131
132    pub fn grant(&self, grant: TreeApprovalGrant) {
133        self.grants
134            .lock()
135            .unwrap_or_else(|poisoned| poisoned.into_inner())
136            .insert(
137                ApprovalKey {
138                    root: grant.root_session_id,
139                    fingerprint: grant.operation_fingerprint.clone(),
140                    runtime: grant.runtime_id.clone(),
141                },
142                grant,
143            );
144    }

146    /// The grant of tree `root` for `operation` on the runtime with ID
147    /// `runtime` (`None` for the base layer). A grant only applies to calls
148    /// on the runtime ID it was given for.
149    #[must_use]
150    pub fn matching(
151        &self,
152        root: SessionId,
153        operation: &PreparedOperationIdentity,
154        runtime: Option<&cookie_agent_protocol::RuntimeId>,
155    ) -> Option<TreeApprovalGrant> {
156        let fingerprint = OperationFingerprint::from_prepared_operation(operation);
157        self.grants
158            .lock()
159            .unwrap_or_else(|poisoned| poisoned.into_inner())
160            .get(&ApprovalKey {
161                root,
162                fingerprint,
163                runtime: runtime.cloned(),
164            })
165            .filter(|grant| {
166                grant.capabilities == operation.capabilities()
167                    && grant.resources == operation.resources()
168            })
169            .cloned()
170    }
171
172    #[must_use]
173    pub fn for_root(&self, root: SessionId) -> Vec<TreeApprovalGrant> {
174        self.grants
175            .lock()
176            .unwrap_or_else(|poisoned| poisoned.into_inner())
177            .iter()
178            .filter(|(key, _)| key.root == root)
179            .map(|(_, grant)| grant.clone())
180            .collect()
181    }
182
183    pub fn invalidate_grants(&self, ids: &HashSet<TreeApprovalGrantId>) {
184        self.grants
185            .lock()
186            .unwrap_or_else(|poisoned| poisoned.into_inner())
187            .retain(|_, grant| !ids.contains(&grant.grant_id));
188    }
189}

228        operation: &PreparedOperationIdentity,
229        policy_labels: &[Option<String>],

235    #[must_use]
236    pub fn decide_operation_with_overlay<'a>(
237        &self,
238        policy: &AgentSnapshot,
239        overlay: Option<&SessionPermissionOverlay>,
240        operation: &PreparedOperationIdentity,
241        policy_labels: &[Option<String>],
242        scope: impl Into<PermissionScope<'a>>,
243    ) -> PermissionDecision {
244        let scope = scope.into();
245        assert!(!operation.resources().is_empty());
246        assert_eq!(operation.resources().len(), policy_labels.len());
247        let evaluations = operation
248            .resources()
249            .iter()
250            .zip(policy_labels)
251            .map(|(resource, normalized)| {
252                let (candidates, effect, reason) = match normalized {
253                    Some(normalized) => {
254                        let candidates = matching_rules(
255                            policy,
256                            overlay,
257                            &resource.capability,
258                            normalized,
259                            scope,
260                        );
261                        let (effect, reason) = effective_permission_with_overlay(
262                            policy,
263                            overlay,
264                            &resource.capability,
265                            normalized,
266                            scope,
267                        );
268                        (candidates, effect, reason)
269                    }
270                    None => {
271                        let candidates =
272                            matching_loose_rules(policy, overlay, &resource.capability, scope);
273                        let (effect, reason) = effective_loose_permission_with_overlay(
274                            policy,
275                            overlay,
276                            &resource.capability,
277                            scope,
278                        );
279                        (candidates, effect, reason)
280                    }
281                };
282                ApprovalEvaluation {
283                    resource_digest: resource.binding_digest.clone(),
284                    effect,
285                    trace: DecisionTrace {
286                        action: resource.capability.clone(),
287                        normalized_resource: normalized
288                            .clone()
289                            .unwrap_or_else(|| UNSCOPED_PERMISSION_RESOURCE_DISPLAY.to_owned()),
290                        candidates,
291                        effect,
292                        precedence_reason: reason,
293                    },
294                }
295            })
296            .collect::<Vec<_>>();
297        let effect = aggregate_effect(&evaluations);
298        PermissionDecision {
299            effect,
300            evaluations,
301        }
302    }

304    #[must_use]
305    pub fn decide_operation_with_grants<'a>(
306        &self,
307        policy: &AgentSnapshot,
308        overlay: Option<&SessionPermissionOverlay>,
309        grants: Option<&SessionPermissionOverlay>,
310        operation: &PreparedOperationIdentity,
311        policy_labels: &[Option<String>],
312        scope: impl Into<PermissionScope<'a>>,
313    ) -> PermissionDecision {
314        let scope = scope.into();
315        let mut decision =
316            self.decide_operation_with_overlay(policy, overlay, operation, policy_labels, scope);
317        let Some(grants) = grants else {
318            return decision;
319        };
320        for (evaluation, (resource, normalized)) in decision
321            .evaluations
322            .iter_mut()
323            .zip(operation.resources().iter().zip(policy_labels))
324        {
325            if evaluation.effect != PermissionEffect::Ask {
326                continue;
327            }
328            let granted = grants.rules.iter().any(|rule| {
329                rule.effect == PermissionEffect::Allow
330                    && rule.action == resource.capability
331                    && rule.applies_to(scope.runtime)
332                    && normalized.as_ref().is_some_and(|normalized| {
333                        permission_pattern_matches(
334                            rule.resource.as_str(),
335                            &resource.capability,
336                            normalized,
337                            scope.workspace,
338                        )
339                    })
340            });
341            if granted {
342                evaluation.effect = PermissionEffect::Allow;
343                evaluation.trace.effect = PermissionEffect::Allow;
344                evaluation.trace.precedence_reason =
345                    "turn-scoped skill grant allows an otherwise ask operation".into();
346            }
347        }
348        decision.effect = aggregate_effect(&decision.evaluations);
349        decision
350    }

358    /// Whether a tool with `permission_name` of the runtime with ID `runtime`
359    /// (`None` for the base layer) is visible. A tool that needs no permission
360    /// has no name to ask about and is always visible.
361    #[must_use]
362    pub fn tool_visible_with_overlay(
363        policy: &AgentSnapshot,
364        overlay: Option<&SessionPermissionOverlay>,
365        permission_name: &str,
366        runtime: Option<&str>,
367    ) -> bool {
368        let Ok(action) = Self::action_for_permission_name(permission_name) else {
369            return false;
370        };
371        tool_visible(&policy.permissions, overlay, &action, runtime)
372    }
373
374    #[must_use]
375    pub fn tool_visible_with_grants<'a>(
376        policy: &AgentSnapshot,
377        overlay: Option<&SessionPermissionOverlay>,
378        grants: Option<&SessionPermissionOverlay>,
379        permission_name: &str,
380        scope: impl Into<PermissionScope<'a>>,
381    ) -> bool {
382        let scope = scope.into();
383        if Self::tool_visible_with_overlay(policy, overlay, permission_name, scope.runtime) {
384            return true;
385        }
386        let Ok(action) = Self::action_for_permission_name(permission_name) else {
387            return false;
388        };
389        grants.is_some_and(|grants| {
390            grants.rules.iter().any(|grant| {
391                grant.action == action
392                    && grant.effect == PermissionEffect::Allow
393                    && grant.applies_to(scope.runtime)
394                    && effective_permission_with_overlay(
395                        policy,
396                        overlay,
397                        &action,
398                        grant.resource.as_str(),
399                        scope,
400                    )
401                    .0 != PermissionEffect::Deny
402            })
403        })
404    }

407fn aggregate_effect(evaluations: &[ApprovalEvaluation]) -> PermissionEffect {
408    if evaluations
409        .iter()
410        .any(|evaluation| evaluation.effect == PermissionEffect::Deny)
411    {
412        PermissionEffect::Deny
413    } else if evaluations
414        .iter()
415        .any(|evaluation| evaluation.effect == PermissionEffect::Ask)
416    {
417        PermissionEffect::Ask
418    } else {
419        PermissionEffect::Allow
420    }
421}

423/// The precedence of a rule among the rules that match: a qualified rule
424/// ranks above an unqualified one, then later declarations win.
425fn loose_rank(rule: &PermissionRule, declaration_index: usize) -> (bool, usize) {
426    (rule.runtime.is_some(), declaration_index)
427}

429pub(crate) fn effective_loose_permission<'a>(
430    policy: &AgentSnapshot,
431    action: &PermissionAction,
432    scope: impl Into<PermissionScope<'a>>,
433) -> (PermissionEffect, String) {
434    let scope = scope.into();
435    policy
436        .permissions
437        .iter()
438        .enumerate()
439        .filter(|(_, rule)| {
440            rule.action == *action
441                && rule.applies_to(scope.runtime)
442                && rule.resource.as_str() == "*"
443        })
444        .max_by_key(|(index, rule)| loose_rank(rule, *index))
445        .map_or_else(
446            || {
447                (
448                    PermissionEffect::Deny,
449                    "no bare or `*` rule; deny by default for a permission-name-only check".into(),
450                )
451            },
452            |(_, rule)| {
453                (
454                    rule.effect,
455                    "bare or `*` permission rule applies to the permission-name-only check".into(),
456                )
457            },
458        )
459}

461pub(crate) fn effective_loose_permission_with_overlay<'a>(
462    policy: &AgentSnapshot,
463    overlay: Option<&SessionPermissionOverlay>,
464    action: &PermissionAction,
465    scope: impl Into<PermissionScope<'a>>,
466) -> (PermissionEffect, String) {
467    let scope = scope.into();
468    overlay
469        .and_then(|overlay| {
470            overlay
471                .rules
472                .iter()
473                .enumerate()
474                .filter(|(_, rule)| {
475                    rule.action == *action
476                        && rule.applies_to(scope.runtime)
477                        && rule.resource.as_str() == "*"
478                })
479                .max_by_key(|(index, rule)| loose_rank(rule, *index))
480        })
481        .map_or_else(
482            || effective_loose_permission(policy, action, scope),
483            |(_, rule)| {
484                (
485                    rule.effect,
486                    "session overlay wildcard rule takes precedence over the agent document".into(),
487                )
488            },
489        )
490}

492pub(crate) fn effective_permission<'a>(
493    policy: &AgentSnapshot,
494    action: &PermissionAction,
495    resource: &str,
496    scope: impl Into<PermissionScope<'a>>,
497) -> (PermissionEffect, String) {
498    let scope = scope.into();
499    let winner = policy
500        .permissions
501        .iter()
502        .enumerate()
503        .filter(|(_, rule)| rule_matches(rule, action, resource, scope, &policy.agent))
504        .max_by_key(|(index, rule)| specificity(rule, *index, &policy.agent));
505    winner
506        .map_or_else(
507            || {
508                (
509                    PermissionEffect::Deny,
510                    "no matching rule; deny by default".into(),
511                )
512            },
513            |(_, rule)| {
514                (
515                    rule.effect,
516                    "most-specific matching pattern: more literal characters, then fewer wildcards, then a runtime selector, then later declaration".into(),
517                )
518            },
519        )
520}

522pub(crate) fn effective_permission_with_overlay<'a>(
523    policy: &AgentSnapshot,
524    overlay: Option<&SessionPermissionOverlay>,
525    action: &PermissionAction,
526    resource: &str,
527    scope: impl Into<PermissionScope<'a>>,
528) -> (PermissionEffect, String) {
529    let scope = scope.into();
530    let winner = overlay.and_then(|overlay| {
531        overlay
532            .rules
533            .iter()
534            .enumerate()
535            .filter(|(_, rule)| rule_matches(rule, action, resource, scope, &policy.agent))
536            .max_by_key(|(index, rule)| specificity(rule, *index, &policy.agent))
537    });
538    winner.map_or_else(
539        || effective_permission(policy, action, resource, scope),
540        |(_, rule)| {
541            (
542                rule.effect,
543                "session overlay matching rule takes precedence over the agent document".into(),
544            )
545        },
546    )
547}

549/// Whether `rule` governs `resource` of `action` in `scope` for a run of
550/// `agent`.
551fn rule_matches(
552    rule: &PermissionRule,
553    action: &PermissionAction,
554    resource: &str,
555    scope: PermissionScope<'_>,
556    agent: &AgentId,
557) -> bool {
558    rule.action == *action
559        && rule.applies_to(scope.runtime)
560        && permission_pattern_matches(
561            &rule_pattern(rule, agent),
562            action,
563            resource,
564            scope.workspace,
565        )
566}

568/// In a rule pattern, `agent://self` stands for the run's own agent
569/// (`docs/site/specs/memory.md` section 9.1).
570pub(crate) const SELF_AGENT_PATTERN: &str = "agent://self";
571
572/// The pattern `rule` matches with for a run of `agent`: its resource, with
573/// `agent://self` read as `agent://<agent>`.
574fn rule_pattern<'rule>(rule: &'rule PermissionRule, agent: &AgentId) -> Cow<'rule, str> {
575    if rule.resource.as_str() == SELF_AGENT_PATTERN {
576        Cow::Owned(format!("agent://{agent}"))
577    } else {
578        Cow::Borrowed(rule.resource.as_str())
579    }
580}

582/// The rules an agent document gives its runs (`docs/site/specs/memory.md`
583/// section 9.2): the document's own, then, for each memory action the
584/// document names no key for (with or without a runtime selector), the
585/// default of the agent's mode. `primary` and `all` agents may read and
586/// write all memory, `subagent` agents read all memory and write their own
587/// agent file, and internal agents get nothing.
588pub(crate) fn agent_permission_rules(
589    document: &cookie_agent_config::AgentDocument,
590) -> Vec<PermissionRule> {
591    let permissions = &document.frontmatter.permissions;
592    let mut rules = permissions
593        .iter()
594        .flat_map(|(key, value)| value.rules(key))
595        .collect::<Vec<_>>();
596    let defaults: &[(PermissionAction, &str)] = match document.frontmatter.mode {
597        AgentMode::Primary | AgentMode::All => &[
598            (PermissionAction::MEMORY, "*"),
599            (PermissionAction::MEMORY_WRITE, "*"),
600        ],
601        AgentMode::Subagent => &[
602            (PermissionAction::MEMORY, "*"),
603            (PermissionAction::MEMORY_WRITE, SELF_AGENT_PATTERN),
604        ],
605        AgentMode::Internal => &[],
606    };
607    for (action, resource) in defaults {
608        if !permissions.keys().any(|key| key.action == *action) {
609            rules.push(PermissionRule::new(
610                action.clone(),
611                WildcardPattern::new(*resource).expect("static default pattern"),
612                PermissionEffect::Allow,
613            ));
614        }
615    }
616    rules
617}

619/// Whether `pattern` matches `resource` for a tool whose runtime has
620/// `workspace`. `${workspace_dir}` patterns match the resource made absolute
621/// against the workspace. Without a workspace (the base layer and the global
622/// runtime) nothing can be resolved: `${workspace_dir}` patterns never match,
623/// and neither does any pattern for a relative path resource of a path action
624/// (`read`, `write`).
625fn permission_pattern_matches(
626    pattern: &str,
627    action: &PermissionAction,
628    resource: &str,
629    workspace: Option<&Path>,
630) -> bool {
631    let workspace_pattern =
632        pattern.contains(cookie_agent_protocol::WildcardPattern::WORKSPACE_DIR_EXPRESSION);
633    match workspace {
634        Some(workspace) if workspace_pattern => simple_wildcard_match(
635            &expand_workspace_pattern(pattern, workspace),
636            &absolute_resource(workspace, resource),
637        ),
638        Some(_) => simple_wildcard_match(pattern, resource),
639        None if workspace_pattern => false,
640        None if action.is_path() && !Path::new(resource).is_absolute() => false,
641        None => simple_wildcard_match(pattern, resource),
642    }
643}
644
645fn expand_workspace_pattern(pattern: &str, workspace: &Path) -> String {
646    let workspace = normalized_path(&canonical_workspace(workspace));
647    if workspace == "/" {
648        pattern.replace("${workspace_dir}/", "/").replace(
649            cookie_agent_protocol::WildcardPattern::WORKSPACE_DIR_EXPRESSION,
650            "/",
651        )
652    } else {
653        pattern.replace(
654            cookie_agent_protocol::WildcardPattern::WORKSPACE_DIR_EXPRESSION,
655            workspace.trim_end_matches('/'),
656        )
657    }
658}
659
660fn absolute_resource(workspace: &Path, resource: &str) -> String {
661    let resource_path = Path::new(resource);
662    if resource_path.is_absolute() {
663        normalized_path(resource_path)
664    } else if resource_path == Path::new(".") {
665        normalized_path(&canonical_workspace(workspace))
666    } else {
667        normalized_path(&canonical_workspace(workspace).join(resource_path))
668    }
669}
670
671fn canonical_workspace(workspace: &Path) -> PathBuf {
672    workspace
673        .canonicalize()
674        .unwrap_or_else(|_| workspace.to_owned())
675}
676
677fn normalized_path(path: &Path) -> String {
678    path.to_string_lossy().replace('\\', "/")
679}

681/// Rule precedence among matching rules: more literal characters, then fewer
682/// wildcards, then a runtime selector over none, then later declaration.
683fn specificity(
684    rule: &PermissionRule,
685    declaration_index: usize,
686    agent: &AgentId,
687) -> (usize, std::cmp::Reverse<usize>, bool, usize) {
688    let pattern = rule_pattern(rule, agent);
689    let wildcards = pattern
690        .chars()
691        .filter(|character| matches!(character, '*' | '?'))
692        .count();
693    let literals = pattern.chars().count() - wildcards;
694    (
695        literals,
696        std::cmp::Reverse(wildcards),
697        rule.runtime.is_some(),
698        declaration_index,
699    )
700}

725/// The agent-document rules, then the overlay rules, that `keep` admits.
726fn matched(
727    policy: &AgentSnapshot,
728    overlay: Option<&SessionPermissionOverlay>,
729    keep: impl Fn(&PermissionRule) -> bool,
730) -> Vec<MatchedPermissionRule> {
731    let traced = |layer: &'static str| {
732        move |rule: &PermissionRule| MatchedPermissionRule {
733            source_layer: SafeCode::new(layer).expect("static safe code"),
734            action: rule.action.clone(),
735            runtime: rule.runtime.clone(),
736            resource: rule.resource.clone(),
737            effect: rule.effect,
738        }
739    };
740    policy
741        .permissions
742        .iter()
743        .filter(|rule| keep(rule))
744        .map(traced("agent_document"))
745        .chain(
746            overlay
747                .into_iter()
748                .flat_map(|overlay| overlay.rules.iter())
749                .filter(|rule| keep(rule))
750                .map(traced("session_overlay")),
751        )
752        .collect()
753}

784    /// The actions the tools of every source of `session` are checked under,
785    /// visible or not, in source order.
786    fn declared_permission_actions(
787        &self,
788        session_id: SessionId,
789    ) -> Result<Vec<PermissionAction>, crate::EngineError> {
790        let context = cookie_agent_runtime_api::SessionToolContext::new(session_id);
791        let mut actions = Vec::new();
792        for source in self.session_tool_sources(session_id)? {
793            for tool in source
794                .provider
795                .tools_for_session(&context)
796                .map_err(|error| crate::EngineError::ToolFailed(error.to_string()))?

800                    .as_deref()
801                    .map(PermissionPipeline::action_for_permission_name)
802                    && !actions.contains(&action)
803                {
804                    actions.push(action);
805                }
806            }
807        }
808        Ok(actions)

811    /// Sets the overlay rule for `action` (qualified by `runtime` when given)
812    /// on `resource`. A change that makes the rule stricter for any runtime
813    /// the session uses invalidates that action's tree grants there.
814    pub async fn set_session_permission(
815        &self,
816        session_id: SessionId,
817        action: &PermissionAction,
818        runtime: Option<RuntimeId>,
819        resource: WildcardPattern,
820        effect: PermissionEffect,
821        origin: cookie_agent_protocol::EventOrigin,
822    ) -> Result<SessionPermissionMutationResult, crate::EngineError> {
823        let _mutation = self
824            .inner
825            .approvals
826            .permission_overlay_mutation
827            .lock()
828            .await;
829        let session = self.inner.store.get(session_id)?;
830        let policy = governing_agent(&session);
831        let scopes = self.overlay_effect_scopes(&session, runtime.as_ref());
832        let mut overlay = session.permission_overlay.clone();
833        let old_effects = rule_effects(&policy, &overlay, action, resource.as_str(), &scopes);
834        overlay.rules.retain(|rule| {
835            rule.action != *action || rule.runtime != runtime || rule.resource != resource
836        });
837        overlay.rules.push(PermissionRule {
838            action: action.clone(),
839            runtime: runtime.clone(),
840            resource: resource.clone(),
841            effect,
842        });
843        overlay
844            .validate()
845            .map_err(|error| crate::EngineError::Permission(error.to_string()))?;
846        let new_effects = rule_effects(&policy, &overlay, action, resource.as_str(), &scopes);
847        if stricter(&old_effects, &new_effects) {
848            self.invalidate_action_grants(session_id, &session.meta.origin, action, runtime)?;
849        }
850        self.append(
851            session_id,
852            None,
853            origin,
854            EventPayload::SessionPermissionOverlaySet { overlay },
855        )
856        .await?;
857        let result = self.get_session_permissions(session_id)?;
858        Ok(SessionPermissionMutationResult {
859            permissions: result.permissions,
860        })
861    }

863    /// Removes the overlay rule for `action` (qualified by `runtime` when
864    /// given) on `resource`.
865    pub async fn clear_session_permission(
866        &self,
867        session_id: SessionId,
868        action: &PermissionAction,
869        runtime: Option<RuntimeId>,
870        resource: &WildcardPattern,
871        origin: cookie_agent_protocol::EventOrigin,
872    ) -> Result<SessionPermissionMutationResult, crate::EngineError> {
873        let _mutation = self
874            .inner
875            .approvals
876            .permission_overlay_mutation
877            .lock()
878            .await;
879        let session = self.inner.store.get(session_id)?;
880        let policy = governing_agent(&session);
881        let scopes = self.overlay_effect_scopes(&session, runtime.as_ref());
882        let mut overlay = session.permission_overlay.clone();
883        let old_effects = rule_effects(&policy, &overlay, action, resource.as_str(), &scopes);
884        let prior_len = overlay.rules.len();
885        overlay.rules.retain(|rule| {
886            rule.action != *action || rule.runtime != runtime || &rule.resource != resource
887        });
888        if overlay.rules.len() != prior_len {
889            let new_effects = rule_effects(&policy, &overlay, action, resource.as_str(), &scopes);
890            if stricter(&old_effects, &new_effects) {
891                self.invalidate_action_grants(session_id, &session.meta.origin, action, runtime)?;
892            }
893            self.append(
894                session_id,
895                None,
896                origin,
897                EventPayload::SessionPermissionOverlaySet { overlay },
898            )
899            .await?;
900        }
901        let result = self.get_session_permissions(session_id)?;
902        Ok(SessionPermissionMutationResult {
903            permissions: result.permissions,
904        })
905    }

907    /// The scopes an overlay rule can affect in `session`: for a rule
908    /// qualified by `runtime`, each attachment of that runtime ID (or the
909    /// runtime without a workspace when none is attached); for an
910    /// unqualified rule, the base layer and every attachment.
911    fn overlay_effect_scopes(
912        &self,
913        session: &crate::session::SessionProjection,
914        runtime: Option<&RuntimeId>,
915    ) -> Vec<(Option<String>, Option<std::path::PathBuf>)> {
916        let attachments = self.attachments_of(session);
917        let mut scopes = attachments
918            .iter()
919            .filter(|attachment| runtime.is_none_or(|runtime| &attachment.runtime_id == runtime))
920            .map(|attachment| {
921                (
922                    Some(attachment.runtime_id.as_str().to_owned()),
923                    attachment
924                        .workspace
925                        .as_deref()
926                        .map(std::path::PathBuf::from),
927                )
928            })
929            .collect::<Vec<_>>();
930        match runtime {
931            None => scopes.push((None, None)),
932            Some(runtime) if scopes.is_empty() => {
933                scopes.push((Some(runtime.as_str().to_owned()), None));
934            }
935            Some(_) => {}
936        }
937        scopes
938    }

940    fn invalidate_action_grants(
941        &self,
942        session_id: SessionId,
943        origin: &SessionOrigin,
944        action: &PermissionAction,
945        runtime: Option<RuntimeId>,
946    ) -> Result<(), crate::EngineError> {
947        let root = match origin {
948            SessionOrigin::Root => session_id,
949            SessionOrigin::Delegated {
950                root_session_id, ..
951            } => *root_session_id,
952        };
953        let grants = self
954            .inner
955            .approvals
956            .store
957            .for_root(root)
958            .into_iter()
959            .filter(|grant| {
960                runtime
961                    .as_ref()
962                    .is_none_or(|runtime| grant.runtime_id.as_ref() == Some(runtime))
963                    && grant
964                        .resources
965                        .iter()
966                        .any(|resource| resource.capability == *action)
967            })
968            .collect::<Vec<_>>();
969        let ids = grants
970            .iter()
971            .map(|grant| grant.grant_id)
972            .collect::<HashSet<_>>();
973        if ids.is_empty() {
974            return Ok(());
975        }
976        let digests = grants
977            .iter()
978            .flat_map(|grant| grant.resources.iter())
979            .filter(|resource| resource.capability == *action)
980            .map(|resource| resource.binding_digest.digest().clone())
981            .collect();
982        self.inner.grant_journals.for_root(root)?.invalidate(
983            root,
984            ids.iter().copied().collect(),
985            digests,
986        )?;
987        self.inner.approvals.store.invalidate_grants(&ids);
988        Ok(())
989    }

998/// The agent of the latest visible run, else the session's creation agent.
999fn governing_agent(session: &crate::session::SessionProjection) -> Arc<AgentSnapshot> {
1000    // The log index names the latest visible run; the projection already
1001    // holds its snapshot unless the run started after this projection was
1002    // taken, in which case the log itself is read.
1003    let latest_run = session.log.last_run_started().map(|(_, run, _)| {
1004        session.runs.get(&run).map_or_else(
1005            || {
1006                session
1007                    .log
1008                    .event_snapshot()
1009                    .iter()
1010                    .rev()
1011                    .find_map(|event| match &event.payload {
1012                        EventPayload::RunStarted { agent, .. } => {
1013                            Some(Arc::new(agent.as_ref().clone()))
1014                        }
1015                        _ => None,
1016                    })
1017                    .unwrap_or_else(|| session.creation_agent.clone())
1018            },
1019            |projected| projected.agent.clone(),
1020        )
1021    });
1022    select_governing_agent(&session.creation_agent, latest_run.as_ref()).clone()
1023}

1025fn select_governing_agent<'a, A>(creation_agent: &'a A, latest_run_agent: Option<&'a A>) -> &'a A {
1026    latest_run_agent.unwrap_or(creation_agent)
1027}

1029fn effect_rank(effect: PermissionEffect) -> u8 {
1030    match effect {
1031        PermissionEffect::Allow => 0,
1032        PermissionEffect::Ask => 1,
1033        PermissionEffect::Deny => 2,
1034    }
1035}
1036
1037/// Whether any scope's effect got stricter between `old` and `new`.
1038fn stricter(old: &[PermissionEffect], new: &[PermissionEffect]) -> bool {
1039    old.iter()
1040        .zip(new)
1041        .any(|(old, new)| effect_rank(*new) > effect_rank(*old))
1042}

1044/// The effect of `action` on `resource` in each of `scopes`.
1045fn rule_effects(
1046    policy: &AgentSnapshot,
1047    overlay: &SessionPermissionOverlay,
1048    action: &PermissionAction,
1049    resource: &str,
1050    scopes: &[(Option<String>, Option<std::path::PathBuf>)],
1051) -> Vec<PermissionEffect> {
1052    scopes
1053        .iter()
1054        .map(|(runtime, workspace)| {
1055            let scope = PermissionScope {
1056                runtime: runtime.as_deref(),
1057                workspace: workspace.as_deref(),
1058            };
1059            if resource == "*" {
1060                effective_loose_permission_with_overlay(policy, Some(overlay), action, scope).0
1061            } else {
1062                effective_permission_with_overlay(policy, Some(overlay), action, resource, scope).0
1063            }
1064        })
1065        .collect()
1066}

1068/// The effective permission of every action: the built-in actions, then the
1069/// other actions the session's tools `declared`, then any other action a rule
1070/// names, each group in first-seen order.
1071fn effective_permission_view(
1072    policy: &AgentSnapshot,
1073    overlay: &SessionPermissionOverlay,
1074    declared: &[PermissionAction],
1075) -> Vec<EffectivePermissionAction> {
1076    let mut actions = PermissionAction::BUILT_IN.to_vec();
1077    let named = policy
1078        .permissions
1079        .iter()
1080        .chain(&overlay.rules)
1081        .map(|rule| &rule.action);
1082    for action in declared.iter().chain(named) {
1083        if !actions.contains(action) {
1084            actions.push(action.clone());
1085        }
1086    }
1087    actions
1088        .into_iter()
1089        .map(|action| {
1090            let unqualified_wildcard = |rule: &&PermissionRule| {
1091                rule.action == action && rule.runtime.is_none() && rule.resource.as_str() == "*"
1092            };
1093            let overlay_wildcard = overlay.rules.iter().rev().find(unqualified_wildcard);
1094            let agent_wildcard = policy.permissions.iter().rev().find(unqualified_wildcard);
1095            let (effect, source) = overlay_wildcard.map_or_else(
1096                || {
1097                    agent_wildcard.map_or(
1098                        (PermissionEffect::Deny, PermissionRuleSource::Default),
1099                        |rule| (rule.effect, PermissionRuleSource::AgentDocument),
1100                    )
1101                },
1102                |rule| (rule.effect, PermissionRuleSource::SessionOverlay),
1103            );
1104            let listed = |rule: &&PermissionRule| {
1105                rule.action == action && (rule.runtime.is_some() || rule.resource.as_str() != "*")
1106            };
1107            let entry = |rule: &PermissionRule, source| {
1108                (
1109                    (rule.runtime.clone(), rule.resource.as_str().to_owned()),
1110                    EffectivePermissionRule {
1111                        runtime: rule.runtime.clone(),
1112                        resource: rule.resource.clone(),
1113                        effect: rule.effect,
1114                        source,
1115                    },
1116                )
1117            };
1118            let mut patterns = BTreeMap::new();
1119            if overlay_wildcard.is_none() {
1120                patterns.extend(
1121                    policy
1122                        .permissions
1123                        .iter()
1124                        .filter(listed)
1125                        .map(|rule| entry(rule, PermissionRuleSource::AgentDocument)),
1126                );
1127            }
1128            patterns.extend(
1129                overlay
1130                    .rules
1131                    .iter()
1132                    .filter(listed)
1133                    .map(|rule| entry(rule, PermissionRuleSource::SessionOverlay)),
1134            );
1135            EffectivePermissionAction {
1136                action,
1137                effect,
1138                source,
1139                patterns: patterns.into_values().collect(),
1140            }
1141        })
1142        .collect()
1143}
