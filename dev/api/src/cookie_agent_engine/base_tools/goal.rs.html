195    async fn prepare(
196        &self,
197        ctx: ToolPreparationContext,
198        call: ToolCall,
199    ) -> Result<PreparedTool, ToolError> {
200        let (normalized_arguments, operation, policy_label, executor) = match call.name.as_str() {
201            "goal_get" => {
202                let params: GoalGetParams = parse_args(&call.name, call.arguments)?;
203                let label = "goal:current".to_owned();
204                let resource = prepared_resource(
205                    PermissionAction::READ,
206                    "goal",
207                    label.as_bytes(),
208                    ctx.session.to_string().as_bytes(),
209                    PreparedBindingLifetime::RestartStable,
210                    ApprovalResourceSource::PrimaryOperation,
211                )?;
212                let operation = prepared_operation(
213                    "goal_get",
214                    &params,
215                    vec![(PermissionAction::READ, "get")],
216                    vec![resource],
217                    ctx.session.to_string().as_bytes(),
218                )?;
219                (
220                    serde_json::to_value(params).map_err(tool_error)?,
221                    operation,
222                    label,
223                    GoalExecutor::Get {
224                        engine: self.engine.clone(),
225                        session: ctx.session,
226                    },
227                )
228            }
229            "goal_set" => {
230                let params: GoalSetParams = parse_args_strict(&call.name, call.arguments)?;
231                let label = "goal:current".to_owned();
232                let normalized = serde_json::to_value(&params).map_err(tool_error)?;
233                let binding = serde_json::to_vec(&(ctx.session, &params)).map_err(tool_error)?;
234                let resource = prepared_resource(
235                    PermissionAction::WRITE,
236                    "goal",
237                    b"goal:current",
238                    &binding,
239                    PreparedBindingLifetime::RestartStable,
240                    ApprovalResourceSource::PrimaryOperation,
241                )?;
242                let operation = prepared_operation(
243                    "goal_set",
244                    &params,
245                    vec![(PermissionAction::WRITE, "set")],
246                    vec![resource],
247                    ctx.session.to_string().as_bytes(),
248                )?;
249                (
250                    normalized,
251                    operation,
252                    label,
253                    GoalExecutor::Set {
254                        engine: self.engine.clone(),
255                        session: ctx.session,
256                        params,
257                    },
258                )
259            }
260            "goal_update" => {
261                let params: GoalUpdateParams = parse_args_strict(&call.name, call.arguments)?;
262                let label = "goal:current".to_owned();
263                let normalized = serde_json::to_value(&params).map_err(tool_error)?;
264                let operation = Self::update_operation(ctx.session, &params)?;
265                (
266                    normalized,
267                    operation,
268                    label,
269                    GoalExecutor::Update {
270                        engine: self.engine.clone(),
271                        session: ctx.session,
272                        params,
273                    },
274                )
275            }
276            _ => return Err(ToolError::execution("goal provider received another tool")),
277        };
278
279        PreparedTool::new(operation, normalized_arguments, None, Box::new(executor))?
280            .with_policy_labels(vec![policy_label])
281    }

284#[async_trait]
285impl PreparedExecutor for GoalExecutor {
286    async fn execute(
287        self: Box<Self>,
288        context: ToolExecutionContext,
289    ) -> Result<cookie_agent_runtime_api::ToolCompletion, ToolError> {
290        let result: Result<ToolResult, ToolError> = async move {
291            if context.cancellation.is_cancelled() {
292                return Err(ToolError::execution("goal tool was cancelled"));
293            }
294            let (title, result, goal) = match *self {
295                Self::Get { engine, session } => {
296                    if context.session != session {
297                        return Err(ToolError::operation_changed("session changed"));
298                    }
299                    let result = engine.goal_get(session).await.map_err(tool_error)?;
300                    (
301                        "Current goal",
302                        serde_json::to_value(&result).map_err(tool_error)?,
303                        result.goal,
304                    )
305                }
306                Self::Set {
307                    engine,
308                    session,
309                    params,
310                } => {
311                    if context.session != session {
312                        return Err(ToolError::operation_changed("session changed"));
313                    }
314                    let result = engine.goal_set(session, params).await.map_err(tool_error)?;
315                    (
316                        "Started goal",
317                        serde_json::to_value(&result).map_err(tool_error)?,
318                        Some(result.goal),
319                    )
320                }
321                Self::Update {
322                    engine,
323                    session,
324                    params,
325                } => {
326                    if context.session != session {
327                        return Err(ToolError::operation_changed("session changed"));
328                    }
329                    let result = engine
330                        .goal_update(session, params)
331                        .await
332                        .map_err(tool_error)?;
333                    (
334                        "Updated goal checklist",
335                        serde_json::to_value(&result).map_err(tool_error)?,
336                        Some(result.goal),
337                    )
338                }
339            };
340            let mut result = goal_result(title, result)?;
341            result.display = markdown_display(&goal_markdown(goal.as_ref()));
342            Ok(result)
343        }
344        .await;
345        result.map(cookie_agent_runtime_api::ToolCompletion::single)
346    }
347}

349/// How clients show a goal tool call.
350fn goal_presentation(call: &ToolCall) -> cookie_agent_protocol::PreparedPresentation {
351    let hint = cookie_agent_protocol::ToolHint::GOAL;
352    match call.name.as_str() {
353        "goal_get" if parse_args::<GoalGetParams>(&call.name, call.arguments.clone()).is_ok() => {
354            base_presentation("Read goal", hint, "current goal", Vec::new())
355        }
356        "goal_set" => {
357            match parse_args_strict::<GoalSetParams>(&call.name, call.arguments.clone()) {
358                Ok(params) => base_presentation(
359                    "Start goal",
360                    hint,
361                    &params.objective,
362                    params
363                        .items
364                        .into_iter()
365                        .map(|item| {
366                            (
367                                if item.finished { "done" } else { "to do" },
368                                item.description,
369                            )
370                        })
371                        .collect(),
372                ),
373                Err(_) => cookie_agent_runtime_api::tool_title_only(&call.name),
374            }
375        }
376        "goal_update" => {
377            match parse_args_strict::<GoalUpdateParams>(&call.name, call.arguments.clone()) {
378                Ok(params) => base_presentation(
379                    "Update goal checklist",
380                    hint,
381                    "current goal",
382                    params
383                        .items
384                        .into_iter()
385                        .map(|item| {
386                            (
387                                if item.finished { "done" } else { "to do" },
388                                item.description,
389                            )
390                        })
391                        .collect(),
392                ),
393                Err(_) => cookie_agent_runtime_api::tool_title_only(&call.name),
394            }
395        }
396        _ => cookie_agent_runtime_api::tool_title_only(&call.name),
397    }
398}

400/// A goal as clients see it: its objective, status, and checklist.
401fn goal_markdown(goal: Option<&cookie_agent_protocol::GoalState>) -> String {
402    let Some(goal) = goal else {
403        return "No goal is set.".into();
404    };
405    let status = serde_json::to_value(goal.status)
406        .ok()
407        .and_then(|status| status.as_str().map(str::to_owned))
408        .unwrap_or_default();
409    let mut text = format!(
410        "**{}**\n\n{status} · revision {}\n",
411        goal.objective.trim(),
412        goal.revision
413    );
414    if !goal.items.is_empty() {
415        text.push('\n');
416    }
417    for item in &goal.items {
418        let mark = if item.finished { 'x' } else { ' ' };
419        text.push_str(&format!("- [{mark}] {}\n", item.description.trim()));
420    }
421    text
422}
