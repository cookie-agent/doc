40    pub(crate) fn get_spec() -> ToolSpec {
41        ToolSpec {
42            background: false,
43            background_default: false,
44collision: Default::default(),
45output: Default::default(),
46            concurrency: cookie_agent_runtime_api::ToolConcurrency::Parallel,
47            result_truncation: cookie_agent_runtime_api::ToolResultTruncationPolicy::OptOut,
48            name: "goal_get".into(),
49            permission_name: None,
50            description: "Read the root session's objective, lifecycle status, revision, and full checklist. The root checklist is authoritative; verify work directly or through subagents before treating an item as finished.".into(),
51            parameters: schema::<GoalGetParams>(),
52        }
53    }

55    pub(crate) fn update_spec() -> ToolSpec {
56        ToolSpec {
57            background: false,
58            background_default: false,
59collision: Default::default(),
60output: Default::default(),
61            concurrency: Default::default(),
62            result_truncation: cookie_agent_runtime_api::ToolResultTruncationPolicy::OptOut,
63            name: "goal_update".into(),
64            permission_name: None,
65            description: "Replace the current root goal's full checklist. The current active or paused goal is selected when this update executes, even if the goal changed since this run started. The last accepted replacement wins. This cannot change the objective or issue lifecycle commands; a nonempty all-finished checklist completes the goal. Verify evidence directly or through subagents before marking items finished; the root checklist is authoritative over subagent self-reports.".into(),
66            parameters: strict_schema::<GoalUpdateParams>(),
67        }
68    }

70    fn update_operation(
71        session: SessionId,
72        params: &GoalUpdateParams,
73    ) -> Result<PreparedOperationIdentity, ToolError> {
74        let binding = serde_json::to_vec(&(session, params)).map_err(tool_error)?;
75        let resource = prepared_resource(
76            PermissionAction::WRITE,
77            "goal",
78            b"goal:current",
79            &binding,
80            PreparedBindingLifetime::RestartStable,
81            ApprovalResourceSource::PrimaryOperation,
82        )?;
83        prepared_operation(
84            "goal_update",
85            params,
86            vec![(PermissionAction::WRITE, "replace_checklist")],
87            vec![resource],
88            session.to_string().as_bytes(),
89        )
90    }

160    async fn prepare(
161        &self,
162        ctx: ToolPreparationContext,
163        call: ToolCall,
164    ) -> Result<PreparedTool, ToolError> {
165        let (normalized_arguments, operation, policy_label, executor) = match call.name.as_str() {
166            "goal_get" => {
167                let params: GoalGetParams = parse_args(&call.name, call.arguments)?;
168                let label = "goal:current".to_owned();
169                let resource = prepared_resource(
170                    PermissionAction::READ,
171                    "goal",
172                    label.as_bytes(),
173                    ctx.session.to_string().as_bytes(),
174                    PreparedBindingLifetime::RestartStable,
175                    ApprovalResourceSource::PrimaryOperation,
176                )?;
177                let operation = prepared_operation(
178                    "goal_get",
179                    &params,
180                    vec![(PermissionAction::READ, "get")],
181                    vec![resource],
182                    ctx.session.to_string().as_bytes(),
183                )?;
184                (
185                    serde_json::to_value(params).map_err(tool_error)?,
186                    operation,
187                    label,
188                    GoalExecutor::Get {
189                        engine: self.engine.clone(),
190                        session: ctx.session,
191                    },
192                )
193            }
194            "goal_update" => {
195                let params: GoalUpdateParams = parse_args_strict(&call.name, call.arguments)?;
196                let label = "goal:current".to_owned();
197                let normalized = serde_json::to_value(&params).map_err(tool_error)?;
198                let operation = Self::update_operation(ctx.session, &params)?;
199                (
200                    normalized,
201                    operation,
202                    label,
203                    GoalExecutor::Update {
204                        engine: self.engine.clone(),
205                        session: ctx.session,
206                        params,
207                    },
208                )
209            }
210            _ => return Err(ToolError::execution("goal provider received another tool")),
211        };
212
213        PreparedTool::new(operation, normalized_arguments, None, Box::new(executor))?
214            .with_policy_labels(vec![policy_label])
215    }
216}

218#[async_trait]
219impl PreparedExecutor for GoalExecutor {
220    async fn execute(
221        self: Box<Self>,
222        context: ToolExecutionContext,
223    ) -> Result<cookie_agent_runtime_api::ToolCompletion, ToolError> {
224        let result: Result<ToolResult, ToolError> = async move {
225            if context.cancellation.is_cancelled() {
226                return Err(ToolError::execution("goal tool was cancelled"));
227            }
228            let (title, result, goal) = match *self {
229                Self::Get { engine, session } => {
230                    if context.session != session {
231                        return Err(ToolError::operation_changed("session changed"));
232                    }
233                    let result = engine.goal_get(session).await.map_err(tool_error)?;
234                    (
235                        "Current goal",
236                        serde_json::to_value(&result).map_err(tool_error)?,
237                        result.goal,
238                    )
239                }
240                Self::Update {
241                    engine,
242                    session,
243                    params,
244                } => {
245                    if context.session != session {
246                        return Err(ToolError::operation_changed("session changed"));
247                    }
248                    let result = engine
249                        .goal_update(session, params)
250                        .await
251                        .map_err(tool_error)?;
252                    (
253                        "Updated goal checklist",
254                        serde_json::to_value(&result).map_err(tool_error)?,
255                        Some(result.goal),
256                    )
257                }
258            };
259            let mut result = goal_result(title, result)?;
260            result.display = markdown_display(&goal_markdown(goal.as_ref()));
261            Ok(result)
262        }
263        .await;
264        result.map(cookie_agent_runtime_api::ToolCompletion::single)
265    }
266}

268/// How clients show a goal tool call.
269fn goal_presentation(call: &ToolCall) -> cookie_agent_protocol::PreparedPresentation {
270    let hint = cookie_agent_protocol::ToolHint::GOAL;
271    match call.name.as_str() {
272        "goal_get" if parse_args::<GoalGetParams>(&call.name, call.arguments.clone()).is_ok() => {
273            base_presentation("Read goal", hint, "current goal", Vec::new())
274        }
275        "goal_update" => {
276            match parse_args_strict::<GoalUpdateParams>(&call.name, call.arguments.clone()) {
277                Ok(params) => base_presentation(
278                    "Update goal checklist",
279                    hint,
280                    "current goal",
281                    params
282                        .items
283                        .into_iter()
284                        .map(|item| {
285                            (
286                                if item.finished { "done" } else { "to do" },
287                                item.description,
288                            )
289                        })
290                        .collect(),
291                ),
292                Err(_) => cookie_agent_runtime_api::tool_title_only(&call.name),
293            }
294        }
295        _ => cookie_agent_runtime_api::tool_title_only(&call.name),
296    }
297}

299/// A goal as clients see it: its objective, status, and checklist.
300fn goal_markdown(goal: Option<&cookie_agent_protocol::GoalState>) -> String {
301    let Some(goal) = goal else {
302        return "No goal is set.".into();
303    };
304    let status = serde_json::to_value(goal.status)
305        .ok()
306        .and_then(|status| status.as_str().map(str::to_owned))
307        .unwrap_or_default();
308    let mut text = format!(
309        "**{}**\n\n{status} · revision {}\n",
310        goal.objective.trim(),
311        goal.revision
312    );
313    if !goal.items.is_empty() {
314        text.push('\n');
315    }
316    for item in &goal.items {
317        let mark = if item.finished { 'x' } else { ' ' };
318        text.push_str(&format!("- [{mark}] {}\n", item.description.trim()));
319    }
320    text
321}

323fn goal_result(title: &str, result: serde_json::Value) -> Result<ToolResult, ToolError> {
324    Ok(ToolResult {
325        display: Vec::new(),

327        retained_output: None,

330        metadata: serde_json::Value::Null,

332        additional_messages: Vec::new(),
333        loaded_tools: Vec::new(),
334    })

476    fn update_fingerprint_binds_session_and_exact_items_only() {
477        use cookie_agent_protocol::{
478            GoalItem, GoalUpdateParams, OperationFingerprint, SessionId, Sha256Digest,
479        };
480
481        let session = SessionId::new_v7();
482        let params = GoalUpdateParams {
483            items: vec![GoalItem {
484                description: "Verify".into(),
485                finished: false,
486            }],
487        };
488        let operation = GoalTools::update_operation(session, &params).unwrap();
489        let fingerprint = OperationFingerprint::from_prepared_operation(&operation);
490        assert_eq!(
491            fingerprint,
492            OperationFingerprint::from_prepared_operation(
493                &GoalTools::update_operation(session, &params).unwrap()
494            )
495        );
496        assert_ne!(
497            fingerprint,
498            OperationFingerprint::from_prepared_operation(
499                &GoalTools::update_operation(SessionId::new_v7(), &params).unwrap()
500            )
501        );
502        let mut changed = params.clone();
503        changed.items[0].finished = true;
504        assert_ne!(
505            fingerprint,
506            OperationFingerprint::from_prepared_operation(
507                &GoalTools::update_operation(session, &changed).unwrap()
508            )
509        );
510        changed = params.clone();
511        changed.items[0].description = "Verify again".into();
512        assert_ne!(
513            GoalTools::update_operation(session, &params)
514                .unwrap()
515                .resources()[0]
516                .binding_digest,
517            GoalTools::update_operation(session, &changed)
518                .unwrap()
519                .resources()[0]
520                .binding_digest
521        );
522        let wire = serde_json::to_value(&operation).unwrap();
523        assert_eq!(
524            wire["normalized_arguments_digest"],
525            serde_json::to_value(Sha256Digest::of_bytes(
526                &serde_json::to_vec(&params).unwrap()
527            ))
528            .unwrap()
529        );
530        assert_eq!(
531            wire["execution_context_digest"],
532            serde_json::to_value(Sha256Digest::of_bytes(session.to_string().as_bytes())).unwrap()
533        );
534        assert_eq!(
535            operation.resources()[0].capability,
536            cookie_agent_protocol::PermissionAction::WRITE
537        );
538    }
