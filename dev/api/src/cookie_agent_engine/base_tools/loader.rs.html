1//! `tool_search`, the loader tool of deferred tools
2//! (`docs/site/specs/stable-prefix.md` section 2).

4//! A base-layer tool that declares no permission: it sees and loads only the
5//! tools the session's latest request already offers. The engine publishes

27/// How many matches a keyword query loads when the call sets no `limit`.
28const DEFAULT_SEARCH_LIMIT: usize = 5;
29
30/// The prefix of a query that names the tools to load exactly.
31const SELECT_PREFIX: &str = "select:";
32
33#[derive(Debug, Deserialize, JsonSchema, Serialize)]
34#[serde(deny_unknown_fields)]
35struct ToolSearchArgs {
36    /// "select:<name>,<name>" loads those tools by exact name; any other text is keywords, such as "browser click", and loads the best matches. "+word" requires the word in a tool's name, as in "+slack send".
37    query: String,
38    /// At most how many keyword matches to load; 5 when absent, and every match when 0 or less. A select: query loads every name it lists.
39    #[serde(default, skip_serializing_if = "Option::is_none")]
40    limit: Option<i64>,
41}

48impl LoaderTools {
49    pub(crate) fn new(engine: Engine) -> Self {
50        Self { engine }
51    }
52
53    #[must_use]
54    pub fn search_spec() -> ToolSpec {
55        ToolSpec {
56            background: false,
57            background_default: false,
58            collision: Default::default(),
59            output: Default::default(),
60            concurrency: cookie_agent_runtime_api::ToolConcurrency::Parallel,
61            result_truncation: Default::default(),
62            name: TOOL_SEARCH.into(),
63            permission_name: None,
64            description: "Load deferred tools so you can call them. Use \"select:<name>,<name>\" to load tools by the exact names an announcement lists, or keywords to find and load the best matches by name, group, description, and parameters; \"+word\" requires the word in a tool's name. Loaded tools stay callable.".into(),
65            parameters: schema::<ToolSearchArgs>(),
66        }
67    }
68
69    fn check(name: &str) -> Result<(), ToolError> {
70        if name == TOOL_SEARCH {
71            Ok(())
72        } else {
73            Err(ToolError::execution(
74                "loader provider received another tool",
75            ))
76        }
77    }
78
79    fn argument(name: &str, arguments: &serde_json::Value) -> Result<String, ToolError> {
80        Self::check(name)?;
81        Ok(parse_args::<ToolSearchArgs>(TOOL_SEARCH, arguments.clone())?.query)
82    }
83}

95    fn get_permission_name(tool_name: &str) -> Result<Option<&'static str>, ToolError> {
96        Self::check(tool_name)?;
97        Ok(None)
98    }

100    fn get_permission_resource(
101        &self,
102        name: &str,
103        arguments: &serde_json::Value,
104    ) -> Result<(Option<&'static str>, Option<String>), ToolError> {
105        Ok((None, Some(Self::argument(name, arguments)?)))
106    }
107
108    fn get_display_argument(
109        &self,
110        name: &str,
111        arguments: &serde_json::Value,
112    ) -> Result<String, ToolError> {
113        Self::argument(name, arguments)
114    }
115
116    fn presentation(&self, call: &ToolCall) -> cookie_agent_protocol::PreparedPresentation {
117        match Self::argument(&call.name, &call.arguments) {
118            Ok(argument) => match selected(&argument) {
119                Some(names) => {
120                    let names = names.join(", ");
121                    base_presentation(
122                        &format!("Load {names}"),
123                        cookie_agent_protocol::ToolHint::GENERIC,
124                        &names,
125                        vec![("query", argument.clone())],
126                    )
127                }
128                None => base_presentation(
129                    &format!("Search tools: {argument}"),
130                    cookie_agent_protocol::ToolHint::GENERIC,
131                    &argument,
132                    vec![("query", argument.clone())],
133                ),
134            },
135            Err(_) => cookie_agent_runtime_api::tool_title_only(&call.name),
136        }
137    }
138
139    async fn prepare(
140        &self,
141        ctx: ToolPreparationContext,
142        call: ToolCall,
143    ) -> Result<PreparedTool, ToolError> {
144        Self::check(&call.name)?;
145        let args: ToolSearchArgs = parse_args(TOOL_SEARCH, call.arguments)?;
146        let request = match selected(&args.query) {
147            Some(names) if names.is_empty() => {
148                return Err(ToolError::execution("select: must name at least one tool"));
149            }
150            Some(names) => LoaderRequest::Select { names },
151            None => LoaderRequest::Search {
152                query: args.query.clone(),
153                limit: match args.limit {
154                    Some(limit) if limit <= 0 => usize::MAX,
155                    Some(limit) => usize::try_from(limit).unwrap_or(usize::MAX),
156                    None => DEFAULT_SEARCH_LIMIT,
157                },
158            },
159        };
160        let normalized = serde_json::to_value(&args).map_err(tool_error)?;
161        // No rule is evaluated; the capability only names the operation.
162        let action = PermissionAction::new(TOOL_SEARCH).map_err(tool_error)?;
163        let binding = serde_json::to_vec(&normalized).map_err(tool_error)?;
164        let resource = prepared_resource(
165            action.clone(),
166            "deferred-tools",
167            TOOL_SEARCH.as_bytes(),
168            &binding,
169            PreparedBindingLifetime::ProcessLocal,
170            ApprovalResourceSource::PrimaryOperation,
171        )?;
172        let operation = prepared_operation(
173            TOOL_SEARCH,
174            &normalized,
175            vec![(action, "load")],
176            vec![resource],
177            ctx.session.to_string().as_bytes(),
178        )?;
179        PreparedTool::new(
180            operation,
181            normalized,
182            None,
183            Box::new(LoaderExecutor {
184                engine: self.engine.clone(),
185                request,
186            }),
187        )
188    }
189}

191/// The names a `select:` query lists, or `None` for a keyword query.
192fn selected(query: &str) -> Option<Vec<String>> {
193    let names = query.trim_start().strip_prefix(SELECT_PREFIX)?;
194    Some(
195        names
196            .split(',')
197            .map(str::trim)
198            .filter(|name| !name.is_empty())
199            .map(str::to_owned)
200            .collect(),
201    )
202}

214#[async_trait]
215impl PreparedExecutor for LoaderExecutor {
216    async fn execute(
217        self: Box<Self>,
218        context: ToolExecutionContext,
219    ) -> Result<cookie_agent_runtime_api::ToolCompletion, ToolError> {
220        let view = self.engine.deferral_view(context.session);
221        let result = match self.request {
222            LoaderRequest::Search { query, limit } => search_result(&view, &query, limit),
223            LoaderRequest::Select { names } => load_result(&view, &names, None),
224        };
225        Ok(cookie_agent_runtime_api::ToolCompletion::single(result))
226    }
227}

243/// The first sentence of a description, on one line.
244fn first_sentence(description: &str) -> String {
245    let line = description.lines().next().unwrap_or_default();
246    let end = line
247        .char_indices()
248        .find(|(index, character)| {
249            *character == '.'
250                && line[index + 1..]
251                    .chars()
252                    .next()
253                    .is_none_or(char::is_whitespace)
254        })
255        .map_or(line.len(), |(index, _)| index + 1);
256    line[..end].trim().to_owned()
257}

259/// The words a tool is found by (section 5.1).
260fn haystack(entry: &DeferredEntry) -> String {
261    let mut text = format!(
262        "{} {} {} {} {}",
263        entry.name,
264        entry.name.replace(['_', '-'], " "),
265        entry.group.name,
266        entry.group.summary.as_deref().unwrap_or_default(),
267        entry.description
268    );
269    if let Some(properties) = entry
270        .parameters
271        .get("properties")
272        .and_then(serde_json::Value::as_object)
273    {
274        for (name, property) in properties {
275            text.push(' ');
276            text.push_str(name);
277            if let Some(description) = property
278                .get("description")
279                .and_then(serde_json::Value::as_str)
280            {
281                text.push(' ');
282                text.push_str(description);
283            }
284        }
285    }
286    text.to_lowercase()
287}

289/// Loads the best keyword matches for `query`, at most `limit` of them.
290pub(crate) fn search_result(view: &DeferralView, query: &str, limit: usize) -> ToolResult {
291    let words = query
292        .split_whitespace()
293        .map(str::to_lowercase)
294        .collect::<Vec<_>>();
295    // `+word` must be in the tool's name; every word counts toward the rank.
296    let required = words
297        .iter()
298        .filter_map(|word| word.strip_prefix('+'))
299        .filter(|word| !word.is_empty())
300        .collect::<Vec<_>>();
301    let terms = words
302        .iter()
303        .map(|word| word.trim_start_matches('+'))
304        .filter(|word| !word.is_empty())
305        .collect::<Vec<_>>();
306    if view.deferred.is_empty() {
307        return result(
308            "No loadable tools",
309            "No tools are loadable in this session.".into(),
310            Vec::new(),
311        );
312    }
313    let mut matches = view
314        .deferred
315        .iter()
316        .filter(|entry| {
317            let name = entry.name.to_lowercase();
318            required.iter().all(|word| name.contains(word))
319        })
320        .filter_map(|entry| {
321            let haystack = haystack(entry);
322            let score = terms
323                .iter()
324                .filter(|term| haystack.contains(**term))
325                .count();
326            (score > 0).then_some((score, entry))
327        })
328        .collect::<Vec<_>>();
329    matches.sort_by(|(left_score, left), (right_score, right)| {
330        right_score
331            .cmp(left_score)
332            .then_with(|| left.name.cmp(&right.name))
333    });
334    if matches.is_empty() {
335        return result(
336            "No matching tools",
337            format!(
338                "No loadable tools match \"{query}\". Try other words, a group's name, or select:<name>."
339            ),
340            Vec::new(),
341        );
342    }
343    let total = matches.len();
344    let names = matches
345        .into_iter()
346        .take(limit)
347        .map(|(_, entry)| entry.name.clone())
348        .collect::<Vec<_>>();
349    let more = (names.len() < total).then(|| {
350        format!(
351            "{} more match; raise limit or select them by name.",
352            total - names.len()
353        )
354    });
355    load_result(view, &names, more)
356}

358/// Loads `names`, recording the tools that newly load, and says why the
359/// others did not; `more` ends the output.
360pub(crate) fn load_result(
361    view: &DeferralView,
362    names: &[String],
363    more: Option<String>,
364) -> ToolResult {
365    let mut loaded = Vec::new();
366    let mut lines = Vec::new();
367    let mut notes = Vec::new();
368    let mut seen = std::collections::HashSet::new();
369    for name in names {
370        if !seen.insert(name.as_str()) {
371            continue;
372        }
373        let entry = view.deferred.iter().find(|entry| &entry.name == name);
374        match entry {
375            _ if view.eager.contains(name) => {
376                notes.push(format!("{name}: already callable, no need to load it"));
377            }
378            None => notes.push(format!("{name}: no loadable tool has this name")),
379            Some(entry) => {
380                let already = if view.loaded.contains(name) {
381                    " (already loaded)"
382                } else {
383                    loaded.push(name.clone());
384                    ""
385                };
386                lines.push(format!(
387                    "- {} [{}]{already}: {}",
388                    entry.name,
389                    entry.group.name,
390                    first_sentence(&entry.description)
391                ));
392            }
393        }
394    }
395    let mut output = match loaded.len() {
396        0 => "Loaded no new tools.".to_owned(),
397        1 => "Loaded 1 tool:".to_owned(),
398        count => format!("Loaded {count} tools:"),
399    };
400    for line in lines {
401        output.push('\n');
402        output.push_str(&line);
403    }
404    for note in notes.into_iter().chain(more) {
405        output.push('\n');
406        output.push_str(&note);
407    }
408    let instructions = provider_instructions(view, &loaded);
409    let mut result = result(
410        if loaded.is_empty() {
411            "No tools loaded"
412        } else {
413            "Tools loaded"
414        },
415        output,
416        loaded,
417    );
418    result.additional_messages.extend(instructions);
419    result
420}

422/// The prompt sections of the deferred providers whose tools `loaded`
423/// loads first, as one message after the result (deferred tools in cookie,
424/// section 7.3). Stored with the result, so history never shifts.
425fn provider_instructions(
426    view: &DeferralView,
427    loaded: &[String],
428) -> Option<cookie_agent_protocol::ToolEmittedMessage> {
429    let mut providers = Vec::new();
430    for name in loaded {
431        if let Some(entry) = view.deferred.iter().find(|entry| &entry.name == name)
432            && view.sections.contains_key(&entry.provider)
433            && !providers.contains(&&entry.provider)
434        {
435            providers.push(&entry.provider);
436        }
437    }
438    let blocks = providers
439        .into_iter()
440        .filter_map(|provider| view.sections.get(provider))
441        .cloned()
442        .collect::<Vec<_>>();
443    if blocks.is_empty() {
444        return None;
445    }
446    cookie_agent_protocol::ToolEmittedMessage::new(
447        cookie_agent_protocol::ToolEmittedMessageRole::System,
448        vec![cookie_agent_protocol::ToolEmittedContent::Text(
449            blocks.join("\n\n"),
450        )],
451    )
452    .ok()
453}
