1//! The memory tools of the `global` runtime (`docs/site/specs/memory.md`
2//! section 8, as section 16.9 amends it): `memory_search` and
3//! `memory_read` under the `memory` action, `memory_write`, `memory_edit`,
4//! and `memory_delete` under `memory_write`, each with the memory URL as
5//! resource. `memory_search` is checked per result instead (section 9.3),
6//! so it declares no permission. An edit applies when its `oldString`
7//! occurs once in the body, which preparation checks before the call is
8//! approved; a replacing write needs the session to have seen the memory as
9//! it is (section 16.9.8), and the store refuses it otherwise. A delete
10//! deletes.
11//!
12//! `memory_read` pages a memory's body as the coding runtime's `read` pages
13//! a file, from a 1-based `offset`, and the write tools word their results
14//! and errors as `write` and `edit` do
15//! ([`cookie_agent_runtime_api::text_tools`]).
16//!
17//! The provider also contributes the `<tool_instructions
18//! provider="builtin.memory">` section (section 16.9.1): part A, what the
19//! snapshot includes and when to search, for every run offered a memory
20//! tool, and part B, the rubric, for a run also offered the write tools.
21//!
22//! The provider is built with the `global` runtime, before the engine
23//! exists, and holds the engine through a handle bound once the engine is
24//! made. Composition publishes its tools only to runs that have memory,
25//! and its write tools only to runs that can both read and write some
26//! memory ([`crate::runtime::memory_runs`]); the permission rules alone
27//! decide what a run may change (section 16.9.11).

106/// The body of the memory provider's section for a run with `access`:
107/// part A for a root or a `delegated` run, and part B when the run may
108/// write. `None` for a run offered no memory tool.
109pub(crate) fn prompt_section(access: MemoryAccess, delegated: bool) -> Option<String> {
110    if !access.available {
111        return None;
112    }
113    let part_a = if delegated {
114        PART_A_DELEGATED
115    } else {
116        PART_A_ROOT
117    };
118    Some(if access.writable {
119        format!("{part_a}\n\n{PART_B}")
120    } else {
121        part_a.to_owned()
122    })
123}

190fn spec(
191    name: &str,
192    permission_name: Option<&str>,
193    description: &str,
194    parameters: serde_json::Value,
195) -> ToolSpec {
196    let reads = matches!(name, SEARCH | READ);
197    ToolSpec {
198        background: false,
199        background_default: false,
200        collision: Default::default(),
201        output: Default::default(),
202        concurrency: if reads {
203            ToolConcurrency::Parallel

214        permission_name: permission_name.map(str::to_owned),
215        description: description.into(),
216        parameters,
217    }

301/// `keywords` as memory_write takes them: a list, or one string of
302/// comma-separated keywords, which agents often send for a list (some send
303/// the list's JSON as a string, which reads as that list).
304fn deserialize_keywords<'de, D>(deserializer: D) -> Result<Vec<String>, D::Error>
305where
306    D: serde::Deserializer<'de>,
307{
308    #[derive(Deserialize)]
309    #[serde(untagged)]
310    enum Keywords {
311        List(Vec<String>),
312        Text(String),
313    }
314    Ok(match Keywords::deserialize(deserializer)? {
315        Keywords::List(keywords) => keywords,
316        Keywords::Text(text) => split_keywords(&text),
317    })
318}
319
320/// The keywords of `text`: the list its JSON holds, else its
321/// comma-separated parts, trimmed, the empty ones left out.
322fn split_keywords(text: &str) -> Vec<String> {
323    let text = text.trim();
324    if text.starts_with('[')
325        && let Ok(keywords) = serde_json::from_str::<Vec<String>>(text)
326    {
327        return keywords;
328    }
329    text.split(',')
330        .map(str::trim)
331        .filter(|keyword| !keyword.is_empty())
332        .map(str::to_owned)
333        .collect()
334}

386impl MemoryCall {
387    /// The call's URL; `None` for a search.
388    fn url(&self) -> Option<&MemoryUrl> {
389        match self {
390            Self::Search { .. } => None,
391            Self::Read { url, .. }
392            | Self::Write { url, .. }
393            | Self::Edit { url, .. }
394            | Self::Delete { url } => Some(url),
395        }
396    }

403/// Checks a call's arguments: strict members, a valid URL and memory, the
404/// bounds of a search, and a `reason` exactly when the write says `new`.
405fn check(name: &str, arguments: serde_json::Value) -> Result<MemoryCall, ToolError> {
406    match name {
407        SEARCH => {
408            let args: SearchArgs = parse_args(SEARCH, arguments)?;
409            let length = args.query.chars().count();
410            if args.query.trim().is_empty() || length > MAX_QUERY_CHARS {
411                return Err(ToolError::execution(format!(
412                    "the query must have 1 to {MAX_QUERY_CHARS} characters and a search term; it has {length}"
413                )));
414            }
415            let limit = args.limit.unwrap_or(DEFAULT_LIMIT);
416            if !(1..=MAX_LIMIT).contains(&limit) {
417                return Err(ToolError::execution(format!(
418                    "limit must be 1 to {MAX_LIMIT}; it is {limit}"
419                )));
420            }
421            Ok(MemoryCall::Search {
422                query: args.query,
423                layer: args.layer.map(LayerArg::layer),
424                limit: limit as usize,
425            })
426        }
427        READ => {
428            let args: ReadArgs = parse_args(READ, arguments)?;
429            Ok(MemoryCall::Read {
430                url: parse_url(&args.url)?,
431                window: LineWindow::new(args.offset, args.limit)?,
432            })
433        }
434        WRITE => {
435            let args: WriteArgs = parse_args(WRITE, arguments)?;
436            let url = parse_url(&args.url)?;
437            match (&args.reason, args.new) {
438                (None, true) => {
439                    return Err(ToolError::execution(
440                        "`new: true` needs a one-line `reason` saying why this memory is not one of the near ones",
441                    ));
442                }
443                (Some(_), false) => {
444                    return Err(ToolError::execution(
445                        "`reason` goes with `new: true`, for a memory that looks like ones that exist",
446                    ));
447                }
448                (Some(reason), true)
449                    if reason.trim().is_empty()
450                        || reason.contains('\n')
451                        || reason.chars().count() > MAX_REASON_CHARS =>
452                {
453                    return Err(ToolError::execution(format!(
454                        "`reason` must be one line of 1 to {MAX_REASON_CHARS} characters"
455                    )));
456                }
457                _ => {}
458            }
459            let described = args.description.is_some();
460            let keyworded = !args.keywords.is_empty();
461            // An agent file's description, and a project overview's, is the
462            // daemon's, which the store puts in place of this one when it
463            // writes.
464            let description = match (args.description, &url) {
465                (Some(description), _) => description,
466                (None, MemoryUrl::Agent { agent }) => format!("Memory of the `{agent}` agent"),
467                (None, MemoryUrl::Project { project, .. })
468                    if crate::memory::projects::is_overview(&url) =>
469                {
470                    crate::memory::projects::overview_description(project.as_str())
471                }
472                (None, _) => {
473                    return Err(ToolError::execution(format!(
474                        "{url} needs a description: one line, 1 to 200 characters"
475                    )));
476                }
477            };
478            let document =
479                MemoryDocument::new(description, args.keywords, args.body).map_err(|reason| {
480                    ToolError::execution(format!("{url} would not be a valid memory: {reason}"))
481                })?;
482            Ok(MemoryCall::Write {
483                url,
484                document,
485                described,
486                keyworded,
487                new: args.new,
488            })
489        }
490        EDIT => {
491            let args: EditArgs = parse_args(EDIT, arguments)?;
492            Ok(MemoryCall::Edit {
493                url: parse_url(&args.url)?,
494                old: args.old_string,
495                new: args.new_string,
496            })
497        }
498        DELETE => {
499            let args: DeleteArgs = parse_args(DELETE, arguments)?;
500            Ok(MemoryCall::Delete {
501                url: parse_url(&args.url)?,
502            })
503        }
504        other => Err(ToolError::execution(format!(
505            "the memory provider received another tool, `{other}`"
506        ))),
507    }
508}

510fn permission_name(tool: &str) -> Result<Option<&'static str>, ToolError> {
511    match tool {
512        SEARCH => Ok(None),
513        READ => Ok(Some("memory")),
514        WRITE | EDIT | DELETE => Ok(Some("memory_write")),
515        other => Err(ToolError::execution(format!(
516            "the memory provider received another tool, `{other}`"
517        ))),

568    async fn prepare(
569        &self,
570        ctx: ToolPreparationContext,
571        call: ToolCall,
572    ) -> Result<PreparedTool, ToolError> {
573        let engine = self.engine()?;
574        let memory_call = check(&call.name, call.arguments.clone())?;
575        let preview = {
576            let root = engine.memory_root().clone();
577            let shown = preview_input(&memory_call);
578            // An edit that cannot apply fails here, before it is approved,
579            // as the coding runtime's `edit` does.
580            let edit = match &memory_call {
581                MemoryCall::Edit { url, old, .. } => Some((url.clone(), old.clone())),
582                _ => None,
583            };
584            blocking(move || {
585                if let Some((url, old)) = edit {
586                    root.check_edit(&url, &old)?;
587                }
588                Ok::<_, StoreError>(shown.and_then(|shown| preview(&root, shown)))
589            })
590            .await?
591            .map_err(store_error)?
592        };
593        let presentation =
594            preview.map(|preview| call_presentation(&call, true).with_preview([preview]));
595        let (action, operation_name) = match &memory_call {
596            MemoryCall::Search { .. } => (PermissionAction::MEMORY, "search"),
597            MemoryCall::Read { .. } => (PermissionAction::MEMORY, "read"),
598            MemoryCall::Write { .. } => (PermissionAction::MEMORY_WRITE, "write"),
599            MemoryCall::Edit { .. } => (PermissionAction::MEMORY_WRITE, "edit"),
600            MemoryCall::Delete { .. } => (PermissionAction::MEMORY_WRITE, "delete"),
601        };
602        let label = memory_call
603            .url()
604            .map_or_else(|| "search".to_owned(), ToString::to_string);
605        let resource = prepared_resource(
606            action.clone(),
607            "memory",
608            label.as_bytes(),
609            label.as_bytes(),
610            PreparedBindingLifetime::RestartStable,
611            ApprovalResourceSource::PrimaryOperation,
612        )?;
613        let operation = prepared_operation(
614            &call.name,
615            &call.arguments,
616            vec![(action, operation_name)],
617            vec![resource],
618            &ctx.context_bytes(),
619        )?;
620        let mut prepared = PreparedTool::new(
621            operation,
622            call.arguments,
623            None,
624            Box::new(MemoryExecutor {
625                engine,
626                call: memory_call,
627            }),
628        )?;
629        if let Some(presentation) = presentation {
630            prepared = prepared.with_presentation(presentation);
631        }
632        prepared.with_policy_labels(vec![label])
633    }
634}

636/// What a write, edit, or delete previews, read before the call is
637/// approved: the URL, and the new memory a write gives or the replacement
638/// an edit makes.
639enum PreviewInput {
640    Write {
641        url: MemoryUrl,
642        text: String,
643    },
644    Edit {
645        url: MemoryUrl,
646        old: String,
647        new: String,
648    },
649    Delete {
650        url: MemoryUrl,
651    },
652}
653
654fn preview_input(call: &MemoryCall) -> Option<PreviewInput> {
655    match call {
656        MemoryCall::Search { .. } | MemoryCall::Read { .. } => None,
657        MemoryCall::Write { url, document, .. } => Some(PreviewInput::Write {
658            url: url.clone(),
659            text: shown_text(&document.render()),
660        }),
661        MemoryCall::Edit { url, old, new } => Some(PreviewInput::Edit {
662            url: url.clone(),
663            old: old.clone(),
664            new: new.clone(),
665        }),
666        MemoryCall::Delete { url } => Some(PreviewInput::Delete { url: url.clone() }),
667    }
668}

670/// The diff a write, edit, or delete would make to the memory as it is
671/// now, as the coding runtime's tools preview theirs. `None` when the call
672/// will fail anyway (an edit's `old` not found once, a memory that is not
673/// there to edit or delete) or the memory cannot be read.
674fn preview(root: &crate::memory::MemoryRoot, input: PreviewInput) -> Option<DisplayBlock> {
675    let current = |url: &MemoryUrl| match root.read(url) {
676        Ok(read) => Some(Some(read.text)),
677        Err(StoreError::NotFound { .. }) => Some(None),
678        Err(_) => None,
679    };
680    let block = match input {
681        PreviewInput::Write { url, text } => match current(&url)? {
682            Some(old) => preview_diff(
683                &url,
684                TextChange::Modified {
685                    old: &shown_text(&old),
686                    new: &text,
687                },
688            ),
689            None => preview_diff(&url, TextChange::Created { text: &text }),
690        },
691        PreviewInput::Edit { url, old, new } => {
692            let text = current(&url)??;
693            let shown = shown_text(&text);
694            let at = replacement_at(&text, &shown, &old)?;
695            preview_diff(
696                &url,
697                TextChange::Replaced {
698                    text: &shown,
699                    ranges: &[(at, at + old.len())],
700                    new: &new,
701                },
702            )
703        }
704        PreviewInput::Delete { url } => preview_diff(
705            &url,
706            TextChange::Deleted {
707                text: &shown_text(&current(&url)??),
708            },
709        ),
710    };
711    cookie_agent_runtime_api::display::fit_block(block, None).ok()
712}

714fn preview_diff(url: &MemoryUrl, change: TextChange<'_>) -> DisplayBlock {
715    text_diff(
716        &url.to_string(),
717        Some("markdown".to_owned()),
718        change,
719        DiffAsOf::Prepare,
720    )
721}

723/// Where in `shown`, the memory file `text` without its stamps, an edit
724/// replaces `old`: its one occurrence in the body. `None` when `old` is
725/// empty or occurs in the body other than once.
726fn replacement_at(text: &str, shown: &str, old: &str) -> Option<usize> {
727    let document = MemoryDocument::parse(text.as_bytes()).ok()?;
728    let body = document.body();
729    let start = shown.len().checked_sub(body.len())?;
730    if old.is_empty() || shown[start..] != *body || body.matches(old).count() != 1 {
731        return None;
732    }
733    Some(start + body.find(old)?)
734}

736/// How clients show a call: the title names its URL or query, and the
737/// arguments add what the title does not show. A previewed call's diff shows
738/// a write's frontmatter and body and an edit's replacement in place, so
739/// they leave the arguments, as the coding runtime's `edit` and `write` do.
740fn call_presentation(
741    call: &ToolCall,
742    previewed: bool,
743) -> cookie_agent_protocol::PreparedPresentation {
744    let Some(primary) = display_argument(&call.name, &call.arguments) else {
745        return cookie_agent_runtime_api::tool_title_only(&call.name);
746    };
747    let text = |member: &str| {
748        call.arguments
749            .get(member)
750            .and_then(serde_json::Value::as_str)
751            .map(str::to_owned)
752    };
753    let (title, hint, entries) = match call.name.as_str() {
754        SEARCH => {
755            let mut entries = vec![("query", primary.clone())];
756            entries.extend(text("layer").map(|layer| ("layer", layer)));
757            entries.extend(
758                call.arguments
759                    .get("limit")
760                    .and_then(serde_json::Value::as_u64)
761                    .map(|limit| ("limit", limit.to_string())),
762            );
763            (
764                "Search memory".to_owned(),
765                cookie_agent_protocol::ToolHint::SEARCH,
766                entries,
767            )
768        }
769        READ => {
770            // The title names the memory; only a window adds to it.
771            let mut entries = Vec::new();
772            for member in ["offset", "limit"] {
773                entries.extend(
774                    call.arguments
775                        .get(member)
776                        .and_then(serde_json::Value::as_u64)
777                        .map(|value| (member, value.to_string())),
778                );
779            }
780            (
781                format!("Read memory {primary}"),
782                cookie_agent_protocol::ToolHint::FILE_READ,
783                entries,
784            )
785        }
786        WRITE if previewed => (
787            format!("Write memory {primary}"),
788            cookie_agent_protocol::ToolHint::FILE_WRITE,
789            text("reason")
790                .map(|value| vec![("reason", value)])
791                .unwrap_or_default(),
792        ),
793        WRITE => {
794            let mut entries = vec![("url", primary.clone())];
795            entries.extend(text("description").map(|value| ("description", value)));
796            entries.extend(call.arguments.get("keywords").and_then(|keywords| {
797                let keywords = match keywords {
798                    serde_json::Value::Array(keywords) => keywords
799                        .iter()
800                        .filter_map(serde_json::Value::as_str)
801                        .map(str::to_owned)
802                        .collect(),
803                    serde_json::Value::String(text) => split_keywords(text),
804                    _ => return None,
805                };
806                Some(("keywords", keywords.join(", ")))
807            }));
808            entries.extend(text("body").map(|value| ("body", value)));
809            entries.extend(text("reason").map(|value| ("reason", value)));
810            (
811                format!("Write memory {primary}"),
812                cookie_agent_protocol::ToolHint::FILE_WRITE,
813                entries,
814            )
815        }
816        EDIT if previewed => (
817            format!("Edit memory {primary}"),
818            cookie_agent_protocol::ToolHint::FILE_EDIT,
819            Vec::new(),
820        ),
821        EDIT => {
822            let mut entries = vec![("url", primary.clone())];
823            entries.extend(text("oldString").map(|value| ("oldString", value)));
824            entries.extend(text("newString").map(|value| ("newString", value)));
825            (
826                format!("Edit memory {primary}"),
827                cookie_agent_protocol::ToolHint::FILE_EDIT,
828                entries,
829            )
830        }
831        DELETE => (
832            format!("Delete memory {primary}"),
833            cookie_agent_protocol::ToolHint::FILE_WRITE,
834            Vec::new(),
835        ),
836        _ => return cookie_agent_runtime_api::tool_title_only(&call.name),
837    };
838    crate::base_tools::base_presentation(&title, hint, &primary, entries)
839}

883/// The frontmatter keys the daemon stamps on every save.
884const STAMP_KEYS: [&str; 2] = ["created", "modified"];
885
886/// A memory file as its diff shows it: without the stamp lines of its
887/// frontmatter, which the daemon rewrites on every save and the memory
888/// editors leave out, the rest kept byte for byte.
889fn shown_text(text: &str) -> String {
890    let Some(opening) = ["---\n", "---\r\n"]
891        .into_iter()
892        .find(|opening| text.starts_with(opening))
893    else {
894        return text.to_owned();
895    };
896    let mut kept = String::with_capacity(text.len());
897    kept.push_str(opening);
898    let mut lines = text[opening.len()..].split_inclusive('\n');
899    for line in lines.by_ref() {
900        if line.trim_end() == "---" {
901            kept.push_str(line);
902            break;
903        }
904        let stamp = STAMP_KEYS.iter().any(|key| {
905            line.strip_prefix(key)
906                .is_some_and(|rest| rest.starts_with(':'))
907        });
908        if !stamp {
909            kept.push_str(line);
910        }
911    }
912    lines.for_each(|line| kept.push_str(line));
913    kept
914}

916/// The diff of the memory at `url` changing as `change` says, built as the
917/// coding runtime's `edit` and `write` tools build theirs, labelled with
918/// the URL, and fit to the display budget with its complete content kept
919/// in the session's artifact store.
920fn memory_diff(
921    url: &MemoryUrl,
922    change: TextChange<'_>,
923    context: &ToolExecutionContext,
924) -> Result<Vec<Open<DisplayBlock>>, ToolError> {
925    let label = url.to_string();
926    let name = label.rsplit(['/', ':']).next().unwrap_or("memory");
927    let block = fit_executed_block(
928        text_diff(
929            &label,
930            Some("markdown".to_owned()),
931            change,
932            DiffAsOf::Execute,
933        ),
934        context.artifacts.as_ref(),
935        context.session,
936        format!("{name}.diff"),
937    )?;
938    Ok(vec![Open::Known(block)])
939}
940
941/// The diff of an edit that replaced `old` with `new` in the body: the
942/// `edit` tool's replacement diff when that one replacement is all that
943/// changed, else a line diff of the whole file (an `agent://` file or a
944/// project's overview whose frontmatter the daemon rewrote too).
945fn edit_diff(
946    url: &MemoryUrl,
947    edited: &crate::memory::store::Edited,
948    old: &str,
949    new: &str,
950    context: &ToolExecutionContext,
951) -> Result<Vec<Open<DisplayBlock>>, ToolError> {
952    let before = shown_text(&edited.before);
953    let after = shown_text(&edited.after);
954    if let Some(at) = replacement_at(&edited.before, &before, old)
955        && format!("{}{new}{}", &before[..at], &before[at + old.len()..]) == after
956    {
957        return memory_diff(
958            url,
959            TextChange::Replaced {
960                text: &before,
961                ranges: &[(at, at + old.len())],
962                new,
963            },
964            context,
965        );
966    }
967    memory_diff(
968        url,
969        TextChange::Modified {
970            old: &before,
971            new: &after,
972        },
973        context,
974    )
975}

995#[async_trait]
996impl PreparedExecutor for MemoryExecutor {
997    async fn execute(
998        self: Box<Self>,
999        context: ToolExecutionContext,
1000    ) -> Result<ToolCompletion, ToolError> {
1001        let engine = self.engine;
1002        let root = engine.memory_root().clone();
1003        let limits = engine.memory_limits();
1004        let session = context.session;
1005        let caller = {
1006            let engine = engine.clone();
1007            let run = context.run;
1008            blocking(move || engine.memory_caller(run))
1009                .await?
1010                .map_err(ToolError::execution)?
1011        };
1012        match self.call {
1013            MemoryCall::Search {
1014                query,
1015                layer,
1016                limit,
1017            } => {
1018                let visibility = caller.clone();
1019                let hits = blocking(move || {
1020                    root.search(&query, layer, limit, |url| visibility.visibility(url))
1021                })
1022                .await?
1023                .map_err(|error| ToolError::execution(error.to_string()))?;
1024                let output = search_output(&hits, &caller, Timestamp::now());
1025                let count = hits.len();
1026                Ok(result(
1027                    format!(
1028                        "Found {count} {}",
1029                        if count == 1 { "memory" } else { "memories" }
1030                    ),
1031                    output,
1032                    search_display(&hits),
1033                    serde_json::json!({
1034                        "results": hits.iter().map(|hit| hit.url.to_string()).collect::<Vec<_>>(),
1035                    }),
1036                ))
1037            }
1038            MemoryCall::Read { url, window } => {
1039                let read = {
1040                    let url = url.clone();
1041                    blocking(move || {
1042                        let read = root.read(&url)?;
1043                        root.record_reads([&url], Timestamp::now());
1044                        Ok::<_, StoreError>(read)
1045                    })
1046                    .await?
1047                    .map_err(store_error)?
1048                };
1049                engine
1050                    .inner
1051                    .memory_seen
1052                    .saw(session, url.clone(), read.entry.revision.clone());
1053                let (output, page) =
1054                    read_output(&url, &read, &caller, &limits, Timestamp::now(), window)?;
1055                let excerpt = cookie_agent_runtime_api::display::fit_block(
1056                    page.excerpt_block(url.to_string(), Some("markdown".to_owned())),
1057                    None,
1058                )?;
1059                Ok(result(
1060                    format!("Read {url}"),
1061                    output,
1062                    vec![Open::Known(excerpt)],
1063                    serde_json::json!({
1064                        "url": url.to_string(),
1065                        "offset": window.offset,
1066                        "limit": window.limit,
1067                        "total_lines": page.total_lines,
1068                        "next_offset": page.next_offset,
1069                    }),
1070                ))
1071            }
1072            MemoryCall::Write {
1073                url,
1074                document,
1075                described,
1076                keyworded,
1077                new,
1078            } => {
1079                let seen = engine.inner.memory_seen.seen(session, &url);
1080                let owners = engine.clone();
1081                let target = url.clone();
1082                let written = blocking(move || {
1083                    root.write(
1084                        &target,
1085                        &document.render(),
1086                        WriteMode::Model {
1087                            seen: seen.as_ref(),
1088                            new,
1089                        },
1090                        crate::runtime::EngineMemoryOwners(&owners),
1091                        &limits,
1092                        Timestamp::now(),
1093                    )
1094                })
1095                .await?
1096                .map_err(store_error)?;
1097                engine
1098                    .inner
1099                    .memory_seen
1100                    .saw(session, url.clone(), written.revision.clone());
1101                let after = shown_text(&written.text);
1102                let before = written.replaced.as_deref().map(shown_text);
1103                let display = memory_diff(
1104                    &url,
1105                    match &before {
1106                        Some(old) => TextChange::Modified { old, new: &after },
1107                        None => TextChange::Created { text: &after },
1108                    },
1109                    &context,
1110                )?;
1111                let mut output =
1112                    text_tools::wrote(written.created, &url.to_string(), written.text.len());
1113                // What the call left out was never the model's to keep.
1114                if let Some(mut kept) = written.kept {
1115                    if !described {
1116                        kept.description = None;
1117                    }
1118                    if !keyworded {
1119                        kept.keywords.clear();
1120                    }
1121                    let notice = kept.describe(&url);
1122                    if !notice.is_empty() {
1123                        output.push(' ');
1124                        output.push_str(&notice);
1125                    }
1126                }
1127                if !written.near.is_empty() {
1128                    output.push_str(&near_output(&written.near));
1129                }
1130                Ok(result(
1131                    format!("Wrote {url}"),
1132                    output,
1133                    display,
1134                    serde_json::json!({ "url": url.to_string() }),
1135                ))
1136            }
1137            MemoryCall::Edit { url, old, new } => {
1138                let target = url.clone();
1139                let owners = engine.clone();
1140                let replacement = (old.clone(), new.clone());
1141                let edited = blocking(move || {
1142                    root.edit(
1143                        &target,
1144                        &old,
1145                        &new,
1146                        crate::runtime::EngineMemoryOwners(&owners),
1147                        &limits,
1148                        Timestamp::now(),
1149                    )
1150                })
1151                .await?
1152                .map_err(store_error)?;
1153                // A write of its own counts as seeing the memory.
1154                engine
1155                    .inner
1156                    .memory_seen
1157                    .saw(session, url.clone(), edited.revision.clone());
1158                let display = edit_diff(&url, &edited, &replacement.0, &replacement.1, &context)?;
1159                Ok(result(
1160                    format!("Edited {url}"),
1161                    text_tools::edited(&url.to_string(), 1),
1162                    display,
1163                    serde_json::json!({ "url": url.to_string() }),
1164                ))
1165            }
1166            MemoryCall::Delete { url } => {
1167                let target = url.clone();
1168                let removed = blocking(move || root.delete(&target, None))
1169                    .await?
1170                    .map_err(store_error)?;
1171                engine.inner.memory_seen.forget(session, &url);
1172                let removed = shown_text(&removed);
1173                let display = memory_diff(&url, TextChange::Deleted { text: &removed }, &context)?;
1174                Ok(result(
1175                    format!("Deleted {url}"),
1176                    format!("Deleted {url}."),
1177                    display,
1178                    serde_json::json!({ "url": url.to_string() }),
1179                ))
1180            }
1181        }
1182    }
1183}

1185/// The memories near a new one, as its write's result lists them
1186/// (section 16.9.5).
1187fn near_output(near: &[NearMemory]) -> String {
1188    let mut text = String::from("\n\nMemories near it:");
1189    for memory in near {
1190        text.push_str(&format!("\n- {} — {}", memory.url, memory.description));
1191    }
1192    text.push_str("\nIf one of these already covers it, edit that and delete this.");
1193    text
1194}

1196fn yes_no(value: bool) -> &'static str {
1197    if value { "yes" } else { "no" }
1198}

1200/// What `memory_read` returns (section 16.9.4): whether the memory is
1201/// related and editable, its stamps with their ages, its size and limit,
1202/// its description and keywords, then the page `window` of its body,
1203/// numbered as `read` numbers a file's lines, and the page itself. No URL
1204/// and no revision. Fails when the window starts past the body's last line.
1205fn read_output(
1206    url: &MemoryUrl,
1207    read: &crate::memory::store::MemoryRead,
1208    caller: &MemoryCaller,
1209    limits: &crate::memory::limits::MemoryLimits,
1210    now: Timestamp,
1211    window: LineWindow,
1212) -> Result<(String, NumberedPage), ToolError> {
1213    let entry = &read.entry;
1214    let max = limits.max_bytes(url) as u64;
1215    let mut lines = vec![
1216        format!("related: {}", yes_no(caller.related(url))),
1217        format!("editable: {}", yes_no(caller.editable(url))),
1218        format!(
1219            "created: {} ({} ago)",
1220            entry.created,
1221            format::age(entry.created, now)
1222        ),
1223        format!(
1224            "modified: {} ({} ago)",
1225            entry.modified,
1226            format::age(entry.modified, now)
1227        ),
1228        if entry.size > max {
1229            format!("size: {} bytes, over its limit", entry.size)
1230        } else {
1231            format!("size: {} bytes", entry.size)
1232        },
1233        format!("max size: {max} bytes"),
1234    ];
1235    let body = match &entry.document {
1236        Ok(document) => {
1237            lines.push(format!("description: {}", document.description()));
1238            if !document.keywords().is_empty() {
1239                lines.push(format!("keywords: {}", document.keywords().join(", ")));
1240            }
1241            document.body()
1242        }
1243        Err(reason) => {
1244            lines.push(format!(
1245                "This memory is invalid ({reason}); search and the snapshot skip it until it is fixed."
1246            ));
1247            read.text.as_str()
1248        }
1249    };
1250    let mut output = format!("{}\n\n", lines.join("\n"));
1251    let page = text_tools::page_text(body, window, MAX_PAGE_BYTES.saturating_sub(output.len()));
1252    page.check_in_range(&url.to_string())?;
1253    output.push_str(&page.text_with_continuation());
1254    Ok((output, page))
1255}

1257/// The search results as the model reads them (section 16.9.4): one
1258/// paragraph per memory, in rank order.
1259fn search_output(hits: &[SearchHit], caller: &MemoryCaller, now: Timestamp) -> String {
1260    if hits.is_empty() {
1261        return "No memories match.".to_owned();
1262    }
1263    hits.iter()
1264        .map(|hit| {
1265            let mut text = format!(
1266                "{} — {}\n  {} · {} · {} old",
1267                hit.url,
1268                hit.description,
1269                if caller.related(&hit.url) {
1270                    "related"
1271                } else {
1272                    "unrelated"
1273                },
1274                if caller.editable(&hit.url) {
1275                    "editable"
1276                } else {
1277                    "read-only"
1278                },
1279                format::age(hit.modified, now)
1280            );
1281            if !hit.full {
1282                text.push_str("\n  (reading it needs approval)");
1283            } else if !hit.keywords.is_empty() {
1284                text.push_str(&format!("\n  keywords: {}", hit.keywords.join(", ")));
1285            }
1286            text
1287        })
1288        .collect::<Vec<_>>()
1289        .join("\n\n")
1290}
