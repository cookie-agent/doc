1//! The cron tools of the `global` runtime (`docs/site/specs/cron.md`
2//! section 8): `cron_create`, `cron_list`, `cron_update`, `cron_delete`,
3//! and `cron_run`, provider `builtin.cron`.
4//!
5//! Every tool but `cron_list` is checked under the `cron` action with the
6//! target kind of each job it touches as a resource (`session`, `subagent`,
7//! or `new`). `cron_list` declares no permission: it is offered with the
8//! other cron tools, and it shows only the jobs whose kind the rules allow
9//! or ask for. Targets are resolved at preparation, against the calling
10//! session: `"self"`, a session handle below it, or a new-session template
11//! whose defaults are the calling session's. An agent never names a
12//! permission mode or a model.
13//!
14//! Like the memory tools, the provider is built with the `global` runtime,
15//! before the engine exists, and holds the engine through a handle bound
16//! once the engine is made.
17

380/// The cron tools, holding the engine once it is bound.
381#[derive(Default)]
382pub(crate) struct CronTools {
383    engine: OnceLock<Weak<Inner>>,
384}
385
386impl CronTools {
387    pub(crate) fn new() -> Self {
388        Self::default()
389    }
390
391    /// Gives the provider its engine. Later calls are ignored.
392    pub(crate) fn bind(&self, engine: &Engine) {
393        let _ = self.engine.set(Arc::downgrade(&engine.inner));
394    }
395
396    fn engine(&self) -> Result<Engine, ToolError> {
397        self.engine
398            .get()
399            .and_then(Weak::upgrade)
400            .map(|inner| Engine { inner })
401            .ok_or_else(|| ToolError::execution("the engine is gone"))
402    }
403}

405impl Engine {
406    /// The agent of the calling run, as it creates or changes jobs.
407    fn cron_agent(
408        &self,
409        session: SessionId,
410        run: cookie_agent_protocol::RunId,
411    ) -> Result<CronAgentActor, ToolError> {
412        // A call is prepared within its run; the session's own agent
413        // stands in for a run that is no longer active.
414        let agent = match self.active_run_policy(run) {
415            Some(policy) => policy.agent.agent.clone(),
416            None => self
417                .inner
418                .store
419                .get(session)
420                .map_err(|error| refused(error.to_string()))?
421                .creation_agent
422                .agent
423                .clone(),
424        };
425        Ok(CronAgentActor { agent, session })
426    }
427
428    /// The target an agent's `target` names (section 8): its own session,
429    /// a session below it, or a new-session template with the calling
430    /// session's agent, runtimes, and project as defaults. Anything else is
431    /// refused whatever the rules say.
432    fn resolve_tool_target(
433        &self,
434        caller: &CronAgentActor,
435        target: TargetArg,
436    ) -> Result<CronTarget, ToolError> {
437        let projection = self
438            .inner
439            .store
440            .get(caller.session)
441            .map_err(|error| refused(error.to_string()))?;
442        match target {
443            TargetArg::Session(reference) if reference == "self" => {
444                Ok(match projection.meta.origin {
445                    SessionOrigin::Root => CronTarget::Session(caller.session),
446                    SessionOrigin::Delegated { .. } => CronTarget::Subagent(caller.session),
447                })
448            }
449            TargetArg::Session(reference) => self
450                .resolve_descendant(caller.session, &reference)
451                .map(CronTarget::Subagent)
452                .map_err(|error| {
453                    refused(format!(
454                        "target `{reference}` is not a session below this one: {error}"
455                    ))
456                }),
457            TargetArg::New { new } => {
458                let attachments = self.attachments_of(&projection);
459                let global = attachments
460                    .iter()
461                    .any(|attachment| attachment.runtime_id.as_str() == RuntimeId::GLOBAL);
462                let runtimes = new.runtimes.unwrap_or_else(|| {
463                    attachments
464                        .iter()
465                        .filter(|attachment| attachment.runtime_id.as_str() != RuntimeId::GLOBAL)
466                        .map(|attachment| ProjectRuntime {
467                            runtime_id: attachment.runtime_id.clone(),
468                            capability: Some(attachment.capability.clone()),
469                            workspace: attachment.workspace.clone(),
470                        })
471                        .collect()
472                });
473                let project = match new.project {
474                    Some(name) => Some(self.project_by_name(&name)?),
475                    None => crate::blocking::off_worker(|| self.session_project(caller.session))
476                        .map(|project| project.id),
477                };
478                Ok(CronTarget::New(CronTemplate {
479                    agent: new.agent.unwrap_or_else(|| caller.agent.clone()),
480                    preset: projection.meta.creation_selection.preset.clone(),
481                    model: None,
482                    permission_rules: new.permission_rules.unwrap_or_default(),
483                    runtimes: Some(runtimes),
484                    global,
485                    cwd: None,
486                    project,
487                }))
488            }
489        }
490    }
491
492    /// The project named `name`, or with ID `name`.
493    fn project_by_name(&self, name: &str) -> Result<ProjectId, ToolError> {
494        let projects = crate::blocking::off_worker(|| self.inner.projects.list())
495            .map_err(|error| refused(error.to_string()))?;
496        projects
497            .into_iter()
498            .find(|project| project.name.as_str() == name || project.id.as_str() == name)
499            .map(|project| project.id)
500            .ok_or_else(|| refused(format!("there is no project `{name}`")))
501    }
502
503    /// Whether the calling run's rules allow or ask for `cron` on `kind`.
504    fn cron_kind_visible(
505        &self,
506        session: SessionId,
507        run: cookie_agent_protocol::RunId,
508        kind: CronTargetKind,
509    ) -> bool {
510        let Some(policy) = self.active_run_policy(run) else {
511            return false;
512        };
513        let overlay = self
514            .inner
515            .store
516            .get(session)
517            .map(|projection| projection.permission_overlay.clone())
518            .unwrap_or_default();
519        kind_visible(&policy.agent, &overlay, kind)
520    }
521
522    /// How a target reads in a tool result: a session by its handle.
523    fn target_text(&self, target: &CronTarget) -> String {
524        let handle = |session: SessionId| {
525            self.inner
526                .store
527                .summary(session)
528                .ok()
529                .and_then(|summary| summary.meta.short_id)
530                .unwrap_or_else(|| session.to_string())
531        };
532        match target {
533            CronTarget::Session(session) => format!("session {}", handle(*session)),
534            CronTarget::Subagent(session) => format!("subagent {}", handle(*session)),
535            CronTarget::New(template) => {
536                let mut text = format!("new session (agent {}", template.agent);
537                if let Some(project) = &template.project {
538                    let name = self
539                        .inner
540                        .projects
541                        .get(project)
542                        .map_or_else(|_| project.to_string(), |project| project.name.to_string());
543                    text.push_str(&format!(", project {name}"));
544                }
545                text.push(')');
546                text
547            }
548        }
549    }
550}

552/// Whether `agent`'s rules and `overlay` allow or ask for `cron` on jobs of
553/// `kind`, so that `cron_list` shows them.
554fn kind_visible(
555    agent: &cookie_agent_protocol::AgentSnapshot,
556    overlay: &cookie_agent_protocol::SessionPermissionOverlay,
557    kind: CronTargetKind,
558) -> bool {
559    let (effect, _) = crate::permissions::effective_permission_with_overlay(
560        agent,
561        Some(overlay),
562        &PermissionAction::CRON,
563        kind.as_str(),
564        crate::permissions::PermissionScope::runtime(RuntimeId::GLOBAL, None),
565    );
566    matches!(effect, PermissionEffect::Allow | PermissionEffect::Ask)
567}

569/// `at` in local time and UTC.
570fn when(at: jiff::Timestamp) -> String {
571    let local = at
572        .to_zoned(crate::runtime::cron_api::host_zone())
573        .strftime("%Y-%m-%d %H:%M:%S %Z")
574        .to_string();
575    format!("{local} ({at})")
576}

578fn schedule_text(job: &CronJob) -> String {
579    match &job.schedule {
580        CronSchedule::Cron(expression) => match job.max_runs {
581            CronMaxRuns::Count(count) => format!("cron `{expression}`, max_runs {count}"),
582            CronMaxRuns::Unlimited => format!("cron `{expression}`, max_runs 0 (no limit)"),
583        },
584        CronSchedule::At(at) => format!("once at {}", when(*at)),
585        CronSchedule::In(delay) => format!("once in {delay}"),
586    }
587}

589fn last_fire_text(job: &CronJob) -> String {
590    use cookie_agent_protocol::CronFireOutcome;
591    let Some(fire) = job.history.last() else {
592        return "never fired".into();
593    };
594    let outcome = match fire.outcome.known() {
595        Some(CronFireOutcome::Pending) => "being delivered".to_owned(),
596        Some(CronFireOutcome::Delivered) => match &fire.run_end {
597            Some(end) => format!("delivered, run {}", status_name(end.status)),
598            None => "delivered".to_owned(),
599        },
600        Some(CronFireOutcome::Skipped { reason }) => format!("skipped ({})", skip_text(*reason)),
601        Some(CronFireOutcome::Failed { error }) => format!("failed: {error}"),
602        None => "unknown".to_owned(),
603    };
604    format!("{outcome} at {}", when(fire.fired_at))
605}

607fn status_name(status: cookie_agent_protocol::SessionStatus) -> &'static str {
608    use cookie_agent_protocol::SessionStatus;
609    match status {
610        SessionStatus::Idle => "idle",
611        SessionStatus::Running => "running",
612        SessionStatus::Completed => "completed",
613        SessionStatus::Failed => "failed",
614        SessionStatus::Cancelled => "cancelled",
615        SessionStatus::Interrupted => "interrupted",
616        SessionStatus::Unknown => "unknown",
617    }
618}

620fn skip_text(reason: cookie_agent_protocol::CronSkipReason) -> &'static str {
621    use cookie_agent_protocol::CronSkipReason;
622    match reason {
623        CronSkipReason::PreviousFireNotDelivered => "previous fire not delivered",
624        CronSkipReason::PreviousRunStillRunning => "previous run still running",
625        CronSkipReason::DaemonWasDown => "daemon was down",
626        CronSkipReason::Interrupted => "interrupted",
627        CronSkipReason::FiringDisabled => "firing disabled",
628        CronSkipReason::DaemonRestarting => "daemon restarting",
629        CronSkipReason::SessionHeld => "session held",
630        CronSkipReason::Unknown => "unknown",
631    }
632}
633
634fn enabled_text(job: &CronJob) -> String {
635    use cookie_agent_protocol::CronDisabledReason;
636    if job.enabled {
637        return "yes".into();
638    }
639    match job
640        .disabled_reason
641        .as_ref()
642        .and_then(|reason| reason.known())
643    {
644        Some(CronDisabledReason::Finished) => "no (finished)".into(),
645        Some(CronDisabledReason::Failed { failures }) => {
646            format!("no (failed {failures} times in a row)")
647        }
648        Some(CronDisabledReason::Disabled { by }) => match by {
649            cookie_agent_protocol::CronActor::User => "no (disabled by the user)".into(),
650            cookie_agent_protocol::CronActor::Agent(agent) => {
651                format!("no (disabled by agent {})", agent.agent)
652            }
653        },
654        None => "no".into(),
655    }
656}

658/// One job as a tool result shows it.
659fn job_text(engine: &Engine, job: &CronJob) -> String {
660    let label = job.name.clone().unwrap_or_else(|| {
661        let line = job.prompt.lines().next().unwrap_or_default();
662        line.chars().take(40).collect()
663    });
664    let next = job.next_fire.map_or_else(|| "none".to_owned(), when);
665    format!(
666        "{} \"{}\"\n  schedule: {}\n  target: {}\n  next fire: {next}\n  enabled: {}\n  last fire: {}",
667        job.id,
668        label,
669        schedule_text(job),
670        engine.target_text(&job.target),
671        enabled_text(job),
672        last_fire_text(job),
673    )
674}

734    fn presentation(&self, call: &ToolCall) -> cookie_agent_protocol::PreparedPresentation {

775
776    async fn prepare(
777        &self,
778        ctx: ToolPreparationContext,
779        call: ToolCall,
780    ) -> Result<PreparedTool, ToolError> {
781        let engine = self.engine()?;
782        permission_name(&call.name)?;
783        let (call_kind, kinds) = match call.name.as_str() {
784            CREATE => {
785                let args: CreateArgs = parse_args(CREATE, call.arguments.clone())?;
786                let caller = engine.cron_agent(ctx.session, ctx.run)?;
787                let target = engine.resolve_tool_target(&caller, args.target)?;
788                let kind = target.kind();
789                (
790                    CronCall::Create(CronCreateParams {
791                        name: args.name,
792                        prompt: args.prompt,
793                        schedule: args.schedule.into(),
794                        max_runs: max_runs_arg(args.max_runs)?,
795                        target,
796                        delivery: args.delivery,
797                        permission_mode: None,
798                        max_consecutive_failures: None,
799                        enabled: None,
800                    }),
801                    vec![kind],
802                )
803            }
804            LIST => {
805                let args: ListArgs = parse_args(LIST, call.arguments.clone())?;
806                (CronCall::List(args.target), Vec::new())
807            }
808            UPDATE => {
809                let args: UpdateArgs = parse_args(UPDATE, call.arguments.clone())?;
810                let job = engine
811                    .cron_store()
812                    .job(&args.id)
813                    .ok_or_else(|| refused(format!("there is no cron job {}", args.id)))?;
814                let mut kinds = vec![job.target.kind()];
815                let target = match args.target {
816                    Some(target) => {
817                        let caller = engine.cron_agent(ctx.session, ctx.run)?;
818                        let target = engine.resolve_tool_target(&caller, target)?;
819                        if target.kind() != kinds[0] {
820                            kinds.push(target.kind());
821                        }
822                        Some(target)
823                    }
824                    None => None,
825                };
826                let mut params = CronUpdateParams::new(args.id);
827                params.name = args.name.map(Some);
828                params.prompt = args.prompt;
829                params.schedule = args.schedule.map(Into::into);
830                params.max_runs = max_runs_arg(args.max_runs)?;
831                params.target = target;
832                params.delivery = args.delivery;
833                params.enabled = args.enabled;
834                (CronCall::Update(params), kinds)
835            }
836            DELETE | RUN => {
837                let args: IdArgs = parse_args(&call.name, call.arguments.clone())?;
838                let job = engine
839                    .cron_store()
840                    .job(&args.id)
841                    .ok_or_else(|| refused(format!("there is no cron job {}", args.id)))?;
842                let kind = job.target.kind();
843                (
844                    if call.name == DELETE {
845                        CronCall::Delete(args.id)
846                    } else {
847                        CronCall::Run(args.id)
848                    },
849                    vec![kind],
850                )
851            }
852            _ => unreachable!("checked by permission_name"),
853        };
854        let executor = Box::new(CronExecutor {
855            engine,
856            call: call_kind,
857            call_id: call.id,
858        });
859        if kinds.is_empty() {
860            return PreparedTool::new(
861                prepared_operation(
862                    &call.name,
863                    &call.arguments,
864                    vec![(PermissionAction::CRON, "list")],
865                    vec![prepared_resource(
866                        PermissionAction::CRON,
867                        "cron",
868                        b"list",
869                        b"list",
870                        PreparedBindingLifetime::RestartStable,
871                        ApprovalResourceSource::PrimaryOperation,
872                    )?],
873                    &ctx.context_bytes(),
874                )?,
875                call.arguments,
876                None,
877                executor,
878            );
879        }
880        let labels = kinds
881            .iter()
882            .map(|kind| kind.as_str().to_owned())
883            .collect::<Vec<_>>();
884        let resources = labels
885            .iter()
886            .map(|label| {
887                prepared_resource(
888                    PermissionAction::CRON,
889                    "cron",
890                    label.as_bytes(),
891                    label.as_bytes(),
892                    PreparedBindingLifetime::RestartStable,
893                    ApprovalResourceSource::PrimaryOperation,
894                )
895            })
896            .collect::<Result<Vec<_>, _>>()?;
897        let operation = prepared_operation(
898            &call.name,
899            &call.arguments,
900            vec![(
901                PermissionAction::CRON,
902                call.name.trim_start_matches("cron_"),
903            )],
904            resources,
905            &ctx.context_bytes(),
906        )?;
907        PreparedTool::new(operation, call.arguments, None, executor)?.with_policy_labels(labels)
908    }
909}

911struct CronExecutor {
912    engine: Engine,
913    call: CronCall,
914    call_id: cookie_agent_protocol::ToolCallId,
915}
916
917#[async_trait]
918impl PreparedExecutor for CronExecutor {
919    async fn execute(
920        self: Box<Self>,
921        context: ToolExecutionContext,
922    ) -> Result<ToolCompletion, ToolError> {
923        let engine = self.engine;
924        let session = context.session;
925        let error = |error: crate::EngineError| refused(error.to_string());
926        match self.call {
927            CronCall::Create(params) => {
928                let caller = engine.cron_agent(session, context.run)?;
929                let job = engine
930                    .create_cron_as(CronRequester::Agent(caller), params)
931                    .await
932                    .map_err(error)?;
933                let next = job.next_fire.map_or_else(|| "none".to_owned(), when);
934                Ok(text_result(
935                    "Scheduled a prompt",
936                    format!(
937                        "Created cron job {}; next fire {next}.\n\n{}",
938                        job.id,
939                        job_text(&engine, &job)
940                    ),
941                ))
942            }
943            CronCall::List(filter) => {
944                let jobs = engine
945                    .list_crons()
946                    .jobs
947                    .into_iter()
948                    .filter(|job| filter.is_none_or(|kind| job.target.kind() == kind))
949                    .filter(|job| engine.cron_kind_visible(session, context.run, job.target.kind()))
950                    .collect::<Vec<_>>();
951                let mut output = if jobs.is_empty() {
952                    "No cron jobs.".to_owned()
953                } else {
954                    jobs.iter()
955                        .map(|job| job_text(&engine, job))
956                        .collect::<Vec<_>>()
957                        .join("\n\n")
958                };
959                if !engine.list_crons().firing {
960                    output.push_str(
961                        "\n\nFiring is off ([cron] enabled = false): due fires are skipped.",
962                    );
963                }
964                Ok(text_result(
965                    &format!(
966                        "Listed {} cron {}",
967                        jobs.len(),
968                        if jobs.len() == 1 { "job" } else { "jobs" }
969                    ),
970                    output,
971                ))
972            }
973            CronCall::Update(params) => {
974                let caller = engine.cron_agent(session, context.run)?;
975                let job = engine
976                    .update_cron_as(CronRequester::Agent(caller), params)
977                    .await
978                    .map_err(error)?;
979                Ok(text_result(
980                    "Changed a cron job",
981                    format!(
982                        "Changed cron job {}.\n\n{}",
983                        job.id,
984                        job_text(&engine, &job)
985                    ),
986                ))
987            }
988            CronCall::Delete(id) => {
989                engine.delete_cron(id.clone()).await.map_err(error)?;
990                Ok(text_result(
991                    "Deleted a cron job",
992                    format!("Deleted cron job {id}."),
993                ))
994            }
995            CronCall::Run(id) => engine.cron_run_tool(context, self.call_id, id).await,
996        }
997    }
998}
