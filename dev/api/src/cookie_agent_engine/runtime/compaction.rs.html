48    pub(crate) fn record_committed_turn(
49        &mut self,
50        serialized_context_bytes: usize,
51        input_tokens: Option<u64>,
52    ) {
53        if serialized_context_bytes > 0
54            && let Some(input_tokens) = input_tokens.filter(|tokens| *tokens > 0)
55        {
56            self.tokens_per_byte = input_tokens as f64 / serialized_context_bytes as f64;
57        }
58    }

60    pub(crate) fn estimated_context_tokens(self, serialized_context_bytes: usize) -> Option<u64> {
61        (self.tokens_per_byte > 0.0)
62            .then(|| (serialized_context_bytes as f64 * self.tokens_per_byte).ceil() as u64)
63    }

164    pub async fn compact_session_result(
165        &self,
166        session: SessionId,
167        focus: Option<&str>,
168        origin: cookie_agent_protocol::EventOrigin,
169    ) -> Result<SessionCompactResult, EngineError> {
170        self.ensure_appendable(session)?;
171        // An ACP agent compacts its own conversation, when it can.
172        if let Some(result) = self
173            .compact_acp_session(session, focus, origin.clone())
174            .await?
175        {
176            return Ok(result);
177        }
178        let focus = focus.map(str::to_owned);
179        let (reply, receiver) = tokio::sync::oneshot::channel();
180        self.on_actor(session, move |engine| {
181            engine.spawn_compaction(session, focus, origin, reply);
182            Ok(())
183        })
184        .await?;
185        receiver.await.map_err(|_| EngineError::ActorStopped)?
186    }

188    /// Reserves the session for compaction and runs it off the actor, sending
189    /// the outcome to `reply`; a session already held for compaction or a
190    /// run start is rejected as running. Called on the session actor.
191    pub(super) fn spawn_compaction(
192        &self,
193        session: SessionId,
194        focus: Option<String>,
195        origin: cookie_agent_protocol::EventOrigin,
196        reply: tokio::sync::oneshot::Sender<Result<SessionCompactResult, EngineError>>,
197    ) {
198        if !self.reserve_compaction(session) {
199            let _ = reply.send(Err(EngineError::SessionRunning(session)));
200            return;
201        }
202        let engine = self.clone();
203        tokio::spawn(async move {
204            #[cfg(test)]
205            let hook = engine
206                .inner
207                .test_hooks
208                .compaction_execution_hook
209                .lock()
210                .expect("compaction execution hook lock poisoned")
211                .take();
212            #[cfg(test)]
213            if let Some(hook) = hook {
214                if let Some(reached) = hook
215                    .reached
216                    .lock()
217                    .expect("compaction execution reached lock poisoned")
218                    .take()
219                {
220                    let _ = reached.send(());
221                }
222                hook.release.notified().await;
223            }
224            let mut result = engine
225                .compact_session_direct(session, focus.as_deref(), origin)
226                .await;
227            if let Err(error) = engine.finish_compaction(session).await
228                && result.is_ok()
229            {
230                result = Err(error);
231            }
232            let _ = reply.send(result);
233        });
234    }

236    pub(super) async fn compact_session_direct(
237        &self,
238        session: SessionId,
239        focus: Option<&str>,
240        origin: cookie_agent_protocol::EventOrigin,
241    ) -> Result<SessionCompactResult, EngineError> {
242        let projection = self.inner.store.get(session)?;
243        if projection.status == SessionStatus::Running {
244            return Err(EngineError::SessionRunning(session));
245        }
246        let events = projection.log.event_snapshot();
247        let run = projection
248            .log
249            .last_run_started()
250            .map(|(_, run, _)| run)
251            .ok_or(EngineError::NoRunnableModel)?;
252        let mut policy = self.historical_title_policy(&events, run)?;
253        // The tools a run of the session would send now, as automatic
254        // compaction sends its run's: the summary request stays a prefix of
255        // the conversation's, for its cache and for the reasoning it
256        // replays. Delegation targets come from the registry and the goal
257        // tools follow the goal, as at run admission.
258        policy.historical_delegation = false;
259        policy.goal_tools_enabled = matches!(
260            projection.meta.origin,
261            cookie_agent_protocol::SessionOrigin::Root
262        )
263            && crate::goal_projection::GoalProducerProjection::from_events(&events)
264                .goal
265                .is_some_and(|goal| {
266                    matches!(
267                        goal.status,
268                        cookie_agent_protocol::GoalStatus::Active
269                            | cookie_agent_protocol::GoalStatus::Paused
270                    )
271                });
272        let binding = active_compaction_binding(&policy, &events, run)?;
273        let internal_policy = self.internal_agent_policy(
274            InternalAgentKind::ContextCompaction,
275            &policy,
276            Some(binding),
277        )?;
278        let tools = self.tool_definitions(session, &policy)?;
279        let before = projection.log.latest_checkpoint_seq();
280        let cancellation = CancellationToken::new();
281        self.inner
282            .compaction
283            .manual
284            .lock()
285            .unwrap_or_else(|poisoned| poisoned.into_inner())
286            .insert(session, cancellation.clone());
287        let _registered = ManualCompaction {
288            engine: self,
289            session,
290        };
291        match self
292            .maybe_compact_context(CompactionInput {
293                session,
294                run,
295                cancellation: &cancellation,
296                binding,
297                owner_policy: &policy,
298                internal_policy: &internal_policy,
299                tools: &tools,
300                events,
301                force: true,
302                overflow_recovery: false,
303                focus,
304                origin,
305            })
306            .await
307        {
308            Ok(_) => {}
309            Err(EngineError::CompactionCancelled(reason)) => {
310                return Ok(SessionCompactResult {
311                    compacted: false,
312                    cancellation_reason: Some(reason),
313                });
314            }
315            Err(_) if cancellation.is_cancelled() => {
316                return Ok(SessionCompactResult {
317                    compacted: false,
318                    cancellation_reason: Some("compaction cancelled".into()),
319                });
320            }
321            Err(error) => return Err(error),
322        }
323        Ok(SessionCompactResult {
324            compacted: projection.log.latest_checkpoint_seq() > before,
325            cancellation_reason: cancellation
326                .is_cancelled()
327                .then(|| "compaction cancelled".to_owned()),
328        })
329    }

344    pub(super) async fn maybe_compact_context(
345        &self,
346        mut input: CompactionInput<'_>,
347    ) -> Result<crate::events::EventSnapshot, EngineError> {
348        let Some(context_limit) = input.binding.descriptor.capabilities.limits.context else {
349            return Ok(input.events);
350        };
351        // Triggers and retained-history budgets use the input budget, not the whole context
352        // window: a model's input limit or its reserved output can be well below the window.
353        let input_budget = policy::input_token_budget(
354            input.binding,
355            policy::effective_max_output_tokens(
356                input.binding,
357                input.owner_policy.agent.max_output_tokens,
358            ),
359        )
360        .unwrap_or(context_limit);
361        let config = &self.inner.config.load().runtime.context_compaction;
362        let trigger_tokens = resolve_compaction_trigger(input_budget, &config.trigger);
363        if !compaction_gate(input.force, config.auto_compaction, trigger_tokens) {
364            return Ok(input.events);
365        }
366        let active_run = self
367            .inner
368            .sessions
369            .active
370            .lock()
371            .ok()
372            .and_then(|runs| runs.get(&input.run).cloned());
373        if !input.force
374            && active_run.as_ref().is_some_and(|run| {
375                run.auto_compaction_failures.load(Ordering::Relaxed) >= MAX_AUTO_COMPACTION_FAILURES
376            })
377        {
378            return Ok(input.events);
379        }
380        let projection = self.inner.store.get(input.session)?;
381        if !input.force {
382            let log = &projection.log;
383            let Some((usage_seq, observed_tokens)) = log.latest_real_usage() else {
384                return Ok(input.events);
385            };
386            let last_checkpoint_seq = log.latest_checkpoint_seq();
387            if usage_seq < last_checkpoint_seq {
388                return Ok(input.events);
389            }
390            if !usage_reaches_compaction_trigger(observed_tokens, trigger_tokens) {
391                return Ok(input.events);
392            }
393        }
394
395        let requested_input_through_seq = input.events.last().map_or(0, |event| event.seq);
396        let current_events = projection.log.event_snapshot();
397        if projection
398            .log
399            .checkpoint_covers_input(requested_input_through_seq)
400        {
401            return Ok(current_events);
402        }
403
404        let has_producer_input =
405            crate::goal_projection::GoalProducerProjection::from_events(&input.events)
406                .messages
407                .iter()
408                .any(|message| {
409                    !message.consumed
410                        && !message.discarded
411                        && message.admission.is_some_and(|(run, _)| run == input.run)
412                });
413        let producer_claim = if !has_producer_input {
414            None
415        } else {
416            Some(
417                self.claim_existing_producer_inputs(input.session, input.run)
418                    .await?,
419            )
420        };
421        if let Some(claim) = &producer_claim {
422            input.events = Arc::clone(&claim.events);
423        }
424
425        let mut compaction_focus = input.focus.map(str::to_owned);
426        let mut additions = Vec::new();
427        let context_id = crate::plugin::plugin_context_id();
428        for hooked in self.inner.plugins.interception_plugins(
429            cookie_agent_protocol::ExtensionInterceptionHook::SessionBeforeCompact,
430        ) {
431            let plugin = hooked.name.clone();
432            let result = self
433                .inner
434                .plugins
435                .intercept_named::<_, cookie_agent_protocol::ExtensionSessionBeforeCompactResult>(
436                    &plugin,
437                    cookie_agent_protocol::PLUGIN_INTERCEPT_SESSION_BEFORE_COMPACT_METHOD,
438                    &ExtensionSessionBeforeCompactParams {
439                        session_id: input.session,
440                        context_id: context_id.clone(),
441                        checkpoint_id: format!("{}:{requested_input_through_seq}", input.session),
442                        additions: additions.clone(),
443                        instructions: compaction_focus.clone(),
444                    },
445                    Some(input.session),
446                    Some(&context_id),
447                )
448                .await;
449            match result {
450                Ok(result) => {
451                    if result.cancel {
452                        let reason = result
453                            .reason
454                            .unwrap_or_else(|| "compaction cancelled by plugin".into());
455                        self.record_plugin_diagnostic(
456                            input.session,
457                            plugin,
458                            PluginDiagnosticKind::HookBlocked,
459                            reason.clone(),
460                        );
461                        return Err(EngineError::CompactionCancelled(reason));
462                    }
463                    if let Some(instructions) = result.instructions_override {
464                        if instructions.len() > 64 * 1024 {
465                            self.record_plugin_diagnostic(
466                                input.session,
467                                plugin.clone(),
468                                PluginDiagnosticKind::InvalidModification,
469                                "plugin compaction instruction override exceeds the 64 KiB limit"
470                                    .into(),
471                            );
472                        } else {
473                            compaction_focus = Some(instructions);
474                        }
475                    }
476                    if let Some(addendum) = result.addendum.filter(|value| !value.is_empty()) {
477                        let addition_bytes = additions.iter().map(String::len).sum::<usize>();
478                        if addition_bytes.saturating_add(addendum.len()) > 64 * 1024 {
479                            self.record_plugin_diagnostic(
480                                input.session,
481                                plugin,
482                                PluginDiagnosticKind::InvalidModification,
483                                "plugin compaction additions exceed the 64 KiB limit".into(),
484                            );
485                            continue;
486                        }
487                        let focus = compaction_focus.get_or_insert_with(String::new);
488                        if !focus.is_empty() {
489                            focus.push('\n');
490                        }
491                        focus.push_str(&addendum);
492                        additions.push(addendum);
493                    }
494                }
495                Err(error) => {
496                    if let Some(failure) = self.hook_failed(
497                        input.session,
498                        &hooked,
499                        cookie_agent_protocol::ExtensionInterceptionHook::SessionBeforeCompact,
500                        error,
501                    ) {
502                        return Err(EngineError::PluginHookFailed(failure));
503                    }
504                }
505            }
506        }
507
508        let composed_prompt = self.run_agent_prompt(input.session, input.run)?;
509        let mut events = input.events.clone();
510        // Native compaction runs beside the summary when the model has it and
511        // the provider record allows it (native compaction spec section 3).
512        let native_model = (input.binding.descriptor.capabilities.compaction
513            == CompactionCapability::Native
514            && self
515                .inner
516                .native_compaction
517                .allows_compaction(input.binding))
518        .then(|| policy::resolve_model(input.binding, &input.owner_policy.runtime).ok())
519        .flatten();
520        // Native compaction keeps its durable elision policy: when its request
521        // would not fit, bulky tool outputs are elided in the log first.
522        // Summary compaction checks the actual internal-agent request and only
523        // prunes a private retry snapshot.
524        if let Some(model) = &native_model {
525            let raw_fits = if let Some(raw_fits) = raw_fit_from_real_usage(
526                input.overflow_recovery,
527                projection
528                    .log
529                    .latest_real_usage()
530                    .map(|(_, observed_tokens)| observed_tokens),
531                |tokens| compaction_input_fits(input.binding, input.internal_policy, tokens),
532            ) {
533                raw_fits
534            } else {
535                let view =
536                    self.native_compaction_view(&events, input.binding, &composed_prompt, model)?;
537                let raw_fit_tokens =
538                    self.estimated_request_tokens(input.session, &view.history, input.tools)?;
539                compaction_input_fits(input.binding, input.internal_policy, raw_fit_tokens)
540            };
541            if !raw_fits {
542                events = self
543                    .stage_tool_output_elision(input.session, &events, input.origin.clone())
544                    .await?;
545            }
546        }
547        // The summary always builds on the latest summary, never on a window.
548        let context = model_history::assemble_summary_view(
549            &events,
550            &self.inner.artifacts,
551            input.binding,
552            &composed_prompt,
553        )?;
554        let context_tokens_before =
555            self.estimated_request_tokens(input.session, &context.history, input.tools)?;
556        let input_through_seq = events.last().map_or(0, |event| event.seq);
557        let previous = projection.log.latest_checkpoint_seq();
558        // The summary replaces what follows the latest summary: a window
559        // alone, which builds before 1.29.0 wrote, is summarized from the log.
560        let summary_base = {
561            let refs = crate::events::event_refs(&events);
562            model_history::latest_summary_checkpoint(&refs)
563                .map(|(event, commit, _)| (event.seq, commit.boundaries.recent_from_seq))
564        };
565        let mut source_from_seq = summary_base.map_or(1, |(seq, _)| seq.saturating_add(1));
566        if let Some((_, Some(recent_from))) = summary_base {
567            source_from_seq = source_from_seq.min(recent_from);
568        }
569        let boundaries = ContextCheckpointBoundaries {
570            source_from_seq,
571            source_through_seq: input_through_seq,
572            input_through_seq,
573            prior_checkpoint_seq: (previous > 0).then_some(previous),
574            recent_from_seq: None,
575        };
576        let summary_limit = SummaryByteLimit::new(config.max_summary_bytes as u64)
577            .map_err(|error| EngineError::from(ModelError::invalid_request(error.to_string())))?;
578        let checkpoint_prefix =
579            model_history::checkpoint_retained_history(&context.history, &events, None);
580        let prior_summary_count = usize::from(summary_base.is_some());
581
582        let native = async {
583            let Some(model) = &native_model else {
584                return NativeOutcome::default();
585            };
586            self.native_compaction_attempt(
587                &input,
588                model,
589                &events,
590                &composed_prompt,
591                compaction_focus.clone(),
592                input_through_seq,
593            )
594            .await
595        };
596        let summary = async {
597            // Reserve an output-sized summary before selecting a suffix. The reserve is the
598            // compaction agent's own effective output cap, which it inherits from the owner run
599            // when its document declares none. The final fit check uses the actual replay
600            // projection and the same calibrated estimator as the input.
601            let output_reserve = internal_agent_output_limit(input.binding, input.internal_policy)
602                .unwrap_or(DEFAULT_COMPACTION_OUTPUT_RESERVE_TOKENS);
603            let retained_limit = input_budget
604                .saturating_sub(output_reserve)
605                .min(if trigger_tokens > 0 {
606                    trigger_tokens - 1
607                } else {
608                    input_budget
609                })
610                .min(context_tokens_before.saturating_sub(1));
611            let summary_output_limit = input
612                .internal_policy
613                .models
614                .iter()
615                .filter_map(|binding| internal_agent_output_limit(binding, input.internal_policy))
616                .max()
617                .unwrap_or(DEFAULT_COMPACTION_OUTPUT_RESERVE_TOKENS);
618            let summary_reserve_bytes = summary_output_limit
619                .saturating_mul(4)
620                .min(config.max_summary_bytes as u64)
621                as usize;
622            let summary_reserve = "x".repeat(summary_reserve_bytes);
623            let base = model_history::project_summary_context(
624                &events,
625                &self.inner.artifacts,
626                input.binding,
627                &composed_prompt,
628                input_through_seq,
629                None,
630                &summary_reserve,
631            )?;
632            let base_tokens =
633                self.estimated_request_tokens(input.session, &base.history, input.tools)?;
634            let keep_recent_tokens = effective_recent_budget(
635                config.keep_recent_tokens,
636                input_budget,
637                retained_limit.saturating_sub(base_tokens),
638            );
639            let recent_from_seq = select_recent_tail(
640                model_history::compaction_tail_candidates(&events),
641                keep_recent_tokens,
642                base_tokens,
643                retained_limit,
644                |candidate| {
645                    let projected = model_history::project_summary_context(
646                        &events,
647                        &self.inner.artifacts,
648                        input.binding,
649                        &composed_prompt,
650                        input_through_seq,
651                        Some(candidate),
652                        &summary_reserve,
653                    )?;
654                    self.estimated_request_tokens(input.session, &projected.history, input.tools)
655                },
656            )?;
657            let prefix = model_history::compaction_prefix_history(
658                &events,
659                &self.inner.artifacts,
660                input.binding,
661                &composed_prompt,
662                recent_from_seq,
663            )?;
664            if prefix.len() <= checkpoint_prefix.len().saturating_add(prior_summary_count) {
665                return Ok::<Option<SummaryParts>, EngineError>(None);
666            }
667            // The summarizer sees only the span the checkpoint replaces; the
668            // retained recent suffix follows the summary verbatim.
669            let instruction = compaction_instruction(
670                &input.internal_policy.agent.composed_prompt,
671                compaction_focus.as_deref(),
672                prior_summary_count > 0,
673            );
674            let retry_prefix = prefix.clone();
675            let mut summary = self
676                .run_internal_history_agent(
677                    input.session,
678                    Some(input.run),
679                    InternalAgentKind::ContextCompaction,
680                    input.internal_policy,
681                    InternalAgentHistoryInput {
682                        history: compaction_history(prefix, &instruction),
683                        summary_source: instruction.clone(),
684                        // First trial keeps the session tool definitions so the
685                        // request stays a cache-friendly prefix of the latest
686                        // conversation request; a tool-call answer is rejected by
687                        // reject_non_text and counted as a compaction failure.
688                        tools: input.tools.to_vec(),
689                        reject_non_text: true,
690                    },
691                    InternalAgentExecution {
692                        cancellation: input.cancellation,
693                        actor_direct: false,
694                    },
695                )
696                .await;
697            // A replayed reasoning block the provider binds to a conversation
698            // prefix that changed since (Anthropic's thinking-binding check)
699            // fails the summary as it fails a turn; the summary is tried once
700            // more without reasoning, which it does not need
701            // (`docs/site/specs/thinking-replay-recovery.md`).
702            if matches!(&summary, Err(EngineError::Model(error))
703                if super::model_loop::is_thinking_binding_mismatch(error))
704                && !input.cancellation.is_cancelled()
705            {
706                let mut history = retry_prefix;
707                model_history::strip_reasoning(&mut history);
708                summary = self
709                    .run_internal_history_agent(
710                        input.session,
711                        Some(input.run),
712                        InternalAgentKind::ContextCompaction,
713                        input.internal_policy,
714                        InternalAgentHistoryInput {
715                            history: compaction_history(history, &instruction),
716                            summary_source: instruction.clone(),
717                            tools: input.tools.to_vec(),
718                            reject_non_text: true,
719                        },
720                        InternalAgentExecution {
721                            cancellation: input.cancellation,
722                            actor_direct: false,
723                        },
724                    )
725                    .await;
726            }
727            if matches!(&summary, Err(EngineError::Model(error))
728                if error.kind == oven_sdk::ModelErrorKind::ContextLength)
729                && !input.cancellation.is_cancelled()
730            {
731                let history = pruned_compaction_history(
732                    &events,
733                    &self.inner.artifacts,
734                    input.session,
735                    input.binding,
736                    &composed_prompt,
737                    recent_from_seq,
738                )?;
739                summary = self
740                    .run_internal_history_agent(
741                        input.session,
742                        Some(input.run),
743                        InternalAgentKind::ContextCompaction,
744                        input.internal_policy,
745                        InternalAgentHistoryInput {
746                            history: compaction_history(history, &instruction),
747                            summary_source: instruction,
748                            // The pruned retry already rewrites tool outputs, so it
749                            // drops the tool definitions too: cache affinity is
750                            // lost either way, and fewer tools means less to send.
751                            tools: Vec::new(),
752                            reject_non_text: true,
753                        },
754                        InternalAgentExecution {
755                            cancellation: input.cancellation,
756                            actor_direct: false,
757                        },
758                    )
759                    .await;
760            }
761            let Ok(summary) = summary else {
762                if !input.force
763                    && let Some(run) = &active_run
764                    && run.auto_compaction_failures.fetch_add(1, Ordering::Relaxed) + 1
765                        == MAX_AUTO_COMPACTION_FAILURES
766                    && !run
767                        .auto_compaction_diagnostic_emitted
768                        .swap(true, Ordering::Relaxed)
769                {
770                    let engine = self.clone();
771                    let session = input.session;
772                    tokio::spawn(async move {
773                        let _ = engine
774                            .on_actor(session, move |engine| {
775                                engine.append_direct(
776                                    session,
777                                    None,
778                                    super::event_origin("engine:auto-compact"),
779                                    Event::Diagnostic {
780                                        source: cookie_agent_protocol::DiagnosticSource::engine("auto-compact")
781                                            .expect("static diagnostic source is valid"),
782                                        kind: cookie_agent_protocol::EngineDiagnosticKind::AutoCompactionDisabled.label(),
783                                        message: "automatic context compaction disabled after 3 consecutive failures for this run".into(),
784                                        count: None,
785                                        detail: None,
786                                    },
787                                )
788                            })
789                            .await;
790                    });
791                }
792                return Ok::<Option<SummaryParts>, EngineError>(None);
793            };
794            if summary.text.trim().is_empty() {
795                return Ok::<Option<SummaryParts>, EngineError>(None);
796            }
797            let checkpoint = InternalSummaryCheckpoint::new(
798                summary.text,
799                summary.invocation_id,
800                summary.internal_run_id,
801                summary_limit,
802            )
803            .map_err(|error| EngineError::from(ModelError::invalid_response(error.to_string())))?;
804            let retained_context = model_history::project_summary_context(
805                &events,
806                &self.inner.artifacts,
807                input.binding,
808                &composed_prompt,
809                input_through_seq,
810                recent_from_seq,
811                checkpoint.summary(),
812            )?;
813            let input_tokens_after = self.estimated_request_tokens(
814                input.session,
815                &retained_context.history,
816                input.tools,
817            )?;
818            let actual_base = model_history::project_summary_context(
819                &events,
820                &self.inner.artifacts,
821                input.binding,
822                &composed_prompt,
823                input_through_seq,
824                None,
825                checkpoint.summary(),
826            )?;
827            let actual_base_tokens =
828                self.estimated_request_tokens(input.session, &actual_base.history, input.tools)?;
829            let keep_recent_tokens = effective_recent_budget(
830                config.keep_recent_tokens,
831                input_budget,
832                retained_limit.saturating_sub(actual_base_tokens),
833            );
834            if input_tokens_after > retained_limit
835                || input_tokens_after.saturating_sub(actual_base_tokens) > keep_recent_tokens
836            {
837                return Ok::<Option<SummaryParts>, EngineError>(None);
838            }
839            let mut boundaries = boundaries.clone();
840            boundaries.recent_from_seq = recent_from_seq;
841            if let Some(recent_from_seq) = recent_from_seq {
842                boundaries.source_from_seq = boundaries.source_from_seq.min(recent_from_seq);
843            }
844            let budgets = ContextCheckpointBudgets {
845                context_limit_tokens: context_limit,
846                trigger_tokens: trigger_tokens.max(1).min(context_limit),
847                input_tokens_before: context_tokens_before,
848                input_tokens_after,
849                max_summary_bytes: summary_limit,
850                keep_recent_tokens,
851            };
852            Ok::<Option<SummaryParts>, EngineError>(Some((checkpoint, boundaries, budgets)))
853        };
854        let (native, summary) = tokio::join!(native, summary);
855        if let Some((state, reason)) = native.mark {
856            self.inner
857                .native_compaction
858                .mark(input.binding, state, reason);
859        }
860        let Some((checkpoint, boundaries, budgets)) = summary? else {
861            return Ok(events);
862        };
863        let mut commit = ContextCheckpointCommit {
864            checkpoint: ContextCheckpoint::InternalSummary {
865                checkpoint,
866                native: native.window,
867            },
868            boundaries,
869            budgets,
870        };
871        if commit.validate().is_err() {
872            // A window the boundaries cannot hold leaves the summary alone.
873            if let ContextCheckpoint::InternalSummary { native, .. } = &mut commit.checkpoint {
874                *native = None;
875            }
876            if commit.validate().is_err() {
877                return Ok(events);
878            }
879        }
880        self.append(
881            input.session,
882            Some(input.run),
883            input.origin.clone(),
884            Event::ContextCheckpointCommitted { commit },
885        )
886        .await?;
887        if let Some(run) = active_run {
888            run.auto_compaction_failures.store(0, Ordering::Relaxed);
889        }
890        Ok(self.inner.store.log(input.session)?.event_snapshot())
891    }

915    /// The context a request on `binding` sends (native compaction spec
916    /// section 4): the latest checkpoint's window when the provider record
917    /// allows windows, the window belongs to the binding's adapter,
918    /// provider, and model, and `model` accepts it; the summary otherwise.
919    pub(super) fn request_context(
920        &self,
921        events: &[Arc<StoredEvent>],
922        binding: &cookie_agent_protocol::FrozenModelBinding,
923        composed_prompt: &str,
924        model: &dyn oven_sdk::LanguageModel,
925    ) -> Result<model_history::ModelContext, EngineError> {
926        let use_windows = self.inner.native_compaction.allows_windows(binding);
927        let context = model_history::assemble_model_context_with(
928            events,
929            &self.inner.artifacts,
930            binding,
931            composed_prompt,
932            use_windows,
933        )?;
934        let Some(window) = &context.native_context else {
935            return Ok(context);
936        };
937        // The adapter checks the window's scope (its endpoint and account)
938        // against its own before any I/O; a window it would refuse serves no
939        // request, so the summary does.
940        let probe = ModelRequest::new(context.history.clone()).with_native_context(window.clone());
941        match model.validate_request(&probe) {
942            Err(error) if error.kind == oven_sdk::ModelErrorKind::NativeContext => {
943                Ok(model_history::assemble_model_context_with(
944                    events,
945                    &self.inner.artifacts,
946                    binding,
947                    composed_prompt,
948                    false,
949                )?)
950            }
951            _ => Ok(context),
952        }
953    }

955    /// One native compaction call beside the summary: the window it returns,
956    /// or how its failure marks the provider.
957    async fn native_compaction_attempt(
958        &self,
959        input: &CompactionInput<'_>,
960        model: &policy::ResolvedRuntimeModel,
961        events: &[Arc<StoredEvent>],
962        composed_prompt: &str,
963        focus: Option<String>,
964        input_through_seq: u64,
965    ) -> NativeOutcome {
966        let Ok(view) = self.native_compaction_view(events, input.binding, composed_prompt, model)
967        else {
968            return NativeOutcome::default();
969        };
970        let Ok(header_context) = self.model_header_context(input.session) else {
971            return NativeOutcome::default();
972        };
973        let Some(capabilities) = input.owner_policy.model_capabilities(input.binding) else {
974            return NativeOutcome::default();
975        };
976        let mut request = ModelRequest::new(view.history)
977            .with_tools(input.tools.to_vec())
978            .with_header_context(header_context);
979        let adapter = input.binding.descriptor.adapter_id.as_str();
980        crate::media::fit_media(
981            &mut request.history,
982            &input.binding.selection.model,
983            &policy::user_input_media(adapter, &capabilities),
984            &policy::tool_input_media(adapter, &capabilities),
985        );
986        if crate::media::elide_excess_media(&mut request.history, &capabilities).is_err() {
987            return NativeOutcome::default();
988        }
989        let seeded = view.native_context.is_some();
990        if let Some(native_context) = view.native_context {
991            request = request.with_native_context(native_context);
992        }
993        let cache_strategy = input
994            .owner_policy
995            .cache_strategy(input.binding, input.session);
996        let request = model.prepare_request_with_cache_strategy(request, cache_strategy.as_ref());
997        let compact_request =
998            with_native_compaction_instructions(CompactionRequest::new(request), adapter, focus);
999        if !model.model().supports_compaction(&compact_request) {
1000            return NativeOutcome {
1001                window: None,
1002                mark: Some((
1003                    native_compaction::MarkState::Unsupported,
1004                    "the adapter cannot build the compact request".into(),
1005                )),
1006            };
1007        }
1008        let resolved_model = wire_model(input.binding);
1009        self.publish_transient(
1010            input.session,
1011            input.run,
1012            Event::NativeCompactionStarted {
1013                resolved_model: resolved_model.clone(),
1014            },
1015        );
1016        let abort = AbortBridge::new(input.cancellation.child_token());
1017        let result = model.model().compact(compact_request, abort.signal()).await;
1018        self.publish_transient(
1019            input.session,
1020            input.run,
1021            Event::NativeCompactionFinished { resolved_model },
1022        );
1023        match result {
1024            Ok(result) => NativeOutcome {
1025                window: model_history::persist_native_context(result.native_context, input.binding)
1026                    .ok()
1027                    .filter(|window| window.validate_for_binding(input.binding).is_ok())
1028                    .map(|window| CheckpointNativeWindow {
1029                        window,
1030                        source_through_seq: input_through_seq,
1031                    }),
1032                mark: None,
1033            },
1034            Err(error) => NativeOutcome {
1035                window: None,
1036                mark: native_compaction::compact_failure_mark(&error, seeded),
1037            },
1038        }
1039    }

1041    async fn stage_tool_output_elision(
1042        &self,
1043        session: SessionId,
1044        events: &[Arc<StoredEvent>],
1045        origin: cookie_agent_protocol::EventOrigin,
1046    ) -> Result<crate::events::EventSnapshot, EngineError> {
1047        let protected_turns = events
1048            .iter()
1049            .rev()
1050            .filter_map(|event| match &event.payload {
1051                Event::ModelTurnCommitted { model_turn_seq, .. } => Some(*model_turn_seq),
1052                _ => None,
1053            })
1054            .take(2)
1055            .collect::<HashSet<_>>();
1056        let starts = events
1057            .iter()
1058            .filter_map(|event| match &event.payload {
1059                Event::ToolCallStarted { start } => {
1060                    Some((start.tool_call_id, start.owner.model_turn_seq))
1061                }
1062                _ => None,
1063            })
1064            .collect::<HashMap<_, _>>();
1065        let already_elided = events
1066            .iter()
1067            .filter_map(|event| match event.payload {
1068                Event::ToolOutputElided { tool_call_id, .. } => Some(tool_call_id),
1069                _ => None,
1070            })
1071            .collect::<HashSet<_>>();
1072        for event in events {
1073            let Event::ToolCallTerminated { termination } = &event.payload else {
1074                continue;
1075            };
1076            let Some(result) = &termination.result else {
1077                continue;
1078            };
1079            let Some(model_turn_seq) = starts.get(&termination.tool_call_id) else {
1080                continue;
1081            };
1082            if !should_elide_tool_output(
1083                *model_turn_seq,
1084                &protected_turns,
1085                already_elided.contains(&termination.tool_call_id),
1086                elidable_bytes(result),
1087            ) {
1088                continue;
1089            }
1090            let (retained, _) = self
1091                .inner
1092                .artifacts
1093                .retain(session, result.output.as_bytes())?;
1094            self.append(
1095                session,
1096                event.run_id,
1097                origin.clone(),
1098                Event::ToolOutputElided {
1099                    tool_call_id: termination.tool_call_id,
1100                    original_bytes: result.output.len() as u64,
1101                    retained,
1102                },
1103            )
1104            .await?;
1105        }
1106        Ok(self.inner.store.log(session)?.event_snapshot())
1107    }

1161/// Resolves the trigger against the model's input budget (see `policy::input_token_budget`).
1162pub(super) fn resolve_compaction_trigger(
1163    input_budget: u64,
1164    trigger: &ContextCompactionTrigger,
1165) -> u64 {
1166    match trigger {
1167        ContextCompactionTrigger::Percent { percent } => input_budget
1168            .saturating_mul(u64::from(*percent))
1169            .saturating_div(100),
1170        ContextCompactionTrigger::BufferTokens { buffer_tokens } => {
1171            input_budget.saturating_sub(*buffer_tokens)
1172        }
1173    }
1174}

1188fn select_recent_tail(
1189    candidates: Vec<u64>,
1190    budget: u64,
1191    base_tokens: u64,
1192    retained_limit: u64,
1193    mut estimate: impl FnMut(u64) -> Result<u64, EngineError>,
1194) -> Result<Option<u64>, EngineError> {
1195    let mut selected = None;
1196    if budget == 0 {
1197        return Ok(selected);
1198    }
1199    for candidate in candidates.into_iter().rev() {
1200        let tokens = estimate(candidate)?;
1201        if tokens > retained_limit || tokens.saturating_sub(base_tokens) > budget {
1202            break;
1203        }
1204        selected = Some(candidate);
1205    }
1206    Ok(selected)
1207}

1220pub(crate) fn serialized_fit_request_bytes(
1221    history: &[oven_sdk::HistoryTurn],
1222    tools: &[ToolDefinition],
1223) -> Result<usize, EngineError> {
1224    // Media costs are flat-ish, usually hundreds to low-thousands of tokens per part, and do not
1225    // track serialized byte size. Text parts count their raw length, image/PDF/audio files count
1226    // nothing, and video alone gets a flat surrogate because frame sampling can be materially
1227    // expensive. Everything else counts its serialized size. Overestimation can trigger
1228    // unnecessary compaction and discard attachments, while real usage calibrates residual error.
1229    let mut part_bytes = 0_usize;
1230    let history = history
1231        .iter()
1232        .map(|turn| fit::FitTurn::new(turn, &mut part_bytes))
1233        .collect::<Vec<_>>();
1234    let mut writer = CountingWriter::default();
1235    serde_json::to_writer(&mut writer, &(history, tools))
1236        .map_err(|error| EngineError::from(ModelError::invalid_request(error.to_string())))?;
1237    Ok(writer.bytes.saturating_add(part_bytes))
1238}

1470fn elidable_bytes(result: &PersistedToolResult) -> usize {
1471    let emitted_bytes = result
1472        .additional_messages
1473        .iter()
1474        .map(|message| {
1475            let marker_bytes =
1476                usize::from(message.role == cookie_agent_protocol::ToolEmittedMessageRole::System)
1477                    .saturating_mul(fit_part_bytes(FitPart::Text(
1478                        model_history::TOOL_EMITTED_SYSTEM_USER_MARKER,
1479                    )));
1480            message.content.iter().fold(marker_bytes, |total, part| {
1481                total.saturating_add(match part {
1482                    ToolEmittedContent::Text(text) => fit_part_bytes(FitPart::Text(text)),
1483                    ToolEmittedContent::File(attachment) => {
1484                        fit_part_bytes(FitPart::File(attachment.mime_type.as_str()))
1485                    }
1486                })
1487            })
1488        })
1489        .fold(0_usize, usize::saturating_add);
1490    result.attachments.iter().fold(
1491        fit_part_bytes(FitPart::Text(&result.output)).saturating_add(emitted_bytes),
1492        |total, attachment| {
1493            total.saturating_add(fit_part_bytes(FitPart::File(attachment.mime_type.as_str())))
1494        },
1495    )
1496}

1588fn pruned_compaction_history(
1589    events: &[Arc<StoredEvent>],
1590    store: &super::artifacts::ArtifactRouter,
1591    session: SessionId,
1592    binding: &cookie_agent_protocol::FrozenModelBinding,
1593    composed_prompt: &str,
1594    recent_from_seq: Option<u64>,
1595) -> Result<Vec<oven_sdk::HistoryTurn>, EngineError> {
1596    // Emitted messages lose their tool ownership after assembly. Remove them in a private
1597    // snapshot first, then prune only results in the active checkpoint-aware history.
1598    let mut events = events.to_vec();
1599    for event in &mut events {
1600        let carries_result = matches!(
1601            &event.payload,
1602            Event::ToolCallTerminated { termination } if termination.result.is_some()
1603        );
1604        // Copy only the records this private snapshot changes.
1605        if carries_result
1606            && let Event::ToolCallTerminated { termination } = &mut Arc::make_mut(event).payload
1607            && let Some(result) = &mut termination.result
1608        {
1609            result.additional_messages.clear();
1610            result.attachments.clear();
1611        }
1612    }
1613    let mut history = model_history::compaction_prefix_history(
1614        &events,
1615        store,
1616        binding,
1617        composed_prompt,
1618        recent_from_seq,
1619    )?;
1620    let mut retrieval_calls = HashSet::new();
1621    for turn in &mut history {
1622        match turn {
1623            oven_sdk::HistoryTurn::Assistant(turn) => {
1624                for part in &turn.message.content {
1625                    if let oven_sdk::AssistantPart::ToolCall(call) = part {
1626                        if call.name == "read_artifact" {
1627                            retrieval_calls.insert(call.id.clone());
1628                        } else {
1629                            retrieval_calls.remove(&call.id);
1630                        }
1631                    }
1632                }
1633                for part in &mut turn.message.content {
1634                    if let oven_sdk::AssistantPart::ToolResult(result) = part {
1635                        prune_compaction_result(result, &retrieval_calls, store, session)?;
1636                    }
1637                }
1638            }
1639            oven_sdk::HistoryTurn::Tool(message) => {
1640                for result in &mut message.results {
1641                    prune_compaction_result(result, &retrieval_calls, store, session)?;
1642                }
1643            }
1644            _ => {}
1645        }
1646    }
1647    Ok(history)
1648}

1650/// What an elision marker adds when the retained content was not plain
1651/// text: how [`line_structured_content`] lays it out.
1652const LINE_STRUCTURED_NOTE: &str = " Stored as text: a `[tool content: ...]` header line, then each part under its own `[part N: kind]` line, text with its own line breaks and JSON pretty-printed.";
1653
1654/// Tool content that is not plain text, laid out so `read_artifact` pages
1655/// it line by line: a header line naming its kind, then each part under a
1656/// `[part N: kind]` line, text with its own newlines, JSON pretty-printed
1657/// over several lines, and a file by its media type and name (its bytes
1658/// are not kept). Earlier builds stored such content as one line of
1659/// serialized JSON, which a page cuts after 4 KiB.
1660fn line_structured_content(content: &oven_sdk::ToolContent) -> serde_json::Result<String> {
1661    fn push_section(text: &mut String, header: &str, body: &str) {
1662        text.push_str(header);
1663        text.push('\n');
1664        text.push_str(body);
1665        if !body.is_empty() && !body.ends_with('\n') {
1666            text.push('\n');
1667        }
1668    }
1669    let mut text = String::new();
1670    match content {
1671        oven_sdk::ToolContent::Text(body) => text.push_str(body),
1672        oven_sdk::ToolContent::Json(value) => {
1673            push_section(
1674                &mut text,
1675                "[tool content: json]",
1676                &serde_json::to_string_pretty(value)?,
1677            );
1678        }
1679        oven_sdk::ToolContent::Denied { reason } => {
1680            push_section(
1681                &mut text,
1682                "[tool content: denied]",
1683                reason.as_deref().unwrap_or(""),
1684            );
1685        }
1686        oven_sdk::ToolContent::Mixed(parts) => {
1687            text.push_str(&format!(
1688                "[tool content: mixed, {} part{}]\n",
1689                parts.len(),
1690                if parts.len() == 1 { "" } else { "s" }
1691            ));
1692            for (index, part) in parts.iter().enumerate() {
1693                let number = index + 1;
1694                match part {
1695                    oven_sdk::ContentValue::Text(body) => {
1696                        push_section(&mut text, &format!("[part {number}: text]"), body);
1697                    }
1698                    oven_sdk::ContentValue::Json(value) => push_section(
1699                        &mut text,
1700                        &format!("[part {number}: json]"),
1701                        &serde_json::to_string_pretty(value)?,
1702                    ),
1703                    oven_sdk::ContentValue::File(file) => {
1704                        let name = match &file.filename {
1705                            Some(name) => format!(" {}", serde_json::to_string(name)?),
1706                            None => String::new(),
1707                        };
1708                        push_section(
1709                            &mut text,
1710                            &format!(
1711                                "[part {number}: file {}{name}; its bytes are not kept here]",
1712                                file.media_type
1713                            ),
1714                            "",
1715                        );
1716                    }
1717                }
1718            }
1719        }
1720    }
1721    Ok(text)
1722}

1724fn prune_compaction_result(
1725    result: &mut oven_sdk::ToolResultPart,
1726    retrieval_calls: &HashSet<String>,
1727    store: &super::artifacts::ArtifactRouter,
1728    session: SessionId,
1729) -> Result<(), EngineError> {
1730    let marker = if retrieval_calls.contains(&result.tool_call_id) {
1731        // This is a private summary-input copy; the original result stays in history.
1732        "[artifact read output omitted for compaction]".to_owned()
1733    } else {
1734        let content = match &result.content {
1735            oven_sdk::ToolContent::Text(text) => text.clone(),
1736            content => line_structured_content(content)
1737                .map_err(|error| ModelError::invalid_request(error.to_string()))?,
1738        };
1739        let (_, artifact_id) = store.retain(session, content.as_bytes())?;
1740        let mut marker =
1741            model_history::tool_output_elision_marker(&artifact_id, content.len() as u64, 0);
1742        if !matches!(result.content, oven_sdk::ToolContent::Text(_)) {
1743            marker.push_str(LINE_STRUCTURED_NOTE);
1744        }
1745        let read_more = serde_json::json!({
1746            "read_more": {
1747                "tool": "read_artifact",
1748                "arguments": {"artifact": artifact_id}
1749            }
1750        });
1751        marker.push('\n');
1752        marker.push_str(&read_more.to_string());
1753        marker
1754    };
1755    result.content = oven_sdk::ToolContent::Text(marker);
1756    result.metadata = None;
1757    Ok(())
1758}

1760fn compaction_history(
1761    mut history: Vec<oven_sdk::HistoryTurn>,
1762    instruction: &str,
1763) -> Vec<oven_sdk::HistoryTurn> {
1764    // `history` is the span the checkpoint replaces. It starts with the
1765    // session's own system prompt and pinned context, so the summarizer
1766    // request is a cache-friendly prefix of the latest conversation request;
1767    // the compaction agent's prompt rides along in the trailing instruction
1768    // instead of replacing the session system prompt.
1769    history.push(oven_sdk::HistoryTurn::user(oven_sdk::UserMessage::new(
1770        vec![oven_sdk::InputPart::Text(oven_sdk::TextPart::new(
1771            instruction.to_owned(),
1772        ))],
1773    )));
1774    history
1775}
