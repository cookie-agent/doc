8/// Boundaries between composed system-prompt blocks are rendered as a markdown
9/// horizontal rule so block transitions are visible in the assembled prompt.
10pub(crate) const PROMPT_BLOCK_SEPARATOR: &str = "\n\n---\n\n";
11
12pub(crate) fn push_prompt_block(prompt: &mut String, block: &str) {
13    if !prompt.is_empty() {
14        prompt.push_str(PROMPT_BLOCK_SEPARATOR);
15    }
16    prompt.push_str(block);
17}

19/// The prompt block of where the session's runtimes work.
20pub(crate) const RUNTIMES_BLOCK: &str = "runtimes";
21/// The prompt block of the tool providers' `<tool_instructions>` sections.
22pub(crate) const TOOL_INSTRUCTIONS_BLOCK: &str = "tool_instructions";
23/// The prompt block of the `<available_skills>` listing.
24pub(crate) const SKILLS_BLOCK: &str = "skills";

26/// Appends prompt block `key` to `policy`'s composed prompt, updates the
27/// prompt and document fingerprints, and records where the block sits
28/// (stable prefix spec, section 6.1).
29pub(crate) fn append_prompt_block(policy: &mut FrozenRunPolicy, key: &str, block: &str) {
30    let prompt = &mut policy.agent.composed_prompt;
31    push_prompt_block(prompt, block);
32    let start = prompt.len() - block.len();
33    if let (Ok(start), Ok(len)) = (u32::try_from(start), u32::try_from(block.len())) {
34        policy.prompt_blocks.push(PromptBlockSpan {
35            key: key.to_owned(),
36            start,
37            len,
38        });
39    }
40    policy.agent.prompt_fingerprint = Sha256Digest::of_bytes(prompt.as_bytes());
41    let mut document_fingerprint = policy
42        .agent
43        .document_fingerprint
44        .as_str()
45        .as_bytes()
46        .to_vec();
47    document_fingerprint.extend_from_slice(block.as_bytes());
48    policy.agent.document_fingerprint = Sha256Digest::of_bytes(&document_fingerprint);
49}

51/// Whether `block` fits after `policy`'s composed prompt.
52pub(crate) fn prompt_block_fits(policy: &FrozenRunPolicy, block: &str) -> bool {
53    policy
54        .agent
55        .composed_prompt
56        .len()
57        .checked_add(block.len() + PROMPT_BLOCK_SEPARATOR.len())
58        .is_some_and(|bytes| bytes <= AgentSnapshot::MAX_PROMPT_BYTES)
59}

61/// The prompt blocks `spans` locate in `prompt`, in order; `None` when a
62/// span does not fit the prompt.
63pub(crate) fn prompt_block_texts(
64    prompt: &str,
65    spans: &[PromptBlockSpan],
66) -> Option<Vec<(String, String)>> {
67    spans
68        .iter()
69        .map(|span| {
70            let start = usize::try_from(span.start).ok()?;
71            let end = start.checked_add(usize::try_from(span.len).ok()?)?;
72            Some((span.key.clone(), prompt.get(start..end)?.to_owned()))
73        })
74        .collect()
75}

77impl Engine {
78    /// The previous run's prompt blocks, which a run on a model that defers
79    /// tools reuses unless a checkpoint lies after that run's start (stable
80    /// prefix spec, section 6.2).
81    pub(crate) fn reusable_prompt_blocks(
82        &self,
83        session: SessionId,
84    ) -> Result<Option<Vec<(String, String)>>, EngineError> {
85        let fold = self.inner.store.get(session)?.log.prompt_blocks();
86        Ok(if fold.checkpoint_since_run {
87            None
88        } else {
89            fold.run_blocks
90        })
91    }
92
93    /// The prompt blocks a run of `policy` in `session` composes now, in
94    /// prompt order.
95    fn live_prompt_blocks(
96        &self,
97        session: SessionId,
98        policy: &FrozenRunPolicy,
99    ) -> Result<Vec<(String, String)>, EngineError> {
100        let mut blocks = Vec::new();
101        if let Some(block) = self.runtimes_block(session, None, Some(policy)) {
102            blocks.push((RUNTIMES_BLOCK.to_owned(), block));
103        }
104        if let Some(block) = self.live_tool_instructions_block(policy, session)? {
105            blocks.push((TOOL_INSTRUCTIONS_BLOCK.to_owned(), block));
106        }
107        if let Some(block) = self.live_skills_block(session, policy, None)? {
108            blocks.push((SKILLS_BLOCK.to_owned(), block));
109        }
110        Ok(blocks)
111    }
112
113    /// The announcement of the prompt blocks that changed since the model
114    /// last knew them, if any did (stable prefix spec, section 6.3).
115    pub(crate) fn prompt_blocks_announcement(
116        &self,
117        session: SessionId,
118        policy: &FrozenRunPolicy,
119    ) -> Result<Option<cookie_agent_protocol::EventPayload>, EngineError> {
120        let Some(known) = self.inner.store.get(session)?.log.prompt_blocks().known else {
121            return Ok(None);
122        };
123        let live = self.live_prompt_blocks(session, policy)?;
124        let mut blocks = live
125            .iter()
126            .filter(|(key, text)| known.get(key) != Some(text))
127            .map(|(key, text)| AnnouncedPromptBlock {
128                key: key.clone(),
129                text: Some(text.clone()),
130            })
131            .collect::<Vec<_>>();
132        blocks.extend(
133            known
134                .keys()
135                .filter(|key| !live.iter().any(|(live, _)| live == *key))
136                .map(|key| AnnouncedPromptBlock {
137                    key: key.clone(),
138                    text: None,
139                }),
140        );
141        Ok((!blocks.is_empty())
142            .then_some(cookie_agent_protocol::EventPayload::PromptBlocksAnnounced { blocks }))
143    }
144}

146#[cfg(test)]
147mod tests {
148    use super::{PROMPT_BLOCK_SEPARATOR, push_prompt_block};
149
150    #[test]
151    fn separator_is_a_markdown_horizontal_rule() {
152        assert_eq!(PROMPT_BLOCK_SEPARATOR, "\n\n---\n\n");
153    }
154
155    #[test]
156    fn empty_prompt_gets_no_leading_separator() {
157        let mut prompt = String::new();
158        push_prompt_block(&mut prompt, "first");
159        assert_eq!(prompt, "first");
160    }
161
162    #[test]
163    fn blocks_are_joined_by_exactly_one_separator() {
164        let mut prompt = String::new();
165        push_prompt_block(&mut prompt, "first");
166        push_prompt_block(&mut prompt, "second");
167        assert_eq!(prompt, "first\n\n---\n\nsecond");
168        assert_eq!(prompt.matches(PROMPT_BLOCK_SEPARATOR).count(), 1);
169    }
