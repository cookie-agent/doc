33/// How often the scheduler looks at every candidate again, whatever woke
34/// it: configuration reloads and anything a change notice missed land
35/// within it.
36const FULL_PASS_EVERY: Duration = Duration::from_secs(60);
37
38/// How long the TTL of the `chatgpt` provider is assumed to be: the
39/// documented GPT-5.6+ default (section 11.2).
40const CHATGPT_ASSUMED_TTL: Duration = Duration::from_secs(30 * 60);
41
42/// How far back from a usage event the scan looks for the request it
43/// measured.
44const REQUEST_SCAN_LIMIT: usize = 4_096;
45
46/// The TTL a request cached its conversation for: what the cache strategy
47/// it was sent with sets, else what the configuration declares or assumes
48/// (section 4.1). `None` when it cached nothing past its system prompt and
49/// tools: an Anthropic or Bedrock request without a rolling marker.
50pub(crate) fn request_cache_ttl(
51    strategy: Option<&CacheStrategyConfig>,
52    chatgpt: bool,
53    declared: Option<Duration>,
54    fallback: Duration,
55) -> Option<CacheTtlStamp> {
56    let set = |seconds: u64| {
57        Some(CacheTtlStamp {
58            ttl_secs: seconds,
59            source: CacheTtlSource::Set,
60        })
61    };
62    let declared_or = |assumed: Duration| match declared {
63        Some(ttl) => Some(CacheTtlStamp {
64            ttl_secs: ttl.as_secs(),
65            source: CacheTtlSource::Declared,
66        }),
67        None => Some(CacheTtlStamp {
68            ttl_secs: assumed.as_secs().max(1),
69            source: CacheTtlSource::Assumed,
70        }),
71    };
72    if chatgpt {
73        return declared_or(CHATGPT_ASSUMED_TTL);
74    }
75    match strategy {
76        Some(CacheStrategyConfig::Anthropic(strategy)) => match strategy.rolling? {
77            AnthropicCacheTtlConfig::FiveMinutes => set(5 * 60),
78            AnthropicCacheTtlConfig::OneHour => set(60 * 60),
79        },
80        Some(CacheStrategyConfig::Bedrock(strategy)) => {
81            match strategy.messages.last()?.cache_point.ttl {
82                Some(BedrockCacheTtl::OneHour) => set(60 * 60),
83                Some(BedrockCacheTtl::FiveMinutes) | None => set(5 * 60),
84            }
85        }
86        Some(CacheStrategyConfig::OpenAi(strategy)) if strategy.mode.is_some() => {
87            match strategy.ttl.unwrap_or(OpenAiPromptCacheTtl::ThirtyMinutes) {
88                OpenAiPromptCacheTtl::ThirtyMinutes => set(30 * 60),
89                OpenAiPromptCacheTtl::TwelveHours => set(12 * 60 * 60),
90            }
91        }
92        Some(CacheStrategyConfig::OpenAi(strategy))
93            if strategy.prompt_cache_retention
94                == Some(OpenAiPromptCacheRetention::TwentyFourHours) =>
95        {
96            set(24 * 60 * 60)
97        }
98        _ => declared_or(fallback),
99    }
100}

102/// What the rule reads of a session's latest request.
103#[derive(Clone, Debug, PartialEq)]
104pub(crate) struct LatestRequest {
105    /// The sequence of its usage event.
106    pub(crate) usage_seq: u64,
107    /// When it was sent: its prepared request, else its usage event.
108    pub(crate) sent_at: Timestamp,
109    pub(crate) stamp: CacheTtlStamp,
110    /// Its reported input plus output tokens.
111    pub(crate) tokens: u64,
112    pub(crate) model: ResolvedModelRef,
113}

115/// The session's latest request, read back from `events`; `None` when it
116/// has none with a cache TTL, or a checkpoint came after it (rule 5).
117pub(crate) fn latest_request(events: &[Arc<StoredEvent>]) -> Option<LatestRequest> {
118    let mut checkpoint_after = false;
119    let mut found = None;
120    let mut attempt = None;
121    for (scanned, event) in events.iter().rev().enumerate() {
122        match (&event.payload, &found) {
123            (Event::ContextCheckpointCommitted { .. }, None) => checkpoint_after = true,
124            (
125                Event::ModelUsageRecorded {
126                    model_turn_seq,
127                    resolved_model,
128                    usage,
129                    cache_ttl,
130                    ..
131                },
132                None,
133            ) => {
134                if checkpoint_after {
135                    return None;
136                }
137                let stamp = (*cache_ttl)?;
138                let tokens = usage
139                    .input_tokens
140                    .unwrap_or(0)
141                    .saturating_add(usage.output_tokens.unwrap_or(0));
142                found = Some((
143                    *model_turn_seq,
144                    LatestRequest {
145                        usage_seq: event.seq,
146                        sent_at: event.timestamp,
147                        stamp,
148                        tokens,
149                        model: resolved_model.clone(),
150                    },
151                ));
152            }
153            (
154                Event::ModelTurnCommitted {
155                    model_turn_seq,
156                    attempt_id,
157                    ..
158                },
159                Some((turn, _)),
160            ) if model_turn_seq == turn && attempt.is_none() => attempt = Some(*attempt_id),
161            (Event::ModelRequestPrepared { attempt_id, .. }, Some(_))
162            | (Event::ModelAttemptStarted { attempt_id, .. }, Some(_))
163                if Some(*attempt_id) == attempt =>
164            {
165                let (_, mut request) = found.take()?;
166                request.sent_at = event.timestamp;
167                return Some(request);
168            }
169            _ => {}
170        }
171        if found.is_some() && scanned > REQUEST_SCAN_LIMIT {
172            break;
173        }
174    }
175    found.map(|(_, request)| request)
176}

201/// The rule of section 4.2: when, if at all, to compact a session whose
202/// latest request is `request`, at `now`.
203pub(crate) fn plan(
204    config: &CacheExpiryConfig,
205    request: &LatestRequest,
206    state: &SessionState,
207    now: Timestamp,
208) -> Option<Plan> {
209    if config.mode == CacheExpiryMode::Off || state.held || state.busy {
210        return None;
211    }
212    if state.tried == Some(request.usage_seq) {
213        return None;
214    }
215    let ttl = Duration::from_secs(request.stamp.ttl_secs);
216    let expires_at = request.stamp.expires_at(request.sent_at)?;
217    if now >= expires_at {
218        return None;
219    }
220    match (&state.next_fire, config.mode) {
221        // A fire before expiry reads a warm cache.
222        (Some((fire, _)), _) if *fire <= expires_at => return None,
223        (None, CacheExpiryMode::Scheduled) => return None,
224        _ => {}
225    }
226    if !worth_compacting(config, request.tokens, state.recache_cost_pico_usd) {
227        return None;
228    }
229    let slack = SignedDuration::try_from(config.effective_slack(ttl)).ok()?;
230    let due_at = expires_at.checked_sub(slack).ok()?;
231    Some(Plan {
232        due_at,
233        expires_at,
234        next_fire: state.next_fire.clone(),
235        usage_seq: request.usage_seq,
236    })
237}

239/// Condition 4: a floor at `0` or less is off; with both off everything
240/// passes, and otherwise reaching either one is enough. The cost floor
241/// needs a priced model.
242pub(crate) fn worth_compacting(
243    config: &CacheExpiryConfig,
244    tokens: u64,
245    recache_cost_pico_usd: Option<u64>,
246) -> bool {
247    let token_floor = u64::try_from(config.min_tokens)
248        .ok()
249        .filter(|floor| *floor > 0);
250    let cost_floor = (config.min_cost_usd > 0.0).then_some(config.min_cost_usd);
251    if token_floor.is_none() && cost_floor.is_none() {
252        return true;
253    }
254    #[allow(clippy::cast_precision_loss)]
255    let cost_usd = recache_cost_pico_usd.map(|pico| pico as f64 / 1e12);
256    token_floor.is_some_and(|floor| tokens >= floor)
257        || cost_floor.is_some_and(|floor| cost_usd.is_some_and(|cost| cost >= floor))
258}

260/// The scheduler's state, owned by [`super::Inner`].
261#[derive(Default)]
262pub(crate) struct CacheExpiryState {
263    /// The plan of each session that has one.
264    plans: Mutex<HashMap<SessionId, Plan>>,
265    /// Each session's latest request as last read, for sessions read while
266    /// resident; a paged-out session's stays as it was.
267    requests: Mutex<HashMap<SessionId, Option<(LatestRequest, bool)>>>,
268    /// The usage event each session's last compaction before expiry
269    /// followed (rule 6).
270    tried: Mutex<HashMap<SessionId, u64>>,
271    /// Sessions to read again.
272    dirty: Mutex<HashSet<SessionId>>,
273    /// Every candidate is to be looked at again.
274    full_pass: AtomicBool,
275    /// Sessions compacting now.
276    running: Mutex<HashSet<SessionId>>,
277    wake: Arc<Notify>,
278    tasks: Mutex<Vec<JoinHandle<()>>>,
279}

338    /// Starts the scheduler (section 4.3), like the cron scheduler, and the
339    /// task that turns session index changes into reads.
340    pub(in crate::runtime) fn start_cache_expiry_scheduler(&self) {
341        let Some(runtime) = self.inner.runtime.clone() else {
342            return;
343        };
344        let weak = Arc::downgrade(&self.inner);
345        let mut changes = self.inner.store.subscribe_index_changes();
346        let forwarder_weak = weak.clone();
347        let forwarder = runtime.spawn(async move {
348            use tokio::sync::broadcast::error::RecvError;
349            loop {
350                let change = changes.recv().await;
351                let Some(inner) = forwarder_weak.upgrade() else {
352                    break;
353                };
354                let state = &inner.cache_expiry;
355                match change {
356                    Ok(cookie_agent_protocol::SessionsChangedNotification::Upserted {
357                        session,
358                    }) => {
359                        lock(&state.dirty).insert(session.session_id);
360                        state.wake.notify_one();
361                    }
362                    Ok(_) => {}
363                    Err(RecvError::Lagged(_)) => {
364                        state.full_pass.store(true, Ordering::Release);
365                        lock(&state.requests).clear();
366                        state.wake.notify_one();
367                    }
368                    Err(RecvError::Closed) => break,
369                }
370            }
371        });
372        let mut crons = self.inner.crons.subscribe();
373        let cron_weak = weak.clone();
374        let cron_watch = runtime.spawn(async move {
375            while crons.changed().await.is_ok() {
376                let Some(inner) = cron_weak.upgrade() else {
377                    break;
378                };
379                inner.cache_expiry.full_pass.store(true, Ordering::Release);
380                inner.cache_expiry.wake.notify_one();
381            }
382        });
383        let task = runtime.spawn(async move {
384            let mut last_full_pass: Option<tokio::time::Instant> = None;
385            loop {
386                let Some(inner) = weak.upgrade() else {
387                    break;
388                };
389                let engine = Engine { inner };
390                if engine.admission_tasks_closing() || engine.restart_due() {
391                    break;
392                }
393                let full = engine
394                    .inner
395                    .cache_expiry
396                    .full_pass
397                    .swap(false, Ordering::AcqRel)
398                    || last_full_pass.is_none_or(|at| at.elapsed() >= FULL_PASS_EVERY);
399                if full {
400                    last_full_pass = Some(tokio::time::Instant::now());
401                }
402                engine.cache_expiry_pass(full).await;
403                let now = Timestamp::now();
404                let next_due = lock(&engine.inner.cache_expiry.plans)
405                    .values()
406                    .map(|plan| plan.due_at)
407                    .min();
408                let until_due = next_due.map(|due| {
409                    let millis = due.as_millisecond().saturating_sub(now.as_millisecond());
410                    Duration::from_millis(u64::try_from(millis).unwrap_or(0))
411                });
412                let sleep = until_due.unwrap_or(FULL_PASS_EVERY).min(FULL_PASS_EVERY);
413                let wake = Arc::clone(&engine.inner.cache_expiry.wake);
414                drop(engine);
415                tokio::select! {
416                    () = tokio::time::sleep(sleep) => {}
417                    () = wake.notified() => {}
418                }
419            }
420        });
421        lock(&self.inner.cache_expiry.tasks).extend([forwarder, cron_watch, task]);
422    }

429    /// One pass: reads the sessions to read, plans every candidate, tells
430    /// clients what changed, and starts the compactions that are due.
431    async fn cache_expiry_pass(&self, full: bool) {
432        let config = self
433            .inner
434            .config
435            .load()
436            .runtime
437            .context_compaction
438            .cache_expiry
439            .clone();
440        let state = &self.inner.cache_expiry;
441        let dirty = std::mem::take(&mut *lock(&state.dirty));
442        {
443            // A paged-out session's history cannot have changed: its last
444            // reading stands. Reading it again would load it back into
445            // memory, as the notice of its own plan would otherwise do.
446            let mut requests = lock(&state.requests);
447            for session in &dirty {
448                if self.inner.store.is_resident(*session) {
449                    requests.remove(session);
450                }
451            }
452        }
453        let mut candidates: HashSet<SessionId> = lock(&state.plans).keys().copied().collect();
454        if config.mode != CacheExpiryMode::Off {
455            candidates.extend(self.inner.crons.waiting_targets());
456            if config.mode == CacheExpiryMode::Always {
457                candidates.extend(
458                    self.inner
459                        .store
460                        .all()
461                        .iter()
462                        .map(|session| session.meta.session_id),
463                );
464            }
465        }
466        if !full {
467            candidates.retain(|session| {
468                dirty.contains(session) || lock(&state.plans).contains_key(session)
469            });
470        }
471        let busy = self.busy_roots();
472        let now = Timestamp::now();
473        let mut due = Vec::new();
474        for session in candidates {
475            let planned = if config.mode == CacheExpiryMode::Off {
476                None
477            } else {
478                self.plan_session(session, &config, &busy, now).await
479            };
480            if let Some(plan) = &planned
481                && plan.due_at <= now
482                && !lock(&state.running).contains(&session)
483            {
484                due.push((session, plan.clone()));
485            }
486            let changed = {
487                let mut plans = lock(&state.plans);
488                let previous = match planned.clone() {
489                    Some(plan) => plans.insert(session, plan),
490                    None => plans.remove(&session),
491                };
492                previous != planned
493            };
494            if changed {
495                self.announce_cache_compaction(session, planned.as_ref(), false);
496            }
497        }
498        for (session, plan) in due {
499            lock(&state.running).insert(session);
500            let engine = self.clone();
501            tokio::spawn(async move {
502                engine.compact_before_expiry(session, plan).await;
503                lock(&engine.inner.cache_expiry.running).remove(&session);
504                engine.cache_expiry_touch(session);
505            });
506        }
507    }

509    /// Tells clients `session`'s plan, `running` or not.
510    fn announce_cache_compaction(&self, session: SessionId, plan: Option<&Plan>, running: bool) {
511        let planned = plan.map(|plan| PlannedCacheCompaction {
512            at: plan.due_at,
513            expires_at: plan.expires_at,
514            next_fire: plan.next_fire.as_ref().map(|(fire, _)| *fire),
515            job_id: plan.next_fire.as_ref().map(|(_, job)| job.clone()),
516            running,
517        });
518        if self.inner.store.record_cache_compaction(session, planned) {
519            self.inner.store.announce_live_waiting(session);
520        }
521    }

523    /// `session`'s plan now. A session paged out is read from its last
524    /// reading; one never read is loaded once.
525    async fn plan_session(
526        &self,
527        session: SessionId,
528        config: &CacheExpiryConfig,
529        busy: &HashSet<SessionId>,
530        now: Timestamp,
531    ) -> Option<Plan> {
532        let state = &self.inner.cache_expiry;
533        let cached = lock(&state.requests).get(&session).cloned();
534        let reading = match (self.inner.store.get_resident(session), cached) {
535            (Some(projection), _) => {
536                let reading = latest_request(&projection.log.event_snapshot())
537                    .map(|request| (request, projection.meta.held.is_some()));
538                lock(&state.requests).insert(session, reading.clone());
539                reading
540            }
541            (None, Some(cached)) => cached,
542            (None, None) => {
543                let store = Arc::clone(&self.inner.store);
544                let projection = super::blocking_io::run(move || store.get(session))
545                    .await
546                    .ok()?
547                    .ok()?;
548                let reading = latest_request(&projection.log.event_snapshot())
549                    .map(|request| (request, projection.meta.held.is_some()));
550                lock(&state.requests).insert(session, reading.clone());
551                reading
552            }
553        };
554        let (request, held) = reading?;
555        let origin = self.inner.store.origin(session).ok()?;
556        let root = root_id(&origin, session);
557        let config_now = self.inner.config.load();
558        let recache_cost_pico_usd = crate::usage::recache_cost_pico_usd(
559            &request.model,
560            request.tokens,
561            &config_now.runtime.pricing,
562            &*self.inner.model_manager.current(),
563        );
564        let session_state = SessionState {
565            held,
566            busy: busy.contains(&root),
567            tried: lock(&state.tried).get(&session).copied(),
568            next_fire: self.inner.crons.next_fire_into(session),
569            recache_cost_pico_usd,
570        };
571        plan(config, &request, &session_state, now)
572    }

574    /// Runs `plan` for `session` (section 4.4): checks the rule again on
575    /// the session as it is now, then compacts it, quietly.
576    async fn compact_before_expiry(&self, session: SessionId, plan: Plan) {
577        let config = self
578            .inner
579            .config
580            .load()
581            .runtime
582            .context_compaction
583            .cache_expiry
584            .clone();
585        lock(&self.inner.cache_expiry.requests).remove(&session);
586        let current = {
587            let _residency = self
588                .lock_residency("checking a compaction before cache expiry of", session)
589                .await;
590            let busy = self.busy_roots();
591            self.plan_session(session, &config, &busy, Timestamp::now())
592                .await
593        };
594        let Some(current) = current.filter(|current| {
595            current.usage_seq == plan.usage_seq && current.due_at <= Timestamp::now()
596        }) else {
597            return;
598        };
599        // Rule 7: a timer that woke after expiry compacts nothing.
600        if Timestamp::now() >= current.expires_at {
601            return;
602        }
603        lock(&self.inner.cache_expiry.tried).insert(session, current.usage_seq);
604        self.announce_cache_compaction(session, Some(&current), true);
605        let reason = CheckpointReason::CacheExpiry {
606            expires_at: current.expires_at,
607            next_fire: current.next_fire.as_ref().map(|(fire, _)| *fire),
608            job_id: current.next_fire.as_ref().map(|(_, job)| job.clone()),
609        };
610        self.inner.store.set_quiet(session, true);
611        let result = self
612            .compact_session_with_reason(
613                session,
614                event_origin(CACHE_EXPIRY_ORIGIN),
615                Some(Open::Known(reason)),
616            )
617            .await;
618        let failure = match result {
619            Ok(result) if result.compacted => None,
620            Ok(result) => result.cancellation_reason,
621            // Someone used the session first: nothing to say.
622            Err(EngineError::SessionRunning(_)) => None,
623            Err(error) => Some(error.to_string()),
624        };
625        if let Some(message) = failure {
626            let _ = self
627                .append(
628                    session,
629                    None,
630                    event_origin(CACHE_EXPIRY_ORIGIN),
631                    Event::Diagnostic {
632                        source: cookie_agent_protocol::DiagnosticSource::engine("cache-expiry")
633                            .expect("static diagnostic source is valid"),
634                        kind: cookie_agent_protocol::EngineDiagnosticKind::CacheExpiryCompaction
635                            .label(),
636                        message: format!(
637                            "compaction before the prompt cache expired failed: {message}"
638                        ),
639                        count: None,
640                        detail: None,
641                    },
642                )
643                .await;
644        }
645        self.inner.store.set_quiet(session, false);
646        lock(&self.inner.cache_expiry.plans).remove(&session);
647        self.announce_cache_compaction(session, None, false);
648    }
649}
