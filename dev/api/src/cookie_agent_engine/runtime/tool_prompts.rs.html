14/// The tools an ACP agent is offered over the MCP bridge, by the provider
15/// that publishes them: which providers' instruction sections its first
16/// prompt carries (`docs/site/specs/acp-agents.md` section 15, 2026-10-23).
17#[derive(Debug, Default)]
18pub(crate) struct OfferedTools {
19    /// `(provider ID, attachment)` of each provider with an offered tool.
20    pub(crate) providers: std::collections::HashSet<(String, Option<String>)>,
21    /// A memory tool is offered, and one that writes.
22    pub(crate) memory: bool,
23    pub(crate) memory_write: bool,
24    /// The `skill` tool is offered.
25    pub(crate) skill: bool,
26}
27
28impl OfferedTools {
29    fn offers(&self, provider: &str, attachment: Option<&str>) -> bool {
30        self.providers
31            .contains(&(provider.to_owned(), attachment.map(str::to_owned)))
32    }
33}

35impl Engine {
36    pub(crate) fn compose_tool_prompt_sections(
37        &self,
38        policy: &mut FrozenRunPolicy,
39        session: SessionId,
40    ) -> Result<(), EngineError> {
41        // A provider whose tools are all deferred gives its sections with
42        // its first load instead (deferred tools in cookie, section 7.3).
43        let deferred = self.deferred_prompt_providers(session, policy)?;
44        let Some(block) = self.tool_prompt_block(policy, session, None, &|id, attachment| {
45            !deferred.contains(&(id.to_owned(), attachment.map(str::to_owned)))
46        })?
47        else {
48            return Ok(());
49        };
50        let composed_bytes = policy
51            .agent
52            .composed_prompt
53            .len()
54            .checked_add(block.len() + PROMPT_BLOCK_SEPARATOR.len())
55            .ok_or_else(|| {
56                EngineError::ToolPrompt("composed prompt byte count overflowed".into())
57            })?;
58        if composed_bytes > AgentSnapshot::MAX_PROMPT_BYTES {
59            return Err(EngineError::ToolPrompt(format!(
60                "composed prompt exceeds {} bytes",
61                AgentSnapshot::MAX_PROMPT_BYTES
62            )));
63        }
64
65        push_prompt_block(&mut policy.agent.composed_prompt, &block);
66        policy.agent.prompt_fingerprint =
67            Sha256Digest::of_bytes(policy.agent.composed_prompt.as_bytes());
68        let mut document_fingerprint = policy
69            .agent
70            .document_fingerprint
71            .as_str()
72            .as_bytes()
73            .to_vec();
74        document_fingerprint.extend_from_slice(block.as_bytes());
75        policy.agent.document_fingerprint = Sha256Digest::of_bytes(&document_fingerprint);
76        Ok(())
77    }
78
79    /// The `<tool_instructions>` sections of `session`'s tool providers
80    /// for a run of `policy`, joined; `None` when there are none. With
81    /// `offered`, only the providers that publish an offered tool give
82    /// sections, and the memory section follows the offered memory tools.
83    pub(crate) fn tool_prompt_block(
84        &self,
85        policy: &FrozenRunPolicy,
86        session: SessionId,
87        offered: Option<&OfferedTools>,
88        keep: &dyn Fn(&str, Option<&str>) -> bool,
89    ) -> Result<Option<String>, EngineError> {
90        let projection = self.inner.store.get(session)?;
91        let depth = session_depth(&projection.meta.origin);
92        let delegate_targets = policy
93            .delegate_targets(depth)
94            .into_iter()
95            .filter_map(|target| {
96                policy
97                    .registry
98                    .descriptors()
99                    .iter()
100                    .find(|descriptor| descriptor.id == target)
101                    .map(|descriptor| (target, descriptor.description.clone()))
102            })
103            .collect();
104        let context = SessionToolContext::for_prompt_composition(session, delegate_targets);
105        let providers = self.session_tool_sources(session)?;
106        // The memory provider decides from the run's frozen policy, with the
107        // visibility test the tool list uses (memory spec section 16.9.1).
108        let mut memory_access = super::memory_runs::memory_access(
109            true,
110            &policy.agent,
111            Some(&projection.permission_overlay),
112        );
113        if let Some(offered) = offered {
114            memory_access.available &= offered.memory;
115            memory_access.writable &= offered.memory_write;
116        }
117        let delegated = matches!(
118            projection.meta.origin,
119            cookie_agent_protocol::SessionOrigin::Delegated { .. }
120        );
121        let mut rendered = Vec::new();
122        let mut total_body_bytes = 0usize;
123
124        for crate::runtimes::SourcedProvider {
125            provider,
126            attachment,
127            ..
128        } in providers
129        {
130            let provider_id = provider.provider_id();
131            validate_provider_id(provider_id)?;
132            if !keep(provider_id, attachment.as_ref().map(|name| name.as_str())) {
133                continue;
134            }
135            if let Some(offered) = offered
136                && provider_id != crate::memory_tools::PROVIDER_ID
137                && !offered.offers(provider_id, attachment.as_ref().map(|name| name.as_str()))
138            {
139                continue;
140            }
141            let sections = if provider_id == crate::memory_tools::PROVIDER_ID {
142                crate::memory_tools::prompt_section(memory_access, delegated)
143                    .map(|body| PromptSection {
144                        title: "Memory".to_owned(),
145                        body,
146                    })
147                    .into_iter()
148                    .collect()
149            } else {
150                provider.prompt_sections(&context).map_err(|error| {
151                    EngineError::ToolPrompt(format!("provider `{provider_id}`: {error}"))
152                })?
153            };
154            let mut provider_body_bytes = 0usize;
155            for section in sections {
156                let section = normalize_section(provider_id, section)?;
157                let body_bytes = section.body.len();
158                if body_bytes > MAX_SECTION_BODY_BYTES {
159                    return Err(section_error(
160                        provider_id,
161                        &section.title,
162                        format!("body exceeds {MAX_SECTION_BODY_BYTES} bytes"),
163                    ));
164                }
165                provider_body_bytes =
166                    provider_body_bytes.checked_add(body_bytes).ok_or_else(|| {
167                        section_error(
168                            provider_id,
169                            &section.title,
170                            "provider byte count overflowed",
171                        )
172                    })?;
173                if provider_body_bytes > MAX_PROVIDER_BODY_BYTES {
174                    return Err(section_error(
175                        provider_id,
176                        &section.title,
177                        format!("provider bodies exceed {MAX_PROVIDER_BODY_BYTES} bytes"),
178                    ));
179                }
180                total_body_bytes = total_body_bytes.checked_add(body_bytes).ok_or_else(|| {
181                    section_error(provider_id, &section.title, "total byte count overflowed")
182                })?;
183                if total_body_bytes > MAX_TOTAL_BODY_BYTES {
184                    return Err(section_error(
185                        provider_id,
186                        &section.title,
187                        format!("all provider bodies exceed {MAX_TOTAL_BODY_BYTES} bytes"),
188                    ));
189                }
190                rendered.push(render_section(provider_id, &section.body));
191            }
192        }
193
194        Ok((!rendered.is_empty()).then(|| rendered.join("\n\n")))
195    }
196}

198fn normalize_section(
199    provider_id: &str,
200    mut section: PromptSection,
201) -> Result<PromptSection, EngineError> {
202    if section.title.trim().is_empty() || section.title.chars().any(char::is_control) {
203        return Err(section_error(
204            provider_id,
205            &section.title,
206            "title must be nonblank and control-free",
207        ));
208    }
209    section.body = section.body.replace("\r\n", "\n").replace('\r', "\n");
210    if section.body.trim().is_empty() {
211        return Err(section_error(
212            provider_id,
213            &section.title,
214            "body must not be blank",
215        ));
216    }
217    if section
218        .body
219        .chars()
220        .any(|character| character.is_control() && character != '\n' && character != '\t')
221    {
222        return Err(section_error(
223            provider_id,
224            &section.title,
225            "body contains a disallowed control character",
226        ));
227    }
228    Ok(section)
229}
230
231fn validate_provider_id(provider_id: &str) -> Result<(), EngineError> {
232    if provider_id.trim().is_empty() || provider_id.chars().any(char::is_control) {
233        return Err(EngineError::ToolPrompt(
234            "provider ID must be nonblank and control-free".into(),
235        ));
236    }
237    Ok(())
238}
239
240fn render_section(provider_id: &str, body: &str) -> String {
241    format!(
242        "<tool_instructions provider=\"{}\">\n{body}\n</tool_instructions>",
243        escape_attribute(provider_id)
244    )
245}
246
247fn escape_attribute(value: &str) -> String {
248    value
249        .replace('&', "&amp;")
250        .replace('<', "&lt;")
251        .replace('>', "&gt;")
252        .replace('"', "&quot;")
253        .replace('\'', "&apos;")
254}
255
256fn section_error(provider_id: &str, title: &str, detail: impl std::fmt::Display) -> EngineError {
257    EngineError::ToolPrompt(format!(
258        "provider `{provider_id}` section `{title}`: {detail}"
259    ))
260}

276    fn section_validation_normalizes_newlines_and_rejects_invalid_content() {
277        let normalized = normalize_section(
278            "test",
279            PromptSection {
280                title: "Valid".into(),
281                body: "one\r\ntwo\rthree\tend".into(),
282            },
283        )
284        .expect("valid section");
285        assert_eq!(normalized.body, "one\ntwo\nthree\tend");
286
287        for section in [

290                body: "body".into(),

292            PromptSection {

294                body: " \n\t ".into(),
295            },
296            PromptSection {
297                title: "Control".into(),
