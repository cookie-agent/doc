1//! The official agents (`docs/site/specs/machines-on-demand.md` section
2//! 2.1): fetched from a versionless source, `<source>/index.json` and
3//! `<source>/<id>.md`, each document checked against the index's SHA-256
4//! and against this build's configuration, and kept in
5//! `<data_root>/official_agents/<id>.md`, which only the daemon writes.
6//! Agents the index no longer lists, or whose `requires_cookie` this version
7//! does not satisfy, are removed; a failed refresh keeps the cached copies
8//! and is reported as a configuration diagnostic. A refresh that changes
9//! anything reloads the configuration, so new runs see it at once; a user
10//! document of the same ID in `agents/` replaces the official one.
11//!
12//! The index also lists templates (`templates`, kept as `_<name>.md`) and
13//! the presets onboarding offers (`acp_presets`, `mcp_presets`, kept in
14//! `.presets.json`), in lists of their own that earlier builds ignore
15//! (`docs/site/specs/acp-onboarding.md` sections 3, 6.1, and 9.1).
16

31/// The official source. Production never changes it; tests serve their own
32/// with [`Engine::set_official_agents_url`].
33pub const OFFICIAL_AGENTS_URL: &str = "https://cookie-agent.github.io/agents";
34/// How often a running daemon refreshes them.
35pub const OFFICIAL_AGENTS_REFRESH: Duration = Duration::from_secs(24 * 60 * 60);
36const MAX_INDEX_BYTES: usize = 1024 * 1024;

38const MAX_AGENT_BYTES: usize = 256 * 1024;
39const MAX_AGENTS: usize = 256;

70/// A preset of the index with the cookie versions it needs.

78/// The presets of `index` this build can use, and why the others were left
79/// out: each must read, need a cookie version this one is, and hold what a
80/// preset must (a command, a server table the configuration takes).
81pub fn usable_presets(
82    acp: &[serde_json::Value],
83    mcp: &[serde_json::Value],
84    version: &str,
85) -> (OfficialPresets, Vec<String>) {
86    let mut presets = OfficialPresets::default();
87    let mut refusals = Vec::new();
88    for value in acp.iter().take(MAX_AGENTS) {
89        let label = preset_label(value);
90        match serde_json::from_value::<Versioned<cookie_agent_protocol::AcpPreset>>(value.clone()) {
91            Err(error) => refusals.push(format!("ACP preset `{label}`: {error}")),
92            Ok(versioned) => match version_satisfies(&versioned.requires_cookie, version) {
93                None => refusals.push(format!(
94                    "ACP preset `{label}`: `{}` is not a version range",
95                    versioned.requires_cookie
96                )),
97                Some(false) => {}
98                Some(true) => match acp_preset_problem(&versioned.preset) {
99                    Some(problem) => refusals.push(format!("ACP preset `{label}`: {problem}")),
100                    None => presets.acp.push(versioned.preset),
101                },
102            },
103        }
104    }
105    for value in mcp.iter().take(MAX_AGENTS) {
106        let label = preset_label(value);
107        match serde_json::from_value::<Versioned<cookie_agent_protocol::McpPreset>>(value.clone()) {
108            Err(error) => refusals.push(format!("MCP preset `{label}`: {error}")),
109            Ok(versioned) => match version_satisfies(&versioned.requires_cookie, version) {
110                None => refusals.push(format!(
111                    "MCP preset `{label}`: `{}` is not a version range",
112                    versioned.requires_cookie
113                )),
114                Some(false) => {}
115                Some(true) => match mcp_preset_problem(&versioned.preset) {
116                    Some(problem) => refusals.push(format!("MCP preset `{label}`: {problem}")),
117                    None => presets.mcp.push(versioned.preset),
118                },
119            },
120        }
121    }
122    (presets, refusals)
123}

125fn preset_label(value: &serde_json::Value) -> String {
126    value
127        .get("id")
128        .and_then(serde_json::Value::as_str)
129        .unwrap_or("?")
130        .to_owned()
131}

133fn acp_preset_problem(preset: &cookie_agent_protocol::AcpPreset) -> Option<String> {
134    if preset.name.trim().is_empty() {
135        return Some("has no name".into());
136    }
137    if preset
138        .command
139        .first()
140        .is_none_or(|program| program.trim().is_empty())
141    {
142        return Some("has no command".into());
143    }
144    None
145}
146
147fn mcp_preset_problem(preset: &cookie_agent_protocol::McpPreset) -> Option<String> {
148    if preset.name.trim().is_empty() {
149        return Some("has no name".into());
150    }
151    if preset.place == cookie_agent_protocol::McpPresetPlace::Unknown {
152        return Some("`where` is neither `global` nor `machine`".into());
153    }
154    mcp_server_problem(
155        &preset.id,
156        &preset.server,
157        preset.place == cookie_agent_protocol::McpPresetPlace::Global,
158    )
159}

161/// Why `server` is not a table the configuration takes for an MCP server
162/// `name`, at the top level (`global`) or a runtime's.
163pub(crate) fn mcp_server_problem(
164    name: &str,
165    server: &serde_json::Map<String, serde_json::Value>,
166    global: bool,
167) -> Option<String> {
168    let value = match toml::Value::try_from(serde_json::Value::Object(server.clone())) {
169        Ok(value) => value,
170        Err(error) => return Some(format!("its server is not a TOML table: {error}")),
171    };
172    let config = match cookie_agent_config::McpServerConfig::deserialize(value) {
173        Ok(config) => config,
174        Err(error) => return Some(format!("its server does not read: {error}")),
175    };
176    let checked = if global {
177        config.validate_top_level(name)
178    } else {
179        config.validate(name)
180    };
181    checked.err().map(|error| error.to_string())
182}

248impl Engine {
249    /// Fetches the official agents from `url` instead of
250    /// [`OFFICIAL_AGENTS_URL`], for tests that serve their own.
251    pub fn set_official_agents_url(&self, url: impl Into<String>) {
252        *lock(&self.inner.official_agents.source_url) = url.into();
253    }
254
255    /// The official agents kept, by ID, with their versions.
256    #[must_use]
257    pub fn official_agents(&self) -> Arc<BTreeMap<String, String>> {
258        Arc::clone(&lock(&self.inner.official_agents.versions))
259    }
260
261    /// The presets onboarding offers: the last refresh's, else the kept
262    /// ones.
263    #[must_use]
264    pub fn official_presets(&self) -> Arc<OfficialPresets> {
265        let state = &self.inner.official_agents;
266        if let Some(presets) = lock(&state.presets).as_ref() {
267            return Arc::clone(presets);
268        }
269        let root = self.configuration_root().ok().map_or_else(
270            || state.root.clone(),
271            |root| cookie_agent_config::official_agents_root(&root),
272        );
273        let presets = Arc::new(read_presets(&root));
274        *lock(&state.presets) = Some(Arc::clone(&presets));
275        presets
276    }
277
278    /// Why the last refresh failed, and why kept official agents were left
279    /// out of the configuration, as configuration diagnostics.
280    pub(crate) fn official_agents_diagnostics(&self) -> Vec<String> {
281        let mut diagnostics = lock(&self.inner.official_agents.last_error)
282            .as_ref()
283            .map(|error| format!("official agents were not refreshed: {error}"))
284            .into_iter()
285            .collect::<Vec<_>>();
286        diagnostics.extend(
287            self.inner
288                .config
289                .load()
290                .official_agent_diagnostics
291                .iter()
292                .cloned(),
293        );
294        diagnostics
295    }
296
297    /// Refreshes the official agents now and, when anything changed,
298    /// reloads the configuration; refreshes do not overlap.
299    pub async fn refresh_official_agents(&self) -> OfficialAgentsReport {
300        let state = &self.inner.official_agents;
301        let _refresh = state.refresh.lock().await;
302        if !self.inner.config.load().runtime.agents.official {
303            *lock(&state.last_error) = None;
304            return OfficialAgentsReport {
305                enabled: false,
306                ..OfficialAgentsReport::default()
307            };
308        }
309        let url = lock(&state.source_url).clone();
310        let config_root = self.configuration_root().ok();
311        let root = config_root.as_deref().map_or_else(
312            || state.root.clone(),
313            cookie_agent_config::official_agents_root,
314        );
315        let mut report = refresh(&url, &root, config_root).await;
316        report.enabled = true;
317        *lock(&state.versions) = Arc::new(report.agents.clone());
318        let presets_changed = lock(&state.presets)
319            .as_deref()
320            .is_none_or(|presets| *presets != report.presets);
321        *lock(&state.presets) = Some(Arc::new(report.presets.clone()));
322        if (!report.updated.is_empty() || !report.removed.is_empty() || presets_changed)
323            && let Err(error) = self.reload_configuration().await
324        {
325            let reload = format!("the configuration did not reload: {error}");
326            report.error = Some(match report.error.take() {
327                Some(error) => format!("{error}; {reload}"),
328                None => reload,
329            });
330        }
331        *lock(&state.last_error) = report.error.clone();
332        report
333    }
334}

350/// One refresh against the source at `url` into `root`. `config_root` is
351/// the user configuration each downloaded document is validated with.
352async fn refresh(url: &str, root: &Path, config_root: Option<PathBuf>) -> OfficialAgentsReport {
353    let mut report = OfficialAgentsReport::default();
354    let kept = {
355        let root = root.to_owned();
356        local(move || read_kept(&root)).await.unwrap_or_default()
357    };
358    let cached_presets = {
359        let root = root.to_owned();
360        local(move || read_presets(&root)).await.unwrap_or_default()
361    };
362    let fail = |mut report: OfficialAgentsReport, error: String, kept: Vec<IndexAgent>| {
363        report.error = Some(error);
364        report.agents = versions(kept);
365        report.presets = cached_presets.clone();
366        report
367    };
368    let client = match source_client() {
369        Ok(client) => client,
370        Err(error) => return fail(report, error.to_string(), kept),
371    };
372    let base = url.trim_end_matches('/');
373    let index = match fetch(&client, &format!("{base}/index.json"), MAX_INDEX_BYTES)
374        .await
375        .and_then(|bytes| {
376            serde_json::from_slice::<SourceIndex>(&bytes)
377                .map_err(|error| format!("index.json does not read: {error}"))
378        })
379        .and_then(|index| {
380            (index.version == 1)
381                .then_some(index)
382                .ok_or_else(|| "index.json is of a version this build does not read".to_owned())
383        }) {
384        Ok(index) => index,
385        Err(error) => return fail(report, error, kept),
386    };
387    let mut refusals = Vec::new();
388    let (presets, preset_refusals) = usable_presets(
389        &index.acp_presets,
390        &index.mcp_presets,
391        env!("CARGO_PKG_VERSION"),
392    );
393    refusals.extend(preset_refusals);
394    {
395        let root = root.to_owned();
396        let bytes = serde_json::to_vec_pretty(&presets).unwrap_or_default();
397        if let Ok(Err(error)) =
398            local(move || write_atomically(&root.join(PRESETS_RECORD), &bytes)).await
399        {
400            refusals.push(format!("the presets were not kept: {error}"));
401        }
402    }
403    report.presets = presets;
404    // A template is kept as `_<name>.md`, under the ID `_<name>`.
405    let templates = index.templates.into_iter().map(|template| IndexAgent {
406        id: format!("_{}", template.id),
407        ..template
408    });
409    let mut wanted = BTreeMap::new();
410    for agent in index
411        .agents
412        .into_iter()
413        .take(MAX_AGENTS)
414        .chain(templates.take(MAX_AGENTS))
415    {
416        if let Err(reason) = valid_entry(&agent) {
417            refusals.push(format!("`{}`: {reason}", agent.id));
418            continue;
419        }
420        match version_satisfies(&agent.requires_cookie, env!("CARGO_PKG_VERSION")) {
421            Some(true) => {
422                wanted.insert(agent.id.clone(), agent);
423            }
424            Some(false) => {}
425            None => refusals.push(format!(
426                "`{}`: `{}` is not a version range",
427                agent.id, agent.requires_cookie
428            )),
429        }
430    }
431    let kept_by_id = kept
432        .into_iter()
433        .map(|agent| (agent.id.clone(), agent))
434        .collect::<BTreeMap<_, _>>();
435    // Download what changed, checked against the index, before anything is
436    // written.
437    let mut downloads = Vec::new();
438    let mut now_kept = BTreeMap::new();
439    for (id, agent) in wanted {
440        let path = root.join(format!("{id}.md"));
441        let current = {
442            let path = path.clone();
443            local(move || std::fs::read(path).ok()).await.ok().flatten()
444        };
445        if current
446            .as_deref()
447            .is_some_and(|bytes| sha256(bytes).eq_ignore_ascii_case(&agent.sha256))
448        {
449            if !kept_by_id.contains_key(&id) {
450                report.updated.push(id.clone());
451            }
452            now_kept.insert(id, agent);
453            continue;
454        }
455        match fetch(&client, &format!("{base}/{id}.md"), MAX_AGENT_BYTES).await {
456            Ok(bytes) if sha256(&bytes).eq_ignore_ascii_case(&agent.sha256) => {
457                downloads.push((id.clone(), bytes));
458                now_kept.insert(id, agent);
459            }
460            Ok(bytes) => {
461                refusals.push(format!(
462                    "`{id}`: does not match the index: its SHA-256 is {}",
463                    sha256(&bytes)
464                ));
465                if let Some(previous) = kept_by_id.get(&id) {
466                    now_kept.insert(id, previous.clone());
467                }
468            }
469            Err(error) => {
470                refusals.push(format!("`{id}`: {error}"));
471                if let Some(previous) = kept_by_id.get(&id) {
472                    now_kept.insert(id, previous.clone());
473                }
474            }
475        }
476    }
477    let previously = kept_by_id.keys().cloned().collect::<BTreeSet<_>>();
478    let root = root.to_owned();
479    let installed = local(move || {
480        install(
481            &root,
482            config_root.as_deref(),
483            downloads,
484            &mut now_kept,
485            &previously,
486        )
487        .map(|outcome| (outcome, now_kept))
488    })
489    .await;
490    match installed {
491        Ok(Ok(((updated, removed, install_refusals), now_kept))) => {
492            report.updated.extend(updated);
493            report.removed = removed;
494            refusals.extend(install_refusals);
495            report.agents = versions(now_kept.into_values());
496        }
497        Ok(Err(error)) | Err(error) => {
498            refusals.push(error);
499            report.agents = versions(kept_by_id.into_values());
500        }
501    }
502    report.updated.sort();
503    report.updated.dedup();
504    if !refusals.is_empty() {
505        report.error = Some(refusals.join("; "));
506    }
507    report
508}

510/// The updated agents, the removed ones, and the documents refused.
511type InstallOutcome = (Vec<String>, Vec<String>, Vec<String>);
512
513/// Checks each download loads with this build and the user's configuration
514/// at `config_root`, writes the ones that do, removes the agents `now_kept`
515/// no longer lists, and records `now_kept`. A refused download keeps the
516/// previous copy, if any.
517fn install(
518    root: &Path,
519    config_root: Option<&Path>,
520    downloads: Vec<(String, Vec<u8>)>,
521    now_kept: &mut BTreeMap<String, IndexAgent>,
522    previously: &BTreeSet<String>,
523) -> Result<InstallOutcome, String> {
524    let mut updated = Vec::new();
525    let mut refusals = Vec::new();
526    let base_tree = config_root
527        .map(|config_root| cookie_agent_config::ConfigTree::read(Some(config_root)))
528        .transpose()
529        .map_err(|error| format!("the configuration does not read: {error}"))?;
530    for (id, bytes) in downloads {
531        if let Err(reason) = loads(base_tree.as_ref(), &id, &bytes) {
532            refusals.push(format!("`{id}` does not load: {reason}"));
533            now_kept.remove(&id);
534            continue;
535        }
536        write_atomically(&root.join(format!("{id}.md")), &bytes)
537            .map_err(|error| format!("`{id}` was not written: {error}"))?;
538        updated.push(id);
539    }
540    // A refused agent that was kept before stays as it was on disk.
541    let kept_before = read_kept(root)
542        .into_iter()
543        .map(|agent| (agent.id.clone(), agent))
544        .collect::<BTreeMap<_, _>>();
545    for refused in refusals.iter().filter_map(|refusal| {
546        refusal
547            .strip_prefix('`')
548            .and_then(|rest| rest.split('`').next())
549    }) {
550        if let Some(previous) = kept_before.get(refused) {
551            now_kept.insert(refused.to_owned(), previous.clone());
552        }
553    }
554    let mut removed = Vec::new();
555    if let Ok(entries) = std::fs::read_dir(root) {
556        for entry in entries.flatten() {
557            let name = entry.file_name().to_string_lossy().into_owned();
558            if name.starts_with('.') {
559                // The record, and temporary files an interrupted write left.
560                if name.ends_with(".tmp") {
561                    let _ = std::fs::remove_file(entry.path());
562                }
563                continue;
564            }
565            let id = name.strip_suffix(".md").unwrap_or(&name);
566            if now_kept.contains_key(id) {
567                continue;
568            }
569            if std::fs::remove_file(entry.path()).is_ok() && previously.contains(id) {
570                removed.push(id.to_owned());
571            }
572        }
573    }
574    let record = now_kept.values().cloned().collect::<Vec<_>>();
575    write_atomically(
576        &root.join(cookie_agent_config::OFFICIAL_AGENTS_RECORD),
577        &serde_json::to_vec_pretty(&record).unwrap_or_default(),
578    )
579    .map_err(|error| format!("the record of kept agents was not written: {error}"))?;
580    Ok((updated, removed, refusals))
581}

583/// Whether the official document `id` loads with this build, beside the
584/// user's configuration `tree` when there is one.
585fn loads(
586    tree: Option<&cookie_agent_config::ConfigTree>,
587    id: &str,
588    bytes: &[u8],
589) -> Result<(), String> {
590    let put = |tree: &mut cookie_agent_config::ConfigTree| -> Result<(), String> {
591        match id.strip_prefix('_') {
592            Some(name) => {
593                cookie_agent_identity::AgentId::new(name).map_err(|error| error.to_string())?;
594                tree.put_official_template(name, bytes.to_vec());
595            }
596            None => {
597                let id =
598                    cookie_agent_identity::AgentId::new(id).map_err(|error| error.to_string())?;
599                tree.put_official_agent(id, bytes.to_vec());
600            }
601        }
602        Ok(())
603    };
604    let mut tree = tree.cloned().unwrap_or_default();
605    tree.clear_official_agents();
606    put(&mut tree)?;
607    match cookie_agent_config::load_from_tree(&tree) {
608        // The user's own configuration may not load; that is not the
609        // official document's fault, so check it alone.
610        Err(_) => {
611            let mut alone = cookie_agent_config::ConfigTree::default();
612            put(&mut alone)?;
613            first_diagnostic(cookie_agent_config::load_from_tree(&alone))
614        }
615        loaded => first_diagnostic(loaded),
616    }
617}
618
619fn first_diagnostic(
620    loaded: Result<cookie_agent_config::LoadedConfiguration, cookie_agent_config::ConfigError>,
621) -> Result<(), String> {
622    let loaded = loaded.map_err(|error| error.to_string())?;
623    match loaded.official_agent_diagnostics.into_iter().next() {
624        Some(diagnostic) => Err(diagnostic),
625        None => Ok(()),
626    }
627}
628
629fn valid_entry(agent: &IndexAgent) -> Result<(), String> {
630    // A template's entry is `_<name>`, its file `_<name>.md`.
631    cookie_agent_identity::AgentId::new(agent.id.strip_prefix('_').unwrap_or(&agent.id))
632        .map_err(|_| "not an agent ID".to_owned())?;
633    if agent.sha256.len() != 64 || !agent.sha256.chars().all(|c| c.is_ascii_hexdigit()) {
634        return Err("has no SHA-256".into());
635    }
636    if agent.version.is_empty() || agent.version.len() > 64 {
637        return Err("has no version".into());
638    }
639    Ok(())
640}
