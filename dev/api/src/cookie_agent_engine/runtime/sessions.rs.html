45    pub fn create_session(&self, selection: RunSelection) -> Result<SessionMeta, EngineError> {
46        self.reconcile_provider_store()?;
47        let runtime = self.current_runtime();
48        if runtime.result.snapshot.models.is_empty() {
49            return Err(EngineError::NoRunnableModel);
50        }
51        let agents = runtime.agents_for_preset(selection.preset.as_deref())?;
52        let agent = resolve_agent(&agents, &selection.agent)?.clone();
53        let policy = freeze_root_agent_policy(
54            &agent,
55            agents,
56            Arc::clone(&runtime),
57            &selection.model,
58            self.inner.config.runtime.delegation.max_depth,
59            policy::ResultLimits {
60                tool_output_max_lines: self.inner.config.runtime.tool_output.max_lines,
61                tool_output_max_bytes: self.inner.config.runtime.tool_output.max_bytes,
62            },
63            self.inner.config.runtime.model_retry,
64        )?;
65        let id = SessionId::new_v7();
66        let cwd_identity = cwd_identity(self.inner.store.cwd())?;
67        // A new root begins an empty tree, so its first handle is trivially
68        // unique; it still goes through the same generation path so every
69        // session carries a handle uniformly.
70        let short_id = super::handles::generate_handle(
71            selection.agent.as_str(),
72            &HashSet::new(),
73            super::handles::random_hex,
74        )?;
75        let creation = Event::SessionCreated {
76            origin: SessionOrigin::Root,
77            short_id: Some(short_id),
78            cwd_identity: cwd_identity.clone(),
79            creation_selection: selection.clone(),
80            creation_agent: Box::new(policy.agent.clone()),
81            runtime_revision: runtime.result.snapshot.runtime_revision.clone(),
82            catalog_revision: runtime.result.snapshot.catalog_revision.clone(),
83            provider_state_revision: runtime.result.snapshot.provider_state_revision.clone(),
84            model_revision: runtime.result.snapshot.model_revision.clone(),
85            agent_revision: runtime.result.snapshot.agent_revision.clone(),
86            recipe_registry_revision: runtime.result.snapshot.recipe_registry_revision.clone(),
87            manifest_revision: runtime.current_manifest.revision.clone(),
88        };
89        self.inner.store.create(
90            id,
91            EventOrigin::new("user").expect("static event origin is valid"),
92            creation,
93        )?;
94        self.spawn_actor(id)?;
95        Ok(self.inner.store.get(id)?.metadata())
96    }

167    pub fn session_tree_usage(&self, id: SessionId) -> Result<SessionTreeUsageResult, EngineError> {
168        // A cold root's index is only a cache. Complete the tree before using
169        // its summaries, otherwise a missing or stale index can produce a
170        // plausible but incomplete rollup.
171        self.inner.store.ensure_tree_for(id)?;
172        let summaries = self.inner.store.tree_summaries(id)?;
173        let mut children: HashMap<SessionId, Vec<(SessionId, InvocationId)>> = HashMap::new();
174        let mut rollups = HashMap::new();
175        for summary in summaries {
176            let session_id = summary.meta.session_id;
177            if let SessionOrigin::Delegated {
178                root_session_id,
179                parent_session_id,
180                invocation_id,
181                ..
182            } = summary.meta.origin
183                && root_session_id == id
184            {
185                let edge = (session_id, invocation_id);
186                children.entry(parent_session_id).or_default().push(edge);
187            }
188            rollups.insert(session_id, summary.usage_rollup);
189        }
190
191        let mut usage = UsageRollup::default();
192        let session_count = merge_session_tree_usage(id, &children, &rollups, &mut usage)?;
193        Ok(SessionTreeUsageResult {
194            session_id: id,
195            usage: crate::usage::with_pricing(
196                usage,
197                &self.inner.config.runtime.pricing,
198                &self.catalog_pricing(),
199            ),
200            session_count,
201        })
202    }

238    pub fn children(
239        &self,
240        id: SessionId,
241    ) -> Result<Vec<cookie_agent_protocol::ChildSummary>, EngineError> {
242        // Listing is a use of the tree: complete it first so nested delegations
243        // and their summaries are in view (§3.2.2). An incomplete tree is
244        // reported, never papered over with the pre-load cache (review L14).
245        self.ensure_tree_loaded(id)?;
246        let known: HashSet<_> = self
247            .inner
248            .delegation_events
249            .entries()
250            .into_iter()
251            .map(|entry| {
252                (
253                    entry.reservation.invocation_id,
254                    entry.reservation.parent_session_id,
255                    entry.reservation.child_session_id,
256                )
257            })
258            .collect();
259        Ok(self
260            .inner
261            .store
262            .tree_summaries(id)?
263            .into_iter()
264            .filter_map(|child| match child.meta.origin {
265                SessionOrigin::Delegated {
266                    parent_session_id,
267                    invocation_id,
268                    ..
269                } if parent_session_id == id
270                    && known.contains(&(
271                        invocation_id,
272                        parent_session_id,
273                        child.meta.session_id,
274                    )) =>
275                {
276                    Some(ChildSummary {
277                        session_id: child.meta.session_id,
278                        agent: child.meta.creation_selection.agent.clone(),
279                        title: child.meta.title.clone(),
280                        title_updated_seq: child.meta.title_updated_seq,
281                        status: child.meta.status,
282                        usage: child.usage,
283                    })
284                }
285                _ => None,
286            })
287            .collect())
288    }

310    /// Adopts `id` and resolves the work its adoption found interrupted.
311    ///
312    /// Adoption schedules delegation recovery instead of running it inline, so
313    /// the wait below is what makes the ordering observable: when this returns,
314    /// every delegation record under the session's root that the adoption
315    /// classified as running without a live run here has been resolved, and the
316    /// root's background-delegation capacity reads settled rather than
317    /// transiently over-counted.
318    ///
319    /// The wait happens here, after the actor has replied, and never inside the
320    /// `Resume` handler: resolving a recovered delegate goes through its parent
321    /// session's mailbox, which is the very mailbox that handler occupies.
322    pub async fn resume(&self, id: SessionId) -> Result<SessionMeta, EngineError> {
323        let meta = self
324            .request(id, |reply| SessionCommand::Resume { reply })
325            .await?;
326        let root = self.inner.store.root_of(id).unwrap_or(id);
327        if !self
328            .await_delegation_recovery(root, DELEGATION_RECOVERY_SETTLE_BOUND)
329            .await
330        {
331            eprintln!(
332                "session {id} delegation recovery still settling after {DELEGATION_RECOVERY_SETTLE_BOUND:?}"
333            );
334        }
335        Ok(meta)
336    }
337    /// Delegated sessions a revert to `cut` deletes: children whose
338    /// reservation the cut removes (and no kept reservation resumes), with all
339    /// their descendants. A running one refuses the revert.
340    fn reverted_children(
341        &self,
342        events: &[Arc<cookie_agent_protocol::StoredEvent>],
343        cut: u64,
344    ) -> Result<HashSet<SessionId>, EngineError> {
345        let mut kept = HashSet::new();
346        let mut removed = Vec::new();
347        for event in events.iter() {
348            if let Event::DelegationReserved { reservation, .. } = &event.payload {
349                if event.seq <= cut {
350                    kept.insert(reservation.child_session_id);
351                } else {
352                    removed.push(reservation.child_session_id);
353                }
354            }
355        }
356        let mut deleted = HashSet::new();
357        let mut pending = removed
358            .into_iter()
359            .filter(|child| !kept.contains(child))
360            .collect::<Vec<_>>();
361        while let Some(child) = pending.pop() {
362            if !deleted.insert(child) {
363                continue;
364            }
365            match self.inner.store.get(child) {
366                Ok(projection) if projection.status == SessionStatus::Running => {
367                    return Err(EngineError::SessionRunning(child));
368                }
369                Ok(_) => {}
370                // Never published, or already gone: nothing on disk to keep.
371                Err(SessionError::Missing(_)) => continue,
372                Err(error) => return Err(error.into()),
373            }
374            pending.extend(
375                self.inner
376                    .store
377                    .children(child)?
378                    .into_iter()
379                    .map(|summary| summary.session_id),
380            );
381        }
382        Ok(deleted)
383    }

385    /// Forgets the in-memory state of sessions a revert deletes.
386    fn forget_deleted_sessions(&self, deleted: &HashSet<SessionId>) {
387        if deleted.is_empty() {
388            return;
389        }
390        fn forget<V>(map: &std::sync::Mutex<HashMap<SessionId, V>>, deleted: &HashSet<SessionId>) {
391            map.lock()
392                .unwrap_or_else(|poisoned| poisoned.into_inner())
393                .retain(|session, _| !deleted.contains(session));
394        }
395        forget(&self.inner.sessions.actors, deleted);
396        forget(&self.inner.sessions.producers, deleted);
397        forget(&self.inner.skills_runtime.grants, deleted);
398        forget(&self.inner.skills_runtime.models, deleted);
399        forget(&self.inner.skills_runtime.pending_child, deleted);
400        forget(&self.inner.compaction.deferred, deleted);
401        forget(&self.inner.compaction.context_token_estimators, deleted);
402        forget(&self.inner.delegation.recovery_pending, deleted);
403        self.inner
404            .delegation
405            .by_session
406            .lock()
407            .unwrap_or_else(|poisoned| poisoned.into_inner())
408            .retain(|child, record| {
409                !deleted.contains(child) && !deleted.contains(&record.parent_session_id)
410            });
411        self.inner
412            .delegation
413            .queue
414            .lock()
415            .unwrap_or_else(|poisoned| poisoned.into_inner())
416            .retain(|child| !deleted.contains(child));
417        self.inner.delegation_events.forget_sessions(deleted);
418    }

420    pub async fn revert_session(
421        &self,
422        session_id: SessionId,
423        through_seq: u64,
424        _origin: EventOrigin,
425    ) -> Result<SessionRevertResult, EngineError> {
426        let context_id = crate::plugin::plugin_context_id();
427        let mut instructions = self
428            .inner
429            .store
430            .get(session_id)?
431            .log
432            .event_snapshot()
433            .iter()
434            .find_map(|event| match &event.payload {
435                cookie_agent_protocol::EventPayload::UserInputSubmitted { input }
436                    if event.seq > through_seq =>
437                {
438                    Some(input.clone())
439                }
440                _ => None,
441            });
442        let mut instructions_override = None;
443        for plugin in self.inner.plugins.interception_plugins(
444            cookie_agent_protocol::ExtensionInterceptionHook::SessionBeforeRevert,
445        ) {
446            let result = self
447                .inner
448                .plugins
449                .intercept_named::<_, cookie_agent_protocol::ExtensionSessionBeforeRevertResult>(
450                    &plugin,
451                    cookie_agent_protocol::PLUGIN_INTERCEPT_SESSION_BEFORE_REVERT_METHOD,
452                    &cookie_agent_protocol::ExtensionSessionBeforeRevertParams {
453                        session_id,
454                        context_id: context_id.clone(),
455                        through_seq,
456                        instructions: instructions.clone(),
457                    },
458                    Some(session_id),
459                    Some(&context_id),
460                )
461                .await;
462            match result {
463                Ok(result)
464                    if result.action
465                        == cookie_agent_protocol::ExtensionSessionBeforeRevertAction::Block =>
466                {
467                    let reason = result
468                        .reason
469                        .unwrap_or_else(|| format!("session revert blocked by {plugin}"));
470                    self.record_plugin_diagnostic(
471                        session_id,
472                        plugin,
473                        cookie_agent_protocol::PluginDiagnosticKind::HookBlocked,
474                        reason.clone(),
475                    );
476                    return Err(EngineError::SessionOperationBlocked(reason));
477                }
478                Ok(result)
479                    if result.action
480                        == cookie_agent_protocol::ExtensionSessionBeforeRevertAction::Override =>
481                {
482                    if let Some(replacement) = result
483                        .instructions_override
484                        .filter(|value| !value.trim().is_empty())
485                    {
486                        instructions = Some(replacement.clone());
487                        instructions_override = Some(replacement);
488                    } else {
489                        self.record_plugin_diagnostic(
490                            session_id,
491                            plugin,
492                            cookie_agent_protocol::PluginDiagnosticKind::InvalidModification,
493                            "empty revert instruction override".into(),
494                        );
495                    }
496                }
497                Ok(_) => {}
498                Err(error) => self.record_interception_error(session_id, plugin, error),
499            }
500        }
501        let session = session_id;
502        self.on_actor_async(session, move |engine| async move {
503            if !engine.reserve_compaction(session) {
504                return Err(EngineError::SessionRunning(session));
505            }
506            let result = (|| {
507                let projection = engine.inner.store.get(session)?;
508                if projection.status == SessionStatus::Running {
509                    return Err(EngineError::SessionRunning(session));
510                }
511                let tip = projection.log.last_event().map_or(0, |event| event.seq);
512                if through_seq == 0 || through_seq > tip {
513                    return Err(SessionError::InvalidSequence {
514                        session_id: session,
515                        through_seq,
516                    }
517                    .into());
518                }
519                // A revert removes the events after the cut: the log keeps no
520                // marker, and later appends reuse the freed sequences.
521                let events = projection.log.all_events();
522                let cut = revert_cut(&events, through_seq);
523                let deleted = engine.reverted_children(&events, cut)?;
524                engine.inner.store.truncate(session, cut)?;
525                // The cut can fall inside a run, with its tool calls, internal
526                // agents or approvals still open: close them as a restart would.
527                engine.reconcile_session_with(session, "session reverted")?;
528                engine
529                    .inner
530                    .compaction
531                    .context_token_estimators
532                    .lock()
533                    .unwrap_or_else(|poisoned| poisoned.into_inner())
534                    .remove(&session);
535                engine.rebuild_visible_tree_grants();
536                engine.inner.delegation_events.reconcile_parent(session)?;
537                engine.reconcile_reverted_producers_direct(session)?;
538                engine.forget_deleted_sessions(&deleted);
539                engine
540                    .inner
541                    .store
542                    .delete_children(&deleted.iter().copied().collect::<Vec<_>>())?;
543                Ok(SessionRevertResult {
544                    session: engine.inner.store.get(session)?.metadata(),
545                    instructions_override,
546                })
547            })();
548            engine.release_compaction_direct(session).await;
549            result
550        })
551        .await
552    }
553    pub async fn fork_session(
554        &self,
555        session_id: SessionId,
556        through_seq: u64,
557        origin: EventOrigin,
558    ) -> Result<SessionForkResult, EngineError> {
559        let context_id = crate::plugin::plugin_context_id();
560        for plugin in self.inner.plugins.interception_plugins(
561            cookie_agent_protocol::ExtensionInterceptionHook::SessionBeforeFork,
562        ) {
563            let result = self
564                .inner
565                .plugins
566                .intercept_named::<_, cookie_agent_protocol::ExtensionSessionBeforeForkResult>(
567                    &plugin,
568                    cookie_agent_protocol::PLUGIN_INTERCEPT_SESSION_BEFORE_FORK_METHOD,
569                    &cookie_agent_protocol::ExtensionSessionBeforeForkParams {
570                        session_id,
571                        context_id: context_id.clone(),
572                        through_seq,
573                    },
574                    Some(session_id),
575                    Some(&context_id),
576                )
577                .await;
578            match result {
579                Ok(result)
580                    if result.action == cookie_agent_protocol::ExtensionAllowBlockAction::Block =>
581                {
582                    let reason = result
583                        .reason
584                        .unwrap_or_else(|| format!("session fork blocked by {plugin}"));
585                    self.record_plugin_diagnostic(
586                        session_id,
587                        plugin,
588                        cookie_agent_protocol::PluginDiagnosticKind::HookBlocked,
589                        reason.clone(),
590                    );
591                    return Err(EngineError::SessionOperationBlocked(reason));
592                }
593                Ok(_) => {}
594                Err(error) => self.record_interception_error(session_id, plugin, error),
595            }
596        }
597        let forked = self.inner.store.fork(session_id, through_seq, origin)?;
598        // A fork into a new tree takes the artifacts its history references
599        // along, so it never resolves anything in the tree it came from
600        // (tree-local D2).
601        let mut referenced = std::collections::HashSet::new();
602        crate::runtime::artifacts::collect_artifact_references_in_events(
603            &self.inner.store.log(forked)?.event_snapshot(),
604            &mut referenced,
605        )?;
606        self.inner
607            .artifacts
608            .copy_into_tree(session_id, forked, referenced)?;
609        let producer_state = self.goal_producer_projection(forked)?;
610        if producer_state.goal.is_some()
611            || producer_state
612                .messages
613                .iter()
614                .any(|message| !message.consumed && !message.discarded)
615        {
616            self.reconcile_producers(forked).await?;
617        }
618        Ok(SessionForkResult { session_id: forked })
619    }

652    pub async fn rename_session(
653        &self,
654        params: SessionRenameParams,
655        _origin: EventOrigin,
656    ) -> Result<SessionRenameResult, EngineError> {
657        let session_id = params.session_id;
658        let reset = matches!(params.change, SessionRenameChange::Reset);
659        // A rename appends only `SessionTitleCommitted`, which producer
660        // reconciliation never reads.
661        let mut result = self
662            .on_actor_unreconciled(session_id, move |engine| {
663                let projection = engine.inner.store.get(session_id)?;
664                if let Some(record) = projection.rename_records.get(&params.client_rename_id) {
665                    if record.conflicts_with(&params) {
666                        return Err(EngineError::RenameConflict);
667                    }
668                    return Ok(SessionRenameResult {
669                        client_rename_id: params.client_rename_id,
670                        session: projection.metadata(),
671                    });
672                }
673                let commit = match params.change {
674                    SessionRenameChange::Set { title } => SessionTitleChange::UserSet {
675                        title,
676                        client_rename_id: params.client_rename_id.clone(),
677                    },
678                    SessionRenameChange::Clear => SessionTitleChange::UserClear {
679                        client_rename_id: params.client_rename_id.clone(),
680                    },
681                    SessionRenameChange::Reset => SessionTitleChange::UserReset {
682                        client_rename_id: params.client_rename_id.clone(),
683                    },
684                };
685                let input_through_seq = projection.log.last_event().map_or(0, |event| event.seq);
686                engine.append_direct(
687                    session_id,
688                    None,
689                    super::event_origin("user"),
690                    Event::SessionTitleCommitted {
691                        input_through_seq,
692                        change: commit,
693                    },
694                )?;
695                Ok(SessionRenameResult {
696                    client_rename_id: params.client_rename_id,
697                    session: engine.inner.store.get(session_id)?.metadata(),
698                })
699            })
700            .await?;
701        if reset {
702            self.generate_title_after_reset(session_id).await?;
703            result.session = self.inner.store.get(session_id)?.metadata();
704        }
705        Ok(result)
706    }
707}

709/// Where a revert to `through_seq` cuts the log. A run the cut would leave
710/// holding nothing but its start goes too: reverting to just before a message
711/// targets the event before it, which is usually that run's `RunStarted`.
712fn revert_cut(events: &[Arc<cookie_agent_protocol::StoredEvent>], through_seq: u64) -> u64 {
713    let mut cut = through_seq;
714    loop {
715        let mut open = None;
716        let mut started = false;
717        for event in events.iter().take_while(|event| event.seq <= cut) {
718            match &event.payload {
719                Event::RunStarted { .. } => {
720                    open = event.run_id.map(|run| (run, event.seq));
721                    started = false;
722                }
723                Event::RunCompleted { .. }
724                | Event::RunFailed { .. }
725                | Event::RunCancelled { .. }
726                | Event::RunInterrupted { .. }
727                    if open.is_some_and(|(run, _)| event.run_id == Some(run)) =>
728                {
729                    open = None;
730                }
731                Event::UserInputSubmitted { .. }
732                | Event::UserInputApplied { .. }
733                | Event::ModelAttemptStarted { .. }
734                | Event::ToolCallStarted { .. }
735                    if open.is_some_and(|(run, _)| event.run_id == Some(run)) =>
736                {
737                    started = true;
738                }
739                _ => {}
740            }
741        }
742        match open {
743            Some((_, start)) if !started && start > 1 => cut = start - 1,
744            _ => return cut,
745        }
746    }
747}

749#[cfg(test)]
750fn validated_tree_usage_edge(
751    origin: &SessionOrigin,
752    child_session_id: SessionId,
753    known: &HashSet<(InvocationId, SessionId, SessionId)>,
754) -> Option<(SessionId, (SessionId, InvocationId))> {
755    match origin {
756        SessionOrigin::Delegated {
757            parent_session_id,
758            invocation_id,
759            ..
760        } if known.contains(&(*invocation_id, *parent_session_id, child_session_id)) => {
761            Some((*parent_session_id, (child_session_id, *invocation_id)))
762        }
763        _ => None,
764    }
765}
766
767fn merge_session_tree_usage(
768    session_id: SessionId,
769    children: &HashMap<SessionId, Vec<(SessionId, InvocationId)>>,
770    rollups: &HashMap<SessionId, UsageRollup>,
771    usage: &mut UsageRollup,
772) -> Result<u64, EngineError> {
773    let mut visited = HashSet::new();
774    let mut pending = vec![(session_id, None)];
775    let mut session_count = 0_u64;
776    while let Some((current_id, incoming_invocation)) = pending.pop() {
777        if !visited.insert(current_id) {
778            let invocation_id = incoming_invocation.expect("only descendants can repeat");
779            return Err(
780                crate::delegation_events::DelegationEventError::Corrupt(invocation_id).into(),
781            );
782        }
783        if let Some(rollup) = rollups.get(&current_id) {
784            crate::usage::merge(usage, rollup);
785            session_count = session_count.saturating_add(1);
786        }
787        if let Some(descendants) = children.get(&current_id) {
788            pending.extend(
789                descendants
790                    .iter()
791                    .map(|(child_id, invocation_id)| (*child_id, Some(*invocation_id))),
792            );
793        }
794    }
795    Ok(session_count)
796}
