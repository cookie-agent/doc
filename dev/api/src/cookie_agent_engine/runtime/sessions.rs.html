45    /// The current runtime and the root policy a session created with
46    /// `selection` would freeze.
47    pub(crate) fn freeze_selection_policy(
48        &self,
49        selection: &RunSelection,
50    ) -> Result<
51        (
52            Arc<crate::runtime_snapshot::PublishedRuntime>,
53            policy::FrozenRunPolicy,
54        ),
55        EngineError,
56    > {
57        self.reconcile_provider_store()?;
58        let runtime = self.current_runtime();
59        if runtime.result.snapshot.models.is_empty() {
60            return Err(EngineError::NoRunnableModel);
61        }
62        let agents = runtime.agents_for_preset(selection.preset.as_deref())?;
63        let agent = resolve_agent(&agents, &selection.agent)?.clone();
64        let policy = freeze_root_agent_policy(
65            &agent,
66            agents,
67            Arc::clone(&runtime),
68            &selection.model,
69            self.inner.config.runtime.delegation.max_depth,
70            policy::ResultLimits::from_config(&self.inner.config.runtime),
71            self.inner.config.runtime.model_retry,
72        )?;
73        Ok((runtime, policy))
74    }

76    /// [`Self::create_session`] that also commits an initial permission
77    /// overlay and sets the new tree's permission mode before the session
78    /// can run anything. The rules are validated before the session exists.
79    pub async fn create_session_with_permissions(
80        &self,
81        selection: RunSelection,
82        mode: Option<PermissionMode>,
83        rules: Vec<PermissionRule>,
84    ) -> Result<SessionMeta, EngineError> {
85        let overlay = SessionPermissionOverlay { rules };
86        overlay
87            .validate()
88            .map_err(|error| EngineError::Permission(error.to_string()))?;
89        let meta = self.create_session(selection)?;
90        let session_id = meta.session_id;
91        if !overlay.rules.is_empty() {
92            self.append(
93                session_id,
94                None,
95                EventOrigin::new("user").expect("static event origin is valid"),
96                Event::SessionPermissionOverlaySet { overlay },
97            )
98            .await?;
99        }
100        if let Some(mode) = mode {
101            self.set_permission_mode(session_id, mode)?;
102        }
103        Ok(meta)
104    }

106    pub fn create_session(&self, selection: RunSelection) -> Result<SessionMeta, EngineError> {
107        let (runtime, policy) = self.freeze_selection_policy(&selection)?;
108        let id = SessionId::new_v7();
109        let cwd_identity = cwd_identity(self.inner.store.cwd())?;
110        // A new root begins an empty tree, so its first handle is trivially
111        // unique; it still goes through the same generation path so every
112        // session carries a handle uniformly.
113        let short_id = super::handles::generate_handle(
114            selection.agent.as_str(),
115            &HashSet::new(),
116            super::handles::random_hex,
117        )?;
118        let creation = Event::SessionCreated {
119            origin: SessionOrigin::Root,
120            short_id: Some(short_id),
121            cwd_identity: cwd_identity.clone(),
122            creation_selection: selection.clone(),
123            creation_agent: Box::new(policy.agent.clone()),
124            runtime_revision: runtime.result.snapshot.runtime_revision.clone(),
125            catalog_revision: runtime.result.snapshot.catalog_revision.clone(),
126            provider_state_revision: runtime.result.snapshot.provider_state_revision.clone(),
127            model_revision: runtime.result.snapshot.model_revision.clone(),
128            agent_revision: runtime.result.snapshot.agent_revision.clone(),
129            recipe_registry_revision: runtime.result.snapshot.recipe_registry_revision.clone(),
130            manifest_revision: runtime.current_manifest.revision.clone(),
131        };
132        self.inner.store.create(
133            id,
134            EventOrigin::new("user").expect("static event origin is valid"),
135            creation,
136        )?;
137        self.spawn_actor(id)?;
138        Ok(self.inner.store.get(id)?.metadata())
139    }

210    pub fn session_tree_usage(&self, id: SessionId) -> Result<SessionTreeUsageResult, EngineError> {
211        // A cold root's index is only a cache. Complete the tree before using
212        // its summaries, otherwise a missing or stale index can produce a
213        // plausible but incomplete rollup.
214        self.inner.store.ensure_tree_for(id)?;
215        let summaries = self.inner.store.tree_summaries(id)?;
216        let mut children: HashMap<SessionId, Vec<(SessionId, InvocationId)>> = HashMap::new();
217        let mut rollups = HashMap::new();
218        for summary in summaries {
219            let session_id = summary.meta.session_id;
220            if let SessionOrigin::Delegated {
221                root_session_id,
222                parent_session_id,
223                invocation_id,
224                ..
225            } = summary.meta.origin
226                && root_session_id == id
227            {
228                let edge = (session_id, invocation_id);
229                children.entry(parent_session_id).or_default().push(edge);
230            }
231            rollups.insert(session_id, summary.usage_rollup);
232        }
233
234        let mut usage = UsageRollup::default();
235        let session_count = merge_session_tree_usage(id, &children, &rollups, &mut usage)?;
236        Ok(SessionTreeUsageResult {
237            session_id: id,
238            usage: crate::usage::with_pricing(
239                usage,
240                &self.inner.config.runtime.pricing,
241                &self.catalog_pricing(),
242            ),
243            session_count,
244        })
245    }

281    pub fn children(
282        &self,
283        id: SessionId,
284    ) -> Result<Vec<cookie_agent_protocol::ChildSummary>, EngineError> {
285        // Listing is a use of the tree: complete it first so nested delegations
286        // and their summaries are in view (§3.2.2). An incomplete tree is
287        // reported, never papered over with the pre-load cache (review L14).
288        self.ensure_tree_loaded(id)?;
289        let known: HashSet<_> = self
290            .inner
291            .delegation_events
292            .entries()
293            .into_iter()
294            .map(|entry| {
295                (
296                    entry.reservation.invocation_id,
297                    entry.reservation.parent_session_id,
298                    entry.reservation.child_session_id,
299                )
300            })
301            .collect();
302        Ok(self
303            .inner
304            .store
305            .tree_summaries(id)?
306            .into_iter()
307            .filter_map(|child| match child.meta.origin {
308                SessionOrigin::Delegated {
309                    parent_session_id,
310                    invocation_id,
311                    ..
312                } if parent_session_id == id
313                    && known.contains(&(
314                        invocation_id,
315                        parent_session_id,
316                        child.meta.session_id,
317                    )) =>
318                {
319                    Some(ChildSummary {
320                        session_id: child.meta.session_id,
321                        agent: child.meta.creation_selection.agent.clone(),
322                        title: child.meta.title.clone(),
323                        title_updated_seq: child.meta.title_updated_seq,
324                        status: child.meta.status,
325                        usage: child.usage,
326                    })
327                }
328                _ => None,
329            })
330            .collect())
331    }

353    /// Adopts `id` and resolves the work its adoption found interrupted.
354    ///
355    /// Adoption schedules delegation recovery instead of running it inline, so
356    /// the wait below is what makes the ordering observable: when this returns,
357    /// every delegation record under the session's root that the adoption
358    /// classified as running without a live run here has been resolved, and the
359    /// root's background-delegation capacity reads settled rather than
360    /// transiently over-counted.
361    ///
362    /// The wait happens here, after the actor has replied, and never inside the
363    /// `Resume` handler: resolving a recovered delegate goes through its parent
364    /// session's mailbox, which is the very mailbox that handler occupies.
365    pub async fn resume(&self, id: SessionId) -> Result<SessionMeta, EngineError> {
366        let meta = self
367            .request(id, |reply| SessionCommand::Resume { reply })
368            .await?;
369        let root = self.inner.store.root_of(id).unwrap_or(id);
370        if !self
371            .await_delegation_recovery(root, DELEGATION_RECOVERY_SETTLE_BOUND)
372            .await
373        {
374            eprintln!(
375                "session {id} delegation recovery still settling after {DELEGATION_RECOVERY_SETTLE_BOUND:?}"
376            );
377        }
378        Ok(meta)
379    }
380    /// Delegated sessions a revert to `cut` deletes: children whose
381    /// reservation the cut removes (and no kept reservation resumes), with all
382    /// their descendants. A running one refuses the revert.
383    fn reverted_children(
384        &self,
385        events: &[Arc<cookie_agent_protocol::StoredEvent>],
386        cut: u64,
387    ) -> Result<HashSet<SessionId>, EngineError> {
388        let mut kept = HashSet::new();
389        let mut removed = Vec::new();
390        for event in events.iter() {
391            if let Event::DelegationReserved { reservation, .. } = &event.payload {
392                if event.seq <= cut {
393                    kept.insert(reservation.child_session_id);
394                } else {
395                    removed.push(reservation.child_session_id);
396                }
397            }
398        }
399        let mut deleted = HashSet::new();
400        let mut pending = removed
401            .into_iter()
402            .filter(|child| !kept.contains(child))
403            .collect::<Vec<_>>();
404        while let Some(child) = pending.pop() {
405            if !deleted.insert(child) {
406                continue;
407            }
408            match self.inner.store.get(child) {
409                Ok(projection) if projection.status == SessionStatus::Running => {
410                    return Err(EngineError::SessionRunning(child));
411                }
412                Ok(_) => {}
413                // Never published, or already gone: nothing on disk to keep.
414                Err(SessionError::Missing(_)) => continue,
415                Err(error) => return Err(error.into()),
416            }
417            pending.extend(
418                self.inner
419                    .store
420                    .children(child)?
421                    .into_iter()
422                    .map(|summary| summary.session_id),
423            );
424        }
425        Ok(deleted)
426    }

428    /// Forgets the in-memory state of sessions a revert deletes.
429    fn forget_deleted_sessions(&self, deleted: &HashSet<SessionId>) {
430        if deleted.is_empty() {
431            return;
432        }
433        fn forget<V>(map: &std::sync::Mutex<HashMap<SessionId, V>>, deleted: &HashSet<SessionId>) {
434            map.lock()
435                .unwrap_or_else(|poisoned| poisoned.into_inner())
436                .retain(|session, _| !deleted.contains(session));
437        }
438        self.kill_deleted_background_jobs(deleted);
439        forget(&self.inner.sessions.actors, deleted);
440        forget(&self.inner.sessions.producers, deleted);
441        forget(&self.inner.skills_runtime.grants, deleted);
442        forget(&self.inner.skills_runtime.models, deleted);
443        forget(&self.inner.skills_runtime.pending_child, deleted);
444        forget(&self.inner.compaction.deferred, deleted);
445        forget(&self.inner.compaction.context_token_estimators, deleted);
446        forget(&self.inner.delegation.recovery_pending, deleted);
447        self.inner
448            .delegation
449            .by_session
450            .lock()
451            .unwrap_or_else(|poisoned| poisoned.into_inner())
452            .retain(|child, record| {
453                !deleted.contains(child) && !deleted.contains(&record.parent_session_id)
454            });
455        self.inner
456            .delegation
457            .queue
458            .lock()
459            .unwrap_or_else(|poisoned| poisoned.into_inner())
460            .retain(|child| !deleted.contains(child));
461        self.inner.delegation_events.forget_sessions(deleted);
462    }

464    pub async fn revert_session(
465        &self,
466        session_id: SessionId,
467        through_seq: u64,
468        _origin: EventOrigin,
469    ) -> Result<SessionRevertResult, EngineError> {
470        let context_id = crate::plugin::plugin_context_id();
471        let mut instructions = self
472            .inner
473            .store
474            .get(session_id)?
475            .log
476            .event_snapshot()
477            .iter()
478            .find_map(|event| match &event.payload {
479                cookie_agent_protocol::EventPayload::UserInputSubmitted { input }
480                    if event.seq > through_seq =>
481                {
482                    Some(input.clone())
483                }
484                _ => None,
485            });
486        let mut instructions_override = None;
487        for plugin in self.inner.plugins.interception_plugins(
488            cookie_agent_protocol::ExtensionInterceptionHook::SessionBeforeRevert,
489        ) {
490            let result = self
491                .inner
492                .plugins
493                .intercept_named::<_, cookie_agent_protocol::ExtensionSessionBeforeRevertResult>(
494                    &plugin,
495                    cookie_agent_protocol::PLUGIN_INTERCEPT_SESSION_BEFORE_REVERT_METHOD,
496                    &cookie_agent_protocol::ExtensionSessionBeforeRevertParams {
497                        session_id,
498                        context_id: context_id.clone(),
499                        through_seq,
500                        instructions: instructions.clone(),
501                    },
502                    Some(session_id),
503                    Some(&context_id),
504                )
505                .await;
506            match result {
507                Ok(result)
508                    if result.action
509                        == cookie_agent_protocol::ExtensionSessionBeforeRevertAction::Block =>
510                {
511                    let reason = result
512                        .reason
513                        .unwrap_or_else(|| format!("session revert blocked by {plugin}"));
514                    self.record_plugin_diagnostic(
515                        session_id,
516                        plugin,
517                        cookie_agent_protocol::PluginDiagnosticKind::HookBlocked,
518                        reason.clone(),
519                    );
520                    return Err(EngineError::SessionOperationBlocked(reason));
521                }
522                Ok(result)
523                    if result.action
524                        == cookie_agent_protocol::ExtensionSessionBeforeRevertAction::Override =>
525                {
526                    if let Some(replacement) = result
527                        .instructions_override
528                        .filter(|value| !value.trim().is_empty())
529                    {
530                        instructions = Some(replacement.clone());
531                        instructions_override = Some(replacement);
532                    } else {
533                        self.record_plugin_diagnostic(
534                            session_id,
535                            plugin,
536                            cookie_agent_protocol::PluginDiagnosticKind::InvalidModification,
537                            "empty revert instruction override".into(),
538                        );
539                    }
540                }
541                Ok(_) => {}
542                Err(error) => self.record_interception_error(session_id, plugin, error),
543            }
544        }
545        let session = session_id;
546        self.on_actor_async(session, move |engine| async move {
547            if !engine.reserve_compaction(session) {
548                return Err(EngineError::SessionRunning(session));
549            }
550            let result = (|| {
551                let projection = engine.inner.store.get(session)?;
552                if projection.status == SessionStatus::Running {
553                    return Err(EngineError::SessionRunning(session));
554                }
555                let tip = projection.log.last_event().map_or(0, |event| event.seq);
556                if through_seq == 0 || through_seq > tip {
557                    return Err(SessionError::InvalidSequence {
558                        session_id: session,
559                        through_seq,
560                    }
561                    .into());
562                }
563                // A revert removes the events after the cut: the log keeps no
564                // marker, and later appends reuse the freed sequences.
565                let events = projection.log.all_events();
566                let cut = revert_cut(&events, through_seq);
567                let deleted = engine.reverted_children(&events, cut)?;
568                engine.inner.store.truncate(session, cut)?;
569                // The cut can fall inside a run, with its tool calls, internal
570                // agents or approvals still open: close them as a restart would.
571                engine.reconcile_session_with(session, "session reverted")?;
572                engine
573                    .inner
574                    .compaction
575                    .context_token_estimators
576                    .lock()
577                    .unwrap_or_else(|poisoned| poisoned.into_inner())
578                    .remove(&session);
579                engine.rebuild_visible_tree_grants();
580                engine.inner.delegation_events.reconcile_parent(session)?;
581                engine.reconcile_reverted_producers_direct(session)?;
582                engine.forget_deleted_sessions(&deleted);
583                engine
584                    .inner
585                    .store
586                    .delete_children(&deleted.iter().copied().collect::<Vec<_>>())?;
587                Ok(SessionRevertResult {
588                    session: engine.inner.store.get(session)?.metadata(),
589                    instructions_override,
590                })
591            })();
592            engine.release_compaction_direct(session).await;
593            result
594        })
595        .await
596    }
597    pub async fn fork_session(
598        &self,
599        session_id: SessionId,
600        through_seq: u64,
601        origin: EventOrigin,
602    ) -> Result<SessionForkResult, EngineError> {
603        let context_id = crate::plugin::plugin_context_id();
604        for plugin in self.inner.plugins.interception_plugins(
605            cookie_agent_protocol::ExtensionInterceptionHook::SessionBeforeFork,
606        ) {
607            let result = self
608                .inner
609                .plugins
610                .intercept_named::<_, cookie_agent_protocol::ExtensionSessionBeforeForkResult>(
611                    &plugin,
612                    cookie_agent_protocol::PLUGIN_INTERCEPT_SESSION_BEFORE_FORK_METHOD,
613                    &cookie_agent_protocol::ExtensionSessionBeforeForkParams {
614                        session_id,
615                        context_id: context_id.clone(),
616                        through_seq,
617                    },
618                    Some(session_id),
619                    Some(&context_id),
620                )
621                .await;
622            match result {
623                Ok(result)
624                    if result.action == cookie_agent_protocol::ExtensionAllowBlockAction::Block =>
625                {
626                    let reason = result
627                        .reason
628                        .unwrap_or_else(|| format!("session fork blocked by {plugin}"));
629                    self.record_plugin_diagnostic(
630                        session_id,
631                        plugin,
632                        cookie_agent_protocol::PluginDiagnosticKind::HookBlocked,
633                        reason.clone(),
634                    );
635                    return Err(EngineError::SessionOperationBlocked(reason));
636                }
637                Ok(_) => {}
638                Err(error) => self.record_interception_error(session_id, plugin, error),
639            }
640        }
641        let forked = self.inner.store.fork(session_id, through_seq, origin)?;
642        // A fork into a new tree takes the artifacts its history references
643        // along, so it never resolves anything in the tree it came from
644        // (tree-local D2).
645        let mut referenced = std::collections::HashSet::new();
646        crate::runtime::artifacts::collect_artifact_references_in_events(
647            &self.inner.store.log(forked)?.event_snapshot(),
648            &mut referenced,
649        )?;
650        self.inner
651            .artifacts
652            .copy_into_tree(session_id, forked, referenced)?;
653        let producer_state = self.goal_producer_projection(forked)?;
654        if producer_state.goal.is_some()
655            || producer_state
656                .messages
657                .iter()
658                .any(|message| !message.consumed && !message.discarded)
659        {
660            self.reconcile_producers(forked).await?;
661        }
662        Ok(SessionForkResult { session_id: forked })
663    }
664    pub fn set_permission_mode(
665        &self,
666        session_id: SessionId,
667        mode: PermissionMode,
668    ) -> Result<(), EngineError> {
669        self.ensure_session_owned(session_id)?;
670        let session = self.inner.store.get(session_id)?;
671        let root = root_id(&session.meta.origin, session_id);
672        self.inner
673            .approvals
674            .permission_modes
675            .lock()
676            .unwrap_or_else(|poisoned| poisoned.into_inner())
677            .insert(root, mode);
678        Ok(())
679    }
680    pub(crate) fn permission_mode(&self, session_id: SessionId) -> PermissionMode {
681        let root = self
682            .inner
683            .store
684            .get(session_id)
685            .map(|session| root_id(&session.meta.origin, session_id))
686            .unwrap_or(session_id);
687        self.inner
688            .approvals
689            .permission_modes
690            .lock()
691            .unwrap_or_else(|poisoned| poisoned.into_inner())
692            .get(&root)
693            .copied()
694            .unwrap_or_default()
695    }
696    pub async fn rename_session(
697        &self,
698        params: SessionRenameParams,
699        _origin: EventOrigin,
700    ) -> Result<SessionRenameResult, EngineError> {
701        let session_id = params.session_id;
702        let reset = matches!(params.change, SessionRenameChange::Reset);
703        // A rename appends only `SessionTitleCommitted`, which producer
704        // reconciliation never reads.
705        let mut result = self
706            .on_actor_unreconciled(session_id, move |engine| {
707                let projection = engine.inner.store.get(session_id)?;
708                if let Some(record) = projection.rename_records.get(&params.client_rename_id) {
709                    if record.conflicts_with(&params) {
710                        return Err(EngineError::RenameConflict);
711                    }
712                    return Ok(SessionRenameResult {
713                        client_rename_id: params.client_rename_id,
714                        session: projection.metadata(),
715                    });
716                }
717                let commit = match params.change {
718                    SessionRenameChange::Set { title } => SessionTitleChange::UserSet {
719                        title,
720                        client_rename_id: params.client_rename_id.clone(),
721                    },
722                    SessionRenameChange::Clear => SessionTitleChange::UserClear {
723                        client_rename_id: params.client_rename_id.clone(),
724                    },
725                    SessionRenameChange::Reset => SessionTitleChange::UserReset {
726                        client_rename_id: params.client_rename_id.clone(),
727                    },
728                };
729                let input_through_seq = projection.log.last_event().map_or(0, |event| event.seq);
730                engine.append_direct(
731                    session_id,
732                    None,
733                    super::event_origin("user"),
734                    Event::SessionTitleCommitted {
735                        input_through_seq,
736                        change: commit,
737                    },
738                )?;
739                Ok(SessionRenameResult {
740                    client_rename_id: params.client_rename_id,
741                    session: engine.inner.store.get(session_id)?.metadata(),
742                })
743            })
744            .await?;
745        if reset {
746            self.generate_title_after_reset(session_id).await?;
747            result.session = self.inner.store.get(session_id)?.metadata();
748        }
749        Ok(result)
750    }
751}

753/// Where a revert to `through_seq` cuts the log. A run the cut would leave
754/// holding nothing but its start goes too: reverting to just before a message
755/// targets the event before it, which is usually that run's `RunStarted`.
756fn revert_cut(events: &[Arc<cookie_agent_protocol::StoredEvent>], through_seq: u64) -> u64 {
757    let mut cut = through_seq;
758    loop {
759        let mut open = None;
760        let mut started = false;
761        for event in events.iter().take_while(|event| event.seq <= cut) {
762            match &event.payload {
763                Event::RunStarted { .. } => {
764                    open = event.run_id.map(|run| (run, event.seq));
765                    started = false;
766                }
767                Event::RunCompleted { .. }
768                | Event::RunFailed { .. }
769                | Event::RunCancelled { .. }
770                | Event::RunInterrupted { .. }
771                    if open.is_some_and(|(run, _)| event.run_id == Some(run)) =>
772                {
773                    open = None;
774                }
775                Event::UserInputSubmitted { .. }
776                | Event::UserInputApplied { .. }
777                | Event::ModelAttemptStarted { .. }
778                | Event::ToolCallStarted { .. }
779                    if open.is_some_and(|(run, _)| event.run_id == Some(run)) =>
780                {
781                    started = true;
782                }
783                _ => {}
784            }
785        }
786        match open {
787            Some((_, start)) if !started && start > 1 => cut = start - 1,
788            _ => return cut,
789        }
790    }
791}

793#[cfg(test)]
794fn validated_tree_usage_edge(
795    origin: &SessionOrigin,
796    child_session_id: SessionId,
797    known: &HashSet<(InvocationId, SessionId, SessionId)>,
798) -> Option<(SessionId, (SessionId, InvocationId))> {
799    match origin {
800        SessionOrigin::Delegated {
801            parent_session_id,
802            invocation_id,
803            ..
804        } if known.contains(&(*invocation_id, *parent_session_id, child_session_id)) => {
805            Some((*parent_session_id, (child_session_id, *invocation_id)))
806        }
807        _ => None,
808    }
809}
810
811fn merge_session_tree_usage(
812    session_id: SessionId,
813    children: &HashMap<SessionId, Vec<(SessionId, InvocationId)>>,
814    rollups: &HashMap<SessionId, UsageRollup>,
815    usage: &mut UsageRollup,
816) -> Result<u64, EngineError> {
817    let mut visited = HashSet::new();
818    let mut pending = vec![(session_id, None)];
819    let mut session_count = 0_u64;
820    while let Some((current_id, incoming_invocation)) = pending.pop() {
821        if !visited.insert(current_id) {
822            let invocation_id = incoming_invocation.expect("only descendants can repeat");
823            return Err(
824                crate::delegation_events::DelegationEventError::Corrupt(invocation_id).into(),
825            );
826        }
827        if let Some(rollup) = rollups.get(&current_id) {
828            crate::usage::merge(usage, rollup);
829            session_count = session_count.saturating_add(1);
830        }
831        if let Some(descendants) = children.get(&current_id) {
832            pending.extend(
833                descendants
834                    .iter()
835                    .map(|(child_id, invocation_id)| (*child_id, Some(*invocation_id))),
836            );
837        }
838    }
839    Ok(session_count)
840}
