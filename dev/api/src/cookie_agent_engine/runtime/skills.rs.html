167    /// The skills `session` sees through its attachments.
168    pub(crate) fn session_skills(&self, session: SessionId) -> Result<SessionSkills, EngineError> {
169        let mut user = Vec::new();
170        let mut project = Vec::new();
171        for attachment in self.session_runtimes(session)? {
172            let Some(instance) = self.inner.runtimes.instance(&attachment) else {
173                continue;
174            };
175            let registry = instance.skills();
176            // The daemon's own global runtime carries the user skills; every
177            // other attachment's skills are its project skills.
178            if attachment.runtime_id.as_str() == cookie_agent_protocol::RuntimeId::GLOBAL {
179                user.push((attachment.runtime_id, registry));
180            } else {
181                project.push((attachment.name, attachment.runtime_id, registry));
182            }
183        }
184        let mut offered = HashMap::<String, usize>::new();
185        for (_, _, registry) in &project {
186            for (name, _) in registry.skills() {
187                *offered.entry(name.clone()).or_default() += 1;
188            }
189        }
190        let published = |attachment: &AttachmentName, name: &str| {
191            if offered.get(name).is_some_and(|count| *count > 1) {
192                format!("{attachment}_{name}")
193            } else {
194                name.to_owned()
195            }
196        };
197        let mut skills = BTreeMap::new();
198        let mut discoveries = Vec::new();
199        for (attachment, runtime, registry) in &project {
200            for (name, _) in registry.skills() {
201                skills.insert(
202                    published(attachment, name),
203                    SessionSkill {
204                        registry: Arc::clone(registry),
205                        key: name.clone(),
206                        attachment: Some(attachment.clone()),
207                        runtime: runtime.clone(),
208                    },
209                );
210            }
211            discoveries.extend(registry.discoveries().iter().map(|discovery| {
212                SessionSkillDiscovery {
213                    name: published(attachment, &discovery.name),
214                    source: discovery.source,
215                    path: discovery.path.clone(),
216                    precedence_winner: discovery.precedence_winner,
217                }
218            }));
219        }
220        for (runtime, registry) in &user {
221            for (name, _) in registry.skills() {
222                if offered.contains_key(name) {
223                    continue;
224                }
225                skills.insert(
226                    name.clone(),
227                    SessionSkill {
228                        registry: Arc::clone(registry),
229                        key: name.clone(),
230                        attachment: None,
231                        runtime: runtime.clone(),
232                    },
233                );
234            }
235            discoveries.extend(registry.discoveries().iter().map(|discovery| {
236                SessionSkillDiscovery {
237                    name: discovery.name.clone(),
238                    source: discovery.source,
239                    path: discovery.path.clone(),
240                    precedence_winner: discovery.precedence_winner
241                        && !offered.contains_key(&discovery.name),
242                }
243            }));
244        }
245        Ok(SessionSkills {
246            skills,
247            discoveries,
248        })
249    }

261    /// The turn-scoped skill grants that apply to tools of `attachment`
262    /// (`None` for the base layer): every user skill's grants, and the
263    /// grants of project skills that attachment contributed. A project
264    /// skill's grants for the base layer's own actions (`skill`, `delegate`,
265    /// `message`) apply to the base layer, which has no attachment.
266    pub(crate) fn skill_grants_for_session(
267        &self,
268        session: SessionId,
269        attachment: Option<&AttachmentName>,
270    ) -> Option<SessionPermissionOverlay> {
271        self.inner
272            .skills_runtime
273            .grants
274            .lock()
275            .unwrap_or_else(|poisoned| poisoned.into_inner())
276            .get(&session)
277            .map(|grants| {
278                let rules = grants
279                    .values()
280                    .flat_map(|grant| {
281                        debug_assert!(grant.source.starts_with("skill:"));
282                        grant.overlay.rules.iter().filter(move |rule| {
283                            match (&grant.attachment, attachment) {
284                                (None, _) => true,
285                                (Some(owner), Some(attachment)) => owner == attachment,
286                                (Some(_), None) => rule.action.is_base_layer(),
287                            }
288                        })
289                    })
290                    .cloned()
291                    .collect();
292                SessionPermissionOverlay { rules }
293            })
294    }

296    /// The daemon's own skills, without a session: the global runtime's
297    /// user and official skills. `permission_effect` is `ask` and `visible`
298    /// says whether the skill wins its name, since no agent decides them.
299    #[must_use]
300    pub fn list_daemon_skills(&self) -> SkillsListResult {
301        let registry = self.inner.runtimes.global().skills();
302        let mut skills = registry
303            .discoveries()
304            .iter()
305            .filter_map(|discovery| {
306                let document = registry.get(&discovery.name)?;
307                let mut descriptor = skill_descriptor(
308                    &discovery.name,
309                    document,
310                    discovery.precedence_winner,
311                    discovery.precedence_winner,
312                );
313                descriptor.location = discovery.path.to_string_lossy().into_owned();
314                descriptor.source = discovery.source;
315                Some(descriptor)
316            })
317            .collect::<Vec<_>>();
318        self.mark_official_versions(&mut skills);
319        skills.sort_by(|left, right| {
320            left.name
321                .cmp(&right.name)
322                .then_with(|| right.precedence_winner.cmp(&left.precedence_winner))
323                .then_with(|| left.location.cmp(&right.location))
324        });
325        SkillsListResult { skills }
326    }

328    /// Gives each official skill of `skills` its version.
329    fn mark_official_versions(&self, skills: &mut [SkillDescriptor]) {
330        if !skills
331            .iter()
332            .any(|skill| skill.source == cookie_agent_protocol::SkillSource::Official)
333        {
334            return;
335        }
336        let versions = self.official_skills();
337        for skill in skills {
338            if skill.source == cookie_agent_protocol::SkillSource::Official {
339                skill.version = versions.get(&skill.name).cloned();
340            }
341        }
342    }

344    pub fn list_skills(&self, session: SessionId) -> Result<SkillsListResult, EngineError> {
345        let projection = self.inner.store.get(session)?;
346        let policy = permissions::governing_agent_for_skills(&projection);
347        let layers = Layers::new(&policy.exposure, &projection.tool_exposure, false);
348        let view = self.session_skills(session)?;
349        let mut skills = view
350            .discoveries
351            .iter()
352            .filter_map(|discovery| {
353                let skill = view.get(&discovery.name)?;
354                let mut descriptor = skill_descriptor(
355                    &discovery.name,
356                    skill.document(),
357                    discovery.precedence_winner,
358                    model_sees(&layers, &discovery.name, skill),
359                );
360                descriptor.location = discovery.path.to_string_lossy().into_owned();
361                descriptor.source = discovery.source;
362                Some(descriptor)
363            })
364            .collect::<Vec<_>>();
365        self.mark_official_versions(&mut skills);
366        // By name, the skill that wins its name first, then by location:
367        // the order does not depend on how the platform spells paths.
368        skills.sort_by(|left, right| {
369            left.name
370                .cmp(&right.name)
371                .then_with(|| right.precedence_winner.cmp(&left.precedence_winner))
372                .then_with(|| left.location.cmp(&right.location))
373        });
374        Ok(SkillsListResult { skills })
375    }

406    pub fn get_model_skill(
407        &self,
408        session: SessionId,
409        name: &str,
410        args: &str,
411    ) -> Result<SkillsGetResult, EngineError> {
412        let projection = self.inner.store.get(session)?;
413        let policy = permissions::governing_agent_for_skills(&projection);
414        let layers = Layers::new(&policy.exposure, &projection.tool_exposure, false);
415        let view = self.session_skills(session)?;
416        let visible =
417            |skill_name: &str, skill: &SessionSkill| model_sees(&layers, skill_name, skill);
418        let Some(skill) = view.get(name).filter(|skill| visible(name, skill)) else {
419            if view.get(name).is_some() && !layers.skill_on(name) {
420                return Err(EngineError::ToolFailed(format!(
421                    "skill `{name}` is off in this session"
422                )));
423            }
424            let hints = view
425                .iter()
426                .filter(|(skill_name, skill)| visible(skill_name, skill))
427                .map(|(skill_name, _)| skill_name.as_str())
428                .collect::<Vec<_>>()
429                .join(", ");
430            return Err(EngineError::ToolFailed(format!(
431                "unknown or unavailable model skill `{name}`; valid skills: {hints}"
432            )));
433        };
434        Ok(SkillsGetResult {
435            skill: skill_descriptor(name, skill.document(), true, true),
436            rendered: self.loaded_skill_text(skill, args),
437        })
438    }

456    pub(super) async fn install_prepared_skill(
457        &self,
458        session: SessionId,
459        run: Option<RunId>,
460        plan: &PreparedSkillInvocation,
461    ) -> Result<SkillInvocation, EngineError> {
462        let projection = self.inner.store.get(session)?;
463        let name = &plan.name;
464        let duplicate = projection
465            .log
466            .state()
467            .skill_loaded(name, plan.rendered.as_bytes());
468        let event = if duplicate {
469            EventPayload::SkillInvocationNoted { name: name.clone() }
470        } else {
471            EventPayload::SkillLoaded {
472                name: name.clone(),
473                rendered_body: plan.rendered.clone(),
474                source_path: plan.source_path.clone(),
475                args: plan.args.clone(),
476                base_dir: plan.base_dir.clone(),
477                supporting_files: plan.supporting_files.clone(),
478            }
479        };
480        self.append(session, run, super::event_origin("engine:skills"), event)
481            .await?;
482        let mut overlays = self
483            .inner
484            .skills_runtime
485            .grants
486            .lock()
487            .unwrap_or_else(|poisoned| poisoned.into_inner());
488        let per_skill = overlays.entry(session).or_default();
489        if plan.grants.is_empty() {
490            per_skill.remove(name);
491        } else {
492            per_skill.insert(
493                name.clone(),
494                SkillGrantOverlay {
495                    source: format!("skill:{name}"),
496                    attachment: plan.attachment.clone(),
497                    overlay: SessionPermissionOverlay {
498                        rules: plan.grants.clone(),
499                    },
500                },
501            );
502        }
503        if per_skill.is_empty() {
504            overlays.remove(&session);
505        }
506        drop(overlays);
507        if let Some(model) = &plan.model {
508            self.inner
509                .skills_runtime
510                .models
511                .lock()
512                .unwrap_or_else(|poisoned| poisoned.into_inner())
513                .insert(session, model.clone());
514        }
515        Ok(SkillInvocation {
516            name: name.clone(),
517            rendered: plan.rendered.clone(),
518            context: plan.context,
519        })
520    }

606            .skills_runtime
607            .direct_calls

614            } else {
615                error
616            }

620    pub async fn execute_skill_fork(
621        &self,
622        session: SessionId,
623        run: RunId,
624        call_id: ToolCallId,
625        name: &str,
626        args: &str,
627    ) -> Result<cookie_agent_protocol::PersistedToolResult, EngineError> {
628        let plan = self.prepare_skill_invocation(session, name, args)?;
629        let agent_type = self
630            .delegate_targets(session)?
631            .into_iter()
632            .next()
633            .ok_or_else(|| {
634                EngineError::ToolFailed("fork skill requires an available delegation target".into())
635            })?;
636        let active = self
637            .inner
638            .sessions
639            .active
640            .lock()
641            .unwrap_or_else(|poisoned| poisoned.into_inner())
642            .get(&run)
643            .cloned()
644            .filter(|active| active.session == session)
645            .ok_or(EngineError::MissingRun(run))?;
646        self.inner
647            .skills_runtime
648            .pending_forks
649            .lock()
650            .unwrap_or_else(|poisoned| poisoned.into_inner())
651            .insert(call_id, plan.clone());
652        let result = self
653            .execute_prepared_call(
654                active,
655                run,
656                ToolCall {
657                    id: call_id,
658                    name: "delegate_subagent".into(),
659                    arguments: serde_json::json!({
660                        "description": format!("Skill {name}"),
661                        "prompt": format!("Apply the staged skill `{name}`."),
662                        "agent_type": agent_type,
663                        "background": false,
664                        "resume_session_id": null,
665                        "inherit_context": false,
666                    }),
667                },
668            )
669            .await;
670        self.inner
671            .skills_runtime
672            .pending_forks
673            .lock()
674            .unwrap_or_else(|poisoned| poisoned.into_inner())
675            .remove(&call_id);
676        result
677    }

679    async fn execute_prepared_call(
680        &self,
681        active: Arc<ActiveRun>,
682        run: RunId,
683        call: ToolCall,
684    ) -> Result<cookie_agent_protocol::PersistedToolResult, EngineError> {
685        let binding = active
686            .policy
687            .selected_suffix
688            .first()
689            .ok_or(EngineError::NoRunnableModel)?;
690        let turn_context = Arc::new(TurnAgentContext {
691            agent: active.policy.agent.agent.clone(),
692            model: binding.selection.model.clone(),
693            adapter: crate::policy::wire_adapter(binding.descriptor.adapter_id.as_str()),
694            input_media: crate::policy::tool_input_media(
695                binding.descriptor.adapter_id.as_str(),
696                &active
697                    .policy
698                    .model_capabilities(binding)
699                    .ok_or(EngineError::NoRunnableModel)?,
700            ),
701        });
702        let prepared = self
703            .prepare_tool_call(
704                active.session,
705                run,
706                call,
707                &active.policy,
708                turn_context.clone(),
709            )
710            .await;
711        self.execute_tool(active, run, prepared, turn_context, false)
712            .await
713            .map_err(|failure| EngineError::ToolFailed(failure.message))
714    }

820    /// The listing of the skills a run of `policy` in `session` may see,
821    /// loaded with `tool` and sized for a context of `context_tokens`; `None`
822    /// when it sees none. The model loop puts it in the system prompt, and
823    /// an ACP agent's first prompt carries it.
824    pub(crate) fn skill_listing(
825        &self,
826        session: SessionId,
827        policy: &FrozenRunPolicy,
828        tool: &str,
829        context_tokens: u64,
830    ) -> Result<Option<String>, EngineError> {
831        let projection = self.inner.store.get(session)?;
832        let view = self.session_skills(session)?;
833        let layers = Layers::of(&projection, policy);
834        let visible = view
835            .iter()
836            .filter(|(name, skill)| model_sees(&layers, name, skill))
837            .map(|(name, skill)| (name.as_str(), skill.document()))
838            .collect::<Vec<_>>();
839        if visible.is_empty() {
840            return Ok(None);
841        }
842        cookie_agent_config::render_skill_listing(visible, tool, context_tokens)
843            .map(Some)
844            .map_err(|error| EngineError::Config(Box::new(error)))
845    }

847    /// The plan for invoking `skill`, known to the session as `name`. A
848    /// project skill's `allowed-tools` grants carry the attachment that
849    /// contributed it.
850    fn prepared_skill_invocation(
851        &self,
852        name: &str,
853        skill: &SessionSkill,
854        args: &str,
855    ) -> PreparedSkillInvocation {
856        let document = skill.document();
857        PreparedSkillInvocation {
858            name: name.to_owned(),
859            attachment: skill.attachment.clone(),
860            args: args.to_owned(),
861            rendered: self.loaded_skill_text(skill, args),
862            source_path: document.path.to_string_lossy().into_owned(),
863            base_dir: document.base_dir.to_string_lossy().into_owned(),
864            supporting_files: document
865                .supporting_files
866                .iter()
867                .map(|path| path.to_string_lossy().into_owned())
868                .collect(),
869            grants: document
870                .frontmatter
871                .allowed_tools
872                .iter()
873                .map(|grant| PermissionRule {
874                    action: grant.action.clone(),
875                    runtime: None,
876                    resource: grant.pattern.clone(),
877                    effect: PermissionEffect::Allow,
878                })
879                .collect(),
880            model: document.frontmatter.model.clone(),
881            context: document.frontmatter.context,
882        }
883    }

885    /// What loading `skill` gives the model: its rendered instructions, and
886    /// where its files are when it has any besides `SKILL.md`. The machine
887    /// is named whenever the daemon can: the text outlives the attachments
888    /// of the moment, since a loaded skill stays pinned across compaction.
889    fn loaded_skill_text(&self, skill: &SessionSkill, args: &str) -> String {
890        let document = skill.document();
891        let rendered = render_skill(document, args);
892        if document.supporting_files.is_empty() {
893            return rendered;
894        }
895        let directory = document.base_dir.to_string_lossy();
896        let machine = self
897            .runtime_machine(skill.runtime.as_str())
898            .map_or_else(String::new, |machine| format!(" (machine: {machine})"));
899        format!(
900            "{}\n\nSkill directory: {directory}{machine}\n\
901             Relative paths in this skill are relative to this directory.\n",
902            rendered.trim_end()
903        )
904    }

907    /// its model's context.
908    pub(crate) fn live_skills_block(
909        &self,
910        session: SessionId,
911        policy: &FrozenRunPolicy,
912    ) -> Result<Option<String>, EngineError> {

914            .descriptor
915            .capabilities
916            .limits
917            .context
918            .unwrap_or(8_192);
919        self.skill_listing(session, policy, "the `skill` tool", context_tokens)

923        &self,
924        session: SessionId,
925        policy: &mut FrozenRunPolicy,
926    ) -> Result<(), EngineError> {

932            crate::runtime::prompt_blocks::SKILLS_BLOCK,

934        );
935        Ok(())
936    }

939/// The skill named `name` in `view`, or an error listing the valid names.
940fn session_skill<'a>(view: &'a SessionSkills, name: &str) -> Result<&'a SessionSkill, EngineError> {
941    view.get(name).ok_or_else(|| {
942        let names = view
943            .iter()
944            .map(|(name, _)| name.as_str())
945            .collect::<Vec<_>>()
946            .join(", ");
947        EngineError::ToolFailed(format!("unknown skill `{name}`; valid skills: {names}"))
948    })
949}

997fn render_skill(skill: &SkillDocument, args: &str) -> String {
998    let positional = args.split_whitespace().collect::<Vec<_>>();
999    let mut variables = BTreeMap::new();
1000    variables.insert("ARGUMENTS".to_owned(), args.to_owned());
1001    variables.insert(
1002        "{COOKIE_SKILL_DIR}".to_owned(),
1003        skill.base_dir.to_string_lossy().into_owned(),
1004    );
1005    for (index, value) in positional.into_iter().enumerate() {
1006        variables.insert((index + 1).to_string(), value.to_owned());
1007    }
1008    let mut output = String::new();
1009    let bytes = skill.body.as_bytes();
1010    let mut index = 0;
1011    while index < bytes.len() {
1012        if bytes[index] == b'$' {
1013            let rest = &skill.body[index + 1..];
1014            let key = if rest.starts_with("{COOKIE_SKILL_DIR}") {
1015                Some("{COOKIE_SKILL_DIR}")
1016            } else if rest.starts_with("ARGUMENTS") {
1017                Some("ARGUMENTS")
1018            } else {
1019                let digits = rest.bytes().take_while(u8::is_ascii_digit).count();
1020                (digits > 0).then_some(&rest[..digits])
1021            };
1022            if let Some(key) = key {
1023                output.push_str(variables.get(key).map_or("", String::as_str));
1024                index += 1 + key.len();
1025                continue;
1026            }
1027        }
1028        let character = skill.body[index..]
1029            .chars()
1030            .next()
1031            .expect("character boundary");
1032        output.push(character);
1033        index += character.len_utf8();
1034    }
1035    output
1036}
