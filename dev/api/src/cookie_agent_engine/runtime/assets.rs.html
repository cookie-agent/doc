23/// The two stores a session's asset tools reach: its tree's, and its
24/// project's (or the shared store, for a session in no project).
25#[derive(Clone, Debug, Eq, PartialEq)]
26pub(crate) struct SessionAssetStores {
27    pub(crate) session: AssetStore,
28    pub(crate) project: AssetStore,
29}
30
31impl SessionAssetStores {
32    /// The store an asset of `scope` goes to.
33    pub(crate) fn of(&self, scope: cookie_agent_protocol::AssetScope) -> &AssetStore {
34        match scope {
35            cookie_agent_protocol::AssetScope::Session => &self.session,
36            cookie_agent_protocol::AssetScope::Project => &self.project,
37        }
38    }
39}

54    /// Runs `call` on the asset stores off the async workers.
55    pub(crate) async fn assets_blocking<T: Send + 'static>(
56        &self,
57        call: impl FnOnce(&Engine, &AssetStores) -> Result<T, EngineError> + Send + 'static,
58    ) -> Result<T, EngineError> {
59        let engine = self.clone();
60        tokio::task::spawn_blocking(move || {
61            let stores = Arc::clone(&engine.inner.assets);
62            call(&engine, &stores)
63        })
64        .await
65        .map_err(|error| AssetError::Store(format!("the request did not finish: {error}")))?
66    }

68    /// The stores `session`'s asset tools reach. Blocking: it reads the
69    /// session's project.
70    pub(crate) fn session_asset_stores(&self, session: SessionId) -> SessionAssetStores {
71        let root = self.root_session_of(session);
72        let project = match self.session_project(session) {
73            Some(project) => AssetStore::Project {
74                project: project.id,
75            },
76            None => AssetStore::Shared,
77        };
78        SessionAssetStores {
79            session: AssetStore::Session { session: root },
80            project,
81        }
82    }

84    /// Refuses a store a client names that does not exist: a project the
85    /// daemon does not have, or a session store not named by a root whose
86    /// folder exists. Blocking.
87    fn check_client_store(&self, store: &AssetStore) -> Result<(), AssetError> {
88        match store {
89            AssetStore::Shared => Ok(()),
90            AssetStore::Project { project } => {
91                if self.inner.projects.exists(project) {
92                    Ok(())
93                } else {
94                    Err(AssetError::InvalidStore(format!(
95                        "there is no project {project}"
96                    )))
97                }
98            }
99            AssetStore::Session { session } => {
100                let root_dir = self.inner.store.root_dir(*session);
101                if root_dir.is_dir() {
102                    Ok(())
103                } else {
104                    Err(AssetError::InvalidStore(format!(
105                        "{session} is not a root session of this daemon"
106                    )))
107                }
108            }
109        }
110    }

126    /// `assets.search`: the session trees' own stores, found by listing
127    /// `session-assets/` and filtered by their root's project, and the
128    /// projects' stores, newest first. No session's log is read, and each
129    /// store's index comes from memory while its file is unchanged.
130    pub async fn search_assets(
131        &self,
132        params: AssetsSearchParams,
133    ) -> Result<AssetsSearchResult, EngineError> {
134        self.assets_blocking(move |engine, stores| {
135            let mut result = AssetsSearchResult::default();
136            if params.scope != Some(AssetScope::Project) {
137                for root in stores.session_roots()? {
138                    // A tree deleted by an older build, or not a root.
139                    let Some(meta) = engine.inner.store.root_meta(root) else {
140                        continue;
141                    };
142                    let project = meta
143                        .project
144                        .filter(|project| engine.project_exists(project));
145                    let wanted = match &params.project {
146                        None => true,
147                        Some(AssetProjectFilter::NoProject) => project.is_none(),
148                        Some(AssetProjectFilter::Project { project: wanted }) => {
149                            project.as_ref() == Some(wanted)
150                        }
151                    };
152                    if !wanted {
153                        continue;
154                    }
155                    let assets = stores.list(&AssetStore::Session { session: root })?;
156                    if assets.is_empty() {
157                        continue;
158                    }
159                    result.sessions.push(AssetSession {
160                        session: root,
161                        title: meta.title,
162                        project,
163                    });
164                    result.assets.extend(assets);
165                }
166            }
167            if params.scope != Some(AssetScope::Session) {
168                let kept = match &params.project {
169                    None => {
170                        let mut projects = engine.project_ids().into_iter().collect::<Vec<_>>();
171                        projects.sort();
172                        projects
173                            .into_iter()
174                            .map(|project| AssetStore::Project { project })
175                            .chain([AssetStore::Shared])
176                            .collect()
177                    }
178                    Some(AssetProjectFilter::NoProject) => vec![AssetStore::Shared],
179                    Some(AssetProjectFilter::Project { project }) => {
180                        let store = AssetStore::Project {
181                            project: project.clone(),
182                        };
183                        engine.check_client_store(&store)?;
184                        vec![store]
185                    }
186                };
187                for store in kept {
188                    result.assets.extend(stores.list(&store)?);
189                }
190            }
191            result
192                .assets
193                .sort_by_key(|asset| std::cmp::Reverse(asset.updated.at));
194            Ok(result)
195        })
196        .await
197    }

210    /// `asset.read`.
211    pub async fn read_asset(
212        &self,
213        params: AssetReadParams,
214    ) -> Result<AssetReadResult, EngineError> {
215        self.assets_blocking(move |engine, stores| {
216            engine.check_client_store(&params.store)?;
217            let length = params.length.min(cookie_agent_protocol::ASSET_CHUNK_BYTES);
218            let (bytes, asset) = stores.read(&params.store, params.asset, params.offset, length)?;
219            Ok(AssetReadResult {
220                data: base64::engine::general_purpose::STANDARD.encode(bytes),
221                total_bytes: asset.byte_length,
222                media_type: asset.media_type,
223                sha256: asset.sha256,
224            })
225        })
226        .await
227    }

229    /// `asset.thumbnail`: the thumbnail kept of the asset's content, or
230    /// one drawn now and kept.
231    pub async fn asset_thumbnail(
232        &self,
233        params: AssetThumbnailParams,
234    ) -> Result<AssetThumbnailResult, EngineError> {
235        let store = params.store.clone();
236        let source = self
237            .assets_blocking(move |engine, stores| {
238                engine.check_client_store(&params.store)?;
239                Ok(stores.thumbnail_source(&params.store, params.asset)?)
240            })
241            .await?;
242        let mut sha256 = source.asset.sha256.clone();
243        let made = match source.kept {
244            Some(kept) => Made::Picture(kept),
245            None => self.draw_thumbnail(store, source, &mut sha256).await,
246        };
247        let (picture, retry_later) = match made {
248            Made::Picture(picture) => (Some(picture), false),
249            Made::Cannot => (None, false),
250            Made::Unavailable => (None, true),
251        };
252        Ok(AssetThumbnailResult {
253            sha256,
254            thumbnail: picture.map(|picture| AssetThumbnail {
255                media_type: picture.format.media_type().to_owned(),
256                data: base64::engine::general_purpose::STANDARD.encode(&picture.bytes),
257                width: picture.width,
258                height: picture.height,
259            }),
260            retry_later,
261        })
262    }

264    /// Draws the thumbnail of `source`, an asset of `store` with none kept,
265    /// and keeps it. `sha256` becomes the content drawn, which another
266    /// write may have changed while this one waited its turn. A kind
267    /// without thumbnails, or content that did not draw before, is
268    /// [`Made::Cannot`].
269    async fn draw_thumbnail(
270        &self,
271        store: AssetStore,
272        source: ThumbnailSource,
273        sha256: &mut String,
274    ) -> Made {
275        let stores = Arc::clone(&self.inner.assets);
276        if Drawer::of(&source.asset.media_type).is_none() || stores.undrawable(sha256) {
277            return Made::Cannot;
278        }
279        let Some(_turn) = stores.draw_turn().await else {
280            return Made::Unavailable;
281        };
282        // Another request may have drawn it while this one waited.
283        let id = source.asset.id;
284        let Ok(source) = self
285            .assets_blocking({
286                let store = store.clone();
287                move |_, stores| Ok(stores.thumbnail_source(&store, id)?)
288            })
289            .await
290        else {
291            return Made::Unavailable;
292        };
293        sha256.clone_from(&source.asset.sha256);
294        if let Some(kept) = source.kept {
295            return Made::Picture(kept);
296        }
297        let Some(drawer) = Drawer::of(&source.asset.media_type) else {
298            return Made::Cannot;
299        };
300        if stores.undrawable(sha256) {
301            return Made::Cannot;
302        }
303        let made = thumbnail::make(source.blob, drawer).await;
304        match &made {
305            Made::Picture(picture) => {
306                let kept = picture.clone();
307                let drawn = sha256.clone();
308                if let Err(error) = self
309                    .assets_blocking(move |_, stores| {
310                        Ok(stores.keep_thumbnail(&store, &drawn, &kept)?)
311                    })
312                    .await
313                {
314                    eprintln!("cookie-agent: a thumbnail was not kept: {error}");
315                }
316            }
317            Made::Cannot => stores.mark_undrawable(sha256),
318            Made::Unavailable => {}
319        }
320        made
321    }

337    /// `asset.delete`.
338    pub async fn delete_asset(
339        &self,
340        params: AssetDeleteParams,
341    ) -> Result<AssetDeleteResult, EngineError> {
342        self.assets_blocking(move |engine, stores| {
343            engine.check_client_store(&params.store)?;
344            stores.delete(&params.store, params.asset)?;
345            engine.forget_shares_of_asset(params.asset);
346            Ok(AssetDeleteResult {})
347        })
348        .await
349    }

351    /// Every asset `session`'s tools can reach, the tree's first and then
352    /// the project's, each newest first. Blocking.
353    pub(crate) fn session_assets(
354        &self,
355        stores: &SessionAssetStores,
356    ) -> Result<Vec<Asset>, AssetError> {
357        let mut assets = self.inner.assets.list(&stores.session)?;
358        assets.extend(self.inner.assets.list(&stores.project)?);
359        Ok(assets)
360    }
