1//! Attaching and detaching machines from a session's own run
2//! (`docs/site/specs/machines-on-demand.md` sections 3 and 4):
3//! `attach_machine` and `detach_machine` record the usual
4//! `runtime_attached` and `runtime_detached` events while the session runs,
5//! and no machine may carry two attachments of the session that offer a
6//! tool of the same name.
7

10use cookie_agent_protocol::{AgentMdEntry, MachineName, RuntimeAttachment, RuntimeId, SessionId};
11use cookie_agent_runtime_api::SessionToolContext;

13use super::{Engine, EngineError, attachments::attached_event};
14use crate::runtimes::RuntimeInstance;

16/// What `attach_machine` tells the model at once, since the runtimes block
17/// is frozen for the run.
18pub(crate) struct AttachedMachine {
19    pub(crate) attachment: RuntimeAttachment,
20    pub(crate) machine: Option<MachineName>,
21    pub(crate) platform: Option<String>,
22    pub(crate) home: Option<String>,
23    /// The workspace's `AGENTS.md` files, as a run would load them.
24    pub(crate) agent_md: Vec<AgentMdEntry>,
25}
26
27impl Engine {
28    /// Attaches `capability` of `runtime_id` (at `workspace` when it takes
29    /// one) to `session` from the session's own run. The same checks as
30    /// [`Engine::attach_runtime`] apply, except that the session may be
31    /// running.
32    pub(crate) async fn attach_from_run(
33        &self,
34        session: SessionId,
35        runtime_id: &str,
36        capability: &str,
37        workspace: Option<PathBuf>,
38        preset: Option<&str>,
39    ) -> Result<AttachedMachine, EngineError> {
40        self.ensure_session_owned(session)?;
41        self.refuse_second_acp_runtime(session, runtime_id)?;
42        let instance = self
43            .ensure_runtime(runtime_id, Some(capability), workspace.as_deref())
44            .await?;
45        if let Some(connection) = instance.connection() {
46            connection.retry_failed_mcp();
47        }
48        self.refuse_overlap(session, &instance)?;
49        let attachment = self
50            .on_actor(session, move |engine| {
51                let projection = engine.inner.store.get(session)?;
52                let mut attachments = engine.attachments_of(&projection);
53                let attachment = super::attachments::push_instance(&mut attachments, &instance)?;
54                engine.append_direct(
55                    session,
56                    None,
57                    super::event_origin("tool:attach_machine"),
58                    attached_event(&attachment),
59                )?;
60                Ok(attachment)
61            })
62            .await?;
63        let agent_md = match self.inner.runtimes.instance(&attachment) {
64            Some(instance) if attachment.workspace.is_some() => instance
65                .agent_md(preset)
66                .await
67                .ok()
68                .flatten()
69                .map(|context| context.entries)
70                .unwrap_or_default(),
71            _ => Vec::new(),
72        };
73        let listing = self
74            .runtime_templates()
75            .into_iter()
76            .find(|template| template.runtime_id == runtime_id)
77            .map(|template| template.listing);
78        Ok(AttachedMachine {
79            machine: self.runtime_machine(runtime_id),
80            platform: self
81                .inner
82                .runtimes
83                .instance(&attachment)
84                .and_then(|instance| {
85                    instance
86                        .info()
87                        .workspace()
88                        .and_then(|workspace| workspace.platform().map(str::to_owned))
89                })
90                .or_else(|| {
91                    listing
92                        .as_ref()
93                        .and_then(|listing| listing.platform.clone())
94                }),
95            home: listing.and_then(|listing| listing.home),
96            agent_md,
97            attachment,
98        })
99    }

101    /// Detaches the attachment of `capability` of `runtime_id` from
102    /// `session`, from the session's own run. Refused while one of the
103    /// session's background calls runs on it.
104    pub(crate) async fn detach_from_run(
105        &self,
106        session: SessionId,
107        runtime_id: &str,
108        capability: &str,
109    ) -> Result<RuntimeAttachment, EngineError> {
110        self.ensure_session_owned(session)?;
111        let projection = self.inner.store.get(session)?;
112        let attachment = self
113            .attachments_of(&projection)
114            .into_iter()
115            .find(|attachment| {
116                attachment.runtime_id.as_str() == runtime_id
117                    && attachment.capability.as_str() == capability
118            })
119            .ok_or_else(|| {
120                EngineError::Runtime(format!(
121                    "this session has no attachment of `{runtime_id}/{capability}`"
122                ))
123            })?;
124        if attachment.runtime_id.as_str() == RuntimeId::GLOBAL {
125            return Err(EngineError::Runtime(
126                "`global` cannot be detached".to_owned(),
127            ));
128        }
129        let running = self.background_calls_on(session, &attachment);
130        if !running.is_empty() {
131            return Err(EngineError::Runtime(format!(
132                "calls still run on `{runtime_id}/{capability}`: {}; stop them with \
133                 stop_background first",
134                running.join(", ")
135            )));
136        }
137        let name = attachment.name.clone();
138        self.on_actor(session, move |engine| {
139            engine.append_direct(
140                session,
141                None,
142                super::event_origin("tool:detach_machine"),
143                super::Event::RuntimeDetached { name },
144            )
145        })
146        .await?;
147        Ok(attachment)
148    }

150    /// Whether `agent` of `preset` is an ACP agent.
151    pub(crate) fn selects_acp_agent(
152        &self,
153        agent: &cookie_agent_protocol::AgentId,
154        preset: Option<&str>,
155    ) -> bool {
156        self.current_runtime()
157            .agents_for_preset(preset)
158            .ok()
159            .and_then(|agents| agents.get(agent).map(|agent| agent.acp.is_some()))
160            .unwrap_or(false)
161    }

163    /// The preset of run `run`, while it runs.
164    pub(crate) fn run_preset(&self, run: cookie_agent_protocol::RunId) -> Option<String> {
165        self.inner
166            .sessions
167            .active
168            .lock()
169            .unwrap_or_else(|poisoned| poisoned.into_inner())
170            .get(&run)
171            .and_then(|active| active.policy.preset.clone())
172    }

174    /// `session`'s attachment of `target`'s runtime capability, if any.
175    pub(crate) fn attachment_of_capability(
176        &self,
177        session: SessionId,
178        target: &crate::DelegatedCapability,
179    ) -> Option<RuntimeAttachment> {
180        let projection = self.inner.store.get(session).ok()?;
181        self.attachments_of(&projection)
182            .into_iter()
183            .find(|attachment| {
184                attachment.runtime_id == target.runtime_id
185                    && attachment.capability == target.capability
186            })
187    }

189    /// `session`'s background calls still running on `attachment`: a call
190    /// of one of its tools, by its bare or its `<attachment>_` name.
191    fn background_calls_on(
192        &self,
193        session: SessionId,
194        attachment: &RuntimeAttachment,
195    ) -> Vec<String> {
196        let Some(instance) = self.inner.runtimes.instance(attachment) else {
197            return Vec::new();
198        };
199        let bare = tool_names(&instance, &SessionToolContext::new(session));
200        let tools = bare
201            .iter()
202            .map(|tool| format!("{}_{tool}", attachment.name))
203            .chain(bare.iter().cloned())
204            .collect();
205        self.inner.background.running_calls_of(session, &tools)
206    }

208    /// A session runs its ACP agent on the one attached runtime that offers
209    /// it, so a second such runtime is refused.
210    pub(crate) fn refuse_second_acp_runtime(
211        &self,
212        session: SessionId,
213        runtime_id: &str,
214    ) -> Result<(), EngineError> {
215        let Some(offers) = self.session_acp_offers(session)? else {
216            return Ok(());
217        };
218        let attached = self.session_runtime_ids(session)?;
219        let offering = offers
220            .runtimes()
221            .into_iter()
222            .filter(|offering| attached.contains(offering))
223            .collect::<Vec<_>>();
224        if offers
225            .runtimes()
226            .iter()
227            .any(|offering| offering.as_str() == runtime_id)
228            && let Some(other) = offering.iter().find(|other| other.as_str() != runtime_id)
229        {
230            return Err(EngineError::Runtime(format!(
231                "runtime `{runtime_id}` offers ACP agent `{}`, which this session runs on \
232                 runtime `{other}`; a session runs it on one runtime, so detach `{other}` \
233                 first",
234                offers.id
235            )));
236        }
237        Ok(())
238    }

240    /// Refuses `instance` when one of its tools has the name of a tool that
241    /// another of `session`'s attachments on the same machine offers
242    /// (machines on demand, section 4). A runtime the daemon cannot place
243    /// is its own machine.
244    pub(crate) fn refuse_overlap(
245        &self,
246        session: SessionId,
247        instance: &Arc<RuntimeInstance>,
248    ) -> Result<(), EngineError> {
249        let projection = self.inner.store.get(session)?;
250        self.refuse_overlap_with(
251            &self.attachments_of(&projection),
252            instance,
253            &SessionToolContext::new(session),
254        )
255    }
256
257    /// [`Engine::refuse_overlap`] against the attachments `existing`, for a
258    /// session that may not exist yet (its creation, a subagent's
259    /// capabilities): tool lists are read with `context`.
260    pub(crate) fn refuse_overlap_with(
261        &self,
262        existing: &[RuntimeAttachment],
263        instance: &Arc<RuntimeInstance>,
264        context: &SessionToolContext,
265    ) -> Result<(), EngineError> {
266        let runtime_id = instance.info().id().to_owned();
267        if runtime_id == RuntimeId::GLOBAL {
268            return Ok(());
269        }
270        let Some(machine) = self.runtime_machine(&runtime_id) else {
271            return Ok(());
272        };
273        let offered = tool_names(instance, context);
274        if offered.is_empty() {
275            return Ok(());
276        }
277        for attachment in existing {
278            if attachment.runtime_id.as_str() == RuntimeId::GLOBAL
279                || self
280                    .runtime_machine(attachment.runtime_id.as_str())
281                    .as_ref()
282                    != Some(&machine)
283            {
284                continue;
285            }
286            let Some(other) = self.inner.runtimes.instance(attachment) else {
287                continue;
288            };
289            if Arc::ptr_eq(&other, instance) {
290                // Attaching the instance twice is refused on its own.
291                continue;
292            }
293            let shared = tool_names(&other, context)
294                .intersection(&offered)
295                .cloned()
296                .collect::<Vec<_>>();
297            if !shared.is_empty() {
298                return Err(EngineError::Runtime(overlap_refusal(
299                    attachment, &machine, &shared, instance,
300                )));
301            }
302        }
303        Ok(())
304    }
305}

307/// The names of the tools `instance` offers a session.
308fn tool_names(instance: &RuntimeInstance, context: &SessionToolContext) -> BTreeSet<String> {
309    instance
310        .sources()
311        .1
312        .iter()
313        .filter_map(|provider| provider.tools_for_session(context).ok())
314        .flatten()
315        .map(|tool| tool.name)
316        .collect()
317}
318
319fn overlap_refusal(
320    existing: &RuntimeAttachment,
321    machine: &MachineName,
322    shared: &[String],
323    instance: &RuntimeInstance,
324) -> String {
325    let tools = shared
326        .iter()
327        .map(|tool| format!("`{tool}`"))
328        .collect::<Vec<_>>()
329        .join(", ");
330    let mut message = format!(
331        "machine `{machine}` already has `{}/{}` attached{}, which offers {tools}: one machine \
332         offers each tool once. Use absolute paths to reach another directory there, or detach \
333         it first",
334        existing.runtime_id,
335        existing.capability,
336        existing
337            .workspace
338            .as_deref()
339            .map(|workspace| format!(" at {workspace}"))
340            .unwrap_or_default(),
341    );
342    if instance.info().id() != existing.runtime_id.as_str() {
343        message.push_str(&format!(
344            ". If runtime `{}` is a container or another environment with files of its own, \
345             give it a machine of its own with `cookie machine unassign {}`",
346            instance.info().id(),
347            instance.info().id()
348        ));
349    }
350    message
351}
