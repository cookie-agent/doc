1//! Attaching and detaching machines from a session's own run
2//! (`docs/site/specs/machines-on-demand.md` sections 3 and 4):
3//! `attach_machine` and `detach_machine` record the usual
4//! `runtime_attached` and `runtime_detached` events while the session runs,
5//! and no machine may carry two attachments of the session that offer a
6//! tool of the same name.
7

18/// What `attach_machine` tells the model at once, since the runtimes block
19/// is frozen for the run.
20pub(crate) struct AttachedMachine {
21    pub(crate) attachment: RuntimeAttachment,
22    pub(crate) machine: Option<MachineName>,
23    pub(crate) platform: Option<String>,
24    pub(crate) home: Option<String>,
25    /// The workspace's `AGENTS.md` files, as a run would load them.
26    pub(crate) agent_md: Vec<AgentMdEntry>,
27    /// The tools the attachment brings, sorted by name; `None` when the
28    /// attaching run is gone or they cannot be listed.
29    pub(crate) tools: Option<Vec<AttachedTool>>,
30}

32/// One tool an attachment brings to a session, as its run publishes it.
33#[derive(Clone, Debug, PartialEq)]
34pub(crate) struct AttachedTool {
35    /// The name the model calls it by.
36    pub(crate) name: String,
37    /// The `machine` a call names to run it on the attachment's machine,
38    /// when the tool is routed (machines on demand, section 5).
39    pub(crate) machine: Option<String>,
40    /// The group it is announced under while deferred: the model finds it
41    /// with `tool_search` before calling it; `None` once it is declared.
42    pub(crate) deferred: Option<String>,
43    /// The first line of its description.
44    pub(crate) summary: String,
45}

47impl Engine {
48    /// Attaches `capability` of `runtime_id` (at `workspace` when it takes
49    /// one) to `session` from the session's own run. The same checks as
50    /// [`Engine::attach_runtime`] apply, except that the session may be
51    /// running.
52    pub(crate) async fn attach_from_run(
53        &self,
54        session: SessionId,
55        runtime_id: &str,
56        capability: &str,
57        workspace: Option<PathBuf>,
58        policy: Option<&FrozenRunPolicy>,
59    ) -> Result<AttachedMachine, EngineError> {
60        let preset = policy.and_then(|policy| policy.preset.as_deref());
61        self.ensure_session_owned(session)?;
62        self.refuse_second_acp_runtime(session, runtime_id)?;
63        let instance = self
64            .ensure_runtime(runtime_id, Some(capability), workspace.as_deref())
65            .await?;
66        if let Some(connection) = instance.connection() {
67            connection.retry_failed_mcp();
68        }
69        let runtime_id_owned = runtime_id.to_owned();
70        let attachment = self
71            .on_actor(session, move |engine| {
72                // Checked on the actor, so of two attaches under way only
73                // the first can pass.
74                engine.refuse_second_acp_runtime(session, runtime_id_owned.as_str())?;
75                let projection = engine.inner.store.get(session)?;
76                let mut attachments = engine.attachments_of(&projection);
77                engine.refuse_overlap_with(
78                    &attachments,
79                    &instance,
80                    &SessionToolContext::new(session),
81                )?;
82                let attachment =
83                    crate::runtime::attachments::push_instance(&mut attachments, &instance)?;
84                engine.append_direct(
85                    session,
86                    None,
87                    crate::runtime::event_origin("engine:machines"),
88                    attached_event(&attachment),
89                )?;
90                Ok(attachment)
91            })
92            .await?;
93        let agent_md = match self.inner.runtimes.instance(&attachment) {
94            Some(instance) if attachment.workspace.is_some() => instance
95                .agent_md(preset)
96                .await
97                .ok()
98                .flatten()
99                .map(|context| context.entries)
100                .unwrap_or_default(),
101            _ => Vec::new(),
102        };
103        let listing = self
104            .runtime_templates()
105            .into_iter()
106            .find(|template| template.runtime_id == runtime_id)
107            .map(|template| template.listing);
108        Ok(AttachedMachine {
109            machine: self.runtime_machine(runtime_id),
110            platform: self
111                .inner
112                .runtimes
113                .instance(&attachment)
114                .and_then(|instance| {
115                    instance
116                        .info()
117                        .workspace()
118                        .and_then(|workspace| workspace.platform().map(str::to_owned))
119                })
120                .or_else(|| {
121                    listing
122                        .as_ref()
123                        .and_then(|listing| listing.platform.clone())
124                }),
125            home: listing.and_then(|listing| listing.home),
126            agent_md,
127            // The attachment stands even when its tools cannot be listed.
128            tools: policy.and_then(|policy| {
129                self.attachment_tools(session, policy, &attachment.name)
130                    .ok()
131            }),
132            attachment,
133        })
134    }

136    /// The tools `attachment` brings to `session` as a run of `policy`
137    /// publishes them now: by their published names, with the `machine` that
138    /// routes a call there and the group of those still deferred, the tools
139    /// the agent hides from its ACP agent left out.
140    pub(crate) fn attachment_tools(
141        &self,
142        session: SessionId,
143        policy: &FrozenRunPolicy,
144        attachment: &AttachmentName,
145    ) -> Result<Vec<AttachedTool>, EngineError> {
146        let composition = self.compose_for_policy(session, policy)?;
147        let (entries, _) = self.route_tools(session, composition.tools)?;
148        let hide = self
149            .current_runtime()
150            .agents
151            .get(&policy.agent.agent)
152            .and_then(|agent| agent.document.frontmatter.acp.clone())
153            .map(|acp| acp.hidden_tools())
154            .unwrap_or_default();
155        let defers = self.defers_tools(policy);
156        let fold = self.inner.store.get(session)?.log.deferral();
157        let config = self.inner.config.load();
158        let mut tools = entries
159            .into_iter()
160            .filter(|(tool, _)| {
161                !hide
162                    .iter()
163                    .any(|pattern| cookie_agent_config::simple_wildcard_match(pattern, &tool.name))
164            })
165            .filter_map(|(tool, routes)| {
166                // A routed tool runs here through its member of this machine.
167                let (machine, source, spec) = match routes {
168                    Some(routes) => {
169                        let (machine, member) =
170                            routes.machines.into_iter().find(|(_, member)| {
171                                member.source.attachment.as_ref() == Some(attachment)
172                            })?;
173                        (Some(machine), member.source, member.spec)
174                    }
175                    None if tool.source.attachment.as_ref() == Some(attachment) => {
176                        (None, tool.source, tool.spec)
177                    }
178                    None => return None,
179                };
180                // A tool outside the round is deferred until a result loads
181                // it: the policy defers it, or it arrived mid-round
182                // (deferred tools, section 8.2).
183                let deferred = defers
184                    && !fold.loaded.contains(&tool.name)
185                    && match &fold.round {
186                        Some(round) => !round.tools.iter().any(|eager| eager.name == tool.name),
187                        None => {
188                            !fold.folded.contains(&tool.name)
189                                && crate::deferred_tools::policy_defers(
190                                    &source,
191                                    &spec.name,
192                                    &config.runtime.deferred_tools,
193                                )
194                        }
195                    };
196                let deferred =
197                    deferred.then(|| crate::deferred_tools::group_of(&source, &spec.name).name);
198                Some(AttachedTool {
199                    name: tool.name,
200                    machine,
201                    deferred,
202                    summary: spec
203                        .description
204                        .lines()
205                        .next()
206                        .unwrap_or_default()
207                        .trim()
208                        .to_owned(),
209                })
210            })
211            .collect::<Vec<_>>();
212        tools.sort_by(|left, right| left.name.cmp(&right.name));
213        Ok(tools)
214    }

216    /// Detaches the attachment of `capability` of `runtime_id` from
217    /// `session`, from the session's own run. Refused while one of the
218    /// session's background calls runs on it.
219    pub(crate) async fn detach_from_run(
220        &self,
221        session: SessionId,
222        runtime_id: &str,
223        capability: &str,
224    ) -> Result<RuntimeAttachment, EngineError> {
225        self.ensure_session_owned(session)?;
226        let projection = self.inner.store.get(session)?;
227        let attachment = self
228            .attachments_of(&projection)
229            .into_iter()
230            .find(|attachment| {
231                attachment.runtime_id.as_str() == runtime_id
232                    && attachment.capability.as_str() == capability
233            })
234            .ok_or_else(|| {
235                EngineError::Runtime(format!(
236                    "this session has no attachment of `{runtime_id}/{capability}`"
237                ))
238            })?;
239        if attachment.runtime_id.as_str() == RuntimeId::GLOBAL {
240            return Err(EngineError::Runtime(
241                "`global` cannot be detached".to_owned(),
242            ));
243        }
244        let running = self.background_calls_on(session, &attachment);
245        if !running.is_empty() {
246            return Err(EngineError::Runtime(format!(
247                "calls still run on `{runtime_id}/{capability}`: {}; stop them with \
248                 stop_background first",
249                running.join(", ")
250            )));
251        }
252        let name = attachment.name.clone();
253        self.on_actor(session, move |engine| {
254            engine.append_direct(
255                session,
256                None,
257                crate::runtime::event_origin("engine:machines"),
258                crate::runtime::Event::RuntimeDetached { name },
259            )
260        })
261        .await?;
262        Ok(attachment)
263    }

265    /// Whether `agent` of `preset` is an ACP agent.
266    pub(crate) fn selects_acp_agent(
267        &self,
268        agent: &cookie_agent_protocol::AgentId,
269        preset: Option<&str>,
270    ) -> bool {
271        self.current_runtime()
272            .agents_for_preset(preset)
273            .ok()
274            .and_then(|agents| agents.get(agent).map(|agent| agent.acp.is_some()))
275            .unwrap_or(false)
276    }

278    /// The policy of run `run`, while it runs.
279    pub(crate) fn run_policy(
280        &self,
281        run: cookie_agent_protocol::RunId,
282    ) -> Option<Arc<FrozenRunPolicy>> {
283        self.inner
284            .sessions
285            .active
286            .lock()
287            .unwrap_or_else(|poisoned| poisoned.into_inner())
288            .get(&run)
289            .map(|active| Arc::clone(&active.policy))

292    /// `session`'s attachment of `target`'s runtime capability, if any.
293    pub(crate) fn attachment_of_capability(
294        &self,
295        session: SessionId,
296        target: &crate::DelegatedCapability,
297    ) -> Option<RuntimeAttachment> {
298        let projection = self.inner.store.get(session).ok()?;
299        self.attachments_of(&projection)
300            .into_iter()
301            .find(|attachment| {
302                attachment.runtime_id == target.runtime_id
303                    && attachment.capability == target.capability
304            })
305    }

307    /// `session`'s background calls still running on `attachment`: a call
308    /// of one of its tools, by its bare or its `<attachment>_` name.
309    fn background_calls_on(
310        &self,
311        session: SessionId,
312        attachment: &RuntimeAttachment,
313    ) -> Vec<String> {
314        let Some(instance) = self.inner.runtimes.instance(attachment) else {
315            return Vec::new();
316        };
317        let bare = tool_names(&instance, &SessionToolContext::new(session));
318        let tools = bare
319            .iter()
320            .map(|tool| format!("{}_{tool}", attachment.name))
321            .chain(bare.iter().cloned())
322            .collect();
323        self.inner.background.running_calls_of(session, &tools)
324    }

326    /// A session runs its ACP agent on the one attached runtime that offers
327    /// it, so a second such runtime is refused.
328    pub(crate) fn refuse_second_acp_runtime(
329        &self,
330        session: SessionId,
331        runtime_id: &str,
332    ) -> Result<(), EngineError> {
333        let Some(offers) = self.session_acp_offers(session)? else {
334            return Ok(());
335        };
336        let attached = self.session_runtime_ids(session)?;
337        let offering = offers
338            .runtimes()
339            .into_iter()
340            .filter(|offering| attached.contains(offering))
341            .collect::<Vec<_>>();
342        if offers
343            .runtimes()
344            .iter()
345            .any(|offering| offering.as_str() == runtime_id)
346            && let Some(other) = offering.iter().find(|other| other.as_str() != runtime_id)
347        {
348            return Err(EngineError::Runtime(format!(
349                "runtime `{runtime_id}` offers ACP agent `{}`, which this session runs on \
350                 runtime `{other}`; a session runs it on one runtime, so detach `{other}` \
351                 first",
352                offers.id
353            )));
354        }
355        Ok(())
356    }

358    /// Refuses `instance` when one of its tools has the name of a tool that
359    /// another attachment among `existing` on the same machine offers
360    /// (machines on demand, section 4); a runtime the daemon cannot place is
361    /// its own machine. The session may not exist yet (its creation, a
362    /// subagent's capabilities): tool lists are read with `context`.
363    pub(crate) fn refuse_overlap_with(
364        &self,
365        existing: &[RuntimeAttachment],
366        instance: &Arc<RuntimeInstance>,
367        context: &SessionToolContext,
368    ) -> Result<(), EngineError> {
369        let runtime_id = instance.info().id().to_owned();
370        if runtime_id == RuntimeId::GLOBAL {
371            return Ok(());
372        }
373        let Some(machine) = self.runtime_machine(&runtime_id) else {
374            return Ok(());
375        };
376        let offered = tool_names(instance, context);
377        if offered.is_empty() {
378            return Ok(());
379        }
380        for attachment in existing {
381            if attachment.runtime_id.as_str() == RuntimeId::GLOBAL
382                || self
383                    .runtime_machine(attachment.runtime_id.as_str())
384                    .as_ref()
385                    != Some(&machine)
386            {
387                continue;
388            }
389            let Some(other) = self.inner.runtimes.instance(attachment) else {
390                continue;
391            };
392            if Arc::ptr_eq(&other, instance) {
393                // Attaching the instance twice is refused on its own.
394                continue;
395            }
396            let shared = tool_names(&other, context)
397                .intersection(&offered)
398                .cloned()
399                .collect::<Vec<_>>();
400            if !shared.is_empty() {
401                return Err(EngineError::Runtime(overlap_refusal(
402                    attachment, &machine, &shared, instance,
403                )));
404            }
405        }
406        Ok(())
407    }

410/// The names of the tools `instance` offers a session.
411fn tool_names(instance: &RuntimeInstance, context: &SessionToolContext) -> BTreeSet<String> {
412    instance
413        .sources()
414        .1
415        .iter()
416        .filter_map(|provider| provider.tools_for_session(context).ok())
417        .flatten()
418        .map(|tool| tool.name)
419        .collect()
420}
421
422fn overlap_refusal(
423    existing: &RuntimeAttachment,
424    machine: &MachineName,
425    shared: &[String],
426    instance: &RuntimeInstance,
427) -> String {
428    let tools = shared
429        .iter()
430        .map(|tool| format!("`{tool}`"))
431        .collect::<Vec<_>>()
432        .join(", ");
433    let mut message = format!(
434        "machine `{machine}` already has `{}/{}` attached{}, which offers {tools}: one machine \
435         offers each tool once. Use absolute paths to reach another directory there, or detach \
436         it first",
437        existing.runtime_id,
438        existing.capability,
439        existing
440            .workspace
441            .as_deref()
442            .map(|workspace| format!(" at {workspace}"))
443            .unwrap_or_default(),
444    );
445    if instance.info().id() != existing.runtime_id.as_str() {
446        message.push_str(&format!(
447            ". If runtime `{}` is a container or another environment with files of its own, \
448             give it a machine of its own with `cookie machine unassign {}`",
449            instance.info().id(),
450            instance.info().id()
451        ));
452    }
453    message
