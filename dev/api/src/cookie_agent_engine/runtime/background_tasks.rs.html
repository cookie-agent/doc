1//! Background jobs a tool call leaves running after it returns.
2//!
3//! A tool reserves a job while its call executes, returns its own result, and
4//! hands the engine a future that resolves when the job ends. The engine owns
5//! the rest: the job's output capture, a `Tool` producer registration, and the
6//! completion message that wakes the session. A job only reports back once the
7//! call that started it is durably committed as completed; a call that never
8//! commits, a revert past it, or deleting its session kills the job instead.
9//! Engine shutdown drops the job future, which kills whatever it owns.
10

72/// What a finished background job reports to the agent. The engine renders
73/// `<element attributes…>`, the body, and the job's output preview (truncated
74/// like any tool result), then the closing tag.
75pub struct BackgroundReport {
76    /// One-line summary shown on the notification row.
77    pub summary: String,
78    /// Element name wrapping the agent-visible report, such as `background_bash`.
79    pub element: String,
80    /// Attributes on the opening tag, in order.
81    pub attributes: Vec<(String, String)>,
82    /// Report lines placed before the output preview.
83    pub body: String,
84}

86/// A reserved background job. [`BackgroundJob::progress`] captures the job's
87/// output like a foreground call's; [`BackgroundJob::launch`] hands over the
88/// future that runs it. Dropping the job unlaunched releases the reservation.
89pub struct BackgroundJob {
90    progress: ProgressSink,
91    launch: Launch,
92}

94struct Launch {
95    engine: Engine,
96    session: SessionId,
97    tool_call_id: ToolCallId,
98    producer_id: ProducerId,
99    capture: OutputCapture,
100    progress: mpsc::Receiver<ToolProgress>,
101    kill: CancellationToken,
102    arm: oneshot::Receiver<()>,
103}
104
105impl Drop for Launch {
106    fn drop(&mut self) {
107        self.engine
108            .release_background_job(self.session, self.tool_call_id, self.producer_id);
109        self.capture.release_publication();
110    }
111}

113impl BackgroundJob {
114    pub(crate) async fn reserve(
115        capability: &BackgroundCapability,
116        session: SessionId,
117    ) -> Result<Self, ToolError> {
118        let inner = capability
119            .inner
120            .upgrade()
121            .ok_or_else(|| ToolError::execution("engine is shutting down"))?;
122        let engine = Engine { inner };
123        let tool_call_id = capability.tool_call_id;
124        let producer_id = engine
125            .register_producer(session, tool_authority(tool_call_id))
126            .await
127            .map_err(|error| ToolError::execution(error.to_string()))?;
128        let (sender, progress) = mpsc::channel(64);
129        let (arm_sender, arm) = oneshot::channel();
130        let kill = CancellationToken::new();
131        engine
132            .inner
133            .background
134            .jobs
135            .lock()
136            .unwrap_or_else(|p| p.into_inner())
137            .insert(
138                tool_call_id,
139                Job {
140                    session,
141                    kill: kill.clone(),
142                    arm: Some(arm_sender),
143                },
144            );
145        let capture = match OutputCapture::with_limits(
146            engine.inner.artifacts.clone(),
147            session,
148            ToolOutputDeclaration::Named {
149                streams: vec!["stdout".into(), "stderr".into()],
150            },
151            capability.limits,
152        )
153        .await
154        {
155            Ok(capture) => capture,
156            Err(error) => {
157                engine.release_background_job(session, tool_call_id, producer_id);
158                return Err(error);
159            }
160        };
161        Ok(Self {
162            progress: ProgressSink::with_capture(
163                sender,
164                OutputHub::new(tool_call_id, 0),
165                capture.clone(),
166            ),
167            launch: Launch {
168                engine,
169                session,
170                tool_call_id,
171                producer_id,
172                capture,
173                progress,
174                kill,
175                arm,
176            },
177        })
178    }

180    /// The sink the job writes its output to.
181    #[must_use]
182    pub fn progress(&self) -> ProgressSink {
183        self.progress.clone()
184    }

186    /// Runs `job` detached from the call. Its report is delivered once the
187    /// call commits; the job is dropped (killing what it owns) if the call
188    /// never commits, is reverted, or its session is deleted.
189    pub fn launch(
190        self,
191        job: impl Future<Output = BackgroundReport> + Send + 'static,
192    ) -> Result<(), ToolError> {
193        let Self { progress, launch } = self;
194        drop(progress);
195        let engine = launch.engine.clone();
196        let runtime = engine
197            .inner
198            .runtime
199            .clone()
200            .or_else(|| tokio::runtime::Handle::try_current().ok())
201            .ok_or_else(|| ToolError::execution("engine is shutting down"))?;
202        if engine.spawn_admission_task(&runtime, launch.run(job)) {
203            Ok(())
204        } else {
205            Err(ToolError::execution("engine is shutting down"))
206        }
207    }
208}

210impl Launch {
211    async fn run(mut self, job: impl Future<Output = BackgroundReport> + Send) {
212        let report = {
213            let mut job = std::pin::pin!(job);
214            loop {
215                tokio::select! {
216                    report = &mut job => break report,
217                    () = self.kill.cancelled() => return,
218                    // Display previews need a reader; the capture already has the bytes.
219                    Some(_) = self.progress.recv() => {}
220                }
221            }
222        };
223        tokio::select! {
224            armed = &mut self.arm => if armed.is_err() { return },
225            () = self.kill.cancelled() => return,
226        }
227        let (output, retained) = match self
228            .capture
229            .finish(ToolCompletion::streamed(job_result()), false)
230            .await
231        {
232            Ok(result) => (Some(result.output), result.retained_output),
233            Err(_) => (None, None),
234        };
235        let body = render_notification(&report, output.as_deref());
236        loop {
237            let (summary, body, retained) =
238                (report.summary.clone(), body.clone(), retained.clone());
239            let (session, tool_call_id, producer_id) =
240                (self.session, self.tool_call_id, self.producer_id);
241            let committed = self
242                .engine
243                .on_actor(session, move |engine| {
244                    engine.commit_background_completion_direct(
245                        session,
246                        tool_call_id,
247                        producer_id,
248                        &summary,
249                        body,
250                        retained,
251                    )
252                })
253                .await;
254            if committed.is_ok()
255                || self.engine.admission_tasks_closing()
256                || self.kill.is_cancelled()
257            {
258                break;
259            }
260            tokio::time::sleep(Duration::from_millis(100)).await;
261        }
262    }
263}

286/// Wraps a job's report and output preview in its element.
287fn render_notification(report: &BackgroundReport, output: Option<&str>) -> String {
288    let mut body = format!("<{}", report.element);
289    for (name, value) in &report.attributes {
290        let value = value
291            .replace('&', "&amp;")
292            .replace('"', "&quot;")
293            .replace('<', "&lt;");
294        body.push_str(&format!(" {name}=\"{value}\""));
295    }
296    body.push_str(">\n");
297    body.push_str(report.body.trim_end());
298    if let Some(output) = output
299        .map(str::trim_end)
300        .filter(|output| !output.is_empty())
301    {
302        body.push('\n');
303        body.push_str(output);
304    }
305    body.push_str(&format!("\n</{}>", report.element));
306    body
307}

309/// Whether a termination is the starting call's own committed completion,
310/// the only state in which its job may report back.
311pub(super) fn completed_start(termination: &ToolCallTermination) -> bool {
312    termination.result.is_some()
313        && match termination.outcome {
314            ToolTerminationOutcome::Completed => true,
315            ToolTerminationOutcome::Cancelled => termination
316                .error
317                .as_ref()
318                .is_some_and(|error| error.code.as_str() == super::CANCELLED_AFTER_COMPLETION),
319            _ => false,
320        }
321}

323impl Engine {
324    fn admission_tasks_closing(&self) -> bool {
325        self.inner
326            .delegation
327            .admission_tasks_closing
328            .load(std::sync::atomic::Ordering::Acquire)
329    }
330
331    /// Settles the job of a just-appended termination: its committed
332    /// completion arms it, anything else (including a failed append) kills it.
333    pub(super) fn settle_background_job(&self, tool_call_id: ToolCallId, committed: bool) {
334        let mut jobs = self
335            .inner
336            .background
337            .jobs
338            .lock()
339            .unwrap_or_else(|p| p.into_inner());
340        let Some(job) = jobs.get_mut(&tool_call_id) else {
341            return;
342        };
343        if committed {
344            if let Some(arm) = job.arm.take() {
345                let _ = arm.send(());
346            }
347        } else {
348            job.kill.cancel();
349        }
350    }
351
352    /// Kills the jobs whose starting call no longer survives in `session`'s log.
353    pub(super) fn kill_reverted_background_jobs(
354        &self,
355        session: SessionId,
356    ) -> Result<(), EngineError> {
357        let events = self.inner.store.log(session)?.event_snapshot();
358        let surviving: HashSet<_> = events
359            .iter()
360            .filter_map(|event| match &event.payload {
361                super::Event::ToolCallTerminated { termination }
362                    if completed_start(termination) =>
363                {
364                    Some(termination.tool_call_id)
365                }
366                _ => None,
367            })
368            .collect();
369        for (tool_call_id, job) in self
370            .inner
371            .background
372            .jobs
373            .lock()
374            .unwrap_or_else(|p| p.into_inner())
375            .iter()
376        {
377            if job.session == session && !surviving.contains(tool_call_id) {
378                job.kill.cancel();
379            }
380        }
381        Ok(())
382    }
383
384    /// Kills every job started in a deleted session.
385    pub(super) fn kill_deleted_background_jobs(&self, deleted: &HashSet<SessionId>) {
386        for job in self
387            .inner
388            .background
389            .jobs
390            .lock()
391            .unwrap_or_else(|p| p.into_inner())
392            .values()
393        {
394            if deleted.contains(&job.session) {
395                job.kill.cancel();
396            }
397        }
398    }

400    fn release_background_job(
401        &self,
402        session: SessionId,
403        tool_call_id: ToolCallId,
404        producer_id: ProducerId,
405    ) {
406        if let Some(job) = self
407            .inner
408            .background
409            .jobs
410            .lock()
411            .unwrap_or_else(|p| p.into_inner())
412            .remove(&tool_call_id)
413        {
414            job.kill.cancel();
415        }
416        self.drop_producer_registration(session, producer_id);
417    }
418
419    /// Accepts a finished job's report once, if its starting call still
420    /// stands in the session's log.
421    pub(super) fn commit_background_completion_direct(
422        &self,
423        session: SessionId,
424        tool_call_id: ToolCallId,
425        producer_id: ProducerId,
426        summary: &str,
427        body: String,
428        retained_output: Option<RetainedToolOutput>,
429    ) -> Result<(), EngineError> {
430        let authority = tool_authority(tool_call_id);
431        let events = self.inner.store.log(session)?.event_snapshot();
432        let started = events.iter().any(|event| {
433            matches!(
434                &event.payload,
435                super::Event::ToolCallTerminated { termination }
436                    if termination.tool_call_id == tool_call_id && completed_start(termination)
437            )
438        });
439        let accepted = events.iter().any(|event| {
440            matches!(
441                &event.payload,
442                super::Event::ProducerMessageAccepted { producer_owner, .. }
443                    if *producer_owner == authority.owner
444            )
445        });
446        if !started || accepted {
447            return Ok(());
448        }
449        self.accept_producer_with_output_direct(
450            &authority,
451            ExtensionProducerSendParams {
452                session_id: session,
453                producer_id,
454                mode: ProducerDeliveryMode::Steer,
455                idempotency_key: ProducerIdempotencyKey::new(COMPLETION_KEY)
456                    .expect("static background idempotency key is valid"),
457                description: super::producers::producer_description("Background: ", summary),
458                body,
459            },
460            None,
461            retained_output,
462        )
463        .map(|_| ())
464    }

471    #[test]
472    fn notification_wraps_the_report_and_output_in_its_element() {
473        let report = BackgroundReport {
474            summary: "bash exited with code 0: make".into(),
475            element: "background_bash".into(),

477            body: "Command: make\nStatus: exited with code 0 after 1.0s\n".into(),

479        assert_eq!(
480            render_notification(&report, Some("[stdout]\nok\n\n[stderr]\n")),
481            "<background_bash from=\"42&quot;&lt;&amp;\">\nCommand: make\nStatus: exited with \
482             code 0 after 1.0s\n[stdout]\nok\n\n[stderr]\n</background_bash>"
483        );
484        assert_eq!(
485            render_notification(&report, None),
486            "<background_bash from=\"42&quot;&lt;&amp;\">\nCommand: make\nStatus: exited with \
487             code 0 after 1.0s\n</background_bash>"
488        );
489    }
