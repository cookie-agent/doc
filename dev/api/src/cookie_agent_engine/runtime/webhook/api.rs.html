1//! Webhook requests (`docs/site/specs/webhooks.md` sections 4, 6, 9, 10,
2//! and 13): listing, creating, changing, rotating, and deleting webhooks,
3//! reading a secret or an event, and test events, for clients and for the
4//! webhook tools, and the checks every definition passes.
5//!
6//! A client may set anything. An agent's request never names a permission
7//! mode or a model, and never reads or rotates a secret after it created
8//! the webhook; its targets are checked by the tools before they reach

48fn check_name(name: Option<String>) -> Result<Option<String>, WebhookError> {
49    let Some(name) = name else {
50        return Ok(None);
51    };
52    let name = name.trim().to_owned();
53    if name.is_empty() {
54        return Err(WebhookError::invalid("name", "must not be blank"));
55    }
56    if name.chars().count() > cookie_agent_protocol::WEBHOOK_NAME_MAX_CHARS {
57        return Err(WebhookError::invalid(
58            "name",
59            format!(
60                "must be at most {} characters",
61                cookie_agent_protocol::WEBHOOK_NAME_MAX_CHARS
62            ),
63        ));
64    }
65    if name.chars().any(char::is_control) {
66        return Err(WebhookError::invalid(
67            "name",
68            "must not contain control characters",
69        ));
70    }
71    Ok(Some(name))
72}

74fn check_prompt(prompt: Option<String>) -> Result<Option<String>, WebhookError> {
75    let Some(prompt) = prompt else {
76        return Ok(None);
77    };
78    if prompt.trim().is_empty() {
79        return Err(WebhookError::invalid("prompt", "must not be blank"));
80    }
81    if prompt.len() > cookie_agent_protocol::WEBHOOK_PROMPT_MAX_BYTES {
82        return Err(WebhookError::invalid(
83            "prompt",
84            format!(
85                "must be at most {} bytes",
86                cookie_agent_protocol::WEBHOOK_PROMPT_MAX_BYTES
87            ),
88        ));
89    }
90    Ok(Some(prompt))
91}

93fn check_filter(conditions: &[WebhookCondition]) -> Result<(), WebhookError> {
94    let mut read = Vec::new();
95    for condition in conditions {
96        let Some(condition) = crate::webhooks::store::condition(condition) else {
97            return Err(WebhookError::invalid(
98                "filter",
99                "each condition names exactly one of `header` and `json`",
100            ));
101        };
102        read.push(condition);
103    }
104    filter::validate(&read).map_err(|error| WebhookError::invalid("filter", error.to_string()))
105}
106
107fn check_headers(headers: Vec<String>) -> Result<Vec<String>, WebhookError> {
108    if headers.len() > cookie_agent_protocol::WEBHOOK_HEADERS_MAX {
109        return Err(WebhookError::invalid(
110            "headers",
111            format!(
112                "at most {} headers",
113                cookie_agent_protocol::WEBHOOK_HEADERS_MAX
114            ),
115        ));
116    }
117    let mut seen = HashSet::new();
118    let mut kept = Vec::new();
119    for header in headers {
120        ValueSource::Header(header.clone())
121            .validate()
122            .map_err(|error| WebhookError::invalid("headers", error.to_string()))?;
123        if seen.insert(header.to_ascii_lowercase()) {
124            kept.push(header);
125        }
126    }
127    Ok(kept)
128}

130/// The stored check for `verify`: its secret, else `kept` when the kind
131/// and shape stay, else a new one. Whether the secret is new.
132fn stored_verify(
133    verify: WebhookVerify,
134    kept: Option<&StoredVerify>,
135) -> Result<(StoredVerify, bool), WebhookError> {
136    let refused = |reason: String| WebhookError::invalid("verify", reason);
137    let hmac = verify.kind == WebhookVerifyKind::HmacSha256;
138    if !hmac && (verify.header.is_some() || verify.encoding.is_some() || verify.prefix.is_some()) {
139        return Err(refused(
140            "header, encoding, and prefix belong to hmac_sha256".into(),
141        ));
142    }
143    let (secret, fresh) = match (
144        verify.secret,
145        kept.filter(|kept| kept.kind() == verify.kind),
146    ) {
147        (Some(secret), _) => (secret, true),
148        (None, Some(kept)) => (kept.secret().to_owned(), false),
149        (None, None) => (
150            generate_secret(if verify.kind == WebhookVerifyKind::Standard {
151                SecretKind::Standard
152            } else {
153                SecretKind::Plain
154            })
155            .map_err(|error| WebhookError::Store(error.to_string()))?,
156            true,
157        ),
158    };
159    let stored = match verify.kind {
160        WebhookVerifyKind::Github => StoredVerify::Github { secret },
161        WebhookVerifyKind::Standard => StoredVerify::Standard { secret },
162        WebhookVerifyKind::Bearer => StoredVerify::Bearer { token: secret },
163        WebhookVerifyKind::HmacSha256 => StoredVerify::HmacSha256 {
164            secret,
165            header: verify
166                .header
167                .ok_or_else(|| refused("hmac_sha256 needs a header".into()))?,
168            encoding: verify.encoding.unwrap_or(WebhookSignatureEncoding::Hex),
169            prefix: verify.prefix.filter(|prefix| !prefix.is_empty()),
170        },
171    };
172    stored
173        .verifier()
174        .validate()
175        .map_err(|error| refused(error.to_string()))?;
176    Ok((stored, fresh))
177}

179/// A secret for `verify`'s kind, made now.
180fn new_secret(verify: &StoredVerify) -> Result<String, WebhookError> {
181    generate_secret(if verify.kind() == WebhookVerifyKind::Standard {
182        SecretKind::Standard
183    } else {
184        SecretKind::Plain
185    })
186    .map_err(|error| WebhookError::Store(error.to_string()))
187}

189/// A URL token no webhook in `webhooks` has.
190fn new_token(webhooks: &[StoredWebhook]) -> Result<String, WebhookError> {
191    loop {
192        let token = crate::webhooks::ids::new_url_token()
193            .map_err(|error| WebhookError::Store(error.to_string()))?;
194        if webhooks.iter().all(|webhook| webhook.url_token != token) {
195            return Ok(token);
196        }
197    }
198}

200/// Refuses what only a client may set in an agent's request: a permission
201/// mode, or a model pin.
202fn refuse_client_fields(
203    permission_mode: bool,
204    target: Option<&WebhookTarget>,
205) -> Result<(), WebhookError> {
206    if permission_mode {
207        return Err(WebhookError::invalid(
208            "permission_mode",
209            "only a client can set a webhook's permission mode",
210        ));
211    }
212    if let Some(WebhookTarget::New(template)) = target
213        && template.model.is_some()
214    {
215        return Err(WebhookError::invalid(
216            "target",
217            "only a client can pin a webhook's model",
218        ));
219    }
220    Ok(())
221}

224    /// Whether `target`'s session exists and has the kind the target names.
225    fn check_webhook_session(&self, target: &StoredWebhookTarget) -> Result<(), WebhookError> {
226        let Some(session) = target.session() else {
227            return Ok(());
228        };
229        let summary = self.inner.store.summary(session).map_err(|_| {
230            WebhookError::invalid("target", format!("there is no session {session}"))
231        })?;
232        match (target, &summary.meta.origin) {
233            (StoredWebhookTarget::Session(_), SessionOrigin::Root)
234            | (StoredWebhookTarget::Subagent(_), SessionOrigin::Delegated { .. }) => Ok(()),
235            (StoredWebhookTarget::Session(_), _) => Err(WebhookError::invalid(
236                "target",
237                format!("session {session} is a subagent's session; target it as `subagent`"),
238            )),
239            _ => Err(WebhookError::invalid(
240                "target",
241                format!("session {session} is a root session; target it as `session`"),
242            )),
243        }
244    }

246    /// Checks a definition and fills in the defaults of its kind.
247    fn check_webhook_definition(&self, definition: Definition) -> Result<Definition, WebhookError> {
248        self.check_webhook_session(&definition.target)?;
249        let (delivery, permission_mode, max_concurrent) = match &definition.target {
250            StoredWebhookTarget::New(target) => {
251                self.check_cron_template(&target.template)
252                    .map_err(|error| match error {
253                        crate::crons::CronError::Invalid { reason, .. } => {
254                            WebhookError::invalid("target", reason)
255                        }
256                        other => WebhookError::Store(other.to_string()),
257                    })?;
258                if let Some(key) = &target.thread_key {
259                    source(key)
260                        .ok_or_else(|| {
261                            WebhookError::invalid(
262                                "target",
263                                "thread_key names exactly one of `header` and `json`",
264                            )
265                        })?
266                        .validate()
267                        .map_err(|error| {
268                            WebhookError::invalid("target", format!("thread_key: {error}"))
269                        })?;
270                }
271                let delivery = if target.thread_key.is_some() {
272                    Some(definition.delivery.unwrap_or(ProducerDeliveryMode::Queue))
273                } else if definition.delivery.is_some() {
274                    return Err(WebhookError::invalid(
275                        "delivery",
276                        "a new-session webhook without a thread key starts a session for every \
277                         event; delivery applies to existing sessions and thread keys",
278                    ));
279                } else {
280                    None
281                };
282                (
283                    delivery,
284                    Some(
285                        definition
286                            .permission_mode
287                            .unwrap_or(PermissionMode::AutoApproveN),
288                    ),
289                    definition.max_concurrent,
290                )
291            }
292            StoredWebhookTarget::Session(_) | StoredWebhookTarget::Subagent(_) => {
293                if definition.permission_mode.is_some() {
294                    return Err(WebhookError::invalid(
295                        "permission_mode",
296                        "an existing-session webhook runs with its session's permission mode",
297                    ));
298                }
299                if definition.max_concurrent.is_some() {
300                    return Err(WebhookError::invalid(
301                        "max_concurrent",
302                        "applies to new-session webhooks",
303                    ));
304                }
305                (
306                    Some(definition.delivery.unwrap_or(ProducerDeliveryMode::Queue)),
307                    None,
308                    None,
309                )
310            }
311        };
312        check_filter(&definition.filter)?;
313        Ok(Definition {
314            name: check_name(definition.name)?,
315            prompt: check_prompt(definition.prompt)?,
316            headers: check_headers(definition.headers)?,
317            delivery,
318            permission_mode,
319            max_concurrent,
320            ..definition
321        })
322    }

324    /// Runs `change` on the webhook store off the async workers.
325    pub(crate) async fn webhooks_blocking<T: Send + 'static>(
326        &self,
327        change: impl FnOnce(&Engine, &mut Vec<StoredWebhook>) -> Result<T, WebhookError>
328        + Send
329        + 'static,
330    ) -> Result<T, EngineError> {
331        let engine = self.clone();
332        let (value, _) = tokio::task::spawn_blocking(move || {
333            let store = std::sync::Arc::clone(&engine.inner.webhooks);
334            store.mutate(|webhooks| change(&engine, webhooks))
335        })
336        .await
337        .map_err(|error| WebhookError::Store(format!("the request did not finish: {error}")))??;
338        Ok(value)
339    }

399    /// `webhooks.rotate`: a new URL token, or a new signing secret.
400    pub async fn rotate_webhook(
401        &self,
402        id: WebhookId,
403        what: WebhookRotate,
404    ) -> Result<WebhookResult, EngineError> {
405        self.ensure_not_shutting_down()?;
406        let (webhook, secret) = self
407            .webhooks_blocking(move |_, webhooks| {
408                let token = match what {
409                    WebhookRotate::Url => Some(new_token(webhooks)?),
410                    WebhookRotate::Secret => None,
411                };
412                let webhook = webhooks
413                    .iter_mut()
414                    .find(|webhook| webhook.id == id)
415                    .ok_or_else(|| WebhookError::NotFound(id.clone()))?;
416                let mut secret = None;
417                match what {
418                    WebhookRotate::Url => {
419                        webhook.url_token = token.expect("made above");
420                    }
421                    WebhookRotate::Secret => {
422                        let verify = webhook.verify.as_ref().ok_or_else(|| {
423                            WebhookError::invalid("verify", "the webhook has no signature check")
424                        })?;
425                        let fresh = new_secret(verify)?;
426                        webhook.verify = Some(verify.with_secret(fresh.clone()));
427                        secret = Some(fresh);
428                    }
429                }
430                Ok((webhook.clone(), secret))
431            })
432            .await?;
433        Ok(WebhookResult {
434            webhook: self.inner.webhooks.wire_of(&webhook),
435            secret,
436        })
437    }

452    /// `webhooks.event`: one event with its request.
453    pub async fn webhook_event(
454        &self,
455        params: WebhookEventParams,
456    ) -> Result<WebhookEventResult, EngineError> {
457        let webhook = self
458            .inner
459            .webhooks
460            .webhook(&params.id)
461            .ok_or_else(|| WebhookError::NotFound(params.id.clone()))?;
462        let stored =
463            webhook
464                .event(&params.event)
465                .cloned()
466                .ok_or_else(|| WebhookError::EventNotFound {
467                    id: params.id.clone(),
468                    event: params.event.clone(),
469                })?;
470        let store = std::sync::Arc::clone(&self.inner.webhooks);
471        let (id, event) = (params.id.clone(), params.event.clone());
472        let body = tokio::task::spawn_blocking(move || store.read_body(&id, &event))
473            .await
474            .map_err(|error| WebhookError::Store(error.to_string()))??;
475        let wire = self
476            .inner
477            .webhooks
478            .wire_of(&webhook)
479            .history
480            .into_iter()
481            .find(|event| event.id == params.event)
482            .expect("the event is in the history");
483        let (body_text, body_base64) = match body {
484            None => (None, None),
485            Some(bytes) => match String::from_utf8(bytes) {
486                Ok(text) => (Some(text), None),
487                Err(error) => (
488                    None,
489                    Some(base64::engine::general_purpose::STANDARD.encode(error.into_bytes())),
490                ),
491            },
492        };
493        Ok(WebhookEventResult {
494            event: wire,
495            method: stored.method,
496            headers: stored.headers,
497            body_text,
498            body_base64,
499        })
500    }

502    /// `webhooks.test`: an event that skips the checks, the filter, and the
503    /// limits, delivered as any other.
504    pub async fn test_webhook(
505        &self,
506        params: WebhookTestParams,
507    ) -> Result<WebhookTestResult, EngineError> {
508        self.ensure_not_shutting_down()?;
509        let webhook = self
510            .inner
511            .webhooks
512            .webhook(&params.id)
513            .ok_or_else(|| WebhookError::NotFound(params.id.clone()))?;
514        let mut headers = params
515            .headers
516            .into_iter()
517            .map(|header| (header.name, header.value))
518            .collect::<Vec<_>>();
519        if !headers
520            .iter()
521            .any(|(name, _)| name.eq_ignore_ascii_case("content-type"))
522        {
523            headers.push((
524                "Content-Type".to_owned(),
525                params
526                    .content_type
527                    .unwrap_or_else(|| "application/json".to_owned()),
528            ));
529        }
530        let request = WebhookRequest {
531            method: "POST".to_owned(),
532            headers,
533            body: params.body.unwrap_or_else(|| "{}".to_owned()).into_bytes(),
534        };
535        let event = self.store_webhook_event(&webhook, request, true).await?;
536        self.spawn_webhook_pump(webhook.id.clone());
537        let webhook = self.inner.webhooks.webhook(&webhook.id).map_or_else(
538            || self.inner.webhooks.wire_of(&webhook),
539            |webhook| self.inner.webhooks.wire_of(&webhook),
540        );
541        Ok(WebhookTestResult { webhook, event })
542    }

544    /// Creates a webhook for `requester`; the result carries its secret.
545    pub(crate) async fn create_webhook_as(
546        &self,
547        requester: CronRequester,
548        params: WebhookCreateParams,
549    ) -> Result<WebhookResult, EngineError> {
550        self.ensure_not_shutting_down()?;
551        if matches!(requester, CronRequester::Agent(_)) {
552            refuse_client_fields(params.permission_mode.is_some(), Some(&params.target))?;
553        }
554        let now = Timestamp::now();
555        let (webhook, secret) = self
556            .webhooks_blocking(move |engine, webhooks| {
557                let definition = engine.check_webhook_definition(Definition {
558                    name: params.name,
559                    prompt: params.prompt,
560                    target: params.target.into(),
561                    delivery: params.delivery,
562                    filter: params.filter,
563                    headers: params.headers,
564                    max_per_hour: params.max_per_hour,
565                    max_concurrent: params.max_concurrent,
566                    permission_mode: params.permission_mode,
567                    max_consecutive_failures: params.max_consecutive_failures,
568                })?;
569                let verify = params
570                    .verify
571                    .map(|verify| stored_verify(verify, None))
572                    .transpose()?;
573                let secret = verify
574                    .as_ref()
575                    .map(|(verify, _)| verify.secret().to_owned());
576                let id = crate::webhooks::ids::new_webhook_id(|id| {
577                    webhooks.iter().any(|webhook| webhook.id.as_str() == id)
578                })
579                .map_err(|error| WebhookError::Store(error.to_string()))?;
580                let enabled = params.enabled.unwrap_or(true);
581                let webhook = StoredWebhook {
582                    id: WebhookId::new(id).expect("eight base-36 characters"),
583                    name: definition.name,
584                    url_token: new_token(webhooks)?,
585                    prompt: definition.prompt,
586                    target: definition.target,
587                    delivery: definition.delivery,
588                    verify: verify.map(|(verify, _)| verify),
589                    filter: definition.filter,
590                    headers: definition.headers,
591                    max_per_hour: definition.max_per_hour,
592                    max_concurrent: definition.max_concurrent,
593                    permission_mode: definition.permission_mode,
594                    max_consecutive_failures: definition.max_consecutive_failures,
595                    enabled,
596                    disabled_reason: (!enabled).then(|| StoredDisabled::Disabled {
597                        by: requester.actor(),
598                    }),
599                    created_by: requester.actor(),
600                    created_at: now,
601                    counts: Default::default(),
602                    recent: Vec::new(),
603                    delivery_ids: Vec::new(),
604                    consecutive_failures: 0,
605                    events: Vec::new(),
606                    threads: crate::threads::ThreadMap::default(),
607                };
608                webhooks.push(webhook.clone());
609                Ok((webhook, secret))
610            })
611            .await?;
612        Ok(WebhookResult {
613            webhook: self.inner.webhooks.wire_of(&webhook),
614            secret,
615        })
616    }

618    /// Changes a webhook for `requester`. Enabling it resets its failure
619    /// count and delivers what waited.
620    pub(crate) async fn update_webhook_as(
621        &self,
622        requester: CronRequester,
623        params: WebhookUpdateParams,
624    ) -> Result<WebhookResult, EngineError> {
625        self.ensure_not_shutting_down()?;
626        if matches!(requester, CronRequester::Agent(_)) {
627            refuse_client_fields(params.permission_mode.is_some(), params.target.as_ref())?;
628        }
629        let id = params.id.clone();
630        let (webhook, secret) = self
631            .webhooks_blocking(move |engine, webhooks| {
632                let webhook = webhooks
633                    .iter_mut()
634                    .find(|webhook| webhook.id == params.id)
635                    .ok_or_else(|| WebhookError::NotFound(params.id.clone()))?;
636                let target: StoredWebhookTarget = params
637                    .target
638                    .clone()
639                    .map_or_else(|| webhook.target.clone(), Into::into);
640                let kind_changed = target.kind() != webhook.target.kind();
641                let in_session = target.session().is_some();
642                let threaded = target
643                    .new_target()
644                    .is_some_and(|target| target.thread_key.is_some());
645                let threads_kept = !kind_changed
646                    && target.new_target().map(|target| &target.thread_key)
647                        == webhook.target.new_target().map(|target| &target.thread_key);
648                let delivery = params.delivery.or(
649                    if (in_session || threaded) && !kind_changed && webhook.delivery.is_some() {
650                        webhook.delivery
651                    } else {
652                        None
653                    },
654                );
655                let permission_mode = match params.permission_mode {
656                    Some(mode) => mode,
657                    None if !in_session && !kind_changed => webhook.permission_mode,
658                    None => None,
659                };
660                let max_concurrent = match params.max_concurrent {
661                    Some(limit) => limit,
662                    None if !in_session => webhook.max_concurrent,
663                    None => None,
664                };
665                let definition = engine.check_webhook_definition(Definition {
666                    name: match params.name.clone() {
667                        Some(name) => name,
668                        None => webhook.name.clone(),
669                    },
670                    prompt: match params.prompt.clone() {
671                        Some(prompt) => prompt,
672                        None => webhook.prompt.clone(),
673                    },
674                    target,
675                    delivery,
676                    filter: params
677                        .filter
678                        .clone()
679                        .unwrap_or_else(|| webhook.filter.clone()),
680                    headers: params
681                        .headers
682                        .clone()
683                        .unwrap_or_else(|| webhook.headers.clone()),
684                    max_per_hour: match params.max_per_hour {
685                        Some(limit) => limit,
686                        None => webhook.max_per_hour,
687                    },
688                    max_concurrent,
689                    permission_mode,
690                    max_consecutive_failures: match params.max_consecutive_failures {
691                        Some(limit) => limit,
692                        None => webhook.max_consecutive_failures,
693                    },
694                })?;
695                let mut secret = None;
696                match params.verify.clone() {
697                    None => {}
698                    Some(None) => webhook.verify = None,
699                    Some(Some(verify)) => {
700                        let (verify, fresh) = stored_verify(verify, webhook.verify.as_ref())?;
701                        if fresh {
702                            secret = Some(verify.secret().to_owned());
703                        }
704                        webhook.verify = Some(verify);
705                    }
706                }
707                webhook.name = definition.name;
708                webhook.prompt = definition.prompt;
709                webhook.target = definition.target;
710                webhook.delivery = definition.delivery;
711                webhook.filter = definition.filter;
712                webhook.headers = definition.headers;
713                webhook.max_per_hour = definition.max_per_hour;
714                webhook.max_concurrent = definition.max_concurrent;
715                webhook.permission_mode = definition.permission_mode;
716                webhook.max_consecutive_failures = definition.max_consecutive_failures;
717                if !threads_kept {
718                    webhook.threads = crate::threads::ThreadMap::default();
719                }
720                match params.enabled {
721                    Some(false) if webhook.enabled => {
722                        webhook.disable(StoredDisabled::Disabled {
723                            by: requester.actor(),
724                        });
725                    }
726                    Some(true) if !webhook.enabled => {
727                        webhook.enabled = true;
728                        webhook.disabled_reason = None;
729                        webhook.consecutive_failures = 0;
730                    }
731                    _ => {}
732                }
733                Ok((webhook.clone(), secret))
734            })
735            .await?;
736        // Enabling it, or a new target, lets what waited go.
737        self.spawn_webhook_pump(id);
738        Ok(WebhookResult {
739            webhook: self.inner.webhooks.wire_of(&webhook),
740            secret,
741        })
742    }

744    /// Deletes the webhooks that target any of `deleted`, which are deleted
745    /// sessions; deleted sessions also leave every webhook's threads
746    /// (section 13).
747    pub(crate) fn delete_webhooks_targeting(&self, deleted: &HashSet<SessionId>) {
748        if !self.inner.webhooks.webhooks().iter().any(|webhook| {
749            webhook
750                .target
751                .session()
752                .is_some_and(|session| deleted.contains(&session))
753                || webhook.threads.names_any(deleted)
754        }) {
755            return;
756        }
757        let deleted = deleted.clone();
758        if let Err(error) = self.webhooks_now(move |_, webhooks| {
759            webhooks.retain(|webhook| {
760                !webhook
761                    .target
762                    .session()
763                    .is_some_and(|session| deleted.contains(&session))
764            });
765            for webhook in webhooks.iter_mut() {
766                webhook.threads.forget_sessions(&deleted);
767            }
768            Ok(())
769        }) {
770            eprintln!("cookie-agent: webhooks of deleted sessions: {error}");
771        }
772    }

774    /// Deletes the new-session webhooks whose template names `project`,
775    /// which is being deleted.
776    pub(crate) async fn delete_webhooks_of_project(
777        &self,
778        project: ProjectId,
779    ) -> Result<(), EngineError> {
780        self.webhooks_blocking(move |_, webhooks| {
781            webhooks.retain(|webhook| {
782                webhook
783                    .target
784                    .new_target()
785                    .is_none_or(|target| target.template.project.as_ref() != Some(&project))
786            });
787            Ok(())
788        })
789        .await
790    }
