1//! Receiving a request on a webhook's URL (`docs/site/specs/webhooks.md`
2//! section 5): the checks in order, each with its answer, and storing the
3//! event before the `202`. The daemon's HTTP route answers `405` for other
4//! methods and reads the body up to [`Engine::webhook_body_limit`] (`413`)
5//! after [`Engine::webhook_receives`] (`404`); everything else is here.
6
7use cookie_agent_protocol::WebhookHeader;
8use jiff::Timestamp;

10use crate::runtime::{Engine, EngineError};

12    filter,

14    request::{WebhookRequest, delivery_id, parse_body, thread_key},
15    store::{Noise, StoredEvent, StoredEventOutcome, StoredWebhook, WebhookError, new_event_id},
16};

18/// How long a sender is asked to wait when the backlog is full.
19const BACKLOG_RETRY_AFTER_SECONDS: u64 = 60;
20
21/// The daemon's answer to a webhook request.
22#[derive(Clone, Debug, Eq, PartialEq)]
23pub struct WebhookReply {
24    pub status: u16,
25    /// A short JSON object.
26    pub body: serde_json::Value,
27    /// Seconds for `Retry-After`, with `429` and `503`.
28    pub retry_after: Option<u64>,
29}
30
31impl WebhookReply {
32    fn of(status: u16, body: serde_json::Value) -> Self {
33        Self {
34            status,
35            body,
36            retry_after: None,
37        }
38    }
39
40    /// The answer to a URL that names no webhook, a disabled one, or any
41    /// while webhooks are off: the same for every case.
42    #[must_use]
43    pub fn not_found() -> Self {
44        Self::of(404, serde_json::json!({"error": "not_found"}))
45    }
46}
47
48/// The headers an event keeps: all but the secret ones.
49fn kept_headers(request: &WebhookRequest, secret: &[String]) -> Vec<WebhookHeader> {
50    request
51        .headers
52        .iter()
53        .filter(|(name, _)| {
54            let lower = name.to_ascii_lowercase();
55            !ALWAYS_HIDDEN_HEADERS.contains(&lower.as_str())
56                && !secret
57                    .iter()
58                    .any(|header| header.eq_ignore_ascii_case(&lower))
59        })
60        .map(|(name, value)| WebhookHeader {
61            name: name.clone(),
62            value: value.clone(),
63        })
64        .collect()
65}

67impl Engine {
68    /// The most body bytes a webhook request may have
69    /// (`[webhooks] max_body_bytes`).
70    #[must_use]
71    pub fn webhook_body_limit(&self) -> u64 {
72        self.inner.config.load().runtime.webhooks.max_body_bytes
73    }
74
75    /// Whether `token` names an enabled webhook while webhooks are on.
76    #[must_use]
77    pub fn webhook_receives(&self, token: &str) -> bool {
78        self.receiving_webhook(token).is_some()
79    }
80
81    fn receiving_webhook(&self, token: &str) -> Option<StoredWebhook> {
82        if !self.inner.config.load().runtime.webhooks.enabled
83            || !crate::webhooks::ids::is_url_token(token)
84        {
85            return None;
86        }
87        self.inner
88            .webhooks
89            .by_token(token)
90            .filter(|webhook| webhook.enabled)
91    }
92
93    /// Receives `request` on the URL of `token` (section 5.1): the check,
94    /// duplicates, the filter, the rate, the backlog, then the event is
95    /// stored and delivered afterwards.
96    pub async fn receive_webhook(&self, token: &str, request: WebhookRequest) -> WebhookReply {
97        let Some(webhook) = self.receiving_webhook(token) else {
98            return WebhookReply::not_found();
99        };
100        let store = &self.inner.webhooks;
101        let now = Timestamp::now();
102        let verifier = Self::webhook_verifier(&webhook);
103        if let Some(verifier) = &verifier
104            && verifier.verify(&request, now).is_err()
105        {
106            store.note_noise(&webhook.id, Noise::Rejected, now);
107            return WebhookReply::of(401, serde_json::json!({"error": "unauthorized"}));
108        }
109        let delivery = delivery_id(
110            &request,
111            verifier
112                .as_ref()
113                .and_then(|verifier| verifier.delivery_header()),
114        );
115        if delivery.as_deref().is_some_and(|id| webhook.seen(id)) {
116            store.note_noise(&webhook.id, Noise::Duplicate, now);
117            return WebhookReply::of(200, serde_json::json!({"duplicate": true}));
118        }
119        let parsed = parse_body(&request);
120        if !filter::matches(&webhook.conditions(), &request, &parsed) {
121            store.note_noise(&webhook.id, Noise::Filtered, now);
122            return WebhookReply::of(204, serde_json::json!({}));
123        }
124        let config = self.inner.config.load().runtime.webhooks;
125        let rate = self.webhook_rate_limit(&webhook);
126        if let Some(limit) = rate
127            && webhook.recent_count(now) >= limit
128        {
129            store.note_noise(&webhook.id, Noise::RateLimited, now);
130            let oldest = webhook
131                .recent
132                .iter()
133                .copied()
134                .filter(|at| now.duration_since(*at) < jiff::SignedDuration::from_hours(1))
135                .min()
136                .unwrap_or(now);
137            let wait = jiff::SignedDuration::from_hours(1) - now.duration_since(oldest);
138            return WebhookReply {
139                retry_after: Some(u64::try_from(wait.as_secs().max(1)).unwrap_or(1)),
140                ..WebhookReply::of(429, serde_json::json!({"error": "rate_limited"}))
141            };
142        }
143        if webhook.waiting_count() >= config.backlog {
144            store.note_noise(&webhook.id, Noise::BacklogFull, now);
145            return WebhookReply {
146                retry_after: Some(BACKLOG_RETRY_AFTER_SECONDS),
147                ..WebhookReply::of(503, serde_json::json!({"error": "backlog_full"}))
148            };
149        }
150        match self.store_webhook_event(&webhook, request, false).await {
151            Ok(event) => {
152                self.spawn_webhook_pump(webhook.id.clone());
153                WebhookReply::of(202, serde_json::json!({"event": event}))
154            }
155            Err(EngineError::Webhook(WebhookError::Invalid { .. })) => {
156                // Another request with the same delivery ID, or over the
157                // rate, got in first.
158                store.note_noise(&webhook.id, Noise::Duplicate, now);
159                WebhookReply::of(200, serde_json::json!({"duplicate": true}))
160            }
161            Err(error) => {
162                eprintln!("cookie-agent: webhook {}: {error}", webhook.id);
163                WebhookReply::of(500, serde_json::json!({"error": "not_stored"}))
164            }
165        }
166    }
167
168    /// The events per hour `webhook` takes; `None` for no limit.
169    fn webhook_rate_limit(&self, webhook: &StoredWebhook) -> Option<usize> {
170        let limit = match webhook.max_per_hour {
171            Some(limit) => u64::try_from(limit).ok().filter(|limit| *limit > 0),
172            None => self.inner.config.load().runtime.webhooks.per_hour_limit(),
173        };
174        limit.map(|limit| usize::try_from(limit).unwrap_or(usize::MAX))
175    }
176
177    /// Stores an event of `webhook` for `request`: its body, then its entry
178    /// as pending (section 12). A test event skips the duplicate and rate
179    /// checks and counts toward no rate. The event's ID.
180    pub(crate) async fn store_webhook_event(
181        &self,
182        webhook: &StoredWebhook,
183        request: WebhookRequest,
184        test: bool,
185    ) -> Result<String, EngineError> {
186        let verifier = Self::webhook_verifier(webhook);
187        let delivery = if test {
188            None
189        } else {
190            delivery_id(
191                &request,
192                verifier
193                    .as_ref()
194                    .and_then(|verifier| verifier.delivery_header()),
195            )
196        };
197        let thread = webhook
198            .thread_source()
199            .and_then(|source| thread_key(&request, &parse_body(&request), &source));
200        let secret = verifier
201            .as_ref()
202            .map(crate::webhooks::verify::Verifier::secret_headers)
203            .unwrap_or_default();
204        let event_id = new_event_id(webhook)?;
205        let now = Timestamp::now();
206        let event = StoredEvent {
207            id: event_id.clone(),
208            received_at: now,
209            test,
210            delivery_id: delivery.clone(),
211            thread_key: thread,
212            bytes: request.body.len() as u64,
213            content_type: request.header("content-type").map(str::to_owned),
214            method: request.method.clone(),
215            headers: kept_headers(&request, &secret),
216            outcome: StoredEventOutcome::Pending,
217            session_id: None,
218            message_id: None,
219            run_id: None,
220            run_end: None,
221        };
222        let rate = self.webhook_rate_limit(webhook);
223        let history = self.inner.config.load().runtime.webhooks.history;
224        let store = std::sync::Arc::clone(&self.inner.webhooks);
225        let id = webhook.id.clone();
226        let body = request.body;
227        let written_event = event_id.clone();
228        let result = tokio::task::spawn_blocking(move || {
229            store.write_body(&id, &written_event, &body)?;
230            let stored = store.mutate(|webhooks| {
231                let webhook = webhooks
232                    .iter_mut()
233                    .find(|webhook| webhook.id == id)
234                    .ok_or_else(|| WebhookError::NotFound(id.clone()))?;
235                if !test {
236                    // Checked again under the lock: a request racing this
237                    // one may have taken the ID or the last slot.
238                    if delivery.as_deref().is_some_and(|id| webhook.seen(id))
239                        || rate.is_some_and(|limit| webhook.recent_count(now) >= limit)
240                    {
241                        return Err(WebhookError::invalid("event", "taken by another request"));
242                    }
243                    webhook.note_accepted(delivery.as_deref(), now, rate.is_some());
244                }
245                webhook.push_event(event, history);
246                Ok(())
247            });
248            if stored.is_err() {
249                store.remove_body(&id, &written_event);
250            }
251            stored
252        })
253        .await
254        .map_err(|error| WebhookError::Store(error.to_string()))?;
255        result?;
256        Ok(event_id)
257    }
258}

260#[cfg(test)]
261mod tests {
262    use super::*;
263
264    #[test]
265    fn kept_headers_leave_out_secret_ones() {
266        let request = WebhookRequest {
267            method: "POST".into(),
268            headers: vec![
269                ("Authorization".into(), "Bearer x".into()),
270                ("X-Hub-Signature-256".into(), "sha256=00".into()),
271                ("X-Custom-Signature".into(), "abc".into()),
272                ("X-GitHub-Event".into(), "issues".into()),
273            ],
274            body: Vec::new(),
275        };
276        let kept = kept_headers(&request, &["x-custom-signature".to_owned()]);
277        let names = kept
278            .iter()
279            .map(|header| header.name.as_str())
280            .collect::<Vec<_>>();
281        assert_eq!(names, ["X-GitHub-Event"]);
282    }
