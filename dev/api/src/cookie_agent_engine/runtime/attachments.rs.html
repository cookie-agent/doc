20/// A runtime ID reserved for one runtime dialing in, from
21/// [`Engine::claim_dial_in_runtime`] until it is accepted or dropped.
22pub struct DialInClaim {
23    engine: Engine,
24    template: Arc<crate::runtimes::DialInTemplate>,
25}
26
27impl DialInClaim {
28    /// Takes the runtime that dialed in over `transport`: initializes it,
29    /// makes it the runtime of the claimed ID, and reopens the instances
30    /// its sessions use.
31    pub async fn accept(
32        self,
33        transport: Box<dyn cookie_agent_protocol::Transport>,
34    ) -> Result<(), DialInRefusal> {
35        let inner = &self.engine.inner;
36        let runtime_id = crate::runtimes::RuntimeTemplate::runtime_id(self.template.as_ref());
37        let settings = crate::runtimes::runtime_settings(&inner.config.load(), runtime_id);
38        let client = self
39            .template
40            .connect(
41                cookie_agent_runtime_api::RuntimeConnection::new(transport),
42                &settings,
43                Some(Arc::clone(&inner.mcp_hub)),
44            )
45            .await
46            .map_err(|reason| DialInRefusal::Initialize(runtime_id.to_owned(), reason))?;
47        inner.runtimes.watch_connection(&self.template, &client);
48        inner.runtimes.revive(runtime_id);
49        Ok(())
50    }
51}
52
53impl Drop for DialInClaim {
54    fn drop(&mut self) {
55        self.template.release();
56    }
57}

93/// The name a session attaches an instance under: the name the instance
94/// reports, suffixed with `_2`, `_3`, ... while the session already has an
95/// attachment of that name.
96pub(crate) fn allocate_attachment_name(
97    existing: &[RuntimeAttachment],
98    reported: &str,
99) -> Result<AttachmentName, EngineError> {
100    let taken = |name: &str| {
101        existing
102            .iter()
103            .any(|attachment| attachment.name.as_str() == name)
104    };
105    let invalid = |name: &str| EngineError::Runtime(format!("invalid attachment name `{name}`"));
106    if !taken(reported) {
107        return AttachmentName::new(reported).map_err(|_| invalid(reported));
108    }
109    (2..=existing.len() + 1)
110        .map(|suffix| format!("{reported}_{suffix}"))
111        .find(|candidate| !taken(candidate))
112        .ok_or_else(|| invalid(reported))
113        .and_then(|name| AttachmentName::new(name.as_str()).map_err(|_| invalid(&name)))
114}

116/// Whether `attachment` refers to `instance`.
117fn refers_to(attachment: &RuntimeAttachment, instance: &RuntimeInstance) -> bool {
118    attachment.runtime_id.as_str() == instance.info().id()
119        && &attachment.capability == instance.info().capability()
120        && attachment.workspace.as_deref().map(std::path::Path::new)
121            == instance
122                .info()
123                .workspace()
124                .map(|workspace| workspace.root())
125}

127/// `instance` added to `attachments` under its reported name. Attaching one
128/// instance twice is an error.
129pub(crate) fn push_instance(
130    attachments: &mut Vec<RuntimeAttachment>,
131    instance: &RuntimeInstance,
132) -> Result<RuntimeAttachment, EngineError> {
133    if let Some(existing) = attachments
134        .iter()
135        .find(|attachment| refers_to(attachment, instance))
136    {
137        return Err(EngineError::Runtime(format!(
138            "capability `{}` of runtime `{}` is already attached as `{}`",
139            instance.info().capability(),
140            instance.info().id(),
141            existing.name
142        )));
143    }
144    let name = allocate_attachment_name(attachments, instance.info().attachment_name())?;
145    let attachment = instance.attachment(name)?;
146    attachments.push(attachment.clone());
147    Ok(attachment)
148}

188    /// The tool sources of a session with `attachments`: the base layer, then
189    /// each attachment's running instance, and after the first attachment of
190    /// a runtime that has tools of its own (its runtime-scoped MCP servers')
191    /// those, once per runtime however many of its workspaces the session
192    /// attaches. They compose under the runtime's ID and run without a
193    /// workspace. An attachment whose instance is not running contributes
194    /// nothing.
195    pub(crate) fn tool_sources_for(
196        &self,
197        attachments: &[RuntimeAttachment],
198    ) -> Vec<SourcedProvider> {
199        let mut sources = self
200            .inner
201            .runtimes
202            .base_providers()
203            .into_iter()
204            .map(|provider| SourcedProvider {
205                attachment: None,
206                generation: 0,
207                origin: crate::ToolOrigin::Base,
208                provider,
209            })
210            .collect::<Vec<_>>();
211        let mut shared_runtimes = std::collections::HashSet::new();
212        for attachment in attachments {
213            let Some(instance) = self.inner.runtimes.instance(attachment) else {
214                continue;
215            };
216            let origin = instance.origin();
217            let (generation, providers) = instance.sources();
218            sources.extend(providers.into_iter().map(|provider| SourcedProvider {
219                attachment: Some(attachment.name.clone()),
220                generation,
221                origin: origin.clone(),
222                provider,
223            }));
224            if let Some((generation, shared)) = instance.shared_source()
225                && shared_runtimes.insert(attachment.runtime_id.clone())
226            {
227                sources.push(SourcedProvider {
228                    attachment: Some(attachment.name.clone()),
229                    generation,
230                    origin: crate::ToolOrigin::Runtime(shared.info),
231                    provider: shared.provider,
232                });
233            }
234        }
235        sources
236    }

246    /// The instance of `capability` of `runtime_id` at `workspace` that a
247    /// session is about to attach, as creating a session, attaching a
248    /// runtime, and delegating with capabilities do: opened when it is not
249    /// running, and never over a runtime that is not online (see
250    /// `RuntimeRegistry::ensure`). A dial-in runtime this daemon has not
251    /// seen since it started, but remembers, is refused as not connected,
252    /// with when it was last seen.
253    pub(crate) async fn ensure_runtime(
254        &self,
255        runtime_id: &str,
256        capability: Option<&str>,
257        workspace: Option<&std::path::Path>,
258    ) -> Result<Arc<RuntimeInstance>, EngineError> {
259        if let Some(offline) = self.remembered_offline(runtime_id) {
260            return Err(EngineError::Runtime(offline));
261        }
262        self.inner
263            .runtimes
264            .ensure(runtime_id, capability, workspace)
265            .await
266    }

268    /// Why dial-in runtime `runtime_id` cannot be used, when this daemon
269    /// has not seen it connect since it started but remembers it: it is
270    /// not connected, last seen when its record says.
271    pub(crate) fn remembered_offline(&self, runtime_id: &str) -> Option<String> {
272        if self.inner.runtimes.knows(runtime_id) {
273            return None;
274        }
275        let record = self
276            .inner
277            .machines
278            .record(&RuntimeId::new(runtime_id).ok()?)?;
279        (record.kind == crate::machines::RecordKind::DialIn)
280            .then(|| crate::runtimes::not_connected(runtime_id, Some(record.last_seen)))
281    }

283    /// The attachments a new session gets for `runtimes`, starting the
284    /// instances that are not running.
285    pub(crate) async fn resolve_session_runtimes(
286        &self,
287        runtimes: &SessionRuntimes,
288    ) -> Result<Vec<RuntimeAttachment>, EngineError> {
289        let mut attachments = Vec::new();
290        if runtimes.global {
291            push_instance(&mut attachments, self.inner.runtimes.global())?;
292        }
293        match &runtimes.runtimes {
294            None => match &runtimes.cwd {
295                Some(cwd) => {
296                    let instance = self
297                        .ensure_runtime(
298                            RuntimeId::LOCAL,
299                            Some(cookie_agent_protocol::CapabilityName::CODING),
300                            Some(cwd),
301                        )
302                        .await?;
303                    push_instance(&mut attachments, &instance)?;
304                }
305                None => {
306                    push_instance(&mut attachments, self.inner.runtimes.local())?;
307                }
308            },
309            Some(requests) => {
310                for request in requests {
311                    let instance = self
312                        .ensure_runtime(
313                            &request.runtime_id,
314                            request.capability.as_deref(),
315                            request.workspace.as_deref(),
316                        )
317                        .await?;
318                    push_instance(&mut attachments, &instance)?;
319                }
320            }
321        }
322        Ok(attachments)
323    }

325    /// Makes sure every instance `session` is attached to is running, as when
326    /// the session is opened in a daemon that has not started them. Returns
327    /// the attachments that could not be started.
328    pub(crate) async fn ensure_session_runtimes(
329        &self,
330        session: SessionId,
331    ) -> Result<Vec<(AttachmentName, EngineError)>, EngineError> {
332        let mut failures = Vec::new();
333        for attachment in self.session_runtimes(session)? {
334            if let Err(error) = self.inner.runtimes.ensure_attachment(&attachment).await {
335                failures.push((attachment.name, error));
336            }
337        }
338        Ok(failures)
339    }

341    /// Connects what `session` uses, as when it is created or opened: every
342    /// attached instance is started, and every MCP server of an attached
343    /// instance whose last connection attempt failed is retried. The next
344    /// run waits for those attempts to finish; failures reach clients
345    /// viewing the session through the failure broadcasts.
346    pub(crate) async fn connect_session(&self, session: SessionId) {
347        self.ensure_session_runtimes_logged(session).await;
348        for attachment in self.session_runtimes(session).unwrap_or_default() {
349            if attachment.runtime_id.as_str() == RuntimeId::GLOBAL {
350                self.retry_global_mcp();
351            } else if let Some(connection) = self
352                .inner
353                .runtimes
354                .instance(&attachment)
355                .and_then(|instance| instance.connection())
356            {
357                connection.retry_failed_mcp();
358            }
359        }
360    }

362    /// Waits until the MCP servers of `session`'s attached runtime instances
363    /// have finished the connection attempts [`Self::connect_session`]
364    /// started, so a run's tool list is not built from a partial snapshot.
365    pub(crate) async fn await_session_mcp(&self, session: SessionId) {
366        for attachment in self.session_runtimes(session).unwrap_or_default() {
367            if let Some(connection) = self
368                .inner
369                .runtimes
370                .instance(&attachment)
371                .and_then(|instance| instance.connection())
372            {
373                connection.mcp_settled().await;
374            }
375        }
376    }

378    /// Retries the global runtime's failed MCP servers (see
379    /// [`Self::connect_session`]).
380    fn retry_global_mcp(&self) {
381        if let Some(runtime) = &self.inner.runtime {
382            self.inner.mcp.retry_failed(runtime);
383        }
384    }

386    /// [`Self::ensure_session_runtimes`] that reports failures on stderr:
387    /// the attachment then publishes no tools.
388    pub(crate) async fn ensure_session_runtimes_logged(&self, session: SessionId) {
389        match self.ensure_session_runtimes(session).await {
390            Ok(failures) => {
391                for (name, error) in failures {
392                    eprintln!(
393                        "cookie-agent: session {session}: runtime `{name}` is unavailable: {error}"
394                    );
395                }
396            }
397            Err(error) => {
398                eprintln!("cookie-agent: session {session}: runtimes not checked: {error}");
399            }
400        }
401    }

403    /// Attaches `capability` of `runtime_id` (its only capability when
404    /// `None`; at `workspace`, when the capability takes one) to `session`,
405    /// opening the instance when needed. Rejected while the session has an
406    /// active run; a failed start records nothing.
407    pub async fn attach_runtime(
408        &self,
409        session: SessionId,
410        runtime_id: &str,
411        capability: Option<&str>,
412        workspace: Option<PathBuf>,
413        origin: EventOrigin,
414    ) -> Result<RuntimeAttachment, EngineError> {
415        self.ensure_session_owned(session)?;
416        if self.inner.store.get(session)?.status == SessionStatus::Running {
417            return Err(EngineError::SessionRunning(session));
418        }
419        let instance = self
420            .ensure_runtime(runtime_id, capability, workspace.as_deref())
421            .await?;
422        if runtime_id == RuntimeId::GLOBAL {
423            self.retry_global_mcp();
424        } else if let Some(connection) = instance.connection() {
425            connection.retry_failed_mcp();
426        }
427        self.on_actor(session, move |engine| {
428            let projection = engine.inner.store.get(session)?;
429            if projection.status == SessionStatus::Running {
430                return Err(EngineError::SessionRunning(session));
431            }
432            let mut attachments = engine.attachments_of(&projection);
433            let attachment = push_instance(&mut attachments, &instance)?;
434            engine.append_direct(session, None, origin, attached_event(&attachment))?;
435            Ok(attachment)
436        })
437        .await
438    }

440    /// Removes the attachment `name` from `session`. Rejected while the
441    /// session has an active run.
442    pub async fn detach_runtime(
443        &self,
444        session: SessionId,
445        name: AttachmentName,
446        origin: EventOrigin,
447    ) -> Result<(), EngineError> {
448        self.ensure_session_owned(session)?;
449        self.on_actor(session, move |engine| {
450            let projection = engine.inner.store.get(session)?;
451            if projection.status == SessionStatus::Running {
452                return Err(EngineError::SessionRunning(session));
453            }
454            if !engine
455                .attachments_of(&projection)
456                .iter()
457                .any(|attachment| attachment.name == name)
458            {
459                return Err(EngineError::Runtime(format!(
460                    "session has no runtime attached as `{name}`"
461                )));
462            }
463            engine.append_direct(session, None, origin, Event::RuntimeDetached { name })?;
464            Ok(())
465        })
466        .await
467    }

518    /// Retries the instance of `capability` of `runtime_id` (its only
519    /// capability when `None`; at `workspace`, when the capability takes
520    /// one) now, as `runtimes.reconnect` and the `/runtime`
521    /// panel do: a failed instance starts again under a new generation, one
522    /// that is not running starts, and a ready or reconnecting one is left as
523    /// it is. Retrying an instance also retries the failed MCP servers it
524    /// hosts. Returns the instance afterwards.
525    pub async fn reconnect_runtime(
526        &self,
527        runtime_id: &str,
528        capability: Option<&str>,
529        workspace: Option<PathBuf>,
530    ) -> Result<RuntimeInstanceSummary, EngineError> {
531        let instance = self
532            .ensure_runtime(runtime_id, capability, workspace.as_deref())
533            .await?;
534        if runtime_id == RuntimeId::GLOBAL {
535            self.retry_global_mcp();
536        } else if let Some(connection) = instance.connection() {
537            connection.retry_failed_mcp();
538        }
539        self.inner
540            .runtimes
541            .summary(&instance)
542            .ok_or_else(|| EngineError::Runtime(format!("runtime `{runtime_id}` stopped")))
543    }
