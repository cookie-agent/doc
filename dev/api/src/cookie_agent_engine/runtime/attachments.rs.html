20/// A runtime ID reserved for one runtime dialing in, from
21/// [`Engine::claim_dial_in_runtime`] until it is accepted or dropped.
22pub struct DialInClaim {
23    engine: Engine,
24    template: Arc<crate::runtimes::DialInTemplate>,
25}
26
27impl DialInClaim {
28    /// Takes the runtime that dialed in over `transport`: initializes it,
29    /// makes it the runtime of the claimed ID, and reopens the instances
30    /// its sessions use.
31    pub async fn accept(
32        self,
33        transport: Box<dyn cookie_agent_protocol::Transport>,
34    ) -> Result<(), DialInRefusal> {
35        let inner = &self.engine.inner;
36        let runtime_id = crate::runtimes::RuntimeTemplate::runtime_id(self.template.as_ref());
37        let settings = crate::runtimes::runtime_settings(&inner.config.load(), runtime_id);
38        let client = self
39            .template
40            .connect(
41                cookie_agent_runtime_api::RuntimeConnection::new(transport),
42                &settings,
43                Some(Arc::clone(&inner.mcp_hub)),
44            )
45            .await
46            .map_err(|reason| DialInRefusal::Initialize(runtime_id.to_owned(), reason))?;
47        inner.runtimes.watch_connection(&self.template, &client);
48        inner.runtimes.revive(runtime_id);
49        Ok(())
50    }
51}
52
53impl Drop for DialInClaim {
54    fn drop(&mut self) {
55        self.template.release();
56    }
57}

93/// The name a session attaches an instance under: the name the instance
94/// reports, suffixed with `_2`, `_3`, ... while the session already has an
95/// attachment of that name.
96pub(crate) fn allocate_attachment_name(
97    existing: &[RuntimeAttachment],
98    reported: &str,
99) -> Result<AttachmentName, EngineError> {
100    let taken = |name: &str| {
101        existing
102            .iter()
103            .any(|attachment| attachment.name.as_str() == name)
104    };
105    let invalid = |name: &str| EngineError::Runtime(format!("invalid attachment name `{name}`"));
106    if !taken(reported) {
107        return AttachmentName::new(reported).map_err(|_| invalid(reported));
108    }
109    (2..=existing.len() + 1)
110        .map(|suffix| format!("{reported}_{suffix}"))
111        .find(|candidate| !taken(candidate))
112        .ok_or_else(|| invalid(reported))
113        .and_then(|name| AttachmentName::new(name.as_str()).map_err(|_| invalid(&name)))
114}

116/// Whether `attachment` refers to `instance`.
117fn refers_to(attachment: &RuntimeAttachment, instance: &RuntimeInstance) -> bool {
118    attachment.runtime_id.as_str() == instance.info().id()
119        && &attachment.capability == instance.info().capability()
120        && attachment.workspace.as_deref().map(std::path::Path::new)
121            == instance
122                .info()
123                .workspace()
124                .map(|workspace| workspace.root())
125}

127/// `instance` added to `attachments` under its reported name. Attaching one
128/// instance twice is an error.
129pub(crate) fn push_instance(
130    attachments: &mut Vec<RuntimeAttachment>,
131    instance: &RuntimeInstance,
132) -> Result<RuntimeAttachment, EngineError> {
133    if let Some(existing) = attachments
134        .iter()
135        .find(|attachment| refers_to(attachment, instance))
136    {
137        return Err(EngineError::Runtime(format!(
138            "capability `{}` of runtime `{}` is already attached as `{}`",
139            instance.info().capability(),
140            instance.info().id(),
141            existing.name
142        )));
143    }
144    let name = allocate_attachment_name(attachments, instance.info().attachment_name())?;
145    let attachment = instance.attachment(name)?;
146    attachments.push(attachment.clone());
147    Ok(attachment)
148}

188    /// The tool sources of a session with `attachments`: the base layer, then
189    /// each attachment's running instance, and after the first attachment of
190    /// a runtime that has tools of its own (its runtime-scoped MCP servers')
191    /// those, once per runtime however many of its workspaces the session
192    /// attaches. They compose under the runtime's ID and run without a
193    /// workspace. An attachment whose instance is not running contributes
194    /// nothing.
195    pub(crate) fn tool_sources_for(
196        &self,
197        attachments: &[RuntimeAttachment],
198    ) -> Vec<SourcedProvider> {
199        let mut sources = self
200            .inner
201            .runtimes
202            .base_providers()
203            .into_iter()
204            .map(|provider| SourcedProvider {
205                attachment: None,
206                generation: 0,
207                origin: crate::ToolOrigin::Base,
208                provider,
209            })
210            .collect::<Vec<_>>();
211        let mut shared_runtimes = std::collections::HashSet::new();
212        for attachment in attachments {
213            let Some(instance) = self.inner.runtimes.instance(attachment) else {
214                continue;
215            };
216            let origin = instance.origin();
217            let (generation, providers) = instance.sources();
218            sources.extend(providers.into_iter().map(|provider| SourcedProvider {
219                attachment: Some(attachment.name.clone()),
220                generation,
221                origin: origin.clone(),
222                provider,
223            }));
224            if let Some((generation, shared)) = instance.shared_source()
225                && shared_runtimes.insert(attachment.runtime_id.clone())
226            {
227                sources.push(SourcedProvider {
228                    attachment: Some(attachment.name.clone()),
229                    generation,
230                    origin: crate::ToolOrigin::Runtime(shared.info),
231                    provider: shared.provider,
232                });
233            }
234        }
235        sources
236    }

246    /// The instance of `capability` of `runtime_id` at `workspace` that a
247    /// session is about to attach, as creating a session, attaching a
248    /// runtime, and delegating with capabilities do: opened when it is not
249    /// running, and never over a runtime that is not online (see
250    /// `RuntimeRegistry::ensure`). A dial-in runtime this daemon has not
251    /// seen since it started, but remembers, is refused as not connected,
252    /// with when it was last seen.
253    pub(crate) async fn ensure_runtime(
254        &self,
255        runtime_id: &str,
256        capability: Option<&str>,
257        workspace: Option<&std::path::Path>,
258    ) -> Result<Arc<RuntimeInstance>, EngineError> {
259        if let Some(offline) = self.remembered_offline(runtime_id) {
260            return Err(EngineError::Runtime(offline));
261        }
262        self.inner
263            .runtimes
264            .ensure(runtime_id, capability, workspace)
265            .await
266    }

268    /// Why dial-in runtime `runtime_id` cannot be used, when this daemon
269    /// has not seen it connect since it started but remembers it: it is
270    /// not connected, last seen when its record says.
271    pub(crate) fn remembered_offline(&self, runtime_id: &str) -> Option<String> {
272        if self.inner.runtimes.knows(runtime_id) {
273            return None;
274        }
275        let record = self
276            .inner
277            .machines
278            .record(&RuntimeId::new(runtime_id).ok()?)?;
279        (record.kind == crate::machines::RecordKind::DialIn)
280            .then(|| crate::runtimes::not_connected(runtime_id, Some(record.last_seen)))
281    }

283    /// The attachments a new session gets for `runtimes`, starting the
284    /// instances that are not running.
285    ///
286    /// `acp`: the session runs an ACP agent, whose bridge offers cookie's
287    /// tools under their attachments' names and which takes the other
288    /// workspaces of its machine as additional directories, so the
289    /// one-attachment-per-tool-and-machine rule does not apply.
290    pub(crate) async fn resolve_session_runtimes(
291        &self,
292        runtimes: &SessionRuntimes,
293        acp: bool,
294    ) -> Result<Vec<RuntimeAttachment>, EngineError> {
295        let mut attachments = Vec::new();
296        if runtimes.global {
297            push_instance(&mut attachments, self.inner.runtimes.global())?;
298        }
299        match &runtimes.runtimes {
300            None => match &runtimes.cwd {
301                Some(cwd) => {
302                    let instance = self
303                        .ensure_runtime(
304                            RuntimeId::LOCAL,
305                            Some(cookie_agent_protocol::CapabilityName::CODING),
306                            Some(cwd),
307                        )
308                        .await?;
309                    push_instance(&mut attachments, &instance)?;
310                }
311                None => {
312                    push_instance(&mut attachments, self.inner.runtimes.local())?;
313                }
314            },
315            Some(requests) => {
316                // The session does not exist yet; tool lists do not depend
317                // on which session asks.
318                let context =
319                    cookie_agent_runtime_api::SessionToolContext::new(SessionId::new_v7());
320                for request in requests {
321                    let instance = self
322                        .ensure_runtime(
323                            &request.runtime_id,
324                            request.capability.as_deref(),
325                            request.workspace.as_deref(),
326                        )
327                        .await?;
328                    if !acp {
329                        self.refuse_overlap_with(&attachments, &instance, &context)?;
330                    }
331                    push_instance(&mut attachments, &instance)?;
332                }
333            }
334        }
335        Ok(attachments)
336    }

348    /// [`Self::ensure_session_runtimes`] for the attachments `wanted` keeps.
349    async fn ensure_session_runtimes_where(
350        &self,
351        session: SessionId,
352        wanted: impl Fn(&RuntimeAttachment) -> bool,
353    ) -> Result<Vec<(AttachmentName, EngineError)>, EngineError> {
354        let mut failures = Vec::new();
355        for attachment in self.session_runtimes(session)? {
356            if !wanted(&attachment) {
357                continue;
358            }
359            if let Err(error) = self.inner.runtimes.ensure_attachment(&attachment).await {
360                failures.push((attachment.name, error));
361            }
362        }
363        Ok(failures)
364    }

366    /// Connects what `session` uses, as when it is created or opened: every
367    /// attached instance is started, except on a runtime a client
368    /// disconnected, and every MCP server of an attached instance whose last
369    /// connection attempt failed is retried. The next run waits for those
370    /// attempts to finish; failures reach clients viewing the session
371    /// through the failure broadcasts.
372    pub(crate) async fn connect_session(&self, session: SessionId) {
373        let result = self
374            .ensure_session_runtimes_where(session, |attachment| {
375                !self
376                    .inner
377                    .runtimes
378                    .is_parked(attachment.runtime_id.as_str())
379            })
380            .await;
381        Self::log_session_runtimes(session, result);
382        for attachment in self.session_runtimes(session).unwrap_or_default() {
383            if attachment.runtime_id.as_str() == RuntimeId::GLOBAL {
384                self.retry_global_mcp();
385            } else if let Some(connection) = self
386                .inner
387                .runtimes
388                .instance(&attachment)
389                .and_then(|instance| instance.connection())
390            {
391                connection.retry_failed_mcp();
392            }
393        }
394    }

396    /// Waits until the MCP servers of `session`'s attached runtime instances
397    /// have finished the connection attempts [`Self::connect_session`]
398    /// started, so a run's tool list is not built from a partial snapshot.
399    pub(crate) async fn await_session_mcp(&self, session: SessionId) {
400        for attachment in self.session_runtimes(session).unwrap_or_default() {
401            if let Some(connection) = self
402                .inner
403                .runtimes
404                .instance(&attachment)
405                .and_then(|instance| instance.connection())
406            {
407                connection.mcp_settled().await;
408            }
409        }
410    }

412    /// Reconnects the failed MCP servers of `session`'s attachments that a
413    /// tool it loaded comes from, and waits for the attempts (deferred tools
414    /// in cookie, section 11), so a loaded tool does not leave the request
415    /// for a server that would come back. Servers nobody loaded from are
416    /// left to the session's own retries.
417    ///
418    /// The attempts run together, and each server's go through
419    /// [`LoadedMcpRetries`]: one at a time whoever asks, and at most one per
420    /// [`LOADED_MCP_RETRY_INTERVAL`], so a dead server costs a step one
421    /// attempt now and then rather than one on every step of every session.
422    pub(crate) async fn reconnect_loaded_mcp(&self, session: SessionId) {
423        let Ok(stored) = self.inner.store.get(session) else {
424            return;
425        };
426        let fold = stored.log.deferral();
427        drop(stored);
428        // A loaded tool is announced under its server's name.
429        let servers = fold
430            .announced
431            .iter()
432            .filter(|(_, group)| group.tools.iter().any(|tool| fold.loaded.contains(tool)))
433            .map(|(name, _)| name.as_str())
434            .collect::<std::collections::HashSet<_>>();
435        if servers.is_empty() {
436            return;
437        }
438        let retries = &self.inner.loaded_mcp_retries;
439        let mut attempts = Vec::<futures_util::future::BoxFuture<'_, ()>>::new();
440        for attachment in self.session_runtimes(session).unwrap_or_default() {
441            if attachment.runtime_id.as_str() == RuntimeId::GLOBAL {
442                for status in self.inner.mcp.statuses() {
443                    if status.state == crate::McpServerState::Failed
444                        && servers.contains(status.server.as_str())
445                    {
446                        let key = LoadedMcpKey {
447                            instance: None,
448                            server: status.server.clone(),
449                        };
450                        attempts.push(Box::pin(retries.attempt(key, async move {
451                            let _ = self.inner.mcp.reconnect_server(&status.server).await;
452                        })));
453                    }
454                }
455            } else if let Some(connection) = self
456                .inner
457                .runtimes
458                .instance(&attachment)
459                .and_then(|instance| instance.connection())
460            {
461                for server in connection.mcp_servers() {
462                    if server.report.state
463                        == cookie_agent_runtime_api::protocol::McpServerState::Failed
464                        && servers.contains(server.report.name.as_str())
465                    {
466                        let key = LoadedMcpKey {
467                            instance: Some((
468                                attachment.runtime_id.to_string(),
469                                attachment.capability.to_string(),
470                                attachment.workspace.clone(),
471                            )),
472                            server: server.report.name.clone(),
473                        };
474                        let connection = connection.clone();
475                        attempts.push(Box::pin(retries.attempt(key, async move {
476                            let _ = connection.reconnect_mcp(&server.report.name).await;
477                        })));
478                    }
479                }
480            }
481        }
482        futures_util::future::join_all(attempts).await;
483    }

518    /// Attaches `capability` of `runtime_id` (its only capability when
519    /// `None`; at `workspace`, when the capability takes one) to `session`,
520    /// opening the instance when needed. Rejected while the session has an
521    /// active run; a failed start records nothing.
522    pub async fn attach_runtime(
523        &self,
524        session: SessionId,
525        runtime_id: &str,
526        capability: Option<&str>,
527        workspace: Option<PathBuf>,
528        origin: EventOrigin,
529    ) -> Result<RuntimeAttachment, EngineError> {
530        self.ensure_session_owned(session)?;
531        if self.inner.store.get(session)?.status == SessionStatus::Running {
532            return Err(EngineError::SessionRunning(session));
533        }
534        self.refuse_second_acp_runtime(session, runtime_id)?;
535        let instance = self
536            .ensure_runtime(runtime_id, capability, workspace.as_deref())
537            .await?;
538        if runtime_id == RuntimeId::GLOBAL {
539            self.retry_global_mcp();
540        } else if let Some(connection) = instance.connection() {
541            connection.retry_failed_mcp();
542        }
543        let runtime_id_owned = runtime_id.to_owned();
544        let attachment = self
545            .on_actor(session, move |engine| {
546                let projection = engine.inner.store.get(session)?;
547                if projection.status == SessionStatus::Running {
548                    return Err(EngineError::SessionRunning(session));
549                }
550                // Checked on the actor, so of two attaches under way only
551                // the first can pass.
552                engine.refuse_second_acp_runtime(session, runtime_id_owned.as_str())?;
553                let mut attachments = engine.attachments_of(&projection);
554                if engine.session_acp_offers(session)?.is_none() {
555                    engine.refuse_overlap_with(
556                        &attachments,
557                        &instance,
558                        &cookie_agent_runtime_api::SessionToolContext::new(session),
559                    )?;
560                }
561                let attachment = push_instance(&mut attachments, &instance)?;
562                engine.append_direct(session, None, origin, attached_event(&attachment))?;
563                Ok(attachment)
564            })
565            .await?;
566        self.announce_tool_changes(session).await;
567        Ok(attachment)
568    }

570    /// Removes the attachment `name` from `session`. Rejected while the
571    /// session has an active run.
572    pub async fn detach_runtime(
573        &self,
574        session: SessionId,
575        name: AttachmentName,
576        origin: EventOrigin,
577    ) -> Result<(), EngineError> {
578        self.ensure_session_owned(session)?;
579        self.on_actor(session, move |engine| {
580            let projection = engine.inner.store.get(session)?;
581            if projection.status == SessionStatus::Running {
582                return Err(EngineError::SessionRunning(session));
583            }
584            if !engine
585                .attachments_of(&projection)
586                .iter()
587                .any(|attachment| attachment.name == name)
588            {
589                return Err(EngineError::Runtime(format!(
590                    "session has no runtime attached as `{name}`"
591                )));
592            }
593            engine.append_direct(session, None, origin, Event::RuntimeDetached { name })?;
594            Ok(())
595        })
596        .await?;
597        self.announce_tool_changes(session).await;
598        Ok(())
599    }

650    /// Retries the instance of `capability` of `runtime_id` (its only
651    /// capability when `None`; at `workspace`, when the capability takes
652    /// one) now, as `runtimes.reconnect` and the `/runtime`
653    /// panel do: a failed instance starts again under a new generation, one
654    /// that is not running starts, and a ready or reconnecting one is left as
655    /// it is. Retrying an instance also retries the failed MCP servers it
656    /// hosts. Returns the instance afterwards.
657    pub async fn reconnect_runtime(
658        &self,
659        runtime_id: &str,
660        capability: Option<&str>,
661        workspace: Option<PathBuf>,
662    ) -> Result<RuntimeInstanceSummary, EngineError> {
663        let instance = self
664            .ensure_runtime(runtime_id, capability, workspace.as_deref())
665            .await?;
666        if runtime_id == RuntimeId::GLOBAL {
667            self.retry_global_mcp();
668        } else if let Some(connection) = instance.connection() {
669            connection.retry_failed_mcp();
670        }
671        self.inner
672            .runtimes
673            .summary(&instance)
674            .ok_or_else(|| EngineError::Runtime(format!("runtime `{runtime_id}` stopped")))
675    }
676
677    /// Connects template runtime `runtime_id` without opening an instance
678    /// (`runtimes.connect`), to try that it starts. A connected runtime is
679    /// left as it is; the error says why it did not connect.
680    pub async fn connect_runtime(&self, runtime_id: &str) -> Result<(), EngineError> {
681        self.inner.runtimes.connect(runtime_id).await
682    }
683
684    /// Disconnects template runtime `runtime_id` (`runtimes.disconnect`):
685    /// its instances stop, and opening a session leaves it down until a
686    /// run, attachment, new session, or delegation uses it. Refused while
687    /// it is busy ([`Self::runtime_activity`]).
688    pub async fn disconnect_runtime(&self, runtime_id: &str) -> Result<(), EngineError> {
689        let activity = self
690            .runtime_activity()
691            .remove(runtime_id)
692            .unwrap_or_default();
693        if !activity.is_idle() {
694            return Err(EngineError::Runtime(format!(
695                "runtime `{runtime_id}` is in use ({activity}); disconnect it once they end"
696            )));
697        }
698        let connections = self.inner.runtimes.disconnect(runtime_id)?;
699        futures_util::future::join_all(
700            connections
701                .iter()
702                .map(crate::runtimes::RuntimeClient::shutdown),
703        )
704        .await;
705        Ok(())
706    }
707
708    /// What keeps each runtime busy, by runtime ID; a runtime nothing uses
709    /// is left out, and so are the built-in `global` and `local`, which are
710    /// never disconnected and which nearly every run uses. A run counts
711    /// against every runtime its session attaches (an ACP agent runs on one
712    /// of them), and a background call against the runtime it executes on.
713    #[must_use]
714    pub fn runtime_activity(&self) -> std::collections::BTreeMap<String, crate::RuntimeActivity> {
715        let sessions = self
716            .inner
717            .sessions
718            .active
719            .lock()
720            .unwrap_or_else(|poisoned| poisoned.into_inner())
721            .values()
722            .map(|run| run.session)
723            .collect::<Vec<_>>();
724        let mut activity = std::collections::BTreeMap::<String, crate::RuntimeActivity>::new();
725        for session in sessions {
726            let mut runtimes = self
727                .session_runtimes(session)
728                .unwrap_or_default()
729                .into_iter()
730                .map(|attachment| attachment.runtime_id.as_str().to_owned())
731                .collect::<Vec<_>>();
732            runtimes.sort_unstable();
733            runtimes.dedup();
734            for runtime in runtimes {
735                activity.entry(runtime).or_default().active_runs += 1;
736            }
737        }
738        for runtime in self
739            .inner
740            .background
741            .pending_runtimes()
742            .into_iter()
743            .chain(self.acp_background_runtimes())
744        {
745            activity.entry(runtime).or_default().background_calls += 1;
746        }
747        activity.retain(|runtime, _| {
748            runtime != cookie_agent_protocol::RuntimeId::GLOBAL
749                && runtime != cookie_agent_protocol::RuntimeId::LOCAL
750        });
751        activity
752    }
753
754    /// Signals whenever a run starts or ends or a background call changes,
755    /// which may change [`Self::runtime_activity`]; bursts coalesce.
756    #[must_use]
757    pub fn subscribe_runtime_activity(&self) -> tokio::sync::watch::Receiver<()> {
758        let (sender, receiver) = tokio::sync::watch::channel(());
759        let mut runs = self.inner.sessions.active.subscribe_changed();
760        let mut background = self.subscribe_background_changes();
761        let task = async move {
762            loop {
763                tokio::select! {
764                    () = sender.closed() => return,
765                    changed = runs.changed() => if changed.is_err() { return },
766                    changed = background.recv() => match changed {
767                        Ok(_) | Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => {}
768                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
769                    },
770                }
771                sender.send_replace(());
772            }
773        };
774        match &self.inner.runtime {
775            Some(runtime) => drop(runtime.spawn(task)),
776            None => drop(tokio::spawn(task)),
777        }
778        receiver
779    }

805/// The least time between two reconnects of one failed MCP server that
806/// [`Engine::reconnect_loaded_mcp`] makes.
807pub(crate) const LOADED_MCP_RETRY_INTERVAL: std::time::Duration =
808    std::time::Duration::from_secs(30);
809
810/// A server [`LoadedMcpRetries`] reconnects: the global runtime's, or one
811/// of a runtime instance's, by its runtime, capability, and workspace,
812/// whichever sessions attach it.
813#[derive(Clone, Debug, Eq, Hash, PartialEq)]
814pub(crate) struct LoadedMcpKey {
815    pub(crate) instance: Option<(String, String, Option<String>)>,
816    pub(crate) server: String,
817}
818
819/// The reconnects of failed MCP servers that loaded tools come from, by
820/// server: one in flight at a time, the callers that come while it is
821/// sharing it, and none sooner than [`LOADED_MCP_RETRY_INTERVAL`] after the
822/// one before.
823#[derive(Default)]
824pub(crate) struct LoadedMcpRetries {
825    /// When each server's latest attempt started.
826    servers: std::sync::Mutex<
827        std::collections::HashMap<
828            LoadedMcpKey,
829            Arc<tokio::sync::Mutex<Option<tokio::time::Instant>>>,
830        >,
831    >,
832}
833
834impl LoadedMcpRetries {
835    /// Runs `attempt` for `key` unless one started within the interval; a
836    /// caller that comes while one runs waits for it and makes none.
837    pub(crate) async fn attempt(
838        &self,
839        key: LoadedMcpKey,
840        attempt: impl std::future::Future<Output = ()>,
841    ) {
842        let slot = Arc::clone(
843            self.servers
844                .lock()
845                .unwrap_or_else(|poisoned| poisoned.into_inner())
846                .entry(key)
847                .or_default(),
848        );
849        let mut started = slot.lock().await;
850        if started.is_some_and(|started| started.elapsed() < LOADED_MCP_RETRY_INTERVAL) {
851            return;
852        }
853        *started = Some(tokio::time::Instant::now());
854        attempt.await;
855    }
856}

904#[cfg(test)]
905mod loaded_mcp_tests {
906    use std::sync::atomic::{AtomicU32, Ordering};
907
908    use super::*;
909
910    fn key(server: &str) -> LoadedMcpKey {
911        LoadedMcpKey {
912            instance: None,
913            server: server.to_owned(),
914        }
915    }
916
917    #[tokio::test(start_paused = true)]
918    async fn a_failed_servers_reconnect_is_shared_and_spaced_out() {
919        let retries = LoadedMcpRetries::default();
920        let attempts = AtomicU32::new(0);
921        let attempt = || async {
922            attempts.fetch_add(1, Ordering::SeqCst);
923            tokio::time::sleep(std::time::Duration::from_secs(4)).await;
924        };
925
926        // Callers that come together share one attempt, and wait for it.
927        let started = tokio::time::Instant::now();
928        futures_util::future::join3(
929            retries.attempt(key("docs"), attempt()),
930            retries.attempt(key("docs"), attempt()),
931            retries.attempt(key("docs"), attempt()),
932        )
933        .await;
934        assert_eq!(attempts.load(Ordering::SeqCst), 1);
935        assert_eq!(started.elapsed(), std::time::Duration::from_secs(4));
936
937        // Another server is its own.
938        retries.attempt(key("search"), attempt()).await;
939        assert_eq!(attempts.load(Ordering::SeqCst), 2);
940
941        // Within the interval nothing is tried, and nothing waits.
942        let started = tokio::time::Instant::now();
943        retries.attempt(key("docs"), attempt()).await;
944        assert_eq!(attempts.load(Ordering::SeqCst), 2);
945        assert_eq!(started.elapsed(), std::time::Duration::ZERO);
946
947        // After it, the server is tried again.
948        tokio::time::advance(LOADED_MCP_RETRY_INTERVAL).await;
949        retries.attempt(key("docs"), attempt()).await;
950        assert_eq!(attempts.load(Ordering::SeqCst), 3);
951    }
