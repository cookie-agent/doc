45/// The name a session attaches an instance under: the name the instance
46/// reports, suffixed with `_2`, `_3`, ... while the session already has an
47/// attachment of that name.
48pub(crate) fn allocate_attachment_name(
49    existing: &[RuntimeAttachment],
50    reported: &str,
51) -> Result<AttachmentName, EngineError> {
52    let taken = |name: &str| {
53        existing
54            .iter()
55            .any(|attachment| attachment.name.as_str() == name)
56    };
57    let invalid = |name: &str| EngineError::Runtime(format!("invalid attachment name `{name}`"));
58    if !taken(reported) {
59        return AttachmentName::new(reported).map_err(|_| invalid(reported));
60    }
61    (2..=existing.len() + 1)
62        .map(|suffix| format!("{reported}_{suffix}"))
63        .find(|candidate| !taken(candidate))
64        .ok_or_else(|| invalid(reported))
65        .and_then(|name| AttachmentName::new(name.as_str()).map_err(|_| invalid(&name)))
66}

68/// Whether `attachment` refers to `instance`.
69fn refers_to(attachment: &RuntimeAttachment, instance: &RuntimeInstance) -> bool {
70    attachment.runtime_id.as_str() == instance.info().id()
71        && attachment.workspace.as_deref().map(std::path::Path::new)
72            == instance
73                .info()
74                .workspace()
75                .map(|workspace| workspace.root())
76}
77
78/// `instance` added to `attachments` under its reported name. Attaching one
79/// instance twice is an error.
80fn push_instance(
81    attachments: &mut Vec<RuntimeAttachment>,
82    instance: &RuntimeInstance,
83) -> Result<RuntimeAttachment, EngineError> {
84    if let Some(existing) = attachments
85        .iter()
86        .find(|attachment| refers_to(attachment, instance))
87    {
88        return Err(EngineError::Runtime(format!(
89            "runtime `{}` is already attached as `{}`",
90            instance.info().id(),
91            existing.name
92        )));
93    }
94    let name = allocate_attachment_name(attachments, instance.info().attachment_name())?;
95    let attachment = instance.attachment(name)?;
96    attachments.push(attachment.clone());
97    Ok(attachment)
98}

100/// The events a new session's creation writes for `attachments`. A log
101/// without any attach or detach event reads as attached to `global` and
102/// `local`, so a session created with no attachment records the detach of
103/// `global` instead.
104pub(crate) fn initial_attachment_events(attachments: &[RuntimeAttachment]) -> Vec<Event> {
105    if attachments.is_empty() {
106        return vec![Event::RuntimeDetached {
107            name: AttachmentName::new(cookie_agent_protocol::RuntimeId::GLOBAL)
108                .expect("`global` is a valid attachment name"),
109        }];
110    }
111    attachments.iter().map(attached_event).collect()
112}

114pub(crate) fn attached_event(attachment: &RuntimeAttachment) -> Event {
115    Event::RuntimeAttached {
116        name: attachment.name.clone(),
117        runtime_id: attachment.runtime_id.clone(),
118        kind: attachment.kind.clone(),
119        workspace: attachment.workspace.clone(),
120    }

124    /// What a new session attaches by default, and what a session whose log
125    /// predates runtimes is attached to: `global`, then `local`.
126    pub(crate) fn default_attachments(&self) -> Vec<RuntimeAttachment> {
127        let mut attachments = Vec::with_capacity(2);
128        for instance in [self.inner.runtimes.global(), self.inner.runtimes.local()] {
129            push_instance(&mut attachments, instance).expect("built-in runtimes attach");
130        }
131        attachments
132    }

135    pub(crate) fn attachments_of(&self, projection: &SessionProjection) -> Vec<RuntimeAttachment> {
136        projection
137            .runtime_attachments
138            .attachments()
139            .map_or_else(|| self.default_attachments(), <[_]>::to_vec)
140    }

150    /// The tool sources of a session with `attachments`: the base layer, then
151    /// each attachment's running instance. An attachment whose instance is
152    /// not running contributes nothing.
153    pub(crate) fn tool_sources_for(
154        &self,
155        attachments: &[RuntimeAttachment],
156    ) -> Vec<SourcedProvider> {
157        let mut sources = self
158            .inner
159            .runtimes
160            .base_providers()
161            .into_iter()
162            .map(|provider| SourcedProvider {
163                attachment: None,
164                generation: 0,
165                origin: crate::ToolOrigin::Base,
166                provider,
167            })
168            .collect::<Vec<_>>();
169        for attachment in attachments {
170            let Some(instance) = self.inner.runtimes.instance(attachment) else {
171                continue;
172            };
173            let origin = instance.origin();
174            let (generation, providers) = instance.sources();
175            sources.extend(providers.into_iter().map(|provider| SourcedProvider {
176                attachment: Some(attachment.name.clone()),
177                generation,
178                origin: origin.clone(),
179                provider,
180            }));
181        }
182        sources
183    }

193    /// The attachments a new session gets for `runtimes`, starting the
194    /// instances that are not running.
195    pub(crate) async fn resolve_session_runtimes(
196        &self,
197        runtimes: &SessionRuntimes,
198    ) -> Result<Vec<RuntimeAttachment>, EngineError> {
199        let mut attachments = Vec::new();
200        if runtimes.global {
201            push_instance(&mut attachments, self.inner.runtimes.global())?;
202        }
203        match &runtimes.runtimes {
204            None => {
205                push_instance(&mut attachments, self.inner.runtimes.local())?;
206            }
207            Some(requests) => {
208                for request in requests {
209                    let instance = self
210                        .inner
211                        .runtimes
212                        .ensure(&request.runtime_id, request.workspace.as_deref())
213                        .await?;
214                    push_instance(&mut attachments, &instance)?;
215                }
216            }
217        }
218        Ok(attachments)
219    }

221    /// Makes sure every instance `session` is attached to is running, as when
222    /// the session is opened in a daemon that has not started them. Returns
223    /// the attachments that could not be started.
224    pub(crate) async fn ensure_session_runtimes(
225        &self,
226        session: SessionId,
227    ) -> Result<Vec<(AttachmentName, EngineError)>, EngineError> {
228        let mut failures = Vec::new();
229        for attachment in self.session_runtimes(session)? {
230            if let Err(error) = self.inner.runtimes.ensure_attachment(&attachment).await {
231                failures.push((attachment.name, error));
232            }
233        }
234        Ok(failures)
235    }

237    /// Connects what `session` uses, as when it is created or opened: every
238    /// attached instance is started, and every MCP server of an attached
239    /// instance whose last connection attempt failed is retried. The next
240    /// run waits for those attempts to finish; failures reach clients
241    /// viewing the session through the failure broadcasts.
242    pub(crate) async fn connect_session(&self, session: SessionId) {
243        self.ensure_session_runtimes_logged(session).await;
244        for attachment in self.session_runtimes(session).unwrap_or_default() {
245            if attachment.runtime_id.as_str() == RuntimeId::GLOBAL {
246                self.retry_global_mcp();
247            } else if let Some(connection) = self
248                .inner
249                .runtimes
250                .instance(&attachment)
251                .and_then(|instance| instance.connection())
252            {
253                connection.retry_failed_mcp();
254            }
255        }
256    }

258    /// Waits until the MCP servers of `session`'s attached runtime instances
259    /// have finished the connection attempts [`Self::connect_session`]
260    /// started, so a run's tool list is not built from a partial snapshot.
261    pub(crate) async fn await_session_mcp(&self, session: SessionId) {
262        for attachment in self.session_runtimes(session).unwrap_or_default() {
263            if let Some(connection) = self
264                .inner
265                .runtimes
266                .instance(&attachment)
267                .and_then(|instance| instance.connection())
268            {
269                connection.mcp_settled().await;
270            }
271        }
272    }

274    /// Retries the global runtime's failed MCP servers (see
275    /// [`Self::connect_session`]).
276    fn retry_global_mcp(&self) {
277        if let Some(runtime) = &self.inner.runtime {
278            self.inner.mcp.retry_failed(runtime);
279        }
280    }

282    /// [`Self::ensure_session_runtimes`] that reports failures on stderr:
283    /// the attachment then publishes no tools.
284    pub(crate) async fn ensure_session_runtimes_logged(&self, session: SessionId) {
285        match self.ensure_session_runtimes(session).await {
286            Ok(failures) => {
287                for (name, error) in failures {
288                    eprintln!(
289                        "cookie-agent: session {session}: runtime `{name}` is unavailable: {error}"
290                    );
291                }
292            }
293            Err(error) => {
294                eprintln!("cookie-agent: session {session}: runtimes not checked: {error}");
295            }
296        }
297    }

299    /// Attaches the instance of `runtime_id` (at `workspace`, when its
300    /// template takes one) to `session`, starting it when needed. Rejected
301    /// while the session has an active run; a failed start records nothing.
302    pub async fn attach_runtime(
303        &self,
304        session: SessionId,
305        runtime_id: &str,
306        workspace: Option<PathBuf>,
307        origin: EventOrigin,
308    ) -> Result<RuntimeAttachment, EngineError> {
309        self.ensure_session_owned(session)?;
310        if self.inner.store.get(session)?.status == SessionStatus::Running {
311            return Err(EngineError::SessionRunning(session));
312        }
313        let instance = self
314            .inner
315            .runtimes
316            .ensure(runtime_id, workspace.as_deref())
317            .await?;
318        if runtime_id == RuntimeId::GLOBAL {
319            self.retry_global_mcp();
320        } else if let Some(connection) = instance.connection() {
321            connection.retry_failed_mcp();
322        }
323        self.on_actor(session, move |engine| {
324            let projection = engine.inner.store.get(session)?;
325            if projection.status == SessionStatus::Running {
326                return Err(EngineError::SessionRunning(session));
327            }
328            let mut attachments = engine.attachments_of(&projection);
329            let attachment = push_instance(&mut attachments, &instance)?;
330            engine.materialize_legacy_attachments(session, &projection, &origin)?;
331            engine.append_direct(session, None, origin, attached_event(&attachment))?;
332            Ok(attachment)
333        })
334        .await
335    }

337    /// Removes the attachment `name` from `session`. Rejected while the
338    /// session has an active run.
339    pub async fn detach_runtime(
340        &self,
341        session: SessionId,
342        name: AttachmentName,
343        origin: EventOrigin,
344    ) -> Result<(), EngineError> {
345        self.ensure_session_owned(session)?;
346        self.on_actor(session, move |engine| {
347            let projection = engine.inner.store.get(session)?;
348            if projection.status == SessionStatus::Running {
349                return Err(EngineError::SessionRunning(session));
350            }
351            if !engine
352                .attachments_of(&projection)
353                .iter()
354                .any(|attachment| attachment.name == name)
355            {
356                return Err(EngineError::Runtime(format!(
357                    "session has no runtime attached as `{name}`"
358                )));
359            }
360            engine.materialize_legacy_attachments(session, &projection, &origin)?;
361            engine.append_direct(session, None, origin, Event::RuntimeDetached { name })?;
362            Ok(())
363        })
364        .await
365    }

367    /// A log that predates runtimes holds no attachment event and reads as
368    /// attached to `global` and `local`. Before its first attach or detach
369    /// those implicit attachments are written, so the fold stays the source
370    /// of truth.
371    fn materialize_legacy_attachments(
372        &self,
373        session: SessionId,
374        projection: &SessionProjection,
375        origin: &EventOrigin,
376    ) -> Result<(), EngineError> {
377        if projection.runtime_attachments.attachments().is_some() {
378            return Ok(());
379        }
380        for attachment in self.default_attachments() {
381            self.append_direct(session, None, origin.clone(), attached_event(&attachment))?;
382        }
383        Ok(())
384    }

386    /// Rejects a delegation narrowing that names an attachment the parent
387    /// does not have.
388    pub(crate) fn validate_delegated_runtimes(
389        &self,
390        parent: SessionId,
391        runtimes: &[AttachmentName],
392    ) -> Result<(), EngineError> {
393        let attached = self.session_runtimes(parent)?;
394        if let Some(unknown) = runtimes
395            .iter()
396            .find(|name| !attached.iter().any(|attachment| &attachment.name == *name))
397        {
398            let names = attached
399                .iter()
400                .map(|attachment| attachment.name.as_str())
401                .collect::<Vec<_>>()
402                .join(", ");
403            return Err(EngineError::ToolFailed(format!(
404                "unknown runtime attachment `{unknown}`; this session has: {names}"
405            )));
406        }
407        Ok(())
408    }

410    /// The attachments a delegated child of `parent` starts with: the
411    /// parent's, in its order, narrowed to `runtimes` when given.
412    pub(crate) fn delegated_attachments(
413        &self,
414        parent: &SessionProjection,
415        runtimes: Option<&[AttachmentName]>,
416    ) -> Vec<RuntimeAttachment> {
417        let mut attachments = self.attachments_of(parent);
418        if let Some(runtimes) = runtimes {
419            attachments.retain(|attachment| runtimes.contains(&attachment.name));
420        }
421        attachments
422    }

447    /// Retries the instance of `runtime_id` (at `workspace`, when its
448    /// template takes one) now, as `runtimes.reconnect` and the `/runtime`
449    /// panel do: a failed instance starts again under a new generation, one
450    /// that is not running starts, and a ready or reconnecting one is left as
451    /// it is. Retrying an instance also retries the failed MCP servers it
452    /// hosts. Returns the instance afterwards.
453    pub async fn reconnect_runtime(
454        &self,
455        runtime_id: &str,
456        workspace: Option<PathBuf>,
457    ) -> Result<RuntimeInstanceSummary, EngineError> {
458        let instance = self
459            .inner
460            .runtimes
461            .ensure(runtime_id, workspace.as_deref())
462            .await?;
463        if runtime_id == RuntimeId::GLOBAL {
464            self.retry_global_mcp();
465        } else if let Some(connection) = instance.connection() {
466            connection.retry_failed_mcp();
467        }
468        self.inner
469            .runtimes
470            .summary(&instance)
471            .ok_or_else(|| EngineError::Runtime(format!("runtime `{runtime_id}` stopped")))
472    }
