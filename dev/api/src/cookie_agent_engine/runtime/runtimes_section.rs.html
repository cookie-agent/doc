11const RUNTIMES_GUIDANCE: &str = concat!(
12    "File tools and bash of a coding runtime resolve relative paths against its workspace. ",
13    "Use workspace-relative paths for files inside it. ",
14    "When several runtimes offer a tool with one name, each is published as ",
15    "`<runtime name>_<tool>`; `prefixed=\"true\"` marks a runtime whose tools carry that prefix.",
16);

18impl Engine {
19    /// The `<runtimes>` block lists the session's attachments: name, kind,
20    /// workspace and platform for a coding runtime, and whether its tools
21    /// are published with the attachment prefix at admission. It is frozen
22    /// with the composed prompt.
23    pub(crate) fn compose_runtimes_section(
24        &self,
25        policy: &mut FrozenRunPolicy,
26        session: SessionId,
27    ) {
28        let attachments = match self.session_runtimes(session) {
29            Ok(attachments) => attachments,
30            Err(error) => {
31                eprintln!("cookie-agent: skipping runtimes prompt section: {error}");
32                return;
33            }
34        };
35        let prefixed = self
36            .compose_for_policy(session, policy)
37            .map(|composition| {
38                composition
39                    .tools
40                    .iter()
41                    .filter(|tool| tool.name != tool.spec.name)
42                    .filter_map(|tool| tool.source.attachment.clone())
43                    .collect::<HashSet<_>>()
44            })
45            .unwrap_or_default();
46        let mut block = String::from("<runtimes>\n");
47        for attachment in &attachments {
48            block.push_str(&format!(
49                "<runtime name=\"{}\" kind=\"{}\"",
50                escape_xml_attribute(attachment.name.as_str()),
51                attachment.kind.as_str()
52            ));
53            if let Some(workspace) = &attachment.workspace {
54                block.push_str(&format!(
55                    " workspace=\"{}\"",
56                    escape_xml_attribute(workspace)
57                ));
58            }
59            if let Some(platform) = self
60                .inner
61                .runtimes
62                .instance(attachment)
63                .and_then(|instance| {
64                    instance
65                        .info()
66                        .workspace()
67                        .map(|workspace| workspace.platform().to_owned())
68                })
69            {
70                block.push_str(&format!(
71                    " platform=\"{}\"",
72                    escape_xml_attribute(&platform)
73                ));
74            }
75            block.push_str(&format!(
76                " prefixed=\"{}\"/>\n",
77                prefixed.contains(&attachment.name)
78            ));
79        }
80        block.push_str("</runtimes>\n");
81        block.push_str(RUNTIMES_GUIDANCE);
82        let Some(composed_bytes) = policy
83            .agent
84            .composed_prompt
85            .len()
86            .checked_add(block.len() + PROMPT_BLOCK_SEPARATOR.len())
87        else {
88            eprintln!(
89                "cookie-agent: skipping runtimes prompt section: composed prompt byte count overflowed"
90            );
91            return;
92        };
93        if composed_bytes > AgentSnapshot::MAX_PROMPT_BYTES {
94            eprintln!(
95                "cookie-agent: skipping runtimes prompt section: composed prompt exceeds {} bytes",
96                AgentSnapshot::MAX_PROMPT_BYTES
97            );
98            return;
99        }
100
101        push_prompt_block(&mut policy.agent.composed_prompt, &block);
102        policy.agent.prompt_fingerprint =
103            Sha256Digest::of_bytes(policy.agent.composed_prompt.as_bytes());
104        let mut document_fingerprint = policy
105            .agent
106            .document_fingerprint
107            .as_str()
108            .as_bytes()
109            .to_vec();
110        document_fingerprint.extend_from_slice(block.as_bytes());
111        policy.agent.document_fingerprint = Sha256Digest::of_bytes(&document_fingerprint);
112    }
113}

115fn escape_xml_attribute(value: &str) -> String {
116    value
117        .replace('&', "&amp;")
118        .replace('<', "&lt;")
119        .replace('>', "&gt;")
120        .replace('"', "&quot;")
121}
