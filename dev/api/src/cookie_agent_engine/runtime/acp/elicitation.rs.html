1//! An ACP agent's questions to the user: `elicitation/create` and
2//! `elicitation/complete` (ACP v1 elicitation; `docs/site/specs/acp-agents.md`
3//! section 15).
4//!
5//! Cookie advertises form and URL elicitation. A session-scoped
6//! `elicitation/create` that arrives while a turn of its session runs
7//! becomes a question of that run: `elicitation_requested`, a wait for the
8//! user's `elicitation.respond`, and `elicitation_resolved`, whose action is
9//! the agent's answer. A question is the agent asking the user, not an
10//! action, so no permission rule or mode answers it; a session no one
11//! attends (permission mode `auto_approve_n`, a cron run's default, or a
12//! headless `cookie run`) declines it at once, the run's cancel or end
13//! cancels it, and the agent's `elicitation/complete` for a URL question
14//! still waiting accepts it. Anything else, a request-scoped question or
15//! one outside a turn, is cancelled; a mode other than form and URL is
16//! refused as invalid params, as ACP asks.
17

90/// A name's sort key in natural order: runs of digits compare as numbers,
91/// so `question_2` comes before `question_10`.
92fn natural_key(name: &str) -> Vec<(bool, u128, String)> {
93    let mut key = Vec::new();
94    let mut chars = name.chars().peekable();
95    while let Some(&first) = chars.peek() {
96        let digits = first.is_ascii_digit();
97        let mut run = String::new();
98        while let Some(&c) = chars.peek() {
99            if c.is_ascii_digit() != digits {
100                break;
101            }
102            run.push(c);
103            chars.next();
104        }
105        let number = if digits {
106            run.parse().unwrap_or(u128::MAX)
107        } else {
108            0
109        };
110        key.push((digits, number, run));
111    }
112    key
113}

128/// The choices of an `oneOf`/`anyOf` list of `{const, title}`, else of an
129/// `enum` of strings (titled by `enumNames` when the agent sends them).
130fn choices(schema: &Value, titled: &str) -> Option<Vec<ElicitationChoice>> {
131    if let Some(options) = schema.get(titled).and_then(Value::as_array) {
132        return Some(
133            options
134                .iter()
135                .filter_map(|option| {
136                    let value = option.get("const")?.as_str()?.to_owned();
137                    Some(ElicitationChoice {
138                        title: text(option, "title").unwrap_or_else(|| value.clone()),
139                        description: text(option, "description"),
140                        value,
141                    })
142                })
143                .collect(),
144        );
145    }
146    let values = schema.get("enum")?.as_array()?;
147    let names = schema.get("enumNames").and_then(Value::as_array);
148    Some(
149        values
150            .iter()
151            .enumerate()
152            .filter_map(|(index, value)| {
153                let value = value.as_str()?.to_owned();
154                let title = names
155                    .and_then(|names| names.get(index))
156                    .and_then(Value::as_str)
157                    .map_or_else(|| value.clone(), str::to_owned);
158                Some(ElicitationChoice {
159                    value,
160                    title,
161                    description: None,
162                })
163            })
164            .collect(),
165    )
166}

168/// One property schema as a field's input; a type cookie does not know is
169/// kept as an unknown input, which no one can fill in.
170fn input_of(schema: &Value) -> Open<ElicitationInput> {
171    let kind = schema.get("type").and_then(Value::as_str).unwrap_or("");
172    let input = match kind {
173        "string" => match choices(schema, "oneOf") {
174            Some(choices) => ElicitationInput::Choice {
175                choices,
176                default: text(schema, "default"),
177            },
178            None => ElicitationInput::Text {
179                min_length: unsigned(schema, "minLength")
180                    .map(|value| u32::try_from(value).unwrap_or(u32::MAX)),
181                max_length: unsigned(schema, "maxLength")
182                    .map(|value| u32::try_from(value).unwrap_or(u32::MAX)),
183                pattern: text(schema, "pattern"),
184                format: text(schema, "format"),
185                default: schema
186                    .get("default")
187                    .and_then(Value::as_str)
188                    .map(str::to_owned),
189            },
190        },
191        "number" => ElicitationInput::Number {
192            minimum: schema.get("minimum").and_then(Value::as_f64),
193            maximum: schema.get("maximum").and_then(Value::as_f64),
194            default: schema.get("default").and_then(Value::as_f64),
195        },
196        "integer" => ElicitationInput::Integer {
197            minimum: schema.get("minimum").and_then(Value::as_i64),
198            maximum: schema.get("maximum").and_then(Value::as_i64),
199            default: schema.get("default").and_then(Value::as_i64),
200        },
201        "boolean" => ElicitationInput::Boolean {
202            default: schema.get("default").and_then(Value::as_bool),
203        },
204        "array" => match schema
205            .get("items")
206            .and_then(|items| choices(items, "anyOf"))
207        {
208            Some(choices) => ElicitationInput::MultiChoice {
209                choices,
210                min_items: unsigned(schema, "minItems"),
211                max_items: unsigned(schema, "maxItems"),
212                default: schema
213                    .get("default")
214                    .and_then(Value::as_array)
215                    .map(|values| {
216                        values
217                            .iter()
218                            .filter_map(Value::as_str)
219                            .map(str::to_owned)
220                            .collect()
221                    })
222                    .unwrap_or_default(),
223            },
224            None => {
225                return Open::Unknown {
226                    tag: "array".to_owned(),
227                    raw: schema.clone(),
228                };
229            }
230        },
231        other => {
232            return Open::Unknown {
233                tag: other.to_owned(),
234                raw: schema.clone(),
235            };
236        }
237    };
238    Open::Known(input)
239}

241/// The form `schema` asks for. The agent's messages reach the daemon
242/// re-encoded by the runtime relay, which loses the order of the schema's
243/// properties, so fields come in the natural order of their names; that
244/// keeps Claude Code's `question_<n>` and `question_<n>_custom` fields in
245/// its order.
246fn form_of(schema: Schema) -> ElicitationForm {
247    let required = schema.required.unwrap_or_default();
248    let mut properties = schema.properties.into_iter().collect::<Vec<_>>();
249    properties.sort_by_cached_key(|(name, _)| natural_key(name));
250    ElicitationForm {
251        title: schema.title.filter(|title| !title.trim().is_empty()),
252        description: schema
253            .description
254            .filter(|description| !description.trim().is_empty()),
255        fields: properties
256            .into_iter()
257            .map(|(name, property)| ElicitationField {
258                required: required.contains(&name),
259                title: text(&property, "title"),
260                description: text(&property, "description"),
261                input: input_of(&property),
262                name,
263            })
264            .collect(),
265    }
266}

268/// The question `create` asks, as cookie records it, and the agent's ID of
269/// a URL question; an error message for a request cookie cannot show.
270fn request_of(
271    create: Create,
272    agent: AgentId,
273    tool_call_id: Option<ToolCallId>,
274) -> Result<(ElicitationRequest, Option<String>), String> {
275    let (form, url, url_id) = match create.mode.as_str() {
276        "form" => (
277            Some(form_of(create.requested_schema.unwrap_or_default())),
278            None,
279            None,
280        ),
281        "url" => {
282            let url = create
283                .url
284                .filter(|url| !url.is_empty())
285                .ok_or("a URL question needs a `url`")?;
286            (None, Some(url), create.elicitation_id)
287        }
288        other => return Err(format!("cookie does not offer elicitation mode `{other}`")),
289    };
290    let request = ElicitationRequest {
291        elicitation_id: ElicitationId::new_v7(),
292        agent,
293        tool_call_id,
294        message: create.message,
295        form,
296        url,
297    };
298    if request.is_valid() {
299        Ok((request, url_id))
300    } else {
301        Err("the question is larger than cookie records, or names a field twice".to_owned())
302    }
303}

305/// The ACP answer to a question.
306fn answer_json(answer: &Answer) -> Value {
307    match answer.action {
308        ElicitationAction::Accept if answer.content.is_empty() => json!({"action": "accept"}),
309        ElicitationAction::Accept => {
310            let content = answer
311                .content
312                .iter()
313                .map(|(name, value)| {
314                    (
315                        name.clone(),
316                        serde_json::to_value(value).unwrap_or(Value::Null),
317                    )
318                })
319                .collect::<Map<_, _>>();
320            json!({"action": "accept", "content": content})
321        }
322        ElicitationAction::Decline => json!({"action": "decline"}),
323        ElicitationAction::Cancel | ElicitationAction::Unknown => json!({"action": "cancel"}),
324    }
325}

334/// Takes an `elicitation/create` of `process`'s agent to the runner of its
335/// session; `request` back for any other request. A request-scoped
336/// question, or one for a session no turn runs in, is cancelled, and a
337/// malformed one refused.
338pub(crate) fn route(process: &AgentProcess, request: IncomingRequest) -> Option<IncomingRequest> {
339    if request.method != CREATE {
340        return Some(request);
341    }
342    let create = match request.params::<Create>() {
343        Ok(create) => create,
344        Err(error) => {
345            tokio::spawn(async move {
346                let _ = request.fail(error).await;
347            });
348            return None;
349        }
350    };
351    let Some(session) = create.session_id.clone() else {
352        cancel(request);
353        return None;
354    };
355    if let Err(SessionEvent::Elicitation(_, request)) = process.send_to(
356        &session,
357        SessionEvent::Elicitation(Box::new(create), request),
358    ) {
359        cancel(request);
360    }
361    None
362}

364/// Whether `notification` is an `elicitation/complete`, which it then
365/// hands to the URL question it names.
366pub(crate) fn completed(process: &AgentProcess, notification: &IncomingNotification) -> bool {
367    if notification.method != COMPLETE {
368        return false;
369    }
370    if let Ok(params) = notification.params::<Value>()
371        && let Some(id) = params.get("elicitationId").and_then(Value::as_str)
372    {
373        process.url_completions.complete(id);
374    }
375    true
376}

378/// The URL questions of one agent process waiting on the user, by the
379/// agent's elicitation ID: an `elicitation/complete` for one ends its wait.
380/// An unknown or finished ID is ignored, as ACP asks.
381#[derive(Default)]
382pub(crate) struct UrlCompletions(Mutex<HashMap<String, oneshot::Sender<()>>>);
383
384impl UrlCompletions {
385    fn watch(&self, id: &str) -> oneshot::Receiver<()> {
386        let (sender, receiver) = oneshot::channel();
387        lock(&self.0).insert(id.to_owned(), sender);
388        receiver
389    }
390
391    fn complete(&self, id: &str) {
392        if let Some(sender) = lock(&self.0).remove(id) {
393            let _ = sender.send(());
394        }
395    }
396
397    fn forget(&self, id: &str) {
398        lock(&self.0).remove(id);
399    }
400}

445const UNATTENDED_REASON: &str =
446    "no one attends this session (permission mode auto_approve_n), so no one can answer";

448impl Engine {
449    /// Asks the user `create`, a question of `agent` (whose call
450    /// `tool_call_id` it belongs to) during run `run`, and answers the
451    /// agent's `request` with what the user, or the daemon for them,
452    /// decides. `url_completions` are the agent process's.
453    #[allow(clippy::too_many_arguments)]
454    pub(in crate::runtime) fn ask_acp_question(
455        &self,
456        active: Arc<ActiveRun>,
457        run: RunId,
458        agent: AgentId,
459        tool_call_id: Option<ToolCallId>,
460        process: &Arc<AgentProcess>,
461        create: Create,
462        request: IncomingRequest,
463    ) -> tokio::task::JoinHandle<()> {
464        let engine = self.clone();
465        let process = Arc::clone(process);
466        tokio::spawn(async move {
467            let (question, url_id) = match request_of(create, agent, tool_call_id) {
468                Ok(question) => question,
469                Err(message) => {
470                    let _ = request.fail(RpcError::remote(-32602, message)).await;
471                    return;
472                }
473            };
474            let completion = url_id
475                .as_deref()
476                .map(|id| process.url_completions.watch(id));
477            let answer = engine
478                .ask_question(&active, run, question, completion)
479                .await;
480            if let Some(id) = &url_id {
481                process.url_completions.forget(id);
482            }
483            let _ = request.respond(&answer_json(&answer)).await;
484        })
485    }

487    /// Records `question` of run `run`, waits for its answer, and records
488    /// that.
489    async fn ask_question(
490        &self,
491        active: &ActiveRun,
492        run: RunId,
493        question: ElicitationRequest,
494        completion: Option<oneshot::Receiver<()>>,
495    ) -> Answer {
496        let id = question.elicitation_id;
497        let session = active.session;
498        let immediate = if self.unattended(session) {
499            Some(Answer::by_daemon(
500                ElicitationAction::Decline,
501                ElicitationAnsweredBy::Unattended,
502                UNATTENDED_REASON,
503            ))
504        } else if question
505            .form
506            .as_ref()
507            .is_some_and(ElicitationForm::has_unknown_required_field)
508        {
509            Some(Answer::by_daemon(
510                ElicitationAction::Decline,
511                ElicitationAnsweredBy::Unsupported,
512                "the form has a required field cookie cannot show",
513            ))
514        } else {
515            None
516        };
517        let (sender, receiver) = oneshot::channel();
518        if immediate.is_none() {
519            lock(&self.inner.acp.questions.0).insert(
520                id,
521                Waiting {
522                    session,
523                    run,
524                    form: question.form.clone(),
525                    answer: sender,
526                },
527            );
528        }
529        let asks_user = immediate.is_none().then(|| {
530            (
531                question.agent.to_string(),
532                question.message.clone(),
533                question.url.is_some(),
534            )
535        });
536        if self
537            .append(
538                session,
539                Some(run),
540                super::super::event_origin("engine:elicitation"),
541                Event::ElicitationRequested { request: question },
542            )
543            .await
544            .is_err()
545        {
546            lock(&self.inner.acp.questions.0).remove(&id);
547            return Answer::by_daemon(
548                ElicitationAction::Cancel,
549                ElicitationAnsweredBy::RunEnded,
550                "the question could not be recorded",
551            );
552        }
553        if let Some((agent, message, url)) = asks_user {
554            self.signal_question_needs_you(session, agent, message, url);
555        }
556        let answer = match immediate {
557            Some(answer) => answer,
558            None => {
559                let completed = async {
560                    match completion {
561                        Some(completion) => completion.await.is_ok(),
562                        None => std::future::pending().await,
563                    }
564                };
565                tokio::select! {
566                    answer = receiver => answer.unwrap_or_else(|_| Answer::by_daemon(
567                        ElicitationAction::Cancel,
568                        ElicitationAnsweredBy::RunEnded,
569                        "the run ended",
570                    )),
571                    () = active.cancellation.cancelled() => Answer::by_daemon(
572                        ElicitationAction::Cancel,
573                        ElicitationAnsweredBy::RunEnded,
574                        "the run was cancelled",
575                    ),
576                    true = completed => Answer::by_daemon(
577                        ElicitationAction::Accept,
578                        ElicitationAnsweredBy::Completed,
579                        "the agent reported the page's interaction finished",
580                    ),
581                }
582            }
583        };
584        lock(&self.inner.acp.questions.0).remove(&id);
585        let _ = self
586            .append(
587                session,
588                Some(run),
589                super::super::event_origin("engine:elicitation"),
590                Event::ElicitationResolved {
591                    resolution: ElicitationResolution {
592                        elicitation_id: id,
593                        action: answer.action,
594                        content: answer.content.clone(),
595                        answered_by: answer.answered_by,
596                        reason: answer.reason.clone(),
597                    },
598                },
599            )
600            .await;
601        answer
602    }

604    /// Whether no one attends `session`: its tree's permission mode is
605    /// `auto_approve_n`, which answers escalated approvals for an absent

607    fn unattended(&self, session: SessionId) -> bool {

611    /// Cancels the questions of run `run` still waiting: the run's turn
612    /// ended.
613    pub(in crate::runtime) fn end_acp_questions(&self, run: RunId) {
614        let ended = lock(&self.inner.acp.questions.0)
615            .iter()
616            .filter(|(_, waiting)| waiting.run == run)
617            .map(|(id, _)| *id)
618            .collect::<Vec<_>>();
619        for id in ended {
620            self.inner.acp.questions.answer(
621                id,
622                Answer::by_daemon(
623                    ElicitationAction::Cancel,
624                    ElicitationAnsweredBy::RunEnded,
625                    "the agent's turn ended",
626                ),
627            );
628        }
629    }

631    /// `elicitation.respond`: hands the user's answer to the question it
632    /// names. An accepted form must fit the form's fields; a refused answer
633    /// leaves the question waiting.
634    pub async fn respond_elicitation(
635        &self,
636        params: ElicitationRespondParams,
637    ) -> Result<cookie_agent_protocol::ElicitationRespondResult, EngineError> {
638        let mut questions = lock(&self.inner.acp.questions.0);
639        let Some(waiting) = questions
640            .get(&params.elicitation_id)
641            .filter(|waiting| waiting.session == params.session_id)
642        else {
643            return Err(EngineError::ElicitationNotPending(params.elicitation_id));
644        };
645        let content = match (params.action, &waiting.form) {
646            (ElicitationAction::Accept, Some(form)) => {
647                let content = params.content.unwrap_or_default();
648                form.check(&content).map_err(|reason| {
649                    EngineError::ElicitationAnswerInvalid(params.elicitation_id, reason)
650                })?;
651                content
652            }
653            (ElicitationAction::Unknown, _) => {
654                return Err(EngineError::ElicitationAnswerInvalid(
655                    params.elicitation_id,
656                    "the action must be accept, decline, or cancel".to_owned(),
657                ));
658            }
659            _ => ElicitationContent::new(),
660        };
661        let waiting = questions
662            .remove(&params.elicitation_id)
663            .expect("the question is waiting");
664        drop(questions);
665        waiting
666            .answer
667            .send(Answer {
668                action: params.action,
669                content,
670                answered_by: ElicitationAnsweredBy::User,
671                reason: None,
672            })
673            .map_err(|_| EngineError::ElicitationNotPending(params.elicitation_id))?;
674        Ok(cookie_agent_protocol::ElicitationRespondResult {})
675    }

677    /// Declines every question waiting in the tree of root session `root`
678    /// for `reason`: no one is there to answer, as in a headless `cookie
679    /// run`. Returns how many it declined.
680    pub fn decline_unattended_questions(&self, root: SessionId, reason: &str) -> usize {
681        let waiting = lock(&self.inner.acp.questions.0)
682            .iter()
683            .map(|(id, waiting)| (*id, waiting.session))
684            .collect::<Vec<_>>();
685        let mut declined = 0;
686        for (id, session) in waiting {
687            let in_tree = self
688                .inner
689                .store
690                .get(session)
691                .is_ok_and(|stored| root_id(&stored.meta.origin, session) == root);
692            if in_tree {
693                self.inner.acp.questions.answer(
694                    id,
695                    Answer::by_daemon(
696                        ElicitationAction::Decline,
697                        ElicitationAnsweredBy::Unattended,
698                        reason,
699                    ),
700                );
701                declined += 1;
702            }
703        }
704        declined
705    }
