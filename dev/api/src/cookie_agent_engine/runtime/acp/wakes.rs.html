1//! Work an ACP agent starts on its own (`docs/site/specs/acp-v2.md`
2//! sections 6 and 7), such as Claude Code's follow-up after a background
3//! command it started ends.
4//!
5//! Between runs a listener reads the session's updates and keeps them for
6//! the next round. When the agent's foreground work starts with no run of
7//! cookie's open (`state_update` `running` or `requires_action`, or a
8//! request to answer), the session accepts a producer message from the
9//! agent itself, owner `acp_agent`, and the run it starts listens: it sends
10//! no prompt, takes what the listener kept, and ends at the work's `idle`.
11

26/// How long the listener waits for a wake's run to take the session's
27/// updates before it listens again.
28const WAKE_START_WAIT: Duration = Duration::from_secs(30);
29
30/// How long a subagent's first work waits for its task: the adapter sends
31/// the subagent's `running` first and its task right after.
32const TASK_WAIT: Duration = Duration::from_secs(2);
33
34/// The body of a wake's producer message: v2 names no cause.
35const WAKE_BODY: &str = "The agent started work on its own.";
36
37/// Whether `event` means the agent's foreground work started: its state
38/// is `running` or `requires_action`, or it asks something to answer.
39fn starts_work(event: &SessionEvent) -> bool {
40    match event {
41        SessionEvent::Update(update) => {
42            update.get("sessionUpdate").and_then(Value::as_str) == Some("state_update")
43                && matches!(
44                    update.get("state").and_then(Value::as_str),
45                    Some("running" | "requires_action")
46                )
47        }
48        SessionEvent::Permission(..)
49        | SessionEvent::BridgeCall { .. }
50        | SessionEvent::Elicitation(..) => true,
51    }
52}
53
54/// The task a message to `live`'s session gave it among what the listener
55/// kept, such as a subagent's prompt from its parent: the wake's body. It
56/// is taken out of what the listener kept, so the run does not show it a
57/// second time.
58fn take_task(live: &LiveSession) -> Option<String> {
59    let mut carried = live
60        .carried
61        .lock()
62        .unwrap_or_else(|poisoned| poisoned.into_inner());
63    let (index, task) = carried
64        .iter()
65        .enumerate()
66        .rev()
67        .find_map(|(index, event)| task_text(live, event).map(|task| (index, task)))?;
68    carried.remove(index);
69    Some(task)
70}
71
72/// The task `event` gives `live`'s session: the text of a message to it.
73fn task_text(live: &LiveSession, event: &SessionEvent) -> Option<String> {
74    let SessionEvent::Update(update) = event else {
75        return None;
76    };
77    (update.get("sessionUpdate").and_then(Value::as_str) == Some("session_message")
78        && update.get("recipientSessionId").and_then(Value::as_str)
79            == Some(live.acp_session_id.as_str()))
80    .then(|| {
81        update
82            .get("content")
83            .and_then(Value::as_array)
84            .into_iter()
85            .flatten()
86            .map(super::wire::content_text)
87            .collect::<Vec<_>>()
88            .join("\n")
89    })
90    .filter(|text| !text.trim().is_empty())
91}

93impl Engine {
94    /// Listens to `live`'s session between runs, until the session is no
95    /// longer `live` or its process ends.
96    pub(super) fn listen_between_runs(&self, live: &LiveSession) {
97        let engine = self.clone();
98        let live = live.clone();
99        tokio::spawn(async move { engine.listen(live).await });
100    }
101
102    async fn listen(self, live: LiveSession) {
103        let (rounds, round_taken) = &*Arc::clone(&live.rounds);
104        loop {
105            // A round is about to take the updates: stay off them.
106            if live.waiting.load(Ordering::SeqCst) > 0 {
107                tokio::select! {
108                    () = tokio::time::sleep(Duration::from_millis(5)) => {}
109                    () = live.process.exited.cancelled() => return,
110                }
111                continue;
112            }
113            let mut events = tokio::select! {
114                events = live.events.lock() => events,
115                () = live.process.exited.cancelled() => {
116                    self.acp_commands_gone(live.session);
117                    return;
118                }
119            };
120            if live.waiting.load(Ordering::SeqCst) > 0 {
121                continue;
122            }
123            let event = tokio::select! {
124                biased;
125                // A round wants the updates: it takes the lock next.
126                () = live.wanted.notified() => {
127                    drop(events);
128                    tokio::task::yield_now().await;
129                    continue;
130                }
131                event = events.recv() => event,
132            };
133            let Some(event) = event else {
134                if live.process.exited.is_cancelled() {
135                    self.acp_commands_gone(live.session);
136                }
137                return;
138            };
139            // Work starts: what already came with it too, such as the task a
140            // subagent got, which follows its `running`.
141            let mut batch = vec![event];
142            if starts_work(&batch[0]) {
143                while let Ok(more) = events.try_recv() {
144                    batch.push(more);
145                }
146                // A subagent's first work starts before its task arrives;
147                // waiting for it lets the run start with the task.
148                let has_task = |batch: &[SessionEvent]| {
149                    batch
150                        .iter()
151                        .chain(
152                            live.carried
153                                .lock()
154                                .unwrap_or_else(|poisoned| poisoned.into_inner())
155                                .iter(),
156                        )
157                        .any(|event| task_text(&live, event).is_some())
158                };
159                if !has_task(&batch) && self.awaits_first_task(live.session) {
160                    let deadline = tokio::time::Instant::now() + TASK_WAIT;
161                    while let Ok(Some(more)) =
162                        tokio::time::timeout_at(deadline, events.recv()).await
163                    {
164                        let task = task_text(&live, &more).is_some();
165                        batch.push(more);
166                        if task {
167                            break;
168                        }
169                    }
170                }
171            }
172            drop(events);
173            if !self.still_live(&live) {
174                return;
175            }
176            let mut wakes = false;
177            for event in batch {
178                // A command running in the background ended: the session
179                // shows it before the run it may wake. The run that takes
180                // the update finds the command already gone.
181                if let SessionEvent::Update(update) = &event
182                    && update.get("sessionUpdate").and_then(Value::as_str)
183                        == Some("terminal_update")
184                    && let Some(read) = TerminalUpdate::read("terminal_update", update)
185                    && let Some(exit) = read.exit
186                    && let Some(ended) = self.acp_terminal_exited(live.session, &read.id, exit)
187                {
188                    let _ = self
189                        .append(
190                            live.session,
191                            None,
192                            event_origin("engine:acp"),
193                            Event::AcpSessionUpdated {
194                                update: AcpSessionUpdate {
195                                    background_ended: Some(ended),
196                                    ..AcpSessionUpdate::default()
197                                },
198                            },
199                        )
200                        .await;
201                }
202                // A subagent announced between runs gets its session at
203                // once, so its own work shows while this session is idle.
204                if let SessionEvent::Update(update) = &event
205                    && update.get("sessionUpdate").and_then(Value::as_str)
206                        == Some("subagent_update")
207                {
208                    self.acp_subagent_update(&live, None, update).await;
209                    continue;
210                }
211                wakes |= starts_work(&event);
212                live.carried
213                    .lock()
214                    .unwrap_or_else(|poisoned| poisoned.into_inner())
215                    .push(event);
216            }
217            if !wakes {
218                continue;
219            }
220            // Listen again once the wake's run took the updates; its round
221            // holds them until the work ends.
222            let before = rounds.load(Ordering::SeqCst);
223            let taken = round_taken.notified();
224            tokio::pin!(taken);
225            taken.as_mut().enable();
226            let task = take_task(&live);
227            if self
228                .start_acp_wake(live.session, live.agent.id.clone(), task)
229                .await
230                .is_err()
231            {
232                continue;
233            }
234            if rounds.load(Ordering::SeqCst) == before {
235                tokio::select! {
236                    () = taken => {}
237                    () = tokio::time::sleep(WAKE_START_WAIT) => {}
238                    () = live.process.exited.cancelled() => return,
239                }
240            }
241        }
242    }

244    /// Whether `session` is an ACP agent's subagent that has not run yet,
245    /// whose first work comes with a task.
246    fn awaits_first_task(&self, session: SessionId) -> bool {
247        self.inner.store.get(session).is_ok_and(|projection| {
248            projection.log.last_run_started().is_none() && projection.log.is_acp_subagent()
249        })
250    }
251
252    /// Whether `live` is still its cookie session's ACP session.
253    fn still_live(&self, live: &LiveSession) -> bool {
254        self.live_session(live.session)
255            .is_some_and(|known| known.acp_session_id == live.acp_session_id)
256    }

258    /// Records that the session's ACP agent `agent` started work on its
259    /// own, which starts a run to show it; nothing when a run is already
260    /// open, which takes the work in.
261    async fn start_acp_wake(
262        &self,
263        session: SessionId,
264        agent: AgentId,
265        task: Option<String>,
266    ) -> Result<(), EngineError> {
267        self.on_actor(session, move |engine| {
268            let running = engine
269                .inner
270                .sessions
271                .active
272                .lock()
273                .unwrap_or_else(|poisoned| poisoned.into_inner())
274                .values()
275                .any(|run| run.session == session)
276                || engine.inner.store.get(session)?.status == SessionStatus::Running;
277            if running {
278                return Ok(());
279            }
280            engine.append_direct(
281                session,
282                None,
283                event_origin("engine:acp"),
284                Event::ProducerMessageAccepted {
285                    message_id: ProducerMessageId::new_v7(),
286                    producer_owner: ProducerOwner::AcpAgent { agent },
287                    mode: ProducerDeliveryMode::Steer,
288                    idempotency_key: ProducerIdempotencyKey::new(
289                        ProducerMessageId::new_v7().to_string(),
290                    )
291                    .expect("UUID idempotency key"),
292                    description: SafeDisplayText::new(if task.is_some() {
293                        "Started with a task"
294                    } else {
295                        "Woke on its own"
296                    })
297                    .expect("static description"),
298                    body: UserContent::text(task.as_deref().unwrap_or(WAKE_BODY)),
299                    reminder: None,
300                    agent_hop: None,
301                    retained_output: None,
302                },
303            )?;
304            engine.inner.store.persist_buffered_session(session)?;
305            Ok(())
306        })
307        .await
308    }

310    /// Drops the agent's wakes still waiting when a run of `session` is
311    /// stopped: the stop ended that work, so no run starts again to show
312    /// it. A wake a claim holds stays; the run releases it first.
313    pub(super) async fn drop_stopped_acp_wakes(
314        &self,
315        session: SessionId,
316    ) -> Result<(), EngineError> {
317        self.on_actor(session, move |engine| {
318            let projection = engine.goal_producer_projection(session)?;
319            for message in projection.messages.iter().filter(|message| {
320                !message.consumed
321                    && !message.discarded
322                    && message.claims.is_empty()
323                    && matches!(message.producer_owner, ProducerOwner::AcpAgent { .. })
324            }) {
325                engine.append_direct(
326                    session,
327                    None,
328                    event_origin("engine:acp"),
329                    Event::ProducerMessageDiscarded {
330                        message_id: message.message_id,
331                        reminder: None,
332                        producer_owner: message.producer_owner.clone(),
333                    },
334                )?;
335            }
336            engine.inner.store.persist_buffered_session(session)?;
337            Ok(())
338        })
339        .await
340    }
