1//! Cookie's tools for an ACP agent (`docs/site/specs/acp-agents.md`
2//! section 9): the MCP server the daemon serves behind the bridge.
3//!
4//! Each ACP session gets a bridge on the agent's runtime (`agent/mcp_open`),
5//! which the agent reaches as the stdio MCP server `cookie`; the runtime
6//! relays every message of each connection to it as `agent/mcp`. The
7//! daemon answers them here: `initialize`, `ping`, `tools/list` with the
8//! session's tools as cookie composes them for a model request (the merged
9//! rules hide what they deny) minus the agent document's `acp.hide_tools`,
10//! and `tools/call` through the same preparation, permission decision,
11//! approval, and execution as a model's call, inside the agent's turn.
12//!
13//! A call through the bridge runs as a call of the agent's turn: the ACP
14//! runner records it as a cookie tool call of the turn, next to the call
15//! the agent reports for it.
16

27/// The name the agent knows cookie's MCP server by, in `session/new` and
28/// in the names it gives the server's tools (`mcp__cookie__<tool>`).
29pub(crate) const BRIDGE_SERVER: &str = "cookie";
30
31/// The MCP versions the server answers with, newest first; a client asking
32/// for another gets the first.
33const MCP_VERSIONS: [&str; 4] = ["2025-11-25", "2025-06-18", "2025-03-26", "2024-11-05"];
34
35/// The cookie session and agent a bridge serves.
36#[derive(Clone, Debug)]
37pub(crate) struct BridgeBinding {
38    pub session: SessionId,
39    pub agent: RuntimeAgent,
40}

42/// Binds bridge `bridge` of runtime `runtime` to `session`.
43pub(crate) fn bind(
44    inner: &Arc<Inner>,
45    agent: &RuntimeAgent,
46    runtime: &str,
47    bridge: &str,
48    session: SessionId,
49) {
50    inner
51        .acp
52        .processes
53        .bridges
54        .lock()
55        .unwrap_or_else(|poisoned| poisoned.into_inner())
56        .insert(
57            (runtime.to_owned(), bridge.to_owned()),
58            BridgeBinding {
59                session,
60                agent: agent.clone(),
61            },
62        );
63}

73/// Answers one MCP message a bridge connection sent.
74pub(crate) fn message(engine: Engine, client: RuntimeClient, params: AgentMcpParams) {
75    let binding = engine
76        .inner
77        .acp
78        .processes
79        .bridges
80        .lock()
81        .unwrap_or_else(|poisoned| poisoned.into_inner())
82        .get(&(client.runtime_id().to_owned(), params.bridge.clone()))
83        .cloned();
84    let Some(id) = params.message.get("id").cloned() else {
85        // Notifications (`notifications/initialized`, cancellations) need
86        // no answer.
87        return;
88    };
89    let method_name = params
90        .message
91        .get("method")
92        .and_then(Value::as_str)
93        .unwrap_or_default()
94        .to_owned();
95    if method_name.is_empty() {
96        // A reply to a request this server never sends.
97        return;
98    }
99    tokio::spawn(async move {
100        let request = params.message.get("params").cloned().unwrap_or(Value::Null);
101        let answer = match binding {
102            None => Err((-32002, "this bridge serves no session".to_owned())),
103            Some(binding) => engine.mcp_answer(&binding, &method_name, &request).await,
104        };
105        let message = match answer {
106            Ok(result) => json!({"jsonrpc": "2.0", "id": id, "result": result}),
107            Err((code, message)) => {
108                json!({"jsonrpc": "2.0", "id": id, "error": {"code": code, "message": message}})
109            }
110        };
111        let _ = client
112            .agent_notify(
113                method::AGENT_MCP,
114                &AgentMcpParams {
115                    bridge: params.bridge,
116                    connection: params.connection,
117                    message,
118                },
119            )
120            .await;
121    });
122}

124/// Whether `tool` matches one of the `hidden` patterns (`*`, `?`).
125fn hidden(tool: &str, patterns: &[String]) -> bool {
126    patterns
127        .iter()
128        .any(|pattern| cookie_agent_config::simple_wildcard_match(pattern, tool))
129}

131impl Engine {
132    async fn mcp_answer(
133        &self,
134        binding: &BridgeBinding,
135        method: &str,
136        params: &Value,
137    ) -> Result<Value, (i64, String)> {
138        match method {
139            "initialize" => {
140                let asked = params
141                    .get("protocolVersion")
142                    .and_then(Value::as_str)
143                    .unwrap_or_default();
144                let version = MCP_VERSIONS
145                    .iter()
146                    .find(|version| **version == asked)
147                    .unwrap_or(&MCP_VERSIONS[0]);
148                Ok(json!({
149                    "protocolVersion": version,
150                    "capabilities": {"tools": {}},
151                    "serverInfo": {"name": BRIDGE_SERVER, "version": env!("CARGO_PKG_VERSION")},
152                }))
153            }
154            "ping" => Ok(json!({})),
155            "tools/list" => self
156                .bridge_tools(binding)
157                .map(|tools| json!({"tools": tools}))
158                .map_err(|error| (-32603, error.to_string())),
159            "tools/call" => Ok(self.bridge_call(binding, params).await),
160            other => Err((
161                -32601,
162                format!("cookie's MCP server has no method `{other}`"),
163            )),
164        }
165    }
166
167    /// The patterns of the tools the agent of `binding` does not get.
168    fn hidden_tools(&self, binding: &BridgeBinding) -> Vec<String> {
169        self.current_runtime()
170            .agents
171            .get(&binding.agent.id)
172            .and_then(|agent| agent.document.frontmatter.acp.clone())
173            .unwrap_or_default()
174            .hidden_tools()
175    }

177    /// The active run of `binding`'s session, if one runs.
178    fn bridge_run(
179        &self,
180        binding: &BridgeBinding,
181    ) -> Option<(cookie_agent_protocol::RunId, Arc<super::super::ActiveRun>)> {
182        self.inner
183            .sessions
184            .active
185            .lock()
186            .unwrap_or_else(|poisoned| poisoned.into_inner())
187            .iter()
188            .find(|(_, active)| active.session == binding.session)
189            .map(|(run, active)| (*run, Arc::clone(active)))
190    }

192    /// The tools of `session` a run of `policy` offers its ACP agent over
193    /// the bridge, `hide` left out, by the provider that publishes them.
194    pub(crate) fn offered_tools(
195        &self,
196        session: SessionId,
197        policy: &super::super::FrozenRunPolicy,
198        hide: &[String],
199    ) -> Result<super::super::tool_prompts::OfferedTools, crate::EngineError> {
200        let composition = self.compose_for_policy(session, policy)?;
201        let mut offered = super::super::tool_prompts::OfferedTools::default();
202        for tool in composition
203            .tools
204            .iter()
205            .filter(|tool| !hidden(&tool.name, hide))
206        {
207            let provider = tool.source.provider.provider_id();
208            offered.skill |= tool.spec.permission_name.as_deref() == Some("skill");
209            if provider == crate::memory_tools::PROVIDER_ID {
210                offered.memory = true;
211                offered.memory_write |=
212                    tool.spec.permission_name.as_deref() == Some("memory_write");
213            }
214            offered.providers.insert((
215                provider.to_owned(),
216                tool.source
217                    .attachment
218                    .as_ref()
219                    .map(|name| name.as_str().to_owned()),
220            ));
221        }
222        Ok(offered)
223    }

225    /// The session's tools as the agent gets them.
226    fn bridge_tools(&self, binding: &BridgeBinding) -> Result<Vec<Value>, crate::EngineError> {
227        let policy = match self.bridge_run(binding) {
228            Some((_, active)) => Arc::clone(&active.policy),
229            None => {
230                let projection = self.inner.store.get(binding.session)?;
231                let selection = cookie_agent_protocol::RunSelection {
232                    agent: binding.agent.id.clone(),
233                    model: cookie_agent_protocol::acp_model_selection(&binding.agent.id),
234                    preset: projection.meta.creation_selection.preset.clone(),
235                };
236                let memory = self.memory_variables(binding.session);
237                Arc::new(self.freeze_root_selection(&selection, true, &memory)?.0)
238            }
239        };
240        let hide = self.hidden_tools(binding);
241        Ok(self
242            .tool_definitions(binding.session, &policy)?
243            .into_iter()
244            .filter(|tool| !hidden(&tool.name, &hide))
245            .map(|tool| {
246                json!({
247                    "name": tool.name,
248                    "description": tool.description,
249                    "inputSchema": tool.input_schema.as_value(),
250                })
251            })
252            .collect())
253    }

255    /// Runs one `tools/call` as a call of the agent's turn (the runner of
256    /// the session's turn records and runs it): a result for the agent, an
257    /// error one when the call failed or was refused.
258    async fn bridge_call(&self, binding: &BridgeBinding, params: &Value) -> Value {
259        let failed = |message: String| json!({"content": [{"type": "text", "text": message}], "isError": true});
260        let Some(name) = params.get("name").and_then(Value::as_str) else {
261            return failed("the call names no tool".to_owned());
262        };
263        if hidden(name, &self.hidden_tools(binding)) {
264            return failed(format!("tool `{name}` is not offered to this agent"));
265        }
266        let Some(live) = self.live_session(binding.session) else {
267            return failed("cookie's tools can be called only during the agent's turns".to_owned());
268        };
269        let (reply, answer) = tokio::sync::oneshot::channel();
270        let event = super::process::SessionEvent::BridgeCall {
271            name: name.to_owned(),
272            arguments: params
273                .get("arguments")
274                .cloned()
275                .unwrap_or_else(|| json!({})),
276            reply,
277        };
278        if live.process.send_to(&live.acp_session_id, event).is_err() {
279            return failed("cookie's tools can be called only during the agent's turns".to_owned());
280        }
281        answer.await.unwrap_or_else(|_| {
282            failed("the agent's turn ended before the call finished".to_owned())
283        })
284    }
