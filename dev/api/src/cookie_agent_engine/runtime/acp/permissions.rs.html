1//! An ACP agent's own permission requests
2//! (`docs/site/specs/acp-agents.md` section 8).
3//!
4//! A `session/request_permission` names one of the agent's tool calls. Its
5//! ACP kind maps onto one of cookie's actions, and its locations, command,
6//! or URL onto that action's resources; the agent's merged rules decide
7//! each, through the same pipeline as cookie's tools, grants included. An
8//! `ask` goes through the approval flow. The answer is the agent's
9//! allow-once or reject-once option, never an "always" one.
10

22/// What a permission request asks: one action and the resources it needs,
23/// each of which must be allowed.
24#[derive(Clone, Debug, Eq, PartialEq)]
25pub(crate) struct MappedRequest {
26    pub action: PermissionAction,
27    pub resources: Vec<String>,
28}

30/// The first string member of `input` among `names`.
31fn input_string<'a>(input: Option<&'a Value>, names: &[&str]) -> Option<&'a str> {
32    let input = input?;
33    names
34        .iter()
35        .find_map(|name| input.get(*name).and_then(Value::as_str))
36        .filter(|value| !value.is_empty())
37}

39/// The command an `execute` call runs: its raw input's `command` (a string
40/// or an argument vector), else its title.
41fn command_of(call: &ToolCallFields) -> Option<String> {
42    let input = call.raw_input.as_ref();
43    if let Some(command) = input.and_then(|input| input.get("command")) {
44        match command {
45            Value::String(command) if !command.is_empty() => return Some(command.clone()),
46            Value::Array(arguments) => {
47                let words = arguments
48                    .iter()
49                    .filter_map(Value::as_str)
50                    .collect::<Vec<_>>();
51                if !words.is_empty() {
52                    return Some(words.join(" "));
53                }
54            }
55            _ => {}
56        }
57    }
58    input_string(input, &["cmd"])
59        .map(str::to_owned)
60        .or_else(|| call.title.clone())
61}
62
63/// The paths a file call touches: its locations, else its raw input's path,
64/// else its title.
65fn paths_of(call: &ToolCallFields) -> Vec<String> {
66    let locations = call
67        .locations
68        .iter()
69        .flatten()
70        .map(|location| location.path.clone())
71        .filter(|path| !path.is_empty())
72        .collect::<Vec<_>>();
73    if !locations.is_empty() {
74        return locations;
75    }
76    input_string(
77        call.raw_input.as_ref(),
78        &["path", "file_path", "filePath", "abs_path", "absPath"],
79    )
80    .map(str::to_owned)
81    .or_else(|| call.title.clone())
82    .into_iter()
83    .collect()
84}

86/// Maps the tool call a permission request names onto cookie's action
87/// and resources: `call` as the request carries it, merged over `known`,
88/// what the agent reported of the call before.
89pub(crate) fn map_request(call: &ToolCallFields, known: Option<&ToolCallFields>) -> MappedRequest {
90    let merged = merge(known, call);
91    let kind = merged.kind.as_deref().unwrap_or("other");
92    let title = merged.title.clone().unwrap_or_default();
93    match kind {
94        "read" | "search" => MappedRequest {
95            action: PermissionAction::READ,
96            resources: paths_of(&merged),
97        },
98        "edit" | "delete" | "move" => MappedRequest {
99            action: PermissionAction::WRITE,
100            resources: paths_of(&merged),
101        },
102        "execute" => MappedRequest {
103            action: PermissionAction::BASH,
104            resources: command_of(&merged).into_iter().collect(),
105        },
106        "fetch" => MappedRequest {
107            action: PermissionAction::WEBFETCH,
108            resources: input_string(merged.raw_input.as_ref(), &["url", "uri"])
109                .map(str::to_owned)
110                .or_else(|| merged.title.clone())
111                .into_iter()
112                .collect(),
113        },
114        other => MappedRequest {
115            action: PermissionAction::ACP,
116            resources: vec![format!("{other} {title}").trim().to_owned()],
117        },
118    }
119}

121/// `update` over `known`: each field the update carries replaces the known
122/// one.
123pub(crate) fn merge(known: Option<&ToolCallFields>, update: &ToolCallFields) -> ToolCallFields {
124    let mut merged = known.cloned().unwrap_or_default();
125    merged.tool_call_id.clone_from(&update.tool_call_id);
126    if update.title.is_some() {
127        merged.title.clone_from(&update.title);
128    }
129    if update.kind.is_some() {
130        merged.kind.clone_from(&update.kind);
131    }
132    if update.status.is_some() {
133        merged.status.clone_from(&update.status);
134    }
135    if update.content.is_some() {
136        merged.content.clone_from(&update.content);
137    }
138    if update.locations.is_some() {
139        merged.locations.clone_from(&update.locations);
140    }
141    if update.raw_input.is_some() {
142        merged.raw_input.clone_from(&update.raw_input);
143    }
144    if update.raw_output.is_some() {
145        merged.raw_output.clone_from(&update.raw_output);
146    }
147    merged
148}

150/// The option of `kind` (`allow_once` or `reject_once`) the agent offered.
151fn option_of<'a>(options: &'a [wire::PermissionOption], kind: &str) -> Option<&'a str> {
152    options
153        .iter()
154        .find(|option| option.kind == kind)
155        .map(|option| option.option_id.as_str())
156}

158impl Engine {
159    /// Decides ACP permission request `request` of the run `active` of
160    /// `agent`, whose session works in `cwd` on the agent's runtime, home
161    /// `home`: the answer to send the agent. An approval it asks for names
162    /// `tool_call_id`, the cookie call the agent's call is shown as.
163    #[allow(clippy::too_many_arguments)]
164    pub(in crate::runtime) async fn decide_acp_permission(
165        &self,
166        active: &ActiveRun,
167        run: cookie_agent_protocol::RunId,
168        agent: &RuntimeAgent,
169        cwd: &str,
170        home: Option<&str>,
171        request: &wire::PermissionRequest,
172        known: Option<&ToolCallFields>,
173        tool_call_id: Option<ToolCallId>,
174    ) -> Value {
175        let allowed = self
176            .acp_permission_allowed(active, run, agent, cwd, home, request, known, tool_call_id)
177            .await;
178        let option = if allowed {
179            option_of(&request.options, "allow_once")
180        } else {
181            option_of(&request.options, "reject_once")
182        };
183        wire::permission_outcome(option)
184    }
185
186    #[allow(clippy::too_many_arguments)]
187    async fn acp_permission_allowed(
188        &self,
189        active: &ActiveRun,
190        run: cookie_agent_protocol::RunId,
191        agent: &RuntimeAgent,
192        cwd: &str,
193        home: Option<&str>,
194        request: &wire::PermissionRequest,
195        known: Option<&ToolCallFields>,
196        tool_call_id: Option<ToolCallId>,
197    ) -> bool {
198        let mapped = map_request(&request.tool_call, known);
199        if mapped.resources.is_empty() {
200            return false;
201        }
202        let tool = format!("acp:{}", agent.id);
203        let resources = mapped
204            .resources
205            .iter()
206            .map(|resource| {
207                prepared_resource(
208                    mapped.action.clone(),
209                    mapped.action.as_str(),
210                    resource.as_bytes(),
211                    resource.as_bytes(),
212                    PreparedBindingLifetime::RestartStable,
213                    ApprovalResourceSource::PrimaryOperation,
214                )
215            })
216            .collect::<Result<Vec<_>, _>>();
217        let arguments = serde_json::json!({
218            "tool_call": request.tool_call.tool_call_id,
219            "action": mapped.action.as_str(),
220            "resources": mapped.resources,
221        });
222        let Ok(resources) = resources else {
223            return false;
224        };
225        let Ok(operation) = prepared_operation(
226            &tool,
227            &arguments,
228            vec![(mapped.action.clone(), "request_permission")],
229            resources,
230            agent.id.as_str().as_bytes(),
231        ) else {
232            return false;
233        };
234        let Ok(session) = self.inner.store.get(active.session) else {
235            return false;
236        };
237        let overlay = session.permission_overlay.clone();
238        let labels = mapped
239            .resources
240            .iter()
241            .cloned()
242            .map(Some)
243            .collect::<Vec<_>>();
244        let workspace = std::path::Path::new(cwd);
245        let decision = self.inner.permissions.decide_operation_with_grants(
246            &active.policy.agent,
247            Some(&overlay),
248            None,
249            &operation,
250            &labels,
251            crate::permissions::PermissionScope::runtime(agent.runtime.as_str(), Some(workspace))
252                .with_home(home),
253        );
254        match decision.effect {
255            PermissionEffect::Allow => true,
256            PermissionEffect::Deny => false,
257            PermissionEffect::Ask => {
258                let tool_call_id = tool_call_id.unwrap_or_else(ToolCallId::new_v7);
259                let request = match ApprovalRequest::new(
260                    ApprovalId::new_v7(),
261                    1,
262                    tool_call_id,
263                    ApprovalTrigger::PermissionPolicy,
264                    operation,
265                    decision.evaluations,
266                    ApprovalConstraints {
267                        allow_once: true,
268                        allow_tree_grant: !mapped.action.is_path(),
269                        cancellable: true,
270                        expires_at: super::super::approval_flow::approval_expiry(
271                            self.inner.config.load().runtime.approval.timeout_ms,
272                        ),
273                    },
274                ) {
275                    Ok(request) => request,
276                    Err(_) => return false,
277                };
278                let executor: cookie_agent_runtime_api::PreparedExecutorCell =
279                    std::sync::Arc::new(tokio::sync::Mutex::new(Some(Box::new(AnsweredExecutor))));
280                self.await_user_approval(
281                    active,
282                    run,
283                    request,
284                    executor,
285                    true,
286                    super::super::approval_flow::ApprovalToolInput {
287                        tool_call_id,
288                        name: &tool,
289                        origin: None,
290                        normalized_parameters: &arguments,
291                    },
292                )
293                .await
294                .is_ok_and(|outcome| outcome.approved)
295            }
296        }
297    }
298}

300/// The executor behind an approval of an ACP permission request: the
301/// agent runs the call itself, so there is nothing to run or revalidate.
302struct AnsweredExecutor;
303
304#[async_trait::async_trait]
305impl cookie_agent_runtime_api::PreparedExecutor for AnsweredExecutor {
306    async fn revalidate(&self) -> Result<(), cookie_agent_runtime_api::ToolError> {
307        Ok(())
308    }
309
310    async fn execute(
311        self: Box<Self>,
312        _ctx: cookie_agent_runtime_api::ToolExecutionContext,
313    ) -> Result<cookie_agent_runtime_api::ToolCompletion, cookie_agent_runtime_api::ToolError> {
314        Err(cookie_agent_runtime_api::ToolError::execution(
315            "an ACP agent runs its own calls",
316        ))
317    }
