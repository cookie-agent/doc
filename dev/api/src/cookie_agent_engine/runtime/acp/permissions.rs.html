1//! An ACP agent's own permission requests
2//! (`docs/site/specs/acp-permissions.md`, check 2, which revises
3//! `docs/site/specs/acp-agents.md` section 8).
4//!
5//! A `session/request_permission` names one of the agent's tool calls. Its
6//! ACP kind maps onto one of cookie's actions, and its locations, command,
7//! URL, or MCP tool onto that action's resources; the agent's merged rules
8//! decide each, through the same pipeline as cookie's tools. A call of
9//! cookie's own tools is decided by the rules of that call, prepared as the
10//! bridged call will be, and asked about on its row; a `think` is allowed
11//! at once. `deny` rejects and `allow` allows without asking; an `ask`,
12//! no rule, `switch_mode`, every kind that maps onto no action, and a file
13//! call that names no path go to the user, never to the reviewer. A
14//! rejection's note is steered into the agent's turn.
15

30/// What a permission request asks, by its mapping (spec section 8).
31#[derive(Clone, Debug, Eq, PartialEq)]
32pub(crate) enum MappedRequest {
33    /// Allowed at once: a `think`.
34    Allow,
35    /// A call of cookie's own tool, by its name: decided by the rules of
36    /// the call itself.
37    Bridge(String),
38    /// Decided by the agent's rules: one action and the resources it
39    /// needs, each of which must be allowed.
40    Rules {
41        action: PermissionAction,
42        resources: Vec<String>,
43    },
44    /// Decided by the approval flow alone: `switch_mode`, every kind that
45    /// maps onto no action, and a file call that names no path. `label`
46    /// names the kind (an action label no rule decides) or the file call's
47    /// action, `resource` what the request shows.
48    Ask {
49        label: PermissionAction,
50        resource: String,
51    },
52}

68/// The first string member of `input` among `names`.
69fn input_string<'a>(input: Option<&'a Value>, names: &[&str]) -> Option<&'a str> {
70    let input = input?;
71    names
72        .iter()
73        .find_map(|name| input.get(*name).and_then(Value::as_str))
74        .filter(|value| !value.is_empty())
75}

77/// The command an `execute` call runs: its raw input's `command` (a string
78/// or an argument vector), else its title.
79fn command_of(call: &ToolCallFields) -> Option<String> {
80    let input = call.raw_input.as_ref();
81    if let Some(command) = input.and_then(|input| input.get("command")) {
82        match command {
83            Value::String(command) if !command.is_empty() => return Some(command.clone()),
84            Value::Array(arguments) => {
85                let words = arguments
86                    .iter()
87                    .filter_map(Value::as_str)
88                    .collect::<Vec<_>>();
89                if !words.is_empty() {
90                    return Some(words.join(" "));
91                }
92            }
93            _ => {}
94        }
95    }
96    input_string(input, &["cmd"])
97        .map(str::to_owned)
98        .or_else(|| call.title.clone())
99}

101/// The paths a file call touches: its locations, else its raw input's path.
102/// None when it names neither: its title is no path. Labelled
103/// ([`cookie_agent_protocol::paths::path_label`]) as `style` reads them,
104/// the platform of the agent's runtime; as they are when it reported none,
105/// for the rules to read both ways.
106fn paths_of(call: &ToolCallFields, style: Option<PathStyle>) -> Vec<String> {
107    let locations = call
108        .locations
109        .iter()
110        .flatten()
111        .map(|location| location.path.as_str())
112        .filter(|path| !path.is_empty())
113        .collect::<Vec<_>>();
114    let paths = if locations.is_empty() {
115        input_string(
116            call.raw_input.as_ref(),
117            &["path", "file_path", "filePath", "abs_path", "absPath"],
118        )
119        .into_iter()
120        .collect()
121    } else {
122        locations
123    };
124    paths
125        .into_iter()
126        .map(|path| match style {
127            Some(style) => cookie_agent_protocol::paths::path_label(path, style),
128            None => path.to_owned(),
129        })
130        .collect()
131}

133/// A file call of `action`: decided by the rules on the paths it names, or,
134/// naming none, through the approval flow.
135fn file_request(
136    action: PermissionAction,
137    call: &ToolCallFields,
138    style: Option<PathStyle>,
139) -> MappedRequest {
140    let resources = paths_of(call, style);
141    if resources.is_empty() {
142        return MappedRequest::Ask {
143            resource: call
144                .title
145                .as_deref()
146                .map(str::trim)
147                .filter(|title| !title.is_empty())
148                .unwrap_or(action.as_str())
149                .to_owned(),
150            label: action,
151        };
152    }
153    MappedRequest::Rules { action, resources }
154}

156/// The MCP server and tool a call of kind `other` (or of a kind ACP v1
157/// does not know) goes to, as the adapters name them: Claude Code titles
158/// it `mcp__<server>__<tool>`, Codex `Tool: <server>/<tool>` with
159/// `{server, tool}` as its raw input, and Gemini CLI `<tool> (<server> MCP
160/// Server)`. The whole title must have the form, so a command or a path
161/// that merely contains one never matches.
162fn mcp_tool_of(call: &ToolCallFields) -> Option<(String, String)> {
163    let input = call.raw_input.as_ref();
164    let from_input = input_string(input, &["server", "serverName", "server_name"])
165        .zip(input_string(input, &["tool", "toolName", "tool_name"]));
166    if let Some((server, tool)) = from_input {
167        return Some((server.to_owned(), tool.to_owned()));
168    }
169    let title = call.title.as_deref()?.trim();
170    let named = if let Some(rest) = title.strip_prefix("mcp__") {
171        rest.split_once("__")
172    } else if let Some(rest) = title.strip_prefix("Tool: ") {
173        rest.split_once('/')
174    } else if let Some(rest) = title.strip_suffix(" MCP Server)") {
175        rest.rsplit_once(" (").map(|(tool, server)| (server, tool))
176    } else {
177        None
178    };
179    named
180        .filter(|(server, tool)| {
181            [server, tool].iter().all(|part| {
182                !part.is_empty() && !part.chars().any(|c| c.is_whitespace() || c.is_control())
183            })
184        })
185        .map(|(server, tool)| (server.to_owned(), tool.to_owned()))
186}

188/// The tool of cookie's own MCP bridge that `call` is the agent's call of,
189/// by any agent's form that [`mcp_tool_of`] reads, or Claude Code's
190/// `claudeCode.toolName` (`mcp__cookie__<tool>`).
191pub(crate) fn bridge_tool_of(call: &ToolCallFields) -> Option<String> {
192    let from_name = || {
193        call.tool_name()?
194            .strip_prefix("mcp__")?
195            .split_once("__")
196            .map(|(server, tool)| (server.to_owned(), tool.to_owned()))
197    };
198    mcp_tool_of(call)
199        .or_else(from_name)
200        .filter(|(server, tool)| server == BRIDGE_SERVER && !tool.is_empty())
201        .map(|(_, tool)| tool)
202}

204/// A name as cookie publishes an MCP server's tools: ASCII letters,
205/// digits, `_`, and `-` kept, everything else `_`.
206fn mcp_name_part(value: &str) -> String {
207    value
208        .chars()
209        .map(|c| {
210            if c.is_ascii_alphanumeric() || matches!(c, '_' | '-') {
211                c
212            } else {
213                '_'
214            }
215        })
216        .collect()
217}

219/// Maps the tool call a permission request names onto cookie's action
220/// and resources: `call` as the request carries it, merged over `known`,
221/// what the agent reported of the call before, its paths read in `style`.
222pub(crate) fn map_request(
223    call: &ToolCallFields,
224    known: Option<&ToolCallFields>,
225    style: Option<PathStyle>,
226) -> MappedRequest {
227    let merged = merge(known, call);
228    let kind = merged.kind.as_deref().unwrap_or("other");
229    match kind {
230        "read" | "search" => file_request(PermissionAction::READ, &merged, style),
231        "edit" | "delete" | "move" => file_request(PermissionAction::WRITE, &merged, style),
232        "execute" => MappedRequest::Rules {
233            action: PermissionAction::BASH,
234            resources: command_of(&merged).into_iter().collect(),
235        },
236        "fetch" => MappedRequest::Rules {
237            action: PermissionAction::WEBFETCH,
238            resources: input_string(merged.raw_input.as_ref(), &["url", "uri"])
239                .map(str::to_owned)
240                .or_else(|| merged.title.clone())
241                .into_iter()
242                .collect(),
243        },
244        "think" => MappedRequest::Allow,
245        "switch_mode" => ask(kind, &merged),
246        other => match mcp_tool_of(&merged) {
247            Some((server, tool)) if server == BRIDGE_SERVER => MappedRequest::Bridge(tool),
248            Some((server, tool)) => MappedRequest::Rules {
249                action: PermissionAction::MCP,
250                resources: vec![format!(
251                    "{}_{}",
252                    mcp_name_part(&server),
253                    mcp_name_part(&tool)
254                )],
255            },
256            None if KNOWN_KINDS.contains(&other) => ask(other, &merged),
257            None => ask("other", &merged),
258        },
259    }
260}

262fn ask(kind: &str, call: &ToolCallFields) -> MappedRequest {
263    let label = PermissionAction::new(kind)
264        .unwrap_or_else(|_| PermissionAction::new("other").expect("`other` is an action label"));
265    let title = call.title.as_deref().map(str::trim).unwrap_or_default();
266    MappedRequest::Ask {
267        label,
268        resource: if title.is_empty() {
269            kind.to_owned()
270        } else {
271            title.to_owned()
272        },
273    }
274}

276/// `update` over `known`: each field the update carries replaces the known
277/// one.
278pub(crate) fn merge(known: Option<&ToolCallFields>, update: &ToolCallFields) -> ToolCallFields {
279    let mut merged = known.cloned().unwrap_or_default();
280    merged.tool_call_id.clone_from(&update.tool_call_id);
281    if update.name.is_some() {
282        merged.name.clone_from(&update.name);
283    }
284    if update.title.is_some() {
285        merged.title.clone_from(&update.title);
286    }
287    if update.kind.is_some() {
288        merged.kind.clone_from(&update.kind);
289    }
290    if update.status.is_some() {
291        merged.status.clone_from(&update.status);
292    }
293    if update.content.is_some() {
294        merged.content.clone_from(&update.content);
295    }
296    if update.locations.is_some() {
297        merged.locations.clone_from(&update.locations);
298    }
299    if update.raw_input.is_some() {
300        merged.raw_input.clone_from(&update.raw_input);
301    }
302    if update.raw_output.is_some() {
303        merged.raw_output.clone_from(&update.raw_output);
304    }
305    match (&mut merged.meta, &update.meta) {
306        (Some(Value::Object(known)), Some(Value::Object(new))) => {
307            for (key, value) in new {
308                known.insert(key.clone(), value.clone());
309            }
310        }
311        (_, Some(new)) => merged.meta = Some(new.clone()),
312        _ => {}
313    }
314    merged
315}

317/// The arguments of the agent's call of one of cookie's tools, as the
318/// bridged call will get them: Codex's raw input wraps them as
319/// `{server, tool, arguments}`, Claude Code's is the arguments themselves.
320pub(crate) fn bridge_arguments(call: &ToolCallFields) -> Value {
321    let Some(input) = call.raw_input.as_ref().filter(|input| input.is_object()) else {
322        return Value::Object(serde_json::Map::new());
323    };
324    let wrapped = ["tool", "toolName", "tool_name"]
325        .iter()
326        .any(|name| input.get(*name).is_some_and(Value::is_string));
327    match input.get("arguments") {
328        Some(arguments) if wrapped && arguments.is_object() => arguments.clone(),
329        _ if wrapped => Value::Object(serde_json::Map::new()),
330        _ => input.clone(),
331    }
332}

334/// How a permission request was answered: whether it is allowed, the
335/// agent's option the user chose, and the note of a rejection the user gave
336/// one with.
337#[derive(Debug, Default)]
338pub(crate) struct AcpAnswer {
339    pub(crate) allowed: bool,
340    pub(crate) option: Option<String>,
341    pub(crate) note: Option<String>,
342}
343
344impl AcpAnswer {
345    fn decided(allowed: bool) -> Self {
346        Self {
347            allowed,
348            ..Self::default()
349        }
350    }
351
352    /// The option to answer the agent with: the user's, else its
353    /// allow-once or reject-once option.
354    pub(crate) fn outcome(&self, options: &[wire::PermissionOption]) -> Value {
355        let chosen = self
356            .option
357            .as_deref()
358            .filter(|chosen| options.iter().any(|option| option.option_id == *chosen));
359        wire::permission_outcome(chosen.or_else(|| answer_option(options, self.allowed)))
360    }
361}

363/// The agent's options as an approval offers them to the user.
364pub(crate) fn agent_options(
365    options: &[wire::PermissionOption],
366) -> Vec<cookie_agent_protocol::ApprovalAgentOption> {
367    options
368        .iter()
369        .filter(|option| {
370            !option.option_id.is_empty()
371                && option.option_id.len() <= cookie_agent_protocol::MAX_ACP_LABEL_BYTES
372        })
373        .map(|option| cookie_agent_protocol::ApprovalAgentOption {
374            option_id: option.option_id.clone(),
375            name: {
376                let name = option.name.trim();
377                if name.is_empty() {
378                    option.option_id.clone()
379                } else {
380                    bounded_label(name)
381                }
382            },
383            kind: serde_json::from_value(Value::String(option.kind.clone()))
384                .unwrap_or(cookie_agent_protocol::ApprovalAgentOptionKind::RejectOnce),
385        })
386        .collect()
387}

389/// `text` cut to the longest label the protocol carries.
390fn bounded_label(text: &str) -> String {
391    let max = cookie_agent_protocol::MAX_ACP_LABEL_BYTES;
392    if text.len() <= max {
393        return text.to_owned();
394    }
395    let mut end = max;
396    while !text.is_char_boundary(end) {
397        end -= 1;
398    }
399    text[..end].to_owned()
400}

402/// The agent's allow-once or reject-once option, as `allowed` says.
403pub(crate) fn answer_option(options: &[wire::PermissionOption], allowed: bool) -> Option<&str> {
404    option_of(options, if allowed { "allow_once" } else { "reject_once" })
405}
406
407/// The option of `kind` (`allow_once` or `reject_once`) the agent offered.
408fn option_of<'a>(options: &'a [wire::PermissionOption], kind: &str) -> Option<&'a str> {
409    options
410        .iter()
411        .find(|option| option.kind == kind)
412        .map(|option| option.option_id.as_str())
413}

415impl Engine {
416    /// Decides ACP permission request `request` of the run `active` of
417    /// `agent`, whose session works in `cwd` on the agent's runtime, home
418    /// `home`, which runs `platform`: the answer to send the agent. An approval it asks for names
419    /// `tool_call_id`, the cookie call the agent's call is shown as.
420    #[allow(clippy::too_many_arguments)]
421    pub(in crate::runtime) async fn decide_acp_permission(
422        &self,
423        active: &ActiveRun,
424        run: cookie_agent_protocol::RunId,
425        agent: &RuntimeAgent,
426        cwd: &str,
427        home: Option<&str>,
428        platform: Option<&str>,
429        request: &wire::PermissionRequest,
430        known: Option<&ToolCallFields>,
431        tool_call_id: Option<ToolCallId>,
432    ) -> Value {
433        let answer = self
434            .acp_permission_allowed(
435                active,
436                run,
437                agent,
438                cwd,
439                home,
440                platform,
441                request,
442                known,
443                tool_call_id,
444            )
445            .await;
446        if let Some(note) = &answer.note {
447            self.steer_rejection_note(run, request, note).await;
448        }
449        answer.outcome(&request.options)
450    }
451
452    #[allow(clippy::too_many_arguments)]
453    async fn acp_permission_allowed(
454        &self,
455        active: &ActiveRun,
456        run: cookie_agent_protocol::RunId,
457        agent: &RuntimeAgent,
458        cwd: &str,
459        home: Option<&str>,
460        platform: Option<&str>,
461        request: &wire::PermissionRequest,
462        known: Option<&ToolCallFields>,
463        tool_call_id: Option<ToolCallId>,
464    ) -> AcpAnswer {
465        let refused = AcpAnswer::decided(false);
466        let style = platform.map(PathStyle::of_platform);
467        let (action, resources, by_rules) = match map_request(&request.tool_call, known, style) {
468            MappedRequest::Allow => return AcpAnswer::decided(true),
469            // The runner decides these on the call's own row; one that
470            // reaches here has no row to ask on.
471            MappedRequest::Bridge(_) => return refused,
472            MappedRequest::Rules { action, resources } => (action, resources, true),
473            MappedRequest::Ask { label, resource } => (label, vec![resource], false),
474        };
475        if resources.is_empty() {
476            return refused;
477        }
478        let tool = format!("acp:{}", agent.id);
479        let prepared = resources
480            .iter()
481            .map(|resource| {
482                prepared_resource(
483                    action.clone(),
484                    action.as_str(),
485                    resource.as_bytes(),
486                    resource.as_bytes(),
487                    PreparedBindingLifetime::RestartStable,
488                    ApprovalResourceSource::PrimaryOperation,
489                )
490            })
491            .collect::<Result<Vec<_>, _>>();
492        let arguments = serde_json::json!({
493            "tool_call": request.tool_call.tool_call_id,
494            "action": action.as_str(),
495            "resources": resources,
496        });
497        let Ok(prepared) = prepared else {
498            return refused;
499        };
500        let Ok(operation) = prepared_operation(
501            &tool,
502            &arguments,
503            vec![(action.clone(), "request_permission")],
504            prepared,
505            agent.id.as_str().as_bytes(),
506        ) else {
507            return refused;
508        };
509        let evaluations = if !by_rules {
510            // No rule decides it: the kind maps onto no action, or the file
511            // call names no path.
512            operation
513                .resources()
514                .iter()
515                .zip(&resources)
516                .map(|(prepared, resource)| ApprovalEvaluation {
517                    resource_digest: prepared.binding_digest.clone(),
518                    effect: PermissionEffect::Ask,
519                    trace: DecisionTrace {
520                        action: action.clone(),
521                        normalized_resource: resource.clone(),
522                        candidates: Vec::new(),
523                        effect: PermissionEffect::Ask,
524                        precedence_reason: if action.is_path() {
525                            format!(
526                                "an ACP agent's `{}` request names no path and always asks",
527                                action.as_str()
528                            )
529                        } else {
530                            format!(
531                                "an ACP agent's `{}` request maps onto no action and always asks",
532                                action.as_str()
533                            )
534                        },
535                    },
536                })
537                .collect()
538        } else {
539            let Ok(session) = self.inner.store.get(active.session) else {
540                return refused;
541            };
542            let overlay = session.permission_overlay.clone();
543            let labels = resources.iter().cloned().map(Some).collect::<Vec<_>>();
544            let workspace = std::path::Path::new(cwd);
545            let decision = self.inner.permissions.decide_operation_with_grants(
546                &active.policy.agent,
547                Some(&overlay),
548                None,
549                &operation,
550                &labels,
551                crate::permissions::PermissionScope::runtime(
552                    agent.runtime.as_str(),
553                    Some(workspace),
554                )
555                .with_home(home)
556                .with_platform(platform),
557            );
558            match decision.effect {
559                PermissionEffect::Allow => return AcpAnswer::decided(true),
560                PermissionEffect::Deny => return refused,
561                PermissionEffect::Ask => decision.evaluations,
562            }
563        };
564        let tool_call_id = tool_call_id.unwrap_or_else(ToolCallId::new_v7);
565        let request = match ApprovalRequest::new(
566            ApprovalId::new_v7(),
567            1,
568            tool_call_id,
569            ApprovalTrigger::PermissionPolicy,
570            operation,
571            evaluations,
572            ApprovalConstraints {
573                allow_once: true,
574                // An ACP session makes no tree grants: its approvals are
575                // the harness's.
576                allow_tree_grant: false,
577                cancellable: true,
578                // The user's window starts at the escalation.
579                expires_at: None,
580            },
581        ) {
582            Ok(approval) => approval.with_agent_options(agent_options(&request.options)),
583            Err(_) => return refused,
584        };
585        let approval_id = request.approval_id();
586        let executor: cookie_agent_runtime_api::PreparedExecutorCell =
587            std::sync::Arc::new(tokio::sync::Mutex::new(Some(Box::new(AnsweredExecutor))));
588        let approved = self
589            .await_user_approval(
590                active,
591                run,
592                request,
593                executor,
594                true,
595                crate::runtime::approval::flow::ApprovalToolInput {
596                    tool_call_id,
597                    origin: None,
598                },
599            )
600            .await
601            .is_ok_and(|outcome| outcome.approved);
602        self.user_answer(active.session, approval_id, approved)
603    }

605    /// Decides the agent's request to call one of cookie's tools,
606    /// `prepared` as the bridged call will be: `deny` rejects it, `allow`
607    /// and tree grants allow it, and otherwise the user is asked on the
608    /// call's own row, with the agent's `options`.
609    pub(in crate::runtime) async fn decide_bridged_request(
610        &self,
611        active: &ActiveRun,
612        run: cookie_agent_protocol::RunId,
613        prepared: &super::super::PreparedToolCall,
614        options: &[wire::PermissionOption],
615    ) -> AcpAnswer {
616        // A call cookie cannot prepare would fail at the bridge anyway: let
617        // the agent make it and read why.
618        if prepared.prepared.is_err() {
619            return AcpAnswer::decided(true);
620        }
621        let Ok(permission) = self.decide_tool_permission(active, prepared) else {
622            return AcpAnswer::decided(false);
623        };
624        let evaluations = match permission.effect {
625            PermissionEffect::Allow => return AcpAnswer::decided(true),
626            PermissionEffect::Deny => return AcpAnswer::decided(false),
627            PermissionEffect::Ask => permission.evaluations,
628        };
629        let Ok(ready) = prepared.prepared.as_ref() else {
630            return AcpAnswer::decided(false);
631        };
632        let Ok(request) = ApprovalRequest::new(
633            ApprovalId::new_v7(),
634            1,
635            prepared.call.id,
636            ApprovalTrigger::PermissionPolicy,
637            ready.operation().clone(),
638            evaluations,
639            ApprovalConstraints {
640                allow_once: true,
641                allow_tree_grant: false,
642                cancellable: true,
643                expires_at: None,
644            },
645        )
646        .map(|request| request.with_agent_options(agent_options(options))) else {
647            return AcpAnswer::decided(false);
648        };
649        let approval_id = request.approval_id();
650        let executor: cookie_agent_runtime_api::PreparedExecutorCell =
651            std::sync::Arc::new(tokio::sync::Mutex::new(Some(Box::new(AnsweredExecutor))));
652        let approved = self
653            .await_user_approval(
654                active,
655                run,
656                request,
657                executor,
658                true,
659                crate::runtime::approval::flow::ApprovalToolInput {
660                    tool_call_id: prepared.call.id,
661                    origin: prepared.origin(),
662                },
663            )
664            .await
665            .is_ok_and(|outcome| outcome.approved);
666        self.user_answer(active.session, approval_id, approved)
667    }
668
669    /// How approval `approval_id` of `session` ended, `approved` or not:
670    /// with the agent's option the user chose, and the note they rejected
671    /// it with.
672    fn user_answer(
673        &self,
674        session: cookie_agent_protocol::SessionId,
675        approval_id: ApprovalId,
676        approved: bool,
677    ) -> AcpAnswer {
678        let mut answer = AcpAnswer::decided(approved);
679        let Ok(log) = self.inner.store.log(session) else {
680            return answer;
681        };
682        if let Some(recorded) = log.state().user_decision(approval_id) {
683            answer.option.clone_from(&recorded.agent_option);
684            if recorded.decision == cookie_agent_protocol::ApprovalUserDecision::Reject {
685                answer.note = recorded
686                    .feedback
687                    .as_ref()
688                    .map(|feedback| feedback.message.to_string());
689            }
690        }
691        answer
692    }
693
694    /// Steers the note the user rejected `request` with into the agent's
695    /// turn, so the agent reads why: an ACP permission answer carries no
696    /// text.
697    pub(in crate::runtime) async fn steer_rejection_note(
698        &self,
699        run: cookie_agent_protocol::RunId,
700        request: &wire::PermissionRequest,
701        note: &str,
702    ) {
703        let call = request
704            .tool_call
705            .title
706            .as_deref()
707            .or(request.title.as_deref())
708            .map(str::trim)
709            .filter(|title| !title.is_empty());
710        let text = match call {
711            Some(call) => format!("Rejected `{call}`: {note}"),
712            None => format!("Rejected: {note}"),
713        };
714        if let Err(error) = self
715            .steer(
716                run,
717                cookie_agent_protocol::UserContent::text(text),
718                cookie_agent_protocol::EventOrigin::new("user").expect("static event origin"),
719            )
720            .await
721        {
722            eprintln!("cookie-agent: a rejection's note was not steered: {error}");
723        }
724    }
725}
726
727/// The executor behind an approval of an ACP permission request: the
728/// agent runs the call itself, so there is nothing to run or revalidate.
729struct AnsweredExecutor;
730
731#[async_trait::async_trait]
732impl cookie_agent_runtime_api::PreparedExecutor for AnsweredExecutor {
733    async fn revalidate(&self) -> Result<(), cookie_agent_runtime_api::ToolError> {
734        Ok(())
735    }
736
737    async fn execute(
738        self: Box<Self>,
739        _ctx: cookie_agent_runtime_api::ToolExecutionContext,
740    ) -> Result<cookie_agent_runtime_api::ToolCompletion, cookie_agent_runtime_api::ToolError> {
741        Err(cookie_agent_runtime_api::ToolError::execution(
742            "an ACP agent runs its own calls",
743        ))
744    }
745}
