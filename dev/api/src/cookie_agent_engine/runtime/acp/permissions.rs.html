1//! An ACP agent's own permission requests
2//! (`docs/site/specs/acp-agents.md` section 8).
3//!
4//! A `session/request_permission` names one of the agent's tool calls. Its
5//! ACP kind maps onto one of cookie's actions, and its locations, command,
6//! URL, or MCP tool onto that action's resources; the agent's merged rules
7//! decide each, through the same pipeline as cookie's tools. A call of
8//! cookie's own tools, which reach cookie's MCP server where the rules
9//! decide them, and a `think` are allowed at once; `switch_mode` and every
10//! kind that maps onto no action go through the approval flow, as does an
11//! `ask`. The answer is the agent's allow-once or reject-once option, never
12//! an "always" one.
13

27/// What a permission request asks, by its mapping (spec section 8).
28#[derive(Clone, Debug, Eq, PartialEq)]
29pub(crate) enum MappedRequest {
30    /// Allowed at once: a call of cookie's own tools, which cookie's MCP
31    /// server decides when it arrives, or a `think`.
32    Allow,
33    /// Decided by the agent's rules: one action and the resources it
34    /// needs, each of which must be allowed.
35    Rules {
36        action: PermissionAction,
37        resources: Vec<String>,
38    },
39    /// Decided by the approval flow alone: `switch_mode` and every kind
40    /// that maps onto no action. `label` names the kind (an action label
41    /// no rule decides), `resource` what the request shows.
42    Ask {
43        label: PermissionAction,
44        resource: String,
45    },
46}
47
48/// The ACP tool kinds of ACP v1.
49const KNOWN_KINDS: [&str; 10] = [
50    "read",
51    "edit",
52    "delete",
53    "move",
54    "search",
55    "execute",
56    "think",
57    "fetch",
58    "switch_mode",
59    "other",
60];

62/// The first string member of `input` among `names`.
63fn input_string<'a>(input: Option<&'a Value>, names: &[&str]) -> Option<&'a str> {
64    let input = input?;
65    names
66        .iter()
67        .find_map(|name| input.get(*name).and_then(Value::as_str))
68        .filter(|value| !value.is_empty())
69}

71/// The command an `execute` call runs: its raw input's `command` (a string
72/// or an argument vector), else its title.
73fn command_of(call: &ToolCallFields) -> Option<String> {
74    let input = call.raw_input.as_ref();
75    if let Some(command) = input.and_then(|input| input.get("command")) {
76        match command {
77            Value::String(command) if !command.is_empty() => return Some(command.clone()),
78            Value::Array(arguments) => {
79                let words = arguments
80                    .iter()
81                    .filter_map(Value::as_str)
82                    .collect::<Vec<_>>();
83                if !words.is_empty() {
84                    return Some(words.join(" "));
85                }
86            }
87            _ => {}
88        }
89    }
90    input_string(input, &["cmd"])
91        .map(str::to_owned)
92        .or_else(|| call.title.clone())
93}

95/// The paths a file call touches: its locations, else its raw input's path,
96/// else its title.
97fn paths_of(call: &ToolCallFields) -> Vec<String> {
98    let locations = call
99        .locations
100        .iter()
101        .flatten()
102        .map(|location| location.path.clone())
103        .filter(|path| !path.is_empty())
104        .collect::<Vec<_>>();
105    if !locations.is_empty() {
106        return locations;
107    }
108    input_string(
109        call.raw_input.as_ref(),
110        &["path", "file_path", "filePath", "abs_path", "absPath"],
111    )
112    .map(str::to_owned)
113    .or_else(|| call.title.clone())
114    .into_iter()
115    .collect()
116}

118/// The MCP server and tool a call of kind `other` (or of a kind ACP v1
119/// does not know) goes to, as the adapters name them: Claude Code titles
120/// it `mcp__<server>__<tool>`, Codex `Tool: <server>/<tool>` with
121/// `{server, tool}` as its raw input, and Gemini CLI `<tool> (<server> MCP
122/// Server)`. The whole title must have the form, so a command or a path
123/// that merely contains one never matches.
124fn mcp_tool_of(call: &ToolCallFields) -> Option<(String, String)> {
125    let input = call.raw_input.as_ref();
126    let from_input = input_string(input, &["server", "serverName", "server_name"])
127        .zip(input_string(input, &["tool", "toolName", "tool_name"]));
128    if let Some((server, tool)) = from_input {
129        return Some((server.to_owned(), tool.to_owned()));
130    }
131    let title = call.title.as_deref()?.trim();
132    let named = if let Some(rest) = title.strip_prefix("mcp__") {
133        rest.split_once("__")
134    } else if let Some(rest) = title.strip_prefix("Tool: ") {
135        rest.split_once('/')
136    } else if let Some(rest) = title.strip_suffix(" MCP Server)") {
137        rest.rsplit_once(" (").map(|(tool, server)| (server, tool))
138    } else {
139        None
140    };
141    named
142        .filter(|(server, tool)| {
143            [server, tool].iter().all(|part| {
144                !part.is_empty() && !part.chars().any(|c| c.is_whitespace() || c.is_control())
145            })
146        })
147        .map(|(server, tool)| (server.to_owned(), tool.to_owned()))
148}

150/// A name as cookie publishes an MCP server's tools: ASCII letters,
151/// digits, `_`, and `-` kept, everything else `_`.
152fn mcp_name_part(value: &str) -> String {
153    value
154        .chars()
155        .map(|c| {
156            if c.is_ascii_alphanumeric() || matches!(c, '_' | '-') {
157                c
158            } else {
159                '_'
160            }
161        })
162        .collect()
163}

165/// Maps the tool call a permission request names onto cookie's action
166/// and resources: `call` as the request carries it, merged over `known`,
167/// what the agent reported of the call before.
168pub(crate) fn map_request(call: &ToolCallFields, known: Option<&ToolCallFields>) -> MappedRequest {
169    let merged = merge(known, call);
170    let kind = merged.kind.as_deref().unwrap_or("other");
171    match kind {
172        "read" | "search" => MappedRequest::Rules {
173            action: PermissionAction::READ,
174            resources: paths_of(&merged),
175        },
176        "edit" | "delete" | "move" => MappedRequest::Rules {
177            action: PermissionAction::WRITE,
178            resources: paths_of(&merged),
179        },
180        "execute" => MappedRequest::Rules {
181            action: PermissionAction::BASH,
182            resources: command_of(&merged).into_iter().collect(),
183        },
184        "fetch" => MappedRequest::Rules {
185            action: PermissionAction::WEBFETCH,
186            resources: input_string(merged.raw_input.as_ref(), &["url", "uri"])
187                .map(str::to_owned)
188                .or_else(|| merged.title.clone())
189                .into_iter()
190                .collect(),
191        },
192        "think" => MappedRequest::Allow,
193        "switch_mode" => ask(kind, &merged),
194        other => match mcp_tool_of(&merged) {
195            Some((server, _)) if server == BRIDGE_SERVER => MappedRequest::Allow,
196            Some((server, tool)) => MappedRequest::Rules {
197                action: PermissionAction::MCP,
198                resources: vec![format!(
199                    "{}_{}",
200                    mcp_name_part(&server),
201                    mcp_name_part(&tool)
202                )],
203            },
204            None if KNOWN_KINDS.contains(&other) => ask(other, &merged),
205            None => ask("other", &merged),
206        },
207    }
208}

210fn ask(kind: &str, call: &ToolCallFields) -> MappedRequest {
211    let label = PermissionAction::new(kind)
212        .unwrap_or_else(|_| PermissionAction::new("other").expect("`other` is an action label"));
213    let title = call.title.as_deref().map(str::trim).unwrap_or_default();
214    MappedRequest::Ask {
215        label,
216        resource: if title.is_empty() {
217            kind.to_owned()
218        } else {
219            title.to_owned()
220        },
221    }
222}

224/// `update` over `known`: each field the update carries replaces the known
225/// one.
226pub(crate) fn merge(known: Option<&ToolCallFields>, update: &ToolCallFields) -> ToolCallFields {
227    let mut merged = known.cloned().unwrap_or_default();
228    merged.tool_call_id.clone_from(&update.tool_call_id);
229    if update.title.is_some() {
230        merged.title.clone_from(&update.title);
231    }
232    if update.kind.is_some() {
233        merged.kind.clone_from(&update.kind);
234    }
235    if update.status.is_some() {
236        merged.status.clone_from(&update.status);
237    }
238    if update.content.is_some() {
239        merged.content.clone_from(&update.content);
240    }
241    if update.locations.is_some() {
242        merged.locations.clone_from(&update.locations);
243    }
244    if update.raw_input.is_some() {
245        merged.raw_input.clone_from(&update.raw_input);
246    }
247    if update.raw_output.is_some() {
248        merged.raw_output.clone_from(&update.raw_output);
249    }
250    merged
251}

253/// The option of `kind` (`allow_once` or `reject_once`) the agent offered.
254fn option_of<'a>(options: &'a [wire::PermissionOption], kind: &str) -> Option<&'a str> {
255    options
256        .iter()
257        .find(|option| option.kind == kind)
258        .map(|option| option.option_id.as_str())
259}

261impl Engine {
262    /// Decides ACP permission request `request` of the run `active` of
263    /// `agent`, whose session works in `cwd` on the agent's runtime, home
264    /// `home`: the answer to send the agent. An approval it asks for names
265    /// `tool_call_id`, the cookie call the agent's call is shown as.
266    #[allow(clippy::too_many_arguments)]
267    pub(in crate::runtime) async fn decide_acp_permission(
268        &self,
269        active: &ActiveRun,
270        run: cookie_agent_protocol::RunId,
271        agent: &RuntimeAgent,
272        cwd: &str,
273        home: Option<&str>,
274        request: &wire::PermissionRequest,
275        known: Option<&ToolCallFields>,
276        tool_call_id: Option<ToolCallId>,
277    ) -> Value {
278        let allowed = self
279            .acp_permission_allowed(active, run, agent, cwd, home, request, known, tool_call_id)
280            .await;
281        let option = if allowed {
282            option_of(&request.options, "allow_once")
283        } else {
284            option_of(&request.options, "reject_once")
285        };
286        wire::permission_outcome(option)
287    }
288
289    #[allow(clippy::too_many_arguments)]
290    async fn acp_permission_allowed(
291        &self,
292        active: &ActiveRun,
293        run: cookie_agent_protocol::RunId,
294        agent: &RuntimeAgent,
295        cwd: &str,
296        home: Option<&str>,
297        request: &wire::PermissionRequest,
298        known: Option<&ToolCallFields>,
299        tool_call_id: Option<ToolCallId>,
300    ) -> bool {
301        let (action, resources, by_rules) = match map_request(&request.tool_call, known) {
302            MappedRequest::Allow => return true,
303            MappedRequest::Rules { action, resources } => (action, resources, true),
304            MappedRequest::Ask { label, resource } => (label, vec![resource], false),
305        };
306        if resources.is_empty() {
307            return false;
308        }
309        let tool = format!("acp:{}", agent.id);
310        let prepared = resources
311            .iter()
312            .map(|resource| {
313                prepared_resource(
314                    action.clone(),
315                    action.as_str(),
316                    resource.as_bytes(),
317                    resource.as_bytes(),
318                    PreparedBindingLifetime::RestartStable,
319                    ApprovalResourceSource::PrimaryOperation,
320                )
321            })
322            .collect::<Result<Vec<_>, _>>();
323        let arguments = serde_json::json!({
324            "tool_call": request.tool_call.tool_call_id,
325            "action": action.as_str(),
326            "resources": resources,
327        });
328        let Ok(prepared) = prepared else {
329            return false;
330        };
331        let Ok(operation) = prepared_operation(
332            &tool,
333            &arguments,
334            vec![(action.clone(), "request_permission")],
335            prepared,
336            agent.id.as_str().as_bytes(),
337        ) else {
338            return false;
339        };
340        let evaluations = if !by_rules {
341            // No rule decides it: the kind maps onto no action.
342            operation
343                .resources()
344                .iter()
345                .zip(&resources)
346                .map(|(prepared, resource)| ApprovalEvaluation {
347                    resource_digest: prepared.binding_digest.clone(),
348                    effect: PermissionEffect::Ask,
349                    trace: DecisionTrace {
350                        action: action.clone(),
351                        normalized_resource: resource.clone(),
352                        candidates: Vec::new(),
353                        effect: PermissionEffect::Ask,
354                        precedence_reason: format!(
355                            "an ACP agent's `{}` request maps onto no action and always asks",
356                            action.as_str()
357                        ),
358                    },
359                })
360                .collect()
361        } else {
362            let Ok(session) = self.inner.store.get(active.session) else {
363                return false;
364            };
365            let overlay = session.permission_overlay.clone();
366            let labels = resources.iter().cloned().map(Some).collect::<Vec<_>>();
367            let workspace = std::path::Path::new(cwd);
368            let decision = self.inner.permissions.decide_operation_with_grants(
369                &active.policy.agent,
370                Some(&overlay),
371                None,
372                &operation,
373                &labels,
374                crate::permissions::PermissionScope::runtime(
375                    agent.runtime.as_str(),
376                    Some(workspace),
377                )
378                .with_home(home),
379            );
380            match decision.effect {
381                PermissionEffect::Allow => return true,
382                PermissionEffect::Deny => return false,
383                PermissionEffect::Ask => decision.evaluations,
384            }
385        };
386        let tool_call_id = tool_call_id.unwrap_or_else(ToolCallId::new_v7);
387        let request = match ApprovalRequest::new(
388            ApprovalId::new_v7(),
389            1,
390            tool_call_id,
391            ApprovalTrigger::PermissionPolicy,
392            operation,
393            evaluations,
394            ApprovalConstraints {
395                allow_once: true,
396                // A tree grant names an action the rules decide; a request
397                // no rule decides is answered once.
398                allow_tree_grant: by_rules && !action.is_path(),
399                cancellable: true,
400                expires_at: super::super::approval_flow::approval_expiry(
401                    self.inner.config.load().runtime.approval.timeout_ms,
402                ),
403            },
404        ) {
405            Ok(request) => request,
406            Err(_) => return false,
407        };
408        let executor: cookie_agent_runtime_api::PreparedExecutorCell =
409            std::sync::Arc::new(tokio::sync::Mutex::new(Some(Box::new(AnsweredExecutor))));
410        self.await_user_approval(
411            active,
412            run,
413            request,
414            executor,
415            true,
416            super::super::approval_flow::ApprovalToolInput {
417                tool_call_id,
418                name: &tool,
419                origin: None,
420                normalized_parameters: &arguments,
421            },
422        )
423        .await
424        .is_ok_and(|outcome| outcome.approved)
425    }
426}

428/// The executor behind an approval of an ACP permission request: the
429/// agent runs the call itself, so there is nothing to run or revalidate.
430struct AnsweredExecutor;
431
432#[async_trait::async_trait]
433impl cookie_agent_runtime_api::PreparedExecutor for AnsweredExecutor {
434    async fn revalidate(&self) -> Result<(), cookie_agent_runtime_api::ToolError> {
435        Ok(())
436    }
437
438    async fn execute(
439        self: Box<Self>,
440        _ctx: cookie_agent_runtime_api::ToolExecutionContext,
441    ) -> Result<cookie_agent_runtime_api::ToolCompletion, cookie_agent_runtime_api::ToolError> {
442        Err(cookie_agent_runtime_api::ToolError::execution(
443            "an ACP agent runs its own calls",
444        ))
445    }
446}

466    #[test]
467    fn mcp_tools_map_by_each_adapters_naming() {
468        // Claude Code, Codex, and Gemini CLI name an MCP tool each their way.
469        for (fields, expected) in [
470            (
471                json!({"toolCallId": "a", "kind": "other", "title": "mcp__exa__web_search"}),
472                rules(PermissionAction::MCP, "exa_web_search"),
473            ),
474            (
475                json!({"toolCallId": "b", "kind": "other", "title": "Tool: exa/web.search",
476                       "rawInput": {"server": "exa", "tool": "web.search", "arguments": {}}}),
477                rules(PermissionAction::MCP, "exa_web_search"),
478            ),
479            (
480                json!({"toolCallId": "c", "title": "web_search (exa MCP Server)"}),
481                rules(PermissionAction::MCP, "exa_web_search"),
482            ),
483            (
484                json!({"toolCallId": "d", "kind": "other", "title": "mcp__cookie__memory_read"}),
485                MappedRequest::Allow,
486            ),
487            (
488                json!({"toolCallId": "e", "kind": "other", "rawInput": {"server": "cookie", "tool": "cron_list"}}),
489                MappedRequest::Allow,
490            ),
491            (
492                json!({"toolCallId": "f", "kind": "other", "title": "memory_read (cookie MCP Server)"}),
493                MappedRequest::Allow,
494            ),
495        ] {
496            assert_eq!(
497                map_request(&call(fields.clone()), None),
498                expected,
499                "{fields}"
