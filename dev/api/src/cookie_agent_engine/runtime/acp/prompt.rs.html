19/// The context window the first prompt's skill listing is sized for: the
20/// agent's own is unknown, and Claude Code's and Codex's models have at
21/// least this.
22pub(crate) const SKILL_LISTING_CONTEXT_TOKENS: u64 = 200_000;

24/// The most files of one kind the clients attach to one message for an
25/// ACP agent.
26const MAX_FILES_PER_KIND: u32 = 20;
27
28/// What an agent with `caps` takes in a prompt, as the clients check
29/// attachments against: images, audio, and text files, each up to its
30/// upload cap.
31pub(crate) fn input_media(caps: &PromptCapabilities) -> Vec<InputMedia> {
32    let entry = |kind: &str, mime_types: Vec<String>| {
33        let image = kind == MediaKindLabel::IMAGE;
34        InputMedia {
35            kind: MediaKindLabel::new(kind).expect("static media kind"),
36            mime_types,
37            max_bytes: media_sniff::max_attachment_bytes(kind),
38            max_dimension: image.then_some(media_sniff::MAX_IMAGE_DIMENSION),
39            max_pixels: image.then_some(media_sniff::MAX_IMAGE_PIXELS),
40            max_count: MAX_FILES_PER_KIND,
41        }
42    };
43    let uploads = |kind: &str| {
44        upload_media_types(kind)
45            .into_iter()
46            .map(str::to_owned)
47            .collect::<Vec<_>>()
48    };
49    let mut media = Vec::new();
50    if caps.image {
51        media.push(entry(MediaKindLabel::IMAGE, uploads(MediaKindLabel::IMAGE)));
52    }
53    if caps.audio {
54        media.push(entry(MediaKindLabel::AUDIO, uploads(MediaKindLabel::AUDIO)));
55    }
56    if caps.embedded_context {
57        media.push(entry(
58            MediaKindLabel::TEXT,
59            vec![media_sniff::TEXT_MEDIA_TYPE.to_owned()],
60        ));
61    }
62    media
63}

65/// The prompt capabilities `media`, made by [`input_media`], were made from:
66/// what an agent that last started with them takes, without its process.
67pub(crate) fn caps_of(media: &[InputMedia]) -> PromptCapabilities {
68    let takes = |kind: &str| media.iter().any(|entry| entry.kind.as_str() == kind);
69    PromptCapabilities {
70        image: takes(MediaKindLabel::IMAGE),
71        audio: takes(MediaKindLabel::AUDIO),
72        embedded_context: takes(MediaKindLabel::TEXT),
73    }
74}

76/// How a file reaches the agent.
77#[derive(Debug, Eq, PartialEq)]
78enum Delivery {
79    Image,
80    Audio,
81    Text,
82    Note,
83}
84
85/// Why an agent with `caps` (named `agent`) takes `artifact` in no prompt
86/// block: its kind, its media type, or its size against what the clients
87/// are told it takes ([`input_media`]), as [`crate::media::user_file_fit`]
88/// decides for a model; `None` when it takes it.
89pub(crate) fn prompt_misfit(
90    artifact: &ArtifactRef,
91    agent: &dyn std::fmt::Display,
92    caps: &PromptCapabilities,
93) -> Option<String> {
94    crate::media::user_file_fit(
95        crate::media::user_file_kind(&artifact.media_type),
96        &artifact.media_type,
97        artifact.byte_length,
98        agent,
99        &input_media(caps),
100    )
101}

103/// What a prompt of an agent with `caps` sends after `artifact`'s label.
104fn delivery(artifact: &ArtifactRef, caps: &PromptCapabilities) -> Delivery {
105    let media_type = artifact.media_type.as_str();
106    if prompt_misfit(artifact, &"the agent", caps).is_some() {
107        Delivery::Note
108    } else if caps.image && media_type.starts_with("image/") {
109        Delivery::Image
110    } else if caps.audio && media_type.starts_with("audio/") {
111        Delivery::Audio
112    } else if caps.embedded_context && media_type == media_sniff::TEXT_MEDIA_TYPE {
113        Delivery::Text
114    } else {
115        // A PDF or a video: ACP's blob resource would carry it, but the
116        // agents that take embedded context read only text resources.
117        Delivery::Note
118    }
119}

121/// The URI a text file the user attached goes to the agent under: its
122/// name, under a scheme that names no path on the agent's machine.
123fn attachment_uri(artifact: &ArtifactRef) -> String {
124    let name = artifact.filename.as_deref().unwrap_or("attachment.txt");
125    let mut uri = String::from("attachment:///");
126    for byte in name.bytes() {
127        if byte.is_ascii_alphanumeric() || b"-._~".contains(&byte) {
128            uri.push(char::from(byte));
129        } else {
130            uri.push_str(&format!("%{byte:02X}"));
131        }
132    }
133    uri
134}

136/// The media cookie's tools return to an agent with `caps` over the MCP
137/// bridge, as tool results: the images and audio its prompts take
138/// (`docs/site/specs/attachment-handling.md` section 4.3).
139pub(crate) fn bridge_input_media(caps: &PromptCapabilities) -> Vec<ToolInputMedia> {
140    input_media(caps)
141        .into_iter()
142        .filter(|media| {
143            [MediaKindLabel::IMAGE, MediaKindLabel::AUDIO].contains(&media.kind.as_str())
144        })
145        .map(|media| ToolInputMedia {
146            kind: media.kind,
147            mime_types: media.mime_types,
148            max_bytes: media.max_bytes,
149            delivery: MediaDelivery::ToolResult,
150        })
151        .collect()
152}

154/// The prompt blocks of `artifact`, a file the user attached, for an agent
155/// with `caps` whose bridge offers `tools`: its label
156/// ([`cookie_agent_protocol::model_text::user_file_label`]), which names the
157/// artifact for `read_artifact`, `transfer`, and `<attach artifact=…/>`,
158/// then an `image`, an `audio`, or a text `resource` block when the agent
159/// takes it and `read` has its bytes. A file the prompt does not carry is
160/// its label alone, which says what the bridged tools can do with it (as
161/// [`crate::media::missing_file_ending`] decides for a model); one whose
162/// stored copy is gone is its label and a note saying so.
163pub(crate) fn file_blocks(
164    artifact: &ArtifactRef,
165    caps: &PromptCapabilities,
166    tools: FileTools,
167    read: impl FnOnce() -> Option<bytes::Bytes>,
168) -> Vec<Value> {
169    use base64::Engine as _;
170    use cookie_agent_protocol::model_text::{FileLabelEnding, user_file_label};
171    let label = |ending| wire::text_block(&user_file_label(artifact, ending));
172    let how = delivery(artifact, caps);
173    let uploaded = artifact.media_type == cookie_agent_protocol::FILE_UPLOAD_MEDIA_TYPE;
174    // A `file` upload's content says what `read_artifact` can do with it.
175    let bytes = (how != Delivery::Note || uploaded).then(read).flatten();
176    let block = match (&how, &bytes) {
177        (Delivery::Image, Some(bytes)) => Some(json!({
178            "type": "image",
179            "mimeType": artifact.media_type,
180            "data": base64::engine::general_purpose::STANDARD.encode(bytes),
181        })),
182        (Delivery::Audio, Some(bytes)) => Some(json!({
183            "type": "audio",
184            "mimeType": artifact.media_type,
185            "data": base64::engine::general_purpose::STANDARD.encode(bytes),
186        })),
187        (Delivery::Text, Some(bytes)) => std::str::from_utf8(bytes).ok().map(|text| {
188            json!({
189                "type": "resource",
190                "resource": {
191                    "uri": attachment_uri(artifact),
192                    "mimeType": artifact.media_type,
193                    "text": text,
194                },
195            })
196        }),
197        _ => None,
198    };
199    if let Some(block) = block {
200        return vec![label(FileLabelEnding::IncludedBelow), block];
201    }
202    let content = match bytes {
203        None if how != Delivery::Note || uploaded => FileContent::Missing,
204        None if artifact.media_type == media_sniff::TEXT_MEDIA_TYPE => FileContent::Text,
205        None => FileContent::Media(artifact.media_type.clone()),
206        Some(bytes) => crate::media::sniffed_content(&bytes, bytes.len() as u64),
207    };
208    let facts = FileFacts {
209        byte_length: artifact.byte_length,
210        content,
211    };
212    let ending = missing_file_ending(&facts, &bridge_input_media(caps), tools);
213    if facts.content == FileContent::Missing {
214        return vec![
215            label(ending),
216            wire::text_block(&format!(
217                "[{} ({}) is unavailable: its stored copy is missing or damaged]",
218                artifact.filename.as_deref().unwrap_or("file"),
219                artifact.media_type
220            )),
221        ];
222    }
223    vec![label(ending)]
224}

226/// Whether `title` is the start of the context cookie put ahead of the
227/// first prompt: an agent that names its session after its first message
228/// (Codex) names it after cookie's context there, which names nothing. The
229/// title is `agent`'s, so it is that agent's context it may repeat.
230pub(crate) fn title_repeats_context(
231    state: &crate::events::LogState,
232    agent: &cookie_agent_protocol::AgentId,
233    title: &str,
234) -> bool {
235    let words = |text: &str| text.split_whitespace().collect::<Vec<_>>().join(" ");
236    let title = words(title);
237    let title = title.trim_end_matches(['…', '.']).trim_end();
238    if title.is_empty() {
239        return false;
240    }
241    state
242        .acp_prompt_context(agent)
243        .first()
244        .is_some_and(|context| words(context).starts_with(title))
245}

269    #[test]
270    fn files_go_as_the_agent_takes_them_and_the_rest_as_a_note() {
271        let all = caps(true, true, true);
272        let bytes = || Some(bytes::Bytes::from_static(b"hello"));
273        use cookie_agent_protocol::model_text::FileLabelEnding;
274        let label = |artifact: &ArtifactRef, ending| {
275            wire::text_block(&cookie_agent_protocol::model_text::user_file_label(
276                artifact, ending,
277            ))
278        };
279        let both = FileTools {
280            read_artifact: true,
281            transfer: true,
282        };
283        let notes = artifact("text/plain", Some("my notes.md"));
284        assert_eq!(
285            file_blocks(&notes, &all, both, bytes),
286            [
287                label(&notes, FileLabelEnding::IncludedBelow),
288                json!({"type": "resource", "resource": {
289                    "uri": "attachment:///my%20notes.md", "mimeType": "text/plain", "text": "hello",
290                }})
291            ]
292        );
293        let wav = artifact("audio/wav", None);
294        let audio = file_blocks(&wav, &all, both, bytes);
295        assert_eq!(audio[0], label(&wav, FileLabelEnding::IncludedBelow));
296        assert_eq!(audio[1]["type"], "audio");
297        assert_eq!(audio[1]["mimeType"], "audio/wav");
298        assert_eq!(audio[1]["data"], "aGVsbG8=");
299        // Files the agent does not take are their label alone, which says
300        // what the bridged tools can do; the bytes are not even read.
301        for (media_type, caps, tools, ending) in [
302            // `read_artifact` returns no PDF over the bridge.
303            (
304                "application/pdf",
305                all.clone(),
306                both,
307                FileLabelEnding::UseTransfer,
308            ),
309            (
310                "application/pdf",
311                all.clone(),
312                FileTools::default(),
313                FileLabelEnding::OnlyName,
314            ),
315            (
316                "text/plain",
317                caps(true, true, false),
318                both,
319                FileLabelEnding::ReadArtifactReturnsIt,
320            ),
321            (
322                "audio/wav",
323                caps(true, false, true),
324                both,
325                FileLabelEnding::UseTransfer,
326            ),
327        ] {
328            let file = artifact(media_type, Some("x"));
329            let blocks = file_blocks(&file, &caps, tools, || {
330                panic!("read a file the agent cannot take")
331            });
332            assert_eq!(blocks, [label(&file, ending)], "{media_type}");
333        }
334        // A file the store no longer has is its label and a note.
335        let png = artifact("image/png", Some("a.png"));
336        let gone = file_blocks(&png, &all, both, || None);
337        assert_eq!(gone.len(), 2);
338        assert_eq!(gone[0], label(&png, FileLabelEnding::OnlyName));
339        assert_eq!(gone[1]["type"], "text");
340    }

342    #[test]
343    fn the_clients_are_told_what_the_agent_takes() {
344        let kinds = |caps: &PromptCapabilities| {
345            input_media(caps)
346                .iter()
347                .map(|media| media.kind.as_str().to_owned())
348                .collect::<Vec<_>>()
349        };
350        assert_eq!(kinds(&caps(true, true, true)), ["image", "audio", "text"]);
351        assert_eq!(kinds(&caps(false, false, true)), ["text"]);
352        assert!(kinds(&caps(false, false, false)).is_empty());
353        let text = &input_media(&caps(false, false, true))[0];
354        assert_eq!(text.mime_types, ["text/plain"]);
355        let image = &input_media(&caps(true, false, false))[0];
356        assert!(image.mime_types.iter().any(|mime| mime == "image/png"));
357        assert!(image.max_dimension.is_some());
358    }
