1//! The ACP runner (`docs/site/specs/acp-agents.md` sections 5 to 7): what
2//! takes a run of an ACP agent instead of the model loop.
3//!
4//! A run gets the agent's process and the cookie session's ACP session
5//! (resumed, loaded, or new), and sends each prompt round's input as
6//! `session/prompt`. What the agent reports while the prompt runs is
7//! recorded as model turns of the synthetic model `acp/<agent>` and tool
8//! calls of the source `acp`: text and thoughts stream as deltas, each new
9//! tool call commits the turn so far with the call in it, and its updates
10//! become the call's progress and, once it completes or fails, its
11//! termination. The prompt's stop reason ends the run: `end_turn` completes
12//! it with the last text (or runs the next round of queued input),
13//! `cancelled` cancels it, and the rest fail it.
14

83/// How a prompt round ended.
84enum RoundEnd {
85    Completed(Option<String>),
86    Cancelled,
87    Failed(String),
88    /// The agent stopped for `reason` (a stop reason other than
89    /// `end_turn` and `cancelled`): the run fails with `message`, and the
90    /// turn so far commits as finished for `finish`.
91    Stopped {
92        message: String,
93        finish: ModelFinishReason,
94    },
95}

97/// What the run's failure says, and how the turn so far finished, for an
98/// agent's stop reason other than `end_turn` and `cancelled`.
99fn stopped(reason: &str) -> RoundEnd {
100    let (message, finish) = match reason {
101        "max_tokens" => (
102            "the agent hit its output token limit (stop reason `max_tokens`)".to_owned(),
103            ModelFinishReason::Length,
104        ),
105        "max_turn_requests" => (
106            "the agent hit its turn request limit (stop reason `max_turn_requests`)".to_owned(),
107            ModelFinishReason::Other("max_turn_requests".to_owned()),
108        ),
109        "refusal" => (
110            "the agent refused (stop reason `refusal`)".to_owned(),
111            ModelFinishReason::Refused,
112        ),
113        other => (
114            format!("the agent stopped for an unknown reason `{other}`"),
115            ModelFinishReason::Error,
116        ),
117    };
118    RoundEnd::Stopped { message, finish }
119}

121/// One ACP option or mode list in the session's terms.
122fn options_of(
123    config_options: Option<&Vec<Value>>,
124    modes: Option<&wire::SessionModes>,
125) -> (Vec<AcpConfigOption>, bool) {
126    let mut options = config_options
127        .map(|options| options.iter().filter_map(config_option).collect::<Vec<_>>())
128        .unwrap_or_default();
129    let has_mode_option = options
130        .iter()
131        .any(|option| option.category.as_deref() == Some("mode") || option.id == "mode");
132    let modes_as_option =
133        !has_mode_option && modes.is_some_and(|modes| !modes.available_modes.is_empty());
134    if modes_as_option && let Some(modes) = modes {
135        options.push(AcpConfigOption {
136            id: "mode".to_owned(),
137            name: "Mode".to_owned(),
138            kind: AcpConfigOptionKind::Select,
139            description: None,
140            category: Some("mode".to_owned()),
141            current_value: modes.current_mode_id.clone(),
142            values: modes
143                .available_modes
144                .iter()
145                .map(|mode| AcpConfigValue {
146                    value: mode.id.clone(),
147                    name: mode.name.clone(),
148                    description: mode.description.clone(),
149                })
150                .collect(),
151        });
152    }
153    (options, modes_as_option)
154}

156/// A config option as the session shows it: a `select`, its grouped values
157/// flattened, or a `boolean`, whose values are `true` (On) and `false`
158/// (Off). An option of another type is left out, as ACP asks of a client
159/// that does not know it.
160fn config_option(option: &Value) -> Option<AcpConfigOption> {
161    let text =
162        |value: &Value, name: &str| value.get(name).and_then(Value::as_str).map(str::to_owned);
163    match option.get("type").and_then(Value::as_str) {
164        None | Some("select") => {}
165        Some("boolean") => {
166            let on = |value: bool, name: &str| AcpConfigValue {
167                value: value.to_string(),
168                name: name.to_owned(),
169                description: None,
170            };
171            return Some(AcpConfigOption {
172                id: text(option, "id")?,
173                name: text(option, "name").unwrap_or_default(),
174                kind: AcpConfigOptionKind::Boolean,
175                description: text(option, "description"),
176                category: text(option, "category"),
177                current_value: option
178                    .get("currentValue")
179                    .and_then(Value::as_bool)
180                    .unwrap_or(false)
181                    .to_string(),
182                values: vec![on(true, "On"), on(false, "Off")],
183            });
184        }
185        Some(_) => return None,
186    }
187    let mut values = Vec::new();
188    for entry in option
189        .get("options")
190        .and_then(Value::as_array)
191        .into_iter()
192        .flatten()
193    {
194        if let Some(group) = entry.get("options").and_then(Value::as_array) {
195            values.extend(group.iter().filter_map(config_value));
196        } else if let Some(value) = config_value(entry) {
197            values.push(value);
198        }
199    }
200    Some(AcpConfigOption {
201        id: text(option, "id")?,
202        name: text(option, "name").unwrap_or_default(),
203        kind: AcpConfigOptionKind::Select,
204        description: text(option, "description"),
205        category: text(option, "category"),
206        current_value: option
207            .get("currentValue")
208            .map(|value| match value {
209                Value::String(value) => value.clone(),
210                other => other.to_string(),
211            })
212            .unwrap_or_default(),
213        values,
214    })
215}

294    let number = |value: &Value, name: &str| value.get(name).and_then(Value::as_u64);
295    if let Some(usage) = response.filter(|usage| usage.is_object()) {
296        return Some(Usage {

298            input_tokens_no_cache: None,
299            input_tokens_cache_read: number(usage, "cachedReadTokens"),
300            input_tokens_cache_write: number(usage, "cachedWriteTokens"),
301            output_tokens: number(usage, "outputTokens"),
302            output_tokens_text: None,
303            output_tokens_reasoning: number(usage, "thoughtTokens"),

305    }
306    let used = update.and_then(|update| number(update, "used"))?;
307    Some(Usage {
308        input_tokens: Some(used),
309        ..Usage::default()
310    })

335/// A diff block of ACP diff content.
336fn diff_block(path: &str, old: Option<&str>, new: &str) -> DisplayBlock {
337    use cookie_agent_protocol::{DiffAsOf, DiffBlock, DiffChange, DiffHunk, DiffLine, DiffLineOp};
338    let old_text = old.unwrap_or_default();
339    let diff = similar::TextDiff::from_lines(old_text, new);
340    let (mut added, mut removed) = (0_u64, 0_u64);
341    let hunks = diff
342        .grouped_ops(3)
343        .into_iter()
344        .map(|group| {
345            let (first, last) = (
346                group[0].as_tag_tuple(),
347                group[group.len() - 1].as_tag_tuple(),
348            );
349            let old_range = first.1.start..last.1.end;
350            let new_range = first.2.start..last.2.end;
351            let mut lines = Vec::new();
352            for op in &group {
353                for change in diff.iter_changes(op) {
354                    let op = match change.tag() {
355                        similar::ChangeTag::Equal => DiffLineOp::Context,
356                        similar::ChangeTag::Insert => {
357                            added += 1;
358                            DiffLineOp::Add
359                        }
360                        similar::ChangeTag::Delete => {
361                            removed += 1;
362                            DiffLineOp::Remove
363                        }
364                    };
365                    lines.push(DiffLine {
366                        op,
367                        text: change.value().trim_end_matches(['\n', '\r']).to_owned(),
368                    });
369                }
370            }
371            let number = |start: usize, count: usize| start as u64 + u64::from(count > 0);
372            DiffHunk {
373                old_start: number(old_range.start, old_range.len()),
374                old_lines: old_range.len() as u64,
375                new_start: number(new_range.start, new_range.len()),
376                new_lines: new_range.len() as u64,
377                lines,
378            }
379        })
380        .collect();
381    let mut block = DisplayBlock::Diff(DiffBlock {
382        path: path.to_owned(),
383        old_path: None,
384        language: None,
385        change: if old.is_none() {
386            DiffChange::Created
387        } else {
388            DiffChange::Modified
389        },
390        hunks,
391        added,
392        removed,
393        omitted: None,
394        as_of: DiffAsOf::Execute,
395        full: None,
396    });
397    block.sanitize();
398    block
399}

401impl Engine {
402    /// The display blocks and text of a tool call's content: text as
403    /// Markdown, images as media, resource links as links, embedded
404    /// resources as code, diffs as diff blocks, the call's terminal as its
405    /// output (where the content names it, else first), and locations as
406    /// file links.
407    fn call_content(
408        &self,
409        session: SessionId,
410        call: &ToolCallFields,
411        terminal: Option<&Terminal>,
412    ) -> (Vec<DisplayBlock>, String) {
413        let terminal = terminal.filter(|terminal| terminal.shows());
414        let mut terminal_shown = false;
415        let mut blocks = Vec::new();
416        let mut text = String::new();
417        let say = |text: &mut String, line: &str| {
418            if !line.is_empty() {
419                if !text.is_empty() {
420                    text.push('\n');
421                }
422                text.push_str(line);
423            }
424        };
425        for content in call.content.iter().flatten() {
426            match content.get("type").and_then(Value::as_str) {
427                Some("diff") => {
428                    let path = content
429                        .get("path")
430                        .and_then(Value::as_str)
431                        .unwrap_or_default();
432                    let new = content
433                        .get("newText")
434                        .and_then(Value::as_str)
435                        .unwrap_or_default();
436                    let old = content.get("oldText").and_then(Value::as_str);
437                    blocks.push(diff_block(path, old, new));
438                    say(&mut text, &format!("edited {path}"));
439                }
440                Some("content") => {
441                    let block = content.get("content").cloned().unwrap_or(Value::Null);
442                    if let Some(shown) = self.content_block(session, &block) {
443                        blocks.push(shown);
444                    }
445                    say(&mut text, &wire::content_text(&block));
446                }
447                // The output the agent streamed for it; a terminal with none
448                // yet shows nothing.
449                Some("terminal") => {
450                    let id = content.get("terminalId").and_then(Value::as_str);
451                    if let Some(terminal) = terminal
452                        .filter(|terminal| !terminal_shown && id.is_none_or(|id| id == terminal.id))
453                    {
454                        blocks.extend(terminal.blocks());
455                        terminal_shown = true;
456                    }
457                }
458                _ => {}
459            }
460        }
461        if let Some(terminal) = terminal.filter(|_| !terminal_shown) {
462            blocks.splice(0..0, terminal.blocks());
463        }
464        for location in call.locations.iter().flatten() {
465            blocks.push(DisplayBlock::Link {
466                url: match location.line {
467                    Some(line) => format!("file://{}#L{line}", location.path),
468                    None => format!("file://{}", location.path),
469                },
470                label: Some(match location.line {
471                    Some(line) => format!("{}:{line}", location.path),
472                    None => location.path.clone(),
473                }),
474            });
475        }
476        for block in &mut blocks {
477            block.sanitize();
478        }
479        (blocks, text)
480    }
481
482    /// One ACP content block as a display block.
483    fn content_block(&self, session: SessionId, block: &Value) -> Option<DisplayBlock> {
484        let text = |name: &str| block.get(name).and_then(Value::as_str);
485        match text("type")? {
486            "text" => text("text")
487                .filter(|text| !text.trim().is_empty())
488                .map(|text| DisplayBlock::Markdown {
489                    text: text.to_owned(),
490                    omitted_lines: None,
491                }),
492            "image" => {
493                use base64::Engine as _;
494                let media_type = text("mimeType").unwrap_or("image/png");
495                let bytes = base64::engine::general_purpose::STANDARD
496                    .decode(text("data")?)
497                    .ok()?;
498                let (_, sha256) = self.inner.artifacts.retain(session, &bytes).ok()?;
499                Some(DisplayBlock::Media {
500                    artifact: cookie_agent_protocol::ArtifactRef {
501                        sha256,
502                        byte_length: bytes.len() as u64,
503                        media_type: media_type.to_owned(),
504                        filename: None,
505                    },
506                    alt: None,
507                })
508            }
509            "resource_link" => Some(DisplayBlock::Link {
510                url: text("uri")?.to_owned(),
511                label: text("title").or_else(|| text("name")).map(str::to_owned),
512            }),
513            "resource" => {
514                let resource = block.get("resource")?;
515                let field = |name: &str| resource.get(name).and_then(Value::as_str);
516                match field("text") {
517                    Some(body) => Some(DisplayBlock::Code {
518                        text: body.to_owned(),
519                        language: None,
520                        start_line: None,
521                        omitted_lines: None,
522                    }),
523                    None => Some(DisplayBlock::Link {
524                        url: field("uri")?.to_owned(),
525                        label: None,
526                    }),
527                }
528            }
529            _ => None,
530        }
531    }
532}

582/// How a call is presented from what the agent reported of it so far:
583/// its title, its kind's hint, what its raw input names, and the raw input
584/// as its arguments.
585fn presentation_of(fields: &ToolCallFields) -> PreparedPresentation {
586    let title = display_text(fields.title.as_deref().unwrap_or("ACP tool call"));
587    let mut presentation =
588        PreparedPresentation::new(title, primary_argument(fields.raw_input.as_ref()))
589            .with_hint(kind_hint(fields.kind.as_deref()));
590    if let Some(input) = fields
591        .raw_input
592        .clone()
593        .filter(|input| input.as_object().is_none_or(|input| !input.is_empty()))
594    {
595        presentation = presentation.with_arguments([DisplayBlock::Json { value: input }]);
596    }
597    presentation.fit();
598    presentation
599}

601/// The tool name a call's ACP kind records.
602fn kind_name(kind: Option<&str>) -> SafeCode {
603    let kind = kind.unwrap_or("other");
604    let sanitized = kind
605        .chars()
606        .map(|character| {
607            if character.is_ascii_lowercase() || character.is_ascii_digit() {
608                character
609            } else {
610                '_'
611            }
612        })
613        .collect::<String>();
614    SafeCode::new(format!("acp.{sanitized}"))
615        .unwrap_or_else(|_| SafeCode::new("acp.other").expect("static code"))
616}

719    async fn ensure_step(&mut self) -> Result<(), EngineError> {
720        if self.step.is_some() {
721            return Ok(());
722        }
723        let attempt_id = AttemptId::new_v7();
724        let ordinal = self.next_attempt;
725        self.next_attempt += 1;
726        let binding = &self.active.policy.agent.fallback_chain[0];
727        self.engine
728            .append(
729                self.session(),
730                Some(self.run),
731                origin(),
732                Event::ModelAttemptStarted {
733                    attempt_id,
734                    attempt_ordinal: ordinal,
735                    fallback_index: 0,
736                    retry_ordinal: self.retries,
737                    resolved_model: binding.resolved.clone(),
738                    prompt_fingerprint: self.active.policy.agent.prompt_fingerprint.clone(),
739                },
740            )
741            .await?;
742        self.step = Some(Step {
743            attempt_id,
744            parts: Vec::new(),
745            streaming: None,
746            message_ids: Vec::new(),
747        });
748        Ok(())
749    }

751    async fn stream(&mut self, kind: StreamedOutputKind, text: String) -> Result<(), EngineError> {
752        if text.is_empty() {
753            return Ok(());
754        }
755        self.ensure_step().await?;
756        let session = self.session();
757        let run = self.run;
758        let step = self.step.as_mut().expect("step started");
759        self.engine
760            .mark_output_part(session, run, step.attempt_id, &mut step.streaming, kind)
761            .await?;
762        let delta = match kind {
763            StreamedOutputKind::Text => Event::TextDelta {
764                attempt_id: step.attempt_id,
765                text: text.clone(),
766            },
767            StreamedOutputKind::Reasoning => Event::ReasoningDelta {
768                attempt_id: step.attempt_id,
769                text: text.clone(),
770            },
771        };
772        self.engine.publish_transient(session, run, delta);
773        match (step.parts.last_mut(), kind) {
774            (Some(PersistedAssistantPart::Text { text: last, .. }), StreamedOutputKind::Text)
775            | (
776                Some(PersistedAssistantPart::Reasoning { text: last, .. }),
777                StreamedOutputKind::Reasoning,
778            ) => last.push_str(&text),
779            (_, StreamedOutputKind::Text) => step.parts.push(PersistedAssistantPart::Text {
780                text,
781                metadata: None,
782            }),
783            (_, StreamedOutputKind::Reasoning) => {
784                step.parts.push(PersistedAssistantPart::Reasoning {
785                    text,
786                    metadata: None,
787                });
788            }
789        }
790        Ok(())
791    }

793    /// Adds the image content block `block` to the turn as a file part,
794    /// its bytes retained as an artifact; one that cannot be read or kept
795    /// becomes the note `[image]`.
796    async fn image(&mut self, block: &Value) -> Result<(), EngineError> {
797        use base64::Engine as _;
798        let session = self.session();
799        let media_type = block
800            .get("mimeType")
801            .and_then(Value::as_str)
802            .unwrap_or("image/png");
803        let file = block
804            .get("data")
805            .and_then(Value::as_str)
806            .and_then(|data| base64::engine::general_purpose::STANDARD.decode(data).ok())
807            .and_then(|bytes| {
808                let media_type = cookie_agent_protocol::MimeType::new(media_type).ok()?;
809                let (reference, sha256) =
810                    self.engine.inner.artifacts.retain(session, &bytes).ok()?;
811                Some(cookie_agent_protocol::PersistedFilePart {
812                    media_type,
813                    filename: None,
814                    source: cookie_agent_protocol::PersistedFileSource::Artifact {
815                        byte_length: bytes.len() as u64,
816                        sha256: cookie_agent_protocol::Sha256Digest::new(sha256).ok()?,
817                        reference,
818                    },
819                    metadata: None,
820                })
821            });
822        let Some(file) = file else {
823            return self
824                .stream(StreamedOutputKind::Text, "[image]".to_owned())
825                .await;
826        };
827        self.ensure_step().await?;
828        let step = self.step.as_mut().expect("step started");
829        step.parts.push(PersistedAssistantPart::File { file });
830        // Text after it starts a new part.
831        step.streaming = None;
832        Ok(())
833    }

835    /// Commits the step as a model turn finished for `finish`; returns its
836    /// sequence number.
837    async fn commit(&mut self, finish: ModelFinishReason) -> Result<u64, EngineError> {
838        self.ensure_step().await?;
839        let step = self.step.take().expect("step started");
840        let model_turn_seq = self.engine.next_model_turn_seq(self.session())?;
841        self.last_text = step
842            .parts
843            .iter()
844            .filter_map(|part| match part {
845                PersistedAssistantPart::Text { text, .. } => Some(text.as_str()),
846                _ => None,
847            })
848            .collect();
849        let binding = &self.active.policy.agent.fallback_chain[0];
850        let mut response_metadata = cookie_agent_protocol::SafeModelMetadata::default();
851        if !step.message_ids.is_empty() {
852            response_metadata.insert(
853                super::lifecycle::MESSAGE_IDS.to_owned(),
854                Value::from(step.message_ids),
855            );
856        }
857        self.engine
858            .append(
859                self.session(),
860                Some(self.run),
861                origin(),
862                Event::ModelTurnCommitted {
863                    attempt_id: step.attempt_id,
864                    model_turn_seq,
865                    resolved_model: binding.resolved.clone(),
866                    input_through_seq: self.input_through_seq,
867                    turn: PersistedModelTurn {
868                        content: step.parts,
869                        provider_options: Default::default(),
870                        finish_reason: finish,
871                        usage: Usage::default(),
872                        response_metadata,
873                        provider_metadata: Default::default(),
874                        native_replay: None,
875                    },
876                    warnings: Vec::new(),
877                },
878            )
879            .await?;
880        self.last_turn_seq = Some(model_turn_seq);
881        self.retries = 0;
882        if let Some(mut claim) = self.claim.take() {
883            claim.release().await?;
884        }
885        for mut claim in std::mem::take(&mut self.steer_claims) {
886            claim.release().await?;
887        }
888        Ok(model_turn_seq)
889    }

891    /// Takes the session's input to steer and sends it into the agent's
892    /// running turn: the turn so far commits, so the input stands after it,
893    /// then the input is promoted and claimed like a round's, and
894    /// `_session/steering` carries it. `None` when there was none.
895    async fn begin_steer(&mut self) -> Result<Option<PendingSteer>, EngineError> {
896        if self
897            .step
898            .as_ref()
899            .is_some_and(|step| !step.parts.is_empty())
900        {
901            self.commit(ModelFinishReason::Stop).await?;
902        }
903        let (session, run, engine) = (self.session(), self.run, self.engine);
904        engine
905            .request(session, |reply| SessionCommand::PromotePendingOrComplete {
906                run,
907                final_text: None,
908                complete_if_empty: false,
909                reply,
910            })
911            .await?;
912        let claim = engine.prompt_events(session, run).await?;
913        // A cancel that arrived while the input was taken stops the turn
914        // instead: the round's loop forwards it next.
915        if self.active.cancellation.is_cancelled() {
916            return Ok(None);
917        }
918        let caps = self
919            .live
920            .process
921            .init()
922            .agent_capabilities
923            .prompt_capabilities
924            .clone();
925        let (prompt, steered) = engine.input_since(&claim.events, self.input_through_seq, &caps);
926        if prompt.is_empty() {
927            return Ok(None);
928        }
929        let through = claim
930            .events
931            .last()
932            .map_or(self.input_through_seq, |event| event.seq);
933        let (reply, answer) = tokio::sync::oneshot::channel();
934        let process = Arc::clone(&self.live.process);
935        let session_id = self.live.acp_session_id.clone();
936        tokio::spawn(async move {
937            let answer = process
938                .peer
939                .request::<_, wire::SteeringResponse>(
940                    wire::method::SESSION_STEERING,
941                    &wire::SteeringRequest {
942                        session_id: &session_id,
943                        prompt: &prompt,
944                        meta: serde_json::json!({"steering": {"idleBehavior": "promptRequired"}}),
945                    },
946                )
947                .await
948                .map(|response| response.outcome)
949                .map_err(|error| agent_error_message(&error));
950            let _ = reply.send(answer);
951        });
952        Ok(Some(PendingSteer {
953            answer,
954            claim,
955            through,
956            steered,
957        }))
958    }

960    /// What the agent answered a steer: taken (`injected`, or a turn it
961    /// started for it), the turns after cover the input and the session
962    /// records that it was steered; otherwise the next round sends it.
963    async fn end_steer(
964        &mut self,
965        pending: PendingSteer,
966        answer: Result<String, String>,
967    ) -> Result<(), EngineError> {
968        match answer.as_deref() {
969            Ok("injected" | "startedNewTurn") => {
970                self.input_through_seq = self.input_through_seq.max(pending.through);
971                self.steer_claims.push(pending.claim);
972                self.record_update(AcpSessionUpdate {
973                    steered: Some(pending.steered),
974                    ..AcpSessionUpdate::default()
975                })
976                .await
977            }
978            Ok("promptRequired") => {
979                self.unsteered = true;
980                Ok(())
981            }
982            other => {
983                self.unsteered = true;
984                let reason = match other {
985                    Ok(outcome) => format!("it answered `{outcome}`"),
986                    Err(error) => error.clone(),
987                };
988                self.engine
989                    .acp_notice(
990                        self.session(),
991                        Some(self.run),
992                        format!(
993                            "The agent did not take the message into its running turn ({reason}); \
994                             it goes with the agent's next turn."
995                        ),
996                    )
997                    .await;
998                Ok(())
999            }
1000        }
1001    }

1017    /// Handles one `session/update` of the round.
1018    async fn update(&mut self, update: Value) -> Result<(), EngineError> {
1019        let kind = update
1020            .get("sessionUpdate")
1021            .and_then(Value::as_str)
1022            .unwrap_or_default()
1023            .to_owned();
1024        match kind.as_str() {
1025            // An image in the agent's message is media of the turn, as a
1026            // model's generated image is.
1027            "agent_message_chunk"
1028                if update
1029                    .get("content")
1030                    .and_then(|content| content.get("type"))
1031                    .and_then(Value::as_str)
1032                    == Some("image") =>
1033            {
1034                self.image(update.get("content").expect("matched above"))
1035                    .await
1036            }
1037            "agent_message_chunk" | "agent_thought_chunk" => {
1038                let text = update
1039                    .get("content")
1040                    .map(wire::content_text)
1041                    .unwrap_or_default();
1042                let kind = if kind == "agent_message_chunk" {
1043                    StreamedOutputKind::Text
1044                } else {
1045                    StreamedOutputKind::Reasoning
1046                };
1047                self.stream(kind, text).await?;
1048                // The message's ID names it as a fork point later.
1049                if kind == StreamedOutputKind::Text
1050                    && let Some(id) = update.get("messageId").and_then(Value::as_str)
1051                    && let Some(step) = self.step.as_mut()
1052                    && step.message_ids.last().map(String::as_str) != Some(id)
1053                {
1054                    step.message_ids.push(id.to_owned());
1055                }
1056                Ok(())
1057            }
1058            // Subagents stay flat on purpose: Claude Code's Task subagents
1059            // arrive as ordinary tool calls whose
1060            // `_meta.claudeCode.parentToolUseId` names the Task call, and ACP's
1061            // `subagent_update` and child sessions (offered to a client that
1062            // declares `clientCapabilities.subagents`) are unstable. Neither
1063            // nesting by `parentToolUseId` nor child sessions is implemented
1064            // until the subagent API is stable.
1065            "tool_call" | "tool_call_update" => {
1066                let fields: ToolCallFields =
1067                    serde_json::from_value(update.clone()).unwrap_or_default();
1068                self.tool_call(fields).await
1069            }
1070            "plan" => {
1071                self.record_update(AcpSessionUpdate {
1072                    plan: Some(Open::Known(plan_block(&update))),
1073                    ..AcpSessionUpdate::default()
1074                })
1075                .await
1076            }
1077            _ => self.metadata(&kind, &update).await,
1078        }
1079    }

1081    /// An update about the session rather than the turn.
1082    async fn metadata(&mut self, kind: &str, update: &Value) -> Result<(), EngineError> {
1083        match kind {
1084            "available_commands_update" => {
1085                self.record_update(AcpSessionUpdate {
1086                    commands: Some(commands_of(update)),
1087                    ..AcpSessionUpdate::default()
1088                })
1089                .await
1090            }
1091            "config_option_update" => {
1092                let options = update
1093                    .get("configOptions")
1094                    .and_then(Value::as_array)
1095                    .cloned();
1096                let (options, _) = options_of(options.as_ref(), None);
1097                self.live.options = options.clone();
1098                self.engine.remember_live(&self.live);
1099                self.record_update(AcpSessionUpdate {
1100                    options: Some(options),
1101                    ..AcpSessionUpdate::default()
1102                })
1103                .await
1104            }
1105            // The mode the agent switched to, such as Claude Code leaving
1106            // plan mode on its own: the option that shows its modes follows,
1107            // whether cookie made it of the modes or the agent offers its own
1108            // mode option beside them (Claude Code's adapter does).
1109            "current_mode_update" => {
1110                let Some(mode) = update.get("currentModeId").and_then(Value::as_str) else {
1111                    return Ok(());
1112                };
1113                let mut changed = false;
1114                for option in &mut self.live.options {
1115                    let shows_modes =
1116                        option.id == "mode" || option.category.as_deref() == Some("mode");
1117                    if shows_modes
1118                        && option.current_value != mode
1119                        && option.values.iter().any(|value| value.value == mode)
1120                    {
1121                        mode.clone_into(&mut option.current_value);
1122                        changed = true;
1123                    }
1124                }
1125                if !changed {
1126                    return Ok(());
1127                }
1128                self.engine.remember_live(&self.live);
1129                self.record_update(AcpSessionUpdate {
1130                    options: Some(self.live.options.clone()),
1131                    ..AcpSessionUpdate::default()
1132                })
1133                .await
1134            }
1135            "session_info_update" => {
1136                if let Some(title) = update.get("title").and_then(Value::as_str) {
1137                    self.engine
1138                        .set_acp_title(self.session(), self.run, title)
1139                        .await?;
1140                }
1141                Ok(())
1142            }
1143            // The agent's context use: what the clients show as the
1144            // session's context, recorded as it changes.
1145            "usage_update" => {
1146                self.usage_update = Some(update.clone());
1147                match context_of(update).filter(|context| self.context != Some(*context)) {
1148                    Some(context) => {
1149                        self.context = Some(context);
1150                        self.record_update(AcpSessionUpdate {
1151                            context: Some(context),
1152                            ..AcpSessionUpdate::default()
1153                        })
1154                        .await
1155                    }
1156                    None => Ok(()),
1157                }
1158            }
1159            // The session notices RFD: an advisory line for the user.
1160            "notice" => {
1161                let text = |name: &str| update.get(name).and_then(Value::as_str);
1162                let severity = match text("severity") {
1163                    Some("error") => AcpNoticeSeverity::Error,
1164                    Some("warning") => AcpNoticeSeverity::Warning,
1165                    _ => AcpNoticeSeverity::Info,
1166                };
1167                let Some(title) = text("title").filter(|title| !title.trim().is_empty()) else {
1168                    return Ok(());
1169                };
1170                self.notice(severity, title, text("description")).await
1171            }
1172            // The session compaction RFD: a compaction's summary streams,
1173            // then its terminal status says how it went.
1174            "compaction_summary_chunk" => {
1175                if let (Some(id), Some(content)) = (
1176                    update.get("compactionId").and_then(Value::as_str),
1177                    update.get("content"),
1178                ) {
1179                    self.compactions
1180                        .entry(id.to_owned())
1181                        .or_default()
1182                        .push_str(&wire::content_text(content));
1183                }
1184                Ok(())
1185            }
1186            "compaction_update" => {
1187                let id = update
1188                    .get("compactionId")
1189                    .and_then(Value::as_str)
1190                    .unwrap_or_default()
1191                    .to_owned();
1192                match update.get("status").and_then(Value::as_str) {
1193                    Some("completed") => {
1194                        let streamed = self.compactions.remove(&id).unwrap_or_default();
1195                        let summary = update
1196                            .get("summary")
1197                            .and_then(Value::as_array)
1198                            .map(|blocks| {
1199                                blocks
1200                                    .iter()
1201                                    .map(wire::content_text)
1202                                    .collect::<Vec<_>>()
1203                                    .join("\n")
1204                            })
1205                            .filter(|summary| !summary.trim().is_empty())
1206                            .unwrap_or(streamed);
1207                        self.notice(
1208                            AcpNoticeSeverity::Info,
1209                            "The agent compacted its conversation",
1210                            Some(summary.trim()).filter(|summary| !summary.is_empty()),
1211                        )
1212                        .await
1213                    }
1214                    Some("failed") => {
1215                        self.compactions.remove(&id);
1216                        let error = update.get("error").map(|error| match error {
1217                            Value::String(error) => error.clone(),
1218                            other => other
1219                                .get("message")
1220                                .and_then(Value::as_str)
1221                                .map_or_else(|| other.to_string(), str::to_owned),
1222                        });
1223                        self.notice(
1224                            AcpNoticeSeverity::Warning,
1225                            "The agent could not compact its conversation",
1226                            error.as_deref(),
1227                        )
1228                        .await
1229                    }
1230                    Some("cancelled") => {
1231                        self.compactions.remove(&id);
1232                        Ok(())
1233                    }
1234                    _ => Ok(()),
1235                }
1236            }
1237            _ => Ok(()),
1238        }
1239    }

1261    async fn tool_call(&mut self, mut update: ToolCallFields) -> Result<(), EngineError> {
1262        if update.tool_call_id.is_empty() {
1263            return Ok(());
1264        }
1265        let terminal_meta = terminal::take_meta(&mut update.meta);
1266        // Nothing but output: the live display grows by appends.
1267        let output_only = update.title.is_none()
1268            && update.kind.is_none()
1269            && update.status.is_none()
1270            && update.content.is_none()
1271            && update.locations.is_none()
1272            && update.raw_input.is_none()
1273            && update.raw_output.is_none();
1274        if let Some(shown) = self.calls.get_mut(&update.tool_call_id) {
1275            if shown.terminated {
1276                return Ok(());
1277            }
1278            shown.fields = permissions::merge(Some(&shown.fields), &update);
1279            let fields = shown.fields.clone();
1280            let tool_call_id = shown.tool_call_id;
1281            // Its title, kind, or raw input can arrive after it started
1282            // (Claude Code's adapter sends the raw input in an update).
1283            let presentation = presentation_of(&fields);
1284            if presentation != shown.presented {
1285                shown.presented = presentation.clone();
1286                self.record_update(AcpSessionUpdate {
1287                    tool_presentation: Some(cookie_agent_protocol::AcpToolPresentation {
1288                        tool_call_id,
1289                        presentation,
1290                    }),
1291                    ..AcpSessionUpdate::default()
1292                })
1293                .await?;
1294            }
1295            let appends = self
1296                .feed_terminal(&fields.tool_call_id, terminal_meta)
1297                .await;
1298            let index = self
1299                .calls
1300                .get(&fields.tool_call_id)
1301                .and_then(|shown| shown.terminal_index);
1302            if output_only && let Some(index) = index {
1303                self.append_output(tool_call_id, &fields, index, appends);
1304                return Ok(());
1305            }
1306            return self.progress_or_end(&fields.tool_call_id).await;
1307        }
1308        let fields = permissions::merge(None, &update);
1309        let mut call_id = fields.tool_call_id.clone();
1310        while !self.used_call_ids.insert(call_id.clone()) {
1311            call_id.push('\'');
1312        }
1313        let model_call_id = ModelCallId::new(call_id).map_err(|error| {
1314            EngineError::Acp(format!("the agent's tool call ID is unusable: {error}"))
1315        })?;
1316        self.ensure_step().await?;
1317        let step = self.step.as_mut().expect("step started");
1318        step.parts.push(PersistedAssistantPart::ToolCall {
1319            id: model_call_id.clone(),
1320            provider_item_id: None,
1321            name: kind_name(fields.kind.as_deref()),
1322            input: fields
1323                .raw_input
1324                .clone()
1325                .unwrap_or_else(|| serde_json::json!({})),
1326            raw_input: None,
1327            metadata: None,
1328        });
1329        let content_index = (step.parts.len() - 1) as u32;
1330        let model_turn_seq = self.commit(ModelFinishReason::ToolCalls).await?;
1331        let owner = AssistantToolCallRef {
1332            model_turn_seq,
1333            content_index,
1334            model_call_id,
1335            provider_item_id: None,
1336        };
1337        let tool_call_id = ToolCallId::new_v7();
1338        let name = kind_name(fields.kind.as_deref()).as_str().to_owned();
1339        let presented = presentation_of(&fields);
1340        let presentation = ToolPresentation::new(
1341            presented.clone(),
1342            ToolIdentity {
1343                published_name: fields
1344                    .tool_name()
1345                    .map(|name| name.chars().take(128).collect())
1346                    .unwrap_or_else(|| name.clone()),
1347                name,
1348                source: Some(Open::Known(ToolSource::Acp {
1349                    agent: self.live.agent.id.clone(),
1350                    runtime_id: self.live.agent.runtime.clone(),
1351                })),
1352            },
1353        );
1354        let fingerprint = super::super::tool_execution::fallback_operation_fingerprint(
1355            &cookie_agent_runtime_api::ToolCall {
1356                id: tool_call_id,
1357                name: format!("acp_{}", fields.kind.as_deref().unwrap_or("other")),
1358                arguments: fields.raw_input.clone().unwrap_or(Value::Null),
1359            },
1360            None,
1361        );
1362        self.engine
1363            .append(
1364                self.session(),
1365                Some(self.run),
1366                origin(),
1367                Event::ToolCallStarted {
1368                    start: ToolCallStart {
1369                        tool_call_id,
1370                        output: ToolOutputDeclaration::default(),
1371                        owner,
1372                        presentation,
1373                        operation_fingerprint: fingerprint,
1374                        background: false,
1375                    },
1376                },
1377            )
1378            .await?;
1379        self.calls.insert(
1380            fields.tool_call_id.clone(),
1381            ShownCall {
1382                tool_call_id,
1383                fields: fields.clone(),
1384                presented,
1385                terminated: false,
1386                terminal: None,
1387                terminal_index: None,
1388            },
1389        );
1390        self.feed_terminal(&fields.tool_call_id, terminal_meta)
1391            .await;
1392        self.progress_or_end(&fields.tool_call_id).await
1393    }

1395    /// Applies what an update says of call `acp_id`'s terminal, opening the
1396    /// terminal on its first mention; returns the appends its new output
1397    /// makes to the live display.
1398    async fn feed_terminal(&mut self, acp_id: &str, meta: terminal::TerminalMeta) -> Vec<String> {
1399        if meta.is_empty() {
1400            return Vec::new();
1401        }
1402        let session = self.active.session;
1403        let limits = self.active.policy.result_limits.tool_output;
1404        let Some(shown) = self.calls.get_mut(acp_id) else {
1405            return Vec::new();
1406        };
1407        if shown.terminal.is_none() {
1408            let id = meta.id.clone().unwrap_or_else(|| acp_id.to_owned());
1409            shown.terminal =
1410                Some(Terminal::open(self.engine, session, shown.tool_call_id, id, limits).await);
1411        }
1412        let terminal = shown.terminal.as_mut().expect("opened above");
1413        let appends = match &meta.output {
1414            Some(output) => terminal.write(output).await,
1415            None => Vec::new(),
1416        };
1417        if let Some(exit) = meta.exit {
1418            terminal.exit = Some(exit);
1419        }
1420        appends
1421    }

1449    /// Shows the new state of the call the agent knows as `acp_id`, or ends
1450    /// it once it completed or failed.
1451    async fn progress_or_end(&mut self, acp_id: &str) -> Result<(), EngineError> {
1452        let session = self.session();
1453        let Some(shown) = self.calls.get_mut(acp_id) else {
1454            return Ok(());
1455        };
1456        match shown.fields.status.as_deref() {
1457            Some("completed") => self.end_call(acp_id, true).await,
1458            Some("failed") => self.end_call(acp_id, false).await,
1459            _ => {
1460                let fields = &shown.fields;
1461                let tool_call_id = shown.tool_call_id;
1462                let (blocks, _) =
1463                    self.engine
1464                        .call_content(session, fields, shown.terminal.as_ref());
1465                shown.terminal_index = blocks
1466                    .iter()
1467                    .position(|block| matches!(block, DisplayBlock::Terminal { .. }))
1468                    .and_then(|index| u32::try_from(index).ok());
1469                let message = display_text(fields.title.as_deref().unwrap_or("ACP tool call"));
1470                self.engine.publish_transient(
1471                    self.session(),
1472                    self.run,
1473                    Event::ToolCallProgress {
1474                        tool_call_id,
1475                        message,
1476                        display: (!blocks.is_empty()).then(|| {
1477                            Open::Known(cookie_agent_protocol::DisplayDelta::Replace {
1478                                blocks: blocks.into_iter().map(Open::Known).collect(),
1479                            })
1480                        }),
1481                    },
1482                );
1483                Ok(())
1484            }
1485        }
1486    }

1488    /// Terminates the call the agent knows as `acp_id`: completed when
1489    /// `succeeded`, failed otherwise.
1490    async fn end_call(&mut self, acp_id: &str, succeeded: bool) -> Result<(), EngineError> {
1491        let Some(shown) = self.calls.get_mut(acp_id) else {
1492            return Ok(());
1493        };
1494        if shown.terminated {
1495            return Ok(());
1496        }
1497        shown.terminated = true;
1498        let session = self.active.session;
1499        let (mut blocks, mut text) =
1500            self.engine
1501                .call_content(session, &shown.fields, shown.terminal.as_ref());
1502        let terminal = shown.terminal.take();
1503        if let Some(output) = &shown.fields.raw_output {
1504            if text.is_empty() {
1505                text = match output {
1506                    Value::String(output) => output.clone(),
1507                    other => other.to_string(),
1508                };
1509            }
1510            if shown.fields.content.as_ref().is_none_or(Vec::is_empty) {
1511                blocks.extend(raw_output_block(output));
1512            }
1513        }
1514        let title = display_text(shown.fields.title.as_deref().unwrap_or("ACP tool call"));
1515        let tool_call_id = shown.tool_call_id;
1516        let mut result = PersistedToolResult {
1517            title,
1518            output: truncate(&text, PersistedToolResult::MAX_OUTPUT_BYTES),
1519            display: cookie_agent_protocol::fit_result_display(
1520                blocks.into_iter().map(Open::Known).collect(),
1521            ),
1522            background: None,
1523            retained_output: None,
1524            metadata: serde_json::json!({
1525                "acp": {
1526                    "kind": shown.fields.kind,
1527                    "raw_input": shown.fields.raw_input,
1528                    "raw_output": shown.fields.raw_output,
1529                    "tool_name": shown.fields.tool_name(),
1530                }
1531            }),
1532            attachments: Vec::new(),
1533            additional_messages: Vec::new(),
1534        };
1535        // A command's output is the call's output, retained as a native
1536        // command's is; a failed call keeps it too.
1537        let mut failure = None;
1538        let mut partial = false;
1539        if let Some(terminal) = terminal {
1540            failure = terminal.failure();
1541            partial = true;
1542            // A failed call's output counts as incomplete, as the session
1543            // records every failed call's (and so keeps showing it).
1544            result = terminal.finish(self.engine, result, !succeeded).await;
1545            if result.output.is_empty() && !text.is_empty() {
1546                result.output = truncate(&text, PersistedToolResult::MAX_OUTPUT_BYTES);
1547            }
1548        }
1549        let outcome = if succeeded {
1550            Ok(result)
1551        } else {
1552            Err(ToolFailure {
1553                code: ToolCallFailureCode::ExecutionFailed,
1554                message: match failure {
1555                    _ if !text.is_empty() => truncate(&text, 4096),
1556                    Some(failure) => failure,
1557                    None => "the agent's tool call failed".to_owned(),
1558                },
1559                partial_output: partial.then(|| Box::new(result)),
1560            })
1561        };
1562        self.engine
1563            .submit_tool_result_status(self.session(), self.run, tool_call_id, outcome, false)
1564            .await?;
1565        Ok(())
1566    }

1568    /// Ends every call the agent left open, as the round ends.
1569    async fn end_open_calls(&mut self, cancelled: bool) -> Result<(), EngineError> {
1570        let open = self
1571            .calls
1572            .iter()
1573            .filter(|(_, call)| !call.terminated)
1574            .map(|(id, call)| (id.clone(), call.tool_call_id))
1575            .collect::<Vec<_>>();
1576        for (acp_id, tool_call_id) in open {
1577            let mut partial_output = None;
1578            if let Some(call) = self.calls.get_mut(&acp_id) {
1579                call.terminated = true;
1580                // The output a command streamed before the round ended.
1581                if let Some(terminal) = call.terminal.take() {
1582                    let result = PersistedToolResult {
1583                        title: display_text(
1584                            call.fields.title.as_deref().unwrap_or("ACP tool call"),
1585                        ),
1586                        output: String::new(),
1587                        display: cookie_agent_protocol::fit_result_display(
1588                            terminal.blocks().into_iter().map(Open::Known).collect(),
1589                        ),
1590                        background: None,
1591                        retained_output: None,
1592                        metadata: serde_json::json!({"acp": {"kind": call.fields.kind}}),
1593                        attachments: Vec::new(),
1594                        additional_messages: Vec::new(),
1595                    };
1596                    partial_output =
1597                        Some(Box::new(terminal.finish(self.engine, result, true).await));
1598                }
1599            }
1600            self.engine
1601                .submit_tool_result_status(
1602                    self.session(),
1603                    self.run,
1604                    tool_call_id,
1605                    Err(ToolFailure {
1606                        code: ToolCallFailureCode::ExecutionFailed,
1607                        message: if cancelled {
1608                            "the turn was cancelled before the call finished".to_owned()
1609                        } else {
1610                            "the agent ended its turn without finishing the call".to_owned()
1611                        },
1612                        partial_output,
1613                    }),
1614                    cancelled,
1615                )
1616                .await?;
1617        }
1618        Ok(())
1619    }

1621    /// Runs a call of cookie's tool `name` that the agent made through the
1622    /// bridge as a call of this turn: the turn so far commits with the call
1623    /// in it, the call starts as cookie's tool calls do, and it runs through
1624    /// the rules, approvals, and execution of a model's call; `reply` gets
1625    /// its result.
1626    async fn bridge_call(
1627        &mut self,
1628        name: String,
1629        arguments: Value,
1630        reply: tokio::sync::oneshot::Sender<Value>,
1631    ) -> Result<(), EngineError> {
1632        let selection = cookie_agent_protocol::acp_model_selection(&self.live.agent.id);
1633        let turn_context = Arc::new(cookie_agent_runtime_api::TurnAgentContext {
1634            agent: self.live.agent.id.clone(),
1635            model: selection.model,
1636            adapter: cookie_agent_protocol::AdaptorId::new(crate::policy::ACP_ADAPTER)
1637                .expect("static adapter"),
1638            input_media: Vec::new(),
1639        });
1640        let tool_call_id = ToolCallId::new_v7();
1641        let call = cookie_agent_runtime_api::ToolCall {
1642            id: tool_call_id,
1643            name: name.clone(),
1644            arguments: arguments.clone(),
1645        };
1646        let prepared = self
1647            .engine
1648            .prepare_tool_call(
1649                self.session(),
1650                self.run,
1651                call,
1652                &self.active.policy,
1653                Arc::clone(&turn_context),
1654            )
1655            .await;
1656        let model_call_id = ModelCallId::new(format!("cookie:{tool_call_id}"))
1657            .map_err(|error| EngineError::Acp(error.to_string()))?;
1658        let tool_name = SafeCode::new(
1659            name.to_ascii_lowercase()
1660                .chars()
1661                .map(|character| {
1662                    if character.is_ascii_alphanumeric() || matches!(character, '.' | '_' | '-') {
1663                        character
1664                    } else {
1665                        '_'
1666                    }
1667                })
1668                .collect::<String>(),
1669        )
1670        .unwrap_or_else(|_| SafeCode::new("tool").expect("static code"));
1671        self.ensure_step().await?;
1672        let step = self.step.as_mut().expect("step started");
1673        step.parts.push(PersistedAssistantPart::ToolCall {
1674            id: model_call_id.clone(),
1675            provider_item_id: None,
1676            name: tool_name,
1677            input: arguments,
1678            raw_input: None,
1679            metadata: None,
1680        });
1681        let content_index = (step.parts.len() - 1) as u32;
1682        let model_turn_seq = self.commit(ModelFinishReason::ToolCalls).await?;
1683        let operation_fingerprint = prepared.prepared.as_ref().map_or_else(
1684            |_| {
1685                super::super::tool_execution::fallback_operation_fingerprint(
1686                    &prepared.call,
1687                    prepared.permission_name.as_deref(),
1688                )
1689            },
1690            |ready| {
1691                cookie_agent_protocol::OperationFingerprint::from_prepared_operation(
1692                    ready.operation(),
1693                )
1694            },
1695        );
1696        self.engine
1697            .append(
1698                self.session(),
1699                Some(self.run),
1700                origin(),
1701                Event::ToolCallStarted {
1702                    start: ToolCallStart {
1703                        tool_call_id,
1704                        output: ToolOutputDeclaration::default(),
1705                        owner: AssistantToolCallRef {
1706                            model_turn_seq,
1707                            content_index,
1708                            model_call_id,
1709                            provider_item_id: None,
1710                        },
1711                        presentation: prepared.presentation.clone(),
1712                        operation_fingerprint,
1713                        background: false,
1714                    },
1715                },
1716            )
1717            .await?;
1718        let engine = self.engine.clone();
1719        let active = Arc::clone(self.active);
1720        let run = self.run;
1721        self.bridged.push(tokio::spawn(async move {
1722            let result = engine
1723                .execute_tool(Arc::clone(&active), run, prepared, turn_context)
1724                .await;
1725            let answer = match &result {
1726                Ok(result) => {
1727                    let mut text = crate::model_history::tool_result_text(result);
1728                    if text.is_empty() {
1729                        text = result.title.as_str().to_owned();
1730                    }
1731                    serde_json::json!({"content": [{"type": "text", "text": text}], "isError": false})
1732                }
1733                Err(failure) => serde_json::json!({
1734                    "content": [{"type": "text", "text": failure.message}],
1735                    "isError": true,
1736                }),
1737            };
1738            let cancelled = active.cancellation.is_cancelled();
1739            let _ = engine
1740                .submit_tool_result_status(active.session, run, tool_call_id, result, cancelled)
1741                .await;
1742            let _ = reply.send(answer);
1743        }));
1744        Ok(())
1745    }

1812fn truncate(text: &str, max: usize) -> String {
1813    if text.len() <= max {
1814        return text.to_owned();
1815    }
1816    let mut end = max;
1817    while !text.is_char_boundary(end) {
1818        end -= 1;

1824    /// Runs `active`, a run of an ACP agent, to its end.
1825    pub(in crate::runtime) async fn run_acp_loop(
1826        &self,
1827        run_id: RunId,
1828        active: Arc<ActiveRun>,
1829    ) -> Result<(), EngineError> {
1830        let offers =
1831            active.policy.acp_agent().cloned().ok_or_else(|| {
1832                EngineError::Acp("the run's agent is not an ACP agent".to_owned())
1833            })?;
1834        let agent = self.place_acp_agent(active.session, &offers)?;
1835        let document = active
1836            .policy
1837            .registry
1838            .get(&agent.id)
1839            .map(|agent| agent.document.clone())
1840            .ok_or_else(|| EngineError::InvalidRuntimeAgent(agent.id.clone()))?;
1841        let (process, _use) = self
1842            .inner
1843            .acp
1844            .processes
1845            .acquire(&self.inner, &agent)
1846            .await
1847            .map_err(EngineError::Acp)?;
1848        let live = self
1849            .acp_session(active.session, Some(run_id), &agent, &document, &process)
1850            .await?;
1851        let home = process.client.home();
1852        let mut turn = Turn {
1853            engine: self,
1854            run: run_id,
1855            active: &active,
1856            live,
1857            step: None,
1858            next_attempt: 1,
1859            retries: 0,
1860            calls: HashMap::new(),
1861            used_call_ids: Default::default(),
1862            last_turn_seq: None,
1863            last_text: String::new(),
1864            input_through_seq: 0,
1865            claim: None,
1866            usage_update: None,
1867            context: None,
1868            home,
1869            bridged: Vec::new(),
1870            steering: process.init().steering(),
1871            steer_claims: Vec::new(),
1872            unsteered: false,
1873            compactions: HashMap::new(),
1874        };
1875        loop {
1876            if active.cancellation.is_cancelled() {
1877                self.append_run_cancelled_once(&active, run_id, None)?;
1878                return Ok(());
1879            }
1880            let claim = self.prompt_events(active.session, run_id).await?;
1881            let context = if turn.live.first_prompt {
1882                self.acp_first_prompt_context(
1883                    active.session,
1884                    &active.policy,
1885                    &claim.events,
1886                    &document,
1887                )?
1888            } else {
1889                Vec::new()
1890            };
1891            let prompt = self.acp_prompt(&claim.events, &mut turn.live, context, &process);
1892            turn.input_through_seq = claim.events.last().map_or(0, |event| event.seq);
1893            turn.claim = Some(claim);
1894            turn.last_text.clear();
1895            turn.last_turn_seq = None;
1896            turn.usage_update = None;
1897            let end = self.acp_round(&mut turn, prompt).await?;
1898            // A cancel accepted while the agent was finishing its turn ends
1899            // the run as cancelled, as it does a model run's, rather than
1900            // completing it or taking more input.
1901            let end = match end {
1902                RoundEnd::Completed(_) if active.cancellation.is_cancelled() => RoundEnd::Cancelled,
1903                end => end,
1904            };
1905            match end {
1906                RoundEnd::Completed(final_text) => {
1907                    // A title is a nicety: a failure leaves the session
1908                    // untitled, as a model run's does.
1909                    let _ = self
1910                        .acp_fallback_title(&active, run_id, turn.input_through_seq)
1911                        .await;
1912                    // Once the agent has replied in a session, its process
1913                    // can be asked whether its fork takes a fork point.
1914                    let engine = self.clone();
1915                    let live = turn.live.clone();
1916                    tokio::spawn(async move { engine.probe_fork_point(&live).await });
1917                    // Input the agent did not take into its turn goes with
1918                    // the next round.
1919                    let unsteered = std::mem::take(&mut turn.unsteered);
1920                    let promoted = self
1921                        .request(active.session, |reply| {
1922                            SessionCommand::PromotePendingOrComplete {
1923                                run: run_id,
1924                                final_text,
1925                                complete_if_empty: !unsteered,
1926                                reply,
1927                            }
1928                        })
1929                        .await?;
1930                    if !promoted && !unsteered {
1931                        return Ok(());
1932                    }
1933                }
1934                RoundEnd::Cancelled => {
1935                    self.append_run_cancelled_once(&active, run_id, None)?;
1936                    return Ok(());
1937                }
1938                RoundEnd::Failed(reason)
1939                | RoundEnd::Stopped {
1940                    message: reason, ..
1941                } => return Err(EngineError::Acp(reason)),
1942            }
1943        }

1946    /// One prompt round: sends `prompt` and records what the agent reports
1947    /// until its answer.
1948    async fn acp_round(
1949        &self,
1950        turn: &mut Turn<'_>,
1951        prompt: Vec<Value>,
1952    ) -> Result<RoundEnd, EngineError> {
1953        let events = Arc::clone(&turn.live.events);
1954        let mut events = events.lock().await;
1955        // What arrived between rounds, such as the commands an agent
1956        // announces after `session/new`; turn output from before the
1957        // prompt (a load's replay) is not this round's.
1958        while let Ok(event) = events.try_recv() {
1959            match event {
1960                SessionEvent::Update(update) => {
1961                    let kind = update
1962                        .get("sessionUpdate")
1963                        .and_then(Value::as_str)
1964                        .unwrap_or_default()
1965                        .to_owned();
1966                    turn.metadata(&kind, &update).await?;
1967                }
1968                SessionEvent::Permission(_, request) => {
1969                    let _ = request.respond(&wire::permission_outcome(None)).await;
1970                }
1971                SessionEvent::BridgeCall { reply, .. } => {
1972                    let _ = reply.send(bridge_refusal());
1973                }
1974                SessionEvent::Elicitation(_, request) => elicitation::cancel(request),
1975            }
1976        }
1977        let process = Arc::clone(&turn.live.process);
1978        let session_id = turn.live.acp_session_id.clone();
1979        let prompt_request = wire::PromptRequest {
1980            session_id: &session_id,
1981            prompt: &prompt,
1982        };
1983        let request = process
1984            .peer
1985            .request::<_, wire::PromptResponse>(wire::method::SESSION_PROMPT, &prompt_request);
1986        tokio::pin!(request);
1987        let mut cancel_deadline: Option<tokio::time::Instant> = None;
1988        // Input to steer wakes the round while the agent can take it.
1989        let steer_wake = turn
1990            .steering
1991            .then(|| self.inner.acp.processes.watch_steering(turn.session()));
1992        let mut pending_steer: Option<PendingSteer> = None;
1993        let answer = loop {
1994            // Biased, cancellation first: an agent that streams without a
1995            // pause keeps the update branch ready, and a cancel accepted
1996            // meanwhile must still reach it at the next turn of the loop.
1997            tokio::select! {
1998                biased;
1999                () = turn.active.cancellation.cancelled(), if cancel_deadline.is_none() => {
2000                    cancel_deadline = Some(tokio::time::Instant::now() + CANCEL_GRACE);
2001                    let _ = process.peer.notify(
2002                        wire::method::SESSION_CANCEL,
2003                        &wire::SessionRef { session_id: &session_id },
2004                    ).await;
2005                }
2006                () = async {
2007                    match cancel_deadline {
2008                        Some(deadline) => tokio::time::sleep_until(deadline).await,
2009                        None => std::future::pending().await,
2010                    }
2011                } => break None,
2012                () = process.exited.cancelled() => break None,
2013                answer = &mut request => break Some(answer),
2014                answer = async {
2015                    match pending_steer.as_mut() {
2016                        Some(pending) => (&mut pending.answer).await,
2017                        None => std::future::pending().await,
2018                    }
2019                }, if pending_steer.is_some() => {
2020                    let pending = pending_steer.take().expect("a steer is pending");
2021                    let answer = answer.unwrap_or_else(|_| {
2022                        Err("the agent's process ended".to_owned())
2023                    });
2024                    turn.end_steer(pending, answer).await?;
2025                }
2026                event = events.recv() => match event {
2027                    Some(SessionEvent::Update(update)) => turn.update(update).await?,
2028                    Some(SessionEvent::Permission(request, incoming)) => {
2029                        if cancel_deadline.is_some() {
2030                            let _ = incoming.respond(&wire::permission_outcome(None)).await;
2031                        } else {
2032                            turn.permission(*request, incoming);
2033                        }
2034                    }
2035                    Some(SessionEvent::BridgeCall { name, arguments, reply }) => {
2036                        turn.bridge_call(name, arguments, reply).await?;
2037                    }
2038                    Some(SessionEvent::Elicitation(create, incoming)) => {
2039                        if cancel_deadline.is_some() {
2040                            elicitation::cancel(incoming);
2041                        } else {
2042                            turn.elicitation(*create, incoming);
2043                        }
2044                    }
2045                    None => break None,
2046                },
2047                () = async {
2048                    match &steer_wake {
2049                        Some(wake) => wake.notified().await,
2050                        None => std::future::pending().await,
2051                    }
2052                }, if cancel_deadline.is_none() && pending_steer.is_none() => {
2053                    pending_steer = turn.begin_steer().await?;
2054                }
2055            }
2056        };
2057        // Updates the agent sent before its answer may still be on their way.
2058        process.flush().await;
2059        while let Ok(event) = events.try_recv() {
2060            match event {
2061                SessionEvent::Update(update) => turn.update(update).await?,
2062                SessionEvent::Permission(_, incoming) => {
2063                    let _ = incoming.respond(&wire::permission_outcome(None)).await;
2064                }
2065                SessionEvent::BridgeCall { reply, .. } => {
2066                    let _ = reply.send(bridge_refusal());
2067                }
2068                SessionEvent::Elicitation(_, incoming) => elicitation::cancel(incoming),
2069            }
2070        }
2071        drop(events);
2072        if steer_wake.is_some() {
2073            self.inner.acp.processes.unwatch_steering(turn.session());
2074        }
2075        if let Some(mut pending) = pending_steer.take() {
2076            let answer = tokio::time::timeout(CANCEL_GRACE * 5, &mut pending.answer)
2077                .await
2078                .ok()
2079                .and_then(Result::ok)
2080                .unwrap_or_else(|| Err("the agent did not answer the steering request".to_owned()));
2081            turn.end_steer(pending, answer).await?;
2082        }
2083        // A call through the bridge, or a question to the user, ends before
2084        // the turn does.
2085        self.end_acp_questions(turn.run);
2086        for call in std::mem::take(&mut turn.bridged) {
2087            let _ = call.await;
2088        }
2089        let cancelled = turn.active.cancellation.is_cancelled();
2090        let answered_usage = match &answer {
2091            Some(Ok(response)) => response.usage.clone(),
2092            _ => None,
2093        };
2094        let end = match answer {
2095            Some(Ok(response)) => match response.stop_reason.as_str() {
2096                "end_turn" => {
2097                    if turn.step.is_some() || turn.last_turn_seq.is_none() {
2098                        turn.commit(ModelFinishReason::Stop).await?;
2099                    }
2100                    let usage = usage_of(response.usage.as_ref(), turn.usage_update.as_ref());
2101                    self.record_acp_usage(turn, usage).await?;
2102                    RoundEnd::Completed(
2103                        (!turn.last_text.is_empty()).then(|| turn.last_text.clone()),
2104                    )
2105                }
2106                "cancelled" => RoundEnd::Cancelled,
2107                other => stopped(other),
2108            },
2109            Some(Err(error)) if cancelled => {
2110                let _ = error;
2111                RoundEnd::Cancelled
2112            }
2113            Some(Err(error)) => RoundEnd::Failed(match process.exit_reason() {
2114                Some(reason) => reason,
2115                None => agent_error_message(&error),
2116            }),
2117            None if cancelled => RoundEnd::Cancelled,
2118            None => RoundEnd::Failed(
2119                process
2120                    .exit_reason()
2121                    .unwrap_or_else(|| "the agent's process ended".to_owned()),
2122            ),
2123        };
2124        if !matches!(end, RoundEnd::Completed(_)) {
2125            if let Some(step) = turn.step.as_ref()
2126                && !step.parts.is_empty()
2127            {
2128                let finish = match &end {
2129                    RoundEnd::Cancelled => ModelFinishReason::Cancelled,
2130                    RoundEnd::Stopped { finish, .. } => finish.clone(),
2131                    _ => ModelFinishReason::Error,
2132                };
2133                turn.commit(finish).await?;
2134            } else if let Some(step) = turn.step.take() {
2135                turn.retries += 1;
2136                self.append(
2137                    turn.session(),
2138                    Some(turn.run),
2139                    origin(),
2140                    Event::AttemptAbandoned {
2141                        attempt_id: step.attempt_id,
2142                        model_error: None,
2143                    },
2144                )
2145                .await?;
2146            }
2147            // What the agent used before it stopped counts too.
2148            let usage = usage_of(answered_usage.as_ref(), turn.usage_update.as_ref());
2149            self.record_acp_usage(turn, usage).await?;
2150        }
2151        turn.end_open_calls(matches!(end, RoundEnd::Cancelled))
2152            .await?;
2153        if !turn.live.is_live() {
2154            self.forget_live(turn.session(), &turn.live);
2155        }
2156        Ok(end)
2157    }

2264    /// (user input applied, producer messages admitted), and which input
2265    /// that was.
2266    fn input_since(
2267        &self,
2268        events: &crate::events::EventSnapshot,
2269        boundary: u64,
2270        caps: &wire::PromptCapabilities,
2271    ) -> (Vec<Value>, cookie_agent_protocol::AcpSteered) {
2272        let producers = crate::goal_projection::GoalProducerProjection::from_events(events);
2273        let admitted = producers
2274            .messages
2275            .iter()
2276            .filter(|message| !message.discarded)
2277            .filter_map(|message| {
2278                message
2279                    .admission
2280                    .map(|(_, seq)| (seq, (message.message_id, &message.body)))
2281            })
2282            .collect::<HashMap<_, _>>();
2283        let mut submitted = HashMap::new();

2287            match &event.payload {
2288                Event::UserInputSubmitted { content } => {
2289                    submitted.insert(event.seq, content);
2290                }
2291                Event::UserInputApplied { user_input_seq } if event.seq > boundary => {
2292                    if let Some(content) = submitted.get(user_input_seq) {
2293                        self.push_content(&mut prompt, event.session_id, content, caps);
2294                        taken.user_inputs.push(*user_input_seq);
2295                    }
2296                }
2297                Event::ProducerMessageAdmitted { .. } if event.seq > boundary => {
2298                    if let Some((message_id, body)) = admitted.get(&event.seq) {
2299                        self.push_content(&mut prompt, event.session_id, body, caps);
2300                        taken.producer_messages.push(*message_id);
2301                    }
2302                }
2303                _ => {}
2304            }
2305        }
2306        (prompt, taken)
2307    }

2331    /// The ACP session `session` talks to `agent` in: the live one, else the
2332    /// one its log records resumed or loaded, else, when a fork or a revert
2333    /// left that one behind, one branched off it
2334    /// ([`super::lifecycle::Plan`]), else a new one. Starting, resuming,
2335    /// loading, or forking one records it.
2336    pub(crate) async fn acp_session(
2337        &self,
2338        session: SessionId,
2339        run: Option<RunId>,
2340        agent: &RuntimeAgent,
2341        document: &cookie_agent_config::AgentDocument,
2342        process: &Arc<AgentProcess>,
2343    ) -> Result<LiveSession, EngineError> {
2344        let plan = super::lifecycle::plan(
2345            &self.inner.store.get(session)?.log.event_snapshot(),
2346            &agent.id,
2347        );
2348        // The live session serves while the log still makes it the
2349        // session's: a fork or a revert since detached it.
2350        if let Some(live) = self.live_session(session)
2351            && Arc::ptr_eq(&live.process, process)
2352            && live.is_live()
2353            && live.agent.id == agent.id
2354            && matches!(&plan, super::lifecycle::Plan::Resume(record)
2355                if record.acp_session_id == live.acp_session_id)
2356        {
2357            return Ok(live);
2358        }
2359        self.remember_acp_input(
2360            agent,
2361            &process.init().agent_capabilities.prompt_capabilities,
2362        );
2363        let cwd = self.acp_cwd(session, agent, process)?;
2364        let additional_directories = if process
2365            .init()
2366            .agent_capabilities
2367            .session("additionalDirectories")
2368        {
2369            self.acp_additional_directories(session, agent, &cwd)?
2370        } else {
2371            Vec::new()
2372        };
2373        let bridge = self.open_acp_bridge(session, agent, process).await;
2374        // Bound before the agent hears of it: an agent connects its MCP
2375        // servers while it answers `session/new`.
2376        if let Ok(opened) = &bridge {
2377            super::mcp::bind(
2378                &self.inner,
2379                agent,
2380                &process.key.runtime,
2381                &opened.bridge,
2382                session,
2383            );
2384        }
2385        let mcp_servers = match &bridge {
2386            Ok(opened) => {
2387                let (command, args) = opened
2388                    .command
2389                    .split_first()
2390                    .map(|(command, args)| (command.clone(), args.to_vec()))
2391                    .unwrap_or_default();
2392                vec![wire::McpServer {
2393                    name: super::mcp::BRIDGE_SERVER.to_owned(),
2394                    command,
2395                    args,
2396                    env: Vec::new(),
2397                }]
2398            }
2399            Err(reason) => {
2400                self.acp_notice(
2401                    session,
2402                    run,
2403                    format!("cookie's tools are not offered to the agent: {reason}"),
2404                )
2405                .await;
2406                Vec::new()
2407            }
2408        };
2409        let (mut prior, mut branch) = match plan {
2410            super::lifecycle::Plan::New => (None, None),
2411            super::lifecycle::Plan::Resume(record) => (Some(record), None),
2412            super::lifecycle::Plan::Branch {
2413                from,
2414                message_id,
2415                delete,
2416            } => (None, Some((from, message_id, delete))),
2417        };

2419        // recorded on another runtime cannot be resumed or forked here.
2420        if let Some(elsewhere) = prior
2421            .as_ref()
2422            .or(branch.as_ref().map(|(from, _, _)| from))
2423            .filter(|record| record.runtime_id != agent.runtime)
2424            .cloned()
2425        {
2426            self.acp_notice(
2427                session,
2428                run,
2429                format!(
2430                    "The agent's earlier session is on runtime `{}`, which does not run it for \
2431                     this session now; this turn starts a new session on runtime `{}`, which does \
2432                     not remember the conversation so far.",
2433                    elsewhere.runtime_id, agent.runtime
2434                ),
2435            )
2436            .await;
2437            prior = None;
2438            branch = None;
2439        }
2440        let caps = &process.init().agent_capabilities;
2441        let mut opened = None;
2442        let mut lost = false;
2443        let mut forked_from = None;
2444        if let Some((from, message_id, delete)) = &branch {
2445            let forked = self
2446                .branch_acp_session(
2447                    session,
2448                    process,
2449                    from,
2450                    message_id.as_deref(),
2451                    *delete,
2452                    &cwd,
2453                    &additional_directories,
2454                    &mcp_servers,
2455                )
2456                .await?;
2457            if let Some((id, response)) = forked {
2458                let receiver = process.route(&id);
2459                forked_from = Some(from.acp_session_id.clone());
2460                opened = Some((id, response, AcpSessionStart::Forked, receiver));
2461            }
2462        }
2463        if let Some(prior) = &prior {
2464            let receiver = process.route(&prior.acp_session_id);
2465            let request = wire::ExistingSessionRequest {
2466                session_id: &prior.acp_session_id,
2467                cwd: &cwd,
2468                additional_directories: &additional_directories,
2469                mcp_servers: &mcp_servers,
2470            };
2471            let attempt = if caps.session("resume") {
2472                Some((wire::method::SESSION_RESUME, AcpSessionStart::Resumed))
2473            } else if caps.load_session {
2474                Some((wire::method::SESSION_LOAD, AcpSessionStart::Loaded))
2475            } else {
2476                None
2477            };
2478            match attempt {
2479                Some((method, how)) => {
2480                    match tokio::time::timeout(
2481                        SESSION_TIMEOUT,
2482                        process
2483                            .peer
2484                            .request::<_, wire::SessionResponse>(method, &request),
2485                    )
2486                    .await
2487                    {
2488                        Ok(Ok(response)) => {
2489                            opened = Some((prior.acp_session_id.clone(), response, how, receiver));
2490                        }
2491                        _ => {
2492                            process.unroute(&prior.acp_session_id);
2493                            lost = true;
2494                        }
2495                    }
2496                }
2497                None => {
2498                    process.unroute(&prior.acp_session_id);
2499                    lost = true;
2500                }
2501            }
2502        }
2503        let (acp_session_id, response, how, receiver) = match opened {
2504            Some(opened) => opened,
2505            None => {
2506                let response = tokio::time::timeout(
2507                    SESSION_TIMEOUT,
2508                    process.peer.request::<_, wire::SessionResponse>(
2509                        wire::method::SESSION_NEW,
2510                        &wire::NewSessionRequest {
2511                            cwd: &cwd,
2512                            additional_directories: &additional_directories,
2513                            mcp_servers: &mcp_servers,
2514                        },
2515                    ),
2516                )
2517                .await
2518                .map_err(|_| {
2519                    EngineError::Acp(format!(
2520                        "the agent did not answer `session/new` within {} s",
2521                        SESSION_TIMEOUT.as_secs()
2522                    ))
2523                })?
2524                .map_err(|error| {
2525                    EngineError::Acp(
2526                        process
2527                            .exit_reason()
2528                            .unwrap_or_else(|| agent_error_message(&error)),
2529                    )
2530                })?;
2531                let id = response
2532                    .session_id
2533                    .clone()
2534                    .filter(|id| !id.is_empty())
2535                    .ok_or_else(|| {
2536                        EngineError::Acp(
2537                            "the agent answered `session/new` without a session ID".to_owned(),
2538                        )
2539                    })?;

2623    /// except the model and effort options when the document names
2624    /// `models`; then the first `models` entry the agent offers, and its

2627        &self,
2628        session: SessionId,
2629        run: Option<RunId>,

2631        document: &cookie_agent_config::AgentDocument,
2632    ) {

2634        let mode_option = super::options::mode_option(&live.options).cloned();
2635        if let Some(option) = mode_option {

2637            if let Some(wanted) = wanted
2638                && wanted != option.current_value
2639                && let Err(reason) = self.set_live_option(live, &option.id, &wanted).await

2735    /// Sets option `id` of `live` to `value` with the agent.
2736    pub(crate) async fn set_live_option(
2737        &self,
2738        live: &mut LiveSession,
2739        id: &str,
2740        value: &str,
2741    ) -> Result<(), String> {
2742        if live.modes_as_option && id == "mode" {
2743            live.process
2744                .peer
2745                .request::<_, Value>(
2746                    wire::method::SESSION_SET_MODE,
2747                    &wire::SetModeRequest {
2748                        session_id: &live.acp_session_id,
2749                        mode_id: value,
2750                    },
2751                )
2752                .await
2753                .map_err(|error| agent_error_message(&error))?;
2754            for option in &mut live.options {
2755                if option.id == "mode" {
2756                    value.clone_into(&mut option.current_value);
2757                }
2758            }
2759        } else {
2760            // A boolean option takes a boolean (`type: boolean`); a select
2761            // the value's ID, as ACP v1 sends it.
2762            let boolean = live
2763                .options
2764                .iter()
2765                .any(|option| option.id == id && option.kind == AcpConfigOptionKind::Boolean);
2766            let request = if boolean {
2767                wire::SetConfigOptionRequest {
2768                    session_id: &live.acp_session_id,
2769                    config_id: id,
2770                    value_type: Some("boolean"),
2771                    value: Value::Bool(value == "true"),
2772                }
2773            } else {
2774                wire::SetConfigOptionRequest {
2775                    session_id: &live.acp_session_id,
2776                    config_id: id,
2777                    value_type: None,
2778                    value: Value::from(value),
2779                }
2780            };
2781            let response = live
2782                .process
2783                .peer
2784                .request::<_, wire::SetConfigOptionResponse>(
2785                    wire::method::SESSION_SET_CONFIG_OPTION,
2786                    &request,
2787                )
2788                .await
2789                .map_err(|error| agent_error_message(&error))?;
2790            match response.config_options {
2791                Some(options) => {
2792                    let (mut options, _) = options_of(Some(&options), None);
2793                    if live.modes_as_option
2794                        && let Some(mode) = live.options.iter().find(|option| option.id == "mode")
2795                    {
2796                        options.push(mode.clone());
2797                    }
2798                    live.options = options;
2799                }
2800                None => {
2801                    for option in &mut live.options {
2802                        if option.id == id {
2803                            value.clone_into(&mut option.current_value);
2804                        }
2805                    }
2806                }
2807            }
2808        }
2809        self.remember_live(live);
2810        Ok(())
2811    }
