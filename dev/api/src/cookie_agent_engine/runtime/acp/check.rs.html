1//! `acp.check` (`docs/site/specs/acp-onboarding.md` section 5): starts a
2//! declared ACP agent on a runtime, opens a session in the runtime's home,
3//! remembers what the agent offered, and closes the session again, saying
4//! how far it got.
5
6use std::time::Duration;
7
8use cookie_agent_protocol::{AcpCheckResult, AcpCheckStage, AgentId, RuntimeId};
9
10use super::{lifecycle::end_agent_session, wire};
11use crate::Engine;
12
13/// How long the agent gets to answer `session/new`; a first start through
14/// `npx` downloads the adapter before `initialize`, which the process start
15/// waits for on its own.
16const SESSION_TIMEOUT: Duration = Duration::from_secs(120);
17
18fn failed(stage: AcpCheckStage, error: String) -> AcpCheckResult {
19    AcpCheckResult {
20        ok: false,
21        stage,
22        error: Some(error),
23        agent_name: None,
24        agent_version: None,
25        options: Vec::new(),
26    }
27}
28
29impl Engine {
30    /// Checks the ACP agent `agent` declared on runtime `runtime`.
31    pub async fn check_acp_agent(&self, runtime: &RuntimeId, agent: &AgentId) -> AcpCheckResult {
32        let Some(declared) = self
33            .inner
34            .acp
35            .runtime_agents()
36            .into_iter()
37            .find(|candidate| candidate.runtime == *runtime && candidate.agent == *agent)
38        else {
39            return failed(
40                AcpCheckStage::Declared,
41                format!("runtime `{runtime}` does not declare ACP agent `{agent}`"),
42            );
43        };
44        if !declared.enabled {
45            return failed(
46                AcpCheckStage::Declared,
47                format!("ACP agent `{agent}` is turned off on runtime `{runtime}`"),
48            );
49        }
50        if !declared.connected {
51            return failed(
52                AcpCheckStage::Declared,
53                format!("runtime `{runtime}` is not connected"),
54            );
55        }
56        let (process, _use) = match self
57            .inner
58            .acp
59            .processes
60            .acquire(&self.inner, &declared)
61            .await
62        {
63            Ok(acquired) => acquired,
64            Err(error) => return failed(AcpCheckStage::Start, error),
65        };
66        let info = &process.init().info;
67        let text = |key: &str| {
68            info.get(key)
69                .and_then(serde_json::Value::as_str)
70                .filter(|text| !text.is_empty())
71                .map(str::to_owned)
72        };
73        let agent_name = text("title").or_else(|| text("name"));
74        let agent_version = text("version");
75        let cwd = process.client.home().unwrap_or_else(|| "/".to_owned());
76        let response = tokio::time::timeout(
77            SESSION_TIMEOUT,
78            process.peer.request::<_, wire::SessionResponse>(
79                wire::method::SESSION_NEW,
80                &wire::NewSessionRequest {
81                    cwd: &cwd,
82                    additional_directories: &[],
83                    mcp_servers: &[],
84                    system_prompt: None,
85                    meta: wire::session_meta(),
86                },
87            ),
88        )
89        .await;
90        let response = match response {
91            Ok(Ok(response)) => response,
92            Ok(Err(error)) => {
93                let reason = process
94                    .exit_reason()
95                    .unwrap_or_else(|| super::process::agent_error_message(&error));
96                return AcpCheckResult {
97                    agent_name,
98                    agent_version,
99                    ..failed(
100                        AcpCheckStage::Session,
101                        format!(
102                            "the agent did not open a session: {reason}; if it needs a login, \
103                             log in to it on that machine"
104                        ),
105                    )
106                };
107            }
108            Err(_) => {
109                return AcpCheckResult {
110                    agent_name,
111                    agent_version,
112                    ..failed(
113                        AcpCheckStage::Session,
114                        format!(
115                            "the agent did not answer `session/new` within {} s",
116                            SESSION_TIMEOUT.as_secs()
117                        ),
118                    )
119                };
120            }
121        };
122        let options = super::runner::options_of(response.config_options.as_ref());
123        self.remember_acp_options(&declared, &options, true);
124        if let Some(session) = response.session_id.as_deref().filter(|id| !id.is_empty()) {
125            end_agent_session(&process, session, true, true).await;
126        }
127        AcpCheckResult {
128            ok: true,
129            stage: AcpCheckStage::Ready,
130            error: None,
131            agent_name,
132            agent_version,
133            options,
134        }
135    }
136}
