1//! The ACP agent's commands still running after their tool calls ended,
2//! such as Claude Code's commands run in the background
3//! (`docs/site/specs/acp-v2.md` section 9).
4//!
5//! A command's ACP terminal stays open until it exits (`exitStatus`). While
6//! it does after its call ended, the command is background work of the
7//! session: listed beside cookie's background calls (`agent_task`), kept
8//! from eviction, waited for by an idle daemon restart, and keeping the
9//! agent's process from idling out. It ends when the terminal exits, or with
10//! the agent's process.
11
12use cookie_agent_protocol::{
13    AcpBackgroundEnded, BackgroundActor, BackgroundCall, SessionId, ToolCallId,
14};
15
16use super::super::Engine;
17use super::process::ProcessUse;
18use super::runner::LiveSession;
19use super::terminal::Exit;
20
21/// One command of the agent's, running after its call ended.
22pub(crate) struct AgentCommand {
23    session: SessionId,
24    /// The runtime the agent runs on.
25    runtime: cookie_agent_protocol::RuntimeId,
26    /// Its terminal's ID, the job clients name it by.
27    terminal: String,
28    tool_call_id: ToolCallId,
29    tool: String,
30    display: Option<String>,
31    since: jiff::Timestamp,
32    /// Keeps the agent's process from idling out meanwhile.
33    _use: ProcessUse,
34}
35
36impl Engine {
37    fn acp_commands(&self) -> std::sync::MutexGuard<'_, Vec<AgentCommand>> {
38        self.inner
39            .acp
40            .background
41            .lock()
42            .unwrap_or_else(|poisoned| poisoned.into_inner())
43    }
44
45    /// The agent's call `tool_call_id` (tool `tool`, shown as `display`)
46    /// ended while its terminal `terminal` still runs.
47    pub(super) fn acp_command_backgrounded(
48        &self,
49        live: &LiveSession,
50        terminal: String,
51        tool_call_id: ToolCallId,
52        tool: String,
53        display: Option<String>,
54    ) {
55        if live.process.exited.is_cancelled() {
56            return;
57        }
58        {
59            let mut commands = self.acp_commands();
60            if commands
61                .iter()
62                .any(|command| command.session == live.session && command.terminal == terminal)
63            {
64                return;
65            }
66            commands.push(AgentCommand {
67                session: live.session,
68                runtime: live.agent.runtime.clone(),
69                terminal,
70                tool_call_id,
71                tool,
72                display,
73                since: jiff::Timestamp::now(),
74                _use: ProcessUse::hold(&live.process, &self.inner),
75            });
76        }
77        self.notify_background_changed(live.session);
78        self.refresh_live_waiting(live.session);
79    }
80
81    /// Terminal `terminal` of `session` exited, as `exit` says: the
82    /// command that ran in the background on it, when one did, for the
83    /// session to record that it ended.
84    pub(super) fn acp_terminal_exited(
85        &self,
86        session: SessionId,
87        terminal: &str,
88        exit: Exit,
89    ) -> Option<AcpBackgroundEnded> {
90        let AgentCommand {
91            tool_call_id,
92            tool,
93            display,
94            since,
95            _use: held,
96            ..
97        } = {
98            let mut commands = self.acp_commands();
99            let index = commands
100                .iter()
101                .position(|command| command.session == session && command.terminal == terminal)?;
102            commands.remove(index)
103        };
104        // The process may idle once nothing else holds it.
105        drop(held);
106        self.notify_background_changed(session);
107        self.refresh_live_waiting(session);
108        Some(AcpBackgroundEnded {
109            tool_call_id,
110            tool,
111            display,
112            started_at: since,
113            exit_code: exit.code,
114            signal: exit.signal,
115        })
116    }
117
118    /// The agent's process that ran `session`'s commands is gone, and they
119    /// with it, and the work of `session` when it is a subagent of the
120    /// agent's.
121    pub(super) fn acp_commands_gone(&self, session: SessionId) {
122        self.set_acp_subagent_working(session, false);
123        let removed = {
124            let mut commands = self.acp_commands();
125            let before = commands.len();
126            commands.retain(|command| command.session != session);
127            commands.len() != before
128        };
129        if removed {
130            self.notify_background_changed(session);
131            self.refresh_live_waiting(session);
132        }
133    }
134
135    /// `session`'s agent commands, as background calls.
136    pub(crate) fn acp_background_calls(&self, session: SessionId) -> Vec<BackgroundCall> {
137        self.acp_commands()
138            .iter()
139            .filter(|command| command.session == session)
140            .map(|command| BackgroundCall {
141                job: command.terminal.clone(),
142                tool: command.tool.clone(),
143                tool_call_id: command.tool_call_id,
144                display: command.display.clone(),
145                started_at: command.since,
146                detached_at: command.since,
147                by: BackgroundActor::Model,
148                agent_task: true,
149            })
150            .collect()
151    }
152
153    /// Whether `job` is a running agent command of `session`.
154    pub(crate) fn acp_terminal_running(&self, session: SessionId, job: &str) -> bool {
155        self.acp_commands()
156            .iter()
157            .any(|command| command.session == session && command.terminal == job)
158    }
159
160    /// Whether `session` has an agent command running.
161    pub(crate) fn acp_background_running(&self, session: SessionId) -> bool {
162        self.acp_commands()
163            .iter()
164            .any(|command| command.session == session)
165    }
166
167    /// How many agent commands run, in every session.
168    pub(crate) fn acp_background_count(&self) -> usize {
169        self.acp_commands().len()
170    }
171
172    /// The runtime of each agent command that runs, in every session.
173    pub(crate) fn acp_background_runtimes(&self) -> Vec<String> {
174        self.acp_commands()
175            .iter()
176            .map(|command| command.runtime.as_str().to_owned())
177            .collect()
178    }
