1//! The ACP agent's commands still running after their tool calls ended,
2//! such as Claude Code's commands run in the background
3//! (`docs/site/specs/acp-v2.md` section 9).
4//!
5//! A command's ACP terminal stays open until it exits (`exitStatus`). While
6//! it does after its call ended, the command is background work of the
7//! session: listed beside cookie's background calls (`agent_task`), kept
8//! from eviction, waited for by an idle daemon restart, and keeping the
9//! agent's process from idling out. It ends when the terminal exits, or with
10//! the agent's process.
11//!
12//! The agent is the only one that can say a command ended. One it will not
13//! report the end of (it lost track of it) leaves when the user forgets it
14//! ([`Engine::forget_agent_command`]), or when the agent's session it ran
15//! in is replaced by another ([`Engine::acp_session_replaced`]); either
16//! records the end as dropped, which does not settle.
17//!
18//! An agent can take a while to follow up on a command that ended, so an
19//! exited command settles before it counts as gone: it leaves the list at
20//! once, but until the agent's next round, its process's end, or
21//! [`ACP_BACKGROUND_SETTLE`] after the exit, it still holds the session, an
22//! idle restart, and the process as it did while it ran.
23

37/// One command of the agent's, running after its call ended.
38pub(crate) struct AgentCommand {
39    session: SessionId,
40    /// The runtime the agent runs on.
41    runtime: cookie_agent_protocol::RuntimeId,
42    /// The agent's session that runs it.
43    acp_session: String,
44    /// Its terminal's ID, the job clients name it by.
45    terminal: String,
46    tool_call_id: ToolCallId,
47    tool: String,
48    display: Option<String>,
49    since: jiff::Timestamp,
50    /// Once its terminal exited: when it stops waiting for the agent to
51    /// follow up.
52    settles_at: Option<Instant>,
53    /// Keeps the agent's process from idling out meanwhile.
54    _use: ProcessUse,
55}

57impl Engine {
58    fn acp_commands(&self) -> std::sync::MutexGuard<'_, Vec<AgentCommand>> {
59        self.inner
60            .acp
61            .background
62            .lock()
63            .unwrap_or_else(|poisoned| poisoned.into_inner())
64    }

66    /// The agent's call `tool_call_id` (tool `tool`, shown as `display`)
67    /// ended while its terminal `terminal` still runs.
68    pub(super) fn acp_command_backgrounded(
69        &self,
70        live: &LiveSession,
71        terminal: String,
72        tool_call_id: ToolCallId,
73        tool: String,
74        display: Option<String>,
75    ) {
76        if live.process.exited.is_cancelled() {
77            return;
78        }
79        {
80            let mut commands = self.acp_commands();
81            if commands
82                .iter()
83                .any(|command| command.session == live.session && command.terminal == terminal)
84            {
85                return;
86            }
87            commands.push(AgentCommand {
88                session: live.session,
89                runtime: live.agent.runtime.clone(),
90                acp_session: live.acp_session_id.clone(),
91                terminal,
92                tool_call_id,
93                tool,
94                display,
95                since: jiff::Timestamp::now(),
96                settles_at: None,
97                _use: ProcessUse::hold(&live.process, &self.inner),
98            });
99        }
100        self.notify_background_changed(live.session);
101        self.refresh_live_waiting(live.session);
102    }

104    /// Terminal `terminal` of `session` exited, as `exit` says: the
105    /// command that ran in the background on it, when one did, for the
106    /// session to record that it ended. The command settles from here.
107    pub(super) fn acp_terminal_exited(
108        &self,
109        session: SessionId,
110        terminal: &str,
111        exit: Exit,
112    ) -> Option<AcpBackgroundEnded> {
113        let settles_at = Instant::now() + self.acp_settle_window();
114        let ended = {
115            let mut commands = self.acp_commands();
116            let command = commands.iter_mut().find(|command| {
117                command.session == session
118                    && command.terminal == terminal
119                    && command.settles_at.is_none()
120            })?;
121            command.settles_at = Some(settles_at);
122            AcpBackgroundEnded {
123                tool_call_id: command.tool_call_id,
124                tool: command.tool.clone(),
125                display: command.display.clone(),
126                started_at: command.since,
127                exit_code: exit.code,
128                signal: exit.signal,
129                dropped: None,
130            }
131        };
132        let inner = std::sync::Arc::downgrade(&self.inner);
133        tokio::spawn(async move {
134            tokio::time::sleep_until(settles_at).await;
135            if let Some(inner) = inner.upgrade() {
136                Engine { inner }.acp_commands_settle(session, |at| at <= Instant::now());
137            }
138        });
139        self.notify_background_changed(session);
140        self.refresh_live_waiting(session);
141        Some(ended)
142    }

144    /// The user forgot `session`'s agent command `job`, one the agent will
145    /// not report the end of: it leaves the list, stops holding the session
146    /// and the agent's process, and the session records it as forgotten.
147    /// Nothing is ended: cookie runs nothing of the agent's.
148    pub async fn forget_agent_command(
149        &self,
150        session: SessionId,
151        job: &str,
152    ) -> Result<(), EngineError> {
153        let forgotten =
154            self.drop_acp_commands(session, AcpBackgroundDropped::Forgotten, |command| {
155                command.terminal == job && command.settles_at.is_none()
156            });
157        if forgotten.is_empty() {
158            return Err(EngineError::Runtime(
159                if self
160                    .background_calls(session)
161                    .iter()
162                    .any(|call| call.job == job)
163                {
164                    format!("`{job}` is cookie's own background call: stop it instead")
165                } else {
166                    format!("no running agent command `{job}` in this session")
167                },
168            ));
169        }
170        self.record_dropped(session, forgotten).await;
171        Ok(())
172    }

174    /// `session`'s agent now runs in its session `acp_session`: the commands
175    /// of the one it replaced, which reports nothing more of them, leave the
176    /// list, the running ones recorded as dropped.
177    pub(super) async fn acp_session_replaced(&self, session: SessionId, acp_session: &str) {
178        let dropped =
179            self.drop_acp_commands(session, AcpBackgroundDropped::SessionReplaced, |command| {
180                command.acp_session != acp_session
181            });
182        self.record_dropped(session, dropped).await;
183    }

185    /// Takes `session`'s commands that `which` picks off the list: the
186    /// running ones' ends, dropped for `why`. Settling ones go silently, as
187    /// their ends are on record.
188    fn drop_acp_commands(
189        &self,
190        session: SessionId,
191        why: AcpBackgroundDropped,
192        which: impl Fn(&AgentCommand) -> bool,
193    ) -> Vec<AcpBackgroundEnded> {
194        // The process may idle once nothing else holds it, which takes its
195        // own locks: these go after the list's.
196        let taken = {
197            let mut commands = self.acp_commands();
198            let (taken, kept) = std::mem::take(&mut *commands)
199                .into_iter()
200                .partition::<Vec<_>, _>(|command| command.session == session && which(command));
201            *commands = kept;
202            taken
203        };
204        if taken.is_empty() {
205            return Vec::new();
206        }
207        let ended = taken
208            .iter()
209            .filter(|command| command.settles_at.is_none())
210            .map(|command| AcpBackgroundEnded {
211                tool_call_id: command.tool_call_id,
212                tool: command.tool.clone(),
213                display: command.display.clone(),
214                started_at: command.since,
215                exit_code: None,
216                signal: None,
217                dropped: Some(why),
218            })
219            .collect();
220        drop(taken);
221        self.notify_background_changed(session);
222        self.refresh_live_waiting(session);
223        ended
224    }

226    /// Records each of `session`'s dropped commands' ends.
227    async fn record_dropped(&self, session: SessionId, dropped: Vec<AcpBackgroundEnded>) {
228        for ended in dropped {
229            let _ = self
230                .append(
231                    session,
232                    None,
233                    event_origin("engine:acp"),
234                    Event::AcpSessionUpdated {
235                        update: AcpSessionUpdate {
236                            background_ended: Some(ended),
237                            ..AcpSessionUpdate::default()
238                        },
239                    },
240                )
241                .await;
242        }
243    }

245    /// A round of `session`'s agent started, which takes in any follow-up:
246    /// its exited commands stop waiting for one.
247    pub(super) fn acp_commands_followed_up(&self, session: SessionId) {
248        self.acp_commands_settle(session, |_| true);
249    }
250
251    /// Ends `session`'s exited commands whose settling time `due` says.
252    fn acp_commands_settle(&self, session: SessionId, due: impl Fn(Instant) -> bool) {
253        // The process may idle once nothing else holds it, which takes its
254        // own locks: these go after the list's.
255        let settled = {
256            let mut commands = self.acp_commands();
257            let (settled, kept) = std::mem::take(&mut *commands)
258                .into_iter()
259                .partition::<Vec<_>, _>(|command| {
260                    command.session == session && command.settles_at.is_some_and(&due)
261                });
262            *commands = kept;
263            settled
264        };
265        if !settled.is_empty() {
266            drop(settled);
267            self.notify_background_changed(session);
268            self.refresh_live_waiting(session);
269        }
270    }
271
272    /// How long an exited command waits for the agent to follow up.
273    fn acp_settle_window(&self) -> Duration {
274        #[cfg(test)]
275        if let Some(window) = *self
276            .inner
277            .acp
278            .settle_window
279            .lock()
280            .unwrap_or_else(|poisoned| poisoned.into_inner())
281        {
282            return window;
283        }
284        ACP_BACKGROUND_SETTLE
285    }

287    /// The agent's process that ran `session`'s commands is gone, and they
288    /// with it, and the work of `session` when it is a subagent of the
289    /// agent's.
290    pub(super) fn acp_commands_gone(&self, session: SessionId) {
291        self.set_acp_subagent_working(session, false);
292        let removed = {
293            let mut commands = self.acp_commands();
294            let before = commands.len();
295            commands.retain(|command| command.session != session);
296            commands.len() != before
297        };
298        if removed {
299            self.notify_background_changed(session);
300            self.refresh_live_waiting(session);
301        }
302    }

304    /// `session`'s agent commands, as background calls.
305    pub(crate) fn acp_background_calls(&self, session: SessionId) -> Vec<BackgroundCall> {
306        self.acp_commands()
307            .iter()
308            .filter(|command| command.session == session && command.settles_at.is_none())
309            .map(|command| BackgroundCall {
310                job: command.terminal.clone(),
311                tool: command.tool.clone(),
312                tool_call_id: command.tool_call_id,
313                display: command.display.clone(),
314                started_at: command.since,
315                detached_at: command.since,
316                by: BackgroundActor::Model,
317                agent_task: true,
318            })
319            .collect()
320    }

322    /// Whether `job` is a running agent command of `session`.
323    pub(crate) fn acp_terminal_running(&self, session: SessionId, job: &str) -> bool {
324        self.acp_commands().iter().any(|command| {
325            command.session == session && command.terminal == job && command.settles_at.is_none()
326        })
327    }

329    /// Whether `session` has an agent command running.
330    pub(crate) fn acp_background_running(&self, session: SessionId) -> bool {
331        self.acp_commands()
332            .iter()
333            .any(|command| command.session == session && command.settles_at.is_none())
334    }

336    /// Whether `session` has an agent command running or settling, which
337    /// may yet wake it.
338    pub(crate) fn acp_background_pending(&self, session: SessionId) -> bool {
339        self.acp_commands()
340            .iter()
341            .any(|command| command.session == session)

344    /// How many agent commands run or settle, in every session.
