11const JANITOR_INTERVAL: Duration = Duration::from_secs(60);
12/// How long a root session tree sits unused before the janitor pages it out.
13/// Viewing it again, or any command, reloads it from disk.
14const ROOT_IDLE_EVICTION_AFTER: Duration = Duration::from_secs(30 * 60);

17    pub(super) fn start_subagent_janitor(&self) {
18        let Some(runtime) = self.inner.runtime.clone() else {
19            return;
20        };
21        let inner = std::sync::Arc::downgrade(&self.inner);
22        let task = runtime.spawn(async move {
23            loop {
24                tokio::time::sleep(JANITOR_INTERVAL).await;
25                let Some(inner) = inner.upgrade() else {
26                    break;
27                };
28                let engine = Engine { inner };
29                if engine
30                    .inner
31                    .delegation
32                    .admission_tasks_closing
33                    .load(std::sync::atomic::Ordering::Acquire)
34                {
35                    break;
36                }
37                if let Err(error) = engine.evict_idle_subagents().await {
38                    eprintln!("subagent session janitor failed: {error}");
39                }
40                if let Err(error) = engine.evict_idle_roots_with(ROOT_IDLE_EVICTION_AFTER).await {
41                    eprintln!("root session janitor failed: {error}");
42                }
43            }
44        });
45        *self
46            .inner
47            .janitor_task
48            .lock()
49            .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(task);
50    }

52    pub(super) async fn evict_idle_subagents(&self) -> Result<Vec<SessionId>, EngineError> {
53        let cap = self.inner.config.runtime.delegation.max_resident_subagents;
54        let idle_after = self.inner.config.runtime.delegation.idle_eviction_after;
55        self.evict_idle_subagents_with(cap, idle_after).await
56    }

158    async fn evict_idle_subagents_with(
159        &self,
160        cap: usize,
161        idle_after: Duration,
162    ) -> Result<Vec<SessionId>, EngineError> {
163        let _delegation_admission = self.inner.delegation.admission.lock().await;
164        // The cap applies to each tree on its own: one tree's activity never
165        // pages out another tree's children (tree-local C10).
166        let over_cap = self
167            .inner
168            .store
169            .resident_subagent_counts()
170            .into_iter()
171            .filter(|(_, count)| *count > cap)
172            .map(|(root, _)| root)
173            .collect::<std::collections::HashSet<_>>();
174        if over_cap.is_empty() {
175            return Ok(Vec::new());
176        }
177        let now = Timestamp::now();
178        let mut candidates = self
179            .inner
180            .store
181            .all()
182            .into_iter()
183            .filter_map(|session| {
184                let SessionOrigin::Delegated {
185                    root_session_id, ..
186                } = session.meta.origin
187                else {
188                    return None;
189                };
190                if !over_cap.contains(&root_session_id) {
191                    return None;
192                }
193                let ended_at = self.subagent_eviction_ended_at(&session, now, idle_after)?;
194                Some((ended_at, root_session_id, session.meta.session_id))
195            })
196            .collect::<Vec<_>>();
197        candidates.sort_by_key(|(ended_at, _, session_id)| (*ended_at, *session_id));
198
199        let mut evicted = Vec::new();
200        for (_, root, session_id) in candidates {
201            if self
202                .inner
203                .store
204                .resident_subagent_counts()
205                .get(&root)
206                .is_none_or(|count| *count <= cap)
207            {
208                continue;
209            }
210            let _residency = self.inner.sessions.residency_mutation.lock().await;
211            let Some(session) = self.inner.store.get_resident(session_id) else {
212                continue;
213            };
214            if self
215                .subagent_eviction_ended_at(&session, Timestamp::now(), idle_after)
216                .is_none()
217            {
218                continue;
219            }
220            #[cfg(test)]
221            let hook = self
222                .inner
223                .test_hooks
224                .janitor_before_barrier_hook
225                .lock()
226                .expect("janitor barrier hook lock poisoned")
227                .take();
228            #[cfg(test)]
229            if let Some(hook) = hook {
230                if let Some(reached) = hook
231                    .reached
232                    .lock()
233                    .expect("janitor barrier reached lock poisoned")
234                    .take()
235                {
236                    let _ = reached.send(());
237                }
238                hook.release.notified().await;
239            }
240            self.drain_actor(session_id).await?;
241            let Some(session) = self.inner.store.get_resident(session_id) else {
242                continue;
243            };
244            if self
245                .subagent_eviction_ended_at(&session, Timestamp::now(), idle_after)
246                .is_none()
247            {
248                continue;
249            }
250            let last_event_seq = session.meta.last_event_seq;
251            self.inner
252                .sessions
253                .actors
254                .lock()
255                .unwrap_or_else(|poisoned| poisoned.into_inner())
256                .remove(&session_id);
257            if self.inner.store.evict(session_id)? {
258                self.inner
259                    .store
260                    .notify_evicted_subscribers(session_id, last_event_seq);
261                self.clear_evicted_session_caches(session_id, &session.meta.origin);
262                evicted.push(session_id);
263            }
264        }
265        Ok(evicted)
266    }

268    fn subagent_eviction_ended_at(
269        &self,
270        session: &crate::session::SessionProjection,
271        now: Timestamp,
272        idle_after: Duration,
273    ) -> Option<Timestamp> {
274        if !matches!(session.meta.origin, SessionOrigin::Delegated { .. })
275            || session.runs.is_empty()
276            || self.session_has_pending_work(session)
277        {
278            return None;
279        }
280        if let Some(record) = self
281            .inner
282            .delegation
283            .by_session
284            .lock()
285            .unwrap_or_else(|poisoned| poisoned.into_inner())
286            .get(&session.meta.session_id)
287            .copied()
288            && (!matches!(record.state, DelegationState::Finished(_))
289                || record.background && !record.notification_sent)
290        {
291            return None;
292        }
293        let ended_at = session
294            .log
295            .event_snapshot()
296            .iter()
297            .filter(|event| {
298                event.run_id.is_some()
299                    && matches!(
300                        event.payload,
301                        Event::RunCompleted { .. }
302                            | Event::RunFailed { .. }
303                            | Event::RunCancelled { .. }
304                            | Event::RunInterrupted { .. }
305                    )
306            })
307            .map(|event| event.timestamp)
308            .max()?;
309        let idle = Duration::try_from(now.duration_since(ended_at)).ok()?;
310        (idle > idle_after).then_some(ended_at)
311    }

313    /// Whether anything still needs `session` in memory: a live or queued
314    /// run, compaction, an approval, unadmitted input, or producer work.
315    fn session_has_pending_work(&self, session: &crate::session::SessionProjection) -> bool {
316        let session_id = session.meta.session_id;
317        session.status == SessionStatus::Running
318            || session
319                .runs
320                .values()
321                .any(|run| matches!(run.status, SessionStatus::Running | SessionStatus::Idle))
322            || self
323                .inner
324                .sessions
325                .active
326                .lock()
327                .unwrap_or_else(|poisoned| poisoned.into_inner())
328                .values()
329                .any(|active| active.session == session_id)
330            || self
331                .inner
332                .delegation
333                .queue
334                .lock()
335                .unwrap_or_else(|poisoned| poisoned.into_inner())
336                .contains(&session_id)
337            || self
338                .inner
339                .compaction
340                .in_progress
341                .lock()
342                .unwrap_or_else(|poisoned| poisoned.into_inner())
343                .contains(&session_id)
344            || self
345                .inner
346                .approvals
347                .pending
348                .lock()
349                .unwrap_or_else(|poisoned| poisoned.into_inner())
350                .keys()
351                .any(|(pending, _)| *pending == session_id)
352            || has_runless_pending_inputs(&session.log.event_snapshot())
353            || self.producers_pin_session(session_id)
354    }

356    /// Pages out root session trees nobody has used for `idle_after`: no
357    /// event, no live tail, and no pending work anywhere in the tree. A tree
358    /// goes whole or not at all, children first, so an evicted root never
359    /// leaves resident children behind it. Its runtime-only settings (the
360    /// permission mode, context calibration) survive for when it reloads.
361    pub(super) async fn evict_idle_roots_with(
362        &self,
363        idle_after: Duration,
364    ) -> Result<Vec<SessionId>, EngineError> {
365        let _delegation_admission = self.inner.delegation.admission.lock().await;
366        let now = Timestamp::now();
367        let resident = self.inner.store.all();
368        let candidates = resident
369            .iter()
370            .filter(|session| self.root_tree_idle(session, &resident, now, idle_after))
371            .map(|session| session.meta.session_id)
372            .collect::<Vec<_>>();
373        let mut evicted = Vec::new();
374        for root in candidates {
375            let _residency = self.inner.sessions.residency_mutation.lock().await;
376            let still_idle = |engine: &Self| {
377                let resident = engine.inner.store.all();
378                resident.iter().any(|session| {
379                    session.meta.session_id == root
380                        && engine.root_tree_idle(session, &resident, Timestamp::now(), idle_after)
381                })
382            };
383            if !still_idle(self) {
384                continue;
385            }
386            // Drain each tree member's actor so nothing it already accepted
387            // is cut off, then re-check: the drained work may have used it.
388            let members = self
389                .inner
390                .store
391                .all()
392                .into_iter()
393                .filter(|session| root_id(&session.meta.origin, session.meta.session_id) == root)
394                .map(|session| session.meta.session_id)
395                .collect::<Vec<_>>();
396            for member in &members {
397                self.drain_actor(*member).await?;
398            }
399            if !still_idle(self) {
400                continue;
401            }
402            // Children first, root last.
403            let mut ordered = members;
404            ordered.sort_by_key(|member| *member == root);
405            for member in ordered {
406                let Some(session) = self.inner.store.get_resident(member) else {
407                    continue;
408                };
409                let last_event_seq = session.meta.last_event_seq;
410                self.inner
411                    .sessions
412                    .actors
413                    .lock()
414                    .unwrap_or_else(|poisoned| poisoned.into_inner())
415                    .remove(&member);
416                if !self.inner.store.evict(member)? {
417                    // A log never persisted (an empty session) stays put.
418                    continue;
419                }
420                self.inner
421                    .store
422                    .notify_evicted_subscribers(member, last_event_seq);
423                if member == root {
424                    self.inner
425                        .sessions
426                        .producers
427                        .lock()
428                        .unwrap_or_else(|e| e.into_inner())
429                        .remove(&member);
430                    self.inner
431                        .sessions
432                        .producer_projections
433                        .lock()
434                        .unwrap_or_else(|e| e.into_inner())
435                        .remove(&member);
436                } else {
437                    self.clear_evicted_session_caches(member, &session.meta.origin);
438                }
439                evicted.push(member);
440            }
441        }
442        Ok(evicted)
443    }

445    /// Whether `session` is a root whose whole resident tree has been unused
446    /// for `idle_after`.
447    fn root_tree_idle(
448        &self,
449        session: &crate::session::SessionProjection,
450        resident: &[Arc<crate::session::SessionProjection>],
451        now: Timestamp,
452        idle_after: Duration,
453    ) -> bool {
454        if !matches!(session.meta.origin, SessionOrigin::Root) {
455            return false;
456        }
457        let root = session.meta.session_id;
458        let unused = |member: &crate::session::SessionProjection| {
459            // Cheap checks first: the pending-work check scans the log.
460            member.log.last_event().is_some_and(|event| {
461                Duration::try_from(now.duration_since(event.timestamp))
462                    .is_ok_and(|idle| idle > idle_after)
463            }) && !self.inner.store.has_live_tail(member.meta.session_id)
464                && !self.session_has_pending_work(member)
465        };
466        unused(session)
467            && resident
468                .iter()
469                .filter(|member| {
470                    matches!(
471                        member.meta.origin,
472                        SessionOrigin::Delegated { root_session_id, .. } if root_session_id == root
473                    )
474                })
475                .all(|child| {
476                    unused(child)
477                        && self
478                            .subagent_eviction_ended_at(child, now, idle_after)
479                            .is_some()
480                })
481    }

483    /// Waits until `session`'s actor has run everything queued before now.
484    async fn drain_actor(&self, session: SessionId) -> Result<(), EngineError> {
485        let actor = self
486            .inner
487            .sessions
488            .actors
489            .lock()
490            .unwrap_or_else(|poisoned| poisoned.into_inner())
491            .get(&session)
492            .cloned();
493        if let Some(actor) = actor {
494            let (barrier, receiver) = SessionCommand::call(false, |_| async { Ok(()) });
495            actor
496                .send(barrier)
497                .await
498                .map_err(|_| EngineError::ActorStopped)?;
499            receiver.await.map_err(|_| EngineError::ActorStopped)??;
500        }
501        Ok(())
502    }

504    fn clear_evicted_session_caches(&self, session_id: SessionId, origin: &SessionOrigin) {
505        self.inner
506            .sessions
507            .producers
508            .lock()
509            .unwrap_or_else(|e| e.into_inner())
510            .remove(&session_id);
511        self.inner
512            .sessions
513            .producer_projections
514            .lock()
515            .unwrap_or_else(|e| e.into_inner())
516            .remove(&session_id);
517        // Permission modes are tree-root keyed. Paging out a child must retain
518        // the shared mode; an evicted root no longer owns runtime-only state.
519        if root_id(origin, session_id) == session_id {
520            self.inner
521                .approvals
522                .permission_modes
523                .lock()
524                .unwrap_or_else(|poisoned| poisoned.into_inner())
525                .remove(&session_id);
526        }
527        self.inner
528            .compaction
529            .context_token_estimators
530            .lock()
531            .unwrap_or_else(|poisoned| poisoned.into_inner())
532            .remove(&session_id);
533        self.inner
534            .compaction
535            .in_progress
536            .lock()
537            .unwrap_or_else(|poisoned| poisoned.into_inner())
538            .remove(&session_id);
539        self.inner
540            .compaction
541            .deferred
542            .lock()
543            .unwrap_or_else(|poisoned| poisoned.into_inner())
544            .remove(&session_id);
545    }

548pub(super) fn has_runless_pending_inputs(
549    events: &[Arc<cookie_agent_protocol::StoredEvent>],
550) -> bool {
551    let boundary = events
552        .iter()
553        .rev()
554        .find(|event| {
555            matches!(
556                event.payload,
557                Event::RunCompleted { .. }
558                    | Event::RunFailed { .. }
559                    | Event::RunCancelled { .. }
560                    | Event::RunInterrupted { .. }
561            )
562        })
563        .map_or(0, |event| event.seq);
564    let mut pending = 0_usize;
565    for event in events
566        .iter()
567        .filter(|event| event.seq > boundary && event.run_id.is_none())
568    {
569        match event.payload {
570            Event::UserInputAdmitted { .. } => pending += 1,
571            Event::UserInputRecalled { .. } => pending = pending.saturating_sub(1),
572            _ => {}
573        }
574    }
575    pending != 0
576}
