59/// The source of a diagnostic about what `tool`'s provider sent: its plugin
60/// or runtime, else the engine's tool host.
61fn peer_diagnostic_source(tool: &ToolIdentity) -> cookie_agent_protocol::DiagnosticSource {
62    let source = match tool.source.as_ref().and_then(Open::known) {
63        Some(ToolSource::Plugin { plugin }) => {
64            cookie_agent_protocol::DiagnosticSource::plugin(plugin).ok()
65        }
66        Some(
67            ToolSource::Runtime { runtime_id, .. }
68            | ToolSource::Mcp {
69                runtime_id: Some(runtime_id),
70                ..
71            },
72        ) => cookie_agent_protocol::DiagnosticSource::runtime(runtime_id.as_str()).ok(),
73        _ => None,
74    };
75    source.unwrap_or_else(|| {
76        cookie_agent_protocol::DiagnosticSource::engine("tools")
77            .expect("static diagnostic source is valid")
78    })
79}

82    /// Records that the daemon dropped display blocks of kinds this build
83    /// does not know, of tags `tags`, from the `part` (`presentation`,
84    /// `progress`, or `result`) of a call of `tool`. A plugin or runtime
85    /// sends only the kinds the daemon's version includes, so such a block
86    /// comes from a peer newer than its handshake says
87    /// (`docs/site/specs/compatibility-v1.md` section 12.4).
88    pub(super) fn report_unknown_display_blocks(
89        &self,
90        session: SessionId,
91        tool: &ToolIdentity,
92        part: &'static str,
93        mut tags: Vec<String>,
94    ) {
95        if tags.is_empty() {
96            return;
97        }
98        let count = tags.len() as u64;
99        tags.sort_unstable();
100        tags.dedup();
101        tags.truncate(MAX_REPORTED_BLOCK_KINDS);
102        let kinds = tags
103            .iter()
104            .map(|tag| cookie_agent_protocol::diagnostics::sanitize(tag, 64))
105            .collect::<Vec<_>>();
106        let message = format!(
107            "dropped {count} display block{} of a kind this daemon does not know from the {part} of tool `{}`: {}",
108            if count == 1 { "" } else { "s" },
109            cookie_agent_protocol::diagnostics::sanitize(&tool.published_name, 64),
110            kinds.join(", ")
111        );
112        let event = Event::Diagnostic {
113            source: peer_diagnostic_source(tool),
114            kind: cookie_agent_protocol::PeerDiagnosticKind::UnknownDisplayBlock.label(),
115            message,
116            count: None,
117            detail: Some(serde_json::json!({
118                "tool": tool.published_name,
119                "part": part,
120                "kinds": kinds,
121            })),
122        };
123        let engine = self.clone();
124        tokio::spawn(async move {
125            let _ = engine
126                .on_actor(session, move |engine| {
127                    engine.append_direct(session, None, super::event_origin("engine:tools"), event)
128                })
129                .await;
130        });
131    }

133    pub(super) fn decide_tool_permission(
134        &self,
135        active: &ActiveRun,
136        prepared_call: &PreparedToolCall,
137    ) -> Result<PermissionDecision, ToolFailure> {
138        let prepared = prepared_call.prepared.as_ref().map_err(Clone::clone)?;
139        // A resolved tool without a permission name needs no permission.
140        if prepared_call
141            .interception
142            .as_ref()
143            .is_some_and(|interception| interception.permission_name.is_none())
144        {
145            return Ok(PermissionDecision::permission_free());
146        }
147        let permission_overlay = self
148            .inner
149            .store
150            .get(active.session)
151            .map_err(|error| ToolFailure {
152                partial_output: None,
153                code: ToolCallFailureCode::ExecutionFailed,
154                message: error.to_string(),
155            })?
156            .permission_overlay
157            .clone();
158        let grants = self.skill_grants_for_session(active.session, prepared_call.attachment());
159        let decide = |scope: PermissionScope<'_>| {
160            self.inner.permissions.decide_operation_with_grants(
161                &active.policy.agent,
162                Some(&permission_overlay),
163                grants.as_ref(),
164                prepared.operation(),
165                prepared.policy_labels(),
166                scope,
167            )
168        };
169        let scopes = prepared.resource_scopes();
170        if !scopes.is_empty() {
171            // Each resource is checked in its own scope only.
172            let decisions = scopes
173                .iter()
174                .map(|scope| {
175                    decide(
176                        PermissionScope::runtime(&scope.runtime, scope.workspace.as_deref())
177                            .with_home(scope.home.as_deref()),
178                    )
179                })
180                .collect::<Vec<_>>();
181            return Ok(per_resource_decision(decisions));
182        }
183        let runtimes = prepared.permission_runtimes();
184        if runtimes.is_empty() {
185            return Ok(decide(
186                prepared_call
187                    .origin()
188                    .map_or(PermissionScope::BASE, PermissionScope::of),
189            ));
190        }
191        // A call that acts on several runtimes is checked as
192        // `action@runtime` under each; the strictest decision wins, per
193        // resource.
194        Ok(strictest_decision(runtimes.iter().map(|runtime| {
195            decide(PermissionScope::runtime(runtime, None))
196        })))
197    }

199    pub(super) async fn resolve_tool_permission(
200        &self,
201        active: &ActiveRun,
202        run: RunId,
203        prepared_call: &PreparedToolCall,
204        permission: &PermissionDecision,
205    ) -> Result<(), ToolFailure> {
206        let prepared = prepared_call.prepared.as_ref().map_err(Clone::clone)?;
207        match permission.effect {
208            cookie_agent_protocol::PermissionEffect::Allow => Ok(()),
209            cookie_agent_protocol::PermissionEffect::Deny => Err(ToolError::PermissionDenied(
210                denied_tool_failure("policy", "permission denied", None),
211            )
212            .into()),
213            cookie_agent_protocol::PermissionEffect::Ask => {
214                let allow_tree_grant = prepared.operation().resources().iter().all(|resource| {
215                    resource.binding_lifetime
216                        == cookie_agent_protocol::PreparedBindingLifetime::RestartStable
217                        && !resource.capability.is_path()
218                });
219                let request = ApprovalRequest::new(
220                    ApprovalId::new_v7(),
221                    1,
222                    prepared_call.call.id,
223                    ApprovalTrigger::PermissionPolicy,
224                    prepared.operation().clone(),
225                    permission.evaluations.clone(),
226                    ApprovalConstraints {
227                        allow_once: true,
228                        allow_tree_grant,
229                        cancellable: true,
230                        // The user's window, `[approval].timeout_ms`, starts
231                        // when the call escalates to them.
232                        expires_at: None,
233                    },
234                )
235                .map_err(|error| ToolFailure {
236                    partial_output: None,
237                    code: ToolCallFailureCode::ExecutionFailed,
238                    message: error.to_string(),
239                })?;
240                let outcome = self
241                    .await_user_approval(
242                        active,
243                        run,
244                        request,
245                        prepared.executor().clone(),
246                        true,
247                        ApprovalToolInput {
248                            tool_call_id: prepared_call.call.id,
249                            origin: prepared_call.origin(),
250                        },
251                    )
252                    .await
253                    .map_err(|error| ToolFailure {
254                        partial_output: None,
255                        code: ToolCallFailureCode::ExecutionFailed,
256                        message: error.to_string(),
257                    })?;
258                if outcome.approved {
259                    Ok(())
260                } else {
261                    Err(ToolFailure {
262                        partial_output: None,
263                        code: ToolCallFailureCode::ExecutionFailed,
264                        message: outcome.denial_message("policy", "permission refused by user"),
265                    })
266                }
267            }
268        }
269    }

292    pub(super) async fn prepare_published_tool_calls(
293        &self,
294        session_id: SessionId,
295        run: RunId,
296        calls: Vec<(ToolCall, Option<&PublishedTool>)>,
297        policy: &FrozenRunPolicy,
298        turn_context: Arc<crate::TurnAgentContext>,
299    ) -> Vec<PreparedToolCall> {
300        let composition = self.compose_for_policy_with(session_id, policy, true).ok();
301        // Calls grouped by the provider and attachment that answer them, each
302        // with the name its source knows the tool by. A loader call prepares
303        // on its own.
304        let mut groups: Vec<CallGroup> = Vec::new();
305        for (index, (call, published)) in calls.iter().enumerate() {
306            // A routed call finds its machine's tool by its composed name.
307            let composed =
308                published.map_or(call.name.as_str(), |published| published.composed.as_str());
309            let Some(tool) = composition
310                .as_ref()
311                .and_then(|composition| composition.get(composed))
312                .filter(|tool| {
313                    tool.source.provider.provider_id() != crate::base_tools::LOADER_PROVIDER_ID
314                })
315            else {
316                continue;
317            };
318            let entry = (index, tool.spec.clone());
319            if let Some((_, indexes)) = groups.iter_mut().find(|(candidate, _)| {
320                Arc::ptr_eq(&candidate.provider, &tool.source.provider)
321                    && candidate.attachment == tool.source.attachment
322            }) {
323                indexes.push(entry);
324            } else {
325                groups.push((tool.source.clone(), vec![entry]));
326            }
327        }
328        let mut prepared = (0..calls.len()).map(|_| None).collect::<Vec<_>>();
329        for (source, indexes) in groups {
330            let provider_calls = indexes
331                .iter()
332                .map(|(index, spec)| {
333                    let mut call = ToolCall {
334                        name: spec.name.clone(),
335                        ..calls[*index].0.clone()
336                    };
337                    // Resolution reports a malformed `background` itself.
338                    let _ = take_background(spec, &mut call);
339                    call
340                })
341                .collect();
342            let context = ToolPreparationContext {
343                session: session_id,
344                run,
345                origin: source.origin.clone(),
346                turn_context: turn_context.clone(),
347            };
348            let results = source
349                .provider
350                .prepare_parallel(context, provider_calls)
351                .await;
352            for ((index, _), result) in indexes.into_iter().zip(results) {
353                prepared[index] = Some(result);
354            }
355        }
356        // Each call's tool must still be what the model was offered: one
357        // composition after the preparations checks them all.
358        let current = self.compose_for_policy_with(session_id, policy, true).ok();
359        let mut output = Vec::with_capacity(calls.len());
360        for (index, (call, published)) in calls.into_iter().enumerate() {
361            output.push(
362                self.prepare_tool_call_with_publication(
363                    session_id,
364                    run,
365                    call,
366                    policy,
367                    turn_context.clone(),
368                    Some(published),
369                    prepared[index].take(),
370                    current.as_ref(),
371                )
372                .await,
373            );
374        }
375        output
376    }

378    #[allow(clippy::too_many_arguments)]
379    async fn prepare_tool_call_with_publication(
380        &self,
381        session_id: SessionId,
382        run: RunId,
383        call: ToolCall,
384        policy: &FrozenRunPolicy,
385        turn_context: Arc<crate::TurnAgentContext>,
386        published: Option<Option<&PublishedTool>>,
387        prepared_override: Option<Result<PreparedTool, ToolError>>,
388        current: Option<&Composition>,
389    ) -> PreparedToolCall {
390        // A call that fails before it prepares shows the name the model
391        // called, and the tool's source when its publication names it.
392        let mut fallback_presentation = ToolPresentation::new(
393            cookie_agent_runtime_api::tool_title_only(&call.name),
394            tool_identity(
395                &call.name,
396                published
397                    .flatten()
398                    .map(|published| (&published.source, &*published.spec)),
399            ),
400        );
401        if call.arguments.is_null() {
402            // A null argument object is the marked-invalid convention for
403            // provider argument text that could not be parsed. Refuse before
404            // schema validation so a permissive published tool can never run
405            // with unsalvageable input.
406            return PreparedToolCall {
407                intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
408                call,
409                permission_name: None,
410                presentation: fallback_presentation,
411                prepared: Err(ToolFailure {
412                    partial_output: None,
413                    code: ToolCallFailureCode::ExecutionFailed,
414                    message: "tool call arguments are invalid or unsalvageable; refusing to execute with null input"
415                        .to_owned(),
416                }),
417                interception: None,
418                background: None,
419            };
420        }
421        let session = match self.inner.store.get(session_id) {
422            Ok(session) => session,
423            Err(error) => {
424                return PreparedToolCall {
425                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
426                    call,
427                    permission_name: None,
428                    presentation: fallback_presentation,
429                    prepared: Err(ToolFailure {
430                        partial_output: None,
431                        code: ToolCallFailureCode::ExecutionFailed,
432                        message: error.to_string(),
433                    }),
434                    interception: None,
435                    background: None,
436                };
437            }
438        };
439        let depth = session_depth(&session.meta.origin);
440        let permission_overlay = &session.permission_overlay;
441        let delegate_enabled = !policy.delegate_targets(depth).is_empty();
442        let mut provider = match published {
443            Some(Some(published)) => {
444                // The tool must still be what the model was offered: same
445                // attachment, same instance generation, same declaration.
446                // The loaders are published only where a request offered
447                // them, so the check keeps them.
448                let composed;
449                let current = match current {
450                    Some(current) => Some(current),
451                    None => {
452                        composed = self.compose_for_policy_with(session_id, policy, true).ok();
453                        composed.as_ref()
454                    }
455                }
456                .and_then(|composition| composition.get(&published.composed));
457                if current.is_some_and(|current| {
458                    current.source.attachment == published.source.attachment
459                        && current.source.generation == published.source.generation
460                        && current.spec == published.spec
461                }) {
462                    Some((published.source.clone(), published.spec.clone()))
463                } else {
464                    return PreparedToolCall {
465                        intercepted_arguments: Arc::new(std::sync::Mutex::new(
466                            call.arguments.clone(),
467                        )),
468                        prepared: Err(ToolFailure {
469                            partial_output: None,
470                            code: ToolCallFailureCode::OperationChanged,
471                            message: format!(
472                                "tool definition changed after `{}` was published to the model",
473                                call.name
474                            ),
475                        }),
476                        call,
477                        permission_name: None,
478                        presentation: fallback_presentation,
479                        interception: None,
480                        background: None,
481                    };
482                }
483            }
484            Some(None) => {
485                return PreparedToolCall {
486                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
487                    prepared: Err(ToolFailure {
488                        partial_output: None,
489                        code: ToolCallFailureCode::ExecutionFailed,
490                        message: format!("tool `{}` was not published to the model", call.name),
491                    }),
492                    call,
493                    permission_name: None,
494                    presentation: fallback_presentation,
495                    interception: None,
496                    background: None,
497                };
498            }
499            None => self
500                .compose_session_tools(session_id, |_, _| true)
501                .ok()
502                .and_then(|composition| {
503                    composition
504                        .get(&call.name)
505                        .map(|tool| (tool.source.clone(), tool.spec.clone()))
506                }),
507        };
508        if published.is_none()
509            && provider.is_none()
510            && (call.name != "skill" || self.is_direct_skill_call(call.id))
511        {
512            let sources = self.session_tool_sources(session_id).unwrap_or_default();
513            provider = sources.iter().find_map(|candidate| {
514                candidate
515                    .provider
516                    .permission_for_unlisted_tool(&call.name)
517                    .ok()
518                    .flatten()
519                    .map(|permission_name| {
520                        (
521                            candidate.clone(),
522                            Arc::new(crate::ToolSpec {
523                                background: false,
524                                background_default: false,
525                                collision: Default::default(),
526                                output: Default::default(),
527                                concurrency: Default::default(),
528                                name: call.name.clone(),
529                                permission_name: Some(permission_name.to_owned()),
530                                description: String::new(),
531                                parameters: serde_json::json!(true),
532                                result_truncation: Default::default(),
533                            }),
534                        )
535                    })
536            });
537        }
538        let Some((source, tool_spec)) = provider else {
539            return PreparedToolCall {
540                intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
541                prepared: Err(ToolFailure {
542                    partial_output: None,
543                    code: ToolCallFailureCode::ExecutionFailed,
544                    message: format!("tool `{}` is unavailable", call.name),
545                }),
546                call,
547                permission_name: None,
548                presentation: fallback_presentation,
549                interception: None,
550                background: None,
551            };
552        };
553        fallback_presentation.tool = tool_identity(&call.name, Some((&source, &*tool_spec)));
554        let tool = fallback_presentation.tool.clone();
555        let SourcedProvider {
556            attachment,
557            origin,
558            provider,
559            ..
560        } = source;
561        // A prefixed tool runs under the name its runtime knows it by.
562        let mut call = ToolCall {
563            name: tool_spec.name.clone(),
564            ..call
565        };
566        let background = match take_background(&tool_spec, &mut call) {
567            Ok(background) => background,
568            Err(message) => {
569                return PreparedToolCall {
570                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
571                    prepared: Err(ToolFailure {
572                        partial_output: None,
573                        code: ToolCallFailureCode::ExecutionFailed,
574                        message,
575                    }),
576                    call,
577                    permission_name: tool_spec.permission_name.clone(),
578                    presentation: fallback_presentation,
579                    interception: None,
580                    background: None,
581                };
582            }
583        };
584        let grants = self.skill_grants_for_session(session_id, attachment.as_ref());
585        let permission_name = tool_spec.permission_name.clone();
586        let delegation_tool = permission_name.as_deref() == Some("delegate");
587        let enabled = !delegation_tool || delegate_enabled;
588        let direct_skill = call.name == "skill" && self.is_direct_skill_call(call.id);
589        if !enabled
590            || (!direct_skill
591                && !permission_name.as_deref().is_none_or(|permission_name| {
592                    PermissionPipeline::tool_offered(
593                        &policy.agent,
594                        Some(permission_overlay),
595                        grants.as_ref(),
596                        permission_name,
597                        &call.name,
598                        PermissionScope::of(&origin),
599                    )
600                }))
601        {
602            return PreparedToolCall {
603                intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
604                prepared: Err(ToolFailure {
605                    partial_output: None,
606                    code: ToolCallFailureCode::ExecutionFailed,
607                    message: format!("tool `{}` is not enabled for this session", call.name),
608                }),
609                call,
610                permission_name,
611                presentation: fallback_presentation,
612                interception: None,
613                background: None,
614            };
615        }
616        let context = ToolPreparationContext {
617            session: session_id,
618            run,
619            origin,
620            turn_context,
621        };
622        let (prepared, original_resource) = match match prepared_override {
623            Some(prepared) => prepared,
624            None => provider.prepare(context.clone(), call.clone()).await,
625        } {
626            Ok(prepared) => {
627                match apply_permission_resource(
628                    provider.as_ref(),
629                    &call.name,
630                    permission_name.as_deref(),
631                    &tool,
632                    &context.origin,
633                    prepared,
634                ) {
635                    Ok((prepared, resource)) => (Ok(prepared), resource),
636                    Err(error) => (Err(error.into()), None),
637                }
638            }
639            Err(error) => (Err(error.into()), None),
640        };
641        // A provider that reports its metadata at preparation (a runtime
642        // behind the runtime protocol), or that shows the call from what it
643        // prepared, reports how the call is shown with it.
644        let mut prepared_presentation = match &prepared {
645            Ok(prepared) => prepared.presentation().cloned(),
646            Err(_) => None,
647        }
648        .unwrap_or_else(|| provider.presentation(&call));
649        let unknown = prepared_presentation.fit();
650        self.report_unknown_display_blocks(session_id, &tool, "presentation", unknown);
651        let presentation = ToolPresentation::new(prepared_presentation, tool.clone());
652        PreparedToolCall {
653            intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
654            call,
655            permission_name: permission_name.clone(),
656            presentation,
657            prepared,
658            interception: Some(ToolInterceptionContext {
659                attachment,
660                tool,
661                provider,
662                spec: tool_spec,
663                preparation: context,
664                permission_name,
665                permission_resource: original_resource,
666            }),
667            background,
668        }
669    }

671    /// Decides, approves, and runs `prepared`. With `retain_output` the
672    /// call's output is captured as a model's call's is (a call of its own,
673    /// such as an ACP agent's over the bridge); without, the caller owns
674    /// its single output (direct skill dispatch, nested in another call).
675    pub(super) async fn execute_tool(
676        &self,
677        active: Arc<ActiveRun>,
678        run: RunId,
679        prepared: PreparedToolCall,
680        turn_context: Arc<crate::TurnAgentContext>,
681        retain_output: bool,
682    ) -> Result<ToolResult, ToolFailure> {
683        let permission = self.decide_tool_permission(&active, &prepared)?;
684        self.resolve_tool_permission(&active, run, &prepared, &permission)
685            .await?;
686        self.execute_approved_tool(active, run, prepared, turn_context, retain_output)
687            .await
688    }

693        run: RunId,
694        prepared: PreparedToolCall,
695        turn_context: Arc<crate::TurnAgentContext>,
696        retain_output: bool,
697    ) -> Result<ToolResult, ToolFailure> {
698        let engine = self.clone();
699        {
700            let PreparedToolCall {
701                mut call,
702                prepared,
703                interception,
704                intercepted_arguments,
705                presentation,
706                background,
707                ..
708            } = prepared;
709            let mut prepared = prepared?;
710            let runtime = interception
711                .as_ref()
712                .and_then(|interception| interception.preparation.origin.runtime())
713                .map(|runtime| runtime.id().to_owned());
714            let result_truncation = interception
715                .as_ref()
716                .map(|interception| interception.spec.result_truncation)
717                .unwrap_or_default();
718            let output_declaration = interception
719                .as_ref()
720                .map(|interception| interception.spec.output.clone())
721                .unwrap_or_default();
722            if result_truncation == crate::ToolResultTruncationPolicy::OptOut
723                && output_declaration != cookie_agent_protocol::ToolOutputDeclaration::Single
724            {
725                return Err(ToolError::execution(
726                    "self-paginating tools require a single output declaration",
727                )
728                .into());
729            }
730            let operation = prepared.operation().clone();
731            let policy_labels = prepared.policy_labels().to_vec();
732            if let Some(interception) = interception {
733                let context_id = crate::plugin::plugin_context_id();
734                for hooked in engine.inner.plugins.interception_plugins(
735                    cookie_agent_protocol::ExtensionInterceptionHook::ToolBeforeCall,
736                ) {
737                    let plugin = hooked.name.clone();
738                    let result = engine
739                        .inner
740                        .plugins
741                        .intercept_named::<_, cookie_agent_protocol::ExtensionToolBeforeCallResult>(
742                            &plugin,
743                            cookie_agent_protocol::PLUGIN_INTERCEPT_TOOL_BEFORE_CALL_METHOD,
744                            &ExtensionToolBeforeCallParams {
745                                session_id: active.session,
746                                context_id: context_id.clone(),
747                                tool: call.name.clone(),
748                                origin: interception.hook_origin().into(),
749                                arguments: call.arguments.clone(),
750                                permission_name: interception.permission_name.clone(),
751                                resource: interception.permission_resource.clone(),
752                            },
753                            Some(active.session),
754                            Some(&context_id),
755                        )
756                        .await;
757                    let result = match result {
758                        Ok(result) => result,
759                        Err(error) => {
760                            if let Some(failure) = engine.hook_failed(
761                                active.session,
762                                &hooked,
763                                cookie_agent_protocol::ExtensionInterceptionHook::ToolBeforeCall,
764                                error,
765                            ) {
766                                return Err(ToolFailure {
767                                    partial_output: None,
768                                    code: ToolCallFailureCode::ExecutionFailed,
769                                    message: format!("Tool call blocked: {failure}."),
770                                });
771                            }
772                            continue;
773                        }
774                    };
775                    if result.action == ExtensionToolBeforeCallAction::Block {
776                        let message = result
777                            .message_to_model
778                            .or(result.reason)
779                            .unwrap_or_else(|| format!("plugin `{plugin}` blocked the tool call"));
780                        engine.record_plugin_diagnostic(
781                            active.session,
782                            plugin,
783                            PluginDiagnosticKind::HookBlocked,
784                            message.clone(),
785                        );
786                        return Err(ToolFailure {
787                            partial_output: None,
788                            code: ToolCallFailureCode::ExecutionFailed,
789                            message,
790                        });
791                    }
792                    let Some(arguments) = result.modified_arguments else {
793                        continue;
794                    };
795                    let valid_schema = jsonschema::validator_for(&interception.spec.parameters)
796                        .is_ok_and(|validator| validator.is_valid(&arguments));
797                    let invalid_modification = |message: String| {
798                        engine.record_plugin_diagnostic(
799                            active.session,
800                            plugin.clone(),
801                            PluginDiagnosticKind::InvalidModification,
802                            message.clone(),
803                        );
804                        ToolFailure {
805                            partial_output: None,
806                            code: ToolCallFailureCode::ExecutionFailed,
807                            message,
808                        }
809                    };
810                    let invalid_arguments = || {
811                        format!(
812                            "plugin `{plugin}` produced invalid tool arguments or changed the permission resource"
813                        )
814                    };
815                    if !arguments.is_object() || !valid_schema {
816                        return Err(invalid_modification(invalid_arguments()));
817                    }
818                    let mut modified_call = call.clone();
819                    modified_call.arguments = arguments;
820                    // The permission resource of the modified arguments is
821                    // what preparing them reports.
822                    let (candidate, modified_resource) = interception
823                        .provider
824                        .prepare(interception.preparation.clone(), modified_call.clone())
825                        .await
826                        .and_then(|prepared| {
827                            apply_permission_resource(
828                                interception.provider.as_ref(),
829                                &modified_call.name,
830                                interception.permission_name.as_deref(),
831                                &interception.tool,
832                                &interception.preparation.origin,
833                                prepared,
834                            )
835                        })
836                        .map_err(ToolFailure::from)?;
837                    if modified_resource != interception.permission_resource {
838                        return Err(invalid_modification(invalid_arguments()));
839                    }
840                    if candidate.operation().capabilities() != operation.capabilities()
841                        || candidate.operation().resources() != operation.resources()
842                        || candidate.policy_labels() != policy_labels
843                    {
844                        let message = format!(
845                            "plugin `{plugin}` changed the approved permission capability or resource"
846                        );
847                        engine.record_plugin_diagnostic(
848                            active.session,
849                            plugin,
850                            PluginDiagnosticKind::InvalidModification,
851                            message.clone(),
852                        );
853                        return Err(ToolFailure {
854                            partial_output: None,
855                            code: ToolCallFailureCode::ExecutionFailed,
856                            message,
857                        });
858                    }
859                    call = modified_call;
860                    prepared = candidate;
861                    *intercepted_arguments
862                        .lock()
863                        .unwrap_or_else(|poisoned| poisoned.into_inner()) = call.arguments.clone();
864                }
865            }
866            let _serialization_guard = if let Some(key) = prepared.serialization_key() {
867                Some(engine.mutation_lock(key).lock_owned().await)
868            } else {
869                None
870            };
871            let executor = prepared
872                .executor()
873                .lock()
874                .await
875                .take()
876                .ok_or_else(|| ToolFailure {
877                    partial_output: None,
878                    code: ToolCallFailureCode::PreparedCapabilityLost,
879                    message: "prepared executor capability was already consumed or lost".into(),
880                })?;
881            executor.revalidate().await.map_err(ToolFailure::from)?;
882            let (progress_tx, mut progress_rx) = mpsc::channel(64);
883            let hub = engine
884                .inner
885                .output
886                .hubs
887                .lock()
888                .unwrap_or_else(|poisoned| poisoned.into_inner())
889                .entry(call.id)
890                .or_insert_with(|| OutputHub::new(call.id, 64 * 1024))
891                .clone();
892            let capture = if retain_output
893                && result_truncation != crate::ToolResultTruncationPolicy::OptOut
894            {
895                Some(
896                    OutputCapture::with_limits(
897                        engine.inner.artifacts.clone(),
898                        active.session,
899                        output_declaration,
900                        active.policy.result_limits.tool_output,
901                    )
902                    .await
903                    .map_err(ToolFailure::from)?,
904                )
905            } else {
906                None
907            };
908            if let Some(capture) = &capture {
909                engine
910                    .inner
911                    .output
912                    .captures
913                    .lock()
914                    .unwrap_or_else(|p| p.into_inner())
915                    .insert(call.id, capture.clone());
916            }
917            // The call's own token: the run cancels it only while the call is
918            // in the foreground, so a detached call outlives the run.
919            let tool_cancellation = tokio_util::sync::CancellationToken::new();
920            let (detach_point, mut detach_watch) = cookie_agent_runtime_api::DetachPoint::channel();
921            // A display block of a kind this build does not know is dropped,
922            // and reported once per call and part.
923            let unknown_reporter = |part: &'static str| {
924                let engine = engine.clone();
925                let session = active.session;
926                let tool = presentation.tool.clone();
927                let reported = std::sync::atomic::AtomicBool::new(false);
928                move |tags: Vec<String>| {
929                    if !reported.swap(true, std::sync::atomic::Ordering::Relaxed) {
930                        engine.report_unknown_display_blocks(session, &tool, part, tags);
931                    }
932                }
933            };
934            let report_unknown_progress = unknown_reporter("progress");
935            let report_unknown_result = unknown_reporter("result");
936            let invoke = std::panic::AssertUnwindSafe(
937                executor.execute(ToolExecutionContext {
938                    session: active.session,
939                    run,
940                    progress: match &capture {
941                        Some(capture)
942                            if result_truncation != crate::ToolResultTruncationPolicy::OptOut =>
943                        {
944                            capture.progress_sink(progress_tx.clone(), hub.clone())
945                        }
946                        _ => ProgressSink::new(progress_tx.clone()),
947                    }
948                    .with_unknown_block_reporter(Arc::new(report_unknown_progress)),
949                    cancellation: tool_cancellation.clone(),
950                    turn_context,
951                    artifacts: engine.inner.artifacts.clone(),
952                    detach: detach_point,
953                    background: background == Some(true),
954                    producers: None,
955                }),
956            )
957            .catch_unwind()
958            .map(move |result| {
959                let mut result =
960                    result.unwrap_or_else(|_| Err(ToolError::execution("tool execution panicked")));
961                if let Ok(completion) = &mut result {
962                    let unknown =
963                        cookie_agent_protocol::take_unknown_blocks(&mut completion.result.display);
964                    if !unknown.is_empty() {
965                        report_unknown_result(unknown);
966                    }
967                }
968                result
969            });
970            let mut invoke: std::pin::Pin<
971                Box<
972                    dyn std::future::Future<Output = Result<crate::ToolCompletion, ToolError>>
973                        + Send,
974                >,
975            > = Box::pin(invoke);
976            // The model asks for the background up front; the user may ask
977            // while the call runs.
978            let mut detach_by = (background == Some(true))
979                .then_some(super::background_tasks::BackgroundActor::Model);
980            let detach_request = background.map(|_| {
981                let request = Arc::new(tokio::sync::Notify::new());
982                active
983                    .detach_requests
984                    .lock()
985                    .unwrap_or_else(|poisoned| poisoned.into_inner())
986                    .insert(call.id, Arc::clone(&request));
987                request
988            });
989            let _detach_registration = DetachRegistration {
990                active: &active,
991                call: call.id,
992            };
993            let started_at = jiff::Timestamp::now();
994            loop {
995                tokio::select! {
996                    () = async {
997                        match &detach_request {
998                            Some(request) => request.notified().await,
999                            None => std::future::pending().await,
1000                        }
1001                    }, if detach_by.is_none() => {
1002                        detach_by = Some(super::background_tasks::BackgroundActor::User);
1003                    }
1004                    detachable = detach_watch.reached(), if detach_by.is_some() => {
1005                        while let Ok(progress) = progress_rx.try_recv() {
1006                            publish_tool_progress(&engine, active.session, run, &progress);
1007                        }
1008                        // The call's live display ends here; its capture goes on.
1009                        hub.finalize();
1010                        engine.retain_finalized_output_hub(call.id);
1011                        let by = detach_by.expect("detaching on request");
1012                        if detachable.reports_itself {
1013                            // The work reports itself (a subagent): stop
1014                            // waiting; dropping the execution releases it.
1015                            drop(invoke);
1016                            return Ok(super::background_tasks::self_reporting_result(
1017                                by,
1018                                detachable.note.as_deref(),
1019                            ));
1020                        }
1021                        return engine
1022                            .detach_call(super::background_tasks::DetachedCall {
1023                                session: active.session,
1024                                tool_call_id: call.id,
1025                                tool: call.name.clone(),
1026                                runtime: runtime.clone(),
1027                                display: presentation
1028                                    .primary_argument
1029                                    .as_ref()
1030                                    .map(|display| display.as_str().to_owned()),
1031                                by,
1032                                started_at,
1033                                note: detachable.note,
1034                                execution: invoke,
1035                                progress: progress_rx,
1036                                capture,
1037                                truncation: result_truncation,
1038                                cancellation: tool_cancellation,
1039                            })
1040                            .await
1041                            .map_err(ToolFailure::from);
1042                    }
1043                    result = &mut invoke => {
1044                        while let Ok(progress) = progress_rx.try_recv() {
1045                            publish_tool_progress(&engine, active.session, run, &progress);
1046                        }
1047                        // Tool implementations drain their producers before
1048                        // resolving.  Finalizing here makes all emitted deltas
1049                        // precede the completion notification committed by the
1050                        // session actor.
1051                        hub.finalize();
1052                        engine.retain_finalized_output_hub(call.id);
1053                        return match result {
1054                            Ok(result) => {
1055                                let failed = result.failed;
1056                                let result = match finalize_output(capture.as_ref(), result, failed, result_truncation).await {
1057                                    Ok(result) => result,
1058                                    Err(error) => {
1059                                        let mut failure = ToolFailure::from(error);
1060                                        failure.partial_output = finalize_output(capture.as_ref(), incomplete_completion(&failure.message), true, result_truncation).await.ok().map(Box::new);
1061                                        return Err(failure);
1062                                    }
1063                                };
1064                                if failed {
1065                                    let message = cookie_agent_protocol::diagnostics::tool_result(&result);
1066                                    Err(ToolFailure { code: ToolCallFailureCode::ExecutionFailed, message, partial_output: Some(Box::new(result)) })
1067                                } else { Ok(result) }
1068                            }
1069                            Err(error) => {
1070                                let mut failure = ToolFailure::from(error);
1071                                failure.partial_output = finalize_output(capture.as_ref(), incomplete_completion(&failure.message), true, result_truncation).await.ok().map(Box::new);
1072                                Err(failure)
1073                            }
1074                        };
1075                    }
1076                    Some(progress) = progress_rx.recv() => {
1077                        publish_tool_progress(&engine, active.session, run, &progress);
1078                    }
1079                    _ = active.cancellation.cancelled() => {
1080                        // Work that reports itself outlives the run that
1081                        // waited for it: stop waiting instead of cancelling.
1082                        let mut self_reporting = detach_watch.current().filter(|detachable| detachable.reports_itself);
1083                        let mut cleanup_timed_out = false;
1084                        let mut delegate_result = None;
1085                        if self_reporting.is_none() {
1086                            tool_cancellation.cancel();
1087                            let cleanup_deadline =
1088                                tokio::time::Instant::now() + TOOL_CANCELLATION_CLEANUP_TIMEOUT;
1089                            let cleanup = tokio::time::sleep_until(cleanup_deadline);
1090                            tokio::pin!(cleanup);
1091                            // A call may reach its detach point while it winds
1092                            // down: a subagent whose start was still being
1093                            // confirmed when the run was cancelled. It then
1094                            // waits on its child, which only its own turn
1095                            // ends, so stop waiting there as above rather
1096                            // than at the deadline, which loses the handle.
1097                            let mut watch_detach = true;
1098                            loop {
1099                                tokio::select! {
1100                                    detachable = detach_watch.reached(), if watch_detach => {
1101                                        if detachable.reports_itself {
1102                                            self_reporting = Some(detachable);
1103                                            break;
1104                                        }
1105                                        watch_detach = false;
1106                                    }
1107                                    result = &mut invoke => {
1108                                        delegate_result = result.ok().filter(|result| {
1109                                            engine.is_delegate_call_result(active.session, run, call.id, &result.result)
1110                                        });
1111                                        while let Ok(progress) = progress_rx.try_recv() {
1112                                            publish_tool_progress(&engine, active.session, run, &progress);
1113                                        }
1114                                        break;
1115                                    }
1116                                    Some(progress) = progress_rx.recv() => {
1117                                        publish_tool_progress(&engine, active.session, run, &progress);
1118                                    }
1119                                    () = &mut cleanup => {
1120                                        cleanup_timed_out = true;
1121                                        progress_rx.close();
1122                                        break;
1123                                    }
1124                                }
1125                            }
1126                        }
1127                        // Drop unsubmitted I/O admission waits before finalization. Jobs
1128                        // already submitted own their permits and are drained by capture.
1129                        drop(invoke);
1130                        hub.finalize();
1131                        engine.retain_finalized_output_hub(call.id);
1132                        let cause = active.cancel_cause();
1133                        let user_interrupt = cause == Some(super::CancelCause::User);
1134                        if let Some(detachable) = self_reporting {
1135                            let message = match detachable.on_user_interrupt.filter(|_| user_interrupt) {
1136                                Some(message) => message,
1137                                None => {
1138                                    let mut message = interrupted_message(cause.as_ref(), false);
1139                                    message.push_str("\nIts work keeps running and reports itself.");
1140                                    if let Some(note) = detachable.note.as_deref().map(str::trim).filter(|note| !note.is_empty()) {
1141                                        message.push('\n');
1142                                        message.push_str(note);
1143                                    }
1144                                    message
1145                                }
1146                            };
1147                            let partial_output = finalize_output(capture.as_ref(), incomplete_completion(&message), true, result_truncation).await.ok().map(Box::new);
1148                            return Err(ToolFailure {
1149                                code: ToolCallFailureCode::ExecutionFailed,
1150                                message,
1151                                partial_output,
1152                            });
1153                        }
1154                        if !cleanup_timed_out && let Some(result) = delegate_result {
1155                            return finalize_output(capture.as_ref(), result, true, result_truncation).await.map_err(ToolFailure::from);
1156                        }
1157                        // A call that wound down within the grace may supply
1158                        // its own result for a user interrupt.
1159                        let message = detach_watch
1160                            .current()
1161                            .and_then(|detachable| detachable.on_user_interrupt)
1162                            .filter(|_| user_interrupt && !cleanup_timed_out)
1163                            .unwrap_or_else(|| interrupted_message(cause.as_ref(), cleanup_timed_out));
1164                        let partial_output = finalize_output(capture.as_ref(), incomplete_completion(&message), true, result_truncation).await.ok().map(Box::new);
1165                        return Err(ToolFailure {
1166                            code: ToolCallFailureCode::ExecutionFailed,
1167                            message,
1168                            partial_output,
1169                        });
1170                    }
1171                }
1172            }
1173        }
1174    }

1186    /// The tool set of `session` as composed now from the sources `keep`
1187    /// admits. Composition diagnostics are reported once each.
1188    pub(crate) fn compose_session_tools(
1189        &self,
1190        session: SessionId,
1191        mut keep: impl FnMut(&SourcedProvider, &crate::ToolSpec) -> bool,
1192    ) -> Result<Composition, EngineError> {
1193        let context = SessionToolContext::new(session);
1194        let mut candidates = Vec::new();
1195        for source in self.session_tool_sources(session)? {
1196            for tool in source
1197                .provider
1198                .tools_for_session(&context)
1199                .map_err(|error| EngineError::ToolFailed(error.to_string()))?
1200            {
1201                if keep(&source, &tool) {
1202                    candidates.push((source.clone(), tool));
1203                }
1204            }
1205        }
1206        // A round's runtime tools keep their published names until it ends
1207        // (stable prefix spec, section 5).
1208        let pins = self.inner.store.get(session)?.log.round_pins();
1209        let composition = compose(candidates, crate::base_tools::reserved_tool_names(), &pins)
1210            .map_err(|name| {
1211                EngineError::ToolFailed(format!(
1212                    "tool name `{name}` is published twice by one runtime"
1213                ))
1214            })?;
1215        self.inner.runtimes.report(&composition.diagnostics);
1216        Ok(composition)
1217    }

1229
1230    /// [`Self::compose_for_policy`], keeping the loader tools when
1231    /// `keep_loaders`.
1232    fn compose_for_policy_with(
1233        &self,
1234        session: SessionId,
1235        policy: &FrozenRunPolicy,
1236        keep_loaders: bool,
1237    ) -> Result<Composition, EngineError> {
1238        let session_projection = self.inner.store.get(session)?;
1239        let depth = session_depth(&session_projection.meta.origin);
1240        let delegate_enabled = !policy.delegate_targets(depth).is_empty();
1241        // The memory tools follow the snapshot: none at all without a
1242        // `memory` rule, whatever `memory_write` rules say, and the write
1243        // tools only when some memory is both readable and writable.
1244        let memory = super::memory_runs::memory_access(
1245            true,
1246            &policy.agent,
1247            Some(&session_projection.permission_overlay),
1248        );
1249        // One merged rule list per home the runtimes reported, not one per
1250        // tool.
1251        let mut merged_by_home = Vec::<(Option<String>, MergedRules<'_>)>::new();
1252        let mut composition = self.compose_session_tools(session, |source, tool| {
1253            let delegation_tool = tool.permission_name.as_deref() == Some("delegate");
1254            let goal_tool = source.attachment.is_none()
1255                && matches!(tool.name.as_str(), "goal_get" | "goal_set" | "goal_update");
1256            let memory_tool = source.provider.provider_id() == crate::memory_tools::PROVIDER_ID;
1257            let memory_write_tool =
1258                memory_tool && tool.permission_name.as_deref() == Some("memory_write");
1259            (!delegation_tool || delegate_enabled)
1260                && (!goal_tool || policy.goal_tools_enabled)
1261                && (!memory_tool || memory.available)
1262                && (!memory_write_tool || memory.writable)
1263                && tool
1264                    .permission_name
1265                    .as_deref()
1266                    .is_none_or(|permission_name| {
1267                        let scope = PermissionScope::of(&source.origin);
1268                        let index = merged_by_home
1269                            .iter()
1270                            .position(|(home, _)| home.as_deref() == scope.home)
1271                            .unwrap_or_else(|| {
1272                                merged_by_home.push((
1273                                    scope.home.map(str::to_owned),
1274                                    MergedRules::of(
1275                                        &policy.agent,
1276                                        Some(&session_projection.permission_overlay),
1277                                        scope.home,
1278                                    ),
1279                                ));
1280                                merged_by_home.len() - 1
1281                            });
1282                        PermissionPipeline::tool_offered_by(
1283                            &merged_by_home[index].1,
1284                            None,
1285                            permission_name,
1286                            &tool.name,
1287                            scope,
1288                        )
1289                    })
1290        })?;
1291        // `cron_list` comes with the other cron tools, and `list_agents` with
1292        // them or with `delegate_subagent` (cron spec section 8.1).
1293        let from_global = |tool: &ComposedTool, provider: &str| {
1294            tool.source.attachment.is_some() && tool.source.provider.provider_id() == provider
1295        };
1296        let offers_cron = composition.tools.iter().any(|tool| {
1297            from_global(tool, crate::cron_tools::PROVIDER_ID) && tool.spec.permission_name.is_some()
1298        });
1299        let offers_delegate = composition
1300            .tools
1301            .iter()
1302            .any(|tool| tool.source.attachment.is_none() && tool.spec.name == "delegate_subagent");
1303        composition.tools.retain(|tool| {
1304            (offers_cron || !from_global(tool, crate::cron_tools::PROVIDER_ID))
1305                && (offers_cron
1306                    || offers_delegate
1307                    || !from_global(tool, crate::agents_tool::PROVIDER_ID))
1308        });
1309        if !keep_loaders {
1310            composition.tools.retain(|tool| {
1311                tool.source.provider.provider_id() != crate::base_tools::LOADER_PROVIDER_ID
1312            });
1313        }
1314        // `stop_background` only makes sense next to a tool that can run in
1315        // the background.
1316        if !composition.tools.iter().any(|tool| tool.spec.background) {
1317            composition.tools.retain(|tool| {
1318                tool.source.attachment.is_some()
1319                    || tool.spec.name != crate::base_tools::STOP_BACKGROUND_TOOL
1320            });
1321        }
1322        Ok(composition)
1323    }

1325    /// `tools` as they are published: a runtime attachment's tools are
1326    /// routed by `machine` once the session can work on several machines
1327    /// (machines on demand, section 5), a tool all of whose machines are
1328    /// one included, so the schema stays put as machines come and go.
1329    /// Returns each published tool with its routes, and the routed names.
1330    pub(super) fn route_tools(
1331        &self,
1332        session: SessionId,
1333        tools: Vec<ComposedTool>,
1334    ) -> Result<(Vec<RoutedEntry>, Vec<String>), EngineError> {
1335        let machines = self.routing_machines(session)?;
1336        let (routed, composed): (Vec<_>, Vec<_>) = tools.into_iter().partition(|tool| {
1337            machines.as_ref().is_some_and(|machines| {
1338                tool.source
1339                    .attachment
1340                    .as_ref()
1341                    .is_some_and(|attachment| machines.contains_key(attachment))
1342            })
1343        });
1344        let taken = composed
1345            .iter()
1346            .map(|tool| tool.name.clone())
1347            .collect::<HashSet<_>>();
1348        let mut groups = Vec::<(String, Vec<ComposedTool>)>::new();
1349        for tool in routed {
1350            match groups.iter_mut().find(|(name, _)| *name == tool.spec.name) {
1351                Some((_, members)) => members.push(tool),
1352                None => groups.push((tool.spec.name.clone(), vec![tool])),
1353            }
1354        }
1355        let mut routed_names = Vec::new();
1356        let mut entries = composed
1357            .into_iter()
1358            .map(|tool| (tool, None))
1359            .collect::<Vec<_>>();
1360        for (name, members) in groups {
1361            if taken.contains(&name) {
1362                self.inner.runtimes.report(&[format!(
1363                    "tool `{name}` of runtime attachment `{}` is not published: the base layer or `global` has a tool of that name",
1364                    members[0].source.attachment.as_ref().map(ToString::to_string).unwrap_or_default()
1365                )]);
1366                continue;
1367            }
1368            let machines = machines.as_ref().expect("routing machines");
1369            let mut routes = Vec::<(String, PublishedTool)>::new();
1370            for member in members {
1371                // `machine` is the host's, as `background` is.
1372                if member
1373                    .spec
1374                    .parameters
1375                    .pointer("/properties/machine")
1376                    .is_some()
1377                {
1378                    self.inner.runtimes.report(&[format!(
1379                        "tool `{name}` of runtime attachment `{}` is not published: it declares its own `machine` argument",
1380                        member.source.attachment.as_ref().map(ToString::to_string).unwrap_or_default()
1381                    )]);
1382                    continue;
1383                }
1384                let machine = member
1385                    .source
1386                    .attachment
1387                    .as_ref()
1388                    .and_then(|attachment| machines.get(attachment))
1389                    .cloned()
1390                    .unwrap_or_default();
1391                if routes.iter().any(|(known, _)| *known == machine) {
1392                    self.inner.runtimes.report(&[format!(
1393                        "tool `{name}` of runtime attachment `{}` is left out: another attachment on machine `{machine}` offers it",
1394                        member.source.attachment.as_ref().map(ToString::to_string).unwrap_or_default()
1395                    )]);
1396                    continue;
1397                }
1398                routes.push((
1399                    machine,
1400                    PublishedTool {
1401                        composed: member.name.clone(),
1402                        source: member.source,
1403                        spec: member.spec,
1404                        routes: None,
1405                    },
1406                ));
1407            }
1408            let Some((default, first)) = routes.first() else {
1409                continue;
1410            };
1411            let routed = ComposedTool {
1412                name: name.clone(),
1413                source: first.source.clone(),
1414                spec: first.spec.clone(),
1415            };
1416            let default = default.clone();
1417            routed_names.push(name);
1418            entries.push((
1419                routed,
1420                Some(ToolRoutes {
1421                    default: Some(default),
1422                    machines: routes,
1423                }),
1424            ));
1425        }
1426        Ok((entries, routed_names))
1427    }

1429    pub(super) fn published_tool_definitions(
1430        &self,
1431        session: SessionId,
1432        policy: &FrozenRunPolicy,
1433    ) -> Result<PublishedToolSet, EngineError> {
1434        let defers = self.defers_tools(policy);
1435        let composition = self.compose_for_policy_with(session, policy, defers)?;
1436        let notes = self.runtime_tool_notes(session);
1437        let config = self.inner.config.load();
1438        let mut output = Vec::with_capacity(composition.tools.len());
1439        let mut published = std::collections::HashMap::with_capacity(composition.tools.len());
1440        let mut candidates = Vec::new();
1441        let mut loaders = Vec::new();
1442        let (entries, routed_names) = self.route_tools(session, composition.tools.clone())?;
1443        for (ComposedTool { name, source, spec }, routes) in entries {
1444            spec.output.validate().map_err(EngineError::ToolFailed)?;
1445            let mut parameters = spec.parameters.clone();
1446            if let Some(routes) = &routes {
1447                add_machine_argument(&mut parameters, routes.default.as_deref());
1448            }
1449            if spec.background {
1450                add_background_argument(&mut parameters, spec.background_default);
1451            }
1452            let schema = JsonSchema::new(parameters).map_err(|error| {
1453                EngineError::ToolFailed(format!("tool `{name}` has invalid JSON Schema: {error}"))
1454            })?;
1455            let description = match source
1456                .attachment
1457                .as_ref()
1458                .filter(|_| routes.is_none())
1459                .and_then(|name| notes.get(name))
1460            {
1461                Some(note) if spec.description.is_empty() => note.clone(),
1462                Some(note) => format!("{}\n\n{note}", spec.description),
1463                None => spec.description.clone(),
1464            };
1465            let definition = ToolDefinition::new(name.clone(), description, schema);
1466            // A routed tool's name is the tool's own: nothing to pin.
1467            let round_source = crate::runtimes::composition_source(&source)
1468                .filter(|_| routes.is_none())
1469                .map(|label| cookie_agent_protocol::RoundToolSource {
1470                    label,
1471                    tool: spec.name.clone(),
1472                });
1473            if defers {
1474                if source.provider.provider_id() == crate::base_tools::LOADER_PROVIDER_ID {
1475                    loaders.push(definition);
1476                } else {
1477                    candidates.push(crate::deferred_tools::Candidate {
1478                        definition,
1479                        group: crate::deferred_tools::group_of(&source, &spec.name),
1480                        provider: provider_key(&source),
1481                        deferred: crate::deferred_tools::policy_defers(
1482                            &source,
1483                            &spec.name,
1484                            &config.runtime.deferred_tools,
1485                        ),
1486                        source: round_source,
1487                    });
1488                }
1489            } else {
1490                output.push(crate::deferred_tools::EagerCandidate {
1491                    definition,
1492                    provider: provider_key(&source),
1493                    source: round_source,
1494                });
1495            }
1496            published.insert(
1497                name.clone(),
1498                PublishedTool {
1499                    composed: name,
1500                    source,
1501                    spec,
1502                    routes,
1503                },
1504            );
1505        }
1506        let fold = self.inner.store.get(session)?.log.deferral();
1507        if !defers {
1508            let plan = crate::deferred_tools::plan_eager(output, &fold);
1509            declared_defaults(&mut published, &plan.definitions, &routed_names);
1510            return Ok(PublishedToolSet {
1511                definitions: plan.definitions,
1512                tools: published,
1513                departed: plan.departed,
1514                unloaded: HashSet::new(),
1515                pending: plan.pending,
1516                composition,
1517            });
1518        }
1519        let previous = self.deferral_view(session);
1520        let mut plan = crate::deferred_tools::plan_reusing(candidates, &loaders, &fold, &previous);
1521        let unloaded = plan
1522            .view
1523            .deferred
1524            .iter()
1525            .filter(|entry| !fold.loaded.contains(&entry.name))
1526            .map(|entry| entry.name.clone())
1527            .collect();
1528        for provider in &plan.deferred_providers {
1529            // Delivered with the provider's first load; a provider with a
1530            // loaded tool has had them.
1531            if plan
1532                .view
1533                .deferred
1534                .iter()
1535                .any(|entry| &entry.provider == provider && fold.loaded.contains(&entry.name))
1536            {
1537                continue;
1538            }
1539            if let Some(block) =
1540                self.tool_prompt_block(policy, session, None, &|id, attachment| {
1541                    id == provider.0 && attachment == provider.1.as_deref()
1542                })?
1543            {
1544                plan.view.sections.insert(provider.clone(), block);
1545            }
1546        }
1547        declared_defaults(&mut published, &plan.definitions, &routed_names);
1548        // A loader tool is callable only where the request declares it.
1549        let declared = plan
1550            .definitions
1551            .iter()
1552            .map(|definition| definition.name.as_str())
1553            .collect::<HashSet<_>>();
1554        published.retain(|name, _| {
1555            !crate::deferred_tools::is_loader(name) || declared.contains(name.as_str())
1556        });
1557        self.inner
1558            .deferral_views
1559            .lock()
1560            .unwrap_or_else(|poisoned| poisoned.into_inner())
1561            .insert(session, Arc::new(plan.view));
1562        Ok(PublishedToolSet {
1563            definitions: plan.definitions,
1564            tools: published,
1565            departed: plan.departed,
1566            unloaded,
1567            pending: plan.pending,
1568            composition,
1569        })
1570    }

1572    /// The providers whose prompt sections the system prompt of a run of
1573    /// `policy` leaves out: those whose every tool is deferred.
1574    ///
1575    /// With `composition`, the tool set `policy` composes now, the caller
1576    /// has composed it; loader tools in it are left out.
1577    pub(crate) fn deferred_prompt_providers(
1578        &self,
1579        session: SessionId,
1580        policy: &FrozenRunPolicy,
1581        composition: Option<&Composition>,
1582    ) -> Result<HashSet<crate::deferred_tools::ProviderKey>, EngineError> {
1583        if !self.defers_tools(policy) {
1584            return Ok(HashSet::new());
1585        }
1586        let owned;
1587        let composition = match composition {
1588            Some(composition) => composition,
1589            None => {
1590                owned = self.compose_for_policy(session, policy)?;
1591                &owned
1592            }
1593        };
1594        let tools = || {
1595            composition.tools.iter().filter(|tool| {
1596                tool.source.provider.provider_id() != crate::base_tools::LOADER_PROVIDER_ID
1597            })
1598        };
1599        let fold = self.inner.store.get(session)?.log.deferral();
1600        let config = self.inner.config.load();
1601        let keys = tools()
1602            .map(|tool| (tool.name.as_str(), provider_key(&tool.source)))
1603            .collect::<Vec<_>>();
1604        let eager = match &fold.round {
1605            Some(round) => round.tools.iter().map(|tool| tool.name.clone()).collect(),
1606            None => tools()
1607                .filter(|tool| {
1608                    !crate::deferred_tools::policy_defers(
1609                        &tool.source,
1610                        &tool.spec.name,
1611                        &config.runtime.deferred_tools,
1612                    ) || fold.folded.contains(&tool.name)
1613                })
1614                .map(|tool| tool.name.clone())
1615                .collect(),
1616        };
1617        Ok(crate::deferred_tools::deferred_providers(
1618            keys.iter().map(|(name, key)| (*name, key)),
1619            &eager,
1620        )
1621        .into_iter()
1622        .collect())
1623    }

1922pub(super) fn fallback_operation_fingerprint(
1923    call: &ToolCall,
1924    permission_name: Option<&str>,
1925) -> OperationFingerprint {
1926    let action = permission_name
1927        .or_else(|| fallback_permission_name(&call.name))
1928        .and_then(|name| PermissionPipeline::action_for_permission_name(name).ok())
1929        .unwrap_or(cookie_agent_protocol::PermissionAction::READ);
1930    let binding = serde_json::to_vec(&call.arguments).expect("tool arguments serialize");
1931    let operation = PreparedOperationIdentity::new(
1932        Sha256Digest::of_bytes(&binding),
1933        vec![cookie_agent_protocol::ApprovalCapability {
1934            action: action.clone(),
1935            operation: cookie_agent_protocol::PreparedCapabilityOperation::new(format!(
1936                "{}:prepare",
1937                call.name
1938                    .replace(|character: char| !character.is_ascii_alphanumeric(), "_")
1939            ))
1940            .unwrap_or_else(|_| {
1941                cookie_agent_protocol::PreparedCapabilityOperation::new("tool:prepare")
1942                    .expect("static operation")
1943            }),
1944        }],
1945        vec![cookie_agent_protocol::PreparedApprovalResource {
1946            capability: action,
1947            canonical: cookie_agent_protocol::PreparedResourceIdentity::new("tool:prepare-failed")
1948                .expect("static resource identity"),
1949            binding_digest:
1950                cookie_agent_protocol::PreparedResourceDigest::from_canonical_binding_bytes(
1951                    &binding,
1952                ),
1953            binding_lifetime: cookie_agent_protocol::PreparedBindingLifetime::ProcessLocal,
1954            boundary: cookie_agent_protocol::ApprovalBoundary::Exact,
1955            source: cookie_agent_protocol::ApprovalResourceSource::PrimaryOperation,
1956        }],
1957        Sha256Digest::of_bytes(b"prepare-failed"),
1958    )
1959    .expect("fallback prepared identity");
1960    OperationFingerprint::from_prepared_operation(&operation)
1961}

1975/// Applies the permission resource of a prepared call's normalized
1976/// arguments, and returns it. A provider that reported its metadata at
1977/// preparation is not asked again. A call its provider only declared gets
1978/// the approval identity the daemon builds for `tool`, prepared under
1979/// `origin`.
1980pub(crate) fn apply_permission_resource(
1981    provider: &dyn ToolProvider,
1982    name: &str,
1983    expected_permission_name: Option<&str>,
1984    tool: &ToolIdentity,
1985    origin: &crate::ToolOrigin,
1986    prepared: PreparedTool,
1987) -> Result<(PreparedTool, Option<String>), ToolError> {
1988    let (permission_name, resource) = match prepared.declaration() {
1989        Some(declaration) => (
1990            declaration
1991                .permission_name
1992                .clone()
1993                .map(std::borrow::Cow::Owned),
1994            declaration.permission_resource.clone(),
1995        ),
1996        None => {
1997            let (permission_name, resource) =
1998                provider.get_permission_resource(name, prepared.normalized_arguments())?;
1999            (permission_name.map(std::borrow::Cow::Borrowed), resource)
2000        }
2001    };
2002    let permission_name = permission_name.as_deref();
2003    let matches = match (permission_name, expected_permission_name) {
2004        (Some(actual), Some(expected)) => actual == expected,
2005        (None, None) => true,
2006        _ => false,
2007    };
2008    if !matches {
2009        return Err(ToolError::execution(
2010            "tool permission metadata changed between discovery and preparation",
2011        ));
2012    }
2013    let action = permission_name
2014        .map(|permission_name| {
2015            PermissionPipeline::action_for_permission_name(permission_name)
2016                .map_err(|error| ToolError::execution(error.to_string()))
2017        })
2018        .transpose()?;
2019    if prepared.is_declared() {
2020        let operation = crate::permissions::identity::declared_operation(
2021            tool,
2022            &origin.context_bytes(),
2023            action.as_ref(),
2024            resource.as_deref(),
2025            prepared.normalized_arguments(),
2026        )?;
2027        return Ok((prepared.identify(operation, resource.clone())?, resource));
2028    }
2029    let Some(action) = action else {
2030        // A tool that needs no permission evaluates no rule, so the prepared
2031        // capability is not checked against an action.
2032        return match resource {
2033            Some(resource) => Ok((
2034                prepared.with_permission_resource(Some(resource.clone()))?,
2035                Some(resource),
2036            )),
2037            None => Ok((prepared, None)),
2038        };
2039    };
2040    let resources = prepared.operation().resources();
2041    // A call whose resources are each checked in their own scope states
2042    // each one's action, such as `transfer`'s `read` of its source; one of
2043    // them is the tool's own.
2044    let mismatched = if prepared.resource_scopes().is_empty() {
2045        resources
2046            .iter()
2047            .any(|prepared| prepared.capability != action)
2048    } else {
2049        !resources
2050            .iter()
2051            .any(|prepared| prepared.capability == action)
2052    };
2053    if mismatched {
2054        return Err(ToolError::execution(
2055            "prepared permission capability does not match provider metadata",
2056        ));
2057    }
2058    // A provider may return None when it manages policy labels while preparing
2059    // the operation. Providers that need the loose marker set it explicitly.
2060    match resource {
2061        Some(resource) => Ok((
2062            prepared.with_permission_resource(Some(resource.clone()))?,
2063            Some(resource),
2064        )),
2065        None => Ok((prepared, None)),
2066    }
2067}

2069/// The strictest of `decisions` over one operation, resource by resource:
2070/// deny over ask over allow. Allowed when there are none.
2071pub(crate) fn strictest_decision(
2072    decisions: impl IntoIterator<Item = PermissionDecision>,
2073) -> PermissionDecision {
2074    let mut decisions = decisions.into_iter();
2075    let Some(mut decision) = decisions.next() else {
2076        return PermissionDecision::permission_free();
2077    };
2078    for other in decisions {
2079        for (evaluation, candidate) in decision.evaluations.iter_mut().zip(other.evaluations) {
2080            if effect_rank(candidate.effect) > effect_rank(evaluation.effect) {
2081                *evaluation = candidate;
2082            }
2083        }
2084    }
2085    decision.effect = decision
2086        .evaluations
2087        .iter()
2088        .map(|evaluation| evaluation.effect)
2089        .max_by_key(|effect| effect_rank(*effect))
2090        .unwrap_or(cookie_agent_protocol::PermissionEffect::Allow);
2091    decision
2092}
2093
2094/// The decision of a call whose resources were each checked in their own
2095/// scope: `decisions[i]` decided the whole call in resource `i`'s scope, and
2096/// only its evaluation of resource `i` counts.
2097fn per_resource_decision(decisions: Vec<PermissionDecision>) -> PermissionDecision {
2098    let mut decisions = decisions.into_iter().enumerate();
2099    let Some((_, mut decision)) = decisions.next() else {
2100        return PermissionDecision::permission_free();
2101    };
2102    for (index, other) in decisions {
2103        if let (Some(evaluation), Some(candidate)) = (
2104            decision.evaluations.get_mut(index),
2105            other.evaluations.into_iter().nth(index),
2106        ) {
2107            *evaluation = candidate;
2108        }
2109    }
2110    decision.effect = decision
2111        .evaluations
2112        .iter()
2113        .map(|evaluation| evaluation.effect)
2114        .max_by_key(|effect| effect_rank(*effect))
2115        .unwrap_or(cookie_agent_protocol::PermissionEffect::Allow);
2116    decision
2117}
2118
2119/// How strict a permission effect is: deny over ask over allow.
2120const fn effect_rank(effect: cookie_agent_protocol::PermissionEffect) -> u8 {
2121    match effect {
2122        cookie_agent_protocol::PermissionEffect::Allow => 0,
2123        cookie_agent_protocol::PermissionEffect::Ask => 1,
2124        cookie_agent_protocol::PermissionEffect::Deny => 2,
2125    }
2126}
