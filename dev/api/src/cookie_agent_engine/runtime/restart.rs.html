1//! Restarting the daemon from a client
2//! (`docs/site/specs/engine-configuration.md` section 11).
3//!
4//! The engine schedules the restart and decides when it is due; the program
5//! that runs the engine as a daemon does the restart itself, through the
6//! hook it installs with [`Engine::set_restart_hook`]. An engine without one
7//! refuses with [`EngineError::RestartUnsupported`].
8//!
9//! While an `idle` restart waits, new top-level runs are refused with
10//! [`EngineError::Restarting`]: client starts, queued input, producer wakes,
11//! and cron fires (recorded as skipped). A run's delegated children still
12//! start, since they are part of a turn the restart waits for.
13
14use std::{
15    sync::{
16        Mutex,
17        atomic::{AtomicBool, AtomicUsize, Ordering},
18    },
19    time::Duration,
20};
21
22use cookie_agent_protocol::RestartWhen;
23use tokio::sync::{Notify, watch};
24
25use super::{Engine, EngineError};
26
27/// How often a waiting restart counts the active runs.
28const IDLE_POLL: Duration = Duration::from_millis(200);
29
30/// What the program running the engine does once a scheduled restart is
31/// due: stop the daemon and relaunch it. Called at most once.
32pub type RestartHook = Box<dyn FnOnce() + Send>;
33
34/// A scheduled restart, as clients are told about it.
35#[derive(Clone, Copy, Debug, Eq, PartialEq)]
36pub struct RestartStatus {
37    pub when: RestartWhen,
38    /// The runs still active, which an `idle` restart waits for.
39    pub active_runs: u32,
40    /// The restart is due and the daemon is going down.
41    pub due: bool,
42}
43
44/// What [`Engine::schedule_restart`] did.
45#[derive(Clone, Copy, Debug, Eq, PartialEq)]
46pub struct RestartScheduled {
47    /// `false` when a restart was already scheduled and only its `when`
48    /// changed.
49    pub scheduled: bool,
50    pub active_runs: u32,
51}
52
53#[derive(Default)]
54pub(crate) struct RestartState {
55    supported: AtomicBool,
56    hook: Mutex<Option<RestartHook>>,
57    /// Set once a restart is scheduled; never cleared, since the daemon goes
58    /// down.
59    pending: AtomicBool,
60    when: Mutex<Option<RestartWhen>>,
61    /// Run starts in progress, which a waiting restart counts as active.
62    starting: AtomicUsize,
63    status: watch::Sender<Option<RestartStatus>>,
64    wake: Notify,
65}
66
67/// Counts a run start in progress until dropped.
68pub(super) struct RunStart<'a>(&'a AtomicUsize);
69
70impl Drop for RunStart<'_> {
71    fn drop(&mut self) {
72        self.0.fetch_sub(1, Ordering::SeqCst);
73    }
74}
75
76impl Engine {
77    /// Lets clients restart the daemon this engine runs in: `hook` is
78    /// called once a scheduled restart is due, and must stop the daemon
79    /// (shutting this engine down) and relaunch it.
80    pub fn set_restart_hook(&self, hook: RestartHook) {
81        let restart = &self.inner.restart;
82        *restart
83            .hook
84            .lock()
85            .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(hook);
86        restart.supported.store(true, Ordering::Release);
87    }
88
89    /// Whether a restart is scheduled.
90    #[must_use]
91    pub fn restart_pending(&self) -> bool {
92        self.inner.restart.pending.load(Ordering::SeqCst)
93    }
94
95    /// The scheduled restart, `None` until one is; it changes as the active
96    /// runs do and once the restart is due.
97    #[must_use]
98    pub fn subscribe_restart(&self) -> watch::Receiver<Option<RestartStatus>> {
99        self.inner.restart.status.subscribe()
100    }
101
102    /// Schedules a restart `when` asked. A second call changes the `when`
103    /// of the scheduled one; `now` wins over `idle`.
104    pub fn schedule_restart(&self, when: RestartWhen) -> Result<RestartScheduled, EngineError> {
105        let restart = &self.inner.restart;
106        if !restart.supported.load(Ordering::Acquire) {
107            return Err(EngineError::RestartUnsupported);
108        }
109        let when = match when {
110            RestartWhen::Now => RestartWhen::Now,
111            RestartWhen::Idle | RestartWhen::Unknown => RestartWhen::Idle,
112        };
113        let scheduled = {
114            let mut current = restart
115                .when
116                .lock()
117                .unwrap_or_else(|poisoned| poisoned.into_inner());
118            let scheduled = current.is_none();
119            if current.is_none_or(|current| current == RestartWhen::Idle) {
120                *current = Some(when);
121            }
122            scheduled
123        };
124        restart.pending.store(true, Ordering::SeqCst);
125        let active_runs = self.restart_active_runs();
126        if scheduled {
127            let engine = self.clone();
128            let task = async move { engine.wait_for_restart().await };
129            match &self.inner.runtime {
130                Some(runtime) => drop(runtime.spawn(task)),
131                None => drop(tokio::spawn(task)),
132            }
133        } else {
134            restart.wake.notify_one();
135        }
136        Ok(RestartScheduled {
137            scheduled,
138            active_runs,
139        })
140    }
141
142    /// Counts a run start, or refuses a top-level one while a restart is
143    /// scheduled. `top_level` is `false` for a delegated child, which is
144    /// part of a turn already running.
145    pub(super) fn begin_run_start(&self, top_level: bool) -> Result<RunStart<'_>, EngineError> {
146        let restart = &self.inner.restart;
147        // Counted before the check: a restart scheduled after it sees this
148        // start among the active runs.
149        restart.starting.fetch_add(1, Ordering::SeqCst);
150        let start = RunStart(&restart.starting);
151        if top_level && restart.pending.load(Ordering::SeqCst) {
152            return Err(EngineError::Restarting);
153        }
154        Ok(start)
155    }
156
157    fn restart_active_runs(&self) -> u32 {
158        let active = self
159            .inner
160            .sessions
161            .active
162            .lock()
163            .unwrap_or_else(|poisoned| poisoned.into_inner())
164            .len();
165        let starting = self.inner.restart.starting.load(Ordering::SeqCst);
166        u32::try_from(active + starting).unwrap_or(u32::MAX)
167    }
168
169    /// Publishes the restart's progress until it is due, then calls the
170    /// hook.
171    async fn wait_for_restart(self) {
172        let restart = &self.inner.restart;
173        loop {
174            let when = restart
175                .when
176                .lock()
177                .unwrap_or_else(|poisoned| poisoned.into_inner())
178                .unwrap_or(RestartWhen::Idle);
179            let active_runs = self.restart_active_runs();
180            let due = when == RestartWhen::Now || active_runs == 0;
181            let status = RestartStatus {
182                when,
183                active_runs,
184                due,
185            };
186            restart.status.send_if_modified(|current| {
187                let changed = *current != Some(status);
188                *current = Some(status);
189                changed
190            });
191            if due {
192                break;
193            }
194            tokio::select! {
195                () = tokio::time::sleep(IDLE_POLL) => {}
196                () = restart.wake.notified() => {}
197            }
198        }
199        let hook = restart
200            .hook
201            .lock()
202            .unwrap_or_else(|poisoned| poisoned.into_inner())
203            .take();
204        if let Some(hook) = hook {
205            hook();
206        }
207    }
