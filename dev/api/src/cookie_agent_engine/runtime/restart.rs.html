1//! Restarting the daemon from a client
2//! (`docs/site/specs/engine-configuration.md` section 11).
3//!
4//! The engine schedules the restart and decides when it is due; the program
5//! that runs the engine as a daemon does the restart itself, through the
6//! hook it installs with [`Engine::set_restart_hook`]. An engine without one
7//! refuses with [`EngineError::RestartUnsupported`].
8//!
9//! An `idle` restart waits for the active runs and the running background
10//! calls, which the shutdown would otherwise end without a report.
11//! While it waits, new top-level runs are refused with
12//! [`EngineError::Restarting`]: client starts, queued input, producer wakes,
13//! and cron fires (recorded as skipped). A run's delegated children still
14//! start, since they are part of a turn the restart waits for.
15
16use std::{
17    sync::{
18        Mutex,
19        atomic::{AtomicBool, AtomicUsize, Ordering},
20    },
21    time::Duration,
22};
23
24use cookie_agent_protocol::RestartWhen;
25use tokio::sync::{Notify, watch};

27use super::{Engine, EngineError};
28
29/// How often a waiting restart counts the active runs.
30const IDLE_POLL: Duration = Duration::from_millis(200);
31
32/// What the program running the engine does once a scheduled restart is
33/// due: stop the daemon and relaunch it. Called at most once.
34pub type RestartHook = Box<dyn FnOnce() + Send>;
35
36/// A scheduled restart, as clients are told about it.
37#[derive(Clone, Copy, Debug, Eq, PartialEq)]
38pub struct RestartStatus {
39    pub when: RestartWhen,
40    /// The runs still active, which an `idle` restart waits for.
41    pub active_runs: u32,
42    /// The background calls still running, which an `idle` restart waits
43    /// for too.
44    pub background_calls: u32,
45    /// The restart is due and the daemon is going down.
46    pub due: bool,
47}
48
49/// What [`Engine::schedule_restart`] did.
50#[derive(Clone, Copy, Debug, Eq, PartialEq)]
51pub struct RestartScheduled {
52    /// `false` when a restart was already scheduled and only its `when`
53    /// changed.
54    pub scheduled: bool,
55    pub active_runs: u32,
56    pub background_calls: u32,
57}
58
59#[derive(Default)]
60pub(crate) struct RestartState {
61    supported: AtomicBool,
62    hook: Mutex<Option<RestartHook>>,
63    /// Set once a restart is scheduled; never cleared, since the daemon goes
64    /// down.
65    pending: AtomicBool,
66    when: Mutex<Option<RestartWhen>>,
67    /// Run starts in progress, which a waiting restart counts as active.
68    starting: AtomicUsize,
69    status: watch::Sender<Option<RestartStatus>>,
70    wake: Notify,
71}
72
73/// Counts a run start in progress until dropped.
74pub(super) struct RunStart<'a>(&'a AtomicUsize);
75
76impl Drop for RunStart<'_> {
77    fn drop(&mut self) {
78        self.0.fetch_sub(1, Ordering::SeqCst);
79    }
80}
81
82impl Engine {
83    /// Lets clients restart the daemon this engine runs in: `hook` is
84    /// called once a scheduled restart is due, and must stop the daemon
85    /// (shutting this engine down) and relaunch it.
86    pub fn set_restart_hook(&self, hook: RestartHook) {
87        let restart = &self.inner.restart;
88        *restart
89            .hook
90            .lock()
91            .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(hook);
92        restart.supported.store(true, Ordering::Release);
93    }
94
95    /// Whether a restart is scheduled.
96    #[must_use]
97    pub fn restart_pending(&self) -> bool {
98        self.inner.restart.pending.load(Ordering::SeqCst)
99    }
100
101    /// The scheduled restart, `None` until one is; it changes as the active
102    /// runs do and once the restart is due.
103    #[must_use]
104    pub fn subscribe_restart(&self) -> watch::Receiver<Option<RestartStatus>> {
105        self.inner.restart.status.subscribe()
106    }
107
108    /// Schedules a restart `when` asked. A second call changes the `when`
109    /// of the scheduled one; `now` wins over `idle`.
110    pub fn schedule_restart(&self, when: RestartWhen) -> Result<RestartScheduled, EngineError> {
111        let restart = &self.inner.restart;
112        if !restart.supported.load(Ordering::Acquire) {
113            return Err(EngineError::RestartUnsupported);
114        }
115        let when = match when {
116            RestartWhen::Now => RestartWhen::Now,
117            RestartWhen::Idle | RestartWhen::Unknown => RestartWhen::Idle,
118        };
119        let scheduled = {
120            let mut current = restart
121                .when
122                .lock()
123                .unwrap_or_else(|poisoned| poisoned.into_inner());
124            let scheduled = current.is_none();
125            if current.is_none_or(|current| current == RestartWhen::Idle) {
126                *current = Some(when);
127            }
128            scheduled
129        };
130        restart.pending.store(true, Ordering::SeqCst);
131        let active_runs = self.restart_active_runs();
132        let background_calls = self.restart_background_calls();
133        if scheduled {
134            let engine = self.clone();
135            let task = async move { engine.wait_for_restart().await };
136            match &self.inner.runtime {
137                Some(runtime) => drop(runtime.spawn(task)),
138                None => drop(tokio::spawn(task)),
139            }
140        } else {
141            restart.wake.notify_one();
142        }
143        Ok(RestartScheduled {
144            scheduled,
145            active_runs,
146            background_calls,
147        })
148    }
149
150    /// Counts a run start, or refuses a top-level one while a restart is
151    /// scheduled. `top_level` is `false` for a delegated child, which is
152    /// part of a turn already running.
153    pub(super) fn begin_run_start(&self, top_level: bool) -> Result<RunStart<'_>, EngineError> {
154        let restart = &self.inner.restart;
155        // Counted before the check: a restart scheduled after it sees this
156        // start among the active runs.
157        restart.starting.fetch_add(1, Ordering::SeqCst);
158        let start = RunStart(&restart.starting);
159        if top_level && restart.pending.load(Ordering::SeqCst) {
160            return Err(EngineError::Restarting);
161        }
162        Ok(start)
163    }
164
165    fn restart_active_runs(&self) -> u32 {
166        let active = self
167            .inner
168            .sessions
169            .active
170            .lock()
171            .unwrap_or_else(|poisoned| poisoned.into_inner())
172            .len();
173        let starting = self.inner.restart.starting.load(Ordering::SeqCst);
174        u32::try_from(active + starting).unwrap_or(u32::MAX)
175    }
176
177    /// The background calls of every session that have not committed their
178    /// result yet.
179    fn restart_background_calls(&self) -> u32 {
180        u32::try_from(self.inner.background.pending_count() + self.acp_background_count())
181            .unwrap_or(u32::MAX)
182    }
183
184    /// Publishes the restart's progress until it is due, then calls the
185    /// hook.
186    async fn wait_for_restart(self) {
187        let restart = &self.inner.restart;
188        loop {
189            let when = restart
190                .when
191                .lock()
192                .unwrap_or_else(|poisoned| poisoned.into_inner())
193                .unwrap_or(RestartWhen::Idle);
194            let active_runs = self.restart_active_runs();
195            let background_calls = self.restart_background_calls();
196            let due = when == RestartWhen::Now || (active_runs == 0 && background_calls == 0);
197            let status = RestartStatus {
198                when,
199                active_runs,
200                background_calls,
201                due,
202            };
203            restart.status.send_if_modified(|current| {
204                let changed = *current != Some(status);
205                *current = Some(status);
206                changed
207            });
208            if due {
209                break;
210            }
211            tokio::select! {
212                () = tokio::time::sleep(IDLE_POLL) => {}
213                () = restart.wake.notified() => {}
214            }
215        }
216        let hook = restart
217            .hook
218            .lock()
219            .unwrap_or_else(|poisoned| poisoned.into_inner())
220            .take();
221        if let Some(hook) = hook {
222            hook();
223        }
224    }
225}
