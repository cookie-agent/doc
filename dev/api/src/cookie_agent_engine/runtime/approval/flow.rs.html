40    /// The tool failure the agent receives for this refused outcome.
41    pub(crate) fn denial_message(self, source: &str, reason: &str) -> String {
42        match self.reviewer_denial {
43            Some(denial) => super::projection::denied_tool_failure(
44                REVIEWER_DENIAL_SOURCE,
45                denial
46                    .reason
47                    .unwrap_or_else(|| REVIEWER_DENIAL_REASON.to_owned()),
48                None,
49            ),
50            None => super::projection::denied_tool_failure(source, reason, self.feedback),
51        }
52    }

128    pub(in crate::runtime) async fn request_model_approval(
129        &self,
130        active: &ActiveRun,
131        run: RunId,
132        input: ModelApprovalInput<'_>,
133    ) -> Result<ApprovalOutcome, EngineError> {
134        let request = approval_request_for_operation(
135            input.tool.tool_call_id,
136            ApprovalTrigger::ModelToolApproval,
137            input.operation.clone(),
138            input
139                .operation
140                .resources()
141                .iter()
142                .zip(input.policy_labels)
143                .map(|(resource, label)| cookie_agent_protocol::DecisionTrace {
144                    action: resource.capability.clone(),
145                    normalized_resource: label
146                        .clone()
147                        .unwrap_or_else(|| UNSCOPED_PERMISSION_RESOURCE_DISPLAY.to_owned()),
148                    candidates: Vec::new(),
149                    effect: cookie_agent_protocol::PermissionEffect::Ask,
150                    precedence_reason: input
151                        .message
152                        .clone()
153                        .unwrap_or_else(|| "model requested tool approval".into()),
154                })
155                .collect(),
156            false,
157            // The user's window, `[approval].timeout_ms`, starts when the
158            // call escalates to them; the reviewer's own `limits.timeout_ms`
159            // bounds only its model call (see `run_internal_history_agent`).
160            None,
161        );
162        self.await_user_approval(active, run, request, input.executor, false, input.tool)
163            .await
164    }

166    pub(in crate::runtime) async fn await_user_approval(
167        &self,
168        active: &ActiveRun,
169        run: RunId,
170        request: ApprovalRequest,
171        executor: PreparedExecutorCell,
172        allow_prior_grant: bool,
173        tool: ApprovalToolInput<'_>,
174    ) -> Result<ApprovalOutcome, EngineError> {
175        // The request, and any tree grant made from it, belongs to the
176        // runtime of the tool it is for.
177        let request = request.with_runtime(
178            tool.origin
179                .and_then(crate::ToolOrigin::runtime)
180                .and_then(|runtime| cookie_agent_protocol::RuntimeId::new(runtime.id()).ok()),
181        );
182        let approval_id = request.approval_id();
183        let session = self.inner.store.get(active.session)?;
184        let root = root_id(&session.meta.origin, active.session);
185        self.append(
186            active.session,
187            Some(run),
188            crate::runtime::event_origin("engine:approvals"),
189            Event::ApprovalRequested {
190                request: request.clone(),
191            },
192        )
193        .await?;
194
195        let repetitions = doom_loop_repetitions(
196            &self.inner.store.log(active.session)?.event_snapshot(),
197            run,
198            request.operation_fingerprint(),
199        );
200        if repetitions >= 4 {
201            self.append(
202                active.session,
203                Some(run),
204                crate::runtime::event_origin("engine:approvals"),
205                Event::approval_step(
206                    approval_id,
207                    cookie_agent_protocol::ApprovalStep::DoomLoop {
208                        operation_fingerprint: request.operation_fingerprint().clone(),
209                        repetitions,
210                    },
211                ),
212            )
213            .await?;
214            self.append(
215                active.session,
216                Some(run),
217                crate::runtime::event_origin("engine:approvals"),
218                Event::approval_finalized(
219                    approval_id,
220                    cookie_agent_protocol::ApprovalFinalization::Decided {
221                        decision: ApprovalFinalDecision {
222                            outcome: ApprovalFinalOutcome::Rejected,
223                            source: ApprovalDecisionSource::DoomLoopGuard,
224                            reason_code: ApprovalReasonCode::DoomLoopDetected,
225                            feedback: None,
226                            tree_grant_id: None,
227                        },
228                    },
229                ),
230            )
231            .await?;
232            return Ok(ApprovalOutcome {
233                approved: false,
234                feedback: None,
235                reviewer_denial: None,
236            });
237        }
238
239        if allow_prior_grant
240            && let Some(grant) =
241                self.inner
242                    .approvals
243                    .store
244                    .matching(root, request.operation(), request.runtime_id())
245        {
246            let decision = ApprovalInternalDecision {
247                decision: ApprovalInternalDecisionKind::Allow,
248                source: ApprovalDecisionSource::TreeGrant,
249                reason_code: ApprovalReasonCode::TreeGrantMatched,
250                evaluations: approval_evaluations(&request),
251                rationale: None,
252            };
253            self.append(
254                active.session,
255                Some(run),
256                crate::runtime::event_origin("engine:approvals"),
257                Event::approval_step(
258                    approval_id,
259                    cookie_agent_protocol::ApprovalStep::Evaluated { decision },
260                ),
261            )
262            .await?;
263            self.append(
264                active.session,
265                Some(run),
266                crate::runtime::event_origin("engine:approvals"),
267                Event::approval_finalized(
268                    approval_id,
269                    cookie_agent_protocol::ApprovalFinalization::Decided {
270                        decision: ApprovalFinalDecision {
271                            outcome: ApprovalFinalOutcome::Approved,
272                            source: ApprovalDecisionSource::TreeGrant,
273                            reason_code: ApprovalReasonCode::TreeGrantMatched,
274                            feedback: None,
275                            tree_grant_id: Some(grant.grant_id),
276                        },
277                    },
278                ),
279            )
280            .await?;
281            return Ok(ApprovalOutcome {
282                approved: true,
283                feedback: None,
284                reviewer_denial: None,
285            });
286        }
287
288        if approval_evaluations(&request)
289            .iter()
290            .any(|evaluation| evaluation.effect == cookie_agent_protocol::PermissionEffect::Deny)
291        {
292            let decision = ApprovalInternalDecision {
293                decision: ApprovalInternalDecisionKind::Deny,
294                source: ApprovalDecisionSource::Policy,
295                reason_code: ApprovalReasonCode::PolicyDenied,
296                evaluations: approval_evaluations(&request),
297                rationale: None,
298            };
299            self.append(
300                active.session,
301                Some(run),
302                crate::runtime::event_origin("engine:approvals"),
303                Event::approval_step(
304                    approval_id,
305                    cookie_agent_protocol::ApprovalStep::Evaluated { decision },
306                ),
307            )
308            .await?;
309            self.append(
310                active.session,
311                Some(run),
312                crate::runtime::event_origin("engine:approvals"),
313                Event::approval_finalized(
314                    approval_id,
315                    cookie_agent_protocol::ApprovalFinalization::Decided {
316                        decision: ApprovalFinalDecision {
317                            outcome: ApprovalFinalOutcome::Rejected,
318                            source: ApprovalDecisionSource::Policy,
319                            reason_code: ApprovalReasonCode::PolicyDenied,
320                            feedback: None,
321                            tree_grant_id: None,
322                        },
323                    },
324                ),
325            )
326            .await?;
327            return Ok(ApprovalOutcome {
328                approved: false,
329                feedback: None,
330                reviewer_denial: None,
331            });
332        }
333
334        let permission_mode = self.permission_mode(root);
335        if permission_mode == PermissionMode::Yolo {
336            self.append(
337                active.session,
338                Some(run),
339                crate::runtime::event_origin("engine:approvals"),
340                Event::approval_step(
341                    approval_id,
342                    cookie_agent_protocol::ApprovalStep::Evaluated {
343                        decision: ApprovalInternalDecision {
344                            decision: ApprovalInternalDecisionKind::Allow,
345                            source: ApprovalDecisionSource::Policy,
346                            reason_code: ApprovalReasonCode::YoloApproved,
347                            evaluations: approval_evaluations(&request),
348                            rationale: None,
349                        },
350                    },
351                ),
352            )
353            .await?;
354            self.append(
355                active.session,
356                Some(run),
357                crate::runtime::event_origin("engine:approvals"),
358                Event::approval_finalized(
359                    approval_id,
360                    cookie_agent_protocol::ApprovalFinalization::Decided {
361                        decision: ApprovalFinalDecision {
362                            outcome: ApprovalFinalOutcome::Approved,
363                            source: ApprovalDecisionSource::Policy,
364                            reason_code: ApprovalReasonCode::YoloApproved,
365                            feedback: None,
366                            tree_grant_id: None,
367                        },
368                    },
369                ),
370            )
371            .await?;
372            return Ok(ApprovalOutcome {
373                approved: true,
374                feedback: None,
375                reviewer_denial: None,
376            });
377        }
378
379        let review = match permission_mode {
380            PermissionMode::Ask => ReviewOutcome::Unreviewed,
381            PermissionMode::AutoApprove
382            | PermissionMode::AutoApproveN
383            | PermissionMode::AutoApproveY => {
384                #[cfg(test)]
385                let hook = {
386                    self.inner
387                        .test_hooks
388                        .approval_evaluation_hook
389                        .lock()
390                        .expect("approval evaluation hook lock poisoned")
391                        .take()
392                };
393                #[cfg(test)]
394                if let Some(hook) = hook {
395                    if let Some(reached) = hook
396                        .reached
397                        .lock()
398                        .expect("approval evaluation reached lock poisoned")
399                        .take()
400                    {
401                        let _ = reached.send(());
402                    }
403                    hook.release.notified().await;
404                }
405                let approval_policy =
406                    self.active_internal_policy(active, InternalAgentKind::Approval)?;
407                self.review_approval(active, run, &session, &approval_policy, &request)
408                    .await
409            }
410            PermissionMode::Yolo => unreachable!("yolo approvals resolve before prompting"),
411        };
412        let session = active.session;
413        let cancelled = active.cancellation.is_cancelled();
414        let evaluated = (request.clone(), executor.clone());
415        let transition = self
416            .on_actor(session, move |engine| {
417                let (request, executor) = evaluated;
418                engine.approval_evaluation_complete_direct(
419                    session,
420                    run,
421                    request,
422                    executor,
423                    (permission_mode, review),
424                    cancelled,
425                )
426            })
427            .await?;
428        let (mut receiver, expires_at) = match transition {
429            ApprovalEvaluationTransition::Resolved(outcome) => return Ok(outcome),
430            ApprovalEvaluationTransition::Escalated {
431                receiver,
432                expires_at,
433            } => (receiver, expires_at),
434        };
435        let expiry_wait = approval_expiry_wait(expires_at);
436        tokio::select! {
437            decision = &mut receiver => decision.map_err(|_| EngineError::ActorStopped),
438            _ = active.cancellation.cancelled() => {
439                let finalized = self.on_actor(session, move |engine| {
440                    engine.approval_terminal_direct(session, run, approval_id, ApprovalTerminal::Cancelled)
441                }).await?;
442                if finalized {
443                    Ok(ApprovalOutcome {
444                        approved: false,
445                        feedback: Some("cancelled".into()),
446                        reviewer_denial: None,
447                    })
448                } else {
449                    receiver.await.map_err(|_| EngineError::ActorStopped)
450                }
451            },
452            _ = tokio::time::sleep(expiry_wait) => {
453                let finalized = self.on_actor(session, move |engine| {
454                    engine.approval_terminal_direct(session, run, approval_id, ApprovalTerminal::Expired)
455                }).await?;
456                if finalized {
457                    Ok(ApprovalOutcome {
458                        approved: false,
459                        feedback: Some("approval expired unattended".into()),
460                        reviewer_denial: None,
461                    })
462                } else {
463                    receiver.await.map_err(|_| EngineError::ActorStopped)
464                }
465            }
466        }
467    }

469    /// Asks the approval reviewer about `request`, the call under review,
470    /// in the session's reviewer conversation (approval reviewer spec
471    /// section 4).
472    async fn review_approval(
473        &self,
474        active: &ActiveRun,
475        run: RunId,
476        session: &crate::session::SessionProjection,
477        policy: &FrozenInternalAgentPolicy,
478        request: &ApprovalRequest,
479    ) -> ReviewOutcome {
480        let Ok(log) = self.inner.store.log(active.session) else {
481            return ReviewOutcome::Failed;
482        };
483        let events = log.event_snapshot();
484        let inherited = self.reviewer_inheritance(session);
485        // The official `approval` document's body is the whole prompt; the
486        // output contract stays in the request block
487        // (`approval_context::REQUEST_CONTRACT`).
488        let system = expand_internal_prompt(policy, "");
489        // The window of the reviewer's first model, the one a review runs
490        // on unless it falls back (approval reviewer spec section 6).
491        let bounds = approval_context::Bounds::new(
492            policy
493                .models
494                .first()
495                .and_then(|binding| binding.descriptor.capabilities.limits.context),
496            policy
497                .models
498                .first()
499                .and_then(|binding| internal_agent_output_limit(binding, policy)),
500            &system,
501        );
502        let Some(conversation) =
503            approval_context::render(&events, inherited.as_ref(), request.approval_id(), bounds)
504        else {
505            return ReviewOutcome::Failed;
506        };
507        let history = conversation.history(system);
508        let result = self
509            .run_internal_history_agent(
510                active.session,
511                Some(run),
512                InternalAgentKind::Approval,
513                policy,
514                InternalAgentHistoryInput {
515                    history,
516                    summary_source: conversation.request,
517                    tools: Vec::new(),
518                    reject_non_text: true,
519                },
520                InternalAgentExecution {
521                    cancellation: &active.cancellation,
522                    actor_direct: false,
523                },
524            )
525            .await;
526        match result {
527            Ok(result) => approval_context::parse_verdict(&result.text).map_or(
528                ReviewOutcome::Failed,
529                |verdict| ReviewOutcome::Verdict {
530                    decision: verdict.decision,
531                    reason: verdict.reason,
532                },
533            ),
534            Err(_) => ReviewOutcome::Failed,
535        }
536    }

538    /// What a delegated session's reviewer conversation opens with: its
539    /// root's user messages up to the delegation, then each ancestor's
540    /// instructions from its parent, and the root's `AGENTS.md` (approval
541    /// reviewer spec sections 4.3 and 4.5). `None` for a root.
542    fn reviewer_inheritance(
543        &self,
544        session: &crate::session::SessionProjection,
545    ) -> Option<approval_context::Inherited> {
546        let mut origin = session.meta.origin.clone();
547        let mut ancestors = Vec::new();
548        while let SessionOrigin::Delegated {
549            parent_session_id, ..
550        } = origin
551        {
552            let parent = self.inner.store.get(parent_session_id).ok()?;
553            origin = parent.meta.origin.clone();
554            ancestors.push(parent);
555        }
556        let root = ancestors.pop()?;
557        // The root's child on the way down: the delegation that started
558        // this branch.
559        let delegated_at = ancestors
560            .last()
561            .map_or(&session.log, |child| &child.log)
562            .event_snapshot()
563            .first()
564            .map(|created| created.timestamp)?;
565        let root_events = root.log.event_snapshot();
566        let mut entries = approval_context::root_messages_until(&root_events, delegated_at);
567        entries.extend(ancestors.iter().rev().filter_map(|ancestor| {
568            approval_context::delegation_prompt(&ancestor.log.event_snapshot())
569        }));
570        Some(approval_context::Inherited {
571            entries,
572            agents_md: approval_context::agents_md_until(&root_events, delegated_at),
573        })
574    }
575}

577/// Untrusted `text` for the reviewer's tagged blocks: every `<` becomes
578/// `&lt;`, so the text can neither close nor open a block (approval
579/// reviewer spec section 4.7). The blocks' own tags are fixed strings.
580pub(in crate::runtime) fn escape_untrusted(text: &str) -> Cow<'_, str> {
581    if text.contains('<') {
582        Cow::Owned(text.replace('<', "&lt;"))
583    } else {
584        Cow::Borrowed(text)
585    }
586}

598pub(in crate::runtime) fn approval_request_for_operation(
599    tool_call_id: cookie_agent_protocol::ToolCallId,
600    trigger: ApprovalTrigger,
601    operation: PreparedOperationIdentity,
602    traces: Vec<cookie_agent_protocol::DecisionTrace>,
603    allow_tree_grant: bool,
604    expires_at: Option<jiff::Timestamp>,
605) -> ApprovalRequest {
606    let evaluations = operation
607        .resources()
608        .iter()
609        .zip(traces)
610        .map(|(resource, trace)| ApprovalEvaluation {
611            resource_digest: resource.binding_digest.clone(),
612            effect: trace.effect,
613            trace,
614        })
615        .collect::<Vec<_>>();
616    ApprovalRequest::new(
617        ApprovalId::new_v7(),
618        1,
619        tool_call_id,
620        trigger,
621        operation,
622        evaluations,
623        ApprovalConstraints {
624            allow_once: true,
625            allow_tree_grant,
626            cancellable: true,
627            expires_at,
628        },
629    )
630    .expect("prepared approval request is complete")
631}

633/// When an approval escalated now expires: never without a `window`.
634pub(in crate::runtime) fn approval_expiry(
635    window: Option<std::time::Duration>,
636) -> Option<jiff::Timestamp> {
637    jiff::Timestamp::now().checked_add(window?).ok()
638}

640pub(in crate::runtime) fn approval_deadline_exhausted(expires_at: Option<jiff::Timestamp>) -> bool {
641    expires_at.is_some_and(|expires_at| expires_at <= jiff::Timestamp::now())
642}
643
644pub(in crate::runtime) fn approval_expiry_wait(
645    expires_at: Option<jiff::Timestamp>,
646) -> std::time::Duration {
647    let Some(expires_at) = expires_at else {
648        return std::time::Duration::from_secs(100 * 365 * 24 * 60 * 60);
649    };
650    let now = jiff::Timestamp::now();
651    if expires_at <= now {
652        std::time::Duration::ZERO
653    } else {
654        expires_at.duration_since(now).unsigned_abs()
655    }
656}
