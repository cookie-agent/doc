1//! The reviewer conversation as its own log keeps it
2//! (`docs/site/specs/forever-sessions.md` section 5.5).
3//!
4//! The log is the conversation's turns since its latest epoch, each holding
5//! the `user` turn and the verdict as they were sent. A review renders only
6//! its delta: the entries the session's log positions after the previous
7//! turn's cut and up to this review's, read from the session's log from
8//! there with what the previous turn's cursor says of what came before. When
9//! that turn would not fit the window, an epoch starts: a fresh turn rendered
10//! from what the log kept, within the same budgets, with no model call.
11//!
12//! What it renders is what [`super::render`], the render of the whole log the
13//! reviewer used before, gives for the same reviews, with these exceptions,
14//! each by design:
15//!
16//! - **`AGENTS.md` is read afresh at each epoch** of a root session: the
17//!   newest run's records before the epoch's cut, where the render kept those
18//!   before the first review for every epoch. A delegated session keeps what
19//!   it inherited of its root when its log was created.
20//! - **A session with no reviewer log**, as every session an earlier build
21//!   reviewed, starts one with a fresh turn at its next review, carrying the
22//!   last verdicts its own log records; the render would have sent the
23//!   conversation from its first review.
24//! - **What a turn sent stays as sent.** A reviewer model with another window
25//!   moves the next epoch, not the earlier ones, and a fresh turn renders from
26//!   what the log kept, where the render would have reached back to entries
27//!   the earlier bounds had left out.
28//! - **A run's number counts the runs of the session's log**, of which the
29//!   cursor keeps the latest [`RECENT_RUNS`]: a record naming an older run
30//!   would number it anew. A call still open holds its entries for at most
31//!   [`OPEN_CALLS`] calls, a message waiting to be consumed for at most
32//!   [`WAITING_MESSAGES`] messages.
33//! - **A verdict the reviewer's log did not record**, as when the daemon
34//!   stopped between the evaluation and the record, is no turn: its entries
35//!   fall into the next review's, as a failed review's do.
36

67/// The turns of a reviewer log since its latest epoch, from what the log
68/// holds of its window: empty when it holds none, or when the first of them
69/// did not start an epoch, which no log this build writes holds.
70#[must_use]
71pub(in crate::runtime) fn thread_of(events: &[Arc<StoredEvent>]) -> Vec<ReviewerTurn> {
72    let turns = events
73        .iter()
74        .filter_map(|event| match &event.payload {
75            Event::ReviewerTurnRecorded { turn } => Some((**turn).clone()),
76            _ => None,
77        })
78        .collect::<Vec<_>>();
79    let Some(start) = turns.iter().rposition(|turn| turn.epoch.is_some()) else {
80        return Vec::new();
81    };
82    turns[start..].to_vec()
83}

146    /// The label `, starts run N` or `, during run N` of an input in `run`.
147    fn run_label(&mut self, run: Option<RunId>) -> String {
148        let Some(run) = run else {
149            return String::new();
150        };
151        let Some(known) = self
152            .cursor
153            .recent_runs
154            .iter_mut()
155            .find(|known| known.run_id == run)
156        else {
157            return ", during run 0".to_owned();
158        };
159        if known.has_input {
160            format!(", during run {}", known.ordinal)
161        } else {
162            known.has_input = true;
163            format!(", starts run {}", known.ordinal)
164        }
165    }

167    /// Numbers `run` when the pass sees it first, as runs count in the order
168    /// they first appear.
169    fn note_run(&mut self, run: RunId) {
170        if self
171            .cursor
172            .recent_runs
173            .iter()
174            .any(|known| known.run_id == run)
175        {
176            return;
177        }
178        self.cursor.runs += 1;
179        self.cursor.recent_runs.push(ReviewerRun {
180            run_id: run,
181            ordinal: self.cursor.runs,
182            has_input: false,
183        });
184        if self.cursor.recent_runs.len() > RECENT_RUNS {
185            self.cursor.recent_runs.remove(0);
186        }
187    }

205    pub(in crate::runtime) fn observe(&mut self, event: &StoredEvent) {
206        if let Some(run) = event.run_id {
207            self.note_run(run);
208        }
209        match &event.payload {
210            Event::RuntimeAttached {
211                name,
212                runtime_id,
213                kind,
214                workspace,
215                ..
216            } => {
217                let attachment = ReviewerAttachment {
218                    name: name.as_str().to_owned(),
219                    runtime_id: runtime_id.as_str().to_owned(),
220                    kind: kind.as_str().to_owned(),
221                    workspace: workspace.clone(),
222                };
223                match self
224                    .cursor
225                    .attachments
226                    .iter_mut()
227                    .find(|known| known.name == attachment.name)
228                {
229                    Some(known) => *known = attachment,
230                    None => self.cursor.attachments.push(attachment),
231                }
232            }
233            Event::AgentMdLoaded { .. } if self.cursor.inherited.is_none() => {
234                if let Some(text) = agents_md_text(event) {
235                    if self
236                        .cursor
237                        .agents_md
238                        .last()
239                        .is_some_and(|last| last.run_id != event.run_id)
240                    {
241                        self.cursor.agents_md.clear();
242                    }
243                    self.cursor.agents_md.push(ReviewerAgentsMd {
244                        seq: event.seq,
245                        run_id: event.run_id,
246                        text,
247                    });
248                }
249            }
250            Event::UserInputSubmitted { content } => {
251                let origin = event.origin.as_ref().map(|origin| origin.as_str());
252                let when = self.run_label(event.run_id);
253                let text = format!(
254                    "{} {}{when}\n{}",
255                    input_label(origin),
256                    timestamp_label(event.timestamp),
257                    escape_untrusted(&cap_tokens(&content.as_plain_text(), MESSAGE_TOKEN_CAP))
258                );
259                self.message(event.seq, text);
260            }
261            Event::ProducerMessageAccepted {
262                message_id,
263                producer_owner,
264                body,
265                ..
266            } => {
267                self.cursor
268                    .messages
269                    .retain(|message| message.message_id != *message_id);
270                if let Some(label) = producer_label(producer_owner) {
271                    self.cursor.messages.push(ReviewerMessage {
272                        message_id: *message_id,
273                        label,
274                        body: escape_untrusted(&cap_tokens(
275                            &body.as_plain_text(),
276                            MESSAGE_TOKEN_CAP,
277                        ))
278                        .into_owned(),
279                    });
280                    if self.cursor.messages.len() > WAITING_MESSAGES {
281                        self.cursor.messages.remove(0);
282                    }
283                }
284            }
285            Event::ProducerMessageDiscarded { message_id, .. } => {
286                self.cursor
287                    .messages
288                    .retain(|message| message.message_id != *message_id);
289            }
290            Event::ProducerMessageConsumed { message_id, run_id } => {
291                if let Some(position) = self
292                    .cursor
293                    .messages
294                    .iter()
295                    .position(|message| message.message_id == *message_id)
296                {
297                    let message = self.cursor.messages.remove(position);
298                    let when = self.run_label(Some(*run_id));
299                    let text = format!(
300                        "{} {}{when}\n{}",
301                        message.label,
302                        timestamp_label(event.timestamp),
303                        message.body
304                    );
305                    self.message(event.seq, text);
306                }
307            }
308            Event::ModelTurnCommitted {
309                model_turn_seq,
310                turn,
311                ..
312            } => {
313                for (index, part) in turn.content.iter().enumerate() {
314                    if let cookie_agent_protocol::PersistedAssistantPart::ToolCall {
315                        input, ..
316                    } = part
317                    {
318                        self.turn_calls.insert(
319                            (*model_turn_seq, u32::try_from(index).unwrap_or(u32::MAX)),
320                            input.clone(),
321                        );
322                    }
323                }
324            }
325            Event::ToolCallStarted { start } => {
326                let name = start.presentation.tool.name.clone();
327                if let Some(reviews) = &mut self.reviews {
328                    reviews.names.insert(start.tool_call_id, name.clone());
329                }
330                if READ_ONLY_TOOLS.contains(&name.as_str()) {
331                    return;
332                }
333                let arguments = self
334                    .turn_calls
335                    .get(&(start.owner.model_turn_seq, start.owner.content_index))
336                    .cloned()
337                    .or_else(|| self.turns.and_then(|turns| turns(&start.owner)));
338                let call = CallFacts {
339                    name: Some(name),
340                    source: start
341                        .presentation
342                        .tool
343                        .source
344                        .as_ref()
345                        .and_then(|source| source.known().cloned()),
346                    arguments,
347                };
348                let end = |outcome| trajectory_text(&call, outcome).unwrap_or_default();
349                self.cursor
350                    .open_calls
351                    .retain(|open| open.tool_call_id != start.tool_call_id);
352                self.cursor.open_calls.push(ReviewerOpenCall {
353                    tool_call_id: start.tool_call_id,
354                    completed: end("completed"),
355                    failed: end("failed"),
356                    denied: end("denied"),
357                    cancelled: end("cancelled"),
358                });
359                if self.cursor.open_calls.len() > OPEN_CALLS {
360                    self.cursor.open_calls.remove(0);
361                }
362            }
363            Event::ToolCallTerminated { termination } => {
364                if let Some(position) = self
365                    .cursor
366                    .open_calls
367                    .iter()
368                    .position(|open| open.tool_call_id == termination.tool_call_id)
369                {
370                    let open = self.cursor.open_calls.remove(position);
371                    let text = match termination_outcome(termination) {
372                        "denied" => open.denied,
373                        "failed" => open.failed,
374                        "cancelled" => open.cancelled,
375                        _ => open.completed,
376                    };
377                    self.trajectory(event.seq, text);
378                }
379            }
380            Event::ApprovalRequested { request } => {
381                if let Some(reviews) = &mut self.reviews {
382                    reviews
383                        .requests
384                        .insert(request.approval_id(), (event.seq, request.tool_call_id()));
385                    if Some(request.approval_id()) != self.current {
386                        reviews.pending = Some(request.approval_id());
387                    }
388                }
389            }
390            Event::InternalAgentFinished { kind, outcome, .. } => {
391                if let Some(reviews) = &mut self.reviews
392                    && *kind == InternalAgentKind::Approval
393                    && matches!(
394                        outcome.known(),
395                        Some(cookie_agent_protocol::InternalAgentOutcome::Completed { .. })
396                    )
397                    && let Some(approval_id) = reviews.pending.take()
398                {
399                    reviews.reviewed.insert(approval_id);
400                }
401            }
402            Event::ApprovalStep {
403                approval_id,
404                step: Open::Known(cookie_agent_protocol::ApprovalStep::Evaluated { decision }),
405            } => {
406                if let Some(reviews) = &mut self.reviews {
407                    if reviews.pending == Some(*approval_id) {
408                        reviews.pending = None;
409                    }
410                    let decided = match decision.reason_code {
411                        ApprovalReasonCode::InternalAgentAllowed => {
412                            Some(ApprovalInternalDecisionKind::Allow)
413                        }
414                        ApprovalReasonCode::InternalAgentDenied => {
415                            Some(ApprovalInternalDecisionKind::Deny)
416                        }
417                        ApprovalReasonCode::Escalated => Some(ApprovalInternalDecisionKind::Ask),
418                        _ => None,
419                    };
420                    if let Some(decision_kind) = decided.filter(|_| {
421                        reviews.reviewed.contains(approval_id)
422                            && decision.source == ApprovalDecisionSource::InternalAgent
423                    }) && let Some((cut, call)) = reviews.requests.get(approval_id)
424                    {
425                        reviews.verdicts.push(ReviewerVerdict {
426                            cut_seq: *cut,
427                            tool: reviews
428                                .names
429                                .get(call)
430                                .cloned()
431                                .unwrap_or_else(|| "a tool".to_owned()),
432                            verdict: Verdict {
433                                decision: decision_kind,
434                                reason: decision.rationale.clone(),
435                            }
436                            .render(),
437                        });
438                    }
439                }
440            }
441            Event::ApprovalFinalized {
442                outcome:
443                    Open::Known(cookie_agent_protocol::ApprovalFinalization::Decided { decision }),
444                ..
445            } if decision.source == ApprovalDecisionSource::User => {
446                let text = match decision.outcome {
447                    ApprovalFinalOutcome::Approved => Some("[user decision] approved".to_owned()),
448                    ApprovalFinalOutcome::Rejected => Some(match &decision.feedback {
449                        Some(feedback) => format!(
450                            "[user decision] rejected: {}",
451                            escape_untrusted(&cap_tokens(
452                                feedback.message.as_str(),
453                                TRAJECTORY_TOKEN_CAP
454                            ))
455                        ),
456                        None => "[user decision] rejected".to_owned(),
457                    }),
458                    ApprovalFinalOutcome::Cancelled | ApprovalFinalOutcome::Expired => None,
459                };
460                if let Some(text) = text {
461                    self.trajectory(event.seq, text);
462                }
463            }
464            _ => {}
465        }
466    }

506/// The review of `request`, cut at `cut` in the session's log, after
507/// `thread`, the reviewer log's turns since its latest epoch.
508///
509/// `events` is the session's log after the last turn's cut through `cut`, or
510/// from its start when `thread` is empty; `inherited` is a delegated
511/// session's opening, read only then. `call` is what the log records of the
512/// call under review, `turns` where a started call's arguments are when the
513/// pass did not read its turn.
514pub(in crate::runtime) struct Review<'a> {
515    pub(in crate::runtime) thread: &'a [ReviewerTurn],
516    /// A log's first turn that was sent and never recorded, as its review
517    /// failed: with no thread, the next review reads on from its cut instead
518    /// of from the log's start, as it holds what that pass kept.
519    pub(in crate::runtime) unrecorded: Option<&'a ReviewerTurn>,
520    pub(in crate::runtime) inherited: Option<&'a Inherited>,
521    pub(in crate::runtime) request: &'a ApprovalRequest,
522    pub(in crate::runtime) cut: u64,
523    pub(in crate::runtime) call: Option<&'a CallFacts>,
524    pub(in crate::runtime) turns: Option<TurnLookup<'a>>,
525    pub(in crate::runtime) bounds: Bounds,
526}

528impl Review<'_> {
529    /// What the review sends, from `events`, the session's log as
530    /// [`Review`] says.
531    pub(in crate::runtime) fn prepare(
532        &self,
533        events: impl IntoIterator<Item = Arc<StoredEvent>>,
534    ) -> Prepared {
535        let Some(last) = self.thread.last() else {
536            return match self.unrecorded {
537                Some(unrecorded) => self.after_unrecorded(unrecorded, events),
538                None => self.first(events),
539            };
540        };
541        let mut gatherer = Gatherer::from_cursor(&last.cursor, self.turns);
542        for event in events {
543            if event.seq > last.cut_seq && event.seq <= self.cut {
544                gatherer.observe(&event);
545            }
546        }
547        let attachments = attachments_of(&gatherer.cursor);
548        let messages = gatherer
549            .entries
550            .iter()
551            .filter(|entry| entry.message)
552            .count();
553        let mut text = String::from("<context_delta>\n");
554        for entry in &gatherer.entries {
555            text.push_str(&entry.text);
556            text.push_str("\n\n");
557        }
558        if gatherer.entries.is_empty() {
559            text.push_str("[nothing new]\n");
560        } else {
561            text.pop();
562        }
563        text.push_str("</context_delta>\n\n");
564        text.push_str(&request_block(
565            Some(self.request),
566            self.call,
567            &attachments,
568            Since::Delta(messages),
569        ));
570        let used = self.bounds.system_tokens
571            + self
572                .thread
573                .iter()
574                .map(|turn| estimate_tokens(&turn.request) + estimate_tokens(&turn.verdict))
575                .sum::<usize>();
576        if used + estimate_tokens(&text) <= self.bounds.window {
577            return Prepared {
578                conversation: Conversation {
579                    reviews: self
580                        .thread
581                        .iter()
582                        .map(|turn| (turn.request.clone(), turn.verdict.clone()))
583                        .collect(),
584                    request: text.clone(),
585                },
586                turn: self.turn(text, gatherer.entries, None, gatherer.cursor),
587            };
588        }
589        // An epoch: the kept entries of the epoch's first turn, those of each
590        // turn since, and this one's.
591        let epoch = self
592            .thread
593            .first()
594            .and_then(|first| first.epoch.clone())
595            .unwrap_or_default();
596        let mut messages = epoch.messages;
597        let mut trajectory = epoch.trajectory;
598        for entry in self
599            .thread
600            .iter()
601            .flat_map(|turn| turn.entries.iter().cloned())
602            .chain(gatherer.entries)
603        {
604            if entry.message {
605                messages.push(entry);
606            } else {
607                trajectory.push(entry);
608            }
609        }
610        let verdicts = epoch
611            .verdicts
612            .into_iter()
613            .chain(self.thread.iter().map(|turn| ReviewerVerdict {
614                cut_seq: turn.cut_seq,
615                tool: turn.tool.clone(),
616                verdict: turn.verdict.clone(),
617            }))
618            .collect::<Vec<_>>();
619        let kept = Kept {
620            messages,
621            messages_omitted: epoch.messages_omitted,
622            trajectory,
623            trajectory_omitted: epoch.trajectory_omitted,
624        };
625        self.fresh(kept, &verdicts, gatherer.cursor, Since::StartedOver)
626    }
627
628    /// A log's first turn after `unrecorded`, one that was sent and not
629    /// recorded: a fresh turn from what it kept and the entries since its cut,
630    /// as [`Self::first`] gives from the whole log, carrying the same verdicts,
631    /// as no verdict came since.
632    fn after_unrecorded(
633        &self,
634        unrecorded: &ReviewerTurn,
635        events: impl IntoIterator<Item = Arc<StoredEvent>>,
636    ) -> Prepared {
637        let mut gatherer = Gatherer::from_cursor(&unrecorded.cursor, self.turns);
638        for event in events {
639            if event.seq > unrecorded.cut_seq && event.seq <= self.cut {
640                gatherer.observe(&event);
641            }
642        }
643        let epoch = unrecorded.epoch.clone().unwrap_or_default();
644        let mut messages = epoch.messages;
645        let mut trajectory = epoch.trajectory;
646        for entry in gatherer.entries.drain(..) {
647            if entry.message {
648                messages.push(entry);
649            } else {
650                trajectory.push(entry);
651            }
652        }
653        let kept = Kept {
654            messages,
655            messages_omitted: epoch.messages_omitted,
656            trajectory,
657            trajectory_omitted: epoch.trajectory_omitted,
658        };
659        let since = if epoch.verdicts.is_empty() {
660            Since::FirstReview
661        } else {
662            Since::StartedOver
663        };
664        self.fresh(kept, &epoch.verdicts, gatherer.cursor, since)
665    }
666
667    /// The first review of a log: a fresh turn from one pass over the
668    /// session's log, carrying the last verdicts it records.
669    fn first(&self, events: impl IntoIterator<Item = Arc<StoredEvent>>) -> Prepared {
670        let mut gatherer =
671            Gatherer::from_start(self.inherited, Some(self.request.approval_id()), self.turns);
672        for event in events {
673            if event.seq <= self.cut {
674                gatherer.observe(&event);
675            }
676        }
677        let verdicts = gatherer
678            .reviews
679            .take()
680            .map(|reviews| reviews.verdicts)
681            .unwrap_or_default()
682            .into_iter()
683            .filter(|verdict| verdict.cut_seq < self.cut)
684            .collect::<Vec<_>>();
685        let mut messages = self
686            .inherited
687            .map(|inherited| {
688                inherited
689                    .entries
690                    .iter()
691                    .map(|text| ReviewerEntry {
692                        seq: 0,
693                        text: text.clone(),
694                        message: true,
695                    })
696                    .collect::<Vec<_>>()
697            })
698            .unwrap_or_default();
699        let mut trajectory = Vec::new();
700        for entry in std::mem::take(&mut gatherer.entries) {
701            if entry.message {
702                messages.push(entry);
703            } else {
704                trajectory.push(entry);
705            }
706        }
707        let kept = Kept {
708            messages,
709            messages_omitted: None,
710            trajectory,
711            trajectory_omitted: 0,
712        };
713        let since = if verdicts.is_empty() {
714            Since::FirstReview
715        } else {
716            Since::StartedOver
717        };
718        self.fresh(kept, &verdicts, gatherer.cursor, since)
719    }

721    fn turn(
722        &self,
723        request: String,
724        entries: Vec<ReviewerEntry>,
725        epoch: Option<ReviewerEpoch>,
726        cursor: ReviewerCursor,
727    ) -> ReviewerTurn {
728        ReviewerTurn {
729            approval_id: self.request.approval_id(),
730            cut_seq: self.cut,
731            evaluated_seq: 0,
732            tool: self
733                .call
734                .and_then(|call| call.name.clone())
735                .unwrap_or_else(|| "a tool".to_owned()),
736            request,
737            verdict: String::new(),
738            entries,
739            epoch,
740            cursor,
741        }
742    }

744    /// A fresh turn from `kept`, within the budgets, carrying the last of
745    /// `verdicts`: the first turn of an epoch.
746    fn fresh(
747        &self,
748        kept: Kept,
749        verdicts: &[ReviewerVerdict],
750        cursor: ReviewerCursor,
751        since: Since,
752    ) -> Prepared {
753        let carried = &verdicts[verdicts.len().saturating_sub(CARRIED_VERDICTS)..];
754        let carried_entries = carried
755            .iter()
756            .map(|verdict| {
757                (
758                    verdict.cut_seq,
759                    escape_untrusted(&cap_tokens(
760                        &format!(
761                            "[your earlier verdict] on {}: {}",
762                            verdict.tool, verdict.verdict
763                        ),
764                        TRAJECTORY_TOKEN_CAP,
765                    ))
766                    .into_owned(),
767                )
768            })
769            .collect::<Vec<_>>();
770        let (messages, messages_marker, messages_omitted) = budget_messages(
771            &kept.messages,
772            kept.messages_omitted,
773            self.bounds.messages(),
774        );
775        let mut trajectory = kept
776            .trajectory
777            .iter()
778            .map(|entry| (entry.seq, entry.text.as_str()))
779            .chain(
780                carried_entries
781                    .iter()
782                    .map(|(seq, text)| (*seq, text.as_str())),
783            )
784            .collect::<Vec<_>>();
785        trajectory.sort_by_key(|(seq, _)| *seq);
786        let unknown_before = (kept.trajectory_omitted > 0)
787            .then(|| kept.trajectory.first().map(|entry| entry.seq))
788            .flatten();
789        let (trajectory_start, trajectory_marker) = budget_trajectory(
790            &trajectory,
791            usize::try_from(kept.trajectory_omitted).unwrap_or(usize::MAX),
792            unknown_before,
793            self.bounds.trajectory(),
794        );
795
796        let agents_md = match &cursor.inherited {
797            Some(inherited) => inherited.agents_md.clone(),
798            None => agents_md_before(
799                &cursor
800                    .agents_md
801                    .iter()
802                    .map(|loaded| (loaded.seq, loaded.run_id, loaded.text.clone()))
803                    .collect::<Vec<_>>(),
804                self.cut,
805            ),
806        };
807        let mut text = String::from("<context>\n");
808        let mut wrote = false;
809        let mut push = |text: &mut String, entry: &str| {
810            text.push_str(entry);
811            text.push_str("\n\n");
812            wrote = true;
813        };
814        if let Some(agents_md) = &agents_md {
815            push(&mut text, &format!("[project instructions]\n{agents_md}"));
816        }
817        let mut merged = messages
818            .iter()
819            .map(|entry| (entry.seq, entry.text.clone()))
820            .chain(
821                trajectory[trajectory_start..]
822                    .iter()
823                    .map(|(seq, entry)| (*seq, (*entry).to_owned())),
824            )
825            .collect::<Vec<_>>();
826        if let Some((at, count)) = messages_marker {
827            merged.push((at, format!("[… {count} earlier messages omitted …]")));
828        }
829        if let Some((at, count)) = trajectory_marker {
830            merged.push((
831                at,
832                format!("[… {count} earlier trajectory entries omitted …]"),
833            ));
834        }
835        merged.sort_by_key(|(seq, _)| *seq);
836        for (_, entry) in &merged {
837            push(&mut text, entry);
838        }
839        if wrote {
840            text.pop();
841        } else {
842            text.push_str("[nothing new]\n");
843        }
844        text.push_str("</context>\n\n");
845        let attachments = attachments_of(&cursor);
846        text.push_str(&request_block(
847            Some(self.request),
848            self.call,
849            &attachments,
850            since,
851        ));
852        let (trajectory, trajectory_omitted) = hold_trajectory(
853            kept.trajectory,
854            kept.trajectory_omitted,
855            self.bounds.trajectory(),
856        );
857        let epoch = ReviewerEpoch {
858            messages,
859            messages_omitted,
860            trajectory,
861            trajectory_omitted,
862            verdicts: carried.to_vec(),
863        };
864        Prepared {
865            conversation: Conversation {
866                reviews: Vec::new(),
867                request: text.clone(),
868            },
869            turn: self.turn(text, Vec::new(), Some(epoch), cursor),
870        }
871    }
872}

874/// What a fresh turn renders from: the entries a log kept, and what was left
875/// out before them.

877    messages: Vec<ReviewerEntry>,
878    messages_omitted: Option<ReviewerOmitted>,
879    trajectory: Vec<ReviewerEntry>,
880    trajectory_omitted: u64,
881}

889                attachment.name.clone(),
890                (

895            )

900fn tokens(count: usize) -> u64 {
901    u64::try_from(count).unwrap_or(u64::MAX)
902}

904/// The messages a fresh turn keeps of `messages`, the ones a log kept, with
905/// `omitted` left out before (section 6): within `budget`, the first
906/// [`KEPT_FIRST_MESSAGES`] with the user's authority, the first instructions
907/// from a parent agent, and the newest down to half the budget, as the render
908/// of the whole log keeps them. An earlier epoch left out only messages no
909/// later one keeps: never a task, and none newer than what it kept. Returns
910/// what it keeps, where the omission marker goes with how many it stands for,
911/// and what is left out from here on.
912fn budget_messages(
913    messages: &[ReviewerEntry],
914    omitted: Option<ReviewerOmitted>,
915    budget: usize,
916) -> (
917    Vec<ReviewerEntry>,
918    Option<(u64, usize)>,
919    Option<ReviewerOmitted>,
920) {
921    let omitted_tokens = omitted.map_or(0, |omitted| {
922        usize::try_from(omitted.tokens).unwrap_or(usize::MAX)
923    });
924    let total = messages
925        .iter()
926        .map(|entry| estimate_tokens(&entry.text))
927        .sum::<usize>()
928        .saturating_add(omitted_tokens);
929    let marker = |omitted: Option<ReviewerOmitted>| {
930        omitted.map(|omitted| {
931            (
932                omitted.first_seq,
933                usize::try_from(omitted.count).unwrap_or(usize::MAX),
934            )
935        })
936    };
937    if total <= budget {
938        return (messages.to_vec(), marker(omitted), omitted);
939    }
940    let target = budget / 2;
941    let mut keep = vec![false; messages.len()];
942    let mut used = 0;
943    let mut first = 0;
944    let mut instructions = false;
945    for (index, entry) in messages.iter().enumerate() {
946        let task = if has_authority(&entry.text) && first < KEPT_FIRST_MESSAGES {
947            first += 1;
948            true
949        } else if !instructions
950            && entry
951                .text
952                .starts_with("[instructions from the parent agent]")
953        {
954            instructions = true;
955            true
956        } else {
957            false
958        };
959        if task {
960            keep[index] = true;
961            used += estimate_tokens(&entry.text);
962        }
963    }
964    let last_first = keep.iter().rposition(|kept| *kept);
965    for index in (0..messages.len()).rev() {
966        if keep[index] || last_first.is_some_and(|last| index < last) {
967            continue;
968        }
969        let cost = estimate_tokens(&messages[index].text);
970        if used + cost > target {
971            break;
972        }
973        keep[index] = true;
974        used += cost;
975    }
976    let mut left = omitted.unwrap_or(ReviewerOmitted {
977        count: 0,
978        tokens: 0,
979        first_seq: u64::MAX,
980    });
981    let mut kept = Vec::new();
982    for (entry, keep) in messages.iter().zip(keep) {
983        if keep {
984            kept.push(entry.clone());
985        } else {
986            left.count += 1;
987            left.tokens = left
988                .tokens
989                .saturating_add(tokens(estimate_tokens(&entry.text)));
990            left.first_seq = left.first_seq.min(entry.seq);
991        }
992    }
993    let left = (left.count > 0).then_some(left);
994    (kept, marker(left), left)
995}

997/// Where the trajectory a fresh turn keeps starts in `trajectory`, the
998/// entries a log kept with the carried verdicts, in order, and the omission
999/// marker (section 6): it drops its oldest half, as often as it takes to fit
1000/// `budget`, as the render of the whole log drops them. `unknown` older
1001/// entries were let go of, all before `unknown_before`, the first kept, from
1002/// which on what was kept is over the budget: a half that reaches them is
1003/// over it too.
1004fn budget_trajectory(
1005    trajectory: &[(u64, &str)],
1006    unknown: usize,
1007    unknown_before: Option<u64>,
1008    budget: usize,
1009) -> (usize, Option<(u64, usize)>) {
1010    let length = trajectory.len();
1011    let total = unknown + length;
1012    // The newest entries whose order is known: all of them, or those from the
1013    // first kept on.
1014    let known = match unknown_before {
1015        Some(first) if unknown > 0 => trajectory.iter().filter(|(seq, _)| *seq >= first).count(),
1016        _ => length,
1017    };
1018    let over = |remaining: usize| {
1019        remaining > known
1020            || trajectory[length - remaining..]
1021                .iter()
1022                .map(|(_, text)| estimate_tokens(text))
1023                .sum::<usize>()
1024                > budget
1025    };
1026    let mut remaining = total;
1027    let mut dropped = 0;
1028    let mut last_at = None;
1029    while remaining > 0 && over(remaining) {
1030        let half = remaining.div_ceil(2);
1031        last_at = Some(total - remaining + half - 1);
1032        remaining -= half;
1033        dropped += half;
1034    }
1035    let start = length - remaining.min(length);
1036    let marker = last_at.and_then(|at| {
1037        length
1038            .checked_sub(total - at)
1039            .map(|index| (trajectory[index].0, dropped))
1040    });
1041    (start, marker)
1042}
1043
1044/// What a log keeps of `trajectory` for later epochs: the newest entries, up
1045/// to the first that puts them over `budget`, as no later half that fits
1046/// reaches past it. Returns them, and how many were let go of in all.
1047fn hold_trajectory(
1048    trajectory: Vec<ReviewerEntry>,
1049    unknown: u64,
1050    budget: usize,
1051) -> (Vec<ReviewerEntry>, u64) {
1052    let mut used = 0;
1053    let mut start = 0;
1054    for (index, entry) in trajectory.iter().enumerate().rev() {
1055        used += estimate_tokens(&entry.text);
1056        if used > budget {
1057            start = index;
1058            break;
1059        }
1060    }
1061    let let_go = tokens(start);
1062    (
1063        trajectory.into_iter().skip(start).collect(),
1064        unknown + let_go,
1065    )
1066}
