10/// The sessions a tree deletion deleted, and the ACP sessions each owned.
11type DeletedTree = (

16impl Engine {
17    /// Deletes the tree rooted at `session_id` (events, metadata, artifacts,
18    /// and every delegated session) and returns the deleted sessions, the
19    /// root first. Every connection is told through `sessions.changed`.
20    ///
21    /// Refused for a delegated session ([`EngineError::NotARootSession`]),
22    /// while a session of the tree is busy
23    /// ([`EngineError::SessionTreeRunning`]), while another process holds
24    /// the tree ([`EngineError::SessionOwnedByAnotherProcess`]), and when
25    /// the root's last event is no longer `expected_last_seq`
26    /// ([`EngineError::StaleDelete`]). A refusal changes nothing.
27    pub async fn delete_session(
28        &self,
29        session_id: SessionId,
30        expected_last_seq: u64,
31    ) -> Result<Vec<SessionId>, EngineError> {
32        self.ensure_not_shutting_down()?;
33        let root = self.inner.store.root_of(session_id)?;
34        if root != session_id {
35            return Err(EngineError::NotARootSession {
36                session_id,
37                root_session_id: root,
38            });
39        }
40        let deleted = if self.inner.store.holds_tree(root) {
41            // On the root's actor, after every request already queued for
42            // it. Nothing reconciles afterwards: there is nothing left.
43            self.on_actor_unreconciled(root, move |engine| {
44                engine.delete_tree_direct(root, expected_last_seq)
45            })
46            .await?
47        } else {
48            // A tree this process does not hold is not adopted: adoption
49            // would repair it, appending events, before it is deleted.
50            let engine = self.clone();
51            crate::blocking::off_worker(move || engine.delete_tree_direct(root, expected_last_seq))?
52        };
53        let (deleted, acp_sessions) = deleted;
54        self.forget_deleted_tree(root, &deleted);
55        // The agents' own transcripts go with the sessions: each session's
56        // ACP sessions are closed and deleted where the agent can.
57        for (session, records) in acp_sessions {
58            self.delete_acp_sessions(session, records).await;
59        }
60        for session in &deleted {
61            self.close_acp_session(*session).await;
62        }
63        Ok(deleted)
64    }

66    /// Deletes the tree rooted at `root`; returns the deleted sessions and
67    /// the ACP sessions each of them owned.
68    fn delete_tree_direct(
69        &self,
70        root: SessionId,
71        expected_last_seq: u64,
72    ) -> Result<DeletedTree, EngineError> {
73        // Serializes with an adoption of the tree, which would otherwise be
74        // caught halfway.
75        let adoption_lock = self.inner.store.adoption_lock(root);
76        let _adoption = adoption_lock
77            .lock()
78            .unwrap_or_else(|poisoned| poisoned.into_inner());
79        let deletion = self
80            .inner
81            .store
82            .begin_tree_deletion(root)
83            .map_err(|error| match error {
84                SessionError::SessionLocked(_) => EngineError::SessionOwnedByAnotherProcess(root),
85                SessionError::TreeBusy(_) => EngineError::SessionTreeRunning { root, busy: root },
86                error => error.into(),
87            })?;
88        // A running tree is refused as running even when the caller's guard
89        // is stale too: a streaming run moves the root past what was listed.
90        let members = deletion.members().iter().copied().collect::<HashSet<_>>();
91        if deletion.was_held()
92            && let Some(busy) = self.busy_member(&members).or(deletion.running_member())
93        {
94            return Err(EngineError::SessionTreeRunning { root, busy });
95        }
96        let current_last_seq = deletion.root_last_seq();
97        if current_last_seq != expected_last_seq {
98            return Err(EngineError::StaleDelete {
99                session_id: root,
100                current_last_seq,
101            });
102        }
103        // Nothing of the tree may keep running or keep a file open in it.
104        self.kill_deleted_background_jobs(&members);
105        self.inner.uploads.forget_tree(root);
106        self.inner.artifacts.forget_tree(root);
107        let acp_sessions = deletion
108            .members()
109            .iter()
110            .filter_map(|member| {
111                let projection = self.inner.store.get(*member).ok()?;
112                let records = super::acp::lifecycle::own_records(&projection.log.event_snapshot());
113                (!records.is_empty()).then_some((*member, records))
114            })
115            .collect();
116        Ok((deletion.commit()?, acp_sessions))
117    }

119    /// Whether a session of the tree rooted at `root` has an active run, a
120    /// compaction, or a place in the delegation queue, as a delete would
121    /// find it.
122    pub(crate) fn tree_busy(&self, root: SessionId) -> bool {
123        let root_of = |session: &SessionId| self.inner.store.root_of(*session).unwrap_or(*session);
124        let busy =
125            |sessions: Vec<SessionId>| sessions.iter().any(|session| root_of(session) == root);
126        let running = self
127            .inner
128            .sessions
129            .active
130            .lock()
131            .unwrap_or_else(|poisoned| poisoned.into_inner())
132            .values()
133            .map(|run| run.session)
134            .collect::<Vec<_>>();
135        let compacting = self
136            .inner
137            .compaction
138            .in_progress
139            .lock()
140            .unwrap_or_else(|poisoned| poisoned.into_inner())
141            .iter()
142            .copied()
143            .collect::<Vec<_>>();
144        let queued = self
145            .inner
146            .delegation
147            .queue
148            .lock()
149            .unwrap_or_else(|poisoned| poisoned.into_inner())
150            .iter()
151            .copied()
152            .collect::<Vec<_>>();
153        busy(running) || busy(compacting) || busy(queued)
154    }

156    /// A session of `members` with an active run, a compaction or run
157    /// start holding it, or a place in the delegation queue.
158    fn busy_member(&self, members: &HashSet<SessionId>) -> Option<SessionId> {
159        let running = self
160            .inner
161            .sessions
162            .active
163            .lock()
164            .unwrap_or_else(|poisoned| poisoned.into_inner())
165            .values()
166            .map(|run| run.session)
167            .find(|session| members.contains(session));
168        running
169            .or_else(|| {
170                self.inner
171                    .compaction
172                    .in_progress
173                    .lock()
174                    .unwrap_or_else(|poisoned| poisoned.into_inner())
175                    .iter()
176                    .copied()
177                    .find(|session| members.contains(session))
178            })
179            .or_else(|| {
180                self.inner
181                    .delegation
182                    .queue
183                    .lock()
184                    .unwrap_or_else(|poisoned| poisoned.into_inner())
185                    .iter()
186                    .copied()
187                    .find(|session| members.contains(session))
188            })
189    }

191    /// Forgets what the engine keeps for a deleted tree's sessions. The
192    /// actors go too; requests still queued on them find no session.
193    fn forget_deleted_tree(&self, root: SessionId, deleted: &[SessionId]) {
194        let deleted = deleted.iter().copied().collect::<HashSet<_>>();
195        self.forget_deleted_sessions(&deleted);
196        self.inner
197            .approvals
198            .pending
199            .lock()
200            .unwrap_or_else(|poisoned| poisoned.into_inner())
201            .retain(|(session, _), _| !deleted.contains(session));
202        self.inner
203            .approvals
204            .permission_modes
205            .lock()
206            .unwrap_or_else(|poisoned| poisoned.into_inner())
207            .remove(&root);
208        self.inner.grant_journals.forget(root);
209        self.rebuild_visible_tree_grants();
210        self.inner
211            .sessions
212            .forks
213            .lock()
214            .unwrap_or_else(|poisoned| poisoned.into_inner())
215            .retain(|(source, _), (_, fork)| !deleted.contains(source) && !deleted.contains(fork));
216        self.inner
217            .sessions
218            .steer_order
219            .lock()
220            .unwrap_or_else(|poisoned| poisoned.into_inner())
221            .retain(|session, _| !deleted.contains(session));
222        self.inner
223            .compaction
224            .manual
225            .lock()
226            .unwrap_or_else(|poisoned| poisoned.into_inner())
227            .retain(|session, _| !deleted.contains(session));
228    }
