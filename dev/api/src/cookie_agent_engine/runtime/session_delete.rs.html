1//! `session.delete`: deleting a root session's whole tree from disk.
2
3use std::collections::HashSet;
4
5use cookie_agent_protocol::SessionId;
6
7use super::{Engine, EngineError};
8use crate::session::SessionError;
9
10impl Engine {
11    /// Deletes the tree rooted at `session_id` (events, metadata, artifacts,
12    /// and every delegated session) and returns the deleted sessions, the
13    /// root first. Every connection is told through `sessions.changed`.
14    ///
15    /// Refused for a delegated session ([`EngineError::NotARootSession`]),
16    /// while a session of the tree is busy
17    /// ([`EngineError::SessionTreeRunning`]), while another process holds
18    /// the tree ([`EngineError::SessionOwnedByAnotherProcess`]), and when
19    /// the root's last event is no longer `expected_last_seq`
20    /// ([`EngineError::StaleDelete`]). A refusal changes nothing.
21    pub async fn delete_session(
22        &self,
23        session_id: SessionId,
24        expected_last_seq: u64,
25    ) -> Result<Vec<SessionId>, EngineError> {
26        self.ensure_not_shutting_down()?;
27        let root = self.inner.store.root_of(session_id)?;
28        if root != session_id {
29            return Err(EngineError::NotARootSession {
30                session_id,
31                root_session_id: root,
32            });
33        }
34        let deleted = if self.inner.store.holds_tree(root) {
35            // On the root's actor, after every request already queued for
36            // it. Nothing reconciles afterwards: there is nothing left.
37            self.on_actor_unreconciled(root, move |engine| {
38                engine.delete_tree_direct(root, expected_last_seq)
39            })
40            .await?
41        } else {
42            // A tree this process does not hold is not adopted: adoption
43            // would repair it, appending events, before it is deleted.
44            let engine = self.clone();
45            crate::blocking::off_worker(move || engine.delete_tree_direct(root, expected_last_seq))?
46        };
47        self.forget_deleted_tree(root, &deleted);
48        Ok(deleted)
49    }
50
51    fn delete_tree_direct(
52        &self,
53        root: SessionId,
54        expected_last_seq: u64,
55    ) -> Result<Vec<SessionId>, EngineError> {
56        // Serializes with an adoption of the tree, which would otherwise be
57        // caught halfway.
58        let adoption_lock = self.inner.store.adoption_lock(root);
59        let _adoption = adoption_lock
60            .lock()
61            .unwrap_or_else(|poisoned| poisoned.into_inner());
62        let deletion = self
63            .inner
64            .store
65            .begin_tree_deletion(root)
66            .map_err(|error| match error {
67                SessionError::SessionLocked(_) => EngineError::SessionOwnedByAnotherProcess(root),
68                SessionError::TreeBusy(_) => EngineError::SessionTreeRunning { root, busy: root },
69                error => error.into(),
70            })?;
71        // A running tree is refused as running even when the caller's guard
72        // is stale too: a streaming run moves the root past what was listed.
73        let members = deletion.members().iter().copied().collect::<HashSet<_>>();
74        if deletion.was_held()
75            && let Some(busy) = self.busy_member(&members).or(deletion.running_member())
76        {
77            return Err(EngineError::SessionTreeRunning { root, busy });
78        }
79        let current_last_seq = deletion.root_last_seq();
80        if current_last_seq != expected_last_seq {
81            return Err(EngineError::StaleDelete {
82                session_id: root,
83                current_last_seq,
84            });
85        }
86        // Nothing of the tree may keep running or keep a file open in it.
87        self.kill_deleted_background_jobs(&members);
88        self.inner.uploads.forget_tree(root);
89        self.inner.artifacts.forget_tree(root);
90        Ok(deletion.commit()?)
91    }
92
93    /// A session of `members` with an active run, a compaction or run
94    /// start holding it, or a place in the delegation queue.
95    fn busy_member(&self, members: &HashSet<SessionId>) -> Option<SessionId> {
96        let running = self
97            .inner
98            .sessions
99            .active
100            .lock()
101            .unwrap_or_else(|poisoned| poisoned.into_inner())
102            .values()
103            .map(|run| run.session)
104            .find(|session| members.contains(session));
105        running
106            .or_else(|| {
107                self.inner
108                    .compaction
109                    .in_progress
110                    .lock()
111                    .unwrap_or_else(|poisoned| poisoned.into_inner())
112                    .iter()
113                    .copied()
114                    .find(|session| members.contains(session))
115            })
116            .or_else(|| {
117                self.inner
118                    .delegation
119                    .queue
120                    .lock()
121                    .unwrap_or_else(|poisoned| poisoned.into_inner())
122                    .iter()
123                    .copied()
124                    .find(|session| members.contains(session))
125            })
126    }
127
128    /// Forgets what the engine keeps for a deleted tree's sessions. The
129    /// actors go too; requests still queued on them find no session.
130    fn forget_deleted_tree(&self, root: SessionId, deleted: &[SessionId]) {
131        let deleted = deleted.iter().copied().collect::<HashSet<_>>();
132        self.forget_deleted_sessions(&deleted);
133        self.inner
134            .approvals
135            .pending
136            .lock()
137            .unwrap_or_else(|poisoned| poisoned.into_inner())
138            .retain(|(session, _), _| !deleted.contains(session));
139        self.inner
140            .approvals
141            .permission_modes
142            .lock()
143            .unwrap_or_else(|poisoned| poisoned.into_inner())
144            .remove(&root);
145        self.inner.grant_journals.forget(root);
146        self.rebuild_visible_tree_grants();
147        self.inner
148            .sessions
149            .forks
150            .lock()
151            .unwrap_or_else(|poisoned| poisoned.into_inner())
152            .retain(|(source, _), (_, fork)| !deleted.contains(source) && !deleted.contains(fork));
153        self.inner
154            .sessions
155            .steer_order
156            .lock()
157            .unwrap_or_else(|poisoned| poisoned.into_inner())
158            .retain(|session, _| !deleted.contains(session));
159        self.inner
160            .compaction
161            .manual
162            .lock()
163            .unwrap_or_else(|poisoned| poisoned.into_inner())
164            .retain(|session, _| !deleted.contains(session));
165    }
