1//! Firing cron jobs (`docs/site/specs/cron.md` sections 4 and 5): the
2//! scheduler task, in-session and new-session fires, runs now, what the
3//! daemon finds at start, and the run outcomes that count failures.
4//!
5//! A fire is keyed by its job and scheduled time. It is written to the
6//! job's history, as pending, before it is delivered, under the store's
7//! lock, so it is claimed once even by two daemons sharing the data root.
8//! Delivery sends the prompt as a producer message owned by the job's
9//! source (`cron:<job id>`, see [`cron_owner`]) with the fire's key as the
10//! idempotency key, so a delivery retried after a restart is accepted once.
11//! A fire's outcome is the run its prompt lands in, recorded when that run
12//! ends.
13

32/// The longest the scheduler sleeps before a fire before it looks again, so
33/// that a clock change or a suspended host is noticed. With no fire ahead it
34/// sleeps until the jobs change.
35const SCHEDULER_MAX_SLEEP: Duration = Duration::from_secs(60);
36/// The most fires a `daemon was down` entry counts.
37const MISSED_COUNT_LIMIT: u64 = 1_000_000;
38/// The producer name of every fire.
39const FIRE_PRODUCER_NAME: &str = "scheduled";

151/// Removes or disables `jobs[index]`, which has nothing left to fire: an
152/// agent's job is deleted, yours is kept, disabled as finished.
153fn finish(jobs: &mut Vec<StoredJob>, index: usize) {
154    if jobs[index].agent_created() {
155        jobs.remove(index);
156    } else if jobs[index].enabled {
157        jobs[index].disable(StoredDisabled::Finished);
158    }
159}

161/// Finishes `jobs[index]` when it is enabled with no next fire, as a
162/// one-shot is after its time.
163fn finish_if_done(jobs: &mut Vec<StoredJob>, index: usize) {
164    if jobs[index].enabled && jobs[index].next_fire.is_none() {
165        finish(jobs, index);
166    }
167}

169/// Records the outcome of the fire keyed `key` of job `id`: delivered, which
170/// counts toward `max_runs` unless it is a run now, or failed, which counts
171/// toward the failure limit unless it is a run now.
172fn settle_fire(
173    jobs: &mut Vec<StoredJob>,
174    id: &CronJobId,
175    key: &str,
176    result: &Result<Delivery, String>,
177    failure_limit: u32,
178) {
179    let Some(index) = jobs.iter().position(|job| job.id == *id) else {
180        return;
181    };
182    let job = &mut jobs[index];
183    let Some(fire) = job
184        .history
185        .iter_mut()
186        .rev()
187        .find(|fire| fire.key.as_deref() == Some(key))
188    else {
189        return;
190    };
191    if !matches!(fire.outcome, StoredOutcome::Pending) {
192        return;
193    }
194    let counted = !fire.run_now;
195    match result {
196        Ok(delivery) => {
197            fire.outcome = StoredOutcome::Delivered;
198            fire.session_id = Some(delivery.session);
199            fire.message_id = delivery.message;
200            if let Some(thread) = fire.thread_key.clone() {
201                let at = fire.fired_at;
202                job.threads.record(&thread, delivery.session, at);
203            }
204            if counted {
205                job.runs = job.runs.saturating_add(1);
206                if job.schedule.is_one_shot() || job.max_runs.reached(job.runs) {
207                    finish(jobs, index);
208                    return;
209                }
210            }
211        }
212        Err(error) => {
213            fire.outcome = StoredOutcome::Failed {
214                error: error.clone(),
215            };
216            if counted {
217                job.count_failure(failure_limit);
218            }
219        }
220    }
221    if counted {
222        finish_if_done(jobs, index);
223    }
224}

226/// How long the scheduler sleeps at `now` before the earliest fire `next`:
227/// until it is due, but at least `retry_after` while the store fails, and
228/// at most [`SCHEDULER_MAX_SLEEP`]. With no fire ahead it waits for a change.
229pub(crate) fn scheduler_sleep(
230    next: Option<Timestamp>,
231    now: Timestamp,
232    retry_after: Option<Duration>,
233) -> Option<Duration> {
234    next.map(|next| {
235        let millis = next.as_millisecond().saturating_sub(now.as_millisecond());
236        Duration::from_millis(u64::try_from(millis).unwrap_or(0))
237            .max(retry_after.unwrap_or_default())
238            .min(SCHEDULER_MAX_SLEEP)
239    })
240}

242/// Records `end`, the end of `run` of `session`, on every fire awaiting
243/// its run there whose prompt `landed` in it: a failed run counts a
244/// failure, a completed one resets the count, any other end changes
245/// nothing, and a run now counts for neither.
246pub(crate) fn record_run_end(
247    jobs: &mut [StoredJob],
248    session: SessionId,
249    run: RunId,
250    end: StoredRunEnd,
251    landed: impl Fn(Option<ProducerMessageId>) -> bool,
252    failure_limit: u32,
253) {
254    for job in jobs.iter_mut() {
255        let (mut failed, mut completed) = (false, false);
256        for fire in job
257            .history
258            .iter_mut()
259            .filter(|fire| fire.awaits_run() && fire.session_id == Some(session))
260        {
261            if !landed(fire.message_id) {
262                continue;
263            }
264            fire.run_id = Some(run);
265            fire.run_end = Some(end.clone());
266            if !fire.uncounted {
267                failed |= end.status == SessionStatus::Failed;
268                completed |= end.status == SessionStatus::Completed;
269            }
270        }
271        if completed {
272            job.consecutive_failures = 0;
273        }
274        if failed {
275            job.count_failure(failure_limit);
276        }
277    }
278}

281    /// Starts the scheduler, like the residency janitor: it first records
282    /// what the daemon finds at start, then sleeps until the earliest next
283    /// fire or until the jobs change, and fires every due job.
284    pub(in crate::runtime) fn start_cron_scheduler(&self) {
285        let Some(runtime) = self.inner.runtime.clone() else {
286            return;
287        };
288        // Before anything can fire: what is pending now the last daemon left.
289        let left_pending = self.pending_cron_fires();
290        let weak = std::sync::Arc::downgrade(&self.inner);
291        let task = runtime.spawn(async move {
292            {
293                let Some(inner) = weak.upgrade() else {
294                    return;
295                };
296                let engine = Engine { inner };
297                engine.reconcile_crons_at_start(left_pending).await;
298            }
299            // While the store fails, a due fire stays due: the scheduler
300            // waits a growing delay before it tries again.
301            let mut backoff = crate::runtime::helpers::RETRY_BACKOFF.delays();
302            let mut retry_after = None;
303            loop {
304                let Some(inner) = weak.upgrade() else {
305                    break;
306                };
307                let engine = Engine { inner };
308                if engine.admission_tasks_closing() {
309                    break;
310                }
311                let sleep = scheduler_sleep(
312                    engine.inner.crons.earliest_fire(),
313                    Timestamp::now(),
314                    retry_after,
315                );
316                let wake = std::sync::Arc::clone(&engine.inner.cron_scheduler.wake);
317                drop(engine);
318                // A change between reading the earliest fire and waiting is
319                // not lost: `notify_one` keeps a permit for the next wait.
320                match sleep {
321                    Some(sleep) => tokio::select! {
322                        () = tokio::time::sleep(sleep) => {}
323                        () = wake.notified() => {}
324                    },
325                    None => wake.notified().await,
326                }
327                let Some(inner) = weak.upgrade() else {
328                    break;
329                };
330                let engine = Engine { inner };
331                // Once a restart is due, fires are left for the restarted
332                // daemon; a due fire would otherwise wake this loop at once.
333                if engine.admission_tasks_closing() || engine.restart_due() {
334                    break;
335                }
336                if engine.cron_tick(Timestamp::now()).await {
337                    retry_after = Some(backoff.next_delay());
338                } else {
339                    backoff.reset();
340                    retry_after = None;
341                }
342            }
343        });
344        *self
345            .inner
346            .cron_scheduler
347            .task
348            .lock()
349            .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(task);
350    }

362    /// The fires written but not settled, with the job each belongs to.
363    pub(crate) fn pending_cron_fires(&self) -> Vec<PendingFire> {
364        let mut pending = Vec::new();
365        for job in self.inner.crons.jobs() {
366            for fire in &job.history {
367                if let (StoredOutcome::Pending, Some(key)) = (&fire.outcome, &fire.key) {
368                    pending.push(PendingFire {
369                        job: job.clone(),
370                        key: key.clone(),
371                        fired_at: fire.fired_at,
372                        thread_key: fire.thread_key.clone(),
373                    });
374                }
375            }
376        }
377        pending
378    }

380    /// What the daemon finds at start (section 4.1): a fire written but
381    /// neither delivered nor started is `skipped (interrupted)`, or
382    /// delivered when its session shows it was; a job whose next fire passed
383    /// while the daemon was down catches up its latest missed fire within
384    /// `[cron] missed_grace`, which stays its next fire, and gets one
385    /// `skipped (daemon was down)` entry for the rest, moving to its next
386    /// future fire when nothing is caught up. Then every session a job waits
387    /// on says so.
388    ///
389    /// `left_pending` is the fires from [`Self::pending_cron_fires`], taken
390    /// before this daemon could fire any: one it fires itself while this runs
391    /// is on its way, whatever its session shows yet, and is not to be touched.
392    pub(crate) async fn reconcile_crons_at_start(&self, left_pending: Vec<PendingFire>) {
393        let started_at = self.inner.cron_scheduler.started_at;
394        let mut found = Vec::new();
395        for PendingFire {
396            job,
397            key,
398            fired_at,
399            thread_key,
400        } in left_pending
401        {
402            let delivery = self
403                .find_delivery(&job, &key, fired_at, thread_key.as_deref())
404                .await;
405            found.push((job.id.clone(), key, delivery));
406        }
407        let limit = self
408            .inner
409            .config
410            .load()
411            .runtime
412            .cron
413            .max_consecutive_failures;
414        let history = self.inner.config.load().runtime.cron.history;
415        let grace = self.inner.config.load().runtime.cron.missed_grace;
416        let result = self
417            .crons_blocking(move |_, jobs| {
418                for (id, key, delivery) in &found {
419                    match delivery {
420                        Some(delivery) => settle_fire(jobs, id, key, &Ok(*delivery), limit),
421                        None => {
422                            if let Some(index) = jobs.iter().position(|job| job.id == *id) {
423                                if let Some(fire) = jobs[index]
424                                    .history
425                                    .iter_mut()
426                                    .find(|fire| fire.key.as_deref() == Some(key.as_str()))
427                                {
428                                    fire.outcome = StoredOutcome::Skipped {
429                                        reason: StoredSkip::Interrupted,
430                                    };
431                                }
432                                finish_if_done(jobs, index);
433                            }
434                        }
435                    }
436                }
437                let zone = host_zone();
438                let mut index = 0;
439                while index < jobs.len() {
440                    let job = &mut jobs[index];
441                    match job.next_fire {
442                        Some(next) if job.enabled && next < started_at => {
443                            let missed = 1 + job.schedule.count_between(
444                                next,
445                                started_at,
446                                &zone,
447                                MISSED_COUNT_LIMIT,
448                            );
449                            let caught_up = job.schedule.caught_up(
450                                next,
451                                started_at,
452                                grace,
453                                &zone,
454                                MISSED_COUNT_LIMIT,
455                            );
456                            let skipped = missed - u64::from(caught_up.is_some());
457                            if skipped > 0 {
458                                let mut fire = StoredFire::skipped(
459                                    Some(next),
460                                    started_at,
461                                    StoredSkip::DaemonWasDown,
462                                );
463                                fire.missed = Some(skipped);
464                                job.record(fire, history);
465                                job.last_fire = Some(started_at);
466                            }
467                            if let Some(caught_up) = caught_up {
468                                // The scheduler fires it as soon as it
469                                // looks, like any due fire.
470                                job.next_fire = Some(caught_up);
471                                index += 1;
472                                continue;
473                            }
474                            job.next_fire = job.schedule.next_after(started_at, &zone);
475                            let before = jobs.len();
476                            finish_if_done(jobs, index);
477                            if jobs.len() < before {
478                                continue;
479                            }
480                        }
481                        _ => {}
482                    }
483                    index += 1;
484                }
485                Ok(())
486            })
487            .await;
488        if let Err(error) = result {
489            eprintln!("cookie-agent: cron jobs at start: {error}");
490        }
491        for session in self
492            .inner
493            .crons
494            .jobs()
495            .iter()
496            .filter(|job| job.enabled && job.next_fire.is_some())
497            .filter_map(|job| job.target.session())
498        {
499            self.refresh_live_waiting(session);
500        }
501    }

503    /// Where the pending fire `key` of `job` was delivered before a
504    /// restart, if it was: its message in the target session, or the run
505    /// session that records it. A run session created without its prompt
506    /// gets the prompt now.
507    async fn find_delivery(
508        &self,
509        job: &StoredJob,
510        key: &str,
511        fired_at: Timestamp,
512        thread_key: Option<&str>,
513    ) -> Option<Delivery> {
514        let created = || {
515            self.list_sessions()
516                .into_iter()
517                .find(|meta| {
518                    meta.cron
519                        .as_ref()
520                        .is_some_and(|cron| cron.job_id == job.id && cron.fired_at == fired_at)
521                })
522                .map(|meta| meta.session_id)
523        };
524        let session = match &job.target {
525            StoredTarget::Session(session) | StoredTarget::Subagent(session) => *session,
526            // A thread's fire went into the session its key had, or one it
527            // created.
528            StoredTarget::New(_) => match thread_key {
529                Some(thread) => created().or_else(|| {
530                    self.inner
531                        .crons
532                        .job(&job.id)
533                        .and_then(|job| job.threads.session(thread))
534                })?,
535                None => created()?,
536            },
537        };
538        let owner = cron_owner(&job.id);
539        let message = ProducerIdempotencyKey::new(key.to_owned())
540            .ok()
541            .and_then(|key| {
542                self.goal_producer_projection(session)
543                    .ok()?
544                    .accepted(&owner, &key)
545            });
546        let created_here = self
547            .inner
548            .store
549            .summary(session)
550            .ok()
551            .and_then(|summary| summary.meta.cron)
552            .is_some_and(|cron| cron.job_id == job.id && cron.fired_at == fired_at);
553        match (message, &job.target) {
554            (Some(message), _) => Some(Delivery {
555                session,
556                message: Some(message),
557            }),
558            // A session the fire created, which never got its prompt.
559            (None, StoredTarget::New(_)) if created_here => self
560                .send_fire(job, session, key, fired_at, ProducerDeliveryMode::Queue)
561                .await
562                .ok(),
563            (None, _) => None,
564        }
565    }

567    /// Fires every enabled job due at `now`. Whether the store failed to
568    /// claim a fire, which leaves it due.
569    pub(crate) async fn cron_tick(&self, now: Timestamp) -> bool {
570        let due = self.inner.crons.due(now);
571        let fires = due
572            .into_iter()
573            .map(|(id, scheduled)| self.fire_scheduled(id, scheduled, now));
574        futures_util::future::join_all(fires)
575            .await
576            .into_iter()
577            .any(|failed| failed)
578    }

580    /// Why a fire of `job` is skipped now (sections 4.2 and 4.3): its
581    /// previous fire's prompt still waits in its session, or, for a
582    /// new-session job, its previous run session still runs.
583    /// A job with a thread key follows the in-session rule, for the session
584    /// `thread_key`'s thread is.
585    fn skip_reason(&self, job: &StoredJob, thread_key: Option<&str>) -> Option<StoredSkip> {
586        let into = match (&job.target, thread_key) {
587            (StoredTarget::Session(session) | StoredTarget::Subagent(session), _) => Some(*session),
588            (StoredTarget::New(_), Some(thread)) => job.threads.session(thread),
589            (StoredTarget::New(_), None) => None,
590        };
591        if into.is_some_and(|session| self.held_reason(session).is_some()) {
592            return Some(StoredSkip::SessionHeld);
593        }
594        let previous = job
595            .history
596            .iter()
597            .rev()
598            .find(|fire| matches!(fire.outcome, StoredOutcome::Delivered))?;
599        let session = previous.session_id?;
600        let waiting = previous.message_id.is_some_and(|message| {
601            self.goal_producer_projection(session)
602                .ok()
603                .and_then(|projection| {
604                    projection
605                        .messages
606                        .iter()
607                        .find(|record| record.message_id == message)
608                        .map(|record| !record.consumed && !record.discarded)
609                })
610                .unwrap_or(false)
611        });
612        match &job.target {
613            StoredTarget::New(_) if job.thread_key_format().is_some() => {
614                waiting.then_some(StoredSkip::PreviousFireNotDelivered)
615            }
616            StoredTarget::New(_) => {
617                let running = self
618                    .inner
619                    .store
620                    .get(session)
621                    .is_ok_and(|projection| projection.status == SessionStatus::Running);
622                (running || waiting).then_some(StoredSkip::PreviousRunStillRunning)
623            }
624            StoredTarget::Session(_) | StoredTarget::Subagent(_) => {
625                waiting.then_some(StoredSkip::PreviousFireNotDelivered)
626            }
627        }
628    }

630    /// Fires `id`'s fire scheduled at `scheduled`: claims it in the
631    /// history, as skipped or pending, moves the job to its next fire, and
632    /// delivers a pending fire. Whether the store failed to claim it.
633    async fn fire_scheduled(&self, id: CronJobId, scheduled: Timestamp, now: Timestamp) -> bool {
634        let Some(job) = self.inner.crons.job(&id) else {
635            return false;
636        };
637        // A restart that is due would refuse the fire's run: leave it
638        // unclaimed for the restarted daemon, which catches it up within
639        // `[cron] missed_grace`.
640        if self.restart_due() {
641            return false;
642        }
643        let thread_key = job.thread_key_at(scheduled, &host_zone());
644        let skip = if self.inner.config.load().runtime.cron.enabled {
645            self.skip_reason(&job, thread_key.as_deref())
646        } else {
647            Some(StoredSkip::FiringDisabled)
648        };
649        let history = self.inner.config.load().runtime.cron.history;
650        let claimed = self
651            .crons_blocking(move |_, jobs| {
652                let Some(index) = jobs.iter().position(|job| job.id == id) else {
653                    return Ok(None);
654                };
655                let job = &mut jobs[index];
656                let claimed_before = job
657                    .history
658                    .iter()
659                    .any(|fire| !fire.run_now && fire.scheduled_at == Some(scheduled));
660                if !job.enabled || job.next_fire != Some(scheduled) || claimed_before {
661                    return Ok(None);
662                }
663                let fired_at = now;
664                job.last_fire = Some(fired_at);
665                job.next_fire = job
666                    .schedule
667                    .next_after(fired_at.max(scheduled), &host_zone());
668                if let Some(reason) = skip {
669                    job.record(
670                        StoredFire::skipped(Some(scheduled), fired_at, reason),
671                        history,
672                    );
673                    finish_if_done(jobs, index);
674                    return Ok(None);
675                }
676                let key = format!("{}:{scheduled}", job.id);
677                let mut fire =
678                    StoredFire::with_outcome(Some(scheduled), fired_at, StoredOutcome::Pending);
679                fire.key = Some(key.clone());
680                fire.thread_key.clone_from(&thread_key);
681                job.record(fire, history);
682                Ok(Some(Claim {
683                    job: job.clone(),
684                    key,
685                    fired_at,
686                    thread_key,
687                }))
688            })
689            .await;
690        match claimed {
691            Ok(Some(claim)) => {
692                let _ = self.deliver_claim(claim).await;
693                false
694            }
695            Ok(None) => false,
696            Err(error) => {
697                eprintln!("cookie-agent: cron job: {error}");
698                true
699            }
700        }
701    }

703    /// Delivers a claimed fire and records its outcome. The run the prompt
704    /// landed in may have ended before the fire was recorded delivered,
705    /// when nothing awaited it yet; its end is recorded now instead.
706    async fn deliver_claim(&self, claim: Claim) -> Result<Delivery, EngineError> {
707        let result = self
708            .deliver(
709                &claim.job,
710                &claim.key,
711                claim.fired_at,
712                claim.thread_key.as_deref(),
713            )
714            .await;
715        #[cfg(test)]
716        self.wait_for_cron_settle_hook().await;
717        let limit = self
718            .inner
719            .config
720            .load()
721            .runtime
722            .cron
723            .max_consecutive_failures;
724        let id = claim.job.id.clone();
725        let key = claim.key.clone();
726        let recorded = result.clone();
727        self.crons_blocking(move |_, jobs| {
728            settle_fire(jobs, &id, &key, &recorded, limit);
729            Ok(())
730        })
731        .await?;
732        if let Ok(Delivery {
733            session,
734            message: Some(message),
735        }) = result
736            && let Some(run) = self.landed_run(session, message)
737            && let Err(error) = self.settle_cron_runs(session, run).await
738        {
739            eprintln!("cookie-agent: cron run outcome: {error}");
740        }
741        result.map_err(EngineError::ToolFailed)
742    }

745    /// that consumed it or that it started.
746    fn landed_run(&self, session: SessionId, message: ProducerMessageId) -> Option<RunId> {
747        let producers = self.goal_producer_projection(session).ok()?;
748        let record = producers

755    /// Delivers `job`'s prompt: into its session, or into a new session
756    /// made from its template.
757    async fn deliver(
758        &self,
759        job: &StoredJob,
760        key: &str,
761        fired_at: Timestamp,
762        thread_key: Option<&str>,
763    ) -> Result<Delivery, String> {
764        if let (StoredTarget::New(template), Some(thread)) = (&job.target, thread_key) {
765            return self
766                .deliver_to_thread(job, template, key, fired_at, thread)
767                .await;
768        }
769        match &job.target {
770            StoredTarget::Session(session) | StoredTarget::Subagent(session) => {
771                self.send_fire(
772                    job,
773                    *session,
774                    key,
775                    fired_at,
776                    job.delivery.unwrap_or(ProducerDeliveryMode::Queue),
777                )
778                .await
779            }
780            StoredTarget::New(template) => {
781                let session = self
782                    .create_run_session(job, template, fired_at, None)
783                    .await
784                    .map_err(|error| error.to_string())?;
785                self.send_fire(job, session, key, fired_at, ProducerDeliveryMode::Queue)
786                    .await
787            }
788        }
789    }

791    /// Delivers a fire of `job` into its thread `thread`
792    /// (`docs/site/specs/webhooks.md` section 8): the session the key has,
793    /// with the job's delivery, or a new one made from `template`, which
794    /// the job then records.
795    async fn deliver_to_thread(
796        &self,
797        job: &StoredJob,
798        template: &StoredTemplate,
799        key: &str,
800        fired_at: Timestamp,
801        thread: &str,
802    ) -> Result<Delivery, String> {
803        let id = job.id.clone();
804        let target = self
805            .thread_target(
806                &job.thread_owner(),
807                thread,
808                |engine| {
809                    engine
810                        .inner
811                        .crons
812                        .job(&id)
813                        .and_then(|job| job.threads.session(thread))
814                },
815                || async {
816                    self.create_run_session(job, template, fired_at, Some(thread))
817                        .await
818                        .map_err(|error| error.to_string())
819                },
820                |session| {
821                    let id = job.id.clone();
822                    let thread = thread.to_owned();
823                    async move {
824                        self.crons_blocking(move |_, jobs| {
825                            if let Some(job) = jobs.iter_mut().find(|job| job.id == id) {
826                                job.threads.record(&thread, session, fired_at);
827                            }
828                            Ok(())
829                        })
830                        .await
831                        .map_err(|error| error.to_string())
832                    }
833                },
834            )
835            .await?;
836        let mode = if target.created {
837            ProducerDeliveryMode::Queue
838        } else {
839            job.delivery.unwrap_or(ProducerDeliveryMode::Queue)
840        };
841        let delivered = self
842            .send_fire(job, target.session, key, fired_at, mode)
843            .await;
844        drop(target.guard);
845        delivered
846    }

848    /// Sends `job`'s marked prompt to `session` as a producer message, once
849    /// per `key`. The message starts a run on an idle session, waits for
850    /// the current turn as `queue`, or joins it as `steer`. A session that
851    /// paged out is loaded again.
852    pub(crate) async fn send_fire(
853        &self,
854        job: &StoredJob,
855        session: SessionId,
856        key: &str,
857        fired_at: Timestamp,
858        mode: ProducerDeliveryMode,
859    ) -> Result<Delivery, String> {
860        let authority = ProducerAuthority {
861            owner: cron_owner(&job.id),
862            connection_epoch: None,
863        };
864        let key = ProducerIdempotencyKey::new(key.to_owned()).map_err(|error| error.to_string())?;
865        let body = render_scheduled_job(job, fired_at);
866        let description = crate::runtime::messaging::producers::producer_description(
867            "Scheduled job: ",
868            &job.label(),
869        );
870        let producer_id = self
871            .register_producer(session, authority.clone())
872            .await
873            .map_err(|error| error.to_string())?;
874        let accepted = self
875            .on_actor(session, move |engine| {
876                engine.accept_producer_with_output_direct(
877                    &authority,
878                    ProducerSendParams {
879                        session_id: session,
880                        producer_id,
881                        mode,
882                        idempotency_key: key,
883                        description,
884                        body,
885                    },
886                    None,
887                    None,
888                )
889            })
890            .await;
891        self.drop_producer_registration(session, producer_id);
892        accepted
893            .map(|message| Delivery {
894                session,
895                message: Some(message),
896            })
897            .map_err(|error| error.to_string())
898    }

900    /// The model selection a run of `template`'s agent starts with: its
901    /// pin, else the first available model of the agent's chain.
902    pub(crate) fn template_selection(
903        &self,
904        template: &StoredTemplate,
905    ) -> Result<RunSelection, EngineError> {
906        let runtime = self.current_runtime();
907        let agents = runtime.agents_for_preset(template.preset.as_deref())?;
908        let agent = crate::policy::resolve_agent(&agents, &template.agent)?;
909        if !agent.runnable_as_root {
910            return Err(EngineError::DisabledAgent(template.agent.clone()));
911        }
912        // An ACP agent chooses its own model.
913        if agent.acp.is_some() {
914            return Ok(RunSelection {
915                agent: template.agent.clone(),
916                model: cookie_agent_protocol::acp_model_selection(&template.agent),
917                preset: template.preset.clone(),
918                acp_runtime: template.acp_runtime.clone(),
919            });
920        }
921        let model = match &template.model {
922            Some(model) => model.clone(),
923            None => agent
924                .resolved_fallback
925                .iter()
926                .find_map(|fallback| match fallback {
927                    crate::runtime_snapshot::ResolvedAgentFallback::Selection {
928                        selection, ..
929                    } if crate::runtime_snapshot::selection_available(
930                        &runtime.models,
931                        selection,
932                    ) =>
933                    {
934                        Some(selection.clone())
935                    }
936                    _ => None,
937                })
938                .ok_or(EngineError::NoRunnableModel)?,
939        };
940        Ok(RunSelection {
941            agent: template.agent.clone(),
942            model: ModelSelection {
943                model: model.model,
944                variant: model.variant,
945            },
946            preset: template.preset.clone(),
947            acp_runtime: None,
948        })
949    }

951    /// Creates the root session a new-session fire runs in (section 4.3):
952    /// from the template, in its project, with the job's permission mode
953    /// and the template's rules as its overlay, titled for the fire, and
954    /// recording the job.
955    async fn create_run_session(
956        &self,
957        job: &StoredJob,
958        template: &StoredTemplate,
959        fired_at: Timestamp,
960        thread_key: Option<&str>,
961    ) -> Result<SessionId, EngineError> {
962        self.check_cron_template(template)?;
963        let selection = self.template_selection(template)?;
964        let runtimes = crate::runtime::attachments::SessionRuntimes {
965            runtimes: template.runtimes.as_ref().map(|requests| {
966                requests
967                    .iter()
968                    .map(
969                        |request| crate::runtime::attachments::SessionRuntimeRequest {
970                            runtime_id: request.runtime_id.to_string(),
971                            capability: request.capability.as_ref().map(ToString::to_string),
972                            workspace: request.workspace.as_ref().map(std::path::PathBuf::from),
973                        },
974                    )
975                    .collect()
976            }),
977            global: template.global,
978            cwd: template.cwd.as_ref().map(std::path::PathBuf::from),
979        };
980        let generated = self
981            .inner
982            .config
983            .load()
984            .runtime
985            .session_title
986            .generate_on_first_turn;
987        let title = run_title(job, template, fired_at, thread_key, generated)
988            .map_err(EngineError::ToolFailed)?;
989        let meta = self

991                selection,
992                Some(job.permission_mode.unwrap_or(PermissionMode::AutoApproveN)),
993                template.permission_rules.clone(),
994                runtimes,
995                template.project.clone(),
996                Some(crate::runtime::sessions::SessionCreator {

998                        job_id: job.id.clone(),
999                        fired_at,
1000                        thread_key: thread_key.map(str::to_owned),

1002                    title,
1003                }),
1004                None,
1005            )
1006            .await?;
1007        Ok(meta.session_id)
1008    }

1010    /// `crons.run`, or an agent's `cron_run` of an in-session job: fires
1011    /// `id` now, as a scheduled fire would, recorded as a run now that
1012    /// counts toward neither `max_runs` nor the failure count and moves
1013    /// nothing else.
1014    pub(crate) async fn run_cron_now(&self, id: CronJobId) -> Result<CronRunResult, EngineError> {
1015        self.ensure_not_shutting_down()?;
1016        let job = self
1017            .inner
1018            .crons
1019            .job(&id)
1020            .ok_or_else(|| CronError::NotFound(id.clone()))?;
1021        let now = Timestamp::now();
1022        // A run now goes into the thread of the moment.
1023        let thread_key = job.thread_key_at(now, &host_zone());
1024        let skip = self.skip_reason(&job, thread_key.as_deref());
1025        let history = self.inner.config.load().runtime.cron.history;
1026        let claim_key = format!("{}:now:{}", job.id, uuid::Uuid::now_v7());
1027        let claim = self
1028            .crons_blocking(move |_, jobs| {
1029                let job = jobs
1030                    .iter_mut()
1031                    .find(|job| job.id == id)
1032                    .ok_or_else(|| CronError::NotFound(id.clone()))?;
1033                let mut fire = match skip {
1034                    Some(reason) => StoredFire::skipped(None, now, reason),
1035                    None => StoredFire::with_outcome(None, now, StoredOutcome::Pending),
1036                };
1037                fire.run_now = true;
1038                fire.uncounted = true;
1039                fire.thread_key.clone_from(&thread_key);
1040                if skip.is_none() {
1041                    fire.key = Some(claim_key.clone());
1042                }
1043                job.record(fire, history);
1044                Ok(skip.is_none().then(|| Claim {
1045                    job: job.clone(),
1046                    key: claim_key,
1047                    fired_at: now,
1048                    thread_key,
1049                }))
1050            })
1051            .await?;
1052        let session_id = match claim {
1053            Some(claim) => Some(self.deliver_claim(claim).await?.session),
1054            None => None,
1055        };
1056        let job = self
1057            .inner
1058            .crons
1059            .job(&job.id)
1060            .map(|job| job.wire())
1061            .unwrap_or_else(|| job.wire());
1062        Ok(CronRunResult { job, session_id })
1063    }

1067    pub(crate) async fn record_cron_child(

1069        id: CronJobId,

1080                job.record(fire, history);

1082            Ok(())

1084        .await

1087    /// Called once a run of `session` ended: a fire whose prompt landed in
1088    /// it records how it ended, and counts it.
1089    pub(in crate::runtime) fn observe_cron_run(&self, session: SessionId, run: RunId) {
1090        if !self.inner.crons.awaits_run_in(session) {
1091            return;
1092        }
1093        let Some(runtime) = self
1094            .inner
1095            .runtime
1096            .clone()
1097            .or_else(|| tokio::runtime::Handle::try_current().ok())
1098        else {
1099            return;
1100        };
1101        let engine = self.clone();
1102        let _ = self.spawn_admission_task(&runtime, async move {
1103            if let Err(error) = engine.settle_cron_runs(session, run).await {
1104                eprintln!("cookie-agent: cron run outcome: {error}");
1105            }
1106        });
1107    }

1109    /// Records the end of `run` of `session` on every fire whose prompt it
1110    /// took (section 4.4): a failed run counts a failure, a completed one
1111    /// resets the count, a cancelled or interrupted one changes nothing.
1112    /// A fire without a message, an agent's run now in a child session,
1113    /// takes the session's first run that ends.
1114    pub(crate) async fn settle_cron_runs(
1115        &self,
1116        session: SessionId,
1117        run: RunId,
1118    ) -> Result<(), EngineError> {
1119        let projection = self.inner.store.get(session)?;
1120        // A run whose fire settles late may be one the projection let go of.
1121        let Some(status) = projection.run_outcome(run)?.map(|run| run.status) else {
1122            return Ok(());
1123        };
1124        if !matches!(
1125            status,
1126            SessionStatus::Completed
1127                | SessionStatus::Failed
1128                | SessionStatus::Cancelled
1129                | SessionStatus::Interrupted
1130        ) {
1131            return Ok(());
1132        }
1133        let producers = self.goal_producer_projection(session)?;
1134        let landed = move |message: Option<ProducerMessageId>| {
1135            message.is_none_or(|message| {
1136                producers
1137                    .messages
1138                    .iter()
1139                    .find(|record| record.message_id == message)
1140                    .is_some_and(|record| {
1141                        record.consumed_run == Some(run)
1142                            || record
1143                                .admission
1144                                .is_some_and(|(admitted, _)| admitted == run)
1145                    })
1146            })
1147        };
1148        let limit = self
1149            .inner
1150            .config
1151            .load()
1152            .runtime
1153            .cron
1154            .max_consecutive_failures;
1155        let at = Timestamp::now();
1156        self.crons_blocking(move |_, jobs| {
1157            record_run_end(
1158                jobs,
1159                session,
1160                run,
1161                StoredRunEnd { status, at },
1162                landed,
1163                limit,
1164            );
1165            Ok(())
1166        })
1167        .await
1168    }
