12    pub(in crate::runtime) fn start_subagent_janitor(&self) {
13        let Some(runtime) = self.inner.runtime.clone() else {
14            return;
15        };
16        let inner = std::sync::Arc::downgrade(&self.inner);
17        let task = runtime.spawn(async move {
18            loop {
19                tokio::time::sleep(JANITOR_INTERVAL).await;
20                let Some(inner) = inner.upgrade() else {
21                    break;
22                };
23                let engine = Engine { inner };
24                if engine
25                    .inner
26                    .delegation
27                    .admission_tasks_closing
28                    .load(std::sync::atomic::Ordering::Acquire)
29                {
30                    break;
31                }
32                if let Err(error) = engine.evict_idle_subagents().await {
33                    eprintln!("subagent session janitor failed: {error}");
34                }
35                // How long a root session tree sits unused before the janitor
36                // pages it out. Viewing it again, or any command, reloads it.
37                let idle_after = engine
38                    .inner
39                    .config
40                    .load()
41                    .runtime
42                    .server
43                    .session_idle_eviction_after;
44                if let Err(error) = engine.evict_idle_roots_with(idle_after).await {
45                    eprintln!("root session janitor failed: {error}");
46                }
47                let sweeper = engine.clone();
48                if let Err(error) = crate::runtime::blocking_io::run(move || {
49                    sweeper.sweep_upload_leases(std::time::Instant::now());
50                })
51                .await
52                {
53                    eprintln!("upload lease sweep failed: {error}");
54                }
55            }
56        });
57        *self
58            .inner
59            .janitor_task
60            .lock()
61            .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(task);
62    }

65        &self,
66    ) -> Result<Vec<SessionId>, EngineError> {
67        let cap = self
68            .inner
69            .config
70            .load()
71            .runtime
72            .delegation
73            .max_resident_subagents;
74        let idle_after = self
75            .inner
76            .config

81        self.evict_idle_subagents_with(cap, idle_after).await
82    }

84    pub(in crate::runtime) async fn evict_idle_subagents_with(
85        &self,
86        cap: usize,
87        idle_after: Duration,
88    ) -> Result<Vec<SessionId>, EngineError> {
89        let delegation_admission = self.inner.delegation.admission.lock().await;
90        // The cap applies to each tree on its own: one tree's activity never
91        // pages out another tree's children (tree-local C10).
92        let over_cap = self
93            .inner
94            .store
95            .resident_subagent_counts()
96            .into_iter()
97            .filter(|(_, count)| *count > cap)
98            .map(|(root, _)| root)
99            .collect::<std::collections::HashSet<_>>();
100        if over_cap.is_empty() {
101            return Ok(Vec::new());
102        }
103        let now = Timestamp::now();
104        let mut candidates = self
105            .inner
106            .store
107            .all()
108            .into_iter()
109            .filter_map(|session| {
110                let SessionOrigin::Delegated {
111                    root_session_id, ..
112                } = session.meta.origin
113                else {
114                    return None;
115                };
116                if !over_cap.contains(&root_session_id) {
117                    return None;
118                }
119                let ended_at = self.subagent_eviction_ended_at(&session, now, idle_after)?;
120                Some((ended_at, root_session_id, session.meta.session_id))
121            })
122            .collect::<Vec<_>>();
123        candidates.sort_by_key(|(ended_at, _, session_id)| (*ended_at, *session_id));
124
125        let mut evicted = Vec::new();
126        let mut closing = Vec::new();
127        for (_, root, session_id) in candidates {
128            if self
129                .inner
130                .store
131                .resident_subagent_counts()
132                .get(&root)
133                .is_none_or(|count| *count <= cap)
134            {
135                continue;
136            }
137            let _residency = self.inner.sessions.residency_mutation.lock().await;
138            let Some(session) = self.inner.store.get_resident(session_id) else {
139                continue;
140            };
141            if self
142                .subagent_eviction_ended_at(&session, Timestamp::now(), idle_after)
143                .is_none()
144            {
145                continue;
146            }
147            #[cfg(test)]
148            let hook = self
149                .inner
150                .test_hooks
151                .janitor_before_barrier_hook
152                .lock()
153                .expect("janitor barrier hook lock poisoned")
154                .take();
155            #[cfg(test)]
156            if let Some(hook) = hook {
157                if let Some(reached) = hook
158                    .reached
159                    .lock()
160                    .expect("janitor barrier reached lock poisoned")
161                    .take()
162                {
163                    let _ = reached.send(());
164                }
165                hook.release.notified().await;
166            }
167            self.drain_actor(session_id).await?;
168            let Some(session) = self.inner.store.get_resident(session_id) else {
169                continue;
170            };
171            if self
172                .subagent_eviction_ended_at(&session, Timestamp::now(), idle_after)
173                .is_none()
174            {
175                continue;
176            }
177            let last_event_seq = session.meta.last_event_seq;
178            self.inner
179                .sessions
180                .actors
181                .lock()
182                .unwrap_or_else(|poisoned| poisoned.into_inner())
183                .remove(&session_id);
184            if self.inner.store.evict(session_id)? {
185                self.inner
186                    .store
187                    .notify_evicted_subscribers(session_id, last_event_seq);
188                self.clear_evicted_session_caches(session_id);
189                evicted.push(session_id);
190                closing.extend(self.take_acp_session_to_close(session_id));
191            }
192        }
193        drop(delegation_admission);
194        self.finish_evicted_acp_sessions(closing).await;
195        Ok(evicted)
196    }

198    pub(in crate::runtime) fn subagent_eviction_ended_at(
199        &self,
200        session: &crate::session::SessionProjection,
201        now: Timestamp,
202        idle_after: Duration,
203    ) -> Option<Timestamp> {
204        if !matches!(session.meta.origin, SessionOrigin::Delegated { .. })
205            || session.runs.is_empty()
206            || self.session_has_pending_work(session)
207        {
208            return None;
209        }
210        if let Some(record) = self
211            .inner
212            .delegation
213            .by_session
214            .lock()
215            .unwrap_or_else(|poisoned| poisoned.into_inner())
216            .get(&session.meta.session_id)
217            .copied()
218            && (!matches!(record.state, DelegationState::Finished(_))
219                || record.background && !record.notification_sent)
220        {
221            return None;
222        }
223        let ended_at = session
224            .log
225            .event_snapshot()
226            .iter()
227            .filter(|event| {
228                event.run_id.is_some() && matches!(event.payload, Event::RunFinished { .. })
229            })
230            .map(|event| event.timestamp)
231            .max()?;
232        let idle = Duration::try_from(now.duration_since(ended_at)).ok()?;
233        (idle > idle_after).then_some(ended_at)
234    }

236    /// Whether anything still needs `session` in memory: a live or queued
237    /// run, compaction, an approval, unadmitted input, or producer work
238    /// about to happen. A producer that only holds a registration does not
239    /// pin it (`Engine::producers_pin_session`).
240    fn session_has_pending_work(&self, session: &crate::session::SessionProjection) -> bool {
241        let session_id = session.meta.session_id;
242        session.status == SessionStatus::Running
243            || session
244                .runs
245                .values()
246                .any(|run| matches!(run.status, SessionStatus::Running | SessionStatus::Idle))
247            || self
248                .inner
249                .sessions
250                .active
251                .lock()
252                .unwrap_or_else(|poisoned| poisoned.into_inner())
253                .values()
254                .any(|active| active.session == session_id)
255            || self
256                .inner
257                .delegation
258                .queue
259                .lock()
260                .unwrap_or_else(|poisoned| poisoned.into_inner())
261                .contains(&session_id)
262            || self
263                .inner
264                .compaction
265                .in_progress
266                .lock()
267                .unwrap_or_else(|poisoned| poisoned.into_inner())
268                .contains(&session_id)
269            || self
270                .inner
271                .approvals
272                .pending
273                .lock()
274                .unwrap_or_else(|poisoned| poisoned.into_inner())
275                .keys()
276                .any(|(pending, _)| *pending == session_id)
277            || has_runless_pending_inputs(&session.log.event_snapshot())
278            || self.producers_pin_session(session_id)
279            || self.acp_background_running(session_id)
280    }

282    /// Pages out root session trees nobody has used for `idle_after`: no
283    /// event, no live tail, and no pending work anywhere in the tree. A tree
284    /// goes whole or not at all, children first, so an evicted root never
285    /// leaves resident children behind it. Its runtime-only settings (the
286    /// permission mode, context calibration) survive for when it reloads.
287    pub(in crate::runtime) async fn evict_idle_roots_with(
288        &self,
289        idle_after: Duration,
290    ) -> Result<Vec<SessionId>, EngineError> {
291        let delegation_admission = self.inner.delegation.admission.lock().await;
292        let now = Timestamp::now();
293        let resident = self.inner.store.all();
294        let candidates = resident
295            .iter()
296            .filter(|session| self.root_tree_idle(session, &resident, now, idle_after))
297            .map(|session| session.meta.session_id)
298            .collect::<Vec<_>>();
299        let mut evicted = Vec::new();
300        let mut closing = Vec::new();
301        for root in candidates {
302            let _residency = self.inner.sessions.residency_mutation.lock().await;
303            let still_idle = |engine: &Self| {
304                let resident = engine.inner.store.all();
305                resident.iter().any(|session| {
306                    session.meta.session_id == root
307                        && engine.root_tree_idle(session, &resident, Timestamp::now(), idle_after)
308                })
309            };
310            if !still_idle(self) {
311                continue;
312            }
313            // Drain each tree member's actor so nothing it already accepted
314            // is cut off, then re-check: the drained work may have used it.
315            let members = self
316                .inner
317                .store
318                .all()
319                .into_iter()
320                .filter(|session| root_id(&session.meta.origin, session.meta.session_id) == root)
321                .map(|session| session.meta.session_id)
322                .collect::<Vec<_>>();
323            for member in &members {
324                self.drain_actor(*member).await?;
325            }
326            if !still_idle(self) {
327                continue;
328            }
329            // Children first, root last.
330            let mut ordered = members.clone();
331            ordered.sort_by_key(|member| *member == root);
332            for member in ordered {
333                let Some(session) = self.inner.store.get_resident(member) else {
334                    continue;
335                };
336                let last_event_seq = session.meta.last_event_seq;
337                self.inner
338                    .sessions
339                    .actors
340                    .lock()
341                    .unwrap_or_else(|poisoned| poisoned.into_inner())
342                    .remove(&member);
343                if !self.inner.store.evict(member)? {
344                    // A log never persisted (an empty session) stays put.
345                    continue;
346                }
347                self.inner
348                    .store
349                    .notify_evicted_subscribers(member, last_event_seq);
350                if member == root {
351                    self.retain_registrations_on_eviction(member);
352                    self.inner
353                        .sessions
354                        .producer_projections
355                        .lock()
356                        .unwrap_or_else(|e| e.into_inner())
357                        .remove(&member);
358                    self.inner
359                        .deferral_views
360                        .lock()
361                        .unwrap_or_else(|poisoned| poisoned.into_inner())
362                        .remove(&member);
363                } else {
364                    self.clear_evicted_session_caches(member);
365                }
366                evicted.push(member);
367                closing.extend(self.take_acp_session_to_close(member));
368            }
369            if members
370                .iter()
371                .all(|member| !self.inner.store.is_resident(*member))
372            {
373                self.forget_evicted_tree(root, &members);
374            }
375        }
376        drop(delegation_admission);
377        self.finish_evicted_acp_sessions(closing).await;
378        Ok(evicted)
379    }

381    /// Closes the ACP sessions of the sessions an eviction paged out, taken
382    /// as it paged them out. An evicted session's ACP session ends as a
383    /// deleted one's does; its next turn resumes or loads it, or starts a
384    /// new one, once this one closed.
385    async fn finish_evicted_acp_sessions(
386        &self,
387        closing: Vec<crate::runtime::acp::runner::ClosingAcpSession>,
388    ) {
389        #[cfg(test)]
390        let hook = self
391            .inner
392            .test_hooks
393            .acp_eviction_close_hook
394            .lock()
395            .unwrap_or_else(|poisoned| poisoned.into_inner())
396            .take();
397        #[cfg(test)]
398        if let Some(hook) = hook {
399            if let Some(reached) = hook
400                .reached
401                .lock()
402                .unwrap_or_else(|poisoned| poisoned.into_inner())
403                .take()
404            {
405                let _ = reached.send(());
406            }
407            hook.release.notified().await;
408        }
409        for closing in closing {
410            self.finish_acp_session_close(closing).await;
411        }
412    }

414    /// Drops what the engine keeps for the sessions of a tree paged out
415    /// whole. What the tree's load handed over (its delegation records and
416    /// the registry built from them) goes with the tree's load, so the next
417    /// use loads the tree again and hands it over anew.
418    fn forget_evicted_tree(&self, root: SessionId, members: &[SessionId]) {
419        let members = members.iter().copied().collect::<HashSet<_>>();
420        self.inner.store.unload_tree(root, || {
421            self.inner.delegation_events.forget_sessions(&members);
422            self.inner
423                .delegation
424                .by_session
425                .lock()
426                .unwrap_or_else(|poisoned| poisoned.into_inner())
427                .retain(|child, record| {
428                    !members.contains(child) && !members.contains(&record.parent_session_id)
429                });
430        });
431        self.inner.memory_seen.forget_sessions(&members);
432        self.inner
433            .sessions
434            .forks
435            .lock()
436            .unwrap_or_else(|poisoned| poisoned.into_inner())
437            .retain(|(source, _), _| !members.contains(source));
438        // A steer holding its session's order keeps it.
439        self.inner
440            .sessions
441            .steer_order
442            .lock()
443            .unwrap_or_else(|poisoned| poisoned.into_inner())
444            .retain(|session, order| !members.contains(session) || Arc::strong_count(order) > 1);
445    }

447    /// Whether `session` is a root whose whole resident tree has been unused
448    /// for `idle_after`.
449    fn root_tree_idle(
450        &self,
451        session: &crate::session::SessionProjection,
452        resident: &[Arc<crate::session::SessionProjection>],
453        now: Timestamp,
454        idle_after: Duration,
455    ) -> bool {
456        if !matches!(session.meta.origin, SessionOrigin::Root) {
457            return false;
458        }
459        let root = session.meta.session_id;
460        let unused = |member: &crate::session::SessionProjection| {
461            // Cheap checks first: the pending-work check scans the log.
462            member.log.last_event().is_some_and(|event| {
463                Duration::try_from(now.duration_since(event.timestamp))
464                    .is_ok_and(|idle| idle > idle_after)
465            }) && !self.inner.store.has_live_tail(member.meta.session_id)
466                && !self.session_has_pending_work(member)
467        };
468        unused(session)
469            && resident
470                .iter()
471                .filter(|member| {
472                    matches!(
473                        member.meta.origin,
474                        SessionOrigin::Delegated { root_session_id, .. } if root_session_id == root
475                    )
476                })
477                .all(|child| {
478                    unused(child)
479                        && self
480                            .subagent_eviction_ended_at(child, now, idle_after)
481                            .is_some()
482                })
483    }

485    /// Waits until `session`'s actor has run everything queued before now.
486    async fn drain_actor(&self, session: SessionId) -> Result<(), EngineError> {
487        let actor = self
488            .inner
489            .sessions
490            .actors
491            .lock()
492            .unwrap_or_else(|poisoned| poisoned.into_inner())
493            .get(&session)
494            .cloned();
495        if let Some(actor) = actor {
496            let (barrier, receiver) = SessionCommand::call(false, |_| async { Ok(()) });
497            actor
498                .send(barrier)
499                .await
500                .map_err(|_| EngineError::ActorStopped)?;
501            receiver.await.map_err(|_| EngineError::ActorStopped)??;
502        }
503        Ok(())
504    }

506    fn clear_evicted_session_caches(&self, session_id: SessionId) {
507        self.retain_registrations_on_eviction(session_id);
508        self.inner
509            .sessions
510            .producer_projections
511            .lock()
512            .unwrap_or_else(|e| e.into_inner())
513            .remove(&session_id);
514        self.inner
515            .compaction
516            .context_token_estimators
517            .lock()
518            .unwrap_or_else(|poisoned| poisoned.into_inner())
519            .remove(&session_id);
520        self.inner
521            .compaction
522            .in_progress
523            .lock()
524            .unwrap_or_else(|poisoned| poisoned.into_inner())
525            .remove(&session_id);
526        self.inner
527            .compaction
528            .deferred
529            .lock()
530            .unwrap_or_else(|poisoned| poisoned.into_inner())
531            .remove(&session_id);
532        self.inner
533            .deferral_views
534            .lock()
535            .unwrap_or_else(|poisoned| poisoned.into_inner())
536            .remove(&session_id);
537        self.inner
538            .memory_seen
539            .forget_sessions(&HashSet::from([session_id]));
540        let mut steer_order = self
541            .inner
542            .sessions
543            .steer_order
544            .lock()
545            .unwrap_or_else(|poisoned| poisoned.into_inner());
546        if steer_order
547            .get(&session_id)
548            .is_some_and(|order| Arc::strong_count(order) == 1)
549        {
550            steer_order.remove(&session_id);
551        }
552    }

555pub(in crate::runtime) fn has_runless_pending_inputs(
556    events: &[Arc<cookie_agent_protocol::StoredEvent>],
557) -> bool {
558    // Back from the end to the latest run's finish: an input admitted
559    // outside a run since then is pending unless a later record recalled it.
560    let mut recalled = Vec::new();
561    for event in events
562        .iter()
563        .rev()
564        .take_while(|event| !matches!(event.payload, Event::RunFinished { .. }))
565        .filter(|event| event.run_id.is_none())
566    {
567        match &event.payload {
568            Event::UserInputAdmitted { .. } if !recalled.contains(&event.seq) => return true,
569            Event::UserInputRecalled { user_input_seq, .. } => recalled.push(*user_input_seq),
570            _ => {}
571        }
572    }
573    false
