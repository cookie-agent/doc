103    pub(super) async fn request_model_approval(
104        &self,
105        active: &ActiveRun,
106        run: RunId,
107        input: ModelApprovalInput<'_>,
108    ) -> Result<ApprovalOutcome, EngineError> {
109        let request = approval_request_for_operation(
110            ApprovalTrigger::ModelToolApproval,
111            input.operation.clone(),
112            input
113                .operation
114                .resources()
115                .iter()
116                .zip(input.policy_labels)
117                .map(|(resource, label)| cookie_agent_protocol::DecisionTrace {
118                    action: resource.capability,
119                    normalized_resource: label
120                        .clone()
121                        .unwrap_or_else(|| UNSCOPED_PERMISSION_RESOURCE_DISPLAY.to_owned()),
122                    candidates: Vec::new(),
123                    effect: cookie_agent_protocol::PermissionEffect::Ask,
124                    precedence_reason: input
125                        .message
126                        .clone()
127                        .unwrap_or_else(|| "model requested tool approval".into()),
128                })
129                .collect(),
130            false,
131            // The user-facing approval window comes from `[approval].timeout_ms`.
132            // The internal approval agent's own `limits.timeout_ms` is only the
133            // classifier model-call budget (see `run_internal_history_agent`).
134            approval_expiry(self.inner.config.runtime.approval.timeout_ms),
135        );
136        self.await_user_approval(active, run, request, input.executor, false, input.tool)
137            .await
138    }

140    pub(super) async fn await_user_approval(
141        &self,
142        active: &ActiveRun,
143        run: RunId,
144        request: ApprovalRequest,
145        executor: PreparedExecutorCell,
146        allow_prior_grant: bool,
147        tool: ApprovalToolInput<'_>,
148    ) -> Result<ApprovalOutcome, EngineError> {
149        let approval_id = request.approval_id();
150        let session = self.inner.store.get(active.session)?;
151        let root = root_id(&session.meta.origin, active.session);
152        self.append(
153            active.session,
154            Some(run),
155            super::event_origin("engine:approvals"),
156            Event::ApprovalRequested {
157                request: request.clone(),
158            },
159        )
160        .await?;
161
162        let repetitions = doom_loop_repetitions(
163            &self.inner.store.log(active.session)?.event_snapshot(),
164            run,
165            request.operation_fingerprint(),
166        );
167        if repetitions >= 4 {
168            self.append(
169                active.session,
170                Some(run),
171                super::event_origin("engine:approvals"),
172                Event::ApprovalDoomLoopDetected {
173                    approval_id,
174                    operation_fingerprint: request.operation_fingerprint().clone(),
175                    repetitions,
176                },
177            )
178            .await?;
179            self.append(
180                active.session,
181                Some(run),
182                super::event_origin("engine:approvals"),
183                Event::ApprovalFinalized {
184                    approval_id,
185                    decision: ApprovalFinalDecision {
186                        outcome: ApprovalFinalOutcome::Rejected,
187                        source: ApprovalDecisionSource::DoomLoopGuard,
188                        reason_code: ApprovalReasonCode::DoomLoopDetected,
189                        feedback: None,
190                        tree_grant_id: None,
191                    },
192                },
193            )
194            .await?;
195            return Ok(ApprovalOutcome {
196                approved: false,
197                feedback: None,
198            });
199        }
200
201        if allow_prior_grant
202            && let Some(grant) = self
203                .inner
204                .approvals
205                .store
206                .matching(root, request.operation())
207        {
208            let decision = ApprovalInternalDecision {
209                decision: ApprovalInternalDecisionKind::Allow,
210                source: ApprovalDecisionSource::TreeGrant,
211                reason_code: ApprovalReasonCode::TreeGrantMatched,
212                evaluations: approval_evaluations(&request),
213            };
214            self.append(
215                active.session,
216                Some(run),
217                super::event_origin("engine:approvals"),
218                Event::ApprovalEvaluated {
219                    approval_id,
220                    decision,
221                },
222            )
223            .await?;
224            self.append(
225                active.session,
226                Some(run),
227                super::event_origin("engine:approvals"),
228                Event::ApprovalFinalized {
229                    approval_id,
230                    decision: ApprovalFinalDecision {
231                        outcome: ApprovalFinalOutcome::Approved,
232                        source: ApprovalDecisionSource::TreeGrant,
233                        reason_code: ApprovalReasonCode::TreeGrantMatched,
234                        feedback: None,
235                        tree_grant_id: Some(grant.grant_id),
236                    },
237                },
238            )
239            .await?;
240            return Ok(ApprovalOutcome {
241                approved: true,
242                feedback: None,
243            });
244        }
245
246        if approval_evaluations(&request)
247            .iter()
248            .any(|evaluation| evaluation.effect == cookie_agent_protocol::PermissionEffect::Deny)
249        {
250            let decision = ApprovalInternalDecision {
251                decision: ApprovalInternalDecisionKind::Deny,
252                source: ApprovalDecisionSource::Policy,
253                reason_code: ApprovalReasonCode::PolicyDenied,
254                evaluations: approval_evaluations(&request),
255            };
256            self.append(
257                active.session,
258                Some(run),
259                super::event_origin("engine:approvals"),
260                Event::ApprovalEvaluated {
261                    approval_id,
262                    decision,
263                },
264            )
265            .await?;
266            self.append(
267                active.session,
268                Some(run),
269                super::event_origin("engine:approvals"),
270                Event::ApprovalFinalized {
271                    approval_id,
272                    decision: ApprovalFinalDecision {
273                        outcome: ApprovalFinalOutcome::Rejected,
274                        source: ApprovalDecisionSource::Policy,
275                        reason_code: ApprovalReasonCode::PolicyDenied,
276                        feedback: None,
277                        tree_grant_id: None,
278                    },
279                },
280            )
281            .await?;
282            return Ok(ApprovalOutcome {
283                approved: false,
284                feedback: None,
285            });
286        }
287
288        let permission_mode = self
289            .inner
290            .approvals
291            .permission_modes
292            .lock()
293            .unwrap_or_else(|poisoned| poisoned.into_inner())
294            .get(&root)
295            .copied()
296            .unwrap_or_default();
297        if permission_mode == PermissionMode::Yolo {
298            self.append(
299                active.session,
300                Some(run),
301                super::event_origin("engine:approvals"),
302                Event::ApprovalEvaluated {
303                    approval_id,
304                    decision: ApprovalInternalDecision {
305                        decision: ApprovalInternalDecisionKind::Allow,
306                        source: ApprovalDecisionSource::Policy,
307                        reason_code: ApprovalReasonCode::YoloApproved,
308                        evaluations: approval_evaluations(&request),
309                    },
310                },
311            )
312            .await?;
313            self.append(
314                active.session,
315                Some(run),
316                super::event_origin("engine:approvals"),
317                Event::ApprovalFinalized {
318                    approval_id,
319                    decision: ApprovalFinalDecision {
320                        outcome: ApprovalFinalOutcome::Approved,
321                        source: ApprovalDecisionSource::Policy,
322                        reason_code: ApprovalReasonCode::YoloApproved,
323                        feedback: None,
324                        tree_grant_id: None,
325                    },
326                },
327            )
328            .await?;
329            return Ok(ApprovalOutcome {
330                approved: true,
331                feedback: None,
332            });
333        }
334
335        let internal_kind = match permission_mode {
336            PermissionMode::Ask => ApprovalInternalDecisionKind::Ask,
337            PermissionMode::AutoApprove
338            | PermissionMode::AutoApproveN
339            | PermissionMode::AutoApproveY => {
340                #[cfg(test)]
341                let hook = {
342                    self.inner
343                        .test_hooks
344                        .approval_evaluation_hook
345                        .lock()
346                        .expect("approval evaluation hook lock poisoned")
347                        .take()
348                };
349                #[cfg(test)]
350                if let Some(hook) = hook {
351                    if let Some(reached) = hook
352                        .reached
353                        .lock()
354                        .expect("approval evaluation reached lock poisoned")
355                        .take()
356                    {
357                        let _ = reached.send(());
358                    }
359                    hook.release.notified().await;
360                }
361                let approval_policy =
362                    self.active_internal_policy(active, InternalAgentKind::Approval)?;
363                self.evaluate_stateless_approval(
364                    active,
365                    run,
366                    &session,
367                    &approval_policy,
368                    &request,
369                    tool,
370                )
371                .await
372            }
373            PermissionMode::Yolo => unreachable!("yolo approvals resolve before prompting"),
374        };
375        let session = active.session;
376        let cancelled = active.cancellation.is_cancelled();
377        let evaluated = (request.clone(), executor.clone());
378        let transition = self
379            .on_actor(session, move |engine| {
380                let (request, executor) = evaluated;
381                engine.approval_evaluation_complete_direct(
382                    session,
383                    run,
384                    request,
385                    executor,
386                    (permission_mode, internal_kind),
387                    cancelled,
388                )
389            })
390            .await?;
391        let mut receiver = match transition {
392            ApprovalEvaluationTransition::Resolved(outcome) => return Ok(outcome),
393            ApprovalEvaluationTransition::Escalated(receiver) => receiver,
394        };
395        let expiry_wait = approval_expiry_wait(approval_constraints(&request).expires_at);
396        tokio::select! {
397            decision = &mut receiver => decision.map_err(|_| EngineError::ActorStopped),
398            _ = active.cancellation.cancelled() => {
399                let finalized = self.on_actor(session, move |engine| {
400                    engine.approval_terminal_direct(session, run, approval_id, ApprovalTerminal::Cancelled)
401                }).await?;
402                if finalized {
403                    Ok(ApprovalOutcome {
404                        approved: false,
405                        feedback: Some("cancelled".into()),
406                    })
407                } else {
408                    receiver.await.map_err(|_| EngineError::ActorStopped)
409                }
410            },
411            _ = tokio::time::sleep(expiry_wait) => {
412                let finalized = self.on_actor(session, move |engine| {
413                    engine.approval_terminal_direct(session, run, approval_id, ApprovalTerminal::Expired)
414                }).await?;
415                if finalized {
416                    Ok(ApprovalOutcome {
417                        approved: false,
418                        feedback: Some("approval expired unattended".into()),
419                    })
420                } else {
421                    receiver.await.map_err(|_| EngineError::ActorStopped)
422                }
423            }
424        }
425    }

427    async fn evaluate_stateless_approval(
428        &self,
429        active: &ActiveRun,
430        run: RunId,
431        session: &crate::session::SessionProjection,
432        policy: &FrozenInternalAgentPolicy,
433        request: &ApprovalRequest,
434        tool: ApprovalToolInput<'_>,
435    ) -> ApprovalInternalDecisionKind {
436        let events = session.log.event_snapshot();
437        let actions = request
438            .operation()
439            .capabilities()
440            .iter()
441            .map(|capability| capability.action)
442            .collect::<Vec<_>>();
443        let tree_events = self
444            .inner
445            .store
446            .resident_tree_logs(root_id(&session.meta.origin, active.session))
447            .iter()
448            .map(|log| log.event_snapshot())
449            .collect::<Vec<_>>();
450        let prior =
451            prior_approval_decisions(tree_events.iter().map(|events| events.as_slice()), &actions);
452        let prompt = approval_stateless_input(tool, latest_user_message(&events, run), &prior);
453        let history =
454            approval_stateless_history(policy.agent.composed_prompt.clone(), prompt.clone());
455        let result = self
456            .run_internal_history_agent(
457                active.session,
458                Some(run),
459                InternalAgentKind::Approval,
460                policy,
461                InternalAgentHistoryInput {
462                    history,
463                    summary_source: prompt,
464                    tools: Vec::new(),
465                    reject_non_text: true,
466                },
467                InternalAgentExecution {
468                    cancellation: &active.cancellation,
469                    actor_direct: false,
470                },
471            )
472            .await;
473        match result {
474            Ok(result) => {
475                parse_internal_approval(&result.text).unwrap_or(ApprovalInternalDecisionKind::Ask)
476            }
477            Err(_) => ApprovalInternalDecisionKind::Ask,
478        }
479    }

482fn approval_stateless_input(
483    tool: ApprovalToolInput<'_>,
484    latest_user: Option<&str>,
485    prior: &[PriorApprovalDecision],
486) -> String {
487    let tool_call = serde_json::json!({
488        "name": tool.name,
489        "normalized_parameters": canonical_approval_parameters(tool.normalized_parameters),
490    });
491    format!(
492        "{APPROVAL_USER_REQUEST_PREFIX}{}{APPROVAL_USER_REQUEST_SUFFIX}{APPROVAL_PRIOR_DECISIONS_PREFIX}{}{APPROVAL_PRIOR_DECISIONS_SUFFIX}{APPROVAL_TOOL_CALL_PREFIX}{}{APPROVAL_TOOL_CALL_SUFFIX}",
493        latest_user.unwrap_or(APPROVAL_NO_USER_MESSAGE),
494        render_prior_decisions(prior),
495        serde_json::to_string(&tool_call).expect("safe approval tool call serializes")
496    )
497}
498
499fn render_prior_decisions(prior: &[PriorApprovalDecision]) -> String {
500    if prior.is_empty() {
501        return APPROVAL_NO_PRIOR_DECISIONS.to_owned();
502    }
503    prior
504        .iter()
505        .map(|decision| {
506            let outcome = if decision.approved {
507                "approved"
508            } else {
509                "rejected"
510            };
511            let decider = match decision.source {
512                ApprovalDecisionSource::User => "user",
513                ApprovalDecisionSource::TreeGrant => "user (approve all)",
514                _ => "approval reviewer",
515            };
516            let feedback = decision
517                .feedback
518                .as_deref()
519                .map(|feedback| format!(" with feedback {feedback:?}"))
520                .unwrap_or_default();
521            format!(
522                "- {} {}: {outcome} by {decider}{feedback}\n",
523                decision.operations.join(","),
524                decision.resources.join(", "),
525            )
526        })
527        .collect()
528}

530/// The newest [`APPROVAL_PRIOR_DECISION_LIMIT`] approvals across `logs` that
531/// share a permission action with the current request and were decided by the
532/// user, a user tree grant, or the approval reviewer. Oldest first.
533fn prior_approval_decisions<'a>(
534    logs: impl IntoIterator<Item = &'a [Arc<StoredEvent>]>,
535    actions: &[cookie_agent_protocol::PermissionAction],
536) -> Vec<PriorApprovalDecision> {
537    let mut prior = Vec::new();
538    for events in logs {
539        // Finalization follows its request, so a reverse scan meets it first.
540        let mut finalized = HashMap::new();
541        let mut found = 0;
542        for event in events.iter().rev() {
543            match &event.payload {
544                Event::ApprovalFinalized {
545                    approval_id,
546                    decision,
547                } if matches!(
548                    decision.source,
549                    ApprovalDecisionSource::User
550                        | ApprovalDecisionSource::TreeGrant
551                        | ApprovalDecisionSource::InternalAgent
552                ) && matches!(
553                    decision.outcome,
554                    ApprovalFinalOutcome::Approved | ApprovalFinalOutcome::Rejected
555                ) =>
556                {
557                    finalized.insert(*approval_id, decision);
558                }
559                Event::ApprovalRequested { request } => {
560                    let Some(decision) = finalized.remove(&request.approval_id()) else {
561                        continue;
562                    };
563                    let capabilities = request.operation().capabilities();
564                    if !capabilities
565                        .iter()
566                        .any(|capability| actions.contains(&capability.action))
567                    {
568                        continue;
569                    }
570                    prior.push(PriorApprovalDecision {
571                        timestamp: event.timestamp,
572                        operations: capabilities
573                            .iter()
574                            .map(|capability| capability.operation.as_str().to_owned())
575                            .collect(),
576                        resources: request
577                            .evaluations()
578                            .iter()
579                            .map(|evaluation| {
580                                truncate_utf8(
581                                    &evaluation.trace.normalized_resource,
582                                    APPROVAL_PRIOR_DECISION_TEXT_MAX,
583                                )
584                            })
585                            .collect(),
586                        approved: decision.outcome == ApprovalFinalOutcome::Approved,
587                        source: decision.source,
588                        feedback: decision.feedback.as_ref().map(|feedback| {
589                            truncate_utf8(
590                                feedback.message.as_str(),
591                                APPROVAL_PRIOR_DECISION_TEXT_MAX,
592                            )
593                        }),
594                    });
595                    found += 1;
596                    if found == APPROVAL_PRIOR_DECISION_LIMIT {
597                        break;
598                    }
599                }
600                _ => {}
601            }
602        }
603    }
604    prior.sort_by_key(|decision| decision.timestamp);
605    let excess = prior.len().saturating_sub(APPROVAL_PRIOR_DECISION_LIMIT);
606    prior.drain(..excess);
607    prior
608}

610fn canonical_approval_parameters(value: &Value) -> Value {
611    match value {
612        Value::Array(values) => {
613            Value::Array(values.iter().map(canonical_approval_parameters).collect())
614        }
615        Value::Object(values) => {
616            let mut keys = values.keys().collect::<Vec<_>>();
617            keys.sort_unstable();
618            Value::Object(
619                keys.into_iter()
620                    .map(|key| (key.clone(), canonical_approval_parameters(&values[key])))
621                    .collect(),
622            )
623        }
624        _ => value.clone(),
625    }
626}

628fn latest_user_message(events: &[Arc<StoredEvent>], run: RunId) -> Option<&str> {
629    events
630        .iter()
631        .rev()
632        .find_map(|event| match &event.payload {
633            Event::UserInputSubmitted { input } if event.run_id == Some(run) => {
634                Some(input.as_str())
635            }
636            _ => None,
637        })
638        .or_else(|| {
639            events.iter().rev().find_map(|event| match &event.payload {
640                Event::UserInputSubmitted { input } => Some(input.as_str()),
641                _ => None,
642            })
643        })
644}

669pub(super) fn approval_request_for_operation(
670    trigger: ApprovalTrigger,
671    operation: PreparedOperationIdentity,
672    traces: Vec<cookie_agent_protocol::DecisionTrace>,
673    allow_tree_grant: bool,
674    expires_at: Option<jiff::Timestamp>,
675) -> ApprovalRequest {
676    let evaluations = operation
677        .resources()
678        .iter()
679        .zip(traces)
680        .map(|(resource, trace)| ApprovalEvaluation {
681            resource_digest: resource.binding_digest.clone(),
682            effect: trace.effect,
683            trace,
684        })
685        .collect::<Vec<_>>();
686    ApprovalRequest::new(
687        ApprovalId::new_v7(),
688        1,
689        trigger,
690        operation,
691        evaluations,
692        ApprovalConstraints {
693            allow_once: true,
694            allow_tree_grant,
695            cancellable: true,
696            expires_at,
697        },
698    )
699    .expect("prepared approval request is complete")
700}

702pub(super) fn approval_expiry(timeout_ms: u64) -> Option<jiff::Timestamp> {

708pub(super) fn approval_deadline_exhausted(expires_at: Option<jiff::Timestamp>) -> bool {
709    expires_at.is_some_and(|expires_at| expires_at <= jiff::Timestamp::now())
710}
711
712pub(super) fn approval_expiry_wait(expires_at: Option<jiff::Timestamp>) -> std::time::Duration {
713    let Some(expires_at) = expires_at else {
714        return std::time::Duration::from_secs(100 * 365 * 24 * 60 * 60);
715    };
716    let now = jiff::Timestamp::now();
717    if expires_at <= now {
718        std::time::Duration::ZERO
719    } else {
720        expires_at.duration_since(now).unsigned_abs()
721    }
722}

786            approval_id,
787            1,

789            operation,
