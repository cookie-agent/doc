1//! Applying a new configuration generation to a running engine: the
2//! settings tables, agent documents, providers, headers, and MCP servers take
3//! effect at once; `server`, `plugins`, and `runtimes` are reported as needing
4//! a restart. A run keeps what it captured at admission (its agent, result
5//! limits, retry policy, and delegation depth).

16/// The configuration parts that apply only when the daemon restarts.
17#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
18pub enum RestartRequiredPart {
19    Server,
20    Plugins,
21    Runtimes,
22}
23
24impl RestartRequiredPart {
25    #[must_use]
26    pub const fn as_str(self) -> &'static str {
27        match self {
28            Self::Server => "server",
29            Self::Plugins => "plugins",
30            Self::Runtimes => "runtimes",
31        }
32    }
33}

35/// One change of a [`Engine::apply_config_changes`] set.
36#[derive(Clone, Debug)]
37pub enum ConfigChange {
38    /// Creates (`expected_fingerprint` absent) or replaces a document.
39    Put {
40        path: ConfigPath,
41        text: String,
42        expected_fingerprint: Option<String>,
43    },
44    /// Removes a document.
45    Delete {
46        path: ConfigPath,
47        expected_fingerprint: String,
48    },
49    /// Sets one key of `config.toml`; `null` removes it.
50    Set {
51        key: Vec<String>,
52        value: serde_json::Value,
53    },
54}

57#[derive(Clone, Debug, Eq, PartialEq)]
58pub struct ConfigDocumentText {

60    pub text: String,
61    pub fingerprint: String,

73/// Where the engine's configuration stands.
74#[derive(Clone, Debug, Eq, PartialEq)]
75pub struct ConfigurationStatus {
76    /// The revision of the applied generation.
77    pub applied_revision: ConfigRevision,
78    /// Parts whose applied value differs from the one the engine opened
79    /// with.
80    pub restart_required: Vec<RestartRequiredPart>,
81    /// Why the last reload failed, until one succeeds.
82    pub error: Option<String>,
83    /// What the applied configuration names that does not exist yet, such
84    /// as a delegation target no agent has: an ACP agent whose runtime has

106    #[must_use]
107    pub fn configuration_status(&self) -> ConfigurationStatus {
108        let applied = self.inner.config.load();
109        let started = &self.inner.started_config;
110        let mut restart_required = Vec::new();
111        let (now, then) = (&applied.runtime.server, &started.runtime.server);
112        // A token the daemon created at start and wrote to the file is the
113        // one it serves with.
114        let served_token = self
115            .inner
116            .served_server_token
117            .get()
118            .map(String::as_str)
119            .or(then.token.as_deref());
120        if now.host != then.host || now.port != then.port || now.token.as_deref() != served_token {
121            restart_required.push(RestartRequiredPart::Server);
122        }
123        if applied.plugins != started.plugins {
124            restart_required.push(RestartRequiredPart::Plugins);
125        }
126        // A runtime's MCP servers and ACP agents reach it as they change
127        // (`runtime/settings_changed`); how the daemon starts one takes a
128        // restart.
129        if runtime_launches(&applied) != runtime_launches(started) {
130            restart_required.push(RestartRequiredPart::Runtimes);
131        }
132        ConfigurationStatus {
133            applied_revision: applied.revision.clone(),
134            restart_required,
135            diagnostics: self.configuration_diagnostics(),
136            native_compaction: self.inner.native_compaction.marks(),
137            deferred_tools: self.inner.deferred_tools_record.marks(),
138            error: self
139                .inner
140                .config_error
141                .lock()
142                .unwrap_or_else(|poisoned| poisoned.into_inner())
143                .clone(),
144        }
145    }

147    /// The ACP declarations a daemon agent document of the same ID shadows,
148    /// one without an `acp` block or one whose `acp.agent` names another
149    /// declaration: the document's agent keeps the ID and the declarations
150    /// are ignored. Then the documents whose `acp.agent` no runtime
151    /// declares yet.
152    fn shadowed_acp_declarations(&self) -> Vec<String> {
153        let config = self.inner.config.load();
154        let declarations = self.inner.acp.runtime_agents();
155        let mut by_agent =
156            std::collections::BTreeMap::<cookie_agent_protocol::AgentId, Vec<String>>::new();
157        for agent in &declarations {
158            if config.agents.get(&agent.agent).is_some_and(|document| {
159                document
160                    .frontmatter
161                    .acp
162                    .as_ref()
163                    .is_none_or(|settings| settings.declaration(&document.id) != &agent.agent)
164            }) {
165                by_agent
166                    .entry(agent.agent.clone())
167                    .or_default()
168                    .push(agent.runtime.to_string());
169            }
170        }
171        let mut diagnostics = by_agent
172            .into_iter()
173            .map(|(agent, runtimes)| {
174                let declared_on = match &runtimes[..] {
175                    [runtime] => format!("runtime `{runtime}`"),
176                    _ => format!("runtimes {}", runtimes.join(", ")),
177                };
178                match config
179                    .agents
180                    .get(&agent)
181                    .and_then(|document| document.frontmatter.acp.as_ref())
182                    .and_then(|settings| settings.agent.as_ref())
183                {
184                    Some(other) => format!(
185                        "ACP agent `{agent}` of {declared_on} is ignored: the agent document \
186                         `{agent}` runs `{other}` (`acp.agent`), so `{agent}` is that \
187                         document's agent"
188                    ),
189                    None => format!(
190                        "ACP agent `{agent}` of {declared_on} is ignored: the agent document \
191                         `{agent}` has no `acp` block, so `{agent}` is a daemon agent; give the \
192                         document an `acp: {{}}` block to make it the ACP agent's"
193                    ),
194                }
195            })
196            .collect::<Vec<_>>();
197        for document in config.agents.values() {
198            let Some(declaration) = document
199                .frontmatter
200                .acp
201                .as_ref()
202                .and_then(|settings| settings.agent.as_ref())
203            else {
204                continue;
205            };
206            if !declarations.iter().any(|agent| &agent.agent == declaration) {
207                diagnostics.push(format!(
208                    "agent `{}` runs ACP agent `{declaration}` (`acp.agent`), which no runtime \
209                     declares yet; it is an agent once one does",
210                    document.id
211                ));
212            }
213        }
214        diagnostics
215    }

217    /// The diagnostics of the published agents: those of the shared agents,
218    /// then each preset's own.
219    fn configuration_diagnostics(&self) -> Vec<String> {
220        let published = self.current_runtime();
221        let mut diagnostics = self.shadowed_acp_declarations();
222        diagnostics.extend(published.agents.diagnostics());
223        for (preset, registry) in &published.agent_presets {
224            for diagnostic in registry.diagnostics() {
225                if !diagnostics.contains(&diagnostic) {
226                    diagnostics.push(format!("preset `{preset}`: {diagnostic}"));
227                }
228            }
229        }
230        diagnostics.extend(self.official_skills_diagnostic());
231        diagnostics.extend(self.official_agents_diagnostics());
232        diagnostics.extend(self.inner.machines.save_diagnostic());
233        diagnostics.extend(self.default_agent_diagnostic(&published));
234        diagnostics.extend(self.approval_cache_diagnostics(&published));
235        diagnostics.extend(self.shell_environment_diagnostics());
236        diagnostics.truncate(256);
237        diagnostics
238    }

257    /// sessions that choose no agent fall back to the first one that can,

259    fn default_agent_diagnostic(
260        &self,
261        published: &crate::runtime_snapshot::PublishedRuntime,
262    ) -> Option<String> {
263        // Unset, the default is `primary`, and a daemon without it falls
264        // back to its first root agent quietly, as it always has.
265        self.inner.config.load().runtime.agents.default.as_ref()?;
266        let default = published.agents.default_agent();

271        (!runnable).then(|| {
272            format!(
273                "the default agent `{default}` (`[agents] default`) cannot start a session: it does not exist or cannot run as a root; run `cookie agents update`, or write `agents/{default}.md`"
274            )

278    /// What `[approval] cache_ttl` does on each model the approval reviewer
279    /// names: a TTL its provider does not offer, or a provider it cannot
280    /// change (approval reviewer spec section 5). `${parent_model}` is the
281    /// reviewed run's model, unknown until a review.
282    fn approval_cache_diagnostics(
283        &self,
284        published: &crate::runtime_snapshot::PublishedRuntime,
285    ) -> Vec<String> {
286        let ttl = self.inner.config.load().runtime.approval.cache_ttl;
287        let Ok(id) =
288            cookie_agent_protocol::AgentId::new(cookie_agent_config::BUILT_IN_APPROVAL_AGENT_ID)
289        else {
290            return Vec::new();
291        };
292        let Some(approval) = published.agents.get(&id) else {
293            return Vec::new();
294        };
295        approval
296            .resolved_fallback
297            .iter()
298            .filter_map(|fallback| match fallback {
299                crate::runtime_snapshot::ResolvedAgentFallback::Selection { selection, .. } => {
300                    Some(selection)
301                }
302                crate::runtime_snapshot::ResolvedAgentFallback::ParentModel { .. } => None,
303            })
304            .filter_map(|selection| {
305                let binding = crate::model_snapshots::binding_for_selection(
306                    &published.current_manifest,
307                    &published.models,
308                    selection,
309                )
310                .ok()?;
311                let provider = crate::policy::resolve_cache_strategy(
312                    Some(approval),
313                    &binding,
314                    published,
315                    cookie_agent_config::CacheLayer::Root,
316                )
317                .ok()
318                .flatten();
319                let (_, note) = crate::policy::reviewer_cache_strategy(provider, &binding, ttl);
320                let model = selection.model.to_string();
321                Some(match note? {
322                    crate::policy::ReviewerCacheNote::Substituted { used, why } => format!(
323                        "`[approval] cache_ttl = \"{}\"`: the reviewer's model `{model}` caches for {used}, {why}",
324                        ttl.as_str()
325                    ),
326                    crate::policy::ReviewerCacheNote::NoEffect(why) => format!(
327                        "`[approval] cache_ttl = \"{}\"` has no effect on the reviewer's model `{model}`: {why}",
328                        ttl.as_str()
329                    ),
330                })
331            })
332            .collect()
333    }

350    /// The configuration documents on disk, which may differ from the
351    /// applied ones until a reload.
352    pub async fn configuration_documents(&self) -> Result<ConfigDocuments, EngineError> {
353        let root = self.configuration_root()?;
354        let tree = tokio::task::spawn_blocking(move || ConfigTree::read(Some(&root)))
355            .await
356            .map_err(|error| EngineError::Runtime(error.to_string()))?
357            .map_err(config_error)?;
358        Ok(ConfigDocuments {
359            revision: tree.revision(),
360            documents: tree
361                .documents()
362                .map(|(path, bytes)| ConfigDocumentText {
363                    path: path.clone(),
364                    text: String::from_utf8_lossy(bytes).into_owned(),
365                    fingerprint: ConfigTree::fingerprint(bytes),
366                })
367                .collect(),
368            empty_presets: tree.empty_presets().map(str::to_owned).collect(),
369        })
370    }

372    /// Applies `changes` to the configuration on disk as one set: each is
373    /// checked against `expected_revision` and its own fingerprint, the
374    /// result is validated whole, written, and applied. Returns the new
375    /// revision.
376    pub async fn apply_config_changes(
377        &self,
378        expected_revision: ConfigRevision,
379        changes: Vec<ConfigChange>,
380    ) -> Result<ConfigRevision, EngineError> {
381        let edits_config = changes.iter().any(|change| match change {
382            ConfigChange::Put { path, .. } | ConfigChange::Delete { path, .. } => {
383                *path == ConfigPath::config()
384            }
385            ConfigChange::Set { .. } => false,
386        });
387        if edits_config
388            && changes
389                .iter()
390                .any(|change| matches!(change, ConfigChange::Set { .. }))
391        {
392            return Err(EngineError::ConfigChangeRefused(
393                "a set and a put or delete of config.toml cannot go together".into(),
394            ));
395        }
396        self.edit_configuration(move |tree, candidate| {
397            if tree.revision() != expected_revision {
398                return Err(EngineError::ConfigConflict(
399                    "the configuration changed since it was read".into(),
400                ));
401            }
402            for change in changes {
403                apply_change(candidate, change)?;
404            }
405            Ok(())
406        })
407        .await
408    }

410    /// Edits the configuration on disk with `edit`, which gets the tree as
411    /// read and the candidate to change, then validates, writes, and applies
412    /// the candidate. Returns the new revision.
413    pub(crate) async fn edit_configuration(
414        &self,
415        edit: impl FnOnce(&ConfigTree, &mut ConfigTree) -> Result<(), EngineError> + Send + 'static,
416    ) -> Result<ConfigRevision, EngineError> {
417        let _mutation = self.inner.config_mutation.lock().await;
418        let root = self.configuration_root()?;
419        let (revision, candidate) = tokio::task::spawn_blocking(move || {
420            let tree = ConfigTree::read(Some(&root)).map_err(config_error)?;
421            let mut candidate = tree.clone();
422            edit(&tree, &mut candidate)?;
423            let loaded = load_from_tree(&candidate).map_err(config_error)?;
424            write_tree(&tree, &candidate).map_err(|error| match error {
425                ConfigError::ChangedOnDisk(path) => EngineError::ConfigConflict(format!(
426                    "{} changed while it was written",
427                    path.display()
428                )),
429                error => config_error(error),
430            })?;
431            Ok::<_, EngineError>((candidate.revision(), loaded))
432        })
433        .await
434        .map_err(|error| EngineError::Runtime(error.to_string()))??;
435        self.apply_configuration_locked(candidate).await?;
436        Ok(revision)
437    }

439    /// Reads `config.toml` and the agent documents from disk and applies
440    /// them, for edits made by hand. A tree that fails validation leaves the
441    /// applied generation in place and is reported in the status.
442    pub async fn reload_configuration(&self) -> Result<ConfigurationStatus, EngineError> {
443        let _mutation = self.inner.config_mutation.lock().await;
444        let root = self.configuration_root()?;
445        let candidate = tokio::task::spawn_blocking(move || {
446            ConfigTree::read(Some(&root)).and_then(|tree| load_from_tree(&tree))
447        })
448        .await
449        .map_err(|error| EngineError::Runtime(error.to_string()))?;
450        match candidate {
451            Ok(candidate) => {
452                self.apply_configuration_locked(candidate).await?;
453                // The rc files may have changed too, which nothing here
454                // can see.
455                self.inner.runtimes.refresh_shell_environments();
456                Ok(self.configuration_status())
457            }
458            Err(error) => {
459                *self
460                    .inner
461                    .config_error
462                    .lock()
463                    .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(error.to_string());
464                Err(EngineError::Config(Box::new(error)))
465            }
466        }
467    }

469    /// Applies a validated generation; the caller holds `config_mutation`.
470    pub(crate) async fn apply_configuration_locked(
471        &self,
472        mut candidate: LoadedConfiguration,
473    ) -> Result<(), EngineError> {
474        let previous = self.inner.config.load_full();
475        // Skills are read once at start; a reload keeps them.
476        candidate.skills = previous.skills.clone();
477        let mut reasons = vec![RuntimeChangeReason::ConfigReloaded];
478        if agent_fingerprints(&previous) != agent_fingerprints(&candidate) {
479            reasons.push(RuntimeChangeReason::AgentReloaded);
480        }
481        let candidate = Arc::new(candidate);
482        // The runtime mutation lock may be held through another change's
483        // recompile, and this one recompiles too: both stay off the async
484        // workers.
485        let engine = self.clone();
486        let applied = Arc::clone(&candidate);
487        super::on_blocking_pool(move || {
488            let _runtime = engine
489                .inner
490                .runtime_mutation
491                .lock()
492                .unwrap_or_else(|poisoned| poisoned.into_inner());
493            let published = engine.current_runtime();
494            let (_, publication) = engine.inner.model_manager.reload_configuration(
495                applied.runtime.providers.clone(),
496                applied.runtime.headers.clone(),
497                Arc::clone(published.models.catalog()),
498                |models| {
499                    engine
500                        .prepare_publication(&applied, models, &published, reasons)
501                        .map_err(|_| cookie_agent_models::ModelManagerError::RuntimeCompileFailed)
502                },
503            )?;
504            engine.inner.config.store(Arc::clone(&applied));
505            engine.publish(publication);
506            engine.inner.runtimes.update_settings(&applied);
507            Ok(())
508        })
509        .await?;
510        // The publication also wakes the background sync; awaiting one here
511        // makes the files current when the apply or reload answers.
512        self.sync_agent_memory().await;
513        self.drop_unconfigured_runtime_records(&candidate).await;
514        *self
515            .inner
516            .config_error
517            .lock()
518            .unwrap_or_else(|poisoned| poisoned.into_inner()) = None;
519        self.rebase_mcp_servers(&candidate).await;
520        Ok(())
521    }

523    /// Puts the new user layer under the MCP store's runtime overlay and
524    /// brings the registry in line with what changed.
525    async fn rebase_mcp_servers(&self, config: &LoadedConfiguration) {
526        let _mutation = self.inner.mcp_mutation.lock().await;
527        let changes = self
528            .inner
529            .config_store
530            .lock()
531            .unwrap_or_else(|poisoned| poisoned.into_inner())
532            .rebase(config.user_mcp_servers.clone());
533        // A server that fails to start reports through the MCP failure
534        // notifications like any other.
535        for (name, server) in changes {
536            let _ = match server {
537                Some(server) => self.inner.mcp.upsert_server(name, server).await,
538                None => self.inner.mcp.remove_server(&name).await,
539            };
540        }
541    }

544/// Each agent document's fingerprint, shared and per preset.
545fn agent_fingerprints(
546    config: &LoadedConfiguration,
547) -> BTreeMap<(Option<&str>, &str), &cookie_agent_models::Sha256Digest> {
548    config
549        .agents
550        .iter()
551        .map(|(id, document)| ((None, id.as_str()), &document.document_fingerprint))
552        .chain(config.agent_presets.iter().flat_map(|(preset, agents)| {
553            agents.iter().map(move |(id, document)| {
554                (
555                    (Some(preset.as_str()), id.as_str()),
556                    &document.document_fingerprint,
557                )
558            })
559        }))
560        .collect()
561}

567fn apply_change(candidate: &mut ConfigTree, change: ConfigChange) -> Result<(), EngineError> {
568    let current = |candidate: &ConfigTree, path: &ConfigPath| {
569        candidate.get(path).map(ConfigTree::fingerprint)
570    };
571    match change {
572        ConfigChange::Put {
573            path,
574            text,
575            expected_fingerprint,
576        } => {
577            if current(candidate, &path) != expected_fingerprint {
578                return Err(EngineError::ConfigConflict(match expected_fingerprint {
579                    None => format!("{path} already exists"),
580                    Some(_) => format!("{path} changed since it was read"),
581                }));
582            }
583            candidate
584                .put(path, text.into_bytes())
585                .map_err(|error| EngineError::ConfigChangeRefused(error.to_string()))
586        }
587        ConfigChange::Delete {
588            path,
589            expected_fingerprint,
590        } => {
591            if current(candidate, &path).as_deref() != Some(expected_fingerprint.as_str()) {
592                return Err(EngineError::ConfigConflict(format!(
593                    "{path} changed since it was read"
594                )));
595            }
596            candidate.delete(&path);
597            Ok(())
598        }
599        ConfigChange::Set { key, value } => candidate
600            .set_key_json(&key, &value)
601            .map_err(|error| EngineError::ConfigChangeRefused(error.to_string())),
602    }
603}

615/// How the daemon starts each runtime `config` has a template for.
616/// What the daemon says of runtime `runtime`'s shell environment, as it
617/// knows it, when the runtime's configuration asks for `capture`: a failed
618/// capture, or a runtime too old to capture at all.
619pub(super) fn shell_environment_diagnostic(
620    runtime: &str,
621    known: crate::runtimes::RuntimeShellEnvironment,
622    capture: cookie_agent_runtime_api::protocol::ShellCapture,
623) -> Option<String> {
624    const CAPTURES_SINCE: cookie_agent_protocol::SemVer =
625        cookie_agent_protocol::SemVer::new(1, 12, 0);
626    if let Some(error) = known.report.and_then(|report| report.error) {
627        return Some(format!(
628            "runtime `{runtime}`: shell environment capture failed ({error}); using the \
629             runtime's own environment"
630        ));
631    }
632    let version = known
633        .protocol_version
634        .filter(|version| *version < CAPTURES_SINCE)?;
635    (capture != cookie_agent_runtime_api::protocol::ShellCapture::Off).then(|| {
636        format!(
637            "runtime `{runtime}`: runtime protocol {version} does not capture the shell \
638             environment; update cookie on its machine"
639        )
640    })
641}

643fn runtime_launches(config: &LoadedConfiguration) -> Vec<RuntimeLaunch<'_>> {
644    config
