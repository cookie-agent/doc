1//! Applying a new configuration generation to a running engine: the
2//! settings tables, agent documents, providers, headers, and MCP servers take
3//! effect at once; `server`, `plugins`, and `runtimes` are reported as needing
4//! a restart. A run keeps what it captured at admission (its agent, result
5//! limits, retry policy, and delegation depth).
6
7use std::{collections::BTreeMap, sync::Arc};
8
9use cookie_agent_config::{ConfigTree, LoadedConfiguration, load_from_tree};
10use cookie_agent_protocol::{ConfigRevision, RuntimeChangeReason};
11
12use crate::{Engine, EngineError};

14/// The configuration parts that apply only when the daemon restarts.
15#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
16pub enum RestartRequiredPart {
17    Server,
18    Plugins,
19    Runtimes,
20}
21
22impl RestartRequiredPart {
23    #[must_use]
24    pub const fn as_str(self) -> &'static str {
25        match self {
26            Self::Server => "server",
27            Self::Plugins => "plugins",
28            Self::Runtimes => "runtimes",
29        }
30    }
31}
32
33/// Where the engine's configuration stands.
34#[derive(Clone, Debug, Eq, PartialEq)]
35pub struct ConfigurationStatus {
36    /// The revision of the applied generation.
37    pub applied_revision: ConfigRevision,
38    /// Parts whose applied value differs from the one the engine opened
39    /// with.
40    pub restart_required: Vec<RestartRequiredPart>,
41    /// Why the last reload failed, until one succeeds.
42    pub error: Option<String>,
43}
44
45impl Engine {
46    /// The applied configuration generation.
47    #[must_use]
48    pub fn configuration(&self) -> Arc<LoadedConfiguration> {
49        self.inner.config.load_full()
50    }
51
52    #[must_use]
53    pub fn configuration_status(&self) -> ConfigurationStatus {
54        let applied = self.inner.config.load();
55        let started = &self.inner.started_config;
56        let mut restart_required = Vec::new();
57        let (now, then) = (&applied.runtime.server, &started.runtime.server);
58        if now.host != then.host || now.port != then.port || now.token != then.token {
59            restart_required.push(RestartRequiredPart::Server);
60        }
61        if applied.plugins != started.plugins {
62            restart_required.push(RestartRequiredPart::Plugins);
63        }
64        if applied.runtimes != started.runtimes {
65            restart_required.push(RestartRequiredPart::Runtimes);
66        }
67        ConfigurationStatus {
68            applied_revision: applied.revision.clone(),
69            restart_required,
70            error: self
71                .inner
72                .config_error
73                .lock()
74                .unwrap_or_else(|poisoned| poisoned.into_inner())
75                .clone(),
76        }
77    }
78
79    /// The user root the configuration is read from and written to.
80    pub(crate) fn configuration_root(&self) -> Result<std::path::PathBuf, EngineError> {
81        self.inner
82            .config
83            .load()
84            .config_paths
85            .user
86            .as_deref()
87            .and_then(std::path::Path::parent)
88            .map(std::path::Path::to_path_buf)
89            .ok_or_else(|| {
90                EngineError::Runtime("the daemon has no user configuration directory".into())
91            })
92    }
93
94    /// Reads `config.toml` and the agent documents from disk and applies
95    /// them, for edits made by hand. A tree that fails validation leaves the
96    /// applied generation in place and is reported in the status.
97    pub async fn reload_configuration(&self) -> Result<ConfigurationStatus, EngineError> {
98        let _mutation = self.inner.config_mutation.lock().await;
99        let root = self.configuration_root()?;
100        let candidate = tokio::task::spawn_blocking(move || {
101            ConfigTree::read(Some(&root)).and_then(|tree| load_from_tree(&tree))
102        })
103        .await
104        .map_err(|error| EngineError::Runtime(error.to_string()))?;
105        match candidate {
106            Ok(candidate) => {
107                self.apply_configuration_locked(candidate).await?;
108                Ok(self.configuration_status())
109            }
110            Err(error) => {
111                *self
112                    .inner
113                    .config_error
114                    .lock()
115                    .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(error.to_string());
116                Err(EngineError::Config(Box::new(error)))
117            }
118        }
119    }
120
121    /// Applies a validated generation; the caller holds `config_mutation`.
122    pub(crate) async fn apply_configuration_locked(
123        &self,
124        mut candidate: LoadedConfiguration,
125    ) -> Result<(), EngineError> {
126        let previous = self.inner.config.load_full();
127        // Skills are read once at start; a reload keeps them.
128        candidate.skills = previous.skills.clone();
129        let mut reasons = vec![RuntimeChangeReason::ConfigReloaded];
130        if agent_fingerprints(&previous) != agent_fingerprints(&candidate) {
131            reasons.push(RuntimeChangeReason::AgentReloaded);
132        }
133        let candidate = Arc::new(candidate);
134        {
135            let _runtime = self
136                .inner
137                .runtime_mutation
138                .lock()
139                .unwrap_or_else(|poisoned| poisoned.into_inner());
140            let published = self.current_runtime();
141            let (_, publication) = self.inner.model_manager.reload_configuration(
142                candidate.runtime.providers.clone(),
143                candidate.runtime.headers.clone(),
144                Arc::clone(published.models.catalog()),
145                |models| {
146                    self.prepare_publication(&candidate, models, &published, reasons)
147                        .map_err(|_| cookie_agent_models::ModelManagerError::RuntimeCompileFailed)
148                },
149            )?;
150            self.inner.config.store(Arc::clone(&candidate));
151            self.publish(publication);
152        }
153        *self
154            .inner
155            .config_error
156            .lock()
157            .unwrap_or_else(|poisoned| poisoned.into_inner()) = None;
158        self.rebase_mcp_servers(&candidate).await;
159        Ok(())
160    }
161
162    /// Puts the new user layer under the MCP store's runtime overlay and
163    /// brings the registry in line with what changed.
164    async fn rebase_mcp_servers(&self, config: &LoadedConfiguration) {
165        let _mutation = self.inner.mcp_mutation.lock().await;
166        let changes = self
167            .inner
168            .config_store
169            .lock()
170            .unwrap_or_else(|poisoned| poisoned.into_inner())
171            .rebase(config.user_mcp_servers.clone());
172        // A server that fails to start reports through the MCP failure
173        // notifications like any other.
174        for (name, server) in changes {
175            let _ = match server {
176                Some(server) => self.inner.mcp.upsert_server(name, server).await,
177                None => self.inner.mcp.remove_server(&name).await,
178            };
179        }
180    }
181}

183/// Each agent document's fingerprint, shared and per preset.
184fn agent_fingerprints(
185    config: &LoadedConfiguration,
186) -> BTreeMap<(Option<&str>, &str), &cookie_agent_models::Sha256Digest> {
187    config
188        .agents
189        .iter()
190        .map(|(id, document)| ((None, id.as_str()), &document.document_fingerprint))
191        .chain(config.agent_presets.iter().flat_map(|(preset, agents)| {
192            agents.iter().map(move |(id, document)| {
193                (
194                    (Some(preset.as_str()), id.as_str()),
195                    &document.document_fingerprint,
196                )
197            })
198        }))
199        .collect()
200}
