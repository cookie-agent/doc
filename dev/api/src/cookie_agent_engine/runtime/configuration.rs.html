1//! Applying a new configuration generation to a running engine: the
2//! settings tables, agent documents, providers, headers, and MCP servers take
3//! effect at once; `server`, `plugins`, and `runtimes` are reported as needing
4//! a restart. A run keeps what it captured at admission (its agent, result
5//! limits, retry policy, and delegation depth).

16/// The configuration parts that apply only when the daemon restarts.
17#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
18pub enum RestartRequiredPart {
19    Server,
20    Plugins,
21    Runtimes,
22}
23
24impl RestartRequiredPart {
25    #[must_use]
26    pub const fn as_str(self) -> &'static str {
27        match self {
28            Self::Server => "server",
29            Self::Plugins => "plugins",
30            Self::Runtimes => "runtimes",
31        }
32    }
33}

35/// One change of a [`Engine::apply_config_changes`] set.
36#[derive(Clone, Debug)]
37pub enum ConfigChange {
38    /// Creates (`expected_fingerprint` absent) or replaces a document.
39    Put {
40        path: ConfigPath,
41        text: String,
42        expected_fingerprint: Option<String>,
43    },
44    /// Removes a document.
45    Delete {
46        path: ConfigPath,
47        expected_fingerprint: String,
48    },
49    /// Sets one key of `config.toml`; `null` removes it.
50    Set {
51        key: Vec<String>,
52        value: serde_json::Value,
53    },
54}

73/// Where the engine's configuration stands.
74#[derive(Clone, Debug, Eq, PartialEq)]
75pub struct ConfigurationStatus {
76    /// The revision of the applied generation.
77    pub applied_revision: ConfigRevision,
78    /// Parts whose applied value differs from the one the engine opened
79    /// with.
80    pub restart_required: Vec<RestartRequiredPart>,
81    /// Why the last reload failed, until one succeeds.
82    pub error: Option<String>,
83    /// What the applied configuration names that does not exist yet, such
84    /// as a delegation target no agent has: an ACP agent whose runtime has
85    /// not reported it, or a typo. Recomputed whenever the agents change.
86    pub diagnostics: Vec<String>,
87    /// The providers the native compaction record marks.
88    pub native_compaction: Vec<cookie_agent_protocol::NativeCompactionMark>,
89}

105    #[must_use]
106    pub fn configuration_status(&self) -> ConfigurationStatus {
107        let applied = self.inner.config.load();
108        let started = &self.inner.started_config;
109        let mut restart_required = Vec::new();
110        let (now, then) = (&applied.runtime.server, &started.runtime.server);
111        // A token the daemon created at start and wrote to the file is the
112        // one it serves with.
113        let served_token = self
114            .inner
115            .served_server_token
116            .get()
117            .map(String::as_str)
118            .or(then.token.as_deref());
119        if now.host != then.host || now.port != then.port || now.token.as_deref() != served_token {
120            restart_required.push(RestartRequiredPart::Server);
121        }
122        if applied.plugins != started.plugins {
123            restart_required.push(RestartRequiredPart::Plugins);
124        }
125        if applied.runtimes != started.runtimes {
126            restart_required.push(RestartRequiredPart::Runtimes);
127        }
128        ConfigurationStatus {
129            applied_revision: applied.revision.clone(),
130            restart_required,
131            diagnostics: self.configuration_diagnostics(),
132            native_compaction: self.inner.native_compaction.marks(),
133            error: self
134                .inner
135                .config_error
136                .lock()
137                .unwrap_or_else(|poisoned| poisoned.into_inner())
138                .clone(),
139        }
140    }

142    /// The ACP declarations a daemon agent document of the same ID shadows,
143    /// one without an `acp` block or one whose `acp.agent` names another
144    /// declaration: the document's agent keeps the ID and the declarations
145    /// are ignored. Then the documents whose `acp.agent` no runtime
146    /// declares yet.
147    fn shadowed_acp_declarations(&self) -> Vec<String> {
148        let config = self.inner.config.load();
149        let declarations = self.inner.acp.runtime_agents();
150        let mut by_agent =
151            std::collections::BTreeMap::<cookie_agent_protocol::AgentId, Vec<String>>::new();
152        for agent in &declarations {
153            if config.agents.get(&agent.agent).is_some_and(|document| {
154                document
155                    .frontmatter
156                    .acp
157                    .as_ref()
158                    .is_none_or(|settings| settings.declaration(&document.id) != &agent.agent)
159            }) {
160                by_agent
161                    .entry(agent.agent.clone())
162                    .or_default()
163                    .push(agent.runtime.to_string());
164            }
165        }
166        let mut diagnostics = by_agent
167            .into_iter()
168            .map(|(agent, runtimes)| {
169                let declared_on = match &runtimes[..] {
170                    [runtime] => format!("runtime `{runtime}`"),
171                    _ => format!("runtimes {}", runtimes.join(", ")),
172                };
173                match config
174                    .agents
175                    .get(&agent)
176                    .and_then(|document| document.frontmatter.acp.as_ref())
177                    .and_then(|settings| settings.agent.as_ref())
178                {
179                    Some(other) => format!(
180                        "ACP agent `{agent}` of {declared_on} is ignored: the agent document \
181                         `{agent}` runs `{other}` (`acp.agent`), so `{agent}` is that \
182                         document's agent"
183                    ),
184                    None => format!(
185                        "ACP agent `{agent}` of {declared_on} is ignored: the agent document \
186                         `{agent}` has no `acp` block, so `{agent}` is a daemon agent; give the \
187                         document an `acp: {{}}` block to make it the ACP agent's"
188                    ),
189                }
190            })
191            .collect::<Vec<_>>();
192        for document in config.agents.values() {
193            let Some(declaration) = document
194                .frontmatter
195                .acp
196                .as_ref()
197                .and_then(|settings| settings.agent.as_ref())
198            else {
199                continue;
200            };
201            if !declarations.iter().any(|agent| &agent.agent == declaration) {
202                diagnostics.push(format!(
203                    "agent `{}` runs ACP agent `{declaration}` (`acp.agent`), which no runtime \
204                     declares yet; it is an agent once one does",
205                    document.id
206                ));
207            }
208        }
209        diagnostics
210    }

212    /// The diagnostics of the published agents: those of the shared agents,
213    /// then each preset's own.
214    fn configuration_diagnostics(&self) -> Vec<String> {
215        let published = self.current_runtime();
216        let mut diagnostics = self.shadowed_acp_declarations();
217        diagnostics.extend(published.agents.diagnostics());
218        for (preset, registry) in &published.agent_presets {
219            for diagnostic in registry.diagnostics() {
220                if !diagnostics.contains(&diagnostic) {
221                    diagnostics.push(format!("preset `{preset}`: {diagnostic}"));
222                }
223            }
224        }
225        diagnostics.extend(self.official_skills_diagnostic());
226        diagnostics.extend(self.approval_cache_diagnostics(&published));
227        diagnostics.truncate(256);
228        diagnostics
229    }

231    /// What `[approval] cache_ttl` does on each model the approval reviewer
232    /// names: a TTL its provider does not offer, or a provider it cannot
233    /// change (approval reviewer spec section 5). `${parent_model}` is the
234    /// reviewed run's model, unknown until a review.
235    fn approval_cache_diagnostics(
236        &self,
237        published: &crate::runtime_snapshot::PublishedRuntime,
238    ) -> Vec<String> {
239        let ttl = self.inner.config.load().runtime.approval.cache_ttl;
240        let Ok(id) =
241            cookie_agent_protocol::AgentId::new(cookie_agent_config::BUILT_IN_APPROVAL_AGENT_ID)
242        else {
243            return Vec::new();
244        };
245        let Some(approval) = published.agents.get(&id) else {
246            return Vec::new();
247        };
248        approval
249            .resolved_fallback
250            .iter()
251            .filter_map(|fallback| match fallback {
252                crate::runtime_snapshot::ResolvedAgentFallback::Selection { selection, .. } => {
253                    Some(selection)
254                }
255                crate::runtime_snapshot::ResolvedAgentFallback::ParentModel { .. } => None,
256            })
257            .filter_map(|selection| {
258                let binding = crate::model_snapshots::binding_for_selection(
259                    &published.current_manifest,
260                    &published.models,
261                    selection,
262                )
263                .ok()?;
264                let provider = crate::policy::resolve_cache_strategy(
265                    Some(approval),
266                    &binding,
267                    published,
268                    cookie_agent_config::CacheLayer::Root,
269                )
270                .ok()
271                .flatten();
272                let (_, note) = crate::policy::reviewer_cache_strategy(provider, &binding, ttl);
273                let model = selection.model.to_string();
274                Some(match note? {
275                    crate::policy::ReviewerCacheNote::Substituted { used, why } => format!(
276                        "`[approval] cache_ttl = \"{}\"`: the reviewer's model `{model}` caches for {used}, {why}",
277                        ttl.as_str()
278                    ),
279                    crate::policy::ReviewerCacheNote::NoEffect(why) => format!(
280                        "`[approval] cache_ttl = \"{}\"` has no effect on the reviewer's model `{model}`: {why}",
281                        ttl.as_str()
282                    ),
283                })
284            })
285            .collect()
286    }

303    /// The configuration documents on disk, which may differ from the
304    /// applied ones until a reload.
305    pub async fn configuration_documents(&self) -> Result<ConfigDocuments, EngineError> {
306        let root = self.configuration_root()?;
307        let tree = tokio::task::spawn_blocking(move || ConfigTree::read(Some(&root)))
308            .await
309            .map_err(|error| EngineError::Runtime(error.to_string()))?
310            .map_err(config_error)?;
311        Ok(ConfigDocuments {
312            revision: tree.revision(),
313            documents: tree
314                .documents()
315                .map(|(path, bytes)| ConfigDocumentText {
316                    path: path.clone(),
317                    text: String::from_utf8_lossy(bytes).into_owned(),
318                    fingerprint: ConfigTree::fingerprint(bytes),
319                })
320                .collect(),
321            empty_presets: tree.empty_presets().map(str::to_owned).collect(),
322        })
323    }

325    /// Applies `changes` to the configuration on disk as one set: each is
326    /// checked against `expected_revision` and its own fingerprint, the
327    /// result is validated whole, written, and applied. Returns the new
328    /// revision.
329    pub async fn apply_config_changes(
330        &self,
331        expected_revision: ConfigRevision,
332        changes: Vec<ConfigChange>,
333    ) -> Result<ConfigRevision, EngineError> {
334        let edits_config = changes.iter().any(|change| match change {
335            ConfigChange::Put { path, .. } | ConfigChange::Delete { path, .. } => {
336                *path == ConfigPath::config()
337            }
338            ConfigChange::Set { .. } => false,
339        });
340        if edits_config
341            && changes
342                .iter()
343                .any(|change| matches!(change, ConfigChange::Set { .. }))
344        {
345            return Err(EngineError::ConfigChangeRefused(
346                "a set and a put or delete of config.toml cannot go together".into(),
347            ));
348        }
349        self.edit_configuration(move |tree, candidate| {
350            if tree.revision() != expected_revision {
351                return Err(EngineError::ConfigConflict(
352                    "the configuration changed since it was read".into(),
353                ));
354            }
355            for change in changes {
356                apply_change(candidate, change)?;
357            }
358            Ok(())
359        })
360        .await
361    }

363    /// Edits the configuration on disk with `edit`, which gets the tree as
364    /// read and the candidate to change, then validates, writes, and applies
365    /// the candidate. Returns the new revision.
366    pub(crate) async fn edit_configuration(
367        &self,
368        edit: impl FnOnce(&ConfigTree, &mut ConfigTree) -> Result<(), EngineError> + Send + 'static,
369    ) -> Result<ConfigRevision, EngineError> {
370        let _mutation = self.inner.config_mutation.lock().await;
371        let root = self.configuration_root()?;
372        let (revision, candidate) = tokio::task::spawn_blocking(move || {
373            let tree = ConfigTree::read(Some(&root)).map_err(config_error)?;
374            let mut candidate = tree.clone();
375            edit(&tree, &mut candidate)?;
376            let loaded = load_from_tree(&candidate).map_err(config_error)?;
377            write_tree(&tree, &candidate).map_err(|error| match error {
378                ConfigError::ChangedOnDisk(path) => EngineError::ConfigConflict(format!(
379                    "{} changed while it was written",
380                    path.display()
381                )),
382                error => config_error(error),
383            })?;
384            Ok::<_, EngineError>((candidate.revision(), loaded))
385        })
386        .await
387        .map_err(|error| EngineError::Runtime(error.to_string()))??;
388        self.apply_configuration_locked(candidate).await?;
389        Ok(revision)
390    }

392    /// Reads `config.toml` and the agent documents from disk and applies
393    /// them, for edits made by hand. A tree that fails validation leaves the
394    /// applied generation in place and is reported in the status.
395    pub async fn reload_configuration(&self) -> Result<ConfigurationStatus, EngineError> {
396        let _mutation = self.inner.config_mutation.lock().await;
397        let root = self.configuration_root()?;
398        let candidate = tokio::task::spawn_blocking(move || {
399            ConfigTree::read(Some(&root)).and_then(|tree| load_from_tree(&tree))
400        })
401        .await
402        .map_err(|error| EngineError::Runtime(error.to_string()))?;
403        match candidate {
404            Ok(candidate) => {
405                self.apply_configuration_locked(candidate).await?;
406                Ok(self.configuration_status())
407            }
408            Err(error) => {
409                *self
410                    .inner
411                    .config_error
412                    .lock()
413                    .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(error.to_string());
414                Err(EngineError::Config(Box::new(error)))
415            }
416        }
417    }

419    /// Applies a validated generation; the caller holds `config_mutation`.
420    pub(crate) async fn apply_configuration_locked(
421        &self,
422        mut candidate: LoadedConfiguration,
423    ) -> Result<(), EngineError> {
424        let previous = self.inner.config.load_full();
425        // Skills are read once at start; a reload keeps them.
426        candidate.skills = previous.skills.clone();
427        let mut reasons = vec![RuntimeChangeReason::ConfigReloaded];
428        if agent_fingerprints(&previous) != agent_fingerprints(&candidate) {
429            reasons.push(RuntimeChangeReason::AgentReloaded);
430        }
431        let candidate = Arc::new(candidate);
432        // The runtime mutation lock may be held through another change's
433        // recompile, and this one recompiles too: both stay off the async
434        // workers.
435        let engine = self.clone();
436        let applied = Arc::clone(&candidate);
437        super::on_blocking_pool(move || {
438            let _runtime = engine
439                .inner
440                .runtime_mutation
441                .lock()
442                .unwrap_or_else(|poisoned| poisoned.into_inner());
443            let published = engine.current_runtime();
444            let (_, publication) = engine.inner.model_manager.reload_configuration(
445                applied.runtime.providers.clone(),
446                applied.runtime.headers.clone(),
447                Arc::clone(published.models.catalog()),
448                |models| {
449                    engine
450                        .prepare_publication(&applied, models, &published, reasons)
451                        .map_err(|_| cookie_agent_models::ModelManagerError::RuntimeCompileFailed)
452                },
453            )?;
454            engine.inner.config.store(Arc::clone(&applied));
455            engine.publish(publication);
456            Ok(())
457        })
458        .await?;
459        // The publication also wakes the background sync; awaiting one here
460        // makes the files current when the apply or reload answers.
461        self.sync_agent_memory().await;
462        self.drop_unconfigured_runtime_records(&candidate).await;
463        *self
464            .inner
465            .config_error
466            .lock()
467            .unwrap_or_else(|poisoned| poisoned.into_inner()) = None;
468        self.rebase_mcp_servers(&candidate).await;
469        Ok(())
470    }

472    /// Puts the new user layer under the MCP store's runtime overlay and
473    /// brings the registry in line with what changed.
474    async fn rebase_mcp_servers(&self, config: &LoadedConfiguration) {
475        let _mutation = self.inner.mcp_mutation.lock().await;
476        let changes = self
477            .inner
478            .config_store
479            .lock()
480            .unwrap_or_else(|poisoned| poisoned.into_inner())
481            .rebase(config.user_mcp_servers.clone());
482        // A server that fails to start reports through the MCP failure
483        // notifications like any other.
484        for (name, server) in changes {
485            let _ = match server {
486                Some(server) => self.inner.mcp.upsert_server(name, server).await,
487                None => self.inner.mcp.remove_server(&name).await,
488            };
489        }
490    }

493/// Each agent document's fingerprint, shared and per preset.
494fn agent_fingerprints(
495    config: &LoadedConfiguration,
496) -> BTreeMap<(Option<&str>, &str), &cookie_agent_models::Sha256Digest> {
497    config
498        .agents
499        .iter()
500        .map(|(id, document)| ((None, id.as_str()), &document.document_fingerprint))
501        .chain(config.agent_presets.iter().flat_map(|(preset, agents)| {
502            agents.iter().map(move |(id, document)| {
503                (
504                    (Some(preset.as_str()), id.as_str()),
505                    &document.document_fingerprint,
506                )
507            })
508        }))
509        .collect()
510}

516fn apply_change(candidate: &mut ConfigTree, change: ConfigChange) -> Result<(), EngineError> {
517    let current = |candidate: &ConfigTree, path: &ConfigPath| {
518        candidate.get(path).map(ConfigTree::fingerprint)
519    };
520    match change {
521        ConfigChange::Put {
522            path,
523            text,
524            expected_fingerprint,
525        } => {
526            if current(candidate, &path) != expected_fingerprint {
527                return Err(EngineError::ConfigConflict(match expected_fingerprint {
528                    None => format!("{path} already exists"),
529                    Some(_) => format!("{path} changed since it was read"),
530                }));
531            }
532            candidate
533                .put(path, text.into_bytes())
534                .map_err(|error| EngineError::ConfigChangeRefused(error.to_string()))
535        }
536        ConfigChange::Delete {
537            path,
538            expected_fingerprint,
539        } => {
540            if current(candidate, &path).as_deref() != Some(expected_fingerprint.as_str()) {
541                return Err(EngineError::ConfigConflict(format!(
542                    "{path} changed since it was read"
543                )));
544            }
545            candidate.delete(&path);
546            Ok(())
547        }
548        ConfigChange::Set { key, value } => candidate
549            .set_key_json(&key, &value)
550            .map_err(|error| EngineError::ConfigChangeRefused(error.to_string())),
551    }
