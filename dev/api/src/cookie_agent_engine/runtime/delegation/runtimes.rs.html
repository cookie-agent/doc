15/// Whether a session's `delegate_subagent` has `capabilities` (section
16/// 2.1).
17#[derive(Clone, Copy, Debug, Eq, PartialEq)]
18pub(crate) enum CapabilitiesParam {
19    /// A delegated session, or a root without `global`.
20    Absent,
21    /// A root with `global` and other attachments: leaving it out inherits.
22    Optional,
23    /// A root with only `global`: it has no workspace of its own.
24    Required,
25}

27/// Whether `attachments` include the `global` runtime.
28pub(crate) fn has_global(attachments: &[RuntimeAttachment]) -> bool {
29    attachments
30        .iter()
31        .any(|attachment| attachment.runtime_id.as_str() == RuntimeId::GLOBAL)
32}

34/// `<runtime_id>/<capability_id>`, as `delegate_subagent` takes it.
35pub(crate) fn parse_capability(
36    capability: &str,
37    workspace: Option<String>,
38) -> Result<DelegatedCapability, String> {
39    let (runtime, name) = capability.split_once('/').ok_or_else(|| {
40        format!(
41            "`{capability}` is not `<runtime_id>/<capability_id>`; write the runtime and one of its capabilities as list_runtimes shows them, e.g. `labpc/coding`"
42        )
43    })?;
44    let runtime_id = RuntimeId::new(runtime)
45        .map_err(|_| format!("`{runtime}` in `{capability}` is not a runtime ID"))?;
46    let capability_name = CapabilityName::new(name)
47        .map_err(|_| format!("`{name}` in `{capability}` is not a capability name"))?;
48    if runtime_id.as_str() == RuntimeId::GLOBAL {
49        return Err(format!(
50            "`{capability}` cannot be listed: `global` follows this session's root, so the subagent has it exactly when the root does"
51        ));
52    }
53    if workspace
54        .as_deref()
55        .is_some_and(|workspace| workspace.trim().is_empty())
56    {
57        return Err(format!("the workspace of `{capability}` is empty"));
58    }
59    Ok(DelegatedCapability {
60        runtime_id,
61        capability: capability_name,
62        workspace,
63    })
64}

66impl Engine {
67    /// The root session of `session`'s tree.
68    fn root_projection(
69        &self,
70        session: &SessionProjection,
71    ) -> Result<std::sync::Arc<SessionProjection>, EngineError> {
72        match session.meta.origin {
73            SessionOrigin::Delegated {
74                root_session_id, ..
75            } => Ok(self.inner.store.get(root_session_id)?),
76            _ => Ok(self.inner.store.get(session.meta.session_id)?),
77        }
78    }

80    /// Whether the root of `session`'s tree has `global` attached now.
81    pub(crate) fn root_has_global(&self, session: SessionId) -> Result<bool, EngineError> {
82        let projection = self.inner.store.get(session)?;
83        let root = self.root_projection(&projection)?;
84        Ok(has_global(&self.attachments_of(&root)))
85    }

87    /// Whether `session`'s `delegate_subagent` takes `capabilities`.
88    pub(crate) fn delegate_capabilities_param(
89        &self,
90        session: SessionId,
91    ) -> Result<CapabilitiesParam, EngineError> {
92        let projection = self.inner.store.get(session)?;
93        if !matches!(projection.meta.origin, SessionOrigin::Root) {
94            return Ok(CapabilitiesParam::Absent);
95        }
96        let attachments = self.attachments_of(&projection);
97        Ok(if !has_global(&attachments) {
98            CapabilitiesParam::Absent
99        } else if attachments.len() == 1 {
100            CapabilitiesParam::Required
101        } else {
102            CapabilitiesParam::Optional
103        })
104    }

106    /// Checks `capabilities` against the daemon's runtimes now: each runtime
107    /// must exist, and a runtime that has said what it offers must offer the
108    /// capability, with a workspace exactly when the capability takes one.
109    pub(crate) fn check_delegated_capabilities(
110        &self,
111        capabilities: &[DelegatedCapability],
112    ) -> Result<(), String> {
113        if capabilities.is_empty() {
114            return Err("capabilities must name at least one runtime capability".into());
115        }
116        let templates = self.runtime_templates();
117        for requested in capabilities {
118            let written = format!("{}/{}", requested.runtime_id, requested.capability);
119            let Some(template) = templates
120                .iter()
121                .find(|template| template.runtime_id == requested.runtime_id.as_str())
122            else {
123                if let Some(offline) = self.remembered_offline(requested.runtime_id.as_str()) {
124                    return Err(offline);
125                }
126                return Err(format!(
127                    "there is no runtime `{}`; call list_runtimes for the runtimes this daemon has",
128                    requested.runtime_id
129                ));
130            };
131            let offered = &template.listing.capabilities;
132            if offered.is_empty() {
133                // It has not connected yet: starting it tells.
134                continue;
135            }
136            let Some(capability) = offered
137                .iter()
138                .find(|offered| offered.name == requested.capability)
139            else {
140                let names = offered
141                    .iter()
142                    .map(|offered| format!("`{}/{}`", requested.runtime_id, offered.name))
143                    .collect::<Vec<_>>()
144                    .join(", ");
145                return Err(format!(
146                    "runtime `{}` has no capability `{}`; it offers {names}",
147                    requested.runtime_id, requested.capability
148                ));
149            };
150            match (capability.takes_workspace, &requested.workspace) {
151                (true, None) => {
152                    return Err(format!(
153                        "`{written}` takes a workspace: give the directory on that runtime's machine"
154                    ));
155                }
156                (false, Some(_)) => {
157                    return Err(format!("`{written}` takes no workspace"));
158                }
159                _ => {}
160            }
161        }
162        Ok(())
163    }

165    /// The attachments a child of the root `parent` gets for
166    /// `capabilities`: the root's `global` when it has it, then each listed
167    /// capability in order, started first. The first that cannot start
168    /// fails the whole call, naming it.
169    pub(crate) async fn open_delegated_capabilities(
170        &self,
171        parent: &SessionProjection,
172        capabilities: &[DelegatedCapability],
173    ) -> Result<Vec<RuntimeAttachment>, EngineError> {
174        let refused = |message: String| EngineError::ToolFailed(message);
175        if !matches!(parent.meta.origin, SessionOrigin::Root) {
176            return Err(refused(
177                "capabilities can only be given by a root session".into(),
178            ));
179        }
180        let root = self.attachments_of(parent);
181        if !has_global(&root) {
182            return Err(refused(
183                "capabilities need `global` attached to this session".into(),
184            ));
185        }
186        self.check_delegated_capabilities(capabilities)
187            .map_err(refused)?;
188        let mut attachments = root
189            .into_iter()
190            .filter(|attachment| attachment.runtime_id.as_str() == RuntimeId::GLOBAL)
191            .collect::<Vec<_>>();
192        for requested in capabilities {
193            let written = format!("{}/{}", requested.runtime_id, requested.capability);
194            let instance = self
195                .ensure_runtime(
196                    requested.runtime_id.as_str(),
197                    Some(requested.capability.as_str()),
198                    requested.workspace.as_deref().map(PathBuf::from).as_deref(),
199                )
200                .await
201                .map_err(|error| {
202                    refused(match &requested.workspace {
203                        Some(workspace) => {
204                            format!("`{written}` could not be opened at `{workspace}`: {error}")
205                        }
206                        None => format!("`{written}` could not be opened: {error}"),
207                    })
208                })?;
209            self.refuse_overlap_with(
210                &attachments,
211                &instance,
212                &cookie_agent_runtime_api::SessionToolContext::new(SessionId::new_v7()),
213            )
214            .map_err(|error| refused(format!("`{written}`: {error}")))?;
215            crate::runtime::attachments::push_instance(&mut attachments, &instance)
216                .map_err(|error| refused(format!("`{written}`: {error}")))?;
217        }
218        Ok(attachments)
219    }

221    /// The attachments a delegated child of `parent` inherits: its
222    /// parent's, with `global` exactly when the root has it now (section
223    /// 2.2).
224    pub(crate) fn inherited_attachments(
225        &self,
226        parent: &SessionProjection,
227    ) -> Result<Vec<RuntimeAttachment>, EngineError> {
228        let root = self.attachments_of(&*self.root_projection(parent)?);
229        let mut attachments = root
230            .into_iter()
231            .filter(|attachment| attachment.runtime_id.as_str() == RuntimeId::GLOBAL)
232            .collect::<Vec<_>>();
233        attachments.extend(
234            self.attachments_of(parent)
235                .into_iter()
236                .filter(|attachment| attachment.runtime_id.as_str() != RuntimeId::GLOBAL),
237        );
238        Ok(attachments)
239    }

241    /// Records on the root of `child`'s tree the attachments `child`
242    /// started with that the root records neither as its own nor as
243    /// delegated, `global` aside (section 5.3).
244    pub(crate) async fn record_delegated_attachments(
245        &self,
246        root: SessionId,
247        child: SessionId,
248        attachments: Vec<RuntimeAttachment>,
249    ) -> Result<(), EngineError> {
250        self.on_actor(root, move |engine| {
251            let projection = engine.inner.store.get(root)?;
252            if !matches!(projection.meta.origin, SessionOrigin::Root) {
253                return Ok(());
254            }
255            let own = engine.attachments_of(&projection);
256            let mut added: Vec<RuntimeAttachment> = Vec::new();
257            for attachment in attachments {
258                let known = attachment.runtime_id.as_str() == RuntimeId::GLOBAL
259                    || own
260                        .iter()
261                        .chain(&projection.meta.delegated_attachments)
262                        .chain(&added)
263                        .any(|known| crate::session::same_place(known, &attachment));
264                if !known {
265                    added.push(attachment);
266                }
267            }
268            if added.is_empty() {
269                return Ok(());
270            }
271            engine.append_direct(
272                root,
273                None,
274                crate::runtime::event_origin("engine:delegation"),
275                Event::DelegatedAttachmentsAdded {
276                    child_session_id: child,
277                    attachments: added,
278                },
279            )?;
280            Ok(())
281        })
282        .await
283    }
284}

286/// One runtime as `list_runtimes` prints it.
287struct ListedRuntime {
288    id: String,
289    connected: bool,
290    starts_on_use: bool,
291    platform: Option<String>,
292    home: Option<String>,
293    message: Option<String>,
294    capabilities: Vec<cookie_agent_protocol::RuntimeCapabilityInfo>,
295    /// The ACP agents it offers, by ID.
296    acp_agents: Vec<String>,
297    attachments: Vec<(RuntimeAttachment, &'static str)>,
298}

300/// `value` escaped for an XML attribute.
301fn xml_escape(value: &str) -> String {
302    let mut escaped = String::with_capacity(value.len());
303    for character in value.chars() {
304        match character {
305            '&' => escaped.push_str("&amp;"),
306            '<' => escaped.push_str("&lt;"),
307            '>' => escaped.push_str("&gt;"),
308            '"' => escaped.push_str("&quot;"),
309            '\'' => escaped.push_str("&apos;"),
310            character => escaped.push(character),
311        }
312    }
313    escaped
314}
315
316/// ` name="value"` for each attribute that has a value.
317fn attributes(pairs: &[(&str, Option<&str>)]) -> String {
318    pairs
319        .iter()
320        .filter_map(|(name, value)| value.map(|value| format!(" {name}=\"{}\"", xml_escape(value))))
321        .collect()
322}

324/// The platform every runtime of `runtimes` with one reports, when they
325/// agree.
326fn shared_platform(runtimes: &[ListedRuntime]) -> Option<String> {
327    let mut platforms = runtimes
328        .iter()
329        .filter_map(|runtime| runtime.platform.as_deref());
330    let first = platforms.next()?;
331    platforms
332        .all(|platform| platform == first)
333        .then(|| first.to_owned())
334}

336fn write_runtime(text: &mut String, indent: &str, runtime: &ListedRuntime, platform: bool) {
337    let connected = if runtime.connected { "true" } else { "false" };
338    let open = format!(
339        "{indent}<runtime{}",
340        attributes(&[
341            ("id", Some(&runtime.id)),
342            ("connected", Some(connected)),
343            ("starts_on_use", runtime.starts_on_use.then_some("true")),
344            ("platform", runtime.platform.as_deref().filter(|_| platform),),
345            ("home", runtime.home.as_deref()),
346            ("message", runtime.message.as_deref()),
347            (
348                "acp_agents",
349                Some(runtime.acp_agents.join(", "))
350                    .filter(|agents| !agents.is_empty())
351                    .as_deref(),
352            ),
353        ])
354    );
355    if runtime.capabilities.is_empty() && runtime.attachments.is_empty() {
356        text.push_str(&open);
357        text.push_str("/>\n");
358        return;
359    }
360    text.push_str(&open);
361    text.push_str(">\n");
362    for capability in &runtime.capabilities {
363        let takes = if capability.takes_workspace {
364            "true"
365        } else {
366            "false"
367        };
368        text.push_str(&format!(
369            "{indent}  <capability{}/>\n",
370            attributes(&[
371                ("name", Some(capability.name.as_str())),
372                ("takes_workspace", Some(takes)),
373                (
374                    "description",
375                    Some(capability.description.as_str()).filter(|text| !text.is_empty()),
376                ),
377            ])
378        ));
379    }
380    for (attachment, status) in &runtime.attachments {
381        text.push_str(&format!(
382            "{indent}  <attachment{}/>\n",
383            attributes(&[
384                ("name", Some(attachment.name.as_str())),
385                ("capability", Some(attachment.capability.as_str())),
386                ("workspace", attachment.workspace.as_deref()),
387                ("status", Some(status)),
388            ])
389        ));
390    }
391    text.push_str(&format!("{indent}</runtime>\n"));
392}

394impl Engine {
395    /// What `list_runtimes` prints for a call of `session`: every runtime
396    /// but `global`, grouped by machine (the daemon's first, then by name)
397    /// and by runtime ID, with `session`'s own attachments; `<global/>`
398    /// when the root of `session`'s tree has `global`.
399    pub(crate) fn list_runtimes_text(&self, session: SessionId) -> Result<String, EngineError> {
400        use cookie_agent_protocol::RuntimeLifetime;
401
402        use crate::runtimes::RuntimeInstanceStatus;
403
404        let own = self.session_runtimes(session)?;
405        let instances = self.runtime_instances();
406        let acp_agents = self.inner.acp.runtime_agents();
407        let status_of = |attachment: &RuntimeAttachment| -> &'static str {
408            let status = instances.iter().find(|instance| {
409                instance.runtime_id == attachment.runtime_id.as_str()
410                    && instance.capability == attachment.capability.as_str()
411                    && instance.workspace == attachment.workspace
412            });
413            match status.map(|instance| instance.status) {
414                Some(RuntimeInstanceStatus::Starting) => "starting",
415                Some(RuntimeInstanceStatus::Ready) => "ready",
416                Some(RuntimeInstanceStatus::Reconnecting) => "reconnecting",
417                Some(RuntimeInstanceStatus::Failed) => "failed",
418                Some(RuntimeInstanceStatus::Stopped) | None => "stopped",
419            }
420        };
421        let mut machines: Vec<(cookie_agent_protocol::MachineName, Vec<ListedRuntime>)> =
422            Vec::new();
423        let mut unplaced = Vec::new();
424        for template in self.runtime_templates() {
425            if template.runtime_id == RuntimeId::GLOBAL {
426                continue;
427            }
428            let listing = template.listing;
429            let remembered = (!listing.connected)
430                .then(|| self.remembered_runtime(&template.runtime_id))
431                .flatten();
432            let (platform, home) = match remembered {
433                Some(remembered) => (
434                    listing.platform.or(remembered.platform),
435                    listing.home.or(remembered.home),
436                ),
437                None => (listing.platform, listing.home),
438            };
439            let runtime = ListedRuntime {
440                starts_on_use: !listing.connected && template.lifetime == RuntimeLifetime::Spawned,
441                connected: listing.connected,
442                platform,
443                home,
444                message: listing.message.filter(|_| !listing.connected),
445                capabilities: listing.capabilities,
446                acp_agents: acp_agents
447                    .iter()
448                    .filter(|agent| agent.runtime.as_str() == template.runtime_id)
449                    .map(|agent| agent.id.to_string())
450                    .collect(),
451                attachments: own
452                    .iter()
453                    .filter(|attachment| attachment.runtime_id.as_str() == template.runtime_id)
454                    .map(|attachment| (attachment.clone(), status_of(attachment)))
455                    .collect(),
456                id: template.runtime_id.clone(),
457            };
458            match self.runtime_machine(&template.runtime_id) {
459                Some(machine) => match machines.iter_mut().find(|(name, _)| *name == machine) {
460                    Some((_, runtimes)) => runtimes.push(runtime),
461                    None => machines.push((machine, vec![runtime])),
462                },
463                None => unplaced.push(runtime),
464            }
465        }
466        let daemon = self.runtime_machine(RuntimeId::GLOBAL);
467        machines.sort_by(|(left, _), (right, _)| {
468            (Some(left) != daemon.as_ref(), left).cmp(&(Some(right) != daemon.as_ref(), right))
469        });
470        let mut text = String::from("<runtimes>\n");
471        if self.root_has_global(session)? {
472            text.push_str(
473                "  <!-- global is attached to this session's root, so every subagent in this tree has it, whether or not its capabilities list it. -->\n  <global/>\n",
474            );
475        }
476        for (machine, runtimes) in &machines {
477            let platform = shared_platform(runtimes);
478            text.push_str(&format!(
479                "  <machine{}>\n",
480                attributes(&[
481                    ("name", Some(machine.as_str())),
482                    ("platform", platform.as_deref()),
483                ])
484            ));
485            for runtime in runtimes {
486                write_runtime(&mut text, "    ", runtime, platform.is_none());
487            }
488            text.push_str("  </machine>\n");
489        }
490        if !unplaced.is_empty() {
491            // Only a runtime that never connected has no machine: the daemon
492            // learns it as the runtime first connects
493            // (`docs/site/specs/machines-on-demand.md` section 4.1).
494            text.push_str(
495                "  <!-- The daemon does not know these runtimes' machines yet: they are not connected. One is placed on a machine when it connects, as attaching or delegating to it makes it do. -->\n  <not_connected_yet>\n",
496            );
497            for runtime in &unplaced {
498                write_runtime(&mut text, "    ", runtime, true);
499            }
500            text.push_str("  </not_connected_yet>\n");
501        }
502        text.push_str("</runtimes>");
503        Ok(text)
504    }
