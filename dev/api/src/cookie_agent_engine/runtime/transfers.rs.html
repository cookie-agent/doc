1//! Where a `transfer` reads and writes: one of the session's attachments on
2//! the machine it names (`docs/site/specs/machines-on-demand.md` section 6).
3
4use cookie_agent_protocol::{RuntimeId, SessionId};
5use cookie_agent_runtime_api::ResourceScope;

7use super::Engine;
8use crate::runtimes::InstanceConnection;

10/// One end of a transfer on a machine: the attachment's runtime connection,
11/// and where its path permission is checked.
12pub(crate) struct TransferEndpoint {
13    pub(crate) connection: InstanceConnection,
14    /// The machine as the session's machines block names it.
15    pub(crate) machine: String,
16    pub(crate) scope: ResourceScope,
17}
18
19impl Engine {
20    /// The end of a transfer on `machine`: the session's attachment there
21    /// whose runtime takes transfers.
22    pub(crate) fn transfer_endpoint(
23        &self,
24        session: SessionId,
25        machine: &str,
26    ) -> Result<TransferEndpoint, String> {
27        let attachments = self
28            .session_runtimes(session)
29            .map_err(|error| error.to_string())?;
30        let mut machines = std::collections::BTreeSet::new();
31        let mut refusals = Vec::new();
32        for attachment in attachments {
33            if attachment.runtime_id.as_str() == RuntimeId::GLOBAL {
34                continue;
35            }
36            let named = self
37                .runtime_machine(attachment.runtime_id.as_str())
38                .map_or_else(
39                    || attachment.runtime_id.to_string(),
40                    |name| name.to_string(),
41                );
42            machines.insert(named.clone());
43            if named != machine {
44                continue;
45            }
46            let Some(instance) = self.inner.runtimes.instance(&attachment) else {
47                refusals.push(format!(
48                    "runtime `{}` is not connected",
49                    attachment.runtime_id
50                ));
51                continue;
52            };
53            let Some(connection) = instance.connection() else {
54                refusals.push(format!(
55                    "runtime `{}` runs inside the daemon without the runtime protocol",
56                    attachment.runtime_id
57                ));
58                continue;
59            };
60            if !connection.client.files() {
61                refusals.push(format!(
62                    "runtime `{}` speaks runtime protocol {}, which has no transfers (1.13.0 \
63                     and later have them): update it",
64                    attachment.runtime_id,
65                    connection
66                        .client
67                        .protocol_version()
68                        .map_or_else(|| "unknown".to_owned(), |version| version.to_string())
69                ));
70                continue;
71            }
72            let info = instance.info();
73            return Ok(TransferEndpoint {
74                scope: ResourceScope {
75                    runtime: attachment.runtime_id.to_string(),
76                    workspace: info
77                        .workspace()
78                        .map(|workspace| workspace.root().to_owned()),
79                    home: info.home().map(str::to_owned),
80                },
81                connection,
82                machine: named,
83            });
84        }
85        if refusals.is_empty() {
86            let known = if machines.is_empty() {
87                "none".to_owned()
88            } else {
89                machines
90                    .iter()
91                    .map(|name| format!("`{name}`"))
92                    .collect::<Vec<_>>()
93                    .join(", ")
94            };
95            return Err(format!(
96                "machine `{machine}` is not attached to this session (its machines: {known}); \
97                 attach it with attach_machine first"
98            ));
99        }
100        Err(format!(
101            "machine `{machine}` takes no transfers: {}",
102            refusals.join("; ")
103        ))
104    }
105
106    /// Whether `session` has a machine attached whose runtime takes
107    /// transfers, which `transfer` is offered for.
108    pub(crate) fn session_takes_transfers(&self, session: SessionId) -> bool {
109        self.session_runtimes(session)
110            .unwrap_or_default()
111            .iter()
112            .filter(|attachment| attachment.runtime_id.as_str() != RuntimeId::GLOBAL)
113            .any(|attachment| {
114                self.inner
115                    .runtimes
116                    .instance(attachment)
117                    .and_then(|instance| instance.connection())
118                    .is_some_and(|connection| connection.client.files())
119            })
120    }
121
122    /// The most bytes one transfer moves, when `[transfer] max_bytes` sets
123    /// a limit.
124    pub(crate) fn transfer_limit(&self) -> Option<u64> {
125        self.inner.config.load().runtime.transfer.limit()
126    }
127}

129impl Engine {
130    /// The artifact a transfer's `from` names in `session`'s tree, and its
131    /// length; see [`super::artifact_reads::transfer_source`].
132    pub(crate) async fn transfer_artifact(
133        &self,
134        session: SessionId,
135        artifact: String,
136        stream: Option<String>,
137    ) -> Result<(String, u64), crate::ToolError> {
138        let store = std::sync::Arc::clone(&self.inner.artifacts);
139        super::blocking_io::run(move || {
140            super::artifact_reads::transfer_source(&store, session, &artifact, stream.as_deref())
141        })
142        .await?
143    }
144
145    /// `length` bytes of blob `digest` of `session`'s tree from `offset`.
146    pub(crate) async fn transfer_artifact_chunk(
147        &self,
148        session: SessionId,
149        digest: String,
150        offset: u64,
151        length: u64,
152    ) -> Result<Vec<u8>, crate::ToolError> {
153        let store = std::sync::Arc::clone(&self.inner.artifacts);
154        super::blocking_io::run(move || {
155            super::artifact_reads::read_transfer_chunk(&store, session, &digest, offset, length)
156        })
157        .await?
158    }
