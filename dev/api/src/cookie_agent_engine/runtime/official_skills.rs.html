1//! The official skills (`docs/site/specs/default-models-and-onboarding.md`
2//! section 13): fetched from a versionless source, `<source>/index.json`
3//! and `<source>/<name>/<file>`, every file checked against the index's
4//! SHA-256, and kept in `<data_root>/official_skills/<name>/`, which only
5//! the daemon writes. Skills the index no longer lists, or whose
6//! `requires_cookie` this version does not satisfy, are removed; a failed
7//! refresh keeps the cached copies and is reported as a configuration
8//! diagnostic.
9

22/// The official source. Production never changes it; tests serve their own
23/// with [`Engine::set_official_skills_url`].
24pub const OFFICIAL_SKILLS_URL: &str = "https://cookie-agent.github.io/skills";
25/// How often a running daemon refreshes them.
26pub const OFFICIAL_SKILLS_REFRESH: Duration = Duration::from_secs(24 * 60 * 60);
27/// The daemon's record of what it keeps, in the official skills folder; its
28/// leading `.` keeps it out of skill discovery.
29const KEPT_INDEX: &str = ".index.json";
30const MAX_INDEX_BYTES: usize = 1024 * 1024;
31const MAX_FILE_BYTES: usize = 1024 * 1024;
32const MAX_FILES_PER_SKILL: usize = 64;

57pub struct OfficialSkillsReport {

109impl Engine {
110    /// Fetches the official skills from `url` instead of
111    /// [`OFFICIAL_SKILLS_URL`], for tests that serve their own.
112    pub fn set_official_skills_url(&self, url: impl Into<String>) {
113        *lock(&self.inner.official_skills.source_url) = url.into();
114    }
115
116    /// The user skill roots the daemon's skills were loaded from; once set,
117    /// a refresh that changes the official skills reloads the daemon's
118    /// skills from them and the official folder, so new sessions see the
119    /// change without a restart.
120    pub fn set_user_skill_roots(&self, roots: UserSkillRoots) {
121        *lock(&self.inner.official_skills.user_roots) = Some(roots);
122    }
123
124    /// The official skills kept, by name, with their versions.
125    #[must_use]
126    pub fn official_skills(&self) -> Arc<BTreeMap<String, String>> {
127        Arc::clone(&lock(&self.inner.official_skills.versions))
128    }
129
130    /// Why the last refresh failed, as a configuration diagnostic.
131    pub(crate) fn official_skills_diagnostic(&self) -> Option<String> {
132        lock(&self.inner.official_skills.last_error)
133            .as_ref()
134            .map(|error| format!("official skills were not refreshed: {error}"))
135    }
136
137    /// Refreshes the official skills now; refreshes do not overlap.
138    pub async fn refresh_official_skills(&self) -> OfficialSkillsReport {
139        let state = &self.inner.official_skills;
140        let _refresh = state.refresh.lock().await;
141        if !self.inner.config.load().runtime.skills.official {
142            *lock(&state.last_error) = None;
143            return OfficialSkillsReport {
144                enabled: false,
145                ..OfficialSkillsReport::default()
146            };
147        }
148        let url = lock(&state.source_url).clone();
149        let mut report = refresh(&url, &state.root).await;
150        report.enabled = true;
151        *lock(&state.versions) = Arc::new(report.skills.clone());
152        *lock(&state.last_error) = report.error.clone();
153        if !report.updated.is_empty() || !report.removed.is_empty() {
154            self.reload_daemon_skills().await;
155        }
156        report
157    }
158
159    /// Reloads the global runtime's skills from the user roots and the
160    /// official folder, when the roots are known.
161    async fn reload_daemon_skills(&self) {
162        let Some(roots) = lock(&self.inner.official_skills.user_roots).clone() else {
163            return;
164        };
165        let root = self.inner.official_skills.root.clone();
166        let loaded = local(move || {
167            cookie_agent_config::load_skill_roots_with_official(
168                Some(&root),
169                roots.shared.as_deref(),
170                roots.user.as_deref(),
171                &[],
172            )
173            .map_err(|error| error.to_string())
174        })
175        .await
176        .and_then(|loaded| loaded);
177        match loaded {
178            Ok(skills) => self.inner.runtimes.global().replace_skills(skills),
179            Err(error) => {
180                *lock(&self.inner.official_skills.last_error) =
181                    Some(format!("the skills did not load: {error}"));
182            }
183        }
184    }
185}

209/// One refresh against the source at `url` into `root`.
210async fn refresh(url: &str, root: &Path) -> OfficialSkillsReport {
211    let mut report = OfficialSkillsReport::default();
212    let kept = {
213        let root = root.to_owned();
214        local(move || read_kept(&root)).await.unwrap_or_default()
215    };
216    let client = match reqwest::Client::builder()
217        .redirect(reqwest::redirect::Policy::limited(5))
218        .connect_timeout(Duration::from_secs(10))
219        .timeout(Duration::from_secs(30))
220        .user_agent(concat!("cookie-agent/", env!("CARGO_PKG_VERSION")))
221        .build()
222    {
223        Ok(client) => client,
224        Err(error) => {
225            report.error = Some(error.to_string());
226            report.skills = versions(kept);
227            return report;
228        }
229    };
230    let base = url.trim_end_matches('/');
231    let index = match fetch(&client, &format!("{base}/index.json"), MAX_INDEX_BYTES)
232        .await
233        .and_then(|bytes| {
234            serde_json::from_slice::<SourceIndex>(&bytes)
235                .map_err(|error| format!("index.json does not read: {error}"))
236        })
237        .and_then(|index| {
238            (index.version == 1)
239                .then_some(index)
240                .ok_or_else(|| "index.json is of a version this build does not read".to_owned())
241        }) {
242        Ok(index) => index,
243        Err(error) => {
244            report.error = Some(error);
245            report.skills = versions(kept);
246            return report;
247        }
248    };
249    let mut refusals = Vec::new();
250    let mut wanted = BTreeMap::new();
251    for skill in index.skills {
252        if let Err(reason) = valid_entry(&skill) {
253            refusals.push(format!("`{}`: {reason}", skill.name));
254            continue;
255        }
256        match version_satisfies(&skill.requires_cookie, env!("CARGO_PKG_VERSION")) {
257            Some(true) => {
258                wanted.insert(skill.name.clone(), skill);
259            }
260            Some(false) => {}
261            None => refusals.push(format!(
262                "`{}`: `{}` is not a version range",
263                skill.name, skill.requires_cookie
264            )),
265        }
266    }
267    let kept_by_name = kept
268        .into_iter()
269        .map(|skill| (skill.name.clone(), skill))
270        .collect::<BTreeMap<_, _>>();
271    let mut now_kept = BTreeMap::new();
272    for (name, skill) in wanted {
273        let directory = root.join(&name);
274        match update_skill(&client, base, &directory, &skill).await {
275            Ok(changed) => {
276                if changed || !kept_by_name.contains_key(&name) {
277                    report.updated.push(name.clone());
278                }
279                now_kept.insert(name, skill);
280            }
281            Err(reason) => {
282                refusals.push(format!("`{name}`: {reason}"));
283                // The cached copy, if any, stays as it was.
284                if let Some(previous) = kept_by_name.get(&name) {
285                    now_kept.insert(name, previous.clone());
286                }
287            }
288        }
289    }
290    let (removed, written) = {
291        let root = root.to_owned();
292        let record = now_kept.values().cloned().collect::<Vec<_>>();
293        let previously = kept_by_name.keys().cloned().collect::<BTreeSet<_>>();
294        local(move || remove_others_and_record(&root, &record, &previously))
295            .await
296            .unwrap_or_else(|error| (Vec::new(), Err(error)))
297    };
298    report.removed = removed;
299    if let Err(error) = written {
300        refusals.push(format!(
301            "the record of kept skills was not written: {error}"
302        ));
303    }
304    report.skills = versions(now_kept.into_values());
305    if !refusals.is_empty() {
306        report.error = Some(refusals.join("; "));
307    }
308    report
309}

311/// Removes everything in `root` but the skills of `record`, the kept index,
312/// and other dot files: unlisted skills, skills this version does not
313/// satisfy, and leftovers. Then writes `record` as the kept index. The
314/// removed skills that were `previously` kept, and whether the index was
315/// written.
316fn remove_others_and_record(
317    root: &Path,
318    record: &[IndexSkill],
319    previously: &BTreeSet<String>,
320) -> (Vec<String>, Result<(), String>) {
321    let mut removed = Vec::new();
322    if let Ok(entries) = std::fs::read_dir(root) {
323        for entry in entries.flatten() {
324            let name = entry.file_name().to_string_lossy().into_owned();
325            let path = entry.path();
326            // What an interrupted install or removal left.
327            if [STAGING, REMOVING]
328                .iter()
329                .any(|purpose| name.starts_with(&format!(".{purpose}-")))
330            {
331                let _ = std::fs::remove_dir_all(&path);
332                continue;
333            }
334            if name.starts_with('.') || record.iter().any(|skill| skill.name == name) {
335                continue;
336            }
337            let gone = if path.is_dir() {
338                remove_hidden(&path)
339            } else {
340                std::fs::remove_file(&path)
341            };
342            if gone.is_ok() && previously.contains(&name) {
343                removed.push(name);
344            }
345        }
346    }
347    let written = write_atomically(
348        &root.join(KEPT_INDEX),
349        &serde_json::to_vec_pretty(record).unwrap_or_default(),
350    )
351    .map_err(|error| error.to_string());
352    (removed, written)
353}

355fn valid_entry(skill: &IndexSkill) -> Result<(), String> {
356    let name_ok = !skill.name.is_empty()
357        && skill.name.len() <= 64
358        && !skill.name.starts_with('-')
359        && !skill.name.ends_with('-')
360        && !skill.name.contains("--")
361        && skill
362            .name
363            .chars()
364            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-');
365    if !name_ok {
366        return Err("not a skill name".into());
367    }
368    if skill.files.is_empty() || skill.files.len() > MAX_FILES_PER_SKILL {
369        return Err(format!("lists 1 to {MAX_FILES_PER_SKILL} files"));
370    }
371    if !skill.files.iter().any(|file| file.path == "SKILL.md") {
372        return Err("lists no SKILL.md".into());
373    }
374    let mut seen = BTreeSet::new();
375    for file in &skill.files {
376        let safe = !file.path.is_empty()
377            && !file.path.starts_with('/')
378            && !file.path.contains('\\')
379            && file.path.split('/').all(|part| {
380                !part.is_empty() && part != "." && part != ".." && !part.starts_with('.')
381            });
382        if !safe || !seen.insert(file.path.as_str()) {
383            return Err(format!("`{}` is not a file path it may have", file.path));
384        }
385        if file.sha256.len() != 64 || !file.sha256.chars().all(|c| c.is_ascii_hexdigit()) {
386            return Err(format!("`{}` has no SHA-256", file.path));
387        }
388    }
389    Ok(())
390}

392/// Brings `directory` in line with `skill`: downloads the files whose hash
393/// changed, checks every one, and only then writes them and removes files
394/// the index no longer lists. Whether anything changed.
395async fn update_skill(
396    client: &reqwest::Client,
397    base: &str,
398    directory: &Path,
399    skill: &IndexSkill,
400) -> Result<bool, String> {
401    // The files whose copy is missing or differs, found off the workers.
402    let stale = {
403        let directory = directory.to_owned();
404        let files = skill.files.clone();
405        local(move || {
406            files
407                .into_iter()
408                .filter(|file| {
409                    !std::fs::read(directory.join(&file.path))
410                        .is_ok_and(|bytes| sha256(&bytes).eq_ignore_ascii_case(&file.sha256))
411                })
412                .collect::<Vec<_>>()
413        })
414        .await?
415    };
416    let mut downloads = Vec::new();
417    for file in &stale {
418        let local = directory.join(&file.path);
419        let bytes = fetch(
420            client,
421            &format!("{base}/{}/{}", skill.name, file.path),
422            MAX_FILE_BYTES,
423        )
424        .await?;
425        let actual = sha256(&bytes);
426        if !actual.eq_ignore_ascii_case(&file.sha256) {
427            return Err(format!(
428                "`{}` does not match the index: its SHA-256 is {actual}",
429                file.path
430            ));
431        }
432        downloads.push((local, bytes));
433    }
434    let directory = directory.to_owned();
435    let listed = skill
436        .files
437        .iter()
438        .map(|file| directory.join(&file.path))
439        .collect::<BTreeSet<_>>();
440    local(move || install(&directory, downloads, &listed)).await?
441}

443/// Checks a downloaded `SKILL.md` loads, writes `downloads` into
444/// `directory`, and removes the files there not `listed`. Whether anything
445/// changed.
446fn install(
447    directory: &Path,
448    downloads: Vec<(PathBuf, Vec<u8>)>,
449    listed: &BTreeSet<PathBuf>,
450) -> Result<bool, String> {
451    if let Some((_, skill_md)) = downloads
452        .iter()
453        .find(|(path, _)| path.ends_with("SKILL.md") && path.parent() == Some(directory))
454    {
455        let text = std::str::from_utf8(skill_md).map_err(|_| "SKILL.md is not UTF-8")?;
456        cookie_agent_config::parse_skill_document(
457            text,
458            cookie_agent_protocol::SkillSource::Official,
459            &directory.join("SKILL.md"),
460            directory,
461            Vec::new(),
462        )
463        .map_err(|error| format!("SKILL.md does not load: {error}"))?;
464    }
465    let changed = !downloads.is_empty();
466    // A skill without its `SKILL.md` fails a configuration load, so a new
467    // one is written in a hidden directory, which loads skip, and renamed
468    // into place whole.
469    if !directory.join("SKILL.md").is_file() {
470        return install_new(directory, downloads).map(|()| changed);
471    }
472    for (path, bytes) in downloads {
473        write_atomically(&path, &bytes).map_err(|error| error.to_string())?;
474    }
475    let mut removed_any = false;
476    remove_unlisted(directory, listed, &mut removed_any);
477    Ok(changed || removed_any)
478}

480/// The hidden name a skill directory takes while it is written or removed.
481fn hidden_beside(directory: &Path, purpose: &str) -> Option<PathBuf> {
482    let name = directory.file_name()?.to_string_lossy();
483    Some(directory.with_file_name(format!(".{purpose}-{name}-{}", std::process::id())))
484}

486/// Writes a new skill's `downloads` into a hidden directory beside
487/// `directory`, then renames it into place, replacing any partial copy.
488fn install_new(directory: &Path, downloads: Vec<(PathBuf, Vec<u8>)>) -> Result<(), String> {
489    let staging = hidden_beside(directory, STAGING).ok_or("the skill has no directory name")?;
490    let _ = std::fs::remove_dir_all(&staging);
491    let written = (|| {
492        for (path, bytes) in downloads {
493            let relative = path
494                .strip_prefix(directory)
495                .map_err(|_| std::io::Error::other("a file outside the skill"))?;
496            write_atomically(&staging.join(relative), &bytes)?;
497        }
498        if directory.exists() {
499            remove_hidden(directory)?;
500        }
501        std::fs::rename(&staging, directory)
502    })();
503    written.map_err(|error| {
504        let _ = std::fs::remove_dir_all(&staging);
505        error.to_string()
506    })
507}

509/// Removes skill directory `directory`: renamed to a hidden name first, so
510/// a configuration load never finds it half removed.
511fn remove_hidden(directory: &Path) -> std::io::Result<()> {
512    let removing = hidden_beside(directory, REMOVING)
513        .ok_or_else(|| std::io::Error::other("the skill has no directory name"))?;
514    std::fs::rename(directory, &removing)?;
515    std::fs::remove_dir_all(&removing)
516}

518fn remove_unlisted(directory: &Path, listed: &BTreeSet<PathBuf>, removed: &mut bool) {
519    let Ok(entries) = std::fs::read_dir(directory) else {
520        return;
521    };
522    for entry in entries.flatten() {
523        let path = entry.path();
524        if path.is_dir() {
525            remove_unlisted(&path, listed, removed);
526            if std::fs::read_dir(&path).is_ok_and(|mut entries| entries.next().is_none()) {
527                let _ = std::fs::remove_dir(&path);
528            }
529        } else if !listed.contains(&path) && std::fs::remove_file(&path).is_ok() {
530            *removed = true;
531        }
532    }
533}

535async fn fetch(client: &reqwest::Client, url: &str, limit: usize) -> Result<Vec<u8>, String> {
536    let mut response = client
537        .get(url)
538        .send()
539        .await
540        .map_err(|error| format!("{url}: {error}"))?;
541    if !response.status().is_success() {
542        return Err(format!("{url}: HTTP {}", response.status()));
543    }
544    if response
545        .content_length()
546        .is_some_and(|length| length > limit as u64)
547    {
548        return Err(format!("{url}: larger than {limit} bytes"));
549    }
550    let mut bytes = Vec::new();
551    while let Some(chunk) = response
552        .chunk()
553        .await
554        .map_err(|error| format!("{url}: {error}"))?
555    {
556        bytes.extend_from_slice(&chunk);
557        if bytes.len() > limit {
558            return Err(format!("{url}: larger than {limit} bytes"));
559        }
560    }
561    Ok(bytes)
562}

564fn sha256(bytes: &[u8]) -> String {
565    format!("{:x}", Sha256::digest(bytes))
566}

568/// The hidden prefix of a new skill's directory while it is written.
569const STAGING: &str = "staging";
570/// The hidden prefix of a skill's directory while it is removed.
571const REMOVING: &str = "removing";

573/// Writes `bytes` to `path` through a temporary file beside it and a
574/// rename, creating the directories on the way.
575fn write_atomically(path: &Path, bytes: &[u8]) -> std::io::Result<()> {
576    let parent = path
577        .parent()
578        .ok_or_else(|| std::io::Error::other("no parent directory"))?;
579    std::fs::create_dir_all(parent)?;
580    let temporary = parent.join(format!(
581        ".{}.{}.tmp",
582        path.file_name()
583            .map(|name| name.to_string_lossy().into_owned())
584            .unwrap_or_default(),
585        std::process::id()
586    ));
587    std::fs::write(&temporary, bytes)?;
588    std::fs::rename(&temporary, path).inspect_err(|_| {
589        let _ = std::fs::remove_file(&temporary);
590    })
591}

593/// A semver version as ranges compare it: major, minor, patch, and the
594/// pre-release identifiers (build metadata ignored).
595#[derive(Clone, Debug, Eq, PartialEq)]
596struct SemVersion {
597    core: (u64, u64, u64),
598    pre: Vec<String>,
599}
600
601impl SemVersion {
602    fn parse(text: &str) -> Option<Self> {
603        let text = text.split('+').next()?;
604        let (core, pre) = match text.split_once('-') {
605            Some((core, pre)) => (core, Some(pre)),
606            None => (text, None),
607        };
608        let mut parts = core.split('.');
609        let major = parts.next()?.parse().ok()?;
610        let minor = parts.next().map_or(Some(0), |part| part.parse().ok())?;
611        let patch = parts.next().map_or(Some(0), |part| part.parse().ok())?;
612        if parts.next().is_some() {
613            return None;
614        }
615        let pre = match pre {
616            None => Vec::new(),
617            Some(pre) => {
618                let identifiers = pre.split('.').map(str::to_owned).collect::<Vec<_>>();
619                if identifiers.iter().any(|identifier| {
620                    identifier.is_empty()
621                        || !identifier
622                            .chars()
623                            .all(|c| c.is_ascii_alphanumeric() || c == '-')
624                }) {
625                    return None;
626                }
627                identifiers
628            }
629        };
630        Some(Self {
631            core: (major, minor, patch),
632            pre,
633        })
634    }
635}
636
637impl Ord for SemVersion {
638    /// Semver precedence: the core first; then a version without a
639    /// pre-release above one with it; then the identifiers in order,
640    /// numeric ones numerically and below alphanumeric ones, a longer list
641    /// above its prefix.
642    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
643        use std::cmp::Ordering;
644        self.core.cmp(&other.core).then_with(|| {
645            match (self.pre.is_empty(), other.pre.is_empty()) {
646                (true, true) => return Ordering::Equal,
647                (true, false) => return Ordering::Greater,
648                (false, true) => return Ordering::Less,
649                (false, false) => {}
650            }
651            for (left, right) in self.pre.iter().zip(&other.pre) {
652                let ordering = match (left.parse::<u64>(), right.parse::<u64>()) {
653                    (Ok(left), Ok(right)) => left.cmp(&right),
654                    (Ok(_), Err(_)) => Ordering::Less,
655                    (Err(_), Ok(_)) => Ordering::Greater,
656                    (Err(_), Err(_)) => left.cmp(right),
657                };
658                if ordering != Ordering::Equal {
659                    return ordering;
660                }
661            }
662            self.pre.len().cmp(&other.pre.len())
663        })
664    }
665}
666
667impl PartialOrd for SemVersion {
668    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
669        Some(self.cmp(other))
670    }
671}

673/// Whether `version` satisfies the semver range `range`: comparators
674/// (`>=`, `>`, `<=`, `<`, `=`, or a bare version for `=`) separated by
675/// commas or spaces, all of which must hold; `*` matches every version.
676/// Versions compare by plain semver precedence, pre-releases included:
677/// `0.4.0-alpha.7` satisfies `>=0.4.0-alpha.0` and `>=0.3.0-alpha.0` but not
678/// `>=0.4.0` (unlike Cargo's rule, a pre-release may match a comparator of
679/// another version). `None` when `range` or `version` does not read.
680pub fn version_satisfies(range: &str, version: &str) -> Option<bool> {
681    let version = SemVersion::parse(version)?;
682    let mut comparators = 0;
683    for comparator in range
684        .split([',', ' '])
685        .filter(|comparator| !comparator.is_empty())
686    {
687        comparators += 1;
688        if comparator == "*" {
689            continue;
690        }
691        let (operator, bound) = ["<=", ">=", "<", ">", "="]
692            .iter()
693            .find_map(|operator| {
694                comparator
695                    .strip_prefix(operator)
696                    .map(|bound| (*operator, bound))
697            })
698            .unwrap_or(("=", comparator));
699        let bound = SemVersion::parse(bound.trim())?;
700        let holds = match operator {
701            "<=" => version <= bound,
702            ">=" => version >= bound,
703            "<" => version < bound,
704            ">" => version > bound,
705            _ => version == bound,
706        };
707        if !holds {
708            return Some(false);
709        }
710    }
711    (comparators > 0).then_some(true)
712}
