20/// Why a `memory.*` request failed.
21#[derive(Debug, thiserror::Error)]
22pub enum MemoryRequestError {
23    /// The URL is not a memory URL.
24    #[error("{reason}")]
25    InvalidUrl { url: String, reason: String },
26    /// A member of the request is malformed or contradicts another.
27    #[error("{reason}")]
28    InvalidParams { field: &'static str, reason: String },
29    #[error("there is no memory at {url}")]
30    NotFound { url: String },
31    #[error("{url} already exists at revision {revision}")]
32    Exists { url: String, revision: String },
33    #[error("{url} changed since it was read; its current revision is {revision}")]
34    RevisionMismatch { url: String, revision: String },
35    #[error("{url} is not a valid memory: {reason}")]
36    Invalid { url: String, reason: String },
37    #[error("{reason}")]
38    Refused { url: String, reason: String },
39    /// Another process holds the memory lock.
40    #[error("{0}")]
41    Busy(String),
42    /// Reading or writing the memory folder failed.
43    #[error("{0}")]
44    Store(String),
45}
46
47impl From<MemoryError> for MemoryRequestError {
48    fn from(error: MemoryError) -> Self {
49        match error {
50            MemoryError::Contention(_) => Self::Busy(error.to_string()),
51            error => Self::Store(error.to_string()),
52        }
53    }
54}

56impl From<StoreError> for MemoryRequestError {
57    fn from(error: StoreError) -> Self {
58        match error {
59            StoreError::Memory(error) => error.into(),
60            StoreError::NotFound { url } => Self::NotFound {
61                url: url.to_string(),
62            },
63            StoreError::Exists { url, revision } => Self::Exists {
64                url: url.to_string(),
65                revision: revision.to_string(),
66            },
67            StoreError::RevisionMismatch { url, current } => Self::RevisionMismatch {
68                url: url.to_string(),
69                revision: current.to_string(),
70            },
71            StoreError::Invalid { url, reason } => Self::Invalid {
72                url: url.to_string(),
73                reason: reason.to_string(),
74            },
75            StoreError::TooLarge { ref url, .. } => Self::Refused {
76                url: url.to_string(),
77                reason: error.to_string(),
78            },
79            // Edits, seen revisions, and the similar-memory check are a
80            // tool's; a client replaces whole files at a revision.
81            error @ (StoreError::Refused(_)
82            | StoreError::Occurrences { .. }
83            | StoreError::Unseen { .. }
84            | StoreError::Similar { .. }
85            | StoreError::Identical { .. }) => Self::Store(error.to_string()),
86        }
87    }
88}

90fn parse_url(url: &str) -> Result<MemoryUrl, MemoryRequestError> {
91    MemoryUrl::parse(url).map_err(|error| MemoryRequestError::InvalidUrl {
92        url: url.to_owned(),
93        reason: error.to_string(),
94    })
95}

97fn parse_revision(revision: &str) -> Result<MemoryRevision, MemoryRequestError> {
98    MemoryRevision::parse(revision).ok_or_else(|| MemoryRequestError::InvalidParams {
99        field: "revision",
100        reason: format!("`{revision}` is not a memory revision (16 lowercase hex digits)"),
101    })
102}

104const fn layer_of(layer: url::MemoryLayer) -> MemoryLayer {
105    match layer {
106        url::MemoryLayer::User => MemoryLayer::User,
107        url::MemoryLayer::Agent => MemoryLayer::Agent,
108        url::MemoryLayer::Machine => MemoryLayer::Machine,
109        url::MemoryLayer::Project => MemoryLayer::Project,
110    }
111}

113/// The store's scope for `params`: at most one of `agent`, `machine`, and
114/// `project`, each agreeing with `layer` when that is given too.
115fn list_scope(params: MemoryListParams) -> Result<ListScope, MemoryRequestError> {
116    let invalid = |field, reason: &str| MemoryRequestError::InvalidParams {
117        field,
118        reason: reason.to_owned(),
119    };
120    let mut scopes = Vec::new();
121    if let Some(agent) = params.agent {
122        scopes.push(("agent", url::MemoryLayer::Agent, ListScope::Agent(agent)));
123    }
124    if let Some(machine) = params.machine {
125        scopes.push((
126            "machine",
127            url::MemoryLayer::Machine,
128            ListScope::Machine(machine),
129        ));
130    }
131    if let Some(project) = params.project {
132        let name = MemoryName::new(project.as_str()).ok_or_else(|| {
133            invalid(
134                "project",
135                &format!(
136                    "`{project}` is not a project name (a lowercase letter or digit, then up to 63 lowercase letters, digits, or `-`)"
137                ),
138            )
139        })?;
140        scopes.push((
141            "project",
142            url::MemoryLayer::Project,
143            ListScope::Project(name),
144        ));
145    }
146    let layer = match params.layer {
147        None => None,
148        Some(MemoryLayer::User) => Some(url::MemoryLayer::User),
149        Some(MemoryLayer::Agent) => Some(url::MemoryLayer::Agent),
150        Some(MemoryLayer::Machine) => Some(url::MemoryLayer::Machine),
151        Some(MemoryLayer::Project) => Some(url::MemoryLayer::Project),
152        Some(MemoryLayer::Unknown) => {
153            return Err(invalid("layer", "the layer is not one this daemon knows"));
154        }
155    };
156    if scopes.len() > 1 {
157        return Err(invalid(
158            scopes[1].0,
159            "give at most one of agent, machine, and project",
160        ));
161    }
162    match (scopes.pop(), layer) {
163        (Some((field, scope_layer, _)), Some(layer)) if scope_layer != layer => Err(invalid(
164            field,
165            &format!("{field} lists the {scope_layer} layer, not {layer}"),
166        )),
167        (Some((_, _, scope)), _) => Ok(scope),
168        (None, Some(layer)) => Ok(ListScope::Layer(layer)),
169        (None, None) => Ok(ListScope::All),
170    }
171}

173impl Engine {
174    /// Runs a store call on a blocking thread: it reads files and may wait
175    /// for the memory lock.
176    async fn memory_blocking<T: Send + 'static>(
177        &self,
178        call: impl FnOnce(Engine) -> Result<T, MemoryRequestError> + Send + 'static,
179    ) -> Result<T, EngineError> {
180        let engine = self.clone();
181        tokio::task::spawn_blocking(move || call(engine))
182            .await
183            .map_err(|error| {
184                MemoryRequestError::Store(format!("the request did not finish: {error}"))
185            })?
186            .map_err(EngineError::from)
187    }
188
189    /// `memory.list`: the memory files `params` asks for, valid or not,
190    /// with the agent files whose agent the registry no longer has marked.
191    pub async fn list_memory(
192        &self,
193        params: MemoryListParams,
194    ) -> Result<MemoryListResult, EngineError> {
195        let scope = list_scope(params)?;
196        self.memory_blocking(move |engine| {
197            let limits = engine.memory_limits();
198            let listing = engine.memory_root().list(&scope)?;
199            Ok(MemoryListResult {
200                memories: listing
201                    .entries
202                    .into_iter()
203                    .map(|entry| memory_info(&engine, entry, &limits))
204                    .collect(),
205                ignored: listing
206                    .ignored
207                    .into_iter()
208                    .map(|ignored| MemoryIgnoredEntry {
209                        path: ignored.path,
210                        reason: ignored.reason.to_owned(),
211                    })
212                    .collect(),
213            })
214        })
215        .await
216    }

218    /// `memory.read`: one memory's whole text and revision.
219    pub async fn read_memory(
220        &self,
221        params: MemoryReadParams,
222    ) -> Result<MemoryReadResult, EngineError> {
223        let url = parse_url(&params.url)?;
224        self.memory_blocking(move |engine| {
225            let limits = engine.memory_limits();
226            let read = engine.memory_root().read(&url)?;
227            Ok(MemoryReadResult {
228                text: read.text,
229                revision: read.entry.revision.as_str().to_owned(),
230                invalid: read.entry.document.err().map(|reason| reason.to_string()),
231                created: Some(read.entry.created),
232                modified: Some(read.entry.modified),
233                size: Some(read.entry.size),
234                max_size: Some(limits.max_bytes(&url) as u64),
235                url: url.to_string(),
236            })
237        })
238        .await
239    }

241    /// `memory.write`: creates or replaces one memory from its whole text,
242    /// for the user. The user may create a `user://` category; an
243    /// `agent://` file still needs an agent the registry has, and keeps the
244    /// description and keywords the daemon owns, which the result's
245    /// `notice` reports (section 16.7). A `project://` file needs a project
246    /// that exists (`docs/site/specs/projects.md` section 3).
247    pub async fn write_memory(
248        &self,
249        params: MemoryWriteParams,
250    ) -> Result<MemoryWriteResult, EngineError> {
251        let url = parse_url(&params.url)?;
252        let revision = params.revision.as_deref().map(parse_revision).transpose()?;
253        let text = params.text;
254        self.memory_blocking(move |engine| {
255            let limits = engine.memory_limits();
256            let written = engine
257                .memory_root()
258                .write(
259                    &url,
260                    &text,
261                    WriteMode::Client(revision.as_ref()),
262                    super::projects::EngineMemoryOwners(&engine),
263                    &limits,
264                    jiff::Timestamp::now(),
265                )
266                .map_err(|error| match error {
267                    StoreError::Refused(reason) => MemoryRequestError::Refused {
268                        url: url.to_string(),
269                        reason,
270                    },
271                    error => error.into(),
272                })?;
273            Ok(MemoryWriteResult {
274                url: url.to_string(),
275                revision: written.revision.as_str().to_owned(),
276                notice: written.kept.as_ref().map(|kept| kept.describe(&url)),
277            })
278        })
279        .await
280    }
281
282    /// `memory.delete`: deletes one memory at `revision`, for the user, who
283    /// may delete `user://` and `agent://` files too.
284    pub async fn delete_memory(&self, params: MemoryDeleteParams) -> Result<(), EngineError> {
285        let url = parse_url(&params.url)?;
286        let revision = parse_revision(&params.revision)?;
287        self.memory_blocking(move |engine| {
288            engine
289                .memory_root()
290                .delete(&url, Some(&revision))
291                .map(drop)
292                .map_err(|error| match error {
293                    StoreError::Refused(reason) => MemoryRequestError::Refused {
294                        url: url.to_string(),
295                        reason,
296                    },
297                    error => error.into(),
298                })
299        })
300        .await
301    }

311fn memory_info(engine: &Engine, entry: MemoryEntry, limits: &MemoryLimits) -> MemoryInfo {
312    let orphaned = match &entry.url {
313        MemoryUrl::Agent { agent } => !engine.agent_registered(agent),
314        _ => false,
315    };
316    let (description, keywords, invalid) = match entry.document {
317        Ok(document) => (
318            Some(document.description().to_owned()),
319            document.keywords().to_vec(),
320            None,
321        ),
322        Err(reason) => (None, Vec::new(), Some(reason.to_string())),
323    };
324    MemoryInfo {
325        created: Some(entry.created),
326        modified: Some(entry.modified),
327        last_read: engine.memory_root().last_read(&entry.url),
328        max_size: Some(limits.max_bytes(&entry.url) as u64),
329        layer: layer_of(entry.url.layer()),
330        group: entry.url.group().map(str::to_owned),
331        url: entry.url.to_string(),
332        revision: entry.revision.as_str().to_owned(),
333        size: entry.size,
334        description,
335        keywords,
336        invalid,
337        orphaned,
338    }
