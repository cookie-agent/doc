1//! The request prefix each assistant turn was made in, for models that bind
2//! replayed reasoning to it (`docs/site/specs/frozen-context.md`, section 4).
3//!
4//! A turn's fingerprint covers what a request sent before it: the system
5//! prompt and every earlier message (all in the history, reasoning included as
6//! sent) and the eager tools. Deferred tools are left out: Anthropic does not
7//! bind reasoning to them. Cache markers are not in the history yet when it is
8//! taken.
9
10use cookie_agent_protocol::Sha256Digest;
11use oven_sdk::{AssistantPart, HistoryTurn, ToolDefinition};
12use sha2::{Digest as _, Sha256};
13
14/// The fingerprints of one request.
15pub(crate) struct PrefixFingerprints {
16    /// One per assistant turn of the history, in order: the prefix before
17    /// it.
18    pub(crate) assistant: Vec<Sha256Digest>,
19    /// The whole request: the prefix of the turn it makes.
20    pub(crate) whole: Sha256Digest,
21}
22
23pub(crate) fn prefix_fingerprints(
24    history: &[HistoryTurn],
25    tools: &[ToolDefinition],
26) -> Result<PrefixFingerprints, serde_json::Error> {
27    let mut hasher = Sha256::new();
28    hasher.update(b"cookie.prefix.v1\0");
29    for tool in tools.iter().filter(|tool| !tool.is_deferred()) {
30        serde_json::to_writer(HashWriter(&mut hasher), tool)?;
31        hasher.update([0]);
32    }
33    hasher.update([1]);
34    let mut assistant = Vec::new();
35    for turn in history {
36        if matches!(turn, HistoryTurn::Assistant(_)) {
37            assistant.push(digest(hasher.clone()));
38        }
39        serde_json::to_writer(HashWriter(&mut hasher), turn)?;
40        hasher.update([0]);
41    }
42    Ok(PrefixFingerprints {
43        assistant,
44        whole: digest(hasher),
45    })
46}

48/// Whether an assistant turn replays reasoning a provider may bind.
49pub(crate) fn carries_reasoning(turn: &HistoryTurn) -> bool {
50    match turn {
51        HistoryTurn::Assistant(turn) => {
52            turn.finish.native_replay.is_some()
53                || turn
54                    .message
55                    .content
56                    .iter()
57                    .any(|part| matches!(part, AssistantPart::Reasoning(_)))
58        }
59        _ => false,
60    }
61}
62
63fn digest(hasher: Sha256) -> Sha256Digest {
64    let hex = hasher
65        .finalize()
66        .iter()
67        .fold(String::with_capacity(64), |mut hex, byte| {
68            use std::fmt::Write as _;
69            let _ = write!(hex, "{byte:02x}");
70            hex
71        });
72    Sha256Digest::new(hex).expect("SHA-256 hex digest is well-formed")
73}
74
75struct HashWriter<'a>(&'a mut Sha256);
76
77impl std::io::Write for HashWriter<'_> {
78    fn write(&mut self, buffer: &[u8]) -> std::io::Result<usize> {
79        self.0.update(buffer);
80        Ok(buffer.len())
81    }
82
83    fn flush(&mut self) -> std::io::Result<()> {
84        Ok(())
85    }
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91    use oven_sdk::{
92        AssistantMessage, CompletedTurn, Finish, FinishReason, InputPart, JsonSchema,
93        ReasoningPart, SystemMessage, SystemPart, TextPart, UserMessage,
94    };
95
96    fn system(text: &str) -> HistoryTurn {
97        HistoryTurn::system(SystemMessage::new(vec![SystemPart::Text(TextPart::new(
98            text,
99        ))]))
100    }
101
102    fn user(text: &str) -> HistoryTurn {
103        HistoryTurn::user(UserMessage::new(vec![InputPart::Text(TextPart::new(text))]))
104    }
105
106    fn assistant(text: &str) -> HistoryTurn {
107        HistoryTurn::assistant(CompletedTurn::new(
108            AssistantMessage::new(vec![AssistantPart::Text(TextPart::new(text))]),
109            Finish::new(oven_sdk::Usage::default(), FinishReason::Stop),
110        ))
111    }
112
113    fn tool(name: &str, deferred: bool) -> ToolDefinition {
114        let tool = ToolDefinition::new(
115            name,
116            "a tool",
117            JsonSchema::new(serde_json::json!({"type": "object"})).unwrap(),
118        );
119        if deferred { tool.deferred() } else { tool }
120    }
121
122    #[test]
123    fn a_turn_keeps_its_fingerprint_while_what_came_before_it_is_unchanged() {
124        let tools = [tool("read", false)];
125        let first = [system("s"), user("u1"), assistant("a1")];
126        let made = prefix_fingerprints(&first[..2], &tools).unwrap().whole;
127        let later = [system("s"), user("u1"), assistant("a1"), user("u2")];
128        let fingerprints = prefix_fingerprints(&later, &tools).unwrap();
129        assert_eq!(fingerprints.assistant, vec![made.clone()]);
130        assert_ne!(fingerprints.whole, made);
131    }
132
133    #[test]
134    fn a_changed_system_prompt_earlier_message_or_eager_tool_changes_it() {
135        let tools = [tool("read", false)];
136        let base = prefix_fingerprints(&[system("s"), user("u1")], &tools)
137            .unwrap()
138            .whole;
139        for (history, tools) in [
140            (vec![system("t"), user("u1")], vec![tool("read", false)]),
141            (vec![system("s"), user("u0")], vec![tool("read", false)]),
142            (
143                vec![system("s"), user("u1")],
144                vec![tool("read", false), tool("write", false)],
145            ),
146        ] {
147            assert_ne!(prefix_fingerprints(&history, &tools).unwrap().whole, base);
148        }
149    }
150
151    #[test]
152    fn deferred_tools_do_not_count() {
153        let history = [system("s"), user("u1")];
154        assert_eq!(
155            prefix_fingerprints(&history, &[tool("read", false)])
156                .unwrap()
157                .whole,
158            prefix_fingerprints(&history, &[tool("read", false), tool("later", true)])
159                .unwrap()
160                .whole
161        );
162    }
163
164    #[test]
165    fn only_turns_with_reasoning_or_native_replay_carry_reasoning() {
166        assert!(!carries_reasoning(&assistant("a")));
167        assert!(!carries_reasoning(&user("u")));
168        let thought = HistoryTurn::assistant(CompletedTurn::new(
169            AssistantMessage::new(vec![AssistantPart::Reasoning(ReasoningPart::new("think"))]),
170            Finish::new(oven_sdk::Usage::default(), FinishReason::Stop),
171        ));
172        assert!(carries_reasoning(&thought));
173    }
