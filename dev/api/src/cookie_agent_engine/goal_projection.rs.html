11#[derive(Clone, Debug, Eq, PartialEq)]
12pub(crate) struct ProducerMessageRecord {
13    pub message_id: ProducerMessageId,
14    pub producer_owner: ProducerOwner,
15    pub mode: ProducerDeliveryMode,
16    pub idempotency_key: ProducerIdempotencyKey,
17    pub description: SafeDisplayText,

20    /// `send_message` hop depth stamped by the guard. `None` means the message

26    pub claims: HashSet<u64>,
27    pub consumed: bool,
28    pub discarded: bool,

30    pub consumption_recorded: bool,
31    pub consumed_run: Option<RunId>,
32}

49impl GoalProducerProjection {
50    pub(crate) fn next_reminder_kind(&self, goal_id: GoalId) -> GoalReminderKind {
51        // Acceptance and claims are retractable; only committed input coverage introduces a goal.
52        if self.messages.iter().any(|message| {
53            message.consumed
54                && message
55                    .reminder
56                    .is_some_and(|reminder| reminder.goal_id == goal_id)
57        }) {
58            GoalReminderKind::Continuation
59        } else {
60            GoalReminderKind::Started
61        }
62    }
63
64    /// Hop depth a `send_message` from `run` must carry: one more than the
65    /// deepest Agent-owned mail this session's log shows `run` has seen, or `0`
66    /// when the run has seen no agent mail.
67    ///
68    /// The basis is durable observation — admission, a live claim, or committed
69    /// consumption — not payload heuristics, so a chain cannot be reset by
70    /// releasing a claim or by restarting the process. A message with no hop
71    /// metadata is a pre-guard send and counts as depth `0`.
72    ///
73    /// This over-approximates on purpose: any send from a run that consumed
74    /// hop-`N` mail inherits `N + 1`, even when the model treats the send as an
75    /// unrelated topic. The guard's bound, not per-message linkage, is the
76    /// control operators tune.
77    #[must_use]
78    pub(crate) fn inherited_agent_hop(&self, run: RunId) -> u32 {
79        self.messages
80            .iter()
81            .filter(|message| {
82                matches!(message.producer_owner, ProducerOwner::Agent { .. })
83                    && (message.consumed_run == Some(run)
84                        || message
85                            .admission
86                            .is_some_and(|(admitted_run, _)| admitted_run == run)
87                        || self.claims.values().any(|claim| {
88                            claim.run_id == run && claim.message_ids.contains(&message.message_id)
89                        }))
90            })
91            .map(|message| message.agent_hop.unwrap_or(0))
92            .max()
93            .map_or(0, |deepest| deepest.saturating_add(1))
94    }

96    #[must_use]
97    pub(crate) fn from_events(events: &[StoredEvent]) -> Self {
98        let mut projection = Self::default();
99        let mut goal_revisions = HashMap::<cookie_agent_protocol::GoalId, u64>::new();
100        let mut message_indexes = HashMap::<ProducerMessageId, usize>::new();
101        let mut dedup_indexes = HashMap::<(ProducerOwner, ProducerIdempotencyKey), usize>::new();
102        let mut terminal_runs = HashSet::<RunId>::new();
103
104        for event in events {
105            match &event.payload {
106                EventPayload::GoalActivated {
107                    goal_id,
108                    objective,
109                    revision,
110                    selection,
111                } => {
112                    let replaceable = projection
113                        .goal
114                        .as_ref()
115                        .is_none_or(|goal| is_terminal(goal.status));
116                    let distinct = projection
117                        .goal
118                        .as_ref()
119                        .is_none_or(|goal| goal.goal_id != *goal_id);
120                    let unseen = !goal_revisions.contains_key(goal_id);
121                    if objective.trim().is_empty() || !replaceable || !distinct || !unseen {
122                        projection.reject(event.seq, "invalid goal activation");
123                        continue;
124                    }
125                    projection.goal = Some(GoalState {
126                        goal_id: *goal_id,
127                        objective: objective.clone(),
128                        status: GoalStatus::Active,
129                        items: Vec::new(),
130                        revision: *revision,
131                    });
132                    projection.selection = selection.clone();
133                    goal_revisions.insert(*goal_id, *revision);
134                }
135                EventPayload::GoalChecklistRevised {
136                    goal_id,
137                    items,
138                    revision,
139                } => {
140                    let valid_items = items.iter().all(|item| !item.description.trim().is_empty());
141                    let valid = valid_items
142                        && projection.goal.as_ref().is_some_and(|goal| {
143                            goal.goal_id == *goal_id
144                                && !is_terminal(goal.status)
145                                && *revision > goal.revision
146                        });
147                    if !valid {
148                        projection.reject(event.seq, "invalid goal checklist revision");
149                        continue;
150                    }
151                    let goal = projection.goal.as_mut().expect("validated current goal");
152                    goal.items = items.clone();
153                    goal.revision = *revision;
154                    goal_revisions.insert(*goal_id, *revision);
155                }
156                EventPayload::GoalLifecycleChanged {
157                    goal_id,
158                    status,
159                    revision,
160                    selection,
161                } => {
162                    let valid = projection.goal.as_ref().is_some_and(|goal| {
163                        goal.goal_id == *goal_id
164                            && *revision > goal.revision
165                            && valid_lifecycle_change(goal, *status)
166                            && (selection.is_none() || *status == GoalStatus::Active)
167                    });
168                    if !valid {
169                        projection.reject(event.seq, "invalid goal lifecycle transition");
170                        continue;
171                    }
172                    let goal = projection.goal.as_mut().expect("validated current goal");
173                    goal.status = *status;
174                    goal.revision = *revision;
175                    if let Some(selection) = selection {
176                        projection.selection = Some(selection.clone());
177                    }
178                    goal_revisions.insert(*goal_id, *revision);
179                }
180                EventPayload::ProducerMessageAccepted {
181                    message_id,
182                    producer_owner,
183                    mode,
184                    idempotency_key,
185                    description,
186                    body,
187                    reminder,
188                    agent_hop,
189                } => {
190                    if !valid_reminder_owner(producer_owner, reminder.as_ref()) {
191                        projection.reject(event.seq, "invalid producer reminder ownership");
192                        continue;
193                    }
194                    if !valid_hop_owner(producer_owner, *agent_hop) {
195                        projection.reject(event.seq, "invalid producer hop ownership");
196                        continue;
197                    }
198                    let dedup_key = (producer_owner.clone(), idempotency_key.clone());
199                    if let Some(index) = dedup_indexes.get(&dedup_key).copied() {
200                        let prior = &projection.messages[index];
201                        // `agent_hop` is guard metadata, not payload: a replayed
202                        // acceptance for the same key stays idempotent even when
203                        // the stamped depth differs from the stored record.
204                        let exact = prior.message_id == *message_id
205                            && prior.mode == *mode
206                            && prior.description == *description
207                            && prior.body == *body
208                            && prior.reminder == *reminder;
209                        if exact {
210                            continue;
211                        }
212                        projection.reject(event.seq, "conflicting producer idempotency key");
213                        continue;
214                    }
215                    if message_indexes.contains_key(message_id) {
216                        projection.reject(event.seq, "conflicting producer message id");
217                        continue;
218                    }
219                    let index = projection.messages.len();
220                    projection.messages.push(ProducerMessageRecord {
221                        message_id: *message_id,
222                        producer_owner: producer_owner.clone(),
223                        mode: *mode,
224                        idempotency_key: idempotency_key.clone(),
225                        description: description.clone(),
226                        body: body.clone(),
227                        reminder: *reminder,
228                        agent_hop: *agent_hop,
229                        accepted_seq: event.seq,
230                        admission: None,
231                        claims: HashSet::new(),
232                        consumed: false,
233                        discarded: false,
234                        discarded_seq: None,
235                        consumption_recorded: false,
236                        consumed_run: None,
237                    });
238                    message_indexes.insert(*message_id, index);
239                    dedup_indexes.insert(dedup_key, index);
240                }
241                EventPayload::ProducerMessageAdmitted { message_id } => {
242                    let Some(run_id) = event.run_id else {
243                        projection.reject(event.seq, "producer admission is missing run id");
244                        continue;
245                    };
246                    if terminal_runs.contains(&run_id) {
247                        projection.reject(event.seq, "producer admission targets a terminal run");
248                        continue;
249                    }
250                    let Some(index) = message_indexes.get(message_id).copied() else {
251                        projection.reject(event.seq, "producer admission has no accepted message");
252                        continue;
253                    };
254                    let message = &projection.messages[index];
255                    let replaceable = message.admission.is_some_and(|(prior_run, _)| {
256                        prior_run != run_id && terminal_runs.contains(&prior_run)
257                    });
258                    if message.consumed
259                        || message.discarded_seq.is_some()
260                        || (message.admission.is_some() && !replaceable)
261                    {
262                        projection.reject(event.seq, "invalid producer admission");
263                        continue;
264                    }
265                    projection.messages[index].admission = Some((run_id, event.seq));
266                }
267                EventPayload::ProducerMessagesClaimed { message_ids } => {
268                    let Some(run_id) = event.run_id else {
269                        projection.reject(event.seq, "producer claim is missing run id");
270                        continue;
271                    };
272                    let unique = message_ids.iter().copied().collect::<HashSet<_>>();
273                    let indexes = message_ids
274                        .iter()
275                        .filter_map(|message_id| message_indexes.get(message_id).copied())
276                        .collect::<Vec<_>>();
277                    let valid = !message_ids.is_empty()
278                        && unique.len() == message_ids.len()
279                        && indexes.len() == message_ids.len()
280                        && !projection.claims.contains_key(&event.seq)
281                        && indexes.iter().all(|index| {
282                            let message = &projection.messages[*index];
283                            message.accepted_seq < event.seq
284                                && !message.consumed
285                                && message.discarded_seq.is_none()
286                                && message.admission.is_some_and(
287                                    |(admission_run, admission_seq)| {
288                                        admission_run == run_id && admission_seq < event.seq
289                                    },
290                                )
291                        });
292                    if !valid {
293                        projection.reject(event.seq, "invalid producer claim");
294                        continue;
295                    }
296                    for index in indexes {
297                        projection.messages[index].claims.insert(event.seq);
298                    }
299                    projection.claims.insert(
300                        event.seq,
301                        ProducerClaimRecord {
302                            run_id,
303                            message_ids: message_ids.clone(),
304                        },
305                    );
306                }
307                EventPayload::ProducerMessagesReleased { claim_seq } => {
308                    let Some(run_id) = event.run_id else {
309                        projection.reject(event.seq, "producer release is missing run id");
310                        continue;
311                    };
312                    let Some(claim) = projection.claims.get(claim_seq) else {
313                        projection.reject(event.seq, "producer release has no claim");
314                        continue;
315                    };
316                    if *claim_seq == 0 || claim.run_id != run_id {
317                        projection.reject(event.seq, "producer release ownership is invalid");
318                        continue;
319                    }
320                    let message_ids = claim.message_ids.clone();
321                    for message_id in &message_ids {
322                        if let Some(index) = message_indexes.get(message_id) {
323                            projection.messages[*index].claims.remove(claim_seq);
324                        }
325                    }
326                    projection.claims.remove(claim_seq);
327                }
328                EventPayload::ModelTurnCommitted {
329                    input_through_seq, ..
330                } => {
331                    let Some(run_id) = event.run_id else { continue };
332                    for message in &mut projection.messages {
333                        if !message.consumed
334                            && message
335                                .admission
336                                .is_some_and(|(admission_run, admission_seq)| {
337                                    admission_run == run_id && admission_seq <= *input_through_seq
338                                })
339                            && message
340                                .discarded_seq
341                                .is_none_or(|discarded_seq| discarded_seq > *input_through_seq)
342                        {
343                            message.discarded = false;
344                            message.discarded_seq = None;
345                            message.consumed = true;
346                            message.consumed_run = Some(run_id);
347                        }
348                    }
349                }
350                EventPayload::ProducerMessageConsumed { message_id, run_id } => {
351                    let Some(index) = message_indexes.get(message_id).copied() else {
352                        projection
353                            .reject(event.seq, "producer consumption has no accepted message");
354                        continue;
355                    };
356                    let message = &projection.messages[index];
357                    if event.run_id != Some(*run_id)
358                        || message.admission.map(|(run, _)| run) != Some(*run_id)
359                        || message.consumed_run != Some(*run_id)
360                        || message.consumption_recorded
361                    {
362                        projection.reject(event.seq, "invalid producer consumption marker");
363                        continue;
364                    }
365                    projection.messages[index].consumption_recorded = true;
366                }
367                EventPayload::ProducerMessageDiscarded {
368                    message_id,
369                    reminder,
370                    producer_owner,
371                } => {
372                    let Some(index) = message_indexes.get(message_id).copied() else {
373                        projection.reject(event.seq, "producer discard has no accepted message");
374                        continue;
375                    };
376                    let message = &projection.messages[index];
377                    let identity_matches = producer_owner
378                        .as_ref()
379                        .is_some_and(|owner| owner == &message.producer_owner)
380                        && reminder
381                            .as_ref()
382                            .is_none_or(|identity| message.reminder.as_ref() == Some(identity));
383                    let legacy_matches = producer_owner.is_none()
384                        && reminder
385                            .as_ref()
386                            .is_some_and(|identity| message.reminder.as_ref() == Some(identity));
387                    if (!identity_matches && !legacy_matches)
388                        || message.consumed
389                        || !message.claims.is_empty()
390                    {
391                        projection.reject(event.seq, "invalid producer message discard");
392                        continue;
393                    }
394                    if message.discarded_seq.is_none() {
395                        projection.messages[index].discarded = true;
396                        projection.messages[index].discarded_seq = Some(event.seq);
397                    }
398                }
399                EventPayload::RunCompleted { .. }
400                | EventPayload::RunFailed { .. }
401                | EventPayload::RunCancelled { .. }
402                | EventPayload::RunInterrupted { .. } => {
403                    if let Some(run_id) = event.run_id {
404                        terminal_runs.insert(run_id);
405                    }
406                }
407                _ => {}
408            }
409        }
410        projection
411    }

413    fn reject(&mut self, seq: u64, reason: &str) {
414        self.invalid.push((seq, reason.to_owned()));
415    }

418fn is_terminal(status: GoalStatus) -> bool {
419    matches!(status, GoalStatus::Completed | GoalStatus::Cancelled)
420}
421
422fn valid_lifecycle_change(goal: &GoalState, status: GoalStatus) -> bool {
423    match (goal.status, status) {
424        (GoalStatus::Active, GoalStatus::Paused | GoalStatus::Cancelled)
425        | (GoalStatus::Paused, GoalStatus::Active | GoalStatus::Cancelled) => true,
426        (GoalStatus::Active | GoalStatus::Paused, GoalStatus::Completed) => {
427            !goal.items.is_empty() && goal.items.iter().all(|item| item.finished)
428        }
429        _ => false,
430    }
431}
432
433fn valid_reminder_owner(owner: &ProducerOwner, reminder: Option<&GoalReminderIdentity>) -> bool {
434    if matches!(owner, ProducerOwner::Plugin { plugin } if plugin.trim().is_empty()) {
435        return false;
436    }
437    match (owner, reminder) {
438        (ProducerOwner::Goal { goal_id }, Some(reminder)) => *goal_id == reminder.goal_id,
439        (ProducerOwner::Goal { .. }, None) | (_, Some(_)) => false,
440        (_, None) => true,
441    }
442}

444/// Hop metadata belongs exclusively to agent mail. A malformed durable event
445/// that stamps it onto plugin, goal, or delegation mail is quarantined so it
446/// cannot inflate an agent chain it never joined.
447const fn valid_hop_owner(owner: &ProducerOwner, agent_hop: Option<u32>) -> bool {
448    agent_hop.is_none() || matches!(owner, ProducerOwner::Agent { .. })
449}
