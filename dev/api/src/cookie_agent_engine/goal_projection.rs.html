43/// Whether a run is offered the goal tools: every root run, so its agent can
44/// start a goal itself (stable prefix spec, section 3), except an ACP
45/// agent's, which sees them only while the user's goal is active or paused.
46pub(crate) fn goal_tools_offered<E: Borrow<StoredEvent>>(
47    root: bool,
48    acp: bool,
49    events: &[E],
50) -> bool {
51    root && (!acp
52        || GoalProducerProjection::from_events(events)
53            .goal
54            .is_some_and(|goal| matches!(goal.status, GoalStatus::Active | GoalStatus::Paused)))
55}

57#[derive(Clone, Debug, Default, Eq, PartialEq)]
58pub(crate) struct GoalProducerProjection {
59    pub goal: Option<GoalState>,
60    pub selection: Option<RunSelection>,
61    pub messages: Vec<ProducerMessageRecord>,
62    pub claims: HashMap<u64, ProducerClaimRecord>,
63    pub invalid: Vec<(u64, String)>,
64}
65
66impl GoalProducerProjection {
67    pub(crate) fn next_reminder_kind(&self, goal_id: GoalId) -> GoalReminderKind {
68        // Acceptance and claims are retractable; only committed input coverage introduces a goal.
69        if self.messages.iter().any(|message| {
70            message.consumed
71                && message
72                    .reminder
73                    .is_some_and(|reminder| reminder.goal_id == goal_id)
74        }) {
75            GoalReminderKind::Continuation
76        } else {
77            GoalReminderKind::Started
78        }
79    }
80
81    /// Hop depth a `send_message` from `run` must carry: one more than the
82    /// deepest Agent-owned mail this session's log shows `run` has seen, or `0`
83    /// when the run has seen no agent mail.
84    ///
85    /// The basis is durable observation — admission, a live claim, or committed
86    /// consumption — not payload heuristics, so a chain cannot be reset by
87    /// releasing a claim or by restarting the process. A message with no hop
88    /// metadata is a pre-guard send and counts as depth `0`.
89    ///
90    /// This over-approximates on purpose: any send from a run that consumed
91    /// hop-`N` mail inherits `N + 1`, even when the model treats the send as an
92    /// unrelated topic. The guard's bound, not per-message linkage, is the
93    /// control operators tune.
94    #[must_use]
95    pub(crate) fn inherited_agent_hop(&self, run: RunId) -> u32 {
96        self.messages
97            .iter()
98            .filter(|message| {
99                matches!(message.producer_owner, ProducerOwner::Agent { .. })
100                    && (message.consumed_run == Some(run)
101                        || message
102                            .admission
103                            .is_some_and(|(admitted_run, _)| admitted_run == run)
104                        || self.claims.values().any(|claim| {
105                            claim.run_id == run && claim.message_ids.contains(&message.message_id)
106                        }))
107            })
108            .filter_map(|message| message.agent_hop)
109            .max()
110            .map_or(0, |deepest| deepest.saturating_add(1))
111    }
112
113    #[must_use]
114    pub(crate) fn from_events<E: Borrow<StoredEvent>>(events: &[E]) -> Self {
115        let mut fold = GoalProducerFold::default();
116        for event in events {
117            fold.observe(event.borrow());
118        }
119        Arc::unwrap_or_clone(fold.projection)
120    }
121
122    fn reject(&mut self, seq: u64, reason: &str) {
123        self.invalid.push((seq, reason.to_owned()));
124    }
125}

127/// [`GoalProducerProjection`] folded one event at a time, so a log that
128/// grew is folded only through what it gained.
129#[derive(Clone, Debug, Default)]
130pub(crate) struct GoalProducerFold {
131    /// Shared with the readers handed it; copied only when an event changes
132    /// it while one still holds it.
133    projection: Arc<GoalProducerProjection>,
134    goals: GoalFold,
135    message_indexes: HashMap<ProducerMessageId, usize>,
136    dedup_indexes: HashMap<(ProducerOwner, ProducerIdempotencyKey), usize>,
137    terminal_runs: HashSet<RunId>,
138}
139
140impl GoalProducerFold {
141    /// The projection through the events observed so far.
142    #[must_use]
143    pub(crate) fn projection(&self) -> Arc<GoalProducerProjection> {
144        Arc::clone(&self.projection)
145    }
146
147    fn projection_mut(&mut self) -> &mut GoalProducerProjection {
148        Arc::make_mut(&mut self.projection)
149    }
150
151    /// Folds the next event of the log.
152    pub(crate) fn observe(&mut self, event: &StoredEvent) {
153        match &event.payload {
154            EventPayload::GoalActivated { selection, .. } => match self.goals.observe(event) {
155                Ok(()) => {
156                    let goal = self.goals.goal.clone();
157                    let projection = self.projection_mut();
158                    projection.selection = selection.clone();
159                    projection.goal = goal;
160                }
161                Err(reason) => self.projection_mut().reject(event.seq, reason),
162            },
163            EventPayload::GoalChecklistRevised { .. } => match self.goals.observe(event) {
164                Ok(()) => self.projection_mut().goal = self.goals.goal.clone(),
165                Err(reason) => self.projection_mut().reject(event.seq, reason),
166            },
167            EventPayload::GoalLifecycleChanged { selection, .. } => {
168                match self.goals.observe(event) {
169                    Ok(()) => {
170                        let goal = self.goals.goal.clone();
171                        let projection = self.projection_mut();
172                        if let Some(selection) = selection {
173                            projection.selection = Some(selection.clone());
174                        }
175                        projection.goal = goal;
176                    }
177                    Err(reason) => self.projection_mut().reject(event.seq, reason),
178                }
179            }
180            EventPayload::ProducerMessageAccepted {
181                message_id,
182                producer_owner,
183                mode,
184                idempotency_key,
185                description,
186                body,
187                reminder,
188                agent_hop,
189                ..
190            } => {
191                if !valid_reminder_owner(producer_owner, reminder.as_ref()) {
192                    self.projection_mut()
193                        .reject(event.seq, "invalid producer reminder ownership");
194                    return;
195                }
196                if !valid_hop_owner(producer_owner, *agent_hop) {
197                    self.projection_mut()
198                        .reject(event.seq, "invalid producer hop ownership");
199                    return;
200                }
201                let dedup_key = (producer_owner.clone(), idempotency_key.clone());
202                if let Some(index) = self.dedup_indexes.get(&dedup_key).copied() {
203                    let prior = &self.projection.messages[index];
204                    // `agent_hop` is guard metadata, not payload: a replayed
205                    // acceptance for the same key stays idempotent even when
206                    // the stamped depth differs from the stored record.
207                    let exact = prior.message_id == *message_id
208                        && prior.mode == *mode
209                        && prior.description == *description
210                        && prior.body == *body
211                        && prior.reminder == *reminder;
212                    if exact {
213                        return;
214                    }
215                    self.projection_mut()
216                        .reject(event.seq, "conflicting producer idempotency key");
217                    return;
218                }
219                if self.message_indexes.contains_key(message_id) {
220                    self.projection_mut()
221                        .reject(event.seq, "conflicting producer message id");
222                    return;
223                }
224                let index = self.projection.messages.len();
225                self.projection_mut().messages.push(ProducerMessageRecord {
226                    message_id: *message_id,
227                    producer_owner: producer_owner.clone(),
228                    mode: *mode,
229                    idempotency_key: idempotency_key.clone(),
230                    description: description.clone(),
231                    body: body.clone(),
232                    reminder: *reminder,
233                    agent_hop: *agent_hop,
234                    accepted_seq: event.seq,
235                    admission: None,
236                    claims: HashSet::new(),
237                    consumed: false,
238                    discarded: false,
239                    discarded_seq: None,
240                    consumption_recorded: false,
241                    consumed_run: None,
242                });
243                self.message_indexes.insert(*message_id, index);
244                self.dedup_indexes.insert(dedup_key, index);
245            }
246            EventPayload::ProducerMessageAdmitted { message_id } => {
247                let Some(run_id) = event.run_id else {
248                    self.projection_mut()
249                        .reject(event.seq, "producer admission is missing run id");
250                    return;
251                };
252                if self.terminal_runs.contains(&run_id) {
253                    self.projection_mut()
254                        .reject(event.seq, "producer admission targets a terminal run");
255                    return;
256                }
257                let Some(index) = self.message_indexes.get(message_id).copied() else {
258                    self.projection_mut()
259                        .reject(event.seq, "producer admission has no accepted message");
260                    return;
261                };
262                let message = &self.projection.messages[index];
263                let replaceable = message.admission.is_some_and(|(prior_run, _)| {
264                    prior_run != run_id && self.terminal_runs.contains(&prior_run)
265                });
266                if message.consumed
267                    || message.discarded_seq.is_some()
268                    || (message.admission.is_some() && !replaceable)
269                {
270                    self.projection_mut()
271                        .reject(event.seq, "invalid producer admission");
272                    return;
273                }
274                self.projection_mut().messages[index].admission = Some((run_id, event.seq));
275            }
276            EventPayload::ProducerMessagesClaimed { message_ids } => {
277                let Some(run_id) = event.run_id else {
278                    self.projection_mut()
279                        .reject(event.seq, "producer claim is missing run id");
280                    return;
281                };
282                let unique = message_ids.iter().copied().collect::<HashSet<_>>();
283                let indexes = message_ids
284                    .iter()
285                    .filter_map(|message_id| self.message_indexes.get(message_id).copied())
286                    .collect::<Vec<_>>();
287                let valid = !message_ids.is_empty()
288                    && unique.len() == message_ids.len()
289                    && indexes.len() == message_ids.len()
290                    && !self.projection.claims.contains_key(&event.seq)
291                    && indexes.iter().all(|index| {
292                        let message = &self.projection.messages[*index];
293                        message.accepted_seq < event.seq
294                            && !message.consumed
295                            && message.discarded_seq.is_none()
296                            && message
297                                .admission
298                                .is_some_and(|(admission_run, admission_seq)| {
299                                    admission_run == run_id && admission_seq < event.seq
300                                })
301                    });
302                if !valid {
303                    self.projection_mut()
304                        .reject(event.seq, "invalid producer claim");
305                    return;
306                }
307                for index in indexes {
308                    self.projection_mut().messages[index]
309                        .claims
310                        .insert(event.seq);
311                }
312                self.projection_mut().claims.insert(
313                    event.seq,
314                    ProducerClaimRecord {
315                        run_id,
316                        message_ids: message_ids.clone(),
317                    },
318                );
319            }
320            EventPayload::ProducerMessagesReleased { claim_seq } => {
321                let Some(run_id) = event.run_id else {
322                    self.projection_mut()
323                        .reject(event.seq, "producer release is missing run id");
324                    return;
325                };
326                let Some(claim) = self.projection.claims.get(claim_seq) else {
327                    self.projection_mut()
328                        .reject(event.seq, "producer release has no claim");
329                    return;
330                };
331                if *claim_seq == 0 || claim.run_id != run_id {
332                    self.projection_mut()
333                        .reject(event.seq, "producer release ownership is invalid");
334                    return;
335                }
336                let message_ids = claim.message_ids.clone();
337                for message_id in &message_ids {
338                    if let Some(index) = self.message_indexes.get(message_id).copied() {
339                        self.projection_mut().messages[index]
340                            .claims
341                            .remove(claim_seq);
342                    }
343                }
344                self.projection_mut().claims.remove(claim_seq);
345            }
346            EventPayload::ModelTurnCommitted {
347                input_through_seq, ..
348            } => {
349                let Some(run_id) = event.run_id else { return };
350                let covered = |message: &ProducerMessageRecord| {
351                    !message.consumed
352                        && message
353                            .admission
354                            .is_some_and(|(admission_run, admission_seq)| {
355                                admission_run == run_id && admission_seq <= *input_through_seq
356                            })
357                        && message
358                            .discarded_seq
359                            .is_none_or(|discarded_seq| discarded_seq > *input_through_seq)
360                };
361                // Most turns consume nothing; they leave a shared projection
362                // unshared.
363                if !self.projection.messages.iter().any(covered) {
364                    return;
365                }
366                for message in &mut self.projection_mut().messages {
367                    if covered(message) {
368                        message.discarded = false;
369                        message.discarded_seq = None;
370                        message.consumed = true;
371                        message.consumed_run = Some(run_id);
372                    }
373                }
374            }
375            EventPayload::ProducerMessageConsumed { message_id, run_id } => {
376                let Some(index) = self.message_indexes.get(message_id).copied() else {
377                    self.projection_mut()
378                        .reject(event.seq, "producer consumption has no accepted message");
379                    return;
380                };
381                let message = &self.projection.messages[index];
382                if event.run_id != Some(*run_id)
383                    || message.admission.map(|(run, _)| run) != Some(*run_id)
384                    || message.consumed_run != Some(*run_id)
385                    || message.consumption_recorded
386                {
387                    self.projection_mut()
388                        .reject(event.seq, "invalid producer consumption marker");
389                    return;
390                }
391                self.projection_mut().messages[index].consumption_recorded = true;
392            }
393            EventPayload::ProducerMessageDiscarded {
394                message_id,
395                reminder,
396                producer_owner,
397            } => {
398                let Some(index) = self.message_indexes.get(message_id).copied() else {
399                    self.projection_mut()
400                        .reject(event.seq, "producer discard has no accepted message");
401                    return;
402                };
403                let message = &self.projection.messages[index];
404                let identity_matches = producer_owner == &message.producer_owner
405                    && reminder
406                        .as_ref()
407                        .is_none_or(|identity| message.reminder.as_ref() == Some(identity));
408                if !identity_matches || message.consumed || !message.claims.is_empty() {
409                    self.projection_mut()
410                        .reject(event.seq, "invalid producer message discard");
411                    return;
412                }
413                if message.discarded_seq.is_none() {
414                    self.projection_mut().messages[index].discarded = true;
415                    self.projection_mut().messages[index].discarded_seq = Some(event.seq);
416                }
417            }
418            EventPayload::RunFinished { .. } => {
419                if let Some(run_id) = event.run_id {
420                    self.terminal_runs.insert(run_id);
421                }
422            }
423            _ => {}
424        }
425    }

428/// The goal part of a session's log: its current goal, after the
429/// activations, checklist revisions, and lifecycle changes that were valid
430/// where they stand. Shared by the producer projection and the session
431/// projection, which reports an active goal in `SessionMeta::waiting_on`.
432#[derive(Clone, Debug, Default, Eq, PartialEq)]
433pub(crate) struct GoalFold {
434    pub(crate) goal: Option<GoalState>,
435    /// The latest revision of every goal the log activated.
436    revisions: HashMap<GoalId, u64>,
437}
438
439impl GoalFold {
440    /// Applies `event` when it is a goal event, or says why it is invalid
441    /// and leaves the fold unchanged. Any other event is ignored.
442    pub(crate) fn observe(&mut self, event: &StoredEvent) -> Result<(), &'static str> {
443        match &event.payload {
444            EventPayload::GoalActivated {
445                goal_id,
446                objective,
447                revision,
448                ..
449            } => {
450                let replaceable = self
451                    .goal
452                    .as_ref()
453                    .is_none_or(|goal| is_terminal(goal.status));
454                let distinct = self
455                    .goal
456                    .as_ref()
457                    .is_none_or(|goal| goal.goal_id != *goal_id);
458                let unseen = !self.revisions.contains_key(goal_id);
459                if objective.trim().is_empty() || !replaceable || !distinct || !unseen {
460                    return Err("invalid goal activation");
461                }
462                self.goal = Some(GoalState {
463                    goal_id: *goal_id,
464                    objective: objective.clone(),
465                    status: GoalStatus::Active,
466                    items: Vec::new(),
467                    revision: *revision,
468                });
469                self.revisions.insert(*goal_id, *revision);
470            }
471            EventPayload::GoalChecklistRevised {
472                goal_id,
473                items,
474                revision,
475            } => {
476                let valid_items = items.iter().all(|item| !item.description.trim().is_empty());
477                let valid = valid_items
478                    && self.goal.as_ref().is_some_and(|goal| {
479                        goal.goal_id == *goal_id
480                            && !is_terminal(goal.status)
481                            && *revision > goal.revision
482                    });
483                if !valid {
484                    return Err("invalid goal checklist revision");
485                }
486                let goal = self.goal.as_mut().expect("validated current goal");
487                goal.items = items.clone();
488                goal.revision = *revision;
489                self.revisions.insert(*goal_id, *revision);
490            }
491            EventPayload::GoalLifecycleChanged {
492                goal_id,
493                status,
494                revision,
495                selection,
496            } => {
497                let valid = self.goal.as_ref().is_some_and(|goal| {
498                    goal.goal_id == *goal_id
499                        && *revision > goal.revision
500                        && valid_lifecycle_change(goal, *status)
501                        && (selection.is_none() || *status == GoalStatus::Active)
502                });
503                if !valid {
504                    return Err("invalid goal lifecycle transition");
505                }
506                let goal = self.goal.as_mut().expect("validated current goal");
507                goal.status = *status;
508                goal.revision = *revision;
509                self.revisions.insert(*goal_id, *revision);
510            }
511            _ => {}
512        }
513        Ok(())
514    }
515
516    /// Whether the current goal is active, so goal mode keeps scheduling
517    /// runs. A paused, completed, or cancelled goal schedules none.
518    #[must_use]
519    pub(crate) fn is_active(&self) -> bool {
520        self.goal
521            .as_ref()
522            .is_some_and(|goal| goal.status == GoalStatus::Active)
523    }
524}
525
526fn is_terminal(status: GoalStatus) -> bool {
527    matches!(status, GoalStatus::Completed | GoalStatus::Cancelled)
528}
529
530fn valid_lifecycle_change(goal: &GoalState, status: GoalStatus) -> bool {
531    match (goal.status, status) {
532        (GoalStatus::Active, GoalStatus::Paused | GoalStatus::Cancelled)
533        | (GoalStatus::Paused, GoalStatus::Active | GoalStatus::Cancelled) => true,
534        (GoalStatus::Active | GoalStatus::Paused, GoalStatus::Completed) => {
535            !goal.items.is_empty() && goal.items.iter().all(|item| item.finished)
536        }
537        _ => false,
538    }
539}

541fn valid_reminder_owner(owner: &ProducerOwner, reminder: Option<&GoalReminderIdentity>) -> bool {
542    match (owner, reminder) {
543        (ProducerOwner::Goal { goal_id }, Some(reminder)) => *goal_id == reminder.goal_id,
544        (ProducerOwner::Goal { .. }, None) | (_, Some(_)) => false,
545        (_, None) => true,
546    }
547}

549/// Hop metadata belongs exclusively to agent mail. A malformed durable event
550/// that stamps it onto plugin, goal, or delegation mail is quarantined so it
551/// cannot inflate an agent chain it never joined.
552const fn valid_hop_owner(owner: &ProducerOwner, agent_hop: Option<u32>) -> bool {
553    agent_hop.is_some() == matches!(owner, ProducerOwner::Agent { .. })
554}
