45pub(crate) fn checkpoint_retained_history(
46    history: &[HistoryTurn],
47    events: &[StoredEvent],
48    summary: Option<&str>,
49) -> Vec<HistoryTurn> {
50    let pinned_skills = events
51        .iter()
52        .filter(|event| matches!(event.payload, EventPayload::SkillLoaded { .. }))
53        .count();
54    let pinned_agent_md = usize::from(latest_agent_md_event(events).is_some());
55    let pinned_count = pinned_agent_md.saturating_add(pinned_skills);
56    let mut history = history.iter();
57    let mut retained = history.next().cloned().into_iter().collect::<Vec<_>>();
58    retained.extend(
59        history
60            .filter(|turn| !is_framed_summary_turn(turn))
61            .take(pinned_count)
62            .cloned(),
63    );
64    if let Some(summary) = summary {
65        retained.push(HistoryTurn::user(user_text(&framed_compaction_summary(
66            summary,
67        ))));
68    }
69    retained
70}

72fn is_framed_summary_turn(turn: &HistoryTurn) -> bool {
73    matches!(
74        turn,
75        HistoryTurn::User(message)

77                if text.text.starts_with(COMPACTION_SUMMARY_PREFIX)
78                    && text.text.ends_with(COMPACTION_SUMMARY_SUFFIX))

98fn retained_artifact_id(reference: &ArtifactReference) -> Result<&str, HistoryError> {
99    let digest = reference
100        .uri
101        .strip_prefix("artifact://sha256/")
102        .ok_or_else(|| HistoryError::Corrupt("invalid retained artifact reference".into()))?;

104        .map_err(|_| HistoryError::Corrupt("invalid retained artifact digest".into()))?;

129pub(crate) fn persist_turn(
130    mut turn: CompletedTurn,
131    store: &ArtifactRouter,
132    session: SessionId,
133    binding: &FrozenModelBinding,
134) -> Result<(PersistedModelTurn, Vec<SafeErrorMessage>), HistoryError> {
135    turn.finish
136        .provider_metadata
137        .remove("cookie_agent.replay_source_wire_model_id");
138    if let Some(source) = turn
139        .finish
140        .native_replay
141        .as_ref()
142        .and_then(OvenReplayArtifact::source_wire_model_id)
143    {
144        turn.finish.provider_metadata.insert(
145            "cookie_agent.replay_source_wire_model_id".into(),
146            serde_json::Value::String(source.as_str().into()),
147        );
148    }
149    let mut warnings = turn
150        .warnings
151        .iter()
152        .map(|warning| {
153            SafeErrorMessage::new(sanitize_control_free(warning, SafeErrorMessage::MAX_BYTES))
154                .map_err(|error| HistoryError::Corrupt(error.to_string()))
155        })
156        .collect::<Result<Vec<_>, _>>()?;
157    let content = turn
158        .message
159        .content
160        .into_iter()
161        .map(|part| persist_assistant_part(part, store, session, &mut warnings))
162        .collect::<Result<_, _>>()?;
163    Ok((
164        PersistedModelTurn {
165            content,
166            provider_options: turn.message.provider_options,
167            finish_reason: persist_finish_reason(turn.finish.finish_reason),
168            usage: persist_usage(turn.finish.usage),
169            response_metadata: turn.finish.response_metadata,
170            provider_metadata: turn.finish.provider_metadata,
171            native_replay: turn
172                .finish
173                .native_replay
174                .map(|artifact| persist_replay(artifact, binding))
175                .transpose()?,
176        },
177        coalesce_warnings(warnings),
178    ))
179}

181pub(crate) fn replay_decisions(
182    decisions: &[OvenReplayDecision],
183    binding: &FrozenModelBinding,
184) -> Vec<ReplayDecision> {
185    decisions
186        .iter()
187        .map(|decision| ReplayDecision {
188            history_index: decision.history_index as u64,
189            disposition: match &decision.disposition {
190                OvenReplayDisposition::Replayed => ReplayDisposition::Replayed,
191                OvenReplayDisposition::NoArtifact => ReplayDisposition::NoArtifact,
192                OvenReplayDisposition::DiscardedForeignAdapter { found, expected } => {
193                    ReplayDisposition::DiscardedForeignAdapter {
194                        found: exact_adapter_code(found.as_str()),
195                        expected: exact_adapter_code(expected.as_str()),
196                    }
197                }
198                OvenReplayDisposition::DiscardedForeignScope { found, expected } => {
199                    let found_selection = scope_selection(found);
200                    let expected_selection = scope_selection(expected);
201                    if found_selection.model == expected_selection.model {
202                        ReplayDisposition::DiscardedForeignVariant {
203                            found: found_selection.variant,
204                            expected: binding.selection.variant.clone(),
205                        }
206                    } else {
207                        ReplayDisposition::DiscardedForeignModelSelection {
208                            found: found_selection,
209                            expected: binding.selection.clone(),
210                        }
211                    }
212                }
213                OvenReplayDisposition::DiscardedInvalidPayload { reason } => {
214                    ReplayDisposition::DiscardedInvalidPayload {
215                        reason: SafeErrorMessage::new(sanitize_control_free(
216                            reason,
217                            SafeErrorMessage::MAX_BYTES,
218                        ))
219                        .expect("sanitized safe replay error"),
220                    }
221                }
222                OvenReplayDisposition::ReconstructedNormalized => {
223                    ReplayDisposition::ReconstructedNormalizedHistory
224                }
225            },
226        })
227        .collect()
228}

230pub(crate) fn replay_decisions_with_preflight(
231    decisions: &[OvenReplayDecision],
232    binding: &FrozenModelBinding,
233    preflight: &[ReplayDecision],
234) -> Vec<ReplayDecision> {
235    let preflight = preflight
236        .iter()
237        .map(|decision| (decision.history_index, decision.disposition.clone()))
238        .collect::<HashMap<_, _>>();
239    let mut emitted = HashSet::new();
240    let mut merged = Vec::new();
241    for decision in replay_decisions(decisions, binding) {
242        let Some(disposition) = preflight.get(&decision.history_index) else {
243            merged.push(decision);
244            continue;
245        };
246        if emitted.insert(decision.history_index) {
247            merged.push(ReplayDecision {
248                history_index: decision.history_index,
249                disposition: disposition.clone(),
250            });
251        }
252        if !matches!(decision.disposition, ReplayDisposition::NoArtifact) {
253            merged.push(decision);
254        }
255    }
256    for (history_index, disposition) in preflight {
257        if emitted.insert(history_index) {
258            merged.push(ReplayDecision {
259                history_index,
260                disposition,
261            });
262            merged.push(ReplayDecision {
263                history_index,
264                disposition: ReplayDisposition::ReconstructedNormalizedHistory,
265            });
266        } else if !merged.iter().any(|decision| {
267            decision.history_index == history_index
268                && matches!(
269                    decision.disposition,
270                    ReplayDisposition::ReconstructedNormalizedHistory
271                )
272        }) {
273            merged.push(ReplayDecision {
274                history_index,
275                disposition: ReplayDisposition::ReconstructedNormalizedHistory,
276            });
277        }
278    }
279    merged.sort_by_key(|decision| decision.history_index);
280    merged
281}

345fn selected_checkpoint_events(
346    events: &[StoredEvent],
347    source_through_seq: u64,
348    recent_from_seq: Option<u64>,
349    close_dependencies: bool,
350    mode: CheckpointSelectionMode,
351) -> Vec<StoredEvent> {
352    let latest_agent_md_seq = latest_agent_md_event(events).map(|event| event.seq);
353    let pending_producer_admissions = match mode {
354        CheckpointSelectionMode::InternalSummary => GoalProducerProjection::from_events(events)
355            .messages
356            .into_iter()
357            .filter(|message| !message.consumed && !message.discarded)
358            .filter_map(|message| message.admission.map(|(_, seq)| seq))
359            .collect::<HashSet<_>>(),
360        CheckpointSelectionMode::NativeWindow => HashSet::new(),
361    };
362    let mut selected = events
363        .iter()
364        .filter(|event| {
365            is_pinned_event(event, latest_agent_md_seq, source_through_seq)
366                || pending_producer_admissions.contains(&event.seq)
367                || recent_from_seq
368                    .is_some_and(|from| (from..=source_through_seq).contains(&event.seq))
369                || event.seq > source_through_seq
370        })
371        .cloned()
372        .collect::<Vec<_>>();
373    if close_dependencies {
374        close_event_dependencies(events, &mut selected);
375    }
376    selected.sort_by_key(|event| event.seq);
377    selected.dedup_by_key(|event| event.seq);
378    selected
379}

381fn close_event_dependencies(events: &[StoredEvent], selected: &mut Vec<StoredEvent>) {
382    // Applied inputs and late tool completions can refer to events before the retained range.
383    loop {
384        let selected_seqs = selected
385            .iter()
386            .map(|event| event.seq)
387            .collect::<HashSet<_>>();
388        let mut required_seqs = HashSet::new();
389        for event in selected.iter() {
390            match &event.payload {
391                EventPayload::UserInputApplied { user_input_seq } => {
392                    required_seqs.insert(*user_input_seq);
393                }
394                EventPayload::ToolCallStarted { start } => {
395                    if let Some(commit) = events.iter().rev().find(|candidate| {
396                        candidate.seq <= event.seq
397                            && candidate.run_id == event.run_id
398                            && matches!(&candidate.payload,
399                                EventPayload::ModelTurnCommitted { model_turn_seq, turn, .. }
400                                    if *model_turn_seq == start.owner.model_turn_seq
401                                        && turn.content.iter().any(|part| matches!(part,
402                                            PersistedAssistantPart::ToolCall { id, .. }
403                                                if id == &start.owner.model_call_id)))
404                    }) {
405                        required_seqs.insert(commit.seq);
406                    }
407                }
408                EventPayload::ToolCallTerminated { termination } => {
409                    if let Some(start) = events.iter().rev().find(|candidate| {
410                        candidate.seq <= event.seq
411                            && candidate.run_id == event.run_id
412                            && matches!(&candidate.payload,
413                                EventPayload::ToolCallStarted { start }
414                                    if start.tool_call_id == termination.tool_call_id)
415                    }) {
416                        required_seqs.insert(start.seq);
417                    }
418                }
419                _ => {}
420            }
421        }
422        let additions = events
423            .iter()
424            .filter(|event| {
425                required_seqs.contains(&event.seq) && !selected_seqs.contains(&event.seq)
426            })
427            .cloned()
428            .collect::<Vec<_>>();
429        if additions.is_empty() {
430            break;
431        }
432        selected.extend(additions);
433    }
434}

444fn insert_summary(assembled: &mut AssembledHistory, events: &[StoredEvent], summary: &str) {
445    let history_index = 1 + pinned_history_len(events);
446    assembled.history.insert(
447        history_index,
448        HistoryTurn::user(user_text(&framed_compaction_summary(summary))),
449    );
450    for decision in &mut assembled.replay_decisions {
451        if decision.history_index >= history_index as u64 {
452            decision.history_index += 1;
453        }
454    }

483pub(crate) fn compaction_prefix_history(
484    events: &[StoredEvent],
485    store: &ArtifactRouter,
486    binding: &FrozenModelBinding,
487    composed_prompt: &str,
488    recent_from_seq: Option<u64>,
489) -> Result<Vec<HistoryTurn>, HistoryError> {
490    let Some(recent_from_seq) = recent_from_seq else {
491        return Ok(assemble_model_context(events, store, binding, composed_prompt)?.history);
492    };
493    let (mut visible, prior_summary) = if let Some(commit) = latest_checkpoint(events) {
494        let (retained_from, summary, mode) = match &commit.checkpoint {
495            ContextCheckpoint::InternalSummary { checkpoint } => (
496                commit.boundaries.recent_from_seq,
497                Some(checkpoint.summary()),
498                CheckpointSelectionMode::InternalSummary,
499            ),
500            ContextCheckpoint::NativeWindow { .. } => {
501                (None, None, CheckpointSelectionMode::NativeWindow)
502            }
503        };
504        (
505            selected_checkpoint_events(
506                events,
507                commit.boundaries.source_through_seq,
508                retained_from,
509                true,
510                mode,
511            ),
512            summary,
513        )
514    } else {
515        (events.to_vec(), None)
516    };
517    let latest_agent_md_seq = latest_agent_md_event(events).map(|event| event.seq);
518    visible.retain(|event| {
519        event.seq < recent_from_seq || is_pinned_event(event, latest_agent_md_seq, u64::MAX)
520    });
521    close_event_dependencies(events, &mut visible);
522    visible.sort_by_key(|event| event.seq);
523    visible.dedup_by_key(|event| event.seq);
524    let mut assembled =
525        assemble_history_with_replay(&visible, events, store, binding, composed_prompt)?;
526    if let Some(summary) = prior_summary {
527        insert_summary(&mut assembled, events, summary);
528    }
529    Ok(assembled.history)
530}

532pub(crate) fn compaction_tail_candidates(events: &[StoredEvent]) -> Vec<u64> {
533    #[derive(Default)]
534    struct Group {
535        start: u64,
536        end: u64,
537        unresolved_calls: usize,
538    }
539
540    let visible = if let Some(commit) = latest_checkpoint(events) {
541        let (recent_from, mode) = match &commit.checkpoint {
542            ContextCheckpoint::InternalSummary { .. } => (
543                commit.boundaries.recent_from_seq,
544                CheckpointSelectionMode::InternalSummary,
545            ),
546            ContextCheckpoint::NativeWindow { .. } => (None, CheckpointSelectionMode::NativeWindow),
547        };
548        selected_checkpoint_events(
549            events,
550            commit.boundaries.source_through_seq,
551            recent_from,
552            true,
553            mode,
554        )
555    } else {
556        events.to_vec()
557    };
558    let visible_seqs = visible
559        .iter()
560        .map(|event| event.seq)
561        .collect::<HashSet<_>>();
562    let producer_admissions = GoalProducerProjection::from_events(events)
563        .messages
564        .into_iter()
565        .filter(|message| !message.discarded)
566        .filter_map(|message| message.admission.map(|(_, seq)| seq))
567        .collect::<HashSet<_>>();
568    let submitted = events
569        .iter()
570        .filter_map(|event| match event.payload {
571            EventPayload::UserInputSubmitted { .. } => Some(event.seq),
572            _ => None,
573        })
574        .collect::<HashSet<_>>();
575    let mut groups = Vec::<Group>::new();
576    let mut pending = HashMap::<
577        (
578            cookie_agent_protocol::RunId,
579            cookie_agent_protocol::ModelCallId,
580        ),
581        VecDeque<usize>,
582    >::new();
583    let mut started = HashMap::<(cookie_agent_protocol::RunId, ToolCallId), usize>::new();
584    let mut terminated = HashSet::new();
585
586    for event in &visible {
587        match &event.payload {
588            EventPayload::UserInputApplied { user_input_seq }
589                if submitted.contains(user_input_seq) =>
590            {
591                groups.push(Group {
592                    start: *user_input_seq,
593                    end: event.seq,
594                    ..Group::default()
595                });
596            }
597            EventPayload::MessageInjected { role, .. }
598                if *role != cookie_agent_protocol::ExtensionMessageRole::Tool =>
599            {
600                groups.push(Group {
601                    start: event.seq,
602                    end: event.seq,
603                    ..Group::default()
604                });
605            }
606            EventPayload::ProducerMessageAdmitted { .. }
607                if producer_admissions.contains(&event.seq) =>
608            {
609                groups.push(Group {
610                    start: event.seq,
611                    end: event.seq,
612                    ..Group::default()
613                });
614            }
615            EventPayload::DelegatedContextSeeded { turns, .. } if !turns.is_empty() => {
616                groups.push(Group {
617                    start: event.seq,
618                    end: event.seq,
619                    ..Group::default()
620                });
621            }
622            EventPayload::ModelTurnCommitted { turn, .. } => {
623                let in_stream_results = turn
624                    .content
625                    .iter()
626                    .filter_map(|part| match part {
627                        PersistedAssistantPart::ToolResult { tool_call_id, .. } => {
628                            Some(tool_call_id.as_str())
629                        }
630                        _ => None,
631                    })
632                    .collect::<HashSet<_>>();
633                let index = groups.len();
634                let mut unresolved_calls = 0;
635                if let Some(run_id) = event.run_id {
636                    for id in turn.content.iter().filter_map(|part| match part {
637                        PersistedAssistantPart::ToolCall { id, .. }
638                            if !in_stream_results.contains(id.as_str()) =>
639                        {
640                            Some(id)
641                        }
642                        _ => None,
643                    }) {
644                        unresolved_calls += 1;
645                        pending
646                            .entry((run_id, id.clone()))
647                            .or_default()
648                            .push_back(index);
649                    }
650                }
651                groups.push(Group {
652                    start: event.seq,
653                    end: event.seq,
654                    unresolved_calls,
655                });
656            }
657            EventPayload::ToolCallStarted { start } => {
658                let Some(run_id) = event.run_id else { continue };
659                if let Some(index) = pending
660                    .get_mut(&(run_id, start.owner.model_call_id.clone()))
661                    .and_then(VecDeque::pop_front)
662                {
663                    started.insert((run_id, start.tool_call_id), index);
664                }
665            }
666            EventPayload::ToolCallTerminated { termination } => {
667                let Some(run_id) = event.run_id else { continue };
668                if terminated.insert((run_id, termination.tool_call_id))
669                    && let Some(index) = started.get(&(run_id, termination.tool_call_id))
670                {
671                    groups[*index].unresolved_calls =
672                        groups[*index].unresolved_calls.saturating_sub(1);
673                    groups[*index].end = groups[*index].end.max(event.seq);
674                }
675            }
676            EventPayload::ContextRehydrated { .. }
677            | EventPayload::PluginEventAdded { .. }
678            | EventPayload::DelegateFinished { .. }
679            | EventPayload::DelegateFinishedV2 { .. } => groups.push(Group {
680                start: event.seq,
681                end: event.seq,
682                ..Group::default()
683            }),
684            _ => {}
685        }
686    }
687
688    groups.sort_by_key(|group| group.start);
689    groups.dedup_by_key(|group| group.start);
690    let unresolved_from = groups
691        .iter()
692        .filter(|group| group.unresolved_calls != 0)
693        .map(|group| group.start)
694        .min();
695    let spans = groups
696        .iter()
697        .filter(|group| group.end > group.start)
698        .map(|group| (group.start, group.end))
699        .collect::<Vec<_>>();
700    groups
701        .into_iter()
702        .skip(1)
703        .filter(|group| unresolved_from.is_none_or(|start| group.start <= start))
704        .filter(|group| {
705            !spans
706                .iter()
707                .any(|(start, end)| *start < group.start && group.start <= *end)
708        })
709        .filter(|group| visible_seqs.contains(&group.start))
710        .map(|group| group.start)
711        .collect()
712}

719pub(crate) fn assemble_model_context(
720    events: &[StoredEvent],
721    store: &ArtifactRouter,
722    binding: &FrozenModelBinding,
723    composed_prompt: &str,
724) -> Result<ModelContext, HistoryError> {
725    let checkpoint = latest_checkpoint(events);
726    let Some(commit) = checkpoint else {
727        let assembled =
728            assemble_history_with_replay(events, events, store, binding, composed_prompt)?;
729        return Ok(ModelContext {
730            history: assembled.history,
731            native_context: None,
732            replay_decisions: assembled.replay_decisions,
733        });
734    };
735    match &commit.checkpoint {
736        ContextCheckpoint::InternalSummary { checkpoint } => project_summary_context(
737            events,
738            store,
739            binding,
740            composed_prompt,
741            commit.boundaries.source_through_seq,
742            commit.boundaries.recent_from_seq,
743            checkpoint.summary(),
744        ),
745        ContextCheckpoint::NativeWindow { window } => {
746            commit
747                .validate_for_binding(binding)
748                .map_err(|error| HistoryError::Corrupt(error.to_string()))?;
749            let selected = selected_checkpoint_events(
750                events,
751                commit.boundaries.source_through_seq,
752                None,
753                false,
754                CheckpointSelectionMode::NativeWindow,
755            );
756            let assembled =
757                assemble_history_with_replay(&selected, events, store, binding, composed_prompt)?;
758            Ok(ModelContext {
759                history: assembled.history,
760                native_context: Some(restore_native_context(window, binding)?),
761                replay_decisions: assembled.replay_decisions,
762            })
763        }
764    }
765}

776// Message selection must not roll back session-wide replay decisions or pinned context.
777// context_events is the full visible snapshot, before checkpoint or summary-prefix filtering.
778fn assemble_history_with_replay(
779    events: &[StoredEvent],
780    context_events: &[StoredEvent],
781    store: &ArtifactRouter,
782    binding: &FrozenModelBinding,
783    composed_prompt: &str,
784) -> Result<AssembledHistory, HistoryError> {
785    let producer_projection = GoalProducerProjection::from_events(context_events);
786    let current_run = context_events.iter().rev().find_map(|event| {
787        matches!(event.payload, EventPayload::RunStarted { .. })
788            .then_some(event.run_id)
789            .flatten()
790    });
791    let producer_admissions = producer_projection
792        .messages
793        .iter()
794        .filter(|message| !message.discarded)
795        .filter_map(|message| {
796            message
797                .admission
798                .and_then(|(admission_run, admission_seq)| {
799                    (message.consumed || current_run.is_none_or(|run| admission_run == run))
800                        .then_some((admission_seq, message.body.as_str()))
801                })
802        })
803        .collect::<HashMap<_, _>>();
804    let producer_delegations = producer_projection
805        .messages
806        .iter()
807        .filter_map(|message| match &message.producer_owner {
808            cookie_agent_protocol::ProducerOwner::Delegation { invocation_id } => {
809                Some(*invocation_id)
810            }
811            _ => None,
812        })
813        .collect::<HashSet<_>>();
814    let loaded_skills = context_events
815        .iter()
816        .filter_map(|event| match &event.payload {
817            EventPayload::SkillLoaded { rendered_body, .. } => Some(rendered_body.clone()),
818            _ => None,
819        })
820        .collect::<Vec<_>>();
821    let loaded_agent_md = latest_agent_md_event(context_events).and_then(|event| {
822        let EventPayload::AgentMdLoaded { entries } = &event.payload else {
823            return None;
824        };
825        Some(agent_md_turn(entries))
826    });
827    let mut logical = Vec::<LogicalTurn>::new();
828    let mut submitted = HashMap::<u64, String>::new();
829    let mut pending_model_calls = HashMap::<
830        (
831            cookie_agent_protocol::RunId,
832            cookie_agent_protocol::ModelCallId,
833        ),
834        VecDeque<(usize, usize)>,
835    >::new();
836    let mut engine_calls =
837        HashMap::<(cookie_agent_protocol::RunId, ToolCallId), (usize, usize)>::new();
838    let delegation_children = context_events
839        .iter()
840        .filter_map(|event| match event.payload {
841            EventPayload::ToolCallLinked {
842                tool_call_id,
843                child_session_id,
844            } => Some(((event.run_id, tool_call_id), child_session_id)),
845            _ => None,
846        })
847        .collect::<HashMap<_, _>>();
848    let elisions = context_events
849        .iter()
850        .filter_map(|event| match &event.payload {
851            EventPayload::ToolOutputElided {
852                tool_call_id,
853                original_bytes,
854                retained,
855            } => Some((*tool_call_id, (*original_bytes, retained.clone()))),
856            _ => None,
857        })
858        .collect::<HashMap<_, _>>();
859
860    for envelope in events {
861        match &envelope.payload {
862            EventPayload::DelegatedContextSeeded { turns, .. } => {
863                logical.extend(turns.iter().map(|turn| match turn.role {
864                    DelegatedContextRole::User => LogicalTurn::User(user_text(&turn.text)),
865                    DelegatedContextRole::Assistant => {
866                        LogicalTurn::SeedAssistant(turn.text.clone())
867                    }
868                }));
869            }
870            EventPayload::MessageInjected { role, input } => match role {
871                cookie_agent_protocol::ExtensionMessageRole::System => {
872                    logical.push(LogicalTurn::System(input.clone()));
873                }
874                cookie_agent_protocol::ExtensionMessageRole::User => {
875                    logical.push(LogicalTurn::User(user_text(input)));
876                }
877                cookie_agent_protocol::ExtensionMessageRole::Assistant => {
878                    logical.push(LogicalTurn::SeedAssistant(input.clone()));
879                }
880                cookie_agent_protocol::ExtensionMessageRole::Tool => {}
881            },
882            EventPayload::UserInputSubmitted { input, .. } => {
883                submitted.insert(envelope.seq, input.clone());
884            }
885            EventPayload::UserInputApplied { user_input_seq } => {
886                if let Some(input) = submitted.remove(user_input_seq) {
887                    logical.push(LogicalTurn::User(user_text(&input)));
888                }
889            }
890            EventPayload::ProducerMessageAdmitted { .. } => {
891                if let Some(body) = producer_admissions.get(&envelope.seq) {
892                    logical.push(LogicalTurn::User(user_text(body)));
893                }
894            }
895            EventPayload::ModelTurnCommitted {
896                resolved_model,
897                turn,
898                ..
899            } => {
900                let logical_index = logical.len();
901                let mut calls = Vec::new();
902                let in_stream_results = turn
903                    .content
904                    .iter()
905                    .filter_map(|part| match part {
906                        PersistedAssistantPart::ToolResult { tool_call_id, .. } => {
907                            Some(tool_call_id.as_str())
908                        }
909                        _ => None,
910                    })
911                    .collect::<std::collections::HashSet<_>>();
912                for part in &turn.content {
913                    if let PersistedAssistantPart::ToolCall { id, .. } = part {
914                        let call_index = calls.len();
915                        let in_stream_result = in_stream_results.contains(id.as_str());
916                        calls.push(CallRecord {
917                            model_call_id: id.clone(),
918                            engine_call_id: None,
919                            result: None,
920                            additional_messages: Vec::new(),
921                            in_stream_result,
922                        });
923                        if !in_stream_result && let Some(run_id) = envelope.run_id {
924                            pending_model_calls
925                                .entry((run_id, id.clone()))
926                                .or_default()
927                                .push_back((logical_index, call_index));
928                        }
929                    }
930                }
931                logical.push(LogicalTurn::Assistant(Box::new(AssistantRecord {
932                    turn: turn.clone(),
933                    resolved_model: resolved_model.clone(),
934                    run_id: envelope.run_id,
935                    calls,
936                    // A turn that an interrupt cut off is annotated for the next
937                    // request only: the durable event stays a faithful record of
938                    // what the model produced, while the marker tells the model
939                    // that the fragment it sees was not its own ending. Only the
940                    // engine's salvage commits `Aborted`; a provider-reported
941                    // `Cancelled` finish is the provider's ending, not an
942                    // interrupt.
943                    interrupted: turn.finish_reason == ModelFinishReason::Aborted,
944                })));
945            }
946            EventPayload::ToolCallStarted { start } => {
947                let Some(run_id) = envelope.run_id else {
948                    continue;
949                };
950                let occurrence = pending_model_calls
951                    .get_mut(&(run_id, start.owner.model_call_id.clone()))
952                    .and_then(VecDeque::pop_front);
953                if let Some((logical_index, call_index)) = occurrence {
954                    let LogicalTurn::Assistant(assistant) = &mut logical[logical_index] else {
955                        return Err(HistoryError::Corrupt(
956                            "tool call mapped to a non-assistant turn".into(),
957                        ));
958                    };
959                    assistant.calls[call_index].engine_call_id = Some(start.tool_call_id);
960                    engine_calls.insert((run_id, start.tool_call_id), (logical_index, call_index));
961                }
962            }
963            EventPayload::ToolCallTerminated { termination }
964                if termination.outcome == ToolTerminationOutcome::Completed
965                    || termination
966                        .result
967                        .as_ref()
968                        .and_then(|result| result.retained_output.as_ref())
969                        .is_some_and(|output| output.incomplete)
970                    // Opt-out pages have no retained_output. Only the commit boundary's
971                    // finalized-result reason admits them after cancellation.
972                    || (termination.outcome == ToolTerminationOutcome::Cancelled
973                        && termination.result.is_some()
974                        && termination.error.as_ref().is_some_and(|error| {
975                            error.code.as_str() == crate::runtime::CANCELLED_AFTER_COMPLETION
976                        }))
977                    || (termination.outcome == ToolTerminationOutcome::Cancelled
978                        && termination
979                            .result
980                            .as_ref()
981                            .zip(
982                                delegation_children
983                                    .get(&(envelope.run_id, termination.tool_call_id)),
984                            )
985                            .is_some_and(|(result, child)| {
986                                crate::delegation_api::delegate_result_matches_child(result, *child)
987                            })) =>
988            {
989                if let Some(result) = &termination.result {
990                    let (mut result_part, additional_messages) =
991                        if let Some((original_bytes, retained)) =
992                            elisions.get(&termination.tool_call_id)
993                        {
994                            (
995                                ToolResultPart::new(
996                                    String::new(),
997                                    ToolContent::Text(tool_output_elision_marker(
998                                        retained_artifact_id(retained)?,
999                                        *original_bytes,
1000                                        result.additional_messages.len(),
1001                                    )),
1002                                ),
1003                                Vec::new(),
1004                            )
1005                        } else {
1006                            (
1007                                tool_result_part(result, store)?,
1008                                result.additional_messages.clone(),
1009                            )
1010                        };
1011                    result_part.is_error = termination.outcome != ToolTerminationOutcome::Completed;
1012                    if let Some(error) = &termination.error {
1013                        match &mut result_part.content {
1014                            ToolContent::Mixed(values) => {
1015                                values.push(ContentValue::Text(error.message.to_string()))
1016                            }
1017                            ToolContent::Text(text) => {
1018                                text.push('\n');
1019                                text.push_str(error.message.as_str());
1020                            }
1021                            _ => {}
1022                        }
1023                    }
1024                    attach_result(
1025                        &mut logical,
1026                        &engine_calls,
1027                        envelope.run_id,
1028                        termination.tool_call_id,
1029                        result_part,
1030                        additional_messages,
1031                    )?;
1032                }
1033            }
1034            EventPayload::ToolCallTerminated { termination } => {
1035                let message = termination
1036                    .error
1037                    .as_ref()
1038                    .map_or("tool failed", |error| error.message.as_str());
1039                let (content, metadata) = if let Some(denied) = denied_failure(message) {
1040                    let visible_reason = denied.feedback.as_ref().map_or_else(
1041                        || denied.reason.clone(),
1042                        |feedback| format!("{}: {feedback}", denied.reason),
1043                    );
1044                    let mut metadata = BTreeMap::from([
1045                        (
1046                            "denial_source".into(),
1047                            serde_json::Value::String(match denied.source {
1048                                ApprovalDecisionSource::Policy => "policy".into(),
1049                                ApprovalDecisionSource::Model => "model".into(),
1050                                ApprovalDecisionSource::InternalAgent => "internal_agent".into(),
1051                                ApprovalDecisionSource::PermissionMode => "permission_mode".into(),
1052                                ApprovalDecisionSource::TreeGrant => "tree_grant".into(),
1053                                ApprovalDecisionSource::User => "user".into(),
1054                                ApprovalDecisionSource::DoomLoopGuard => "doom_loop_guard".into(),
1055                                ApprovalDecisionSource::System => "system".into(),
1056                            }),
1057                        ),
1058                        (
1059                            "denial_reason".into(),
1060                            serde_json::Value::String(denied.reason),
1061                        ),
1062                    ]);
1063                    if let Some(feedback) = denied.feedback {
1064                        metadata.insert("feedback".into(), serde_json::Value::String(feedback));
1065                    }
1066                    (
1067                        ToolContent::Denied {
1068                            reason: Some(visible_reason),
1069                        },
1070                        Some(metadata),
1071                    )
1072                } else {
1073                    (ToolContent::Text(message.to_owned()), None)
1074                };
1075                attach_result(
1076                    &mut logical,
1077                    &engine_calls,
1078                    envelope.run_id,
1079                    termination.tool_call_id,
1080                    ToolResultPart {
1081                        tool_call_id: String::new(),
1082                        content,
1083                        is_error: true,
1084                        metadata,
1085                    },
1086                    Vec::new(),
1087                )?;
1088            }
1089            EventPayload::ContextRehydrated { files } => {
1090                logical.push(LogicalTurn::Rehydration(files.clone()));
1091            }
1092            EventPayload::PluginEventAdded {
1093                plugin,
1094                name,
1095                payload,
1096            } => {
1097                logical.push(LogicalTurn::User(user_text(&format!(
1098                    "<plugin_event>{}</plugin_event>",
1099                    serde_json::json!({
1100                        "plugin": plugin,
1101                        "name": name,
1102                        "payload": payload,
1103                    })
1104                ))));
1105            }
1106            EventPayload::DelegateFinished {
1107                session_id,
1108                short_id,
1109                status,
1110                preview,
1111                total_lines,
1112            } => {
1113                logical.push(LogicalTurn::User(user_text(
1114                    &crate::runtime::render_subagent_notification(
1115                        preview,
1116                        *status,
1117                        *total_lines,
1118                        short_id.as_deref().unwrap_or(&session_id.to_string()),
1119                    ),
1120                )));
1121            }
1122            EventPayload::DelegateFinishedV2 {
1123                invocation_id,
1124                session_id,
1125                short_id,
1126                status,
1127                preview,
1128                total_lines,
1129                ..
1130            } if !producer_delegations.contains(invocation_id) => {
1131                logical.push(LogicalTurn::User(user_text(
1132                    &crate::runtime::render_subagent_notification(
1133                        preview,
1134                        *status,
1135                        *total_lines,
1136                        short_id.as_deref().unwrap_or(&session_id.to_string()),
1137                    ),
1138                )));
1139            }
1140            _ => {}
1141        }
1142    }
1143
1144    let mut history = vec![HistoryTurn::system(SystemMessage::new(vec![
1145        SystemPart::Text(TextPart::new(composed_prompt)),
1146    ]))];
1147    if let Some(agent_md) = loaded_agent_md {
1148        history.push(HistoryTurn::user(user_text(&agent_md)));
1149    }
1150    history.extend(
1151        loaded_skills
1152            .iter()
1153            .map(|body| HistoryTurn::user(user_text(body))),
1154    );
1155    let mut replay_decisions = Vec::new();
1156    for turn in logical {
1157        match turn {
1158            LogicalTurn::System(text) => {
1159                let Some(HistoryTurn::System(message)) = history.first_mut() else {
1160                    unreachable!("assembled history starts with a system message")
1161                };
1162                message.content.push(SystemPart::Text(TextPart::new(text)));
1163            }
1164            LogicalTurn::User(user) => history.push(HistoryTurn::user(user)),
1165            LogicalTurn::SeedAssistant(text) => {
1166                history.push(HistoryTurn::assistant(CompletedTurn::new(
1167                    AssistantMessage::new(vec![AssistantPart::Text(TextPart::new(text))]),
1168                    Finish::new(oven_sdk::Usage::default(), FinishReason::Stop),
1169                )));
1170            }
1171            LogicalTurn::Assistant(mut assistant) => {
1172                let retained = assistant
1173                    .calls
1174                    .iter()
1175                    .filter(|call| call.in_stream_result || call.result.is_some())
1176                    .map(|call| call.model_call_id.as_str())
1177                    .collect::<std::collections::HashSet<_>>();
1178                let original_call_count = assistant.calls.len();
1179                assistant.turn.content.retain(|part| match part {
1180                    PersistedAssistantPart::ToolCall { id, .. } => retained.contains(id.as_str()),
1181                    PersistedAssistantPart::ToolApproval { tool_call_id, .. } => {
1182                        retained.contains(tool_call_id.as_str())
1183                    }
1184                    _ => true,
1185                });
1186                let forced_disposition = if retained.len() != original_call_count
1187                    && assistant.turn.native_replay.is_some()
1188                {
1189                    assistant.turn.native_replay = None;
1190                    Some(ReplayDisposition::DiscardedInvalidPayload {
1191                        reason: safe_replay_reason(
1192                            "native replay was discarded because normalized tool-call history changed",
1193                        ),
1194                    })
1195                } else {
1196                    None
1197                };
1198                let restored = if assistant.turn.content.is_empty() {
1199                    None
1200                } else {
1201                    Some(restore_turn_with_store(
1202                        &assistant.turn,
1203                        &assistant.resolved_model,
1204                        store,
1205                        binding,
1206                    )?)
1207                };
1208                // `restore_turn_with_store` drops provider reasoning the target
1209                // cannot replay, so a reasoning-only salvage can restore to an
1210                // empty message. Judge emptiness after restore so neither an
1211                // empty assistant turn nor its interrupt marker reaches history.
1212                let has_content = restored
1213                    .as_ref()
1214                    .is_some_and(|(restored, _)| !restored.message.content.is_empty());
1215                if let Some((restored, disposition)) = restored
1216                    && has_content
1217                {
1218                    let history_index = history.len() as u64;
1219                    if let Some(disposition) = forced_disposition.or(disposition) {
1220                        replay_decisions.push(ReplayDecision {
1221                            history_index,
1222                            disposition,
1223                        });
1224                    }
1225                    history.push(HistoryTurn::assistant(restored));
1226                }
1227                let mut results = assistant
1228                    .calls
1229                    .into_iter()
1230                    .filter_map(|call| call.result.map(|result| (result, call.additional_messages)))
1231                    .collect::<Vec<_>>();
1232                if !results.is_empty() {
1233                    if !has_content {
1234                        return Err(HistoryError::Corrupt(
1235                            "tool result has no retained assistant call".into(),
1236                        ));
1237                    }
1238                    let call_order = assistant
1239                        .turn
1240                        .content
1241                        .iter()
1242                        .enumerate()
1243                        .filter_map(|(index, part)| match part {
1244                            PersistedAssistantPart::ToolCall { id, .. } => {
1245                                Some((id.as_str(), index))
1246                            }
1247                            _ => None,
1248                        })
1249                        .collect::<HashMap<_, _>>();
1250                    results.sort_by_key(|(result, _)| {
1251                        call_order
1252                            .get(result.tool_call_id.as_str())
1253                            .copied()
1254                            .unwrap_or(usize::MAX)
1255                    });
1256                    let mut additional_messages = Vec::new();
1257                    let results = results
1258                        .into_iter()
1259                        .map(|(result, messages)| {
1260                            additional_messages.extend(messages);
1261                            result
1262                        })
1263                        .collect();
1264                    history.push(HistoryTurn::tool(ToolMessage::new(results)));
1265                    for message in additional_messages {
1266                        append_tool_emitted_message(&mut history, &message, store)?;
1267                    }
1268                }
1269                // A turn an interrupt cut off is annotated right after it is
1270                // restored: the marker tells the next request that the fragment
1271                // it sees was not the model's own ending. It is placed after the
1272                // turn's own tool results so it never splits a call from its
1273                // result, and is skipped entirely when restore emptied the turn.
1274                if has_content && assistant.interrupted {
1275                    history.push(HistoryTurn::user(user_text(INTERRUPTED_TURN_MARKER)));
1276                }
1277                let _ = assistant.run_id;
1278            }
1279            LogicalTurn::Rehydration(files) => {
1280                let mut calls = Vec::with_capacity(files.len());
1281                let mut results = Vec::with_capacity(files.len());
1282                for (index, file) in files.into_iter().enumerate() {
1283                    let id = format!("context-rehydration-{index}");
1284                    calls.push(AssistantPart::ToolCall(ToolCallPart::new(
1285                        id.clone(),
1286                        "read",
1287                        serde_json::json!({"filePath": file.path.as_str()}),
1288                    )));
1289                    results.push(ToolResultPart::new(id, ToolContent::Text(file.content)));
1290                }
1291                history.push(HistoryTurn::assistant(CompletedTurn::new(
1292                    AssistantMessage::new(calls),
1293                    Finish::new(oven_sdk::Usage::default(), FinishReason::ToolCalls),
1294                )));
1295                history.push(HistoryTurn::tool(ToolMessage::new(results)));
1296            }
1297        }
1298    }
1299    Ok(AssembledHistory {
1300        history,
1301        replay_decisions,
1302    })

1788fn restore_replay(
1789    artifact: &NativeReplayArtifact,
1790    resolved_model: &ResolvedModelRef,
1791    _binding: &FrozenModelBinding,
1792    source_wire_model_id: Option<&str>,
1793) -> (Option<OvenReplayArtifact>, Option<ReplayDisposition>) {
1794    // Persisted-turn integrity: the artifact must have been recorded by the
1795    // adapter the turn itself resolved to, before any current-eligibility
1796    // checks expose its opaque payload. Artifacts record the Oven adapter ID
1797    // while the persisted turn carries the protocol adapter ID, so compare
1798    // through the shared family mapping.
1799    if cookie_agent_models::adapters::wire_adapter_for_protocol(artifact.adapter_id().as_str())
1800        .is_none()
1801        || crate::policy::wire_adapter(artifact.adapter_id().as_str()) != resolved_model.adapter_id
1802    {
1803        return (
1804            None,
1805            Some(ReplayDisposition::DiscardedInvalidPayload {
1806                reason: safe_replay_reason(
1807                    "native replay adapter does not match its persisted model turn",
1808                ),
1809            }),
1810        );
1811    }
1812    if artifact.selection_fingerprint() != &resolved_model.selection_fingerprint
1813        || artifact.scope().provider_id != resolved_model.provider_id
1814        || artifact.scope().model_id != resolved_model.model_id
1815    {
1816        return (
1817            None,
1818            Some(ReplayDisposition::DiscardedInvalidPayload {
1819                reason: safe_replay_reason(
1820                    "native replay identity does not match its persisted model turn",
1821                ),
1822            }),
1823        );
1824    }
1825    // Source attribution is integrity data. Target codecs decide block eligibility.
1826    let scope = match restore_scope(artifact.scope()) {
1827        Ok(scope) => scope,
1828        Err(error) => {
1829            return (
1830                None,
1831                Some(ReplayDisposition::DiscardedInvalidPayload {
1832                    reason: safe_replay_reason(&error.to_string()),
1833                }),
1834            );
1835        }
1836    };
1837    match OvenReplayArtifact::new(
1838        AdapterId::new(artifact.adapter_id().as_str()),
1839        scope,
1840        artifact.payload().clone(),
1841    )
1842    .and_then(|artifact| match source_wire_model_id {
1843        Some(source) => artifact.with_source_wire_model_id(oven_sdk::ModelId::new(source)),
1844        None => Ok(artifact),
1845    }) {
1846        Ok(artifact) => (Some(artifact), None),
1847        Err(error) => (
1848            None,
1849            Some(ReplayDisposition::DiscardedInvalidPayload {
1850                reason: safe_replay_reason(&error.to_string()),
1851            }),
1852        ),
1853    }
1854}

1907        }
1908    };

1913        )?),

1919    )?;
1920    OvenNativeContextWindow::new(
1921        AdapterId::new(window.adapter_id().as_str()),
1922        scope,
1923        window.payload().clone(),

1925    .map_err(|error| HistoryError::Corrupt(error.to_string()))
1926}

2082fn sanitize_control_free(value: &str, maximum: usize) -> String {
2083    let mut output = String::with_capacity(value.len().min(maximum));
2084    for character in value.chars() {
2085        if output.len() >= maximum {
2086            break;
2087        }
2088        let replacement = if character.is_control() {
2089            ' '
2090        } else {
2091            character
2092        };
2093        if output.len() + replacement.len_utf8() > maximum {
2094            break;
2095        }
2096        output.push(replacement);
2097    }
2098    if output.is_empty() {
2099        "unavailable".to_owned()
2100    } else {
2101        output
2102    }
2103}

2105/// Normalize a model-supplied identifier into a valid [`SafeCode`].
2106///
2107/// Providers can emit tool-call names and custom part kinds that are not valid
2108/// `SafeCode` values. Stored history cannot represent those, so coerce them to a
2109/// valid placeholder and record a user-visible warning instead of failing the run.
2110fn normalize_safe_code(
2111    value: String,
2112    label: &str,
2113    warnings: &mut Vec<SafeErrorMessage>,
2114) -> SafeCode {
2115    if is_valid_safe_code(&value) {
2116        return SafeCode::new(value).expect("validated model identifier");
2117    }
2118    let normalized = normalized_code(&value);
2119    let original = sanitize_control_free(&value, SafeErrorMessage::MAX_BYTES);
2120    warnings.push(
2121        SafeErrorMessage::new(sanitize_control_free(
2122            &format!(
2123                "model {label} \"{original}\" was normalized to \"{normalized}\" because it is not a valid identifier"
2124            ),
2125            SafeErrorMessage::MAX_BYTES,
2126        ))
2127        .expect("sanitized normalization warning is valid"),
2128    );
2129    SafeCode::new(normalized).expect("normalized model identifier is valid")
2130}

2132/// Mirrors [`SafeCode::new`] without consuming the caller's string, so valid
2133/// identifiers move into the [`SafeCode`] without an extra allocation.
2134fn is_valid_safe_code(value: &str) -> bool {
2135    (1..=SafeCode::MAX_BYTES).contains(&value.len())
2136        && value.bytes().enumerate().all(|(index, byte)| {
2137            byte.is_ascii_lowercase()
2138                || byte.is_ascii_digit()
2139                || (index > 0 && matches!(byte, b'.' | b'_' | b'-'))
2140        })
2141}

2143/// Deduplicate and bound warnings so the persisted [`PersistedModelTurn`] event
2144/// satisfies `ModelTurnCommitted`'s 256-warning limit. Identical warnings
2145/// collapse first; if the remainder still exceeds the cap, keep the earliest
2146/// warnings and append one deterministic truncation summary.
2147fn coalesce_warnings(warnings: Vec<SafeErrorMessage>) -> Vec<SafeErrorMessage> {
2148    const MAX_WARNINGS: usize = 256;
2149    let mut seen = HashSet::new();
2150    let mut deduped = Vec::with_capacity(warnings.len());
2151    for warning in warnings {
2152        if seen.insert(warning.clone()) {
2153            deduped.push(warning);
2154        }
2155    }
2156    if deduped.len() <= MAX_WARNINGS {
2157        return deduped;
2158    }
2159    let omitted = deduped.len() - (MAX_WARNINGS - 1);
2160    deduped.truncate(MAX_WARNINGS - 1);
2161    deduped.push(
2162        SafeErrorMessage::new(format!("additional warnings truncated ({omitted} omitted)"))
2163            .expect("bounded truncation summary is valid"),
2164    );
2165    deduped
2166}

2168fn normalized_code(value: &str) -> String {
2169    let mut code = value
2170        .bytes()
2171        .take(SafeCode::MAX_BYTES)
2172        .map(|byte| {
2173            if byte.is_ascii_alphanumeric() {
2174                char::from(byte.to_ascii_lowercase())
2175            } else {
2176                '_'
2177            }
2178        })
2179        .collect::<String>();
2180    if code.is_empty() {
2181        code.push_str("unnamed-tool");
2182    }
2183    if !code.as_bytes()[0].is_ascii_lowercase() && !code.as_bytes()[0].is_ascii_digit() {
2184        code.insert(0, 'x');
2185    }
2186    code.truncate(SafeCode::MAX_BYTES);
2187    code
2188}
