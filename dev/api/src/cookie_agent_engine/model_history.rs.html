36pub(crate) fn checkpoint_retained_history(
37    history: &[HistoryTurn],
38    events: &[StoredEvent],
39    summary: Option<&str>,
40) -> Vec<HistoryTurn> {
41    let pinned_skills = events
42        .iter()
43        .filter(|event| matches!(event.payload, EventPayload::SkillLoaded { .. }))
44        .count();
45    let pinned_agent_md = usize::from(latest_agent_md_event(events).is_some());
46    let pinned_count = pinned_agent_md.saturating_add(pinned_skills);
47    let mut history = history.iter();
48    let mut retained = history.next().cloned().into_iter().collect::<Vec<_>>();
49    retained.extend(
50        history
51            .filter(|turn| !is_framed_summary_turn(turn))
52            .take(pinned_count)
53            .cloned(),
54    );
55    if let Some(summary) = summary {
56        retained.push(HistoryTurn::user(user_text(&framed_compaction_summary(
57            summary,
58        ))));
59    }
60    retained
61}

63fn is_framed_summary_turn(turn: &HistoryTurn) -> bool {
64    matches!(
65        turn,
66        HistoryTurn::User(message)
67            if matches!(message.content.as_slice(), [InputPart::Text(text)]
68                if text.text.starts_with(COMPACTION_SUMMARY_PREFIX)
69                    && text.text.ends_with(COMPACTION_SUMMARY_SUFFIX))
70    )

89fn retained_artifact_id(reference: &ArtifactReference) -> Result<&str, HistoryError> {
90    let digest = reference
91        .uri
92        .strip_prefix("artifact://sha256/")
93        .ok_or_else(|| HistoryError::Corrupt("invalid retained artifact reference".into()))?;
94    Sha256Digest::new(digest)
95        .map_err(|_| HistoryError::Corrupt("invalid retained artifact digest".into()))?;
96    Ok(digest)

120pub(crate) fn persist_turn(
121    mut turn: CompletedTurn,
122    store: &ArtifactRouter,
123    session: SessionId,
124    binding: &FrozenModelBinding,
125) -> Result<(PersistedModelTurn, Vec<SafeErrorMessage>), HistoryError> {
126    turn.finish
127        .provider_metadata
128        .remove("cookie_agent.replay_source_wire_model_id");
129    if let Some(source) = turn
130        .finish
131        .native_replay
132        .as_ref()
133        .and_then(OvenReplayArtifact::source_wire_model_id)
134    {
135        turn.finish.provider_metadata.insert(
136            "cookie_agent.replay_source_wire_model_id".into(),
137            serde_json::Value::String(source.as_str().into()),
138        );
139    }
140    let mut warnings = turn
141        .warnings
142        .iter()
143        .map(|warning| {
144            SafeErrorMessage::new(sanitize_control_free(warning, SafeErrorMessage::MAX_BYTES))
145                .map_err(|error| HistoryError::Corrupt(error.to_string()))
146        })
147        .collect::<Result<Vec<_>, _>>()?;
148    let content = turn
149        .message
150        .content
151        .into_iter()
152        .map(|part| persist_assistant_part(part, store, session, &mut warnings))
153        .collect::<Result<_, _>>()?;
154    Ok((
155        PersistedModelTurn {
156            content,
157            provider_options: turn.message.provider_options,
158            finish_reason: persist_finish_reason(turn.finish.finish_reason),
159            usage: persist_usage(turn.finish.usage),
160            response_metadata: turn.finish.response_metadata,
161            provider_metadata: turn.finish.provider_metadata,
162            native_replay: turn
163                .finish
164                .native_replay
165                .map(|artifact| persist_replay(artifact, binding))
166                .transpose()?,
167        },
168        coalesce_warnings(warnings),
169    ))
170}

172pub(crate) fn replay_decisions(
173    decisions: &[OvenReplayDecision],
174    binding: &FrozenModelBinding,
175) -> Vec<ReplayDecision> {
176    decisions
177        .iter()
178        .map(|decision| ReplayDecision {
179            history_index: decision.history_index as u64,
180            disposition: match &decision.disposition {
181                OvenReplayDisposition::Replayed => ReplayDisposition::Replayed,
182                OvenReplayDisposition::NoArtifact => ReplayDisposition::NoArtifact,
183                OvenReplayDisposition::DiscardedForeignAdapter { found, expected } => {
184                    ReplayDisposition::DiscardedForeignAdapter {
185                        found: exact_adapter_code(found.as_str()),
186                        expected: exact_adapter_code(expected.as_str()),
187                    }
188                }
189                OvenReplayDisposition::DiscardedForeignScope { found, expected } => {
190                    let found_selection = scope_selection(found);
191                    let expected_selection = scope_selection(expected);
192                    if found_selection.model == expected_selection.model {
193                        ReplayDisposition::DiscardedForeignVariant {
194                            found: found_selection.variant,
195                            expected: binding.selection.variant.clone(),
196                        }
197                    } else {
198                        ReplayDisposition::DiscardedForeignModelSelection {
199                            found: found_selection,
200                            expected: binding.selection.clone(),
201                        }
202                    }
203                }
204                OvenReplayDisposition::DiscardedInvalidPayload { reason } => {
205                    ReplayDisposition::DiscardedInvalidPayload {
206                        reason: SafeErrorMessage::new(sanitize_control_free(
207                            reason,
208                            SafeErrorMessage::MAX_BYTES,
209                        ))
210                        .expect("sanitized safe replay error"),
211                    }
212                }
213                OvenReplayDisposition::ReconstructedNormalized => {
214                    ReplayDisposition::ReconstructedNormalizedHistory
215                }
216            },
217        })
218        .collect()
219}

221pub(crate) fn replay_decisions_with_preflight(
222    decisions: &[OvenReplayDecision],
223    binding: &FrozenModelBinding,
224    preflight: &[ReplayDecision],
225) -> Vec<ReplayDecision> {
226    let preflight = preflight
227        .iter()
228        .map(|decision| (decision.history_index, decision.disposition.clone()))
229        .collect::<HashMap<_, _>>();
230    let mut emitted = HashSet::new();
231    let mut merged = Vec::new();
232    for decision in replay_decisions(decisions, binding) {
233        let Some(disposition) = preflight.get(&decision.history_index) else {
234            merged.push(decision);
235            continue;
236        };
237        if emitted.insert(decision.history_index) {
238            merged.push(ReplayDecision {
239                history_index: decision.history_index,
240                disposition: disposition.clone(),
241            });
242        }
243        if !matches!(decision.disposition, ReplayDisposition::NoArtifact) {
244            merged.push(decision);
245        }
246    }
247    for (history_index, disposition) in preflight {
248        if emitted.insert(history_index) {
249            merged.push(ReplayDecision {
250                history_index,
251                disposition,
252            });
253            merged.push(ReplayDecision {
254                history_index,
255                disposition: ReplayDisposition::ReconstructedNormalizedHistory,
256            });
257        } else if !merged.iter().any(|decision| {
258            decision.history_index == history_index
259                && matches!(
260                    decision.disposition,
261                    ReplayDisposition::ReconstructedNormalizedHistory
262                )
263        }) {
264            merged.push(ReplayDecision {
265                history_index,
266                disposition: ReplayDisposition::ReconstructedNormalizedHistory,
267            });
268        }
269    }
270    merged.sort_by_key(|decision| decision.history_index);
271    merged
272}

331fn selected_checkpoint_events(
332    events: &[StoredEvent],
333    source_through_seq: u64,
334    recent_from_seq: Option<u64>,
335    close_dependencies: bool,
336    mode: CheckpointSelectionMode,
337) -> Vec<StoredEvent> {
338    let latest_agent_md_seq = latest_agent_md_event(events).map(|event| event.seq);
339    let pending_producer_admissions = match mode {
340        CheckpointSelectionMode::InternalSummary => GoalProducerProjection::from_events(events)
341            .messages
342            .into_iter()
343            .filter(|message| !message.consumed && !message.discarded)
344            .filter_map(|message| message.admission.map(|(_, seq)| seq))
345            .collect::<HashSet<_>>(),
346        CheckpointSelectionMode::NativeWindow => HashSet::new(),
347    };
348    let mut selected = events
349        .iter()
350        .filter(|event| {
351            is_pinned_event(event, latest_agent_md_seq, source_through_seq)
352                || pending_producer_admissions.contains(&event.seq)
353                || recent_from_seq
354                    .is_some_and(|from| (from..=source_through_seq).contains(&event.seq))
355                || event.seq > source_through_seq
356        })
357        .cloned()
358        .collect::<Vec<_>>();
359    if close_dependencies {
360        close_event_dependencies(events, &mut selected);
361    }
362    selected.sort_by_key(|event| event.seq);
363    selected.dedup_by_key(|event| event.seq);
364    selected
365}

367fn close_event_dependencies(events: &[StoredEvent], selected: &mut Vec<StoredEvent>) {
368    // Applied inputs and late tool completions can refer to events before the retained range.
369    loop {
370        let selected_seqs = selected
371            .iter()
372            .map(|event| event.seq)
373            .collect::<HashSet<_>>();
374        let mut required_seqs = HashSet::new();
375        for event in selected.iter() {
376            match &event.payload {
377                EventPayload::UserInputApplied { user_input_seq } => {
378                    required_seqs.insert(*user_input_seq);
379                }
380                EventPayload::ToolCallStarted { start } => {
381                    if let Some(commit) = events.iter().rev().find(|candidate| {
382                        candidate.seq <= event.seq
383                            && candidate.run_id == event.run_id
384                            && matches!(&candidate.payload,
385                                EventPayload::ModelTurnCommitted { model_turn_seq, turn, .. }
386                                    if *model_turn_seq == start.owner.model_turn_seq
387                                        && turn.content.iter().any(|part| matches!(part,
388                                            PersistedAssistantPart::ToolCall { id, .. }
389                                                if id == &start.owner.model_call_id)))
390                    }) {
391                        required_seqs.insert(commit.seq);
392                    }
393                }
394                EventPayload::ToolCallTerminated { termination } => {
395                    if let Some(start) = events.iter().rev().find(|candidate| {
396                        candidate.seq <= event.seq
397                            && candidate.run_id == event.run_id
398                            && matches!(&candidate.payload,
399                                EventPayload::ToolCallStarted { start }
400                                    if start.tool_call_id == termination.tool_call_id)
401                    }) {
402                        required_seqs.insert(start.seq);
403                    }
404                }
405                _ => {}
406            }
407        }
408        let additions = events
409            .iter()
410            .filter(|event| {
411                required_seqs.contains(&event.seq) && !selected_seqs.contains(&event.seq)
412            })
413            .cloned()
414            .collect::<Vec<_>>();
415        if additions.is_empty() {
416            break;
417        }
418        selected.extend(additions);
419    }
420}

430fn insert_summary(assembled: &mut AssembledHistory, events: &[StoredEvent], summary: &str) {
431    let history_index = 1 + pinned_history_len(events);
432    assembled.history.insert(
433        history_index,
434        HistoryTurn::user(user_text(&framed_compaction_summary(summary))),
435    );
436    for decision in &mut assembled.replay_decisions {
437        if decision.history_index >= history_index as u64 {
438            decision.history_index += 1;
439        }
440    }

469pub(crate) fn compaction_prefix_history(
470    events: &[StoredEvent],
471    store: &ArtifactRouter,
472    binding: &FrozenModelBinding,
473    composed_prompt: &str,
474    recent_from_seq: Option<u64>,
475) -> Result<Vec<HistoryTurn>, HistoryError> {
476    let Some(recent_from_seq) = recent_from_seq else {
477        return Ok(assemble_model_context(events, store, binding, composed_prompt)?.history);
478    };
479    let (mut visible, prior_summary) = if let Some(commit) = latest_checkpoint(events) {
480        let (retained_from, summary, mode) = match &commit.checkpoint {
481            ContextCheckpoint::InternalSummary { checkpoint } => (
482                commit.boundaries.recent_from_seq,
483                Some(checkpoint.summary()),
484                CheckpointSelectionMode::InternalSummary,
485            ),
486            ContextCheckpoint::NativeWindow { .. } => {
487                (None, None, CheckpointSelectionMode::NativeWindow)
488            }
489        };
490        (
491            selected_checkpoint_events(
492                events,
493                commit.boundaries.source_through_seq,
494                retained_from,
495                true,
496                mode,
497            ),
498            summary,
499        )
500    } else {
501        (events.to_vec(), None)
502    };
503    let latest_agent_md_seq = latest_agent_md_event(events).map(|event| event.seq);
504    visible.retain(|event| {
505        event.seq < recent_from_seq || is_pinned_event(event, latest_agent_md_seq, u64::MAX)
506    });
507    close_event_dependencies(events, &mut visible);
508    visible.sort_by_key(|event| event.seq);
509    visible.dedup_by_key(|event| event.seq);
510    let mut assembled =
511        assemble_history_with_replay(&visible, events, store, binding, composed_prompt)?;
512    if let Some(summary) = prior_summary {
513        insert_summary(&mut assembled, events, summary);
514    }
515    Ok(assembled.history)
516}

518pub(crate) fn compaction_tail_candidates(events: &[StoredEvent]) -> Vec<u64> {
519    #[derive(Default)]
520    struct Group {
521        start: u64,
522        end: u64,
523        unresolved_calls: usize,
524    }
525
526    let visible = if let Some(commit) = latest_checkpoint(events) {
527        let (recent_from, mode) = match &commit.checkpoint {
528            ContextCheckpoint::InternalSummary { .. } => (
529                commit.boundaries.recent_from_seq,
530                CheckpointSelectionMode::InternalSummary,
531            ),
532            ContextCheckpoint::NativeWindow { .. } => (None, CheckpointSelectionMode::NativeWindow),
533        };
534        selected_checkpoint_events(
535            events,
536            commit.boundaries.source_through_seq,
537            recent_from,
538            true,
539            mode,
540        )
541    } else {
542        events.to_vec()
543    };
544    let visible_seqs = visible
545        .iter()
546        .map(|event| event.seq)
547        .collect::<HashSet<_>>();
548    let producer_admissions = GoalProducerProjection::from_events(events)
549        .messages
550        .into_iter()
551        .filter(|message| !message.discarded)
552        .filter_map(|message| message.admission.map(|(_, seq)| seq))
553        .collect::<HashSet<_>>();
554    let submitted = events
555        .iter()
556        .filter_map(|event| match event.payload {
557            EventPayload::UserInputSubmitted { .. } => Some(event.seq),
558            _ => None,
559        })
560        .collect::<HashSet<_>>();
561    let mut groups = Vec::<Group>::new();
562    let mut pending = HashMap::<
563        (
564            cookie_agent_protocol::RunId,
565            cookie_agent_protocol::ModelCallId,
566        ),
567        VecDeque<usize>,
568    >::new();
569    let mut started = HashMap::<(cookie_agent_protocol::RunId, ToolCallId), usize>::new();
570    let mut terminated = HashSet::new();
571
572    for event in &visible {
573        match &event.payload {
574            EventPayload::UserInputApplied { user_input_seq }
575                if submitted.contains(user_input_seq) =>
576            {
577                groups.push(Group {
578                    start: *user_input_seq,
579                    end: event.seq,
580                    ..Group::default()
581                });
582            }
583            EventPayload::MessageInjected { role, .. }
584                if *role != cookie_agent_protocol::ExtensionMessageRole::Tool =>
585            {
586                groups.push(Group {
587                    start: event.seq,
588                    end: event.seq,
589                    ..Group::default()
590                });
591            }
592            EventPayload::ProducerMessageAdmitted { .. }
593                if producer_admissions.contains(&event.seq) =>
594            {
595                groups.push(Group {
596                    start: event.seq,
597                    end: event.seq,
598                    ..Group::default()
599                });
600            }
601            EventPayload::DelegatedContextSeeded { turns, .. } if !turns.is_empty() => {
602                groups.push(Group {
603                    start: event.seq,
604                    end: event.seq,
605                    ..Group::default()
606                });
607            }
608            EventPayload::ModelTurnCommitted { turn, .. } => {
609                let in_stream_results = turn
610                    .content
611                    .iter()
612                    .filter_map(|part| match part {
613                        PersistedAssistantPart::ToolResult { tool_call_id, .. } => {
614                            Some(tool_call_id.as_str())
615                        }
616                        _ => None,
617                    })
618                    .collect::<HashSet<_>>();
619                let index = groups.len();
620                let mut unresolved_calls = 0;
621                if let Some(run_id) = event.run_id {
622                    for id in turn.content.iter().filter_map(|part| match part {
623                        PersistedAssistantPart::ToolCall { id, .. }
624                            if !in_stream_results.contains(id.as_str()) =>
625                        {
626                            Some(id)
627                        }
628                        _ => None,
629                    }) {
630                        unresolved_calls += 1;
631                        pending
632                            .entry((run_id, id.clone()))
633                            .or_default()
634                            .push_back(index);
635                    }
636                }
637                groups.push(Group {
638                    start: event.seq,
639                    end: event.seq,
640                    unresolved_calls,
641                });
642            }
643            EventPayload::ToolCallStarted { start } => {
644                let Some(run_id) = event.run_id else { continue };
645                if let Some(index) = pending
646                    .get_mut(&(run_id, start.owner.model_call_id.clone()))
647                    .and_then(VecDeque::pop_front)
648                {
649                    started.insert((run_id, start.tool_call_id), index);
650                }
651            }
652            EventPayload::ToolCallTerminated { termination } => {
653                let Some(run_id) = event.run_id else { continue };
654                if terminated.insert((run_id, termination.tool_call_id))
655                    && let Some(index) = started.get(&(run_id, termination.tool_call_id))
656                {
657                    groups[*index].unresolved_calls =
658                        groups[*index].unresolved_calls.saturating_sub(1);
659                    groups[*index].end = groups[*index].end.max(event.seq);
660                }
661            }
662            EventPayload::ContextRehydrated { .. }
663            | EventPayload::PluginEventAdded { .. }
664            | EventPayload::DelegateFinished { .. }
665            | EventPayload::DelegateFinishedV2 { .. } => groups.push(Group {
666                start: event.seq,
667                end: event.seq,
668                ..Group::default()
669            }),
670            _ => {}
671        }
672    }
673
674    groups.sort_by_key(|group| group.start);
675    groups.dedup_by_key(|group| group.start);
676    let unresolved_from = groups
677        .iter()
678        .filter(|group| group.unresolved_calls != 0)
679        .map(|group| group.start)
680        .min();
681    let spans = groups
682        .iter()
683        .filter(|group| group.end > group.start)
684        .map(|group| (group.start, group.end))
685        .collect::<Vec<_>>();
686    groups
687        .into_iter()
688        .skip(1)
689        .filter(|group| unresolved_from.is_none_or(|start| group.start <= start))
690        .filter(|group| {
691            !spans
692                .iter()
693                .any(|(start, end)| *start < group.start && group.start <= *end)
694        })
695        .filter(|group| visible_seqs.contains(&group.start))
696        .map(|group| group.start)
697        .collect()
698}

705pub(crate) fn assemble_model_context(
706    events: &[StoredEvent],
707    store: &ArtifactRouter,
708    binding: &FrozenModelBinding,
709    composed_prompt: &str,
710) -> Result<ModelContext, HistoryError> {
711    let checkpoint = latest_checkpoint(events);
712    let Some(commit) = checkpoint else {
713        let assembled =
714            assemble_history_with_replay(events, events, store, binding, composed_prompt)?;
715        return Ok(ModelContext {
716            history: assembled.history,
717            native_context: None,
718            replay_decisions: assembled.replay_decisions,
719        });
720    };
721    match &commit.checkpoint {
722        ContextCheckpoint::InternalSummary { checkpoint } => project_summary_context(
723            events,
724            store,
725            binding,
726            composed_prompt,
727            commit.boundaries.source_through_seq,
728            commit.boundaries.recent_from_seq,
729            checkpoint.summary(),
730        ),
731        ContextCheckpoint::NativeWindow { window } => {
732            commit
733                .validate_for_binding(binding)
734                .map_err(|error| HistoryError::Corrupt(error.to_string()))?;
735            let selected = selected_checkpoint_events(
736                events,
737                commit.boundaries.source_through_seq,
738                None,
739                false,
740                CheckpointSelectionMode::NativeWindow,
741            );
742            let assembled =
743                assemble_history_with_replay(&selected, events, store, binding, composed_prompt)?;
744            Ok(ModelContext {
745                history: assembled.history,
746                native_context: Some(restore_native_context(window, binding)?),
747                replay_decisions: assembled.replay_decisions,
748            })
749        }
750    }
751}

762// Message selection must not roll back session-wide replay decisions or pinned context.
763// context_events is the full visible snapshot, before checkpoint or summary-prefix filtering.
764fn assemble_history_with_replay(
765    events: &[StoredEvent],
766    context_events: &[StoredEvent],
767    store: &ArtifactRouter,
768    binding: &FrozenModelBinding,
769    composed_prompt: &str,
770) -> Result<AssembledHistory, HistoryError> {
771    let producer_projection = GoalProducerProjection::from_events(context_events);
772    let current_run = context_events.iter().rev().find_map(|event| {
773        matches!(event.payload, EventPayload::RunStarted { .. })
774            .then_some(event.run_id)
775            .flatten()
776    });
777    let producer_admissions = producer_projection
778        .messages
779        .iter()
780        .filter(|message| !message.discarded)
781        .filter_map(|message| {
782            message
783                .admission
784                .and_then(|(admission_run, admission_seq)| {
785                    (message.consumed || current_run.is_none_or(|run| admission_run == run))
786                        .then_some((admission_seq, message.body.as_str()))
787                })
788        })
789        .collect::<HashMap<_, _>>();
790    let producer_delegations = producer_projection
791        .messages
792        .iter()
793        .filter_map(|message| match &message.producer_owner {
794            cookie_agent_protocol::ProducerOwner::Delegation { invocation_id } => {
795                Some(*invocation_id)
796            }
797            _ => None,
798        })
799        .collect::<HashSet<_>>();
800    let loaded_skills = context_events
801        .iter()
802        .filter_map(|event| match &event.payload {
803            EventPayload::SkillLoaded { rendered_body, .. } => Some(rendered_body.clone()),
804            _ => None,
805        })
806        .collect::<Vec<_>>();
807    let loaded_agent_md = latest_agent_md_event(context_events).and_then(|event| {
808        let EventPayload::AgentMdLoaded { entries } = &event.payload else {
809            return None;
810        };
811        Some(agent_md_turn(entries))
812    });
813    let mut logical = Vec::<LogicalTurn>::new();
814    let mut submitted = HashMap::<u64, String>::new();
815    let mut pending_model_calls = HashMap::<
816        (
817            cookie_agent_protocol::RunId,
818            cookie_agent_protocol::ModelCallId,
819        ),
820        VecDeque<(usize, usize)>,
821    >::new();
822    let mut engine_calls =
823        HashMap::<(cookie_agent_protocol::RunId, ToolCallId), (usize, usize)>::new();
824    let delegation_children = context_events
825        .iter()
826        .filter_map(|event| match event.payload {
827            EventPayload::ToolCallLinked {
828                tool_call_id,
829                child_session_id,
830            } => Some(((event.run_id, tool_call_id), child_session_id)),
831            _ => None,
832        })
833        .collect::<HashMap<_, _>>();
834    let elisions = context_events
835        .iter()
836        .filter_map(|event| match &event.payload {
837            EventPayload::ToolOutputElided {
838                tool_call_id,
839                original_bytes,
840                retained,
841            } => Some((*tool_call_id, (*original_bytes, retained.clone()))),
842            _ => None,
843        })
844        .collect::<HashMap<_, _>>();
845
846    for envelope in events {
847        match &envelope.payload {
848            EventPayload::DelegatedContextSeeded { turns, .. } => {
849                logical.extend(turns.iter().map(|turn| match turn.role {
850                    DelegatedContextRole::User => LogicalTurn::User(user_text(&turn.text)),
851                    DelegatedContextRole::Assistant => {
852                        LogicalTurn::SeedAssistant(turn.text.clone())
853                    }
854                }));
855            }
856            EventPayload::MessageInjected { role, input } => match role {
857                cookie_agent_protocol::ExtensionMessageRole::System => {
858                    logical.push(LogicalTurn::System(input.clone()));
859                }
860                cookie_agent_protocol::ExtensionMessageRole::User => {
861                    logical.push(LogicalTurn::User(user_text(input)));
862                }
863                cookie_agent_protocol::ExtensionMessageRole::Assistant => {
864                    logical.push(LogicalTurn::SeedAssistant(input.clone()));
865                }
866                cookie_agent_protocol::ExtensionMessageRole::Tool => {}
867            },
868            EventPayload::UserInputSubmitted { input, .. } => {
869                submitted.insert(envelope.seq, input.clone());
870            }
871            EventPayload::UserInputApplied { user_input_seq } => {
872                if let Some(input) = submitted.remove(user_input_seq) {
873                    logical.push(LogicalTurn::User(user_text(&input)));
874                }
875            }
876            EventPayload::ProducerMessageAdmitted { .. } => {
877                if let Some(body) = producer_admissions.get(&envelope.seq) {
878                    logical.push(LogicalTurn::User(user_text(body)));
879                }
880            }
881            EventPayload::ModelTurnCommitted {
882                resolved_model,
883                turn,
884                ..
885            } => {
886                let logical_index = logical.len();
887                let mut calls = Vec::new();
888                let in_stream_results = turn
889                    .content
890                    .iter()
891                    .filter_map(|part| match part {
892                        PersistedAssistantPart::ToolResult { tool_call_id, .. } => {
893                            Some(tool_call_id.as_str())
894                        }
895                        _ => None,
896                    })
897                    .collect::<std::collections::HashSet<_>>();
898                for part in &turn.content {
899                    if let PersistedAssistantPart::ToolCall { id, .. } = part {
900                        let call_index = calls.len();
901                        let in_stream_result = in_stream_results.contains(id.as_str());
902                        calls.push(CallRecord {
903                            model_call_id: id.clone(),
904                            engine_call_id: None,
905                            result: None,
906                            additional_messages: Vec::new(),
907                            in_stream_result,
908                        });
909                        if !in_stream_result && let Some(run_id) = envelope.run_id {
910                            pending_model_calls
911                                .entry((run_id, id.clone()))
912                                .or_default()
913                                .push_back((logical_index, call_index));
914                        }
915                    }
916                }
917                logical.push(LogicalTurn::Assistant(Box::new(AssistantRecord {
918                    turn: turn.clone(),
919                    resolved_model: resolved_model.clone(),
920                    run_id: envelope.run_id,
921                    calls,
922                })));
923            }
924            EventPayload::ToolCallStarted { start } => {
925                let Some(run_id) = envelope.run_id else {
926                    continue;
927                };
928                let occurrence = pending_model_calls
929                    .get_mut(&(run_id, start.owner.model_call_id.clone()))
930                    .and_then(VecDeque::pop_front);
931                if let Some((logical_index, call_index)) = occurrence {
932                    let LogicalTurn::Assistant(assistant) = &mut logical[logical_index] else {
933                        return Err(HistoryError::Corrupt(
934                            "tool call mapped to a non-assistant turn".into(),
935                        ));
936                    };
937                    assistant.calls[call_index].engine_call_id = Some(start.tool_call_id);
938                    engine_calls.insert((run_id, start.tool_call_id), (logical_index, call_index));
939                }
940            }
941            EventPayload::ToolCallTerminated { termination }
942                if termination.outcome == ToolTerminationOutcome::Completed
943                    || termination
944                        .result
945                        .as_ref()
946                        .and_then(|result| result.retained_output.as_ref())
947                        .is_some_and(|output| output.incomplete)
948                    // Opt-out pages have no retained_output. Only the commit boundary's
949                    // finalized-result reason admits them after cancellation.
950                    || (termination.outcome == ToolTerminationOutcome::Cancelled
951                        && termination.result.is_some()
952                        && termination.error.as_ref().is_some_and(|error| {
953                            error.code.as_str() == crate::runtime::CANCELLED_AFTER_COMPLETION
954                        }))
955                    || (termination.outcome == ToolTerminationOutcome::Cancelled
956                        && termination
957                            .result
958                            .as_ref()
959                            .zip(
960                                delegation_children
961                                    .get(&(envelope.run_id, termination.tool_call_id)),
962                            )
963                            .is_some_and(|(result, child)| {
964                                crate::delegation_api::delegate_result_matches_child(result, *child)
965                            })) =>
966            {
967                if let Some(result) = &termination.result {
968                    let (mut result_part, additional_messages) =
969                        if let Some((original_bytes, retained)) =
970                            elisions.get(&termination.tool_call_id)
971                        {
972                            (
973                                ToolResultPart::new(
974                                    String::new(),
975                                    ToolContent::Text(tool_output_elision_marker(
976                                        retained_artifact_id(retained)?,
977                                        *original_bytes,
978                                        result.additional_messages.len(),
979                                    )),
980                                ),
981                                Vec::new(),
982                            )
983                        } else {
984                            (
985                                tool_result_part(result, store)?,
986                                result.additional_messages.clone(),
987                            )
988                        };
989                    result_part.is_error = termination.outcome != ToolTerminationOutcome::Completed;
990                    if let Some(error) = &termination.error {
991                        match &mut result_part.content {
992                            ToolContent::Mixed(values) => {
993                                values.push(ContentValue::Text(error.message.to_string()))
994                            }
995                            ToolContent::Text(text) => {
996                                text.push('\n');
997                                text.push_str(error.message.as_str());
998                            }
999                            _ => {}
1000                        }
1001                    }
1002                    attach_result(
1003                        &mut logical,
1004                        &engine_calls,
1005                        envelope.run_id,
1006                        termination.tool_call_id,
1007                        result_part,
1008                        additional_messages,
1009                    )?;
1010                }
1011            }
1012            EventPayload::ToolCallTerminated { termination } => {
1013                let message = termination
1014                    .error
1015                    .as_ref()
1016                    .map_or("tool failed", |error| error.message.as_str());
1017                let (content, metadata) = if let Some(denied) = denied_failure(message) {
1018                    let visible_reason = denied.feedback.as_ref().map_or_else(
1019                        || denied.reason.clone(),
1020                        |feedback| format!("{}: {feedback}", denied.reason),
1021                    );
1022                    let mut metadata = BTreeMap::from([
1023                        (
1024                            "denial_source".into(),
1025                            serde_json::Value::String(match denied.source {
1026                                ApprovalDecisionSource::Policy => "policy".into(),
1027                                ApprovalDecisionSource::Model => "model".into(),
1028                                ApprovalDecisionSource::InternalAgent => "internal_agent".into(),
1029                                ApprovalDecisionSource::PermissionMode => "permission_mode".into(),
1030                                ApprovalDecisionSource::TreeGrant => "tree_grant".into(),
1031                                ApprovalDecisionSource::User => "user".into(),
1032                                ApprovalDecisionSource::DoomLoopGuard => "doom_loop_guard".into(),
1033                                ApprovalDecisionSource::System => "system".into(),
1034                            }),
1035                        ),
1036                        (
1037                            "denial_reason".into(),
1038                            serde_json::Value::String(denied.reason),
1039                        ),
1040                    ]);
1041                    if let Some(feedback) = denied.feedback {
1042                        metadata.insert("feedback".into(), serde_json::Value::String(feedback));
1043                    }
1044                    (
1045                        ToolContent::Denied {
1046                            reason: Some(visible_reason),
1047                        },
1048                        Some(metadata),
1049                    )
1050                } else {
1051                    (ToolContent::Text(message.to_owned()), None)
1052                };
1053                attach_result(
1054                    &mut logical,
1055                    &engine_calls,
1056                    envelope.run_id,
1057                    termination.tool_call_id,
1058                    ToolResultPart {
1059                        tool_call_id: String::new(),
1060                        content,
1061                        is_error: true,
1062                        metadata,
1063                    },
1064                    Vec::new(),
1065                )?;
1066            }
1067            EventPayload::ContextRehydrated { files } => {
1068                logical.push(LogicalTurn::Rehydration(files.clone()));
1069            }
1070            EventPayload::PluginEventAdded {
1071                plugin,
1072                name,
1073                payload,
1074            } => {
1075                logical.push(LogicalTurn::User(user_text(&format!(
1076                    "<plugin_event>{}</plugin_event>",
1077                    serde_json::json!({
1078                        "plugin": plugin,
1079                        "name": name,
1080                        "payload": payload,
1081                    })
1082                ))));
1083            }
1084            EventPayload::DelegateFinished {
1085                session_id,
1086                short_id,
1087                status,
1088                preview,
1089                total_lines,
1090            } => {
1091                logical.push(LogicalTurn::User(user_text(
1092                    &crate::runtime::render_subagent_notification(
1093                        preview,
1094                        *status,
1095                        *total_lines,
1096                        short_id.as_deref().unwrap_or(&session_id.to_string()),
1097                    ),
1098                )));
1099            }
1100            EventPayload::DelegateFinishedV2 {
1101                invocation_id,
1102                session_id,
1103                short_id,
1104                status,
1105                preview,
1106                total_lines,
1107                ..
1108            } if !producer_delegations.contains(invocation_id) => {
1109                logical.push(LogicalTurn::User(user_text(
1110                    &crate::runtime::render_subagent_notification(
1111                        preview,
1112                        *status,
1113                        *total_lines,
1114                        short_id.as_deref().unwrap_or(&session_id.to_string()),
1115                    ),
1116                )));
1117            }
1118            _ => {}
1119        }
1120    }
1121
1122    let mut history = vec![HistoryTurn::system(SystemMessage::new(vec![
1123        SystemPart::Text(TextPart::new(composed_prompt)),
1124    ]))];
1125    if let Some(agent_md) = loaded_agent_md {
1126        history.push(HistoryTurn::user(user_text(&agent_md)));
1127    }
1128    history.extend(
1129        loaded_skills
1130            .iter()
1131            .map(|body| HistoryTurn::user(user_text(body))),
1132    );
1133    let mut replay_decisions = Vec::new();
1134    for turn in logical {
1135        match turn {
1136            LogicalTurn::System(text) => {
1137                let Some(HistoryTurn::System(message)) = history.first_mut() else {
1138                    unreachable!("assembled history starts with a system message")
1139                };
1140                message.content.push(SystemPart::Text(TextPart::new(text)));
1141            }
1142            LogicalTurn::User(user) => history.push(HistoryTurn::user(user)),
1143            LogicalTurn::SeedAssistant(text) => {
1144                history.push(HistoryTurn::assistant(CompletedTurn::new(
1145                    AssistantMessage::new(vec![AssistantPart::Text(TextPart::new(text))]),
1146                    Finish::new(oven_sdk::Usage::default(), FinishReason::Stop),
1147                )));
1148            }
1149            LogicalTurn::Assistant(mut assistant) => {
1150                let retained = assistant
1151                    .calls
1152                    .iter()
1153                    .filter(|call| call.in_stream_result || call.result.is_some())
1154                    .map(|call| call.model_call_id.as_str())
1155                    .collect::<std::collections::HashSet<_>>();
1156                let original_call_count = assistant.calls.len();
1157                assistant.turn.content.retain(|part| match part {
1158                    PersistedAssistantPart::ToolCall { id, .. } => retained.contains(id.as_str()),
1159                    PersistedAssistantPart::ToolApproval { tool_call_id, .. } => {
1160                        retained.contains(tool_call_id.as_str())
1161                    }
1162                    _ => true,
1163                });
1164                let forced_disposition = if retained.len() != original_call_count
1165                    && assistant.turn.native_replay.is_some()
1166                {
1167                    assistant.turn.native_replay = None;
1168                    Some(ReplayDisposition::DiscardedInvalidPayload {
1169                        reason: safe_replay_reason(
1170                            "native replay was discarded because normalized tool-call history changed",
1171                        ),
1172                    })
1173                } else {
1174                    None
1175                };
1176                let has_content = !assistant.turn.content.is_empty();
1177                if has_content {
1178                    let history_index = history.len() as u64;
1179                    let (restored, disposition) = restore_turn_with_store(
1180                        &assistant.turn,
1181                        &assistant.resolved_model,
1182                        store,
1183                        binding,
1184                    )?;
1185                    if let Some(disposition) = forced_disposition.or(disposition) {
1186                        replay_decisions.push(ReplayDecision {
1187                            history_index,
1188                            disposition,
1189                        });
1190                    }
1191                    history.push(HistoryTurn::assistant(restored));
1192                }
1193                let mut results = assistant
1194                    .calls
1195                    .into_iter()
1196                    .filter_map(|call| call.result.map(|result| (result, call.additional_messages)))
1197                    .collect::<Vec<_>>();
1198                if !results.is_empty() {
1199                    if !has_content {
1200                        return Err(HistoryError::Corrupt(
1201                            "tool result has no retained assistant call".into(),
1202                        ));
1203                    }
1204                    let call_order = assistant
1205                        .turn
1206                        .content
1207                        .iter()
1208                        .enumerate()
1209                        .filter_map(|(index, part)| match part {
1210                            PersistedAssistantPart::ToolCall { id, .. } => {
1211                                Some((id.as_str(), index))
1212                            }
1213                            _ => None,
1214                        })
1215                        .collect::<HashMap<_, _>>();
1216                    results.sort_by_key(|(result, _)| {
1217                        call_order
1218                            .get(result.tool_call_id.as_str())
1219                            .copied()
1220                            .unwrap_or(usize::MAX)
1221                    });
1222                    let mut additional_messages = Vec::new();
1223                    let results = results
1224                        .into_iter()
1225                        .map(|(result, messages)| {
1226                            additional_messages.extend(messages);
1227                            result
1228                        })
1229                        .collect();
1230                    history.push(HistoryTurn::tool(ToolMessage::new(results)));
1231                    for message in additional_messages {
1232                        append_tool_emitted_message(&mut history, &message, store)?;
1233                    }
1234                }
1235                let _ = assistant.run_id;
1236            }
1237            LogicalTurn::Rehydration(files) => {
1238                let mut calls = Vec::with_capacity(files.len());
1239                let mut results = Vec::with_capacity(files.len());
1240                for (index, file) in files.into_iter().enumerate() {
1241                    let id = format!("context-rehydration-{index}");
1242                    calls.push(AssistantPart::ToolCall(ToolCallPart::new(
1243                        id.clone(),
1244                        "read",
1245                        serde_json::json!({"filePath": file.path.as_str()}),
1246                    )));
1247                    results.push(ToolResultPart::new(id, ToolContent::Text(file.content)));
1248                }
1249                history.push(HistoryTurn::assistant(CompletedTurn::new(
1250                    AssistantMessage::new(calls),
1251                    Finish::new(oven_sdk::Usage::default(), FinishReason::ToolCalls),
1252                )));
1253                history.push(HistoryTurn::tool(ToolMessage::new(results)));
1254            }
1255        }
1256    }
1257    Ok(AssembledHistory {
1258        history,
1259        replay_decisions,
1260    })
1261}

1263fn latest_agent_md_event(events: &[StoredEvent]) -> Option<&StoredEvent> {
1264    let latest_run = events.iter().rev().find_map(|event| {
1265        matches!(event.payload, EventPayload::RunStarted { .. }).then_some(event.run_id)
1266    });
1267    events.iter().rev().find(|event| {
1268        matches!(event.payload, EventPayload::AgentMdLoaded { .. })
1269            && latest_run.is_none_or(|run| event.run_id == run)
1270    })
1271}

1273fn agent_md_turn(entries: &[cookie_agent_protocol::AgentMdEntry]) -> String {
1274    entries
1275        .iter()
1276        .map(|entry| {
1277            let source = escape_xml_attribute(entry.source.as_str());
1278            let marker = entry.truncated.then(|| {
1279                format!(
1280                    "\n[AGENTS.md context truncated; original size: {} bytes]",
1281                    entry.original_bytes
1282                )
1283            });
1284            format!(
1285                "<agent_md source=\"{source}\">\n{}{}\n</agent_md>",
1286                entry.content,
1287                marker.as_deref().unwrap_or_default()
1288            )
1289        })
1290        .collect::<Vec<_>>()
1291        .join("\n\n")
1292}

1308    logical: &mut [LogicalTurn],
1309    engine_calls: &HashMap<(cookie_agent_protocol::RunId, ToolCallId), (usize, usize)>,
1310    run_id: Option<cookie_agent_protocol::RunId>,
1311    tool_call_id: ToolCallId,
1312    mut result: ToolResultPart,
1313    additional_messages: Vec<ToolEmittedMessage>,
1314) -> Result<(), HistoryError> {
1315    let Some(run_id) = run_id else {
1316        return Ok(());
1317    };
1318    let Some(&(logical_index, call_index)) = engine_calls.get(&(run_id, tool_call_id)) else {
1319        return Ok(());
1320    };
1321    let LogicalTurn::Assistant(assistant) = &mut logical[logical_index] else {
1322        return Err(HistoryError::Corrupt(
1323            "tool result mapped to a non-assistant turn".into(),
1324        ));
1325    };
1326    result.tool_call_id = assistant.calls[call_index].model_call_id.to_string();
1327    assistant.calls[call_index].result = Some(result);
1328    assistant.calls[call_index].additional_messages = additional_messages;
1329    Ok(())
1330}

1745fn restore_replay(
1746    artifact: &NativeReplayArtifact,
1747    resolved_model: &ResolvedModelRef,
1748    _binding: &FrozenModelBinding,
1749    source_wire_model_id: Option<&str>,
1750) -> (Option<OvenReplayArtifact>, Option<ReplayDisposition>) {
1751    // Persisted-turn integrity: the artifact must have been recorded by the
1752    // adapter the turn itself resolved to, before any current-eligibility
1753    // checks expose its opaque payload. Artifacts record the Oven adapter ID
1754    // while the persisted turn carries the protocol adapter ID, so compare
1755    // through the shared family mapping.
1756    if cookie_agent_models::adapters::wire_adapter_for_protocol(artifact.adapter_id().as_str())
1757        .is_none()
1758        || crate::policy::wire_adapter(artifact.adapter_id().as_str()) != resolved_model.adapter_id
1759    {
1760        return (
1761            None,
1762            Some(ReplayDisposition::DiscardedInvalidPayload {
1763                reason: safe_replay_reason(
1764                    "native replay adapter does not match its persisted model turn",
1765                ),
1766            }),
1767        );
1768    }
1769    if artifact.selection_fingerprint() != &resolved_model.selection_fingerprint
1770        || artifact.scope().provider_id != resolved_model.provider_id
1771        || artifact.scope().model_id != resolved_model.model_id
1772    {
1773        return (
1774            None,
1775            Some(ReplayDisposition::DiscardedInvalidPayload {
1776                reason: safe_replay_reason(
1777                    "native replay identity does not match its persisted model turn",
1778                ),
1779            }),
1780        );
1781    }
1782    // Source attribution is integrity data. Target codecs decide block eligibility.
1783    let scope = match restore_scope(artifact.scope()) {
1784        Ok(scope) => scope,
1785        Err(error) => {
1786            return (
1787                None,
1788                Some(ReplayDisposition::DiscardedInvalidPayload {
1789                    reason: safe_replay_reason(&error.to_string()),
1790                }),
1791            );
1792        }
1793    };
1794    match OvenReplayArtifact::new(
1795        AdapterId::new(artifact.adapter_id().as_str()),
1796        scope,
1797        artifact.payload().clone(),
1798    )
1799    .and_then(|artifact| match source_wire_model_id {
1800        Some(source) => artifact.with_source_wire_model_id(oven_sdk::ModelId::new(source)),
1801        None => Ok(artifact),
1802    }) {
1803        Ok(artifact) => (Some(artifact), None),
1804        Err(error) => (
1805            None,
1806            Some(ReplayDisposition::DiscardedInvalidPayload {
1807                reason: safe_replay_reason(&error.to_string()),
1808            }),
1809        ),
1810    }
1811}

2039fn sanitize_control_free(value: &str, maximum: usize) -> String {
2040    let mut output = String::with_capacity(value.len().min(maximum));
2041    for character in value.chars() {
2042        if output.len() >= maximum {
2043            break;
2044        }
2045        let replacement = if character.is_control() {
2046            ' '
2047        } else {
2048            character
2049        };
2050        if output.len() + replacement.len_utf8() > maximum {
2051            break;
2052        }
2053        output.push(replacement);
2054    }
2055    if output.is_empty() {
2056        "unavailable".to_owned()
2057    } else {
2058        output
2059    }
2060}

2062/// Normalize a model-supplied identifier into a valid [`SafeCode`].
2063///
2064/// Providers can emit tool-call names and custom part kinds that are not valid
2065/// `SafeCode` values. Stored history cannot represent those, so coerce them to a
2066/// valid placeholder and record a user-visible warning instead of failing the run.
2067fn normalize_safe_code(
2068    value: String,
2069    label: &str,
2070    warnings: &mut Vec<SafeErrorMessage>,
2071) -> SafeCode {
2072    if is_valid_safe_code(&value) {
2073        return SafeCode::new(value).expect("validated model identifier");
2074    }
2075    let normalized = normalized_code(&value);
2076    let original = sanitize_control_free(&value, SafeErrorMessage::MAX_BYTES);
2077    warnings.push(
2078        SafeErrorMessage::new(sanitize_control_free(
2079            &format!(
2080                "model {label} \"{original}\" was normalized to \"{normalized}\" because it is not a valid identifier"
2081            ),
2082            SafeErrorMessage::MAX_BYTES,
2083        ))
2084        .expect("sanitized normalization warning is valid"),
2085    );
2086    SafeCode::new(normalized).expect("normalized model identifier is valid")
2087}
2088
2089/// Mirrors [`SafeCode::new`] without consuming the caller's string, so valid
2090/// identifiers move into the [`SafeCode`] without an extra allocation.
2091fn is_valid_safe_code(value: &str) -> bool {
2092    (1..=SafeCode::MAX_BYTES).contains(&value.len())
2093        && value.bytes().enumerate().all(|(index, byte)| {
2094            byte.is_ascii_lowercase()
2095                || byte.is_ascii_digit()
2096                || (index > 0 && matches!(byte, b'.' | b'_' | b'-'))
2097        })
2098}

2100/// Deduplicate and bound warnings so the persisted [`PersistedModelTurn`] event
2101/// satisfies `ModelTurnCommitted`'s 256-warning limit. Identical warnings
2102/// collapse first; if the remainder still exceeds the cap, keep the earliest
2103/// warnings and append one deterministic truncation summary.
2104fn coalesce_warnings(warnings: Vec<SafeErrorMessage>) -> Vec<SafeErrorMessage> {
2105    const MAX_WARNINGS: usize = 256;
2106    let mut seen = HashSet::new();
2107    let mut deduped = Vec::with_capacity(warnings.len());
2108    for warning in warnings {
2109        if seen.insert(warning.clone()) {
2110            deduped.push(warning);
2111        }
2112    }
2113    if deduped.len() <= MAX_WARNINGS {
2114        return deduped;
2115    }
2116    let omitted = deduped.len() - (MAX_WARNINGS - 1);
2117    deduped.truncate(MAX_WARNINGS - 1);
2118    deduped.push(
2119        SafeErrorMessage::new(format!("additional warnings truncated ({omitted} omitted)"))
2120            .expect("bounded truncation summary is valid"),
2121    );
2122    deduped
2123}
2124
2125fn normalized_code(value: &str) -> String {
2126    let mut code = value
2127        .bytes()
2128        .take(SafeCode::MAX_BYTES)
2129        .map(|byte| {
2130            if byte.is_ascii_alphanumeric() {
2131                char::from(byte.to_ascii_lowercase())
2132            } else {
2133                '_'
2134            }
2135        })
2136        .collect::<String>();
2137    if code.is_empty() {
2138        code.push_str("unnamed-tool");
2139    }
2140    if !code.as_bytes()[0].is_ascii_lowercase() && !code.as_bytes()[0].is_ascii_digit() {
2141        code.insert(0, 'x');
2142    }
2143    code.truncate(SafeCode::MAX_BYTES);
2144    code
2145}
