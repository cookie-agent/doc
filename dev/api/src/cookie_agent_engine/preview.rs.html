1//! Head-and-tail previews of output too long to show whole. Tool output
2//! streams and subagent reports truncate the same way: the first lines, a
3//! marker naming what was left out and how to read it, then the last lines.
4
5use cookie_agent_config::{SubagentOutputConfig, ToolOutputConfig};
6
7/// When a preview truncates and how much of each end it keeps.
8#[derive(Clone, Copy, Debug, Eq, PartialEq)]
9pub(crate) struct PreviewLimits {
10    /// Output within both limits is shown whole.
11    pub(crate) max_lines: usize,
12    pub(crate) max_bytes: usize,
13    /// Lines kept from each end of truncated output. The byte limit is split
14    /// between the ends in the same ratio.
15    pub(crate) head_lines: usize,
16    pub(crate) tail_lines: usize,
17}
18
19impl PreviewLimits {
20    /// Limits that keep half of the lines from each end.
21    #[cfg(any(test, feature = "test-support"))]
22    pub(crate) const fn halves(max_lines: usize, max_bytes: usize) -> Self {
23        Self {
24            max_lines,
25            max_bytes,
26            head_lines: max_lines - max_lines / 2,
27            tail_lines: max_lines / 2,
28        }
29    }
30
31    /// The same split with `max_bytes` capped at `ceiling`.
32    pub(crate) fn with_byte_ceiling(self, ceiling: usize) -> Self {
33        Self {
34            max_bytes: self.max_bytes.min(ceiling),
35            ..self
36        }
37    }
38
39    fn tail_bytes(self) -> usize {
40        let lines = self.head_lines + self.tail_lines;
41        if lines == 0 {
42            return 0;
43        }
44        usize::try_from(self.max_bytes as u128 * self.tail_lines as u128 / lines as u128)
45            .unwrap_or(self.max_bytes)
46    }
47
48    fn head_bytes(self) -> usize {
49        self.max_bytes - self.tail_bytes()
50    }
51}

56        Self {

58            max_bytes: config.max_bytes,

62    }
63}

77/// The two ends a truncated output shows. `head_source` is a prefix of the
78/// output and `tail_source` a suffix; `tail_source_is_whole` says the suffix
79/// starts at the output's first byte. `total_bytes` is the output's length.
80pub(crate) fn split<'a>(
81    head_source: &'a str,
82    tail_source: &'a str,
83    tail_source_is_whole: bool,
84    total_bytes: u64,
85    limits: PreviewLimits,
86) -> (&'a str, &'a str) {
87    let head = head_within(head_source, limits.head_lines, limits.head_bytes());
88    let mut tail = tail_within(
89        tail_source,
90        tail_source_is_whole,
91        limits.tail_lines,
92        limits.tail_bytes(),
93    );
94    // Never repeat what the head already shows.
95    let tail_start = total_bytes.saturating_sub(tail.len() as u64);
96    if let Some(overlap) = (head.len() as u64).checked_sub(tail_start) {
97        tail = &tail[usize::try_from(overlap)
98            .unwrap_or(tail.len())
99            .min(tail.len())..];
100    }
101    (head, tail)
102}

104/// How much of an output with `total_newlines` line breaks and `total_bytes`
105/// bytes lies between `head` and `tail`: in lines when any line break falls
106/// there, otherwise in bytes.
107pub(crate) fn omitted(total_newlines: u64, total_bytes: u64, head: &str, tail: &str) -> String {
108    let lines = total_newlines.saturating_sub(newlines(head) + newlines(tail));
109    let (count, unit) = if lines > 0 {
110        (lines, "line")
111    } else {
112        (
113            total_bytes.saturating_sub((head.len() + tail.len()) as u64),
114            "byte",
115        )
116    };
117    format!("{count} {unit}{}", if count == 1 { "" } else { "s" })
118}
119
120/// The line that stands for the omitted middle and says how to read it.
121pub(crate) fn marker(omitted: &str, read_more: &str) -> String {
122    format!("[… {omitted} omitted. Read more: {read_more}]")
123}
124
125/// Joins a truncated output's ends around its marker, one line break apart.
126pub(crate) fn render(head: &str, marker: &str, tail: &str) -> String {
127    let mut text = head.to_owned();
128    if !text.is_empty() && !text.ends_with('\n') {
129        text.push('\n');
130    }
131    text.push_str(marker);
132    if !tail.is_empty() {
133        text.push('\n');
134        text.push_str(tail);
135    }
136    text
137}

139pub(crate) fn newlines(text: &str) -> u64 {
140    text.bytes().filter(|byte| *byte == b'\n').count() as u64
141}

143/// The longest prefix within `max_lines` lines and `max_bytes` bytes.
144pub(crate) fn head_within(text: &str, max_lines: usize, max_bytes: usize) -> &str {
145    let mut lines = 0;
146    let mut end = 0;
147    for (offset, character) in text.char_indices() {
148        if lines >= max_lines || offset + character.len_utf8() > max_bytes {
149            break;
150        }
151        end = offset + character.len_utf8();
152        if character == '\n' {
153            lines += 1;
154        }
155    }
156    &text[..end]
157}
158
159/// The last whole lines of `text` within `max_lines` lines and `max_bytes`
160/// bytes. A line cut by the byte limit is dropped unless it is all there is.
161fn tail_within(text: &str, starts_at_line: bool, max_lines: usize, max_bytes: usize) -> &str {
162    if max_lines == 0 || max_bytes == 0 {
163        return "";
164    }
165    let mut start = text.len().saturating_sub(max_bytes);
166    while !text.is_char_boundary(start) {
167        start += 1;
168    }
169    let at_line_start = if start == 0 {
170        starts_at_line
171    } else {
172        text.as_bytes()[start - 1] == b'\n'
173    };
174    if !at_line_start
175        && let Some(newline) = text[start..].find('\n')
176        && start + newline + 1 < text.len()
177    {
178        start += newline + 1;
179    }
180    let tail = &text[start..];
181    let body = tail.strip_suffix('\n').unwrap_or(tail);
182    let mut seen = 0;
183    for (index, byte) in body.bytes().enumerate().rev() {
184        if byte == b'\n' {
185            seen += 1;
186            if seen == max_lines {
187                return &tail[index + 1..];
188            }
189        }
190    }
191    tail
192}

194#[cfg(test)]
195mod tests {

198    fn limits(max_lines: usize, max_bytes: usize, head: usize, tail: usize) -> PreviewLimits {
199        PreviewLimits {
200            max_lines,
201            max_bytes,
202            head_lines: head,
203            tail_lines: tail,
204        }
205    }
206
207    #[test]
208    fn ends_follow_their_line_counts_and_share_bytes_in_that_ratio() {
209        let text = (0..10)
210            .map(|line| format!("line{line}\n"))
211            .collect::<String>();
212        let (head, tail) = split(
213            &text,
214            &text,
215            true,
216            text.len() as u64,
217            limits(5, 1_000, 3, 1),
218        );
219        assert_eq!((head, tail), ("line0\nline1\nline2\n", "line9\n"));
220        assert_eq!(omitted(10, text.len() as u64, head, tail), "6 lines");
221        // 30 bytes split 3:1 leaves the tail 7 bytes; they start inside
222        // line8, whose cut part is dropped.
223        let (head, tail) = split(&text, &text, true, text.len() as u64, limits(5, 30, 3, 1));
224        assert_eq!((head, tail), ("line0\nline1\nline2\n", "line9\n"));
225        // A head-only split spends every byte on the head.
226        let (head, tail) = split(&text, &text, true, text.len() as u64, limits(4, 9, 4, 0));
227        assert_eq!((head, tail), ("line0\nlin", ""));
228        assert_eq!(omitted(10, text.len() as u64, head, tail), "9 lines");
229    }
230
231    #[test]
232    fn a_line_without_breaks_is_counted_in_bytes() {
233        let text = "a€bcdefz";
234        let (head, tail) = split(text, text, true, text.len() as u64, limits(1, 4, 1, 1));
235        assert_eq!((head, tail), ("a", "fz"));
236        assert_eq!(omitted(0, text.len() as u64, head, tail), "7 bytes");
237        assert_eq!(render(head, "[…]", tail), "a\n[…]\nfz");
238        assert_eq!(render("", "[…]", ""), "[…]");
239    }
