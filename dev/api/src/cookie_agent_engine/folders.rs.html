1//! Reading a folder of a machine into a store
2//! (`docs/site/specs/folder-artifacts.md` sections 3 and 7): the runtime
3//! sends it as a tar stream (`fs/read_open` of a folder), which is kept in
4//! an anonymous temporary file, then unpacked one file at a time into the
5//! store's blobs, so no more than one file is in memory at once.

21/// The most bytes of tar stream a folder at the limits takes: its files,
22/// a header and padding for each, and long-name records.
23const MAX_TAR_BYTES: u64 = FOLDER_MAX_BYTES + FOLDER_MAX_ENTRIES as u64 * 3 * 512 + (1 << 20);
24/// How many links are followed one after another before a link is left out.
25const MAX_LINK_HOPS: usize = 8;
26/// The main file a folder shows as when none is named, in this order.
27const DEFAULT_MAINS: [&str; 3] = ["index.html", "README.md", "index.md"];
28
29/// Why a folder could not be read.
30#[derive(Debug)]
31pub(crate) enum FolderReadError {
32    /// No main file was named and the folder has none of the defaults.
33    NoMain,
34    /// The named main file is not in the folder.
35    MissingMain(String),
36    /// Anything else, said for the model and the user.
37    Failed(String),
38}
39
40impl FolderReadError {
41    /// The failure as one sentence; `name_main` says how the caller names a
42    /// main file (`main="…"` for a tag, `main` for a tool).
43    pub(crate) fn describe(&self, name_main: &str) -> String {
44        match self {
45            Self::NoMain => format!(
46                "the folder has no index.html, README.md, or index.md, and more than one file \
47                 at its top; name its main file with {name_main}"
48            ),
49            Self::MissingMain(main) => format!("the folder has no file {main:?} to show"),
50            Self::Failed(reason) => reason.clone(),
51        }
52    }
53}
54
55impl From<String> for FolderReadError {
56    fn from(reason: String) -> Self {
57        Self::Failed(reason)
58    }
59}

61/// The folder `route` names, read from `client` into a temporary file. The
62/// route is used up either way.
63pub(crate) async fn read_tar(client: &RuntimeClient, route: u64) -> Result<std::fs::File, String> {
64    use base64::Engine as _;
65    use std::io::Write as _;
66
67    let opened = client.fs_read_open(route).await?;
68    if opened.kind != FsKind::Folder {
69        client.fs_close(opened.handle).await;
70        return Err("it is not a folder".into());
71    }
72    let mut file = match tempfile::tempfile() {
73        Ok(file) => file,
74        Err(error) => {
75            client.fs_close(opened.handle).await;
76            return Err(format!(
77                "the folder could not be kept while it is read: {error}"
78            ));
79        }
80    };
81    let mut written = 0u64;
82    loop {
83        let chunk = match client
84            .fs_read_chunk(opened.handle, MAX_FS_CHUNK_BYTES)
85            .await
86        {
87            Ok(chunk) => chunk,
88            Err(error) => {
89                client.fs_close(opened.handle).await;
90                return Err(error);
91            }
92        };
93        let data = match base64::engine::general_purpose::STANDARD.decode(chunk.data.as_bytes()) {
94            Ok(data) => data,
95            Err(error) => {
96                client.fs_close(opened.handle).await;
97                return Err(format!(
98                    "the runtime sent data that does not decode: {error}"
99                ));
100            }
101        };
102        written += data.len() as u64;
103        if written > MAX_TAR_BYTES {
104            client.fs_close(opened.handle).await;
105            return Err(format!(
106                "its files hold more than the {} a folder may",
107                human_bytes(FOLDER_MAX_BYTES)
108            ));
109        }
110        if let Err(error) = file.write_all(&data) {
111            client.fs_close(opened.handle).await;
112            return Err(format!(
113                "the folder could not be kept while it is read: {error}"
114            ));
115        }
116        if chunk.summary.is_some() {
117            break;
118        }
119    }
120    file.rewind()
121        .map_err(|error| format!("the folder could not be read back: {error}"))?;
122    Ok(file)
123}

125/// Unpacks the tar stream in `file` into a store: `store` keeps each
126/// file's bytes and returns their digest. `main` names the main file, else
127/// the first of [`DEFAULT_MAINS`] the folder has, else its only file at its
128/// top. Blocking.
129pub(crate) fn unpack(
130    file: std::fs::File,
131    main: Option<&str>,
132    mut store: impl FnMut(&[u8]) -> Result<String, String>,
133) -> Result<FolderManifest, FolderReadError> {
134    let mut archive = tar::Archive::new(file);
135    let mut files: BTreeMap<String, FolderEntry> = BTreeMap::new();
136    let mut links: Vec<(String, String)> = Vec::new();
137    let mut bytes = 0u64;
138    let entries = archive
139        .entries()
140        .map_err(|error| format!("the folder's stream does not read: {error}"))?;
141    for entry in entries {
142        let mut entry =
143            entry.map_err(|error| format!("the folder's stream does not read: {error}"))?;
144        let path = entry
145            .path()
146            .map_err(|error| format!("a name in the folder does not read: {error}"))?;
147        let Some(path) = path
148            .to_str()
149            .map(|path| path.trim_end_matches('/').to_owned())
150        else {
151            return Err(FolderReadError::Failed(
152                "a name in the folder is not UTF-8".into(),
153            ));
154        };
155        let kind = entry.header().entry_type();
156        if kind.is_dir() {
157            continue;
158        }
159        validate_folder_path(&path).map_err(|error| error.to_string())?;
160        if kind.is_symlink() {
161            if let Some(target) = entry
162                .link_name()
163                .ok()
164                .flatten()
165                .and_then(|target| target.to_str().map(str::to_owned))
166            {
167                links.push((path, target));
168            }
169            continue;
170        }
171        if !(kind.is_file() || kind == tar::EntryType::Continuous) {
172            continue;
173        }
174        let size = entry.header().size().unwrap_or(u64::MAX);
175        if size > ASSET_MAX_BYTES {
176            return Err(FolderReadError::Failed(format!(
177                "{path} holds {}, more than the {} one file may",
178                human_bytes(size),
179                human_bytes(ASSET_MAX_BYTES)
180            )));
181        }
182        bytes = bytes.saturating_add(size);
183        if bytes > FOLDER_MAX_BYTES {
184            return Err(FolderReadError::Failed(format!(
185                "its files hold more than the {} a folder may",
186                human_bytes(FOLDER_MAX_BYTES)
187            )));
188        }
189        if files.len() >= FOLDER_MAX_ENTRIES {
190            return Err(too_many(files.len() + 1, &files));
191        }
192        let mut content = Vec::with_capacity(size as usize);
193        entry
194            .read_to_end(&mut content)
195            .map_err(|error| format!("{path} does not read: {error}"))?;
196        let sha256 = store(&content)?;
197        let media_type = media_type(&path, &content).to_owned();
198        files.insert(
199            path.clone(),
200            FolderEntry {
201                path,
202                sha256,
203                byte_length: content.len() as u64,
204                media_type,
205            },
206        );
207    }
208    resolve_links(&mut files, &links)?;
209    if files.is_empty() {
210        return Err(FolderReadError::Failed("the folder holds no files".into()));
211    }
212    let main = match main {
213        Some(main) => {
214            let main = main.trim_start_matches("./").trim_end_matches('/');
215            if !files.contains_key(main) {
216                return Err(FolderReadError::MissingMain(main.to_owned()));
217            }
218            main.to_owned()
219        }
220        None => default_main(&files).ok_or(FolderReadError::NoMain)?,
221    };
222    FolderManifest::new(main, files.into_values().collect())
223        .map_err(|error| FolderReadError::Failed(error.to_string()))
224}

226/// The main file of `files` when none is named.
227fn default_main(files: &BTreeMap<String, FolderEntry>) -> Option<String> {
228    if let Some(main) = DEFAULT_MAINS.iter().find(|main| files.contains_key(**main)) {
229        return Some((*main).to_owned());
230    }
231    let mut top = files.keys().filter(|path| !path.contains('/'));
232    match (top.next(), top.next()) {
233        (Some(only), None) => Some(only.clone()),
234        _ => None,
235    }
236}

238/// Adds the files `links` lead to inside the folder: a link to a file is
239/// that file again, and a link to a folder of it is each file under it. A
240/// link that leaves the folder, or leads to nothing, is left out.
241fn resolve_links(
242    files: &mut BTreeMap<String, FolderEntry>,
243    links: &[(String, String)],
244) -> Result<(), FolderReadError> {
245    let targets: HashMap<&str, Option<String>> = links
246        .iter()
247        .map(|(path, target)| (path.as_str(), join_inside(path, target)))
248        .collect();
249    for (path, _) in links {
250        let mut target = targets.get(path.as_str()).cloned().flatten();
251        // A link to a link is followed, a few hops at most.
252        for _ in 0..MAX_LINK_HOPS {
253            match target.as_deref().and_then(|next| targets.get(next)) {
254                Some(next) => target = next.clone(),
255                None => break,
256            }
257        }
258        let Some(target) = target else { continue };
259        if let Some(entry) = files.get(&target).cloned() {
260            files.insert(
261                path.clone(),
262                FolderEntry {
263                    path: path.clone(),
264                    ..entry
265                },
266            );
267        } else {
268            let prefix = format!("{target}/");
269            let under: Vec<FolderEntry> = files
270                .range(prefix.clone()..)
271                .take_while(|(name, _)| name.starts_with(&prefix))
272                .map(|(_, entry)| entry.clone())
273                .collect();
274            for entry in under {
275                let alias = format!("{path}/{}", &entry.path[prefix.len()..]);
276                if validate_folder_path(&alias).is_ok() {
277                    files.insert(
278                        alias.clone(),
279                        FolderEntry {
280                            path: alias,
281                            ..entry
282                        },
283                    );
284                }
285            }
286        }
287        if files.len() > FOLDER_MAX_ENTRIES {
288            return Err(too_many(files.len(), files));
289        }
290    }
291    Ok(())
292}

294/// The path `target`, a link at `link`, leads to inside the folder, or
295/// `None` when it is absolute or climbs out.
296fn join_inside(link: &str, target: &str) -> Option<String> {
297    if target.starts_with('/') || target.contains('\\') {
298        return None;
299    }
300    let mut parts: Vec<&str> = link.split('/').collect();
301    parts.pop();
302    for part in target.split('/') {
303        match part {
304            "" | "." => {}
305            ".." => {
306                parts.pop()?;
307            }
308            part => parts.push(part),
309        }
310    }
311    (!parts.is_empty()).then(|| parts.join("/"))
312}

314/// The refusal of a folder with more than [`FOLDER_MAX_ENTRIES`] files,
315/// naming its largest subfolders by count.
316fn too_many(count: usize, files: &BTreeMap<String, FolderEntry>) -> FolderReadError {
317    let mut by_folder: HashMap<&str, usize> = HashMap::new();
318    for path in files.keys() {
319        if let Some((folder, _)) = path.split_once('/') {
320            *by_folder.entry(folder).or_default() += 1;
321        }
322    }
323    let mut largest: Vec<_> = by_folder.into_iter().collect();
324    largest.sort_by(|a, b| b.1.cmp(&a.1).then(a.0.cmp(b.0)));
325    let named = largest
326        .iter()
327        .take(3)
328        .map(|(folder, count)| format!("{folder}/ ({count}+ files)"))
329        .collect::<Vec<_>>()
330        .join(", ");
331    let mut reason = format!(
332        "the folder holds more than the {FOLDER_MAX_ENTRIES} files a folder may ({count} so far)"
333    );
334    if !named.is_empty() {
335        reason.push_str(&format!("; the largest of its folders: {named}"));
336    }
337    FolderReadError::Failed(reason)
338}

340/// The media type of a folder's file: by content first, then by its
341/// extension, which decides for the files a page loads (scripts, styles,
342/// data, fonts, WebAssembly), since a browser needs their types.
343pub(crate) fn media_type(path: &str, bytes: &[u8]) -> &'static str {
344    let recognized = crate::assets::media_type(path, bytes);
345    if !matches!(recognized, "text/plain" | "application/octet-stream") {
346        return recognized;
347    }
348    let extension = path
349        .rsplit_once('.')
350        .map(|(_, extension)| extension.to_ascii_lowercase())
351        .unwrap_or_default();
352    let text = recognized == "text/plain";
353    match extension.as_str() {
354        "js" | "mjs" | "cjs" if text => "text/javascript",
355        "css" if text => "text/css",
356        "json" | "map" if text => "application/json",
357        "geojson" if text => "application/geo+json",
358        "csv" if text => "text/csv",
359        "tsv" if text => "text/tab-separated-values",
360        "xml" if text => "application/xml",
361        "txt" | "log" if text => "text/plain",
362        "wasm" => "application/wasm",
363        "woff2" => "font/woff2",
364        "woff" => "font/woff",
365        "ttf" => "font/ttf",
366        "otf" => "font/otf",
367        _ => recognized,
368    }
