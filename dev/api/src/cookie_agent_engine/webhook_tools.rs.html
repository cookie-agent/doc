1//! The webhook tools of the `global` runtime
2//! (`docs/site/specs/webhooks.md` section 9): `webhook_create`,
3//! `webhook_list`, `webhook_update`, and `webhook_delete`, provider
4//! `builtin.webhooks`, group `webhooks`, deferred by default.
5//!
6//! Every tool but `webhook_list` is checked under the `webhook` action
7//! with the target kind of each webhook it touches as a resource
8//! (`session`, `subagent`, or `new`), as the cron tools are under `cron`;
9//! root runs ask by default, since a webhook opens a public endpoint.
10//! `webhook_list` shows only the webhooks whose kind the rules allow or ask
11//! for. Targets resolve as the cron tools' do. An agent never names a
12//! permission mode or a model, and sees a webhook's secret only in the
13//! result of the call that made it.
14

84#[serde(deny_unknown_fields)]
85struct WebhookNewArg {
86    /// The agent each new session runs; this session's agent when omitted
87    #[serde(default, skip_serializing_if = "Option::is_none")]
88    agent: Option<AgentId>,
89    /// Rules added to each new session, e.g. {"action": "bash",
90    /// "resource": "git *", "effect": "allow"}
91    #[serde(default, skip_serializing_if = "Option::is_none")]
92    #[schemars(with = "Option<Vec<RuleArg>>")]
93    permission_rules: Option<Vec<PermissionRule>>,
94    /// The runtime capabilities each new session works through, as
95    /// `delegate_subagent` takes them; this session's when omitted
96    #[serde(default, skip_serializing_if = "Option::is_none")]
97    capabilities: Option<Vec<CapabilityArg>>,
98    /// A project's name; this session's project when omitted
99    #[serde(default, skip_serializing_if = "Option::is_none")]
100    project: Option<String>,
101    /// Where each request's thread key is read: {"json": "<JSON Pointer>"}
102    /// or {"header": "<name>"}; events with the same key share one session
103    #[serde(default, skip_serializing_if = "Option::is_none")]
104    thread_key: Option<WebhookValueSource>,
105    /// Instructions each new session gets once as a plain user message
106    /// before its first event, and again after each compaction
107    #[serde(default, skip_serializing_if = "Option::is_none")]
108    thread_prompt: Option<String>,
109    /// What each new session is called, plain text of at most 96
110    /// characters, followed by " · " and the thread key or the arrival
111    /// time; the webhook's name when omitted. Name what the sessions are
112    /// about; nothing in it is read from requests
113    #[serde(default, skip_serializing_if = "Option::is_none")]
114    title: Option<String>,
115    /// true leaves each new session's title to the automatic title after
116    /// its first turn instead of `title`
117    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
118    title_auto: bool,
119}

278fn permission_name(tool: &str) -> Result<Option<&'static str>, ToolError> {
279    match tool {
280        LIST => Ok(None),
281        CREATE | UPDATE | DELETE => Ok(Some("webhook")),
282        other => Err(ToolError::execution(format!(
283            "the webhook provider received another tool, `{other}`"
284        ))),
285    }
286}

333    /// The target a webhook tool's `target` names, resolved as the cron
334    /// tools resolve theirs.
335    fn resolve_webhook_target(
336        &self,
337        caller: &cookie_agent_protocol::CronAgentActor,
338        target: WebhookTargetArg,
339    ) -> Result<WebhookTarget, ToolError> {
340        let (arg, thread_key) = match target {
341            WebhookTargetArg::Session(reference) => (TargetArg::Session(reference), None),
342            WebhookTargetArg::New { new } => (
343                TargetArg::New {
344                    new: NewArg {
345                        agent: new.agent,
346                        permission_rules: new.permission_rules,
347                        capabilities: new.capabilities,
348                        project: new.project,
349                        thread_key: None,
350                        thread_prompt: new.thread_prompt,
351                        title: new.title,
352                        title_auto: new.title_auto,
353                    },
354                },
355                new.thread_key,
356            ),
357        };
358        Ok(match self.resolve_tool_target(caller, arg)? {
359            CronTarget::Session(session) => WebhookTarget::Session(session),
360            CronTarget::Subagent(session) => WebhookTarget::Subagent(session),
361            CronTarget::New(template) => WebhookTarget::New(WebhookTemplate {
362                agent: template.agent,
363                preset: template.preset,
364                model: None,
365                permission_rules: template.permission_rules,
366                runtimes: template.runtimes,
367                global: template.global,
368                cwd: template.cwd,
369                project: template.project,
370                acp_runtime: template.acp_runtime,
371                thread_key,
372                thread_prompt: template.thread_prompt,
373                title: template.title,
374                title_auto: template.title_auto,
375            }),
376        })
377    }

379    /// Whether the calling run's rules allow or ask for `webhook` on
380    /// `kind`.
381    fn webhook_kind_visible(
382        &self,
383        session: SessionId,
384        run: cookie_agent_protocol::RunId,
385        kind: CronTargetKind,
386    ) -> bool {
387        let Some(policy) = self.active_run_policy(run) else {
388            return false;
389        };
390        let overlay = self
391            .inner
392            .store
393            .get(session)
394            .map(|projection| projection.permission_overlay.clone())
395            .unwrap_or_default();
396        let (effect, _) = crate::permissions::effective_permission_with_overlay(
397            &policy.agent,
398            Some(&overlay),
399            &PermissionAction::WEBHOOK,
400            kind.as_str(),
401            crate::permissions::PermissionScope::runtime(RuntimeId::GLOBAL, None),
402        );
403        matches!(effect, PermissionEffect::Allow | PermissionEffect::Ask)
404    }

406    /// One webhook as a tool result shows it.
407    fn webhook_text(&self, webhook: &Webhook) -> String {
408        let target = match &webhook.target {
409            WebhookTarget::Session(session) => self.target_text(&CronTarget::Session(*session)),
410            WebhookTarget::Subagent(session) => self.target_text(&CronTarget::Subagent(*session)),
411            WebhookTarget::New(template) => {
412                let mut text = format!("new session (agent {}", template.agent);
413                if let Some(key) = &template.thread_key {
414                    match (&key.header, &key.json) {
415                        (Some(header), _) => {
416                            text.push_str(&format!(", thread key header {header}"))
417                        }
418                        (_, Some(json)) => text.push_str(&format!(", thread key json {json}")),
419                        _ => {}
420                    }
421                }
422                text.push(')');
423                text
424            }
425        };
426        let check = webhook.verify.as_ref().map_or_else(
427            || "none (the URL is the secret)".to_owned(),
428            |verify| verify.kind.to_string(),
429        );
430        let last = webhook.history.last().map_or_else(
431            || "no events yet".to_owned(),
432            |event| {
433                let outcome = match event.outcome.known() {
434                    Some(WebhookEventOutcome::Pending) => "being delivered".to_owned(),
435                    Some(WebhookEventOutcome::Waiting) => "waiting".to_owned(),
436                    Some(WebhookEventOutcome::Delivered) => match &event.run_end {
437                        Some(end) => format!("delivered, run {}", status_name(end.status)),
438                        None => "delivered".to_owned(),
439                    },
440                    Some(WebhookEventOutcome::Failed { error }) => format!("failed: {error}"),
441                    None => "unknown".to_owned(),
442                };
443                format!("{outcome} at {}", when(event.received_at))
444            },
445        );
446        format!(
447            "{} \"{}\"\n  url: {}\n  target: {target}\n  check: {check}\n  enabled: {}\n  waiting: {}\n  last event: {last}",
448            webhook.id,
449            webhook.name.clone().unwrap_or_default(),
450            self.webhook_url(&webhook.path),
451            if webhook.enabled { "yes" } else { "no" },
452            webhook.waiting,
453        )
454    }
455
456    /// The URL of the webhook at `path` when the daemon knows its public
457    /// base, else the path.
458    pub(crate) fn webhook_url(&self, path: &str) -> String {
459        match self.share_base() {
460            Some(base) => format!("{}{path}", base.trim_end_matches('/')),
461            None => format!("{path} (on the daemon's public address; set [public] url)"),
462        }
463    }

507    fn presentation(&self, call: &ToolCall) -> cookie_agent_protocol::PreparedPresentation {
508        let text = |member: &str| {
509            call.arguments
510                .get(member)
511                .and_then(serde_json::Value::as_str)
512                .map(str::to_owned)
513        };
514        let id = text("id").unwrap_or_default();
515        let title = match call.name.as_str() {
516            CREATE => format!("Create webhook {}", text("name").unwrap_or_default())
517                .trim_end()
518                .to_owned(),
519            LIST => "List webhooks".to_owned(),
520            UPDATE => format!("Change webhook {id}"),
521            DELETE => format!("Delete webhook {id}"),
522            _ => return cookie_agent_runtime_api::tool_title_only(&call.name),
523        };
524        let mut entries = Vec::new();
525        if let Some(target) = call.arguments.get("target") {
526            entries.push((
527                "target",
528                target
529                    .as_str()
530                    .map_or_else(|| target.to_string(), str::to_owned),
531            ));
532        }
533        if let Some(prompt) = text("prompt") {
534            entries.push(("prompt", prompt));
535        }
536        if let Some(verify) = call.arguments.get("verify") {
537            // The secret, if given, stays out of the approval card.
538            let mut shown = verify.clone();
539            if let Some(object) = shown.as_object_mut() {
540                object.remove("secret");
541            }
542            entries.push(("verify", shown.to_string()));
543        }
544        crate::base_tools::base_presentation(
545            &title,
546            cookie_agent_protocol::ToolHint::GENERIC,
547            &id,
548            entries,
549        )
550    }

552    async fn prepare(
553        &self,
554        ctx: ToolPreparationContext,
555        call: ToolCall,
556    ) -> Result<PreparedTool, ToolError> {
557        let engine = self.engine()?;
558        permission_name(&call.name)?;
559        let (call_kind, kinds) = match call.name.as_str() {
560            CREATE => {
561                let args: CreateArgs = parse_args_strict(CREATE, call.arguments.clone())?;
562                let caller = engine.cron_agent(ctx.session, ctx.run)?;
563                let target = engine.resolve_webhook_target(&caller, args.target)?;
564                let kind = target.kind();
565                let mut params = WebhookCreateParams::new(target);
566                params.name = args.name;
567                params.prompt = args.prompt;
568                params.delivery = args.delivery;
569                params.verify = args.verify;
570                params.filter = args.filter.unwrap_or_default();
571                params.headers = args.headers.unwrap_or_default();
572                params.max_per_hour = args.max_per_hour;
573                params.max_concurrent = args.max_concurrent;
574                (WebhookCall::Create(params), vec![kind])
575            }
576            LIST => {
577                let args: ListArgs = parse_args(LIST, call.arguments.clone())?;
578                (WebhookCall::List(args.target), Vec::new())
579            }
580            UPDATE => {
581                let args: UpdateArgs = parse_args_strict(UPDATE, call.arguments.clone())?;
582                let webhook = engine
583                    .inner
584                    .webhooks
585                    .webhook(&args.id)
586                    .ok_or_else(|| refused(format!("there is no webhook {}", args.id)))?;
587                let mut kinds = vec![webhook.target.kind()];
588                let target = match args.target {
589                    Some(target) => {
590                        let caller = engine.cron_agent(ctx.session, ctx.run)?;
591                        let target = engine.resolve_webhook_target(&caller, target)?;
592                        if target.kind() != kinds[0] {
593                            kinds.push(target.kind());
594                        }
595                        Some(target)
596                    }
597                    None => None,
598                };
599                let mut params = WebhookUpdateParams::new(args.id);
600                params.name = args.name.map(Some);
601                params.prompt = args.prompt.map(Some);
602                params.target = target;
603                params.delivery = args.delivery;
604                params.verify = args.verify.map(Some);
605                params.filter = args.filter;
606                params.headers = args.headers;
607                params.max_per_hour = args.max_per_hour.map(Some);
608                params.max_concurrent = args.max_concurrent.map(Some);
609                params.enabled = args.enabled;
610                (WebhookCall::Update(params), kinds)
611            }
612            DELETE => {
613                let args: IdArgs = parse_args(DELETE, call.arguments.clone())?;
614                let webhook = engine
615                    .inner
616                    .webhooks
617                    .webhook(&args.id)
618                    .ok_or_else(|| refused(format!("there is no webhook {}", args.id)))?;
619                (WebhookCall::Delete(args.id), vec![webhook.target.kind()])
620            }
621            _ => unreachable!("checked by permission_name"),
622        };
623        let executor = Box::new(WebhookExecutor {
624            engine,
625            call: call_kind,
626        });
627        if kinds.is_empty() {
628            return PreparedTool::new(
629                prepared_operation(
630                    &call.name,
631                    &call.arguments,
632                    vec![(PermissionAction::WEBHOOK, "list")],
633                    vec![prepared_resource(
634                        PermissionAction::WEBHOOK,
635                        "webhook",
636                        b"list",
637                        b"list",
638                        PreparedBindingLifetime::RestartStable,
639                        ApprovalResourceSource::PrimaryOperation,
640                    )?],
641                    &ctx.context_bytes(),
642                )?,
643                call.arguments,
644                None,
645                executor,
646            );
647        }
648        let labels = kinds
649            .iter()
650            .map(|kind| kind.as_str().to_owned())
651            .collect::<Vec<_>>();
652        let resources = labels
653            .iter()
654            .map(|label| {
655                prepared_resource(
656                    PermissionAction::WEBHOOK,
657                    "webhook",
658                    label.as_bytes(),
659                    label.as_bytes(),
660                    PreparedBindingLifetime::RestartStable,
661                    ApprovalResourceSource::PrimaryOperation,
662                )
663            })
664            .collect::<Result<Vec<_>, _>>()?;
665        let operation = prepared_operation(
666            &call.name,
667            &call.arguments,
668            vec![(
669                PermissionAction::WEBHOOK,
670                call.name.trim_start_matches("webhook_"),
671            )],
672            resources,
673            &ctx.context_bytes(),
674        )?;
675        PreparedTool::new(operation, call.arguments, None, executor)?.with_policy_labels(labels)
676    }
677}
678
679struct WebhookExecutor {
680    engine: Engine,
681    call: WebhookCall,
682}
683
684#[async_trait]
685impl PreparedExecutor for WebhookExecutor {
686    async fn execute(
687        self: Box<Self>,
688        context: ToolExecutionContext,
689    ) -> Result<ToolCompletion, ToolError> {
690        let engine = self.engine;
691        let session = context.session;
692        let error = |error: crate::EngineError| refused(error.to_string());
693        match self.call {
694            WebhookCall::Create(params) => {
695                let caller = engine.cron_agent(session, context.run)?;
696                let created = engine
697                    .create_webhook_as(CronRequester::Agent(caller), params)
698                    .await
699                    .map_err(error)?;
700                let mut output = format!(
701                    "Created webhook {}. Send HTTP POST requests to {}",
702                    created.webhook.id,
703                    engine.webhook_url(&created.webhook.path)
704                );
705                if let Some(secret) = &created.secret {
706                    output.push_str(&format!(
707                        "\nIts secret, shown only now: {secret}\nConfigure the sender with it; you cannot read it again."
708                    ));
709                }
710                output.push_str("\n\n");
711                output.push_str(&engine.webhook_text(&created.webhook));
712                Ok(text_result("Created a webhook", output))
713            }
714            WebhookCall::List(filter) => {
715                let listed = engine.list_webhooks();
716                let webhooks = listed
717                    .webhooks
718                    .into_iter()
719                    .filter(|webhook| filter.is_none_or(|kind| webhook.target.kind() == kind))
720                    .filter(|webhook| {
721                        engine.webhook_kind_visible(session, context.run, webhook.target.kind())
722                    })
723                    .collect::<Vec<_>>();
724                let mut output = if webhooks.is_empty() {
725                    "No webhooks.".to_owned()
726                } else {
727                    webhooks
728                        .iter()
729                        .map(|webhook| engine.webhook_text(webhook))
730                        .collect::<Vec<_>>()
731                        .join("\n\n")
732                };
733                if !listed.receiving {
734                    output.push_str(
735                        "\n\nReceiving is off ([webhooks] enabled = false): every URL answers 404.",
736                    );
737                }
738                Ok(text_result(
739                    &format!(
740                        "Listed {} {}",
741                        webhooks.len(),
742                        if webhooks.len() == 1 {
743                            "webhook"
744                        } else {
745                            "webhooks"
746                        }
747                    ),
748                    output,
749                ))
750            }
751            WebhookCall::Update(params) => {
752                let caller = engine.cron_agent(session, context.run)?;
753                let changed = engine
754                    .update_webhook_as(CronRequester::Agent(caller), params)
755                    .await
756                    .map_err(error)?;
757                let mut output = format!("Changed webhook {}.", changed.webhook.id);
758                if let Some(secret) = &changed.secret {
759                    output.push_str(&format!("\nIts new secret, shown only now: {secret}"));
760                }
761                output.push_str("\n\n");
762                output.push_str(&engine.webhook_text(&changed.webhook));
763                Ok(text_result("Changed a webhook", output))
764            }
765            WebhookCall::Delete(id) => {
766                engine.delete_webhook(id.clone()).await.map_err(error)?;
767                Ok(text_result(
768                    "Deleted a webhook",
769                    format!("Deleted webhook {id}."),
770                ))
771            }
772        }
773    }
774}
