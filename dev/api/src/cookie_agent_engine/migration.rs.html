1//! One-shot migration from the legacy flat project store (v1) to the
2//! hierarchical per-work-dir layout (v2) — spec §6.
3//!
4//! [`SessionStore::open`] runs [`run_if_needed`] before it constructs anything, so
5//! a store is never served from a half-migrated layout. Every step is a rename
6//! guarded by an existence check, which makes the procedure resumable: a crash
7//! leaves `.migrating` in the legacy project directory and the next open picks
8//! the job back up. Migration only runs while no session in the work dir is
9//! owned, which also rules out interrupted appends.

42/// Journal kept in the legacy project dir while a migration is in flight.

79pub(crate) fn run_if_needed(
80    data_root: &Path,
81    cwd: &Path,
82    progress: &dyn Fn(&str),
83) -> Result<Option<Outcome>, SessionError> {
84    let legacy_root = SessionStore::project_dir(data_root, cwd);
85    let legacy_sessions = legacy_root.join(LEGACY_SESSIONS_DIR);
86    let legacy_artifacts = legacy_root.join(ARTIFACTS_DIR);
87    let target = SessionStore::resolve_workdir_dir(data_root, cwd);
88    if !legacy_sessions.is_dir() {
89        // Nothing is left to move. The finishing window is the one place where
90        // the work can be done while the marker still says `in-progress`:
91        // `.migrated` is written before the flat directories are retired, so it
92        // proves verification passed. Claim completion instead of leaving the
93        // store in-progress forever.
94        finish_claimed_completion(&legacy_root, &target, &legacy_sessions, &legacy_artifacts)?;
95        return Ok(None);
96    }
97    if migration_state(&target).as_deref() == Some(MIGRATION_COMPLETE)
98        && !has_remnants(&legacy_sessions)?
99        && !has_remnants(&legacy_artifacts)?
100    {
101        // §6.1: the v2 store is live and the flat directory is a leftover. Point
102        // old binaries at the new home and leave the remnant alone.
103        ensure_tombstone(&legacy_root, &target)?;
104        Journal::clear(&legacy_root)?;
105        return Ok(None);
106    }
107    let _lock = MigrationLock::acquire(&legacy_root)?;
108    // Nothing else writes here while the lock is held, so a temporary left by an
109    // interrupted atomic write is litter; under it the flat project still counts
110    // as unfinished and the store never gets to claim its migration is done.
111    sweep_litter(&legacy_root, &target)?;
112    // Plan first, then lock: knowing where each directory lands is what lets the
113    // same process hold its owner lock across the rename (M1) instead of probing
114    // and releasing.
115    let mut plan = Plan::discover(&legacy_sessions, &target)?;
116    let mut journal = Journal::load(&legacy_root);
117    let resumed = journal.is_resumed();
118    // Which caches were already unreadable before anything moved. Recorded so a
119    // resume reports those sessions as repaired rather than as new damage; the
120    // completion check itself rebuilds them from the event logs (§6.6).
121    plan.opaque.extend(journal.opaque());
122    // Every legacy owner lock, held until the last session directory has moved.
123    let owners = OwnerLocks::acquire(&plan, &legacy_sessions, &target)?;
124    journal.ensure_plan(&plan, &legacy_artifacts)?;
125    progress(&format!(
126        "{} session store migration… {} sessions in {}",
127        if resumed { "resuming" } else { "starting" },
128        plan.pending_total(),
129        legacy_root.display()
130    ));
131    let outcome = migrate(&legacy_root, &target, &plan, &mut journal, progress, owners)?;
132    progress(&format!(
133        "migrating session store… done ({} roots, {} children, {} artifacts)",
134        outcome.roots, outcome.children, outcome.artifacts
135    ));
136    Ok(Some(outcome))
137}

139/// Anything the completed job left behind in a legacy directory: a stray file
140/// means finalization is not finished, so the marker must not claim it is (B1).
141fn has_remnants(directory: &Path) -> Result<bool, SessionError> {
142    if !directory.exists() {
143        return Ok(false);
144    }
145    Ok(!directory_is_empty(directory)?)
146}

148/// Close the one finishing window that can outlive its work: verification
149/// passed, `.migrated` and the tombstone are down, the flat directories are
150/// gone, but a crash landed before `layout.json` was flipped to `complete`.
151/// Claims the completion the store already earned instead of stranding it in
152/// `in-progress` with no legacy directory left to resume from.
153fn finish_claimed_completion(
154    legacy_root: &Path,
155    target: &Path,
156    legacy_sessions: &Path,
157    legacy_artifacts: &Path,
158) -> Result<(), SessionError> {
159    if migration_state(target).as_deref() != Some(MIGRATION_IN_PROGRESS) {
160        return Ok(());
161    }
162    // Every finalization step has to have landed: `.migrated` is written only
163    // after verification, and the retirement of both flat directories is what
164    // `finish` reports as retired.
165    if !legacy_root.join(MIGRATED_MARKER_FILE).is_file()
166        || !legacy_root.join(TOMBSTONE_FILE).is_file()
167        || legacy_sessions.exists()
168        || legacy_artifacts.exists()
169    {
170        return Ok(());
171    }
172    sweep_litter(legacy_root, target)?;
173    write_layout_marker(target, MIGRATION_COMPLETE)?;
174    fsync_directory_tolerant(target)
175}

177/// Steps 2 through 6 of §6.3, resumable at any point.
178fn migrate(
179    legacy_root: &Path,
180    target: &Path,
181    plan: &Plan,
182    journal: &mut Journal,
183    progress: &dyn Fn(&str),
184    owners: OwnerLocks,
185) -> Result<Outcome, SessionError> {
186    let legacy_sessions = legacy_root.join(LEGACY_SESSIONS_DIR);
187    let legacy_artifacts = legacy_root.join(ARTIFACTS_DIR);
188
189    // 2. SCAFFOLD: reserve the v2 work dir and move the project-level files.
190    write_layout_marker(target, MIGRATION_IN_PROGRESS)?;
191    for file in PROJECT_FILES {
192        let source = legacy_root.join(file);
193        if source.is_file() && !target.join(file).exists() {
194            rename(&source, &target.join(file))?;
195        }
196        crash_after("project-file", Some(file))?;
197    }
198    crash_after("scaffold", None)?;
199
200    // 3. ROOTS first so every child has a `subagents/` parent to land in.
201    //    Every planned root goes through `move_session`, not only the still-flat
202    //    ones: the call also reconciles components a crash left half-moved (B2).
203    for root in &plan.roots {
204        let source = legacy_sessions.join(root.to_string());
205        let destination = target.join(root.to_string());
206        journal.consistent(&source, "roots", *root)?;
207        move_session(&source, &destination)?;
208        ensure_private_dir(&destination.join(ARTIFACTS_DIR))?;
209        ensure_private_dir(&destination.join(SUBAGENTS_DIR))?;
210        journal.done("roots", *root)?;
211        crash_after("root", Some(&root.to_string()))?;
212    }
213    progress(&format!(
214        "migrating session store… {} roots moved",
215        plan.pending_roots().count()
216    ));
217    crash_after("roots", None)?;
218
219    // 4. CHILDREN under their discovered root.
220    for (child, root) in plan.all_children() {
221        let source = legacy_sessions.join(child.to_string());
222        let destination = target
223            .join(root.to_string())
224            .join(SUBAGENTS_DIR)
225            .join(child.to_string());
226        journal.consistent(&source, "children", child)?;
227        move_session(&source, &destination)?;
228        journal.done("children", child)?;
229        crash_after("child", Some(&child.to_string()))?;
230    }
231    progress(&format!(
232        "migrating session store… {} child sessions moved",
233        plan.pending_children().count()
234    ));
235    // No session directory moves after this point, so the owner locks can go and
236    // the sidecar files that could not be renamed while held can.
237    owners.release()?;
238    crash_after("children", None)?;
239
240    // 5. ARTIFACTS: place each digest in the tree that references it, shared
241    // otherwise (§6.3 step 5, §6.5). The inventory is the durable plan record
242    // plus whatever the flat store holds now, so a resumed run reconciles blobs
243    // already moved as well as those still to place (B3).
244    let references = tree_references(target, plan)?;
245    let mut outcome = Outcome {
246        roots: plan.roots.len(),
247        children: plan.children.len(),
248        ..Outcome::default()
249    };
250    let mut ledger = BTreeSet::new();
251    let mut touched: BTreeSet<PathBuf> = BTreeSet::new();
252    // The flat artifact directory is the source parent of every rename below;
253    // one sync at the end of the batch makes the removals durable (M2).
254    touched.insert(legacy_artifacts.clone());
255    let mut names = journal.artifact_inventory()?;
256    names.extend(file_names(&legacy_artifacts)?);
257    for name in &names {
258        let placement = is_digest_name_common(name)
259            .then(|| placement_root(&references, name))
260            .flatten();
261        let destination = match placement {
262            Some(root) => {
263                outcome.artifacts += 1;
264                let trees = references.get(name.as_str()).into_iter().flatten();
265                for tree in trees.filter(|tree| **tree != root) {
266                    ledger.insert((name.clone(), *tree));
267                }
268                target.join(root.to_string()).join(ARTIFACTS_DIR).join(name)
269            }
270            None => {
271                outcome.shared_artifacts += 1;
272                target.join(SHARED_ARTIFACTS_DIR).join(name)
273            }
274        };
275        let source = legacy_artifacts.join(name);
276        place_blob(&source, &destination, &mut touched)?;
277        crash_after("artifact", Some(name))?;
278    }
279    // Renames are cheap; one sync per touched directory is enough to make the
280    // batch durable (§6.3 step 5), which a sync per blob would not be.
281    for directory in &touched {
282        // A flat artifact directory that never existed — a store that retained no
283        // output — has nothing to make durable, and neither has its parent.
284        fsync_directory_tolerant(directory)?;
285    }
286    if !ledger.is_empty() {
287        write_cross_ref_ledger(&target.join(CROSS_REFS_FILE), &ledger).map_err(|source| {
288            SessionError::Io {
289                path: target.join(CROSS_REFS_FILE),
290                source,
291            }
292        })?;
293        fsync_directory(target)?;
294        outcome.cross_refs = ledger.len();
295    }
296    progress(&format!(
297        "migrating session store… {} artifacts placed, {} shared",
298        outcome.artifacts, outcome.shared_artifacts
299    ));
300    crash_after("artifacts", None)?;
301
302    // 6. VERIFY against the durable plan, retire the flat project, and only then
303    // let the marker claim completion (B1). `before_verify` is the last crash
304    // point with a journal behind a fully moved store: everything after it is
305    // the finishing window, which the `retire`/`complete` points exercise.
306    crash_after("before_verify", None)?;
307    verify(target, &legacy_sessions, &legacy_artifacts, plan, journal)?;
308    // The journal is dropped as soon as the moves are known good, so a crash in
309    // what follows resumes from a verified plan rather than a stale record.
310    Journal::clear(legacy_root)?;
311    let retired = finish(legacy_root, target, &legacy_sessions, &legacy_artifacts)?;
312    crash_after("retire", None)?;
313    if retired {
314        write_layout_marker(target, MIGRATION_COMPLETE)?;
315        fsync_directory(target)?;
316    } else {
317        // Something an old binary dropped is still in the flat project. The
318        // marker stays in-progress so the next open finishes it: an incomplete
319        // finalization must never become authoritative (B1).
320        write_layout_marker(target, MIGRATION_IN_PROGRESS)?;
321        fsync_directory(target)?;
322        progress(
323            "session store migration… legacy project still holds files, finishing on next open",
324        );
325    }
326    // The marker writes above leave their own temporaries behind when a crash
327    // lands between the sync and the rename.
328    sweep_litter(legacy_root, target)?;
329    crash_after("complete", None)?;
330    Ok(outcome)
331}

333/// Move one artifact into its destination store, reconciling whatever a previous,
334/// interrupted run left behind (B3).
335fn place_blob(
336    source: &Path,
337    destination: &Path,
338    touched: &mut BTreeSet<PathBuf>,
339) -> Result<(), SessionError> {
340    let parent = destination
341        .parent()
342        .filter(|parent| !parent.as_os_str().is_empty())
343        .unwrap_or(destination);
344    if !parent.is_dir() {
345        ensure_private_dir(parent)?;
346    }
347    match (source.is_file(), destination.is_file()) {
348        (true, true) => {
349            // Both exist: the rename survived and its source-side removal did
350            // not. Keep the placed copy, and drop the duplicate only when the
351            // bytes agree — a mismatch is corruption, not a resume.
352            if read_digest(source)? == read_digest(destination)? {
353                fs::remove_file(source).map_err(|source_error| SessionError::Io {
354                    path: source.to_owned(),
355                    source: source_error,
356                })?;
357            } else {
358                return Err(SessionError::Migration(format!(
359                    "{} and {} hold different bytes for the same artifact",
360                    source.display(),
361                    destination.display()
362                )));
363            }
364        }
365        (true, false) => {
366            rename(source, destination)?;
367            touched.insert(parent.to_path_buf());
368        }
369        // Already placed by the run that crashed.
370        (false, true) => {}
371        // The blob is in neither store. It was there when the plan was recorded,
372        // so this is a loss, not a skip: refuse to call the store migrated.
373        (false, false) => {
374            return Err(SessionError::Migration(format!(
375                "artifact {} vanished from both {} and {}",
376                destination
377                    .file_name()
378                    .unwrap_or_default()
379                    .to_string_lossy(),
380                source.display(),
381                destination.display()
382            )));
383        }
384    }
385    Ok(())
386}

388/// Which root owns a digest's bytes.
389fn placement_root(
390    references: &BTreeMap<String, BTreeSet<SessionId>>,
391    digest: &str,
392) -> Option<SessionId> {
393    references
394        .get(digest)?
395        .iter()
396        .min_by_key(|tree| tree.to_string())
397        .copied()
398}

400/// Digest → set of trees whose logs reference it, from the freshly moved logs.
401///
402/// Fail-closed: an unreadable log would make its blobs look unreferenced and
403/// send them to the shared store while the migration still claimed success, so
404/// the scan error aborts the job and leaves `.migrating` for the next open.
405fn tree_references(
406    target: &Path,
407    plan: &Plan,
408) -> Result<BTreeMap<String, BTreeSet<SessionId>>, SessionError> {
409    let mut references: BTreeMap<String, BTreeSet<SessionId>> = BTreeMap::new();
410    for root in &plan.roots {
411        let directory = target.join(root.to_string());
412        let mut live = HashSet::new();
413        let root_log = directory.join(EVENTS_FILE);
414        absorb(
415            scan_artifact_references_in_log(&root_log),
416            &root_log,
417            &mut live,
418        )?;
419        for child in plan.children_of(*root) {
420            let child_log = directory
421                .join(SUBAGENTS_DIR)
422                .join(child.to_string())
423                .join(EVENTS_FILE);
424            absorb(
425                scan_artifact_references_in_log(&child_log),
426                &child_log,
427                &mut live,
428            )?;
429        }
430        for digest in live {
431            references.entry(digest).or_default().insert(*root);
432        }
433    }
434    Ok(references)
435}

437/// Fold one log's references into the set. A planned session must have a
438/// readable event log: the low-level event reader can represent a missing file
439/// as an empty byte stream, but `EventLog::open_read_only` rejects that stream
440/// because it has no `SessionCreated` record. Migration therefore fails closed
441/// rather than silently treating a missing session history as empty.
442fn absorb(
443    result: std::io::Result<HashSet<String>>,
444    path: &Path,
445    into: &mut HashSet<String>,
446) -> Result<(), SessionError> {
447    match result {
448        Ok(found) => {
449            into.extend(found);
450            Ok(())
451        }
452        Err(source) => Err(SessionError::Io {
453            path: path.to_owned(),
454            source,
455        }),
456    }
457}

459/// §6.6. Any mismatch aborts before the completion marker is written, leaving
460/// `.migrating` and an in-progress marker for the next open to resume.
461fn verify(
462    target: &Path,
463    legacy_sessions: &Path,
464    legacy_artifacts: &Path,
465    plan: &Plan,
466    journal: &Journal,
467) -> Result<(), SessionError> {
468    let stranded: Vec<String> = directory_names(legacy_sessions)?
469        .into_iter()
470        .filter(|name| name.parse::<SessionId>().is_ok())
471        .collect();
472    if !stranded.is_empty() {
473        return Err(SessionError::Migration(format!(
474            "{} legacy sessions were not moved",
475            stranded.len()
476        )));
477    }
478    let mut placed_children: HashMap<SessionId, usize> = HashMap::new();
479    for root in &plan.roots {
480        let directory = target.join(root.to_string());
481        check_placed(&directory, *root, plan)?;
482        if !directory.join(ARTIFACTS_DIR).is_dir() || !directory.join(SUBAGENTS_DIR).is_dir() {
483            return Err(SessionError::Migration(format!(
484                "{} is missing artifacts/ or subagents/",
485                directory.display()
486            )));
487        }
488        let subagents = directory.join(SUBAGENTS_DIR);
489        for name in directory_names(&subagents)? {
490            let Ok(child) = name.parse::<SessionId>() else {
491                continue;
492            };
493            *placed_children.entry(child).or_default() += 1;
494            check_placed(&subagents.join(&name), child, plan)?;
495        }
496    }
497    for (child, root) in &plan.children {
498        match placed_children.get(child) {
499            Some(1) => Ok(()),
500            Some(count) => Err(SessionError::Migration(format!(
501                "child {child} landed in {count} roots, expected one"
502            ))),
503            None => Err(SessionError::Migration(format!(
504                "child {child} is not under root {root}",
505            ))),
506        }?;
507    }
508    // Artifact inventory: every digest the flat store held when the plan was
509    // recorded must now sit in exactly one place, with bytes matching its name.
510    // Counting this invocation's moves would let a lost blob pass (§6.6, B3).
511    let inventory = journal.artifact_inventory()?;
512    let placed = placement_inventory(target, plan)?;
513    let still_flat = file_names(legacy_artifacts)?;
514    let still_flat: BTreeSet<&str> = still_flat.iter().map(String::as_str).collect();
515    for name in &inventory {
516        match placed.get(name).map(Vec::as_slice) {
517            None if still_flat.contains(&name.as_str()) => {
518                return Err(SessionError::Migration(format!(
519                    "artifact {name} was never placed and is still in the flat store"
520                )));
521            }
522            None => {
523                return Err(SessionError::Migration(format!(
524                    "artifact {name} is in neither the flat store nor its destination"
525                )));
526            }
527            Some([path]) => {
528                if is_digest_name_common(name) && read_digest(path)? != name.as_str() {
529                    return Err(SessionError::Migration(format!(
530                        "{} does not hash to its name",
531                        path.display()
532                    )));
533                }
534            }
535            Some(paths) => {
536                return Err(SessionError::Migration(format!(
537                    "artifact {name} sits in {} places",
538                    paths.len()
539                )));
540            }
541        }
542    }
543    if let Some(leftover) = still_flat.iter().find(|name| inventory.contains(**name)) {
544        return Err(SessionError::Migration(format!(
545            "artifact {leftover} is still in the flat store"
546        )));
547    }
548    Ok(())
549}

551/// Every artifact placement the migrated store can serve: the shared store plus
552/// each tree's own store, keyed by file name.
553fn placement_inventory(
554    target: &Path,
555    plan: &Plan,
556) -> Result<BTreeMap<String, Vec<PathBuf>>, SessionError> {
557    let mut placed: BTreeMap<String, Vec<PathBuf>> = BTreeMap::new();
558    let mut stores = vec![target.join(SHARED_ARTIFACTS_DIR)];
559    for root in &plan.roots {
560        stores.push(target.join(root.to_string()).join(ARTIFACTS_DIR));
561    }
562    for store in stores {
563        for name in file_names(&store)? {
564            placed
565                .entry(name.clone())
566                .or_default()
567                .push(store.join(&name));
568        }
569    }
570    Ok(placed)
571}

573/// One placed session (§6.6 check 2): scaffold present, and a metadata cache
574/// that parses and names this session.
575///
576/// A cache that was already unreadable in the flat store grants no exemption:
577/// the event log is authoritative, so an unusable cache is rebuilt from the log
578/// and revalidated. A session that neither has a usable cache nor can rebuild
579/// one aborts the migration rather than being declared complete.
580fn check_placed(directory: &Path, id: SessionId, plan: &Plan) -> Result<(), SessionError> {
581    if !directory.is_dir() {
582        return Err(SessionError::Migration(format!(
583            "{} is missing after the move",
584            directory.display()
585        )));
586    }
587    if metadata_is_current(directory, id)? {
588        return Ok(());
589    }
590    if plan.opaque.contains(&id) {
591        eprintln!(
592            "migration: {} arrived without a usable cache; rebuilding it from the event log",
593            directory.display()
594        );
595    }
596    rebuild_metadata(directory, id)?;
597    if metadata_is_current(directory, id)? {
598        return Ok(());
599    }
600    Err(SessionError::Migration(format!(
601        "{} still has no metadata cache naming {id} after rebuilding from its event log",
602        directory.display()
603    )))
604}

606/// Whether the cache in `directory` parses and declares `id`.
607fn metadata_is_current(directory: &Path, id: SessionId) -> Result<bool, SessionError> {
608    let path = meta_path(directory);
609    let text = match fs::read_to_string(&path) {
610        Ok(text) => text,
611        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(false),
612        Err(source) => {
613            return Err(SessionError::Io { path, source });
614        }
615    };
616    // An unparseable cache is a hole in the cache, not in the session: the log
617    // below rebuilds it.
618    match serde_json::from_str::<SessionMeta>(&text) {
619        Ok(meta) => Ok(meta.session_id == id),
620        Err(_) => Ok(false),
621    }
622}

624/// Replace the cache with one folded from the session's own event log, durably
625/// and before anything is allowed to call the store migrated.
626fn rebuild_metadata(directory: &Path, id: SessionId) -> Result<(), SessionError> {
627    let cache = directory.join(SESSION_META_FILE);
628    let events = directory.join(EVENTS_FILE);
629    if !events.is_file() {
630        return Err(SessionError::Migration(format!(
631            "{} has no usable metadata cache and no {} to rebuild one from",
632            directory.display(),
633            EVENTS_FILE
634        )));
635    }
636    let meta = projection(EventLog::open_read_only(events.clone(), id)?)?.meta;
637    if meta.session_id != id {
638        return Err(SessionError::Migration(format!(
639            "{} describes session {}, not {id}",
640            directory.display(),
641            meta.session_id
642        )));
643    }
644    let bytes = serde_json::to_vec_pretty(&meta).map_err(|source| SessionError::Json {
645        path: cache.clone(),
646        source,
647    })?;
648    write_atomically(&cache, &bytes, directory)?;
649    if cache != directory.join(LEGACY_SESSION_META_FILE) {
650        remove_file(&directory.join(LEGACY_SESSION_META_FILE))?;
651    }
652    Ok(())
653}

655/// Write the tombstone and drop the emptied legacy directories. `false` means
656/// something is still in the flat project, so completion must not be claimed (B1).
657fn finish(
658    legacy_root: &Path,
659    target: &Path,
660    legacy_sessions: &Path,
661    legacy_artifacts: &Path,
662) -> Result<bool, SessionError> {
663    write_atomically(
664        &legacy_root.join(MIGRATED_MARKER_FILE),
665        format!("{}\n", target.display()).as_bytes(),
666        legacy_root,
667    )?;
668    write_tombstone(legacy_root, target)?;
669    let mut retired = true;
670    for directory in [legacy_sessions, legacy_artifacts] {
671        if !directory.is_dir() {
672            continue;
673        }
674        if directory_is_empty(directory)? {
675            match fs::remove_dir(directory) {
676                Ok(()) => {}
677                // A racing removal is the outcome we wanted; anything else is not.
678                Err(error) if error.kind() != std::io::ErrorKind::NotFound => {
679                    retired = false;
680                }
681                Err(_) => {}
682            }
683        } else {
684            retired = false;
685        }
686    }
687    fsync_directory(legacy_root).map_err(SessionError::Event)?;
688    Ok(retired)
689}

712/// The plan: which legacy sessions are roots, and where each child lands.
713#[derive(Debug, Default)]
714struct Plan {
715    roots: Vec<SessionId>,
716    children: BTreeMap<SessionId, SessionId>,
717    /// Sessions whose metadata cache could not be parsed. They become roots and
718    /// their artifacts are treated as unreferenced (§6.5).
719    opaque: BTreeSet<SessionId>,
720    /// Sessions still sitting in the flat layout. A resumed run plans over the
721    /// whole tree but only moves these.
722    pending: BTreeSet<SessionId>,
723}
724
725impl Plan {
726    fn pending_total(&self) -> usize {
727        self.pending.len()
728    }
729
730    fn pending_roots(&self) -> impl Iterator<Item = SessionId> + '_ {
731        self.roots
732            .iter()
733            .copied()
734            .filter(|root| self.pending.contains(root))
735    }
736
737    fn pending_children(&self) -> impl Iterator<Item = (SessionId, SessionId)> + '_ {
738        self.children
739            .iter()
740            .filter(|(child, _)| self.pending.contains(child))
741            .map(|(child, root)| (*child, *root))
742    }
743
744    /// Where every planned session belongs in the v2 layout, whether or not it
745    /// has already moved.
746    fn placements(&self, target: &Path) -> Vec<(SessionId, PathBuf)> {
747        self.roots
748            .iter()
749            .map(|root| (*root, target.join(root.to_string())))
750            .chain(self.children.iter().map(|(child, root)| {
751                (
752                    *child,
753                    target
754                        .join(root.to_string())
755                        .join(SUBAGENTS_DIR)
756                        .join(child.to_string()),
757                )
758            }))
759            .collect()
760    }
761
762    /// Every planned child, pending or already moved: `move_session` reconciles
763    /// components a crash left half-moved, so it must run for all of them.
764    fn all_children(&self) -> impl Iterator<Item = (SessionId, SessionId)> + '_ {
765        self.children.iter().map(|(child, root)| (*child, *root))
766    }
767
768    fn children_of(&self, root: SessionId) -> impl Iterator<Item = SessionId> + '_ {
769        self.children
770            .iter()
771            .filter(move |(_, parent)| **parent == root)
772            .map(|(child, _)| *child)
773    }
774
775    /// Plans over the legacy directories *and* whatever a previous, interrupted
776    /// run already moved, so a resume places children under the same roots.
777    fn discover(legacy_sessions: &Path, target: &Path) -> Result<Self, SessionError> {
778        let mut sessions: BTreeMap<SessionId, Option<SessionMeta>> = BTreeMap::new();
779        let mut pending: BTreeSet<SessionId> = BTreeSet::new();
780        for (directory, is_pending) in [(legacy_sessions, true), (target, false)] {
781            for name in directory_names(directory)? {
782                let Ok(id) = name.parse::<SessionId>() else {
783                    continue;
784                };
785                let meta = read_metadata(&directory.join(&name));
786                if meta.is_none() && is_pending {
787                    eprintln!("migration: session {id} has no readable metadata cache");
788                }
789                sessions.entry(id).or_insert_with(|| meta.clone());
790                if is_pending {
791                    pending.insert(id);
792                }
793                // Children a previous run already filed under their root count
794                // as known placements, so a resume does not orphan them.
795                if !is_pending {
796                    let subagents = directory.join(&name).join(SUBAGENTS_DIR);
797                    for child in directory_names(&subagents)? {
798                        let Ok(child) = child.parse::<SessionId>() else {
799                            continue;
800                        };
801                        let meta = read_metadata(&subagents.join(child.to_string()));
802                        sessions.entry(child).or_insert_with(|| meta.clone());
803                    }
804                }
805            }
806        }
807        let mut plan = Plan {
808            pending,
809            ..Plan::default()
810        };
811        for (id, meta) in &sessions {
812            if meta.is_none() {
813                plan.opaque.insert(*id);
814            }
815            match delegated_root(meta.as_ref()).filter(|claimed| *claimed != *id) {
816                // A child lands in the directory of its root, following the
817                // origin chain to a session that is itself a root.
818                Some(claimed) if sessions.contains_key(&claimed) => {
819                    let root = resolve_root(&sessions, claimed);
820                    plan.children.insert(*id, root);
821                }
822                // A root, or an orphan whose root session is gone: a deleted
823                // parent must not strand its transcript (§6.3 step 1).
824                _ => plan.roots.push(*id),
825            }
826        }
827        Ok(plan)
828    }
829}

831/// `root_session_id` of a delegated session (§6.3 step 1).
832fn delegated_root(meta: Option<&SessionMeta>) -> Option<SessionId> {
833    match meta?.origin {
834        SessionOrigin::Root => None,
835        SessionOrigin::Delegated {
836            root_session_id, ..
837        } => Some(root_session_id),
838    }
839}
840
841/// Follow `root_session_id` links until a session that is not itself delegated
842/// is reached, so nested children still land in one tree. Cycles promote.
843fn resolve_root(
844    sessions: &BTreeMap<SessionId, Option<SessionMeta>>,
845    start: SessionId,
846) -> SessionId {
847    let mut candidate = start;
848    let mut seen: BTreeSet<SessionId> = BTreeSet::from([start]);
849    while let Some(claimed) = sessions
850        .get(&candidate)
851        .and_then(|meta| delegated_root(meta.as_ref()))
852    {
853        if claimed == candidate || !sessions.contains_key(&claimed) || !seen.insert(claimed) {
854            break;
855        }
856        candidate = claimed;
857    }
858    candidate
859}

870/// What the store looked like before the first rename, recorded durably so a
871/// resumed run can reconcile and verify against the *original* state (B3, W3).
872#[derive(Clone, Debug, Default, serde::Deserialize, serde::Serialize)]
873struct PlanRecord {
874    /// Every legacy session directory, by id.
875    #[serde(default)]
876    roots: Vec<String>,
877    /// `(child, root)` placements as discovered before any move.
878    #[serde(default)]
879    children: Vec<(String, String)>,
880    /// Sessions whose metadata cache was already unreadable (W3).
881    #[serde(default)]
882    opaque: Vec<String>,
883    /// Every file name in the flat artifact store (B3).
884    #[serde(default)]
885    artifacts: Vec<String>,
886}
887
888/// One journal line.
889#[derive(Debug, serde::Deserialize, serde::Serialize)]
890#[serde(tag = "t", rename_all = "kebab-case")]
891enum Record {
892    Plan(PlanRecord),
893    Moved { phase: String, id: String },
894}
895
896impl Journal {
897    fn load(legacy_root: &Path) -> Self {
898        let mut journal = Self {
899            legacy_root: legacy_root.to_path_buf(),
900            ..Self::default()
901        };
902        let Ok(bytes) = fs::read(legacy_root.join(JOURNAL_FILE)) else {
903            return journal;
904        };
905        for line in String::from_utf8_lossy(&bytes).lines() {
906            // A torn final line is a crash mid-append: the record it describes
907            // never completed, so skipping it is the resume.
908            let Ok(record) = serde_json::from_str::<Record>(line) else {
909                continue;
910            };
911            match record {
912                Record::Plan(plan) => journal.plan = Some(plan),
913                Record::Moved { phase, id } => {
914                    journal.moved.insert(format!("{phase}:{id}"));
915                }
916            }
917        }
918        journal
919    }
920
921    /// Whether a previous, interrupted run left a plan behind.
922    fn is_resumed(&self) -> bool {
923        self.plan.is_some()
924    }
925
926    /// Append the pre-move snapshot once, durably, before anything moves.
927    fn ensure_plan(&mut self, plan: &Plan, legacy_artifacts: &Path) -> Result<(), SessionError> {
928        if self.plan.is_some() {
929            return Ok(());
930        }
931        let record = PlanRecord {
932            roots: plan.roots.iter().map(ToString::to_string).collect(),
933            children: plan
934                .children
935                .iter()
936                .map(|(child, root)| (child.to_string(), root.to_string()))
937                .collect(),
938            opaque: plan.opaque.iter().map(ToString::to_string).collect(),
939            artifacts: file_names(legacy_artifacts)?,
940        };
941        self.append(&Record::Plan(record.clone()))?;
942        self.plan = Some(record);
943        Ok(())
944    }
945
946    /// The original flat artifact inventory.
947    fn artifact_inventory(&self) -> Result<BTreeSet<String>, SessionError> {
948        Ok(self
949            .plan
950            .as_ref()
951            .map(|record| record.artifacts.iter().cloned().collect())
952            .unwrap_or_default())
953    }
954
955    /// Sessions recorded as having unreadable metadata before any move.
956    fn opaque(&self) -> BTreeSet<SessionId> {
957        self.plan
958            .as_ref()
959            .map(|record| {
960                record
961                    .opaque
962                    .iter()
963                    .filter_map(|name| name.parse::<SessionId>().ok())
964                    .collect()
965            })
966            .unwrap_or_default()
967    }
968
969    /// Rewind to the recorded plan: keep the plan, drop every move record.
970    ///
971    /// Verification clears the journal, so a surviving record describes a phase
972    /// that did not finish. Once that phase's work has landed the record is stale,
973    /// and `reconcile_move` has nothing left to trust: the phase has to run again
974    /// because something put the directory back where the plan says it already
975    /// moved (an interrupted `rename_metadata`, a file an old binary dropped into
976    /// the retired flat store). Rewinding makes a correction after the move
977    /// resumable instead of a permanent refusal — `move_session` is idempotent and
978    /// the durable plan still backs the artifact check. Returns how many records
979    /// were dropped.
980    fn forget_moves(&mut self) -> Result<usize, SessionError> {
981        let path = self.legacy_root.join(JOURNAL_FILE);
982        let Ok(text) = fs::read_to_string(&path) else {
983            self.moved.clear();
984            return Ok(0);
985        };
986        let mut kept = String::new();
987        let mut dropped = 0;
988        for line in text.lines() {
989            let parsed = serde_json::from_str::<serde_json::Value>(line).ok();
990            if matches!(&parsed, Some(value) if value["t"] == "moved") {
991                dropped += 1;
992                continue;
993            }
994            kept.push_str(line);
995            kept.push('\n');
996        }
997        if dropped > 0 {
998            write_atomically(&path, kept.as_bytes(), &self.legacy_root)?;
999            self.moved.clear();
1000        }
1001        Ok(dropped)
1002    }
1003
1004    /// Whether the journal's own record of a move contradicts the store.
1005    fn contradicts(&self, source: &Path, phase: &str, id: SessionId) -> bool {
1006        self.moved.contains(&format!("{phase}:{id}")) && source.is_dir()
1007    }
1008
1009    fn done(&mut self, phase: &str, id: SessionId) -> Result<(), SessionError> {
1010        self.append(&Record::Moved {
1011            phase: phase.to_owned(),
1012            id: id.to_string(),
1013        })?;
1014        self.moved.insert(format!("{phase}:{id}"));
1015        Ok(())
1016    }
1017
1018    fn append(&self, record: &Record) -> Result<(), SessionError> {
1019        let path = self.legacy_root.join(JOURNAL_FILE);
1020        let mut line = serde_json::to_vec(record).map_err(|source| SessionError::Json {
1021            path: path.clone(),
1022            source,
1023        })?;
1024        line.push(b'\n');
1025        let mut file = fs::OpenOptions::new()
1026            .append(true)
1027            .create(true)
1028            .open(&path)
1029            .map_err(|source| SessionError::Io {
1030                path: path.clone(),
1031                source,
1032            })?;
1033        file.write_all(&line)
1034            .and_then(|()| file.sync_all())
1035            .map_err(|source| SessionError::Io {
1036                path: path.clone(),
1037                source,
1038            })?;
1039        drop(file);
1040        fsync_directory(&self.legacy_root)?;
1041        Ok(())
1042    }
1043
1044    /// A resume must not find a session the journal already finished still
1045    /// sitting in the flat layout: that means two migrations raced.
1046    fn consistent(
1047        &mut self,
1048        source: &Path,
1049        phase: &str,
1050        id: SessionId,
1051    ) -> Result<(), SessionError> {
1052        if self.contradicts(source, phase, id) {
1053            // The move record is history that stopped being true; rewind to the
1054            // plan and let this phase run again over what is really there.
1055            if self.forget_moves()? == 0 || self.contradicts(source, phase, id) {
1056                return Err(SessionError::Migration(format!(
1057                    "journal records {id} as moved, but {} still exists",
1058                    source.display()
1059                )));
1060            }
1061        }
1062        Ok(())
1063    }
1064
1065    fn clear(legacy_root: &Path) -> Result<(), SessionError> {
1066        match fs::remove_file(legacy_root.join(JOURNAL_FILE)) {
1067            Ok(()) => Ok(()),
1068            Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(()),
1069            Err(source) => Err(SessionError::Io {
1070                path: legacy_root.join(JOURNAL_FILE),
1071                source,
1072            }),
1073        }
1074    }
1075}

1077/// Every legacy session's owner lock, held from the first rename until the last
1078/// session directory has landed, so no live process can slip in behind the probe
1079/// (§6.3 step 0, M1).
1080struct OwnerLocks {
1081    held: Vec<HeldLock>,
1082    /// Windows files `owner.lock` *beside* its session directory. The open
1083    /// handle denies pathname replacement (share mode `READ | WRITE`), so the
1084    /// sidecar cannot move with its directory; `release` relocates it after the
1085    /// handle closes. Unix keeps the lock *inside* the directory, so its list is
1086    /// always empty there.
1087    sidecars: Vec<(PathBuf, PathBuf)>,
1088    /// Windows sidecars left beside a session directory by a crash between the
1089    /// directory move and `release`. The session already lives at its
1090    /// destination, so the duplicate is litter that would keep the retired flat
1091    /// directory non-empty; `release` removes it once the locks are dropped.
1092    stale: Vec<PathBuf>,
1093}
1094
1095impl OwnerLocks {
1096    fn acquire(plan: &Plan, legacy_sessions: &Path, target: &Path) -> Result<Self, SessionError> {
1097        let mut locks = Self {
1098            held: Vec::new(),
1099            sidecars: Vec::new(),
1100            stale: Vec::new(),
1101        };
1102        for (id, destination) in plan.placements(target) {
1103            let flat = legacy_sessions.join(id.to_string());
1104            let owned_flat = flat.is_dir();
1105            let directory = if owned_flat { &flat } else { &destination };
1106            match try_acquire(directory) {
1107                // A stale lock is ours to take; it moves with its session.
1108                Ok(SessionOwnership::Owned(held)) => {
1109                    locks.held.push(held);
1110                    let held_lock = owner_lock_path(directory);
1111                    // A sidecar (parent outside the directory) only exists on
1112                    // Windows; the Unix lock lives inside the directory and is
1113                    // moved by the directory rename itself.
1114                    if held_lock.parent() != Some(directory.as_ref()) {
1115                        if owned_flat {
1116                            if held_lock.is_file() {
1117                                locks
1118                                    .sidecars
1119                                    .push((held_lock, owner_lock_path(&destination)));
1120                            }
1121                        } else {
1122                            // The directory is already at its destination, so a
1123                            // sidecar still at the flat path is a crash orphan.
1124                            let orphan = owner_lock_path(&flat);
1125                            if orphan.is_file() {
1126                                locks.stale.push(orphan);
1127                            }
1128                        }
1129                    }
1130                }
1131                Ok(SessionOwnership::Foreign) => {
1132                    return Err(SessionError::Migration(format!(
1133                        "session {id} is owned by a live process; quit cookie and retry"
1134                    )));
1135                }
1136                Err(source) => {
1137                    return Err(SessionError::Io {
1138                        path: owner_lock_path(directory),
1139                        source,
1140                    });
1141                }
1142            }
1143        }
1144        Ok(locks)
1145    }
1146
1147    /// Drop every lock, then move the sidecar files that could not move with it
1148    /// and drop the crash orphans a resume found. This is the *only* place a
1149    /// sidecar moves: [`move_session`] must not attempt it while the handle is
1150    /// open, because the held handle denies pathname replacement.
1151    fn release(self) -> Result<(), SessionError> {
1152        let Self {
1153            held,
1154            sidecars,
1155            stale,
1156        } = self;
1157        drop(held);
1158        for (source, destination) in sidecars {
1159            if source.is_file() {
1160                replace_path(&source, &destination).map_err(|error| SessionError::Io {
1161                    path: destination.clone(),
1162                    source: error,
1163                })?;
1164            }
1165        }
1166        for path in stale {
1167            remove_file(&path)?;
1168        }
1169        Ok(())
1170    }
1171}

1219/// Exclusion lock for the whole procedure (§6.3 step 0).
1220struct MigrationLock {
1221    file: fs::File,
1222}
1223
1224impl MigrationLock {
1225    fn acquire(legacy_root: &Path) -> Result<Self, SessionError> {
1226        let path = legacy_root.join(MIGRATION_LOCK_FILE);
1227        let file = fs::OpenOptions::new()
1228            .read(true)
1229            .write(true)
1230            .create(true)
1231            // The lock file's contents are never read; an existing one stays.
1232            .truncate(false)
1233            .open(&path)
1234            .map_err(|source| SessionError::Io {
1235                path: path.clone(),
1236                source,
1237            })?;
1238        match file.try_lock_exclusive() {
1239            Ok(()) => Ok(Self { file }),
1240            Err(contention) => Err(SessionError::Migration(format!(
1241                "another cookie process is migrating {}; quit it and retry ({contention})",
1242                legacy_root.display()
1243            ))),
1244        }
1245    }
1246}

1255/// Move one session directory, then reconcile each of its components
1256/// independently. A crash can land between the directory rename and the metadata
1257/// rename, so re-running this repairs whatever the destination is missing (B2).
1258fn move_session(source: &Path, destination: &Path) -> Result<(), SessionError> {
1259    if !destination.is_dir() {
1260        if !source.is_dir() {
1261            return Err(SessionError::Migration(format!(
1262                "session directory {} vanished mid-migration",
1263                source.display()
1264            )));
1265        }
1266        let parent = destination
1267            .parent()
1268            .filter(|parent| !parent.as_os_str().is_empty())
1269            .unwrap_or(destination);
1270        ensure_private_dir(parent)?;
1271        rename(source, destination)?;
1272    }
1273    rename_metadata(destination)?;
1274    // The Windows ownership sidecar is deliberately *not* moved here: the
1275    // migration holds its handle open for the whole procedure and that handle
1276    // denies pathname replacement (`ownership.rs` share mode is `READ | WRITE`).
1277    // `OwnerLocks::release` performs the move exactly once, after the handle is
1278    // closed and after the session directory is durable. Unix keeps the lock
1279    // inside the directory, so it is carried by the directory rename above.
1280    Ok(())
1281}

1283/// `meta.json` → `metadata`, in either layout position, without ever losing a
1284/// cache: two copies are only acceptable when they are the same bytes.
1285fn rename_metadata(directory: &Path) -> Result<(), SessionError> {
1286    let legacy = directory.join(LEGACY_SESSION_META_FILE);
1287    if !legacy.is_file() {
1288        return Ok(());
1289    }
1290    let current = directory.join(SESSION_META_FILE);
1291    if current.is_file() {
1292        if read_digest(&legacy)? != read_digest(&current)? {
1293            return Err(SessionError::Migration(format!(
1294                "{} has two different metadata caches",
1295                directory.display()
1296            )));
1297        }
1298        return remove_file(&legacy);
1299    }
1300    rename(&legacy, &current)
1301}

1324/// Rename and make *both* parents durable: syncing only the destination can roll
1325/// the removal of the source entry back, leaving the item in neither place (M2).
1326fn rename(source: &Path, destination: &Path) -> Result<(), SessionError> {
1327    match fs::rename(source, destination) {
1328        Ok(()) => {
1329            fsync_directory(&parent_of(destination))?;
1330            let source_parent = parent_of(source);
1331            if source_parent != parent_of(destination) {
1332                fsync_directory_tolerant(&source_parent)?;
1333            }
1334            Ok(())
1335        }
1336        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(()),
1337        Err(source_error) => Err(SessionError::Io {
1338            path: destination.to_owned(),
1339            source: source_error,
1340        }),
1341    }
1342}

1344/// A parent that no longer exists has nothing left to sync.
1345fn fsync_directory_tolerant(directory: &Path) -> Result<(), SessionError> {
1346    match fsync_directory(directory) {
1347        Ok(()) => Ok(()),
1348        // A parent that no longer exists has nothing left to sync.
1349        Err(crate::events::EventLogError::Io { source, .. })
1350            if source.kind() == std::io::ErrorKind::NotFound =>
1351        {
1352            Ok(())
1353        }
1354        Err(source) => Err(source.into()),
1355    }
1356}

1404/// Write `path` through a temporary in its own directory, then fsync.
1405///
1406/// The temporary name is unique per call and stale temporaries for this target
1407/// are removed first: a crash between the sync and the rename used to leave a
1408/// fixed-name temporary behind, whose `create_new` then failed for every later
1409/// attempt, wedging the migration permanently.
1410fn write_atomically(path: &Path, bytes: &[u8], parent: &Path) -> Result<(), SessionError> {
1411    let name = path
1412        .file_name()
1413        .map(|name| name.to_string_lossy().into_owned())
1414        .unwrap_or_default();
1415    let temporary = parent.join(format!(".{name}.{}.migration-tmp", Uuid::now_v7()));
1416    remove_stale_temparies(parent, &name)?;
1417    let mut file = fs::File::options()
1418        .write(true)
1419        .create_new(true)
1420        .open(&temporary)
1421        .map_err(|source| SessionError::Io {
1422            path: temporary.clone(),
1423            source,
1424        })?;
1425    let written = file
1426        .write_all(bytes)
1427        .and_then(|()| file.sync_all())
1428        .map_err(|source| SessionError::Io {
1429            path: temporary.clone(),
1430            source,
1431        });
1432    drop(file);
1433    if let Err(failure) = written {
1434        let _ = fs::remove_file(&temporary);
1435        return Err(failure);
1436    }
1437    match replace_path(&temporary, path) {
1438        Ok(()) => {}
1439        Err(source) => {
1440            let _ = fs::remove_file(&temporary);
1441            return Err(SessionError::Io {
1442                path: path.to_owned(),
1443                source,
1444            });
1445        }
1446    }
1447    fsync_directory(parent)?;
1448    Ok(())
1449}

1452/// already exist.
1453///
1454/// Rust's `fs::rename` already passes `MOVEFILE_REPLACE_EXISTING` on Windows,
1455/// so replacement is not what is missing here. What the Windows path adds is
1456/// `MOVEFILE_WRITE_THROUGH` (the marker/temporary ordering stays durable) plus a
1457/// bounded retry on transient `ERROR_ACCESS_DENIED` (5) /
1458/// `ERROR_SHARING_VIOLATION` (32) when an antivirus or indexer briefly holds the
1459/// just-written temporary or the destination. The Unix path is a plain rename.

1462/// Windows migration failures; those come from renaming a held owner-lock
1463/// sidecar (see `OwnerLocks`).

1474/// Sweep every atomic-write temporary the migration procedure can leave in the
1475/// legacy project directory or beside the v2 layout marker.
1476///
1477/// A crash between [`write_atomically`]'s sync and its rename leaves a
1478/// `.<name>.<uuid>.migration-tmp` behind; the uuid makes the name one no other
1479/// writer can hold, so a temporary that outlived its process cannot belong to a
1480/// live write. Left in place it counts as a remnant of the flat project, and the
1481/// store never gets to claim its migration is done.
1482fn sweep_litter(legacy_root: &Path, target: &Path) -> Result<(), SessionError> {
1483    remove_stale_temparies(legacy_root, JOURNAL_FILE)?;
1484    remove_stale_temparies(legacy_root, MIGRATED_MARKER_FILE)?;
1485    remove_stale_temparies(legacy_root, TOMBSTONE_FILE)?;
1486    remove_stale_temparies(target, LAYOUT_MARKER_FILE)?;
1487    Ok(())
1488}

1490/// Drop temporaries an interrupted write left behind for this same target.
1491fn remove_stale_temparies(parent: &Path, name: &str) -> Result<(), SessionError> {
1492    let prefix = format!(".{name}.");
1493    let Ok(entries) = fs::read_dir(parent) else {
1494        return Ok(());
1495    };
1496    for entry in entries {
1497        let entry = match entry {
1498            Ok(entry) => entry,
1499            // An unlistable entry is not a temporary we can reason about; the
1500            // write below does not depend on it.
1501            Err(_) => continue,
1502        };
1503        let file_name = entry.file_name().to_string_lossy().into_owned();
1504        if file_name.starts_with(&prefix) && file_name.ends_with(".migration-tmp") {
1505            match fs::remove_file(entry.path()) {
1506                Ok(()) => {}
1507                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
1508                Err(source) => {
1509                    return Err(SessionError::Io {
1510                        path: entry.path(),
1511                        source,
1512                    });
1513                }
1514            }
1515        }
1516    }
1517    Ok(())
1518}

1543fn named(directory: &Path, want_directory: bool) -> Result<Vec<String>, SessionError> {
1544    let mut names = Vec::new();
1545    let mut entries = match fs::read_dir(directory) {
1546        Ok(entries) => entries,
1547        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(names),
1548        Err(source) => {
1549            return Err(SessionError::Io {
1550                path: directory.to_owned(),
1551                source,
1552            });
1553        }
1554    };
1555    while let Some(entry) = entries
1556        .next()
1557        .transpose()
1558        .map_err(|source| SessionError::Io {
1559            path: directory.to_owned(),
1560            source,
1561        })?
1562    {
1563        let is_dir = entry
1564            .file_type()
1565            .map(|file_type| file_type.is_dir())
1566            .unwrap_or(false);
1567        if is_dir == want_directory {
1568            names.push(entry.file_name().to_string_lossy().into_owned());
1569        }
1570    }
1571    names.sort();
1572    Ok(names)
1573}

1575fn directory_is_empty(directory: &Path) -> Result<bool, SessionError> {
1576    let mut entries = match fs::read_dir(directory) {
1577        Ok(entries) => entries,
1578        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(true),
1579        Err(source) => {
1580            return Err(SessionError::Io {
1581                path: directory.to_owned(),
1582                source,
1583            });
1584        }
1585    };
1586    // An entry that cannot be read means emptiness is unprovable, which is not
1587    // the same as empty: claiming otherwise would finalize an unknown state.
1588    match entries.next() {
1589        None => Ok(true),
1590        Some(Ok(_)) => Ok(false),
1591        Some(Err(source)) => Err(SessionError::Io {
1592            path: directory.to_owned(),
1593            source,
1594        }),
1595    }

2617    /// Everything the startup passes fold from, per session: its durable payloads
2618    /// in log order, plus the store-wide set of restart-stable tree grants the
2619    /// approval store re-installs (§4.3). Comparing the two before and after a
2620    /// move is the registry/approval equivalence §8.2 asks for: a migration
2621    /// renames log files and may not change what they mean.
2622    fn fold_inputs(
2623        logs: &BTreeMap<SessionId, PathBuf>,
2624    ) -> (BTreeMap<SessionId, Vec<String>>, BTreeSet<String>) {
2625        let mut payloads = BTreeMap::new();
2626        let mut grants = BTreeSet::new();
2627        for (id, path) in logs {
2628            let log = EventLog::open_read_only(path.clone(), *id).expect("log reads");
2629            let mut ordered = Vec::new();
2630            for event in log.event_snapshot().iter() {
2631                ordered.push(serde_json::to_string(&event.payload).expect("payload serializes"));
2632                if let EventPayload::TreeApprovalGrantCommitted { grant } = &event.payload
2633                    && crate::session::restart_stable_grant(grant)
2634                {
2635                    grants.insert(serde_json::to_string(grant).expect("grant serializes"));
2636                }
2637            }
2638            payloads.insert(*id, ordered);
2639        }
2640        (payloads, grants)
2641    }
