1//! The memory snapshot's text (`docs/site/specs/memory.md` section 7.4):
2//! the one user turn history assembly renders from the latest
3//! `memory_loaded`. The text depends on the event alone, so equal events
4//! render equal text.
5//!
6//! The sections are rendered one at a time as well, because the snapshot's
7//! limits are measured on rendered text: the user section and the machines
8//! section from their heading to their last line.
9

41fn root_save(agent: &str) -> String {
42    format!(
43        "Save what a later session should know: memory_write for a new memory, memory_edit to \
44         change one. Prefer editing an existing memory over adding a near-duplicate. Give each \
45         memory keywords a later search would use. Never save secrets. Lessons about doing your \
46         job as this agent go in agent://{agent}."
47    )
48}

50fn delegated_save(agent: &str) -> String {
51    format!(
52        "You may save only lessons about doing your job as this agent, in agent://{agent}: \
53         memory_write to create it, memory_edit to change it. Keep it short and current: update \
54         an entry rather than adding a near-duplicate. Never save secrets."
55    )
56}

60/// The heading of the memories the run's rules ask before reading.
61const LISTED: &str = "Listed only, as reading asks first (memory_read):";

63/// Whether `snapshot` holds any memory to show: a user category, the
64/// agent's file, or a machine or project memory, whole, listed, or counted
65/// (section 7.5). One that holds none, as when the run's rules leave
66/// nothing out of what exists, renders no memory turn at all.
67fn shows_memory(snapshot: &MemorySnapshot) -> bool {
68    let agent = snapshot.agent.as_ref().is_some_and(|agent| {
69        agent.omitted
70            || agent.listed.is_some()
71            || agent
72                .memory
73                .as_ref()
74                .is_some_and(|memory| !is_blank(&memory.body))
75    });
76    agent
77        || !snapshot.user.is_empty()
78        || !snapshot.user_omitted.is_empty()
79        || !snapshot.user_listed.is_empty()
80        || snapshot
81            .machines
82            .iter()
83            .any(|machine| !machine.memories.is_empty() || machine.omitted > 0)
84        || snapshot.project.as_ref().is_some_and(|project| {
85            project.overview.is_some() || !project.memories.is_empty() || project.omitted > 0
86        })
87}

89/// The whole memory turn, or `None` for an empty snapshot or one that holds
90/// no memory to show.
91pub(crate) fn snapshot_turn(snapshot: &MemorySnapshot) -> Option<String> {
92    let agent = snapshot.agent.as_ref()?;
93    if !shows_memory(snapshot) {
94        return None;
95    }
96    let id = agent.agent.as_str();
97    let mut parts = vec![BACKGROUND.to_owned()];
98    if snapshot.delegated {
99        parts.push(DELEGATED_SEARCH.to_owned());
100        if snapshot.writable {
101            parts.push(delegated_save(id));
102        }
103        parts.extend(agent_section(agent));
104    } else {
105        parts.push(
106            if snapshot.project.is_some() {
107                PROJECT_SEARCH
108            } else {
109                ROOT_SEARCH
110            }
111            .to_owned(),
112        );
113        if snapshot.writable {
114            parts.push(root_save(id));
115        }
116        parts.push(user_section(
117            &snapshot.user,
118            &snapshot.user_omitted,
119            &snapshot.user_listed,
120        ));
121        parts.extend(agent_section(agent));
122        parts.push(machines_section(&snapshot.machines));
123        parts.extend(snapshot.project.as_ref().map(project_section));
124    }
125    Some(format!(
126        "<system-reminder>\n# Memory\n{}\n</system-reminder>",
127        parts.join("\n\n")
128    ))
129}

131/// `## About the user`: one element per category with a body, then the
132/// empty and the omitted categories by URL, then the categories the run's
133/// rules ask before reading, listed.
134pub(crate) fn user_section(
135    user: &[MemorySnapshotCategory],
136    omitted: &[String],
137    listed: &[MemorySnapshotIndexEntry],
138) -> String {
139    let mut blocks = user
140        .iter()
141        .filter(|category| !is_blank(&category.body))
142        .map(|category| {
143            memory_element(
144                &format!("user://{}", category.category),
145                &category.description,
146                &category.body,
147            )
148        })
149        .collect::<Vec<_>>();
150    let mut lines = Vec::new();
151    let empty = user
152        .iter()
153        .filter(|category| is_blank(&category.body))
154        .map(|category| format!("user://{}", category.category))
155        .collect::<Vec<_>>();
156    if !empty.is_empty() {
157        lines.push(format!("Empty: {}", escape_text(&empty.join(", "))));
158    }
159    if !omitted.is_empty() {
160        let omitted = omitted
161            .iter()
162            .map(|category| format!("user://{category}"))
163            .collect::<Vec<_>>();
164        lines.push(format!(
165            "Not shown (too long): {}",
166            escape_text(&omitted.join(", "))
167        ));
168    }
169    if !lines.is_empty() {
170        blocks.push(lines.join("\n"));
171    }
172    if !listed.is_empty() {
173        blocks.push(listed_block(listed));
174    }
175    section("## About the user", blocks)
176}

178/// The memories the run's rules ask before reading, one index line each.
179fn listed_block(listed: &[MemorySnapshotIndexEntry]) -> String {
180    let mut lines = vec![LISTED.to_owned()];
181    lines.extend(listed.iter().map(index_line));
182    lines.join("\n")
183}

185/// `- <url>: <description> [keywords: …]`, escaped.
186fn index_line(memory: &MemorySnapshotIndexEntry) -> String {
187    let mut line = format!("- {}: {}", memory.url, memory.description);
188    if !memory.keywords.is_empty() {
189        line.push_str(&format!(" [keywords: {}]", memory.keywords.join(", ")));
190    }
191    escape_text(&line)
192}

194/// `## About this agent (<agent>)`: the agent's file, or why it is not
195/// shown. An agent file with an empty body, or none, adds no section at all
196/// (section 16.7): the daemon keeps a file for every agent, and most stay
197/// empty.
198pub(crate) fn agent_section(agent: &MemorySnapshotAgent) -> Option<String> {
199    let url = format!("agent://{}", agent.agent);
200    let block = match &agent.memory {
201        Some(memory) if !is_blank(&memory.body) => {
202            memory_element(&url, &memory.description, &memory.body)
203        }
204        None if agent.omitted => format!("Not shown (too long): {url}"),
205        None => listed_block(std::slice::from_ref(agent.listed.as_ref()?)),
206        Some(_) => return None,
207    };
208    Some(section(
209        &format!("## About this agent ({})", agent.agent),
210        vec![block],
211    ))
212}

214/// What the agent's file adds to the snapshot as rendered text: its
215/// escaped description and body. The snapshot's agent limit applies to it;
216/// the heading and tags come out of the snapshot's reserve.
217pub(crate) fn agent_content_bytes(description: &str, body: &str) -> usize {
218    escape_attribute(description).len() + escape_text(trim_body(body)).len()
219}

229/// `## Project <name>`: the session's project, its overview when it has
230/// one, and the index of its other memories.
231pub(crate) fn project_section(project: &MemorySnapshotProject) -> String {
232    let name = project.project.as_str();
233    let mut blocks = vec![format!(
234        "## Project {name}\nThis session belongs to the {name} project."
235    )];
236    if let Some(overview) = &project.overview {
237        blocks.push(project_overview(&project.project, overview));
238    }
239    blocks.push(project_index(project));
240    blocks.join("\n\n")
241}

243/// The overview's element. Its length counts toward the user section's

245pub(crate) fn project_overview(project: &ProjectName, overview: &MemorySnapshotMemory) -> String {
246    memory_element(
247        &format!("project://{project}/overview"),
248        &overview.description,
249        &overview.body,

253/// The index of the project's other memories. Its length counts toward the
254/// machines index's limit.
255pub(crate) fn project_index(project: &MemorySnapshotProject) -> String {
256    let mut lines = vec!["Other memories of this project (read with memory_read):".to_owned()];
257    lines.extend(project.memories.iter().map(index_line));
258    if project.omitted > 0 {
259        lines.push(format!(
260            "- {} more; search with layer \"project\".",
261            project.omitted
262        ));
263    }
264    if lines.len() == 1 {
265        lines.push(NONE.to_owned());
266    }
267    lines.join("\n")
268}

270fn machine_element(machine: &MemorySnapshotMachine) -> String {
271    let mut open = format!(
272        "<machine name=\"{}\"",
273        escape_attribute(machine.machine.as_str())
274    );
275    if !machine.attachments.is_empty() {
276        let attachments = machine
277            .attachments
278            .iter()
279            .map(|attachment| attachment.as_str())
280            .collect::<Vec<_>>()
281            .join(", ");
282        open.push_str(&format!(
283            " attachments=\"{}\"",
284            escape_attribute(&attachments)
285        ));
286    }
287    open.push('>');
288    let mut lines = machine.memories.iter().map(index_line).collect::<Vec<_>>();
289    if machine.omitted > 0 {
290        lines.push(format!(
291            "- {} more; search with layer \"machine\".",
292            machine.omitted
293        ));
294    }
295    if lines.is_empty() {
296        lines.push("No memories yet.".to_owned());
297    }
298    format!("{open}\n{}\n</machine>", lines.join("\n"))
299}

301fn section(heading: &str, blocks: Vec<String>) -> String {
302    if blocks.is_empty() {
303        format!("{heading}\n\n{NONE}")
304    } else {
305        format!("{heading}\n\n{}", blocks.join("\n\n"))
306    }
307}

309fn memory_element(url: &str, description: &str, body: &str) -> String {
310    format!(
311        "<memory url=\"{}\" description=\"{}\">\n{}\n</memory>",
312        escape_attribute(url),
313        escape_attribute(description),
314        escape_text(trim_body(body))
315    )
316}

318fn is_blank(body: &str) -> bool {
319    body.trim().is_empty()
320}
321
322/// A body without its leading blank lines and trailing whitespace, so the
323/// element's tags sit on their own lines.
324fn trim_body(body: &str) -> &str {
325    body.trim_start_matches(['\n', '\r']).trim_end()
326}
327
328fn escape_attribute(value: &str) -> String {
329    value
330        .replace('&', "&amp;")
331        .replace('<', "&lt;")
332        .replace('>', "&gt;")
333        .replace('"', "&quot;")
334}
335
336/// The turn's own tag names. A `<` that starts one of them, with or without
337/// `/`, in any case, is written as `&lt;`, so memory text can neither open
338/// nor close an element; every other `<` stays, so `Vec<T>` reads as it is.
339const TAG_NAMES: [&str; 3] = ["memory", "machine", "system-reminder"];
340
341fn escape_text(text: &str) -> String {
342    let mut escaped = String::with_capacity(text.len());
343    let mut rest = text;
344    while let Some(index) = rest.find('<') {
345        escaped.push_str(&rest[..index]);
346        let after = &rest[index + 1..];
347        let name = after.strip_prefix('/').unwrap_or(after);
348        let starts_tag = TAG_NAMES.iter().any(|tag| {
349            name.get(..tag.len())
350                .is_some_and(|prefix| prefix.eq_ignore_ascii_case(tag))
351        });
352        escaped.push_str(if starts_tag { "&lt;" } else { "<" });
353        rest = after;
354    }
355    escaped.push_str(rest);
356    escaped
357}
