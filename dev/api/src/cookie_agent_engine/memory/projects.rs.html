1//! The memory folders of projects (`docs/site/specs/projects.md` section 3):
2//! `memory/projects/<name>/`, which the project registry
3//! ([`crate::projects`]) moves, empties, and gives an `overview` memory whose
4//! description the daemon owns.
5//!
6//! Every function that changes a folder takes the held memory lock as proof,
7//! as the machine folders' do.
8
9use std::{fs, io, path::PathBuf};

11use super::{

19/// The memory every project has, loaded in full into its sessions.
20pub(crate) const OVERVIEW: &str = "overview";
21
22/// The description the daemon gives a project's overview.
23pub(crate) fn overview_description(project: &str) -> String {
24    format!("Overview of the `{project}` project, loaded in full into its sessions")
25}
26
27/// The URL of `project`'s overview.
28pub(crate) fn overview_url(project: &MemoryName) -> MemoryUrl {
29    MemoryUrl::Project {
30        project: project.clone(),
31        name: MemoryName::new(OVERVIEW).expect("`overview` is a memory name"),
32    }
33}
34
35/// Whether `url` is a project's overview.
36pub(crate) fn is_overview(url: &MemoryUrl) -> bool {
37    matches!(url, MemoryUrl::Project { name, .. } if name.as_str() == OVERVIEW)
38}
39
40/// `document` with the daemon's description for the overview of `project`.
41pub(crate) fn managed_overview(
42    project: &MemoryName,
43    document: &MemoryDocument,
44) -> Result<MemoryDocument, super::file::InvalidMemory> {
45    MemoryDocument::new(
46        overview_description(project.as_str()),
47        document.keywords().to_vec(),
48        document.body(),
49    )
50}

52impl MemoryRoot {
53    /// `memory/projects/<project>/`.
54    pub(crate) fn project_dir(&self, project: &MemoryName) -> PathBuf {
55        self.root.join(PROJECTS_DIR).join(project.as_str())
56    }
57
58    /// The folders under `memory/projects/` whose names are project names,
59    /// in name order.
60    pub(crate) fn project_folders(&self) -> Result<Vec<MemoryName>, MemoryError> {
61        Ok(read_dir_sorted(&self.root.join(PROJECTS_DIR))?
62            .into_iter()
63            .filter(|(_, kind)| kind.is_dir())
64            .filter_map(|(name, _)| MemoryName::new(name))
65            .collect())
66    }
67
68    /// The memory files (`*.md`) directly in a project's folder; none when
69    /// it has no folder.
70    pub(crate) fn project_memory_count(&self, project: &MemoryName) -> usize {
71        count_memories(&self.project_dir(project))
72    }
73
74    /// Makes sure `project`'s overview exists and carries the daemon's
75    /// description: a missing file is created with an empty body, and a
76    /// valid file with another description has its frontmatter rewritten,
77    /// its keywords and body kept. An invalid file, something other than a
78    /// regular file, or a rewrite past `max_file_bytes` is left alone, and
79    /// so is everything while the memory folder does not exist. Returns
80    /// whether it wrote anything, which it signals.
81    pub(crate) fn ensure_overview(
82        &self,
83        _lock: &MemoryLock,
84        project: &MemoryName,
85        max_file_bytes: usize,
86    ) -> Result<bool, MemoryError> {
87        if !self.root.is_dir() {
88            return Ok(false);
89        }
90        let path = self.file_path(&overview_url(project));
91        let description = overview_description(project.as_str());
92        let text = match read_bytes(&path)? {
93            None => Some(
94                MemoryDocument::new(description, Vec::new(), "")
95                    .expect("the overview's frontmatter is valid")
96                    .render(),
97            ),
98            Some((_, metadata)) if !metadata.is_file() => None,
99            Some((bytes, _)) => MemoryDocument::parse(&bytes, max_file_bytes)
100                .ok()
101                .filter(|document| document.description() != description)
102                .and_then(|document| managed_overview(project, &document).ok())
103                .and_then(|managed| {
104                    std::str::from_utf8(&bytes)
105                        .ok()
106                        .and_then(|text| with_frontmatter(text, &managed))
107                })
108                .filter(|text| text.len() <= max_file_bytes),
109        };
110        let Some(text) = text else {
111            return Ok(false);
112        };
113        write_file(&path, text.as_bytes())?;
114        self.changed();
115        Ok(true)
116    }
117
118    /// Moves `memory/projects/<from>/` to `<to>/`. Nothing moves when
119    /// `from` has no folder; a `to` folder that already exists fails.
120    pub(crate) fn rename_project_folder(
121        &self,
122        _lock: &MemoryLock,
123        from: &MemoryName,
124        to: &MemoryName,
125    ) -> Result<FolderMoves, MemoryError> {
126        let (source, target) = (self.project_dir(from), self.project_dir(to));
127        if fs::symlink_metadata(&target).is_ok() {
128            return Err(MemoryError::Exists(target));
129        }
130        let mut moves = FolderMoves::default();
131        if fs::symlink_metadata(&source).is_err() {
132            return Ok(moves);
133        }
134        fs::rename(&source, &target).map_err(io_error(&source))?;
135        moves.renamed.push((source, target));
136        self.changed();
137        Ok(moves)
138    }
139
140    /// Deletes `memory/projects/<project>/` with everything in it, memory
141    /// files or not, and returns how many memory files it held. Nothing
142    /// happens when it has no folder.
143    pub(crate) fn remove_project_folder(
144        &self,
145        _lock: &MemoryLock,
146        project: &MemoryName,
147    ) -> Result<u64, MemoryError> {
148        let path = self.project_dir(project);
149        let count = count_memories(&path) as u64;
150        match fs::remove_dir_all(&path) {
151            Ok(()) => {
152                self.changed();
153                Ok(count)
154            }
155            Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(0),
156            Err(source) => Err(MemoryError::Io { path, source }),
157        }
158    }
159}
160
161fn count_memories(dir: &std::path::Path) -> usize {
162    fs::read_dir(dir).map_or(0, |entries| {
163        entries
164            .filter_map(Result::ok)
165            .filter(|entry| entry.file_type().is_ok_and(|kind| kind.is_file()))
166            .filter(|entry| {
167                std::path::Path::new(&entry.file_name())
168                    .extension()
169                    .is_some_and(|extension| extension == MEMORY_EXTENSION)
170            })
171            .count()
172    })
173}
