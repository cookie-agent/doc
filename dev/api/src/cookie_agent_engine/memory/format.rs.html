1//! How memory reads out ages and sizes (`docs/site/specs/memory.md`
2//! section 16.9.4 and 16.9.7): `3 weeks`, `5 days`, `1.4 KB`, `812 bytes`.
3//! An age is whole units of the largest unit that fits, from minutes to
4//! years; a size under 1 KB is in bytes, otherwise in KB of 1024 bytes
5//! with at most one decimal.
6
7use jiff::Timestamp;
8
9/// How long ago `then` was at `now`, such as `3 weeks`; `less than a
10/// minute` for anything shorter, a time after `now` included.
11pub(crate) fn age(then: Timestamp, now: Timestamp) -> String {
12    let seconds = now.as_second().saturating_sub(then.as_second());
13    let units: [(i64, &str); 6] = [
14        (365 * 86_400, "year"),
15        (30 * 86_400, "month"),
16        (7 * 86_400, "week"),
17        (86_400, "day"),
18        (3_600, "hour"),
19        (60, "minute"),
20    ];
21    units
22        .iter()
23        .find(|(length, _)| seconds >= *length)
24        .map_or_else(
25            || "less than a minute".to_owned(),
26            |(length, unit)| {
27                let count = seconds / length;
28                format!("{count} {unit}{}", if count == 1 { "" } else { "s" })
29            },
30        )
31}
32
33/// `bytes` for a reader, such as `812 bytes` or `1.4 KB`.
34pub(crate) fn size(bytes: u64) -> String {
35    if bytes < 1024 {
36        return format!("{bytes} {}", if bytes == 1 { "byte" } else { "bytes" });
37    }
38    let tenths = (bytes * 10 + 512) / 1024;
39    if tenths.is_multiple_of(10) {
40        format!("{} KB", tenths / 10)
41    } else {
42        format!("{}.{} KB", tenths / 10, tenths % 10)
43    }
44}

46#[cfg(test)]
47mod tests {
48    use super::*;
49
50    fn at(seconds: i64) -> Timestamp {
51        Timestamp::from_second(seconds).unwrap()
52    }
53
54    #[test]
55    fn ages_use_the_largest_whole_unit() {
56        let now = at(1_000_000_000);
57        for (ago, expected) in [
58            (-5, "less than a minute"),
59            (59, "less than a minute"),
60            (60, "1 minute"),
61            (3_599, "59 minutes"),
62            (3_600, "1 hour"),
63            (2 * 86_400, "2 days"),
64            (6 * 86_400 + 86_399, "6 days"),
65            (7 * 86_400, "1 week"),
66            (21 * 86_400, "3 weeks"),
67            (30 * 86_400, "1 month"),
68            (364 * 86_400, "12 months"),
69            (365 * 86_400, "1 year"),
70            (800 * 86_400, "2 years"),
71        ] {
72            assert_eq!(age(at(1_000_000_000 - ago), now), expected, "{ago}");
73        }
74    }
75
76    #[test]
77    fn sizes_switch_to_kilobytes_at_1024_bytes() {
78        for (bytes, expected) in [
79            (0, "0 bytes"),
80            (1, "1 byte"),
81            (812, "812 bytes"),
82            (1023, "1023 bytes"),
83            (1024, "1 KB"),
84            (1434, "1.4 KB"),
85            (2048, "2 KB"),
86            (2611, "2.5 KB"),
87        ] {
88            assert_eq!(size(bytes), expected, "{bytes}");
89        }
90    }
