1//! Reading, listing, and changing memory files (`docs/site/specs/memory.md`
2//! sections 4 and 8, as section 16.9 amends them: stamps, size limits per
3//! kind of memory, model writes without revisions, and the similar-memory
4//! check).
5//!
6//! Reads never take the lock: every write replaces a file by renaming a
7//! flushed temporary file over it, so a reader sees the old or the new file
8//! whole. On Windows that rename can leave the name briefly unresolvable,
9//! so reads go through `secure_store::read_replaced` and its kin. Writes, edits, and deletes take [`super::MemoryLock`] for their
10//! whole check-and-change, and may wait for it up to a few seconds, so
11//! async callers run them on a blocking thread.
12

57/// Why a write to a project that does not exist is refused
58/// (`docs/site/specs/projects.md` section 3).
59fn unknown_project(url: &MemoryUrl, project: &MemoryName, projects: &[String]) -> StoreError {
60    let existing = if projects.is_empty() {
61        "There are no projects yet".to_owned()
62    } else {
63        format!("The projects are {}", projects.join(", "))
64    };
65    StoreError::Refused(format!(
66        "{url} names `{project}`, which is no project. {existing}; projects are created from a client, not by writing memory"
67    ))
68}

70/// How a write checks the memory it replaces.
71#[derive(Clone, Copy, Debug, Eq, PartialEq)]
72pub(crate) enum WriteMode<'a> {
73    /// A client's `memory.write` (section 10): without a revision the
74    /// memory must not exist; with one, it must exist at that revision.
75    Client(Option<&'a MemoryRevision>),
76    /// A model's `memory_write` (section 16.9.8): replacing a memory needs
77    /// `seen`, the revision the session last saw of it, to be its current
78    /// one. A new memory is first compared with the memories around it
79    /// (section 16.9.5); `new` says the writer already looked, so only an
80    /// identical body refuses it.
81    Model {
82        seen: Option<&'a MemoryRevision>,
83        new: bool,
84    },

87/// Why a store operation failed.
88#[derive(Debug, thiserror::Error)]
89pub(crate) enum StoreError {
90    #[error(transparent)]
91    Memory(#[from] MemoryError),
92    #[error("{url} does not exist")]
93    NotFound { url: MemoryUrl },
94    #[error("{url} already exists at revision {revision}; send that revision to replace it")]
95    Exists {
96        url: MemoryUrl,
97        revision: MemoryRevision,
98    },
99    #[error("{url} changed since it was read; its current revision is {current}")]
100    RevisionMismatch {
101        url: MemoryUrl,
102        current: MemoryRevision,
103    },
104    #[error("{url} is not a valid memory: {reason}")]
105    Invalid {
106        url: MemoryUrl,
107        reason: InvalidMemory,
108    },
109    #[error("{0}")]
110    Refused(String),
111    /// An edit's text occurs `count` times in the body, other than once.
112    #[error(
113        "{}",
114        cookie_agent_runtime_api::text_tools::occurrence_error(old, *count, &format!("the body of {url}"), false)
115    )]
116    Occurrences {
117        url: MemoryUrl,
118        old: String,
119        count: usize,
120    },
121    /// A model's replacing write of a memory its session has not seen as
122    /// it is now. The message leaves the memory's content out: following
123    /// a write, it would read as what the write left.
124    #[error(
125        "{url} was not written: this memory changed since you saw it, or you have not read it. Read it with memory_read, then write it again, or change it with memory_edit."
126    )]
127    Unseen { url: MemoryUrl },
128    /// The change would leave the memory over its limit. The message
129    /// leaves the memory's content out: following a write, it would read
130    /// as what the write left.
131    #[error("{}", too_large_message(url, *size, *max, *current))]
132    TooLarge {
133        url: MemoryUrl,
134        size: usize,
135        max: usize,
136        /// The memory's current size in bytes, when it exists.
137        current: Option<usize>,
138    },
139    /// A new memory looks like memories that exist.
140    #[error("{}", similar_message(url, near))]
141    Similar {
142        url: MemoryUrl,
143        near: Vec<NearMemory>,
144    },
145    /// A new memory has the body of one that exists.
146    #[error("{url} was not created: {existing} already holds exactly this body")]
147    Identical { url: MemoryUrl, existing: MemoryUrl },
148}

150/// Fails unless `old`, not empty, occurs exactly once in `body`, the body
151/// of the memory at `url`.
152fn occurs_once(url: &MemoryUrl, body: &str, old: &str) -> Result<(), StoreError> {
153    let count = if old.is_empty() {
154        0
155    } else {
156        body.matches(old).count()
157    };
158    if count == 1 {
159        return Ok(());
160    }
161    Err(StoreError::Occurrences {
162        url: url.clone(),
163        old: old.to_owned(),
164        count,
165    })
166}

168fn too_large_message(url: &MemoryUrl, size: usize, max: usize, current: Option<usize>) -> String {
169    let (size_text, max_text) = (format::size(size as u64), format::size(max as u64));
170    match current {
171        None => format!(
172            "{url} was not created: it would be {size_text}, over its limit of {max_text}. Make it shorter."
173        ),
174        Some(current) if current > max => format!(
175            "{url} was not written: it is {}, already over its limit of {max_text}, and only a change that makes it smaller is accepted until it fits; this one would leave it at {size_text}. Condense it first: read it with memory_read, then write a shorter version.",
176            format::size(current as u64)
177        ),
178        Some(_) => format!(
179            "{url} was not written: it would be {size_text}, over its limit of {max_text}. Condense it first: read it with memory_read, then merge or drop what it holds."
180        ),
181    }
182}

184fn similar_message(url: &MemoryUrl, near: &[NearMemory]) -> String {
185    let listed = near
186        .iter()
187        .map(|near| format!("- {} — {}", near.url, near.description))
188        .collect::<Vec<_>>()
189        .join("\n");
190    format!(
191        "{url} was not created: memories like it already exist:\n{listed}\nIf one of these covers it, edit that instead. If it is new, write it again with `new: true` and a one-line `reason`."
192    )
193}

230impl KeptMetadata {
231    /// What was kept, in a sentence or two for the writer of `url`.
232    pub(crate) fn describe(&self, url: &MemoryUrl) -> String {
233        let mut parts = Vec::new();
234        if let Some(description) = &self.description {
235            parts.push(format!(
236                "The description of {url} is the daemon's and cannot be changed, so it stays \"{description}\"."
237            ));
238        }
239        if !self.keywords.is_empty() {
240            let keywords = self
241                .keywords
242                .iter()
243                .map(|keyword| format!("`{keyword}`"))
244                .collect::<Vec<_>>()
245                .join(" and ");
246            parts.push(format!(
247                "Its keywords always include {keywords}, so they were added back; other keywords are kept."
248            ));
249        }
250        parts.join(" ")
251    }

328/// Reads a whole file, or `None` when nothing is at `path`. Something other
329/// than a regular file, a symbolic link included, reads as no bytes and
330/// parses as invalid.
331pub(super) fn read_bytes(path: &Path) -> Result<Option<(Vec<u8>, fs::Metadata)>, MemoryError> {
332    let metadata = match replaced(path, || fs::symlink_metadata(path)) {
333        Ok(metadata) => metadata,
334        Err(error) if not_found(&error) => return Ok(None),
335        Err(source) => return Err(io_error(path)(source)),
336    };
337    if !metadata.is_file() {
338        return Ok(Some((Vec::new(), metadata)));
339    }
340    let bytes = read_replaced(path).map_err(io_error(path))?;
341    Ok(Some((bytes, metadata)))
342}

344/// The modification time of the file `metadata` describes, to the second.
345fn modified_time(metadata: &fs::Metadata) -> Timestamp {
346    let time = metadata.modified().unwrap_or(SystemTime::UNIX_EPOCH);
347    to_second(Timestamp::try_from(time).unwrap_or(Timestamp::UNIX_EPOCH))
348}

350fn entry(url: MemoryUrl, bytes: &[u8], metadata: &fs::Metadata) -> MemoryEntry {
351    let document = if metadata.is_file() {
352        MemoryDocument::parse(bytes)
353    } else {
354        Err(InvalidMemory(NOT_FILE.to_owned()))
355    };
356    let file_time = modified_time(metadata);
357    let stamps = document.as_ref().ok();
358    MemoryEntry {
359        url,
360        revision: MemoryRevision::of(bytes),
361        size: bytes.len() as u64,
362        created: stamps
363            .and_then(MemoryDocument::created)
364            .unwrap_or(file_time),
365        modified: stamps
366            .and_then(MemoryDocument::modified)
367            .unwrap_or(file_time),
368        document,
369    }
370}

372/// Reads a memory file for listing and search. A file larger than any
373/// memory may be is not read whole: it is invalid, and its revision is
374/// hashed while streaming.
375fn load(path: &Path, url: MemoryUrl) -> Result<MemoryEntry, MemoryError> {
376    let metadata = replaced(path, || fs::metadata(path)).map_err(io_error(path))?;
377    if metadata.len() > MAX_FILE_BYTES as u64 {
378        use sha2::{Digest as _, Sha256};
379        let mut hasher = Sha256::new();
380        let mut file = replaced(path, || fs::File::open(path)).map_err(io_error(path))?;
381        let size = io::copy(&mut file, &mut hasher).map_err(io_error(path))?;
382        let file_time = modified_time(&metadata);
383        return Ok(MemoryEntry {
384            url,
385            revision: MemoryRevision::from_digest(hasher.finalize().as_slice()),
386            size,
387            created: file_time,
388            modified: file_time,
389            document: Err(too_large(usize::try_from(size).unwrap_or(usize::MAX))),
390        });
391    }
392    let bytes = read_replaced(path).map_err(io_error(path))?;
393    Ok(entry(url, &bytes, &metadata))
394}

407/// The sorted entries of a folder that exists; none when it does not.
408pub(super) fn read_dir_sorted(path: &Path) -> Result<Vec<(String, fs::FileType)>, MemoryError> {
409    let entries = match fs::read_dir(path) {
410        Ok(entries) => entries,
411        Err(error) if not_found(&error) => return Ok(Vec::new()),
412        Err(source) => return Err(io_error(path)(source)),
413    };
414    let mut found = Vec::new();
415    for entry in entries {
416        let entry = entry.map_err(io_error(path))?;
417        let kind = entry.file_type().map_err(io_error(path))?;
418        found.push((entry.file_name().to_string_lossy().into_owned(), kind));
419    }
420    found.sort_by(|left, right| left.0.cmp(&right.0));
421    Ok(found)
422}

429    /// Reads one memory whole: its text, revision, and parsed content.
430    pub(crate) fn read(&self, url: &MemoryUrl) -> Result<MemoryRead, StoreError> {
431        let path = self.file_path(url);
432        let (bytes, metadata) =
433            read_bytes(&path)?.ok_or_else(|| StoreError::NotFound { url: url.clone() })?;
434        Ok(MemoryRead {
435            text: String::from_utf8_lossy(&bytes).into_owned(),
436            entry: entry(url.clone(), &bytes, &metadata),
437        })
438    }

440    /// Every memory file in `scope`, valid or not, and what else lies in
441    /// its folders.
442    pub(crate) fn list(&self, scope: &ListScope) -> Result<MemoryListing, MemoryError> {
443        let mut listing = MemoryListing::default();
444        let wants = |layer: MemoryLayer| match scope {
445            ListScope::All => true,
446            ListScope::Layer(only) => *only == layer,
447            ListScope::Agent(_) => layer == MemoryLayer::Agent,
448            ListScope::Machine(_) => layer == MemoryLayer::Machine,
449            ListScope::Project(_) => layer == MemoryLayer::Project,
450        };
451        if *scope == ListScope::All {
452            self.list_top(&mut listing)?;
453        }
454        if wants(MemoryLayer::User) {
455            self.list_flat(MemoryLayer::User, &mut listing)?;
456        }
457        if wants(MemoryLayer::Agent) {
458            self.list_flat(MemoryLayer::Agent, &mut listing)?;
459        }
460        for layer in [MemoryLayer::Machine, MemoryLayer::Project] {
461            if wants(layer) {
462                let only = match scope {
463                    ListScope::Machine(machine) => Some(machine.as_str()),
464                    ListScope::Project(project) => Some(project.as_str()),
465                    _ => None,
466                };
467                self.list_grouped(layer, only, &mut listing)?;
468            }
469        }
470        if let ListScope::Agent(agent) = scope {
471            listing.entries.retain(
472                |entry| matches!(&entry.url, MemoryUrl::Agent { agent: found } if found == agent),
473            );
474            let path = format!("{}/{agent}.{MEMORY_EXTENSION}", MemoryLayer::Agent.dir());
475            listing.ignored.retain(|ignored| ignored.path == path);
476        }
477        Ok(listing)
478    }

480    /// Entries directly in the memory folder that are not part of the
481    /// layout.
482    fn list_top(&self, listing: &mut MemoryListing) -> Result<(), MemoryError> {
483        for (name, kind) in read_dir_sorted(&self.root)? {
484            if MemoryLayer::ALL.iter().any(|layer| layer.dir() == name) {
485                if !kind.is_dir() {
486                    listing.ignored.push(IgnoredEntry {
487                        path: name,
488                        reason: NOT_FOLDER,
489                    });
490                }
491            } else if name != super::LOCK_FILE && name != super::READS_FILE {
492                listing.ignored.push(IgnoredEntry {
493                    path: name,
494                    reason: UNKNOWN_TOP,
495                });
496            }
497        }
498        Ok(())
499    }

501    /// `user/<category>.md` or `agents/<agent>.md`.
502    fn list_flat(
503        &self,
504        layer: MemoryLayer,
505        listing: &mut MemoryListing,
506    ) -> Result<(), MemoryError> {
507        let dir = self.root.join(layer.dir());
508        for (file_name, kind) in read_dir_sorted(&dir)? {
509            let path = relative(&[layer.dir(), &file_name]);
510            let url = memory_stem(&file_name).and_then(|stem| match layer {
511                MemoryLayer::User => MemoryName::new(stem)
512                    .map(|category| MemoryUrl::User { category })
513                    .ok_or(BAD_NAME),
514                _ => MemoryUrl::parse(&format!("agent://{stem}")).map_err(|_| BAD_AGENT),
515            });
516            self.push(listing, path, &kind, url, &dir.join(&file_name))?;
517        }
518        Ok(())
519    }

521    /// `machines/<machine>/<name>.md` or `projects/<project>/<name>.md`.
522    fn list_grouped(
523        &self,
524        layer: MemoryLayer,
525        only: Option<&str>,
526        listing: &mut MemoryListing,
527    ) -> Result<(), MemoryError> {
528        let dir = self.root.join(layer.dir());
529        for (group, kind) in read_dir_sorted(&dir)? {
530            if only.is_some_and(|only| only != group) {
531                continue;
532            }
533            let group_path = relative(&[layer.dir(), &group]);
534            if !kind.is_dir() {
535                listing.ignored.push(IgnoredEntry {
536                    path: group_path,
537                    reason: NOT_FOLDER,
538                });
539                continue;
540            }
541            let valid_group = match layer {
542                MemoryLayer::Machine => MachineName::new(group.as_str()).is_ok(),
543                _ => MemoryName::new(group.as_str()).is_some(),
544            };
545            if !valid_group {
546                listing.ignored.push(IgnoredEntry {
547                    path: group_path,
548                    reason: if layer == MemoryLayer::Machine {
549                        BAD_MACHINE
550                    } else {
551                        BAD_NAME
552                    },
553                });
554                continue;
555            }
556            let group_dir = dir.join(&group);
557            for (file_name, kind) in read_dir_sorted(&group_dir)? {
558                let path = relative(&[layer.dir(), &group, &file_name]);
559                let url = memory_stem(&file_name).and_then(|stem| {
560                    MemoryUrl::parse(&format!("{layer}://{group}/{stem}")).map_err(|_| BAD_NAME)
561                });
562                self.push(listing, path, &kind, url, &group_dir.join(&file_name))?;
563            }
564        }
565        Ok(())
566    }

568    fn push(
569        &self,
570        listing: &mut MemoryListing,
571        path: String,
572        kind: &fs::FileType,
573        url: Result<MemoryUrl, &'static str>,
574        file: &Path,
575    ) -> Result<(), MemoryError> {
576        if kind.is_dir() {
577            listing.ignored.push(IgnoredEntry {
578                path,
579                reason: NESTED,
580            });
581            return Ok(());
582        }
583        if !kind.is_file() {
584            listing.ignored.push(IgnoredEntry {
585                path,
586                reason: NOT_FILE,
587            });
588            return Ok(());
589        }
590        match url {
591            Ok(url) => match load(file, url) {
592                Ok(entry) => listing.entries.push(entry),
593                // Removed since the folder was read.
594                Err(MemoryError::Io { source, .. }) if not_found(&source) => {}
595                Err(error) => return Err(error),
596            },
597            Err(reason) => listing.ignored.push(IgnoredEntry { path, reason }),
598        }
599        Ok(())
600    }

602    /// The valid memories in the same place as `url` (section 16.9.5): the
603    /// same project, the same machine, the user categories, or the agent's
604    /// file; `url` itself left out.
605    fn neighbours(&self, url: &MemoryUrl) -> Result<Vec<(MemoryUrl, MemoryDocument)>, MemoryError> {
606        let scope = match url {
607            MemoryUrl::User { .. } => ListScope::Layer(MemoryLayer::User),
608            MemoryUrl::Agent { agent } => ListScope::Agent(agent.clone()),
609            MemoryUrl::Machine { machine, .. } => ListScope::Machine(machine.clone()),
610            MemoryUrl::Project { project, .. } => ListScope::Project(project.clone()),
611        };
612        Ok(self
613            .list(&scope)?
614            .entries
615            .into_iter()
616            .filter(|entry| entry.url != *url)
617            .filter_map(|entry| Some((entry.url, entry.document.ok()?)))
618            .collect())
619    }

621    /// Creates or replaces one memory from its whole file text, which must
622    /// be a valid memory. `mode` says how the memory it replaces is checked
623    /// ([`WriteMode`]). The daemon stamps the file (section 16.9.6): a new
624    /// memory is created now, a replaced one keeps when it was created,
625    /// and either is modified now; stamps the text gives are ignored. The
626    /// result must fit the memory's limit in `limits`, unless the memory
627    /// is already over it and the write makes it smaller (section 16.9.7).
628    /// Creating an `agent://` file needs `owners` to know its agent. An
629    /// `agent://` file keeps the frontmatter the daemon owns (section
630    /// 16.7): the daemon's description and keywords for an agent it keeps a
631    /// file for, and the description the file has otherwise; a project's
632    /// overview keeps the daemon's description the same way. The body is
633    /// kept as given and the frontmatter rendered anew. A write to a
634    /// project that does not exist is refused; the first memory of a
635    /// machine creates its folder.
636    pub(crate) fn write(
637        &self,
638        url: &MemoryUrl,
639        text: &str,
640        mode: WriteMode<'_>,
641        owners: impl MemoryOwners,
642        limits: &MemoryLimits,
643        now: Timestamp,
644    ) -> Result<Written, StoreError> {
645        let invalid = |reason| StoreError::Invalid {
646            url: url.clone(),
647            reason,
648        };
649        let document = MemoryDocument::parse(text.as_bytes()).map_err(invalid)?;
650        let lock = self.lock()?;
651        if let MemoryUrl::Project { project, .. } = url
652            && let Some(projects) = owners.unknown_project(project)
653        {
654            return Err(unknown_project(url, project, &projects));
655        }
656        let path = self.file_path(url);
657        let existing = read_bytes(&path)?;
658        let current = existing
659            .as_ref()
660            .map(|(bytes, _)| MemoryRevision::of(bytes));
661        let existing_text = existing
662            .as_ref()
663            .map(|(bytes, _)| String::from_utf8_lossy(bytes).into_owned());
664        let mut near = Vec::new();
665        match (mode, current) {
666            (WriteMode::Client(None), Some(revision)) => {
667                return Err(StoreError::Exists {
668                    url: url.clone(),
669                    revision,
670                });
671            }
672            (WriteMode::Client(Some(_)), None) => {
673                return Err(StoreError::NotFound { url: url.clone() });
674            }
675            (WriteMode::Client(Some(expected)), Some(current)) if current != *expected => {
676                return Err(StoreError::RevisionMismatch {
677                    url: url.clone(),
678                    current,
679                });
680            }
681            (WriteMode::Model { seen, .. }, Some(current)) if seen != Some(&current) => {
682                return Err(StoreError::Unseen { url: url.clone() });
683            }
684            (WriteMode::Model { new, .. }, None) => {
685                match super::similar::compare(&document, &self.neighbours(url)?) {
686                    Similarity::Identical(existing) => {
687                        return Err(StoreError::Identical {
688                            url: url.clone(),
689                            existing,
690                        });
691                    }
692                    Similarity::Strong(found) if !new => {
693                        return Err(StoreError::Similar {
694                            url: url.clone(),
695                            near: found,
696                        });
697                    }
698                    Similarity::Strong(found) | Similarity::Weak(found) => near = found,
699                    Similarity::None => {}
700                }
701            }
702            (WriteMode::Client(_) | WriteMode::Model { .. }, _) => {}
703        }
704        let existing_document = existing
705            .as_ref()
706            .and_then(|(bytes, _)| MemoryDocument::parse(bytes).ok());
707        let mut target = None;
708        match url {
709            MemoryUrl::Agent { agent } => {
710                let status = owners.agent(agent);
711                if existing.is_none() && !status.exists() {
712                    return Err(StoreError::Refused(format!(
713                        "{url} names no agent in the agent registry"
714                    )));
715                }
716                target = match &status {
717                    AgentMemory::Managed(spec) => {
718                        Some(spec.apply(agent, &document).map_err(invalid)?)
719                    }
720                    AgentMemory::Registered | AgentMemory::Unknown => existing_document
721                        .as_ref()
722                        .map(|current| {
723                            MemoryDocument::new(
724                                current.description(),
725                                document.keywords().to_vec(),
726                                document.body(),
727                            )
728                        })
729                        .transpose()
730                        .map_err(invalid)?,
731                };
732            }
733            MemoryUrl::Project { project, .. } if super::projects::is_overview(url) => {
734                target =
735                    Some(super::projects::managed_overview(project, &document).map_err(invalid)?);
736            }
737            _ => {}
738        }
739        let kept = target.as_ref().and_then(|target| {
740            let added = target
741                .keywords()
742                .iter()
743                .filter(|keyword| {
744                    !document
745                        .keywords()
746                        .iter()
747                        .any(|given| given.to_lowercase() == keyword.to_lowercase())
748                })
749                .cloned()
750                .collect::<Vec<_>>();
751            let metadata = KeptMetadata {
752                description: (target.description() != document.description())
753                    .then(|| target.description().to_owned()),
754                keywords: added,
755            };
756            (metadata != KeptMetadata::default()).then_some(metadata)
757        });
758        let created = existing.as_ref().map_or(now, |(_, metadata)| {
759            existing_document
760                .as_ref()
761                .and_then(MemoryDocument::created)
762                .unwrap_or_else(|| modified_time(metadata))
763        });
764        let stamped = target
765            .unwrap_or_else(|| document.clone())
766            .with_stamps(Some(created), Some(now));
767        let written = with_frontmatter(text, &stamped)
768            .ok_or_else(|| invalid(InvalidMemory("the frontmatter was not found".to_owned())))?;
769        check_size(url, &written, existing_text.as_deref(), limits)?;
770        let revision = write_file(&path, written.as_bytes())?;
771        drop(lock);
772        self.changed();
773        Ok(Written {
774            revision,
775            kept,
776            created: existing.is_none(),
777            near,
778            replaced: existing_text,
779            text: written,
780        })
781    }

783    /// Replaces the one occurrence of `old` in the body of a memory with
784    /// `new`, whatever else changed in it since the caller saw it (section
785    /// 16.9.8). Fails, changing nothing, when `old` occurs zero or several
786    /// times, the result is not a valid memory, or it does not fit the
787    /// memory's limit ([`MemoryRoot::write`]). The file is written in the
788    /// form [`MemoryDocument::render`] gives, stamped as modified now, with
789    /// the daemon's description and keywords for an agent file the daemon
790    /// keeps (section 16.7).
791    pub(crate) fn edit(
792        &self,
793        url: &MemoryUrl,
794        old: &str,
795        new: &str,
796        owners: impl MemoryOwners,
797        limits: &MemoryLimits,
798        now: Timestamp,
799    ) -> Result<Edited, StoreError> {
800        let lock = self.lock()?;
801        if let MemoryUrl::Project { project, .. } = url
802            && let Some(projects) = owners.unknown_project(project)
803        {
804            return Err(unknown_project(url, project, &projects));
805        }
806        let path = self.file_path(url);
807        let (bytes, metadata) =
808            read_bytes(&path)?.ok_or_else(|| StoreError::NotFound { url: url.clone() })?;
809        let invalid = |reason| StoreError::Invalid {
810            url: url.clone(),
811            reason,
812        };
813        let document = MemoryDocument::parse(&bytes).map_err(invalid)?;
814        occurs_once(url, document.body(), old)?;
815        let mut edited = document
816            .with_body(document.body().replacen(old, new, 1))
817            .map_err(invalid)?;
818        if let MemoryUrl::Agent { agent } = url
819            && let AgentMemory::Managed(spec) = owners.agent(agent)
820        {
821            edited = spec.apply(agent, &edited).map_err(invalid)?;
822        }
823        if let MemoryUrl::Project { project, .. } = url
824            && super::projects::is_overview(url)
825        {
826            edited = super::projects::managed_overview(project, &edited).map_err(invalid)?;
827        }
828        let created = document
829            .created()
830            .unwrap_or_else(|| modified_time(&metadata));
831        let text = edited.with_stamps(Some(created), Some(now)).render();
832        let before = String::from_utf8_lossy(&bytes).into_owned();
833        check_size(url, &text, Some(&before), limits)?;
834        let revision = write_file(&path, text.as_bytes())?;
835        drop(lock);
836        self.changed();
837        Ok(Edited {
838            revision,
839            before,
840            after: text,
841        })
842    }

844    /// Whether an edit of the memory at `url` replacing `old` would apply
845    /// now: the memory exists, is valid, and `old` occurs once in its body.
846    /// Changes nothing; the edit checks again when it runs
847    /// ([`MemoryRoot::edit`]).
848    pub(crate) fn check_edit(&self, url: &MemoryUrl, old: &str) -> Result<(), StoreError> {
849        let read = self.read(url)?;
850        let document =
851            MemoryDocument::parse(read.text.as_bytes()).map_err(|reason| StoreError::Invalid {
852                url: url.clone(),
853                reason,
854            })?;
855        occurs_once(url, document.body(), old)
856    }

858    /// Deletes a memory, at revision `expected` when one is given (a
859    /// client's delete), and returns the text it held. A model's delete
860    /// deletes, whatever the layer (section 16.9.11). A project's folder
861    /// stays when its last memory goes: it belongs to the project.
862    pub(crate) fn delete(
863        &self,
864        url: &MemoryUrl,
865        expected: Option<&MemoryRevision>,
866    ) -> Result<String, StoreError> {
867        let lock = self.lock()?;
868        let path = self.file_path(url);
869        let (bytes, _) =
870            read_bytes(&path)?.ok_or_else(|| StoreError::NotFound { url: url.clone() })?;
871        let current = MemoryRevision::of(&bytes);
872        if let Some(expected) = expected
873            && current != *expected
874        {
875            return Err(StoreError::RevisionMismatch {
876                url: url.clone(),
877                current,
878            });
879        }
880        fs::remove_file(&path).map_err(io_error(&path))?;
881        drop(lock);
882        self.changed();
883        Ok(String::from_utf8_lossy(&bytes).into_owned())
884    }

886    /// Stamps every valid memory file without stamps with its modification
887    /// time, as `created` and `modified` (section 16.9.6), adding only the
888    /// missing lines before the frontmatter's closing `---`, so the rest of
889    /// the file stays as written. Run once at start; until a file is
890    /// stamped, reads take its modification time instead. Returns how many
891    /// files it stamped, and signals when it stamped any.
892    pub(crate) fn stamp_unstamped(&self) -> Result<usize, MemoryError> {
893        if !self.root.is_dir() {
894            return Ok(0);
895        }
896        let lock = self.lock()?;
897        let mut stamped = 0;
898        for entry in self.list(&ListScope::All)?.entries {
899            let Ok(document) = &entry.document else {
900                continue;
901            };
902            if document.created().is_some() && document.modified().is_some() {
903                continue;
904            }
905            let path = self.file_path(&entry.url);
906            let Some((bytes, metadata)) = read_bytes(&path)? else {
907                continue;
908            };
909            let Some(text) = std::str::from_utf8(&bytes)
910                .ok()
911                .and_then(|text| add_stamps(text, modified_time(&metadata)))
912            else {
913                continue;
914            };
915            write_file(&path, text.as_bytes())?;
916            stamped += 1;
917        }
918        drop(lock);
919        if stamped > 0 {
920            self.changed();
921        }
922        Ok(stamped)
923    }
924}

926/// `text`, a valid memory file, with the stamps it lacks set to `time`,
927/// added as lines just before its frontmatter's closing `---`. `None` when
928/// it has both, or its frontmatter is not found.
929fn add_stamps(text: &str, time: Timestamp) -> Option<String> {
930    let document = MemoryDocument::parse(text.as_bytes()).ok()?;
931    let frontmatter = MemoryDocument::frontmatter_len(text)?;
932    let fence = text[..frontmatter].trim_end_matches(['\n', '\r']).len() - "---".len();
933    let mut lines = String::new();
934    for (key, stamp) in [
935        ("created", document.created()),
936        ("modified", document.modified()),
937    ] {
938        if stamp.is_none() {
939            lines.push_str(&format!("{key}: {time}\n"));
940        }
941    }
942    if lines.is_empty() {
943        return None;
944    }
945    Some(format!("{}{lines}{}", &text[..fence], &text[fence..]))
946}

948/// Whether `text` may replace `current` (absent for a new memory) at
949/// `url`: it fits the memory's limit, or the memory is over it and `text`
950/// is smaller (section 16.9.7).
951fn check_size(
952    url: &MemoryUrl,
953    text: &str,
954    current: Option<&str>,
955    limits: &MemoryLimits,
956) -> Result<(), StoreError> {
957    let max = limits.max_bytes(url);
958    let shrinks_an_oversized = current.is_some_and(|current| current.len() > text.len());
959    if text.len() <= max || shrinks_an_oversized && current.is_some_and(|c| c.len() > max) {
960        return Ok(());
961    }
962    Err(StoreError::TooLarge {
963        url: url.clone(),
964        size: text.len(),
965        max,
966        current: current.map(str::len),
967    })
968}

970/// Writes `bytes` to `path` through a flushed temporary file in the same
971/// folder renamed over it, creating the folder first. Returns the new
972/// revision.
973pub(super) fn write_file(path: &Path, bytes: &[u8]) -> Result<MemoryRevision, MemoryError> {
974    let revision = write_file_unsynced_folder(path, bytes)?;
975    #[cfg(unix)]
976    {
977        let directory = path.parent().expect("memory files have a folder");
978        fs::File::open(directory)
979            .and_then(|directory| directory.sync_all())
980            .map_err(io_error(path))?;
981    }
982    Ok(revision)
983}

985/// [`write_file`] without the fsync of the file's folder, for a caller that
986/// writes several files of one folder and syncs it once after them.
987pub(super) fn write_file_unsynced_folder(
988    path: &Path,
989    bytes: &[u8],
990) -> Result<MemoryRevision, MemoryError> {
991    let directory = path.parent().expect("memory files have a folder");
992    fs::create_dir_all(directory).map_err(io_error(directory))?;
993    let file_name = path
994        .file_name()
995        .expect("memory files have a name")
996        .to_string_lossy();
997    let temporary = directory.join(format!(".{file_name}.{}.tmp", uuid::Uuid::now_v7()));
998    let result = (|| {
999        let mut output = fs::File::create(&temporary)?;
1000        io::Write::write_all(&mut output, bytes)?;
1001        output.sync_all()?;
1002        drop(output);
1003        fs::rename(&temporary, path)
1004    })();
1005    if result.is_err() {
1006        let _ = fs::remove_file(&temporary);
1007    }
1008    result.map_err(io_error(path))?;
1009    Ok(MemoryRevision::of(bytes))
1010}
