1//! Reading, listing, and changing memory files (`docs/site/specs/memory.md`
2//! sections 4 and 8).
3//!
4//! Reads never take the lock: every write replaces a file by renaming a
5//! flushed temporary file over it, so a reader sees the old or the new file
6//! whole. On Windows that rename can leave the name briefly unresolvable,
7//! so reads go through `secure_store::read_replaced` and its kin. Writes, edits, and deletes take [`super::MemoryLock`] for their
8//! whole check-and-change, and may wait for it up to a few seconds, so
9//! async callers run them on a blocking thread.
10

28/// What the store is told about the owners of a URL: the agent of an

50/// Why a write to a project that does not exist is refused
51/// (`docs/site/specs/projects.md` section 3).
52fn unknown_project(url: &MemoryUrl, project: &MemoryName, projects: &[String]) -> StoreError {
53    let existing = if projects.is_empty() {
54        "There are no projects yet".to_owned()
55    } else {
56        format!("The projects are {}", projects.join(", "))
57    };
58    StoreError::Refused(format!(
59        "{url} names `{project}`, which is no project. {existing}; projects are created from a client, not by writing memory"
60    ))
61}

74/// Why a store operation failed.
75#[derive(Debug, thiserror::Error)]
76pub(crate) enum StoreError {
77    #[error(transparent)]
78    Memory(#[from] MemoryError),
79    #[error("{url} does not exist")]
80    NotFound { url: MemoryUrl },
81    #[error("{url} already exists at revision {revision}; send that revision to replace it")]
82    Exists {
83        url: MemoryUrl,
84        revision: MemoryRevision,
85    },
86    #[error("{url} changed since it was read; its current revision is {current}")]
87    RevisionMismatch {
88        url: MemoryUrl,
89        current: MemoryRevision,
90    },
91    #[error("{url} is not a valid memory: {reason}")]
92    Invalid {
93        url: MemoryUrl,
94        reason: InvalidMemory,
95    },
96    #[error("{0}")]
97    Refused(String),
98    #[error("`{old}` does not occur in the body of {url}")]
99    NoOccurrence { url: MemoryUrl, old: String },
100    #[error(
101        "`{old}` occurs {count} times in the body of {url}; include more context so it occurs once"
102    )]
103    SeveralOccurrences {
104        url: MemoryUrl,
105        old: String,
106        count: usize,
107    },
108}

130/// What a write of an `agent://` file kept in place of what it was given

134#[derive(Clone, Debug, Default, Eq, PartialEq)]
135pub(crate) struct KeptMetadata {
136    /// The description the file keeps, when the text gave another.

139    pub(crate) keywords: Vec<String>,

142impl KeptMetadata {
143    /// What was kept, in a sentence or two for the writer of `url`.
144    pub(crate) fn describe(&self, url: &MemoryUrl) -> String {
145        let mut parts = Vec::new();
146        if let Some(description) = &self.description {
147            parts.push(format!(
148                "The description of {url} is the daemon's and cannot be changed, so it stays \"{description}\"."
149            ));
150        }
151        if !self.keywords.is_empty() {
152            let keywords = self
153                .keywords
154                .iter()
155                .map(|keyword| format!("`{keyword}`"))
156                .collect::<Vec<_>>()
157                .join(" and ");
158            parts.push(format!(
159                "Its keywords always include {keywords}, so they were added back; other keywords are kept."
160            ));
161        }
162        parts.join(" ")
163    }
164}

233/// Reads a whole file, or `None` when nothing is at `path`. Something other
234/// than a regular file, a symbolic link included, reads as no bytes and
235/// parses as invalid.
236pub(super) fn read_bytes(path: &Path) -> Result<Option<(Vec<u8>, fs::Metadata)>, MemoryError> {
237    let metadata = match replaced(path, || fs::symlink_metadata(path)) {
238        Ok(metadata) => metadata,
239        Err(error) if not_found(&error) => return Ok(None),
240        Err(source) => return Err(io_error(path)(source)),
241    };
242    if !metadata.is_file() {
243        return Ok(Some((Vec::new(), metadata)));
244    }
245    let bytes = read_replaced(path).map_err(io_error(path))?;
246    Ok(Some((bytes, metadata)))
247}

249fn entry(
250    url: MemoryUrl,
251    bytes: &[u8],
252    metadata: &fs::Metadata,
253    max_file_bytes: usize,
254) -> MemoryEntry {
255    let document = if metadata.is_file() {
256        MemoryDocument::parse(bytes, max_file_bytes)
257    } else {
258        Err(InvalidMemory(NOT_FILE.to_owned()))
259    };
260    MemoryEntry {
261        url,
262        revision: MemoryRevision::of(bytes),
263        size: bytes.len() as u64,
264        modified: metadata.modified().unwrap_or(SystemTime::UNIX_EPOCH),
265        document,
266    }
267}

269/// Reads a memory file for listing and search. A file over the limit is not
270/// read whole: it is invalid, and its revision is hashed while streaming.
271fn load(path: &Path, url: MemoryUrl, max_file_bytes: usize) -> Result<MemoryEntry, MemoryError> {
272    let metadata = replaced(path, || fs::metadata(path)).map_err(io_error(path))?;
273    if metadata.len() > max_file_bytes as u64 {
274        use sha2::{Digest as _, Sha256};
275        let mut hasher = Sha256::new();
276        let mut file = replaced(path, || fs::File::open(path)).map_err(io_error(path))?;
277        let size = io::copy(&mut file, &mut hasher).map_err(io_error(path))?;
278        return Ok(MemoryEntry {
279            url,
280            revision: MemoryRevision::from_digest(hasher.finalize().as_slice()),
281            size,
282            modified: metadata.modified().unwrap_or(SystemTime::UNIX_EPOCH),
283            document: Err(too_large(
284                usize::try_from(size).unwrap_or(usize::MAX),
285                max_file_bytes,
286            )),
287        });
288    }
289    let bytes = read_replaced(path).map_err(io_error(path))?;
290    Ok(entry(url, &bytes, &metadata, max_file_bytes))
291}

304/// The sorted entries of a folder that exists; none when it does not.
305pub(super) fn read_dir_sorted(path: &Path) -> Result<Vec<(String, fs::FileType)>, MemoryError> {
306    let entries = match fs::read_dir(path) {
307        Ok(entries) => entries,
308        Err(error) if not_found(&error) => return Ok(Vec::new()),
309        Err(source) => return Err(io_error(path)(source)),
310    };
311    let mut found = Vec::new();
312    for entry in entries {
313        let entry = entry.map_err(io_error(path))?;
314        let kind = entry.file_type().map_err(io_error(path))?;
315        found.push((entry.file_name().to_string_lossy().into_owned(), kind));
316    }
317    found.sort_by(|left, right| left.0.cmp(&right.0));
318    Ok(found)
319}

326    /// Reads one memory whole: its text, revision, and parsed content.
327    pub(crate) fn read(
328        &self,
329        url: &MemoryUrl,
330        max_file_bytes: usize,
331    ) -> Result<MemoryRead, StoreError> {
332        let path = self.file_path(url);
333        let (bytes, metadata) =
334            read_bytes(&path)?.ok_or_else(|| StoreError::NotFound { url: url.clone() })?;
335        Ok(MemoryRead {
336            text: String::from_utf8_lossy(&bytes).into_owned(),
337            entry: entry(url.clone(), &bytes, &metadata, max_file_bytes),
338        })
339    }

341    /// Every memory file in `scope`, valid or not, and what else lies in
342    /// its folders.
343    pub(crate) fn list(
344        &self,
345        scope: &ListScope,
346        max_file_bytes: usize,
347    ) -> Result<MemoryListing, MemoryError> {
348        let mut listing = MemoryListing::default();
349        let wants = |layer: MemoryLayer| match scope {
350            ListScope::All => true,
351            ListScope::Layer(only) => *only == layer,
352            ListScope::Agent(_) => layer == MemoryLayer::Agent,
353            ListScope::Machine(_) => layer == MemoryLayer::Machine,
354            ListScope::Project(_) => layer == MemoryLayer::Project,
355        };
356        if *scope == ListScope::All {
357            self.list_top(&mut listing)?;
358        }
359        if wants(MemoryLayer::User) {
360            self.list_flat(MemoryLayer::User, &mut listing, max_file_bytes)?;
361        }
362        if wants(MemoryLayer::Agent) {
363            self.list_flat(MemoryLayer::Agent, &mut listing, max_file_bytes)?;
364        }
365        for layer in [MemoryLayer::Machine, MemoryLayer::Project] {
366            if wants(layer) {
367                let only = match scope {
368                    ListScope::Machine(machine) => Some(machine.as_str()),
369                    ListScope::Project(project) => Some(project.as_str()),
370                    _ => None,
371                };
372                self.list_grouped(layer, only, &mut listing, max_file_bytes)?;
373            }
374        }
375        if let ListScope::Agent(agent) = scope {
376            listing.entries.retain(
377                |entry| matches!(&entry.url, MemoryUrl::Agent { agent: found } if found == agent),
378            );
379            let path = format!("{}/{agent}.{MEMORY_EXTENSION}", MemoryLayer::Agent.dir());
380            listing.ignored.retain(|ignored| ignored.path == path);
381        }
382        Ok(listing)
383    }

385    /// Entries directly in the memory folder that are not part of the
386    /// layout.
387    fn list_top(&self, listing: &mut MemoryListing) -> Result<(), MemoryError> {
388        for (name, kind) in read_dir_sorted(&self.root)? {
389            if MemoryLayer::ALL.iter().any(|layer| layer.dir() == name) {
390                if !kind.is_dir() {
391                    listing.ignored.push(IgnoredEntry {
392                        path: name,
393                        reason: NOT_FOLDER,
394                    });
395                }
396            } else if name != super::LOCK_FILE {
397                listing.ignored.push(IgnoredEntry {
398                    path: name,
399                    reason: UNKNOWN_TOP,
400                });
401            }
402        }
403        Ok(())
404    }

406    /// `user/<category>.md` or `agents/<agent>.md`.
407    fn list_flat(
408        &self,
409        layer: MemoryLayer,
410        listing: &mut MemoryListing,
411        max_file_bytes: usize,
412    ) -> Result<(), MemoryError> {
413        let dir = self.root.join(layer.dir());
414        for (file_name, kind) in read_dir_sorted(&dir)? {
415            let path = relative(&[layer.dir(), &file_name]);
416            let url = memory_stem(&file_name).and_then(|stem| match layer {
417                MemoryLayer::User => MemoryName::new(stem)
418                    .map(|category| MemoryUrl::User { category })
419                    .ok_or(BAD_NAME),
420                _ => MemoryUrl::parse(&format!("agent://{stem}")).map_err(|_| BAD_AGENT),
421            });
422            self.push(
423                listing,
424                path,
425                &kind,
426                url,
427                &dir.join(&file_name),
428                max_file_bytes,
429            )?;
430        }
431        Ok(())
432    }

434    /// `machines/<machine>/<name>.md` or `projects/<project>/<name>.md`.
435    fn list_grouped(
436        &self,
437        layer: MemoryLayer,
438        only: Option<&str>,
439        listing: &mut MemoryListing,
440        max_file_bytes: usize,
441    ) -> Result<(), MemoryError> {
442        let dir = self.root.join(layer.dir());
443        for (group, kind) in read_dir_sorted(&dir)? {
444            if only.is_some_and(|only| only != group) {
445                continue;
446            }
447            let group_path = relative(&[layer.dir(), &group]);
448            if !kind.is_dir() {
449                listing.ignored.push(IgnoredEntry {
450                    path: group_path,
451                    reason: NOT_FOLDER,
452                });
453                continue;
454            }
455            let valid_group = match layer {
456                MemoryLayer::Machine => MachineName::new(group.as_str()).is_ok(),
457                _ => MemoryName::new(group.as_str()).is_some(),
458            };
459            if !valid_group {
460                listing.ignored.push(IgnoredEntry {
461                    path: group_path,
462                    reason: if layer == MemoryLayer::Machine {
463                        BAD_MACHINE
464                    } else {
465                        BAD_NAME
466                    },
467                });
468                continue;
469            }
470            let group_dir = dir.join(&group);
471            for (file_name, kind) in read_dir_sorted(&group_dir)? {
472                let path = relative(&[layer.dir(), &group, &file_name]);
473                let url = memory_stem(&file_name).and_then(|stem| {
474                    MemoryUrl::parse(&format!("{layer}://{group}/{stem}")).map_err(|_| BAD_NAME)
475                });
476                self.push(
477                    listing,
478                    path,
479                    &kind,
480                    url,
481                    &group_dir.join(&file_name),
482                    max_file_bytes,
483                )?;
484            }
485        }
486        Ok(())
487    }

489    fn push(
490        &self,
491        listing: &mut MemoryListing,
492        path: String,
493        kind: &fs::FileType,
494        url: Result<MemoryUrl, &'static str>,
495        file: &Path,
496        max_file_bytes: usize,
497    ) -> Result<(), MemoryError> {
498        if kind.is_dir() {
499            listing.ignored.push(IgnoredEntry {
500                path,
501                reason: NESTED,
502            });
503            return Ok(());
504        }
505        if !kind.is_file() {
506            listing.ignored.push(IgnoredEntry {
507                path,
508                reason: NOT_FILE,
509            });
510            return Ok(());
511        }
512        match url {
513            Ok(url) => match load(file, url, max_file_bytes) {
514                Ok(entry) => listing.entries.push(entry),
515                // Removed since the folder was read.
516                Err(MemoryError::Io { source, .. }) if not_found(&source) => {}
517                Err(error) => return Err(error),
518            },
519            Err(reason) => listing.ignored.push(IgnoredEntry { path, reason }),
520        }
521        Ok(())
522    }

524    /// Creates or replaces one memory from its whole file text, which must
525    /// be a valid memory within `max_file_bytes`. Without `expected` the
526    /// file must not exist; with it, the file must exist at that revision.
527    /// Creating a `user://` file is refused for [`Origin::Model`]; creating
528    /// an `agent://` file needs `agents` to know its agent. An `agent://`
529    /// file keeps the frontmatter the daemon owns (section 16.7): the
530    /// daemon's description and keywords for an agent it keeps a file for,
531    /// and the description the file has otherwise; the text's own
532    /// frontmatter is replaced when it differs, its body kept as given.
533    /// A project's overview keeps the daemon's description the same way.
534    /// A write to a project that does not exist is refused; the first
535    /// memory of a machine creates its folder.
536    pub(crate) fn write(
537        &self,
538        url: &MemoryUrl,
539        text: &str,
540        expected: Option<&MemoryRevision>,
541        origin: Origin,
542        owners: impl MemoryOwners,
543        max_file_bytes: usize,
544    ) -> Result<Written, StoreError> {
545        let invalid = |reason| StoreError::Invalid {
546            url: url.clone(),
547            reason,
548        };
549        let document = MemoryDocument::parse(text.as_bytes(), max_file_bytes).map_err(invalid)?;
550        let lock = self.lock()?;
551        if let MemoryUrl::Project { project, .. } = url
552            && let Some(projects) = owners.unknown_project(project)
553        {
554            return Err(unknown_project(url, project, &projects));
555        }
556        let path = self.file_path(url);
557        let existing = read_bytes(&path)?.map(|(bytes, _)| bytes);
558        let current = existing.as_deref().map(MemoryRevision::of);
559        match (current, expected) {
560            (Some(revision), None) => {
561                return Err(StoreError::Exists {
562                    url: url.clone(),
563                    revision,
564                });
565            }
566            (None, Some(_)) => return Err(StoreError::NotFound { url: url.clone() }),
567            (Some(current), Some(expected)) if current != *expected => {
568                return Err(StoreError::RevisionMismatch {
569                    url: url.clone(),
570                    current,
571                });
572            }
573            (None, None) if matches!(url, MemoryUrl::User { .. }) && origin == Origin::Model => {
574                return Err(StoreError::Refused(format!(
575                    "{url} does not exist, and only the user adds user categories"
576                )));
577            }
578            (Some(_), Some(_)) | (None, None) => {}
579        }
580        let mut text = std::borrow::Cow::Borrowed(text);
581        let mut kept = None;
582        let target = match url {
583            MemoryUrl::Agent { agent } => {
584                let status = owners.agent(agent);
585                if existing.is_none() && !status.exists() {
586                    return Err(StoreError::Refused(format!(
587                        "{url} names no agent in the agent registry"
588                    )));
589                }
590                match &status {
591                    AgentMemory::Managed(spec) => {
592                        Some(spec.apply(agent, &document).map_err(invalid)?)
593                    }
594                    AgentMemory::Registered | AgentMemory::Unknown => existing
595                        .as_deref()
596                        .and_then(|bytes| MemoryDocument::parse(bytes, max_file_bytes).ok())
597                        .map(|current| {
598                            MemoryDocument::new(
599                                current.description(),
600                                document.keywords().to_vec(),
601                                document.body(),
602                            )
603                        })
604                        .transpose()
605                        .map_err(invalid)?,
606                }
607            }
608            MemoryUrl::Project { project, .. } if super::projects::is_overview(url) => {
609                Some(super::projects::managed_overview(project, &document).map_err(invalid)?)
610            }
611            _ => None,
612        };
613        if let Some(target) = target.filter(|target| *target != document) {
614            let added = target
615                .keywords()
616                .iter()
617                .filter(|keyword| {
618                    !document
619                        .keywords()
620                        .iter()
621                        .any(|given| given.to_lowercase() == keyword.to_lowercase())
622                })
623                .cloned()
624                .collect::<Vec<_>>();
625            let metadata = KeptMetadata {
626                description: (target.description() != document.description())
627                    .then(|| target.description().to_owned()),
628                keywords: added,
629            };
630            let replaced = with_frontmatter(&text, &target).ok_or_else(|| {
631                invalid(InvalidMemory("the frontmatter was not found".to_owned()))
632            })?;
633            if replaced.len() > max_file_bytes {
634                return Err(invalid(too_large(replaced.len(), max_file_bytes)));
635            }
636            text = std::borrow::Cow::Owned(replaced);
637            kept = (metadata != KeptMetadata::default()).then_some(metadata);
638        }
639        let revision = write_file(&path, text.as_bytes())?;
640        drop(lock);
641        self.changed();
642        Ok(Written { revision, kept })
643    }

645    /// Replaces the one occurrence of `old` in the body of a memory at
646    /// revision `expected` with `new`. Fails, changing nothing, when `old`
647    /// occurs zero or several times or the result is not a valid memory.
648    /// The file is written in the form [`MemoryDocument::render`] gives,
649    /// with the daemon's description and keywords for an agent file the
650    /// daemon keeps (section 16.7).
651    pub(crate) fn edit(
652        &self,
653        url: &MemoryUrl,
654        old: &str,
655        new: &str,
656        expected: &MemoryRevision,
657        owners: impl MemoryOwners,
658        max_file_bytes: usize,
659    ) -> Result<MemoryRevision, StoreError> {
660        let lock = self.lock()?;
661        if let MemoryUrl::Project { project, .. } = url
662            && let Some(projects) = owners.unknown_project(project)
663        {
664            return Err(unknown_project(url, project, &projects));
665        }
666        let path = self.file_path(url);
667        let (bytes, _) =
668            read_bytes(&path)?.ok_or_else(|| StoreError::NotFound { url: url.clone() })?;
669        let current = MemoryRevision::of(&bytes);
670        if current != *expected {
671            return Err(StoreError::RevisionMismatch {
672                url: url.clone(),
673                current,
674            });
675        }
676        let invalid = |reason| StoreError::Invalid {
677            url: url.clone(),
678            reason,
679        };
680        let document = MemoryDocument::parse(&bytes, max_file_bytes).map_err(invalid)?;
681        let count = if old.is_empty() {
682            0
683        } else {
684            document.body().matches(old).count()
685        };
686        match count {
687            0 => {
688                return Err(StoreError::NoOccurrence {
689                    url: url.clone(),
690                    old: old.to_owned(),
691                });
692            }
693            1 => {}
694            count => {
695                return Err(StoreError::SeveralOccurrences {
696                    url: url.clone(),
697                    old: old.to_owned(),
698                    count,
699                });
700            }
701        }
702        let mut edited = document
703            .with_body(document.body().replacen(old, new, 1))
704            .map_err(invalid)?;
705        if let MemoryUrl::Agent { agent } = url
706            && let AgentMemory::Managed(spec) = owners.agent(agent)
707        {
708            edited = spec.apply(agent, &edited).map_err(invalid)?;
709        }
710        if let MemoryUrl::Project { project, .. } = url
711            && super::projects::is_overview(url)
712        {
713            edited = super::projects::managed_overview(project, &edited).map_err(invalid)?;
714        }
715        let text = edited.render();
716        if text.len() > max_file_bytes {
717            return Err(invalid(too_large(text.len(), max_file_bytes)));
718        }
719        let revision = write_file(&path, text.as_bytes())?;
720        drop(lock);
721        self.changed();
722        Ok(revision)
723    }
724
725    /// Removes the body of a valid memory at revision `expected`, keeping
726    /// the file and its frontmatter byte for byte. This is what a model
727    /// tool's delete of a `user://` or `agent://` memory does
728    /// (section 16.6). A body that is already empty is left alone, and
729    /// nothing is written or signalled.
730    pub(crate) fn clear_body(
731        &self,
732        url: &MemoryUrl,
733        expected: &MemoryRevision,
734        max_file_bytes: usize,
735    ) -> Result<ClearedBody, StoreError> {
736        let lock = self.lock()?;
737        let path = self.file_path(url);
738        let (bytes, _) =
739            read_bytes(&path)?.ok_or_else(|| StoreError::NotFound { url: url.clone() })?;
740        let current = MemoryRevision::of(&bytes);
741        if current != *expected {
742            return Err(StoreError::RevisionMismatch {
743                url: url.clone(),
744                current,
745            });
746        }
747        let invalid = |reason| StoreError::Invalid {
748            url: url.clone(),
749            reason,
750        };
751        MemoryDocument::parse(&bytes, max_file_bytes).map_err(invalid)?;
752        let frontmatter = std::str::from_utf8(&bytes)
753            .ok()
754            .and_then(MemoryDocument::frontmatter_len)
755            .ok_or_else(|| invalid(InvalidMemory("the frontmatter was not found".to_owned())))?;
756        if frontmatter == bytes.len() {
757            return Ok(ClearedBody::AlreadyEmpty(current));
758        }
759        let revision = write_file(&path, &bytes[..frontmatter])?;
760        drop(lock);
761        self.changed();
762        Ok(ClearedBody::Cleared(revision))
763    }
764
765    /// Deletes a memory at revision `expected`. Deleting a `user://` or
766    /// `agent://` file is refused for [`Origin::Model`], whose deletes of
767    /// those clear the body instead ([`Self::clear_body`]). A project's
768    /// folder stays when its last memory goes: it belongs to the project.
769    pub(crate) fn delete(
770        &self,
771        url: &MemoryUrl,
772        expected: &MemoryRevision,
773        origin: Origin,
774    ) -> Result<(), StoreError> {
775        if origin == Origin::Model && matches!(url.layer(), MemoryLayer::User | MemoryLayer::Agent)
776        {
777            return Err(StoreError::Refused(format!(
778                "{url} can only be deleted by the user; a model tool clears its body instead"
779            )));
780        }
781        let lock = self.lock()?;
782        let path = self.file_path(url);
783        let (bytes, _) =
784            read_bytes(&path)?.ok_or_else(|| StoreError::NotFound { url: url.clone() })?;
785        let current = MemoryRevision::of(&bytes);
786        if current != *expected {
787            return Err(StoreError::RevisionMismatch {
788                url: url.clone(),
789                current,
790            });
791        }
792        fs::remove_file(&path).map_err(io_error(&path))?;
793        drop(lock);
794        self.changed();
795        Ok(())
796    }
797}

799/// Writes `bytes` to `path` through a flushed temporary file in the same
800/// folder renamed over it, creating the folder first. Returns the new
801/// revision.
802pub(super) fn write_file(path: &Path, bytes: &[u8]) -> Result<MemoryRevision, MemoryError> {
803    let directory = path.parent().expect("memory files have a folder");
804    fs::create_dir_all(directory).map_err(io_error(directory))?;
805    let file_name = path
806        .file_name()
807        .expect("memory files have a name")
808        .to_string_lossy();
809    let temporary = directory.join(format!(".{file_name}.{}.tmp", uuid::Uuid::now_v7()));
810    let result = (|| {
811        let mut output = fs::File::create(&temporary)?;
812        io::Write::write_all(&mut output, bytes)?;
813        output.sync_all()?;
814        drop(output);
815        fs::rename(&temporary, path)?;
816        #[cfg(unix)]
817        fs::File::open(directory)?.sync_all()?;
818        Ok(())
819    })();
820    if result.is_err() {
821        let _ = fs::remove_file(&temporary);
822    }
823    result.map_err(io_error(path))?;
824    Ok(MemoryRevision::of(bytes))
825}
