1//! Whether a new memory repeats one that exists
2//! (`docs/site/specs/memory.md` section 16.9.5). A memory a model creates
3//! is first compared with the memories in the same place: the same
4//! project, the same machine, the user categories, or the agent's file.
5//! The comparison uses descriptions and keywords, lowercased, with filler
6//! words left out, and matches as search does (section 16.9.3): no
7//! embeddings and no model call.
8
9use std::collections::BTreeSet;
10
11use super::{file::MemoryDocument, url::MemoryUrl};
12
13/// The most near memories a result lists.
14pub(crate) const MAX_NEAR: usize = 3;
15
16/// Words that say nothing about what a memory holds.
17const FILLER: [&str; 48] = [
18    "a", "about", "all", "an", "and", "any", "are", "as", "at", "be", "by", "can", "do", "does",
19    "for", "from", "has", "have", "how", "if", "in", "into", "is", "it", "its", "no", "not", "of",
20    "on", "or", "so", "than", "that", "the", "their", "them", "then", "there", "these", "this",
21    "to", "use", "was", "what", "when", "where", "which", "with",
22];
23
24/// A memory near a new one, as a result lists it.
25#[derive(Clone, Debug, Eq, PartialEq)]
26pub(crate) struct NearMemory {
27    pub(crate) url: MemoryUrl,
28    pub(crate) description: String,
29}
30
31/// How a new memory compares with the memories around it.
32#[derive(Clone, Debug, Eq, PartialEq)]
33pub(crate) enum Similarity {
34    /// Nothing near: save it.
35    None,
36    /// Some memories share a word or a keyword: save it, and list them.
37    Weak(Vec<NearMemory>),
38    /// Memories share two or more keywords, or most of the description's
39    /// words: refuse unless the writer says it is new.
40    Strong(Vec<NearMemory>),
41    /// This memory has the same body: always refuse.
42    Identical(MemoryUrl),
43}
44
45/// The lowercased words of `text` that are not filler.
46fn words(text: &str) -> BTreeSet<String> {
47    text.split(|character: char| !character.is_alphanumeric() && character != '-')
48        .map(|word| word.trim_matches('-').to_lowercase())
49        .filter(|word| !word.is_empty() && !FILLER.contains(&word.as_str()))
50        .collect()
51}
52
53fn keywords(document: &MemoryDocument) -> BTreeSet<String> {
54    document
55        .keywords()
56        .iter()
57        .map(|keyword| keyword.trim().to_lowercase())
58        .filter(|keyword| !keyword.is_empty() && !FILLER.contains(&keyword.as_str()))
59        .collect()
60}
61
62/// How `new` compares with `existing`, the other valid memories in its
63/// place.
64pub(crate) fn compare(
65    new: &MemoryDocument,
66    existing: &[(MemoryUrl, MemoryDocument)],
67) -> Similarity {
68    let body = new.body().trim();
69    if !body.is_empty()
70        && let Some((url, _)) = existing
71            .iter()
72            .find(|(_, document)| document.body().trim() == body)
73    {
74        return Similarity::Identical(url.clone());
75    }
76    let new_keywords = keywords(new);
77    let new_words = words(new.description());
78    let mut strong = Vec::new();
79    let mut weak = Vec::new();
80    for (url, document) in existing {
81        let shared_keywords = keywords(document).intersection(&new_keywords).count();
82        let their_words = words(document.description());
83        let shared_words = new_words.intersection(&their_words).count();
84        let keyword_words = new_keywords
85            .iter()
86            .filter(|keyword| their_words.contains(*keyword))
87            .count();
88        let near = NearMemory {
89            url: url.clone(),
90            description: document.description().to_owned(),
91        };
92        let rank = (shared_keywords, shared_words + keyword_words);
93        if shared_keywords >= 2 || (!new_words.is_empty() && shared_words * 2 > new_words.len()) {
94            strong.push((rank, near));
95        } else if rank != (0, 0) {
96            weak.push((rank, near));
97        }
98    }
99    let best = |mut found: Vec<((usize, usize), NearMemory)>| {
100        found.sort_by(|left, right| {
101            right
102                .0
103                .cmp(&left.0)
104                .then_with(|| left.1.url.cmp(&right.1.url))
105        });
106        found
107            .into_iter()
108            .take(MAX_NEAR)
109            .map(|(_, near)| near)
110            .collect::<Vec<_>>()
111    };
112    if !strong.is_empty() {
113        Similarity::Strong(best(strong))
114    } else if !weak.is_empty() {
115        Similarity::Weak(best(weak))
116    } else {
117        Similarity::None
118    }
119}

121#[cfg(test)]
122mod tests {
123    use super::*;
124
125    fn memory(description: &str, keywords: &[&str], body: &str) -> MemoryDocument {
126        MemoryDocument::new(
127            description,
128            keywords
129                .iter()
130                .map(|keyword| (*keyword).to_owned())
131                .collect(),
132            body,
133        )
134        .unwrap()
135    }
136
137    fn existing(entries: &[(&str, MemoryDocument)]) -> Vec<(MemoryUrl, MemoryDocument)> {
138        entries
139            .iter()
140            .map(|(url, document)| (MemoryUrl::parse(url).unwrap(), document.clone()))
141            .collect()
142    }
143
144    fn urls(near: &[NearMemory]) -> Vec<String> {
145        near.iter().map(|near| near.url.to_string()).collect()
146    }
147
148    #[test]
149    fn two_shared_keywords_or_most_description_words_are_a_strong_match() {
150        let around = existing(&[
151            (
152                "project://web/release",
153                memory("How releases are cut", &["release", "cargo-dist"], "- tag"),
154            ),
155            (
156                "project://web/ci",
157                memory("The CI workflows and their gates", &["ci"], "- gates"),
158            ),
159        ]);
160        let keywords = compare(
161            &memory("Publishing", &["Release", "cargo-dist", "x"], "other"),
162            &around,
163        );
164        assert!(
165            matches!(&keywords, Similarity::Strong(near) if urls(near) == ["project://web/release"]),
166            "{keywords:?}"
167        );
168        // Filler words do not count: "the", "and", "their" are left out,
169        // so two of the three words left ("workflows", "gates") match.
170        let words = compare(
171            &memory("The workflows and their gates on main", &[], "other"),
172            &around,
173        );
174        assert!(
175            matches!(&words, Similarity::Strong(near) if urls(near) == ["project://web/ci"]),
176            "{words:?}"
177        );
178    }
179
180    #[test]
181    fn a_shared_word_or_keyword_is_a_weak_match_and_none_is_none() {
182        let around = existing(&[(
183            "machine://labpc/gpu",
184            memory(
185                "CUDA toolkit and driver versions",
186                &["cuda", "gpu"],
187                "- 12.4",
188            ),
189        )]);
190        let weak = compare(
191            &memory("Where scratch builds go", &["cuda", "disk"], "- /scratch"),
192            &around,
193        );
194        assert!(
195            matches!(&weak, Similarity::Weak(near) if urls(near) == ["machine://labpc/gpu"]),
196            "{weak:?}"
197        );
198        let none = compare(&memory("Shell aliases", &["zsh"], "- alias"), &around);
199        assert_eq!(none, Similarity::None);
200    }
201
202    #[test]
203    fn an_identical_body_is_found_whatever_else_differs() {
204        let around = existing(&[(
205            "user://feedback",
206            memory("Feedback", &[], "  - never push on Fridays\n"),
207        )]);
208        assert_eq!(
209            compare(
210                &memory("Unrelated", &[], "- never push on Fridays"),
211                &around
212            ),
213            Similarity::Identical(MemoryUrl::parse("user://feedback").unwrap())
214        );
215        // Empty bodies are not compared.
216        let empty = existing(&[("user://profile", memory("Profile", &[], ""))]);
217        assert_eq!(
218            compare(&memory("Other", &[], " "), &empty),
219            Similarity::None
220        );
221    }
222
223    #[test]
224    fn at_most_three_near_memories_are_listed_best_first() {
225        let around = existing(&[
226            ("project://web/a", memory("a", &["one"], "a")),
227            ("project://web/b", memory("b", &["one", "two"], "b")),
228            (
229                "project://web/c",
230                memory("c", &["one", "two", "three"], "c"),
231            ),
232            ("project://web/d", memory("d", &["one", "two"], "d")),
233            ("project://web/e", memory("e", &["two", "three"], "e")),
234        ]);
235        let found = compare(&memory("new", &["one", "two", "three"], "n"), &around);
236        assert!(
237            matches!(&found, Similarity::Strong(near) if urls(near) == ["project://web/c", "project://web/b", "project://web/d"]),
238            "{found:?}"
239        );
240    }
