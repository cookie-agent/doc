1//! The agent memory files the daemon keeps (`docs/site/specs/memory.md`
2//! section 16.7): one `agents/<agent>.md` for every agent of the agent
3//! registries that is not internal, whose description and two keywords the
4//! daemon owns.
5//!
6//! [`MemoryRoot::sync_agents`] brings the folder in line with the
7//! registries under the memory lock: it creates a missing file with an
8//! empty body, rewrites the frontmatter of a file whose description or
9//! daemon keywords are off (keeping its body and any extra keywords), and
10//! removes the file of an agent no registry has any more when the daemon
11//! made it and nobody has touched it since. Doing nothing writes nothing
12//! and signals nothing.
13

25/// The keywords an agent's mode can give its memory.
26const MODE_KEYWORDS: [&str; 3] = ["primary", "subagent", "all"];
27
28/// What the daemon sets in one agent's memory file: the description, and
29/// the mode keyword that follows the agent ID.
30#[derive(Clone, Debug, Eq, PartialEq)]
31pub(crate) struct AgentMemorySpec {
32    description: String,
33    mode: &'static str,
34}
35
36/// What the memory store knows about the agent of an `agent://` URL.
37#[derive(Clone, Debug, Eq, PartialEq)]
38pub(crate) enum AgentMemory {
39    /// The daemon keeps this agent's file and owns its description.
40    Managed(AgentMemorySpec),
41    /// The registry has the agent, but the daemon keeps no file for it
42    /// (an internal agent).
43    Registered,
44    /// No registry has the agent.
45    Unknown,
46}
47
48impl AgentMemory {
49    /// Whether a file for the agent may be created.
50    pub(crate) const fn exists(&self) -> bool {
51        !matches!(self, Self::Unknown)
52    }
53}

55fn daemon_keywords(agent: &AgentId, mode: &str) -> Vec<String> {
56    let mut keywords = vec![agent.to_string()];
57    if agent.as_str() != mode {
58        keywords.push(mode.to_owned());
59    }
60    keywords
61}
62
63/// The start of every description the daemon writes for `agent`.
64fn description_prefix(agent: &AgentId) -> String {
65    format!("Memory of the `{agent}` agent")
66}
67
68/// `text` cut to `max` characters, ending with `…` when it was cut.
69fn cut(text: &str, max: usize) -> String {
70    match text.char_indices().nth(max) {
71        None => text.to_owned(),
72        Some(_) => {
73            let end = text
74                .char_indices()
75                .nth(max - 1)
76                .map_or(text.len(), |(index, _)| index);
77            format!("{}…", text[..end].trim_end())
78        }
79    }
80}

82impl AgentMemorySpec {
83    /// The spec of `agent`, from its agent document's description and its
84    /// mode keyword (`primary`, `subagent`, or `all`). The description is
85    /// ``Memory of the `<agent>` agent: <description>``, on one line: line
86    /// breaks, tabs, and other control characters become spaces, runs of
87    /// whitespace one space, and the whole is cut to 200 characters.
88    pub(crate) fn new(agent: &AgentId, document_description: &str, mode: &'static str) -> Self {
89        let words = document_description
90            .split(|character: char| character.is_whitespace() || character.is_control())
91            .filter(|word| !word.is_empty())
92            .collect::<Vec<_>>()
93            .join(" ");
94        let prefix = description_prefix(agent);
95        let description = if words.is_empty() {
96            prefix
97        } else {
98            format!("{prefix}: {words}")
99        };
100        Self {
101            description: cut(&description, MAX_DESCRIPTION_CHARS),
102            mode,
103        }
104    }
105
106    /// The daemon's two keywords, the agent ID and the mode; one when the
107    /// agent ID is the mode's name.
108    pub(crate) fn daemon_keywords(&self, agent: &AgentId) -> Vec<String> {
109        daemon_keywords(agent, self.mode)
110    }
111
112    /// The daemon's two keywords, then `existing` without them, without
113    /// any other mode keyword, and without repeats (all compared
114    /// case-insensitively), at most 32 in all.
115    pub(crate) fn keywords(&self, agent: &AgentId, existing: &[String]) -> Vec<String> {
116        let mut keywords = self.daemon_keywords(agent);
117        let mut seen = keywords
118            .iter()
119            .map(|keyword| keyword.to_lowercase())
120            .chain(MODE_KEYWORDS.iter().map(|mode| (*mode).to_owned()))
121            .collect::<Vec<_>>();
122        for keyword in existing {
123            let lower = keyword.to_lowercase();
124            if keywords.len() == MAX_KEYWORDS || seen.contains(&lower) {
125                continue;
126            }
127            seen.push(lower);
128            keywords.push(keyword.clone());
129        }
130        keywords
131    }
132
133    /// `document` with the daemon's description and keywords; its stamps
134    /// stay.
135    pub(crate) fn apply(
136        &self,
137        agent: &AgentId,
138        document: &MemoryDocument,
139    ) -> Result<MemoryDocument, InvalidMemory> {
140        Ok(MemoryDocument::new(
141            self.description.clone(),
142            self.keywords(agent, document.keywords()),
143            document.body(),
144        )?
145        .with_stamps(document.created(), document.modified()))
146    }
147
148    /// A new file's text: the daemon's frontmatter, stamped now, and an
149    /// empty body.
150    fn new_file(&self, agent: &AgentId) -> Result<String, InvalidMemory> {
151        let now = Some(jiff::Timestamp::now());
152        MemoryDocument::new(self.description.clone(), self.daemon_keywords(agent), "")
153            .map(|document| document.with_stamps(now, now).render())
154    }
155}

157/// `text`, a valid memory file, with the frontmatter of `document`, its
158/// stamps included, in place of its own; the body's bytes stay as they are.
159pub(crate) fn with_frontmatter(text: &str, document: &MemoryDocument) -> Option<String> {
160    let body = &text[MemoryDocument::frontmatter_len(text)?..];
161    let mut replaced =
162        MemoryDocument::new(document.description(), document.keywords().to_vec(), "")
163            .ok()?
164            .with_stamps(document.created(), document.modified())
165            .render();
166    replaced.push_str(body);
167    Some(replaced)
168}

170/// Whether `bytes` are an agent file exactly as the daemon writes one, with
171/// an empty body: its description starts as the daemon's do and its
172/// keywords are only the daemon's two. Such a file was made by the daemon
173/// and never touched, so it goes when its agent does.
174fn untouched(agent: &AgentId, bytes: &[u8]) -> bool {
175    let Ok(document) = MemoryDocument::parse(bytes) else {
176        return false;
177    };
178    let prefix = description_prefix(agent);
179    let described = document.description() == prefix
180        || document
181            .description()
182            .strip_prefix(&prefix)
183            .is_some_and(|rest| rest.starts_with(": "));
184    described
185        && document.body().is_empty()
186        && MODE_KEYWORDS
187            .iter()
188            .any(|mode| daemon_keywords(agent, mode) == document.keywords())
189        && document.render().as_bytes() == bytes
190}

192/// What [`MemoryRoot::sync_agents`] changed, by agent.
193#[derive(Clone, Debug, Default, Eq, PartialEq)]
194pub(crate) struct AgentSync {
195    pub(crate) created: Vec<AgentId>,
196    pub(crate) updated: Vec<AgentId>,
197    pub(crate) removed: Vec<AgentId>,
198}
199
200impl AgentSync {
201    pub(crate) fn is_empty(&self) -> bool {
202        self.created.is_empty() && self.updated.is_empty() && self.removed.is_empty()
203    }
204}

206impl MemoryRoot {
207    /// Brings the agent files in line with `desired`, the agents the
208    /// daemon keeps a file for, which is asked for once the memory lock is
209    /// held so that the latest registries win a race between two syncs.
210    /// A file that is invalid, or would grow past `agent_max_bytes`, is left
211    /// as it is. Nothing happens while the memory folder does not exist:
212    /// the next start's first start creates it.
213    pub(crate) fn sync_agents(
214        &self,
215        desired: impl FnOnce() -> BTreeMap<AgentId, AgentMemorySpec>,
216        agent_max_bytes: usize,
217    ) -> Result<AgentSync, MemoryError> {
218        if !self.root.is_dir() {
219            return Ok(AgentSync::default());
220        }
221        let lock = self.lock()?;
222        let desired = desired();
223        let mut sync = AgentSync::default();
224        for (agent, spec) in &desired {
225            let path = self.file_path(&MemoryUrl::Agent {
226                agent: agent.clone(),
227            });
228            let Some((bytes, metadata)) = read_bytes(&path)? else {
229                if let Ok(text) = spec.new_file(agent) {
230                    write_file_unsynced_folder(&path, text.as_bytes())?;
231                    sync.created.push(agent.clone());
232                }
233                continue;
234            };
235            if !metadata.is_file() {
236                continue;
237            }
238            let Ok(document) = MemoryDocument::parse(&bytes) else {
239                continue;
240            };
241            let Ok(managed) = spec.apply(agent, &document) else {
242                continue;
243            };
244            if managed.description() == document.description()
245                && managed.keywords() == document.keywords()
246            {
247                continue;
248            }
249            let Some(text) = std::str::from_utf8(&bytes)
250                .ok()
251                .and_then(|text| with_frontmatter(text, &managed))
252            else {
253                continue;
254            };
255            if text.len() > agent_max_bytes {
256                continue;
257            }
258            write_file_unsynced_folder(&path, text.as_bytes())?;
259            sync.updated.push(agent.clone());
260        }
261        let dir = self.root.join(AGENTS_DIR);
262        for (file_name, kind) in read_dir_sorted(&dir)? {
263            let Some(stem) = file_name.strip_suffix(&format!(".{MEMORY_EXTENSION}")) else {
264                continue;
265            };
266            let Ok(MemoryUrl::Agent { agent }) = MemoryUrl::parse(&format!("agent://{stem}"))
267            else {
268                continue;
269            };
270            if !kind.is_file() || desired.contains_key(&agent) {
271                continue;
272            }
273            let path = dir.join(&file_name);
274            let Some((bytes, _)) = read_bytes(&path)? else {
275                continue;
276            };
277            if untouched(&agent, &bytes) {
278                std::fs::remove_file(&path).map_err(super::store::io_error(&path))?;
279                sync.removed.push(agent);
280            }
281        }
282        // One folder fsync for every file written or removed above.
283        #[cfg(unix)]
284        if !sync.is_empty() {
285            std::fs::File::open(&dir)
286                .and_then(|folder| folder.sync_all())
287                .map_err(super::store::io_error(&dir))?;
288        }
289        drop(lock);
290        if !sync.is_empty() {
291            self.changed();
292        }
293        Ok(sync)
294    }
295}
