1//! Hands memory the process has freed back to the operating system.
2//!
3//! glibc returns little by itself: a freed chunk stays resident unless it
4//! touches the top of the heap, so after a long session is paged out its
5//! memory stays with the process. The janitor asks here on a pass that finds
6//! no run going, since a release holds the allocator's locks while it
7//! walks the free lists, and memory goes back once enough has been freed
8//! since the last time to be worth that walk.
9
10/// How many bytes must have been freed since the last release before the
11/// next one. As much as this can stay resident and unused.
12#[cfg(all(target_os = "linux", target_env = "gnu"))]
13const RELEASE_AFTER_FREED: usize = 32 * 1024 * 1024;
14
15/// What the janitor remembers between its passes.
16#[derive(Clone, Copy, Default)]
17pub(crate) struct FreedMemory {
18    /// The allocator's free bytes after the last release, or less if the
19    /// process has since allocated from them. Released pages still count as
20    /// free, so only what is above this was freed since.
21    settled: usize,
22}
23
24impl FreedMemory {
25    /// Whether `free` bytes hold `threshold` more than last settled.
26    #[cfg_attr(not(all(target_os = "linux", target_env = "gnu")), allow(dead_code))]
27    fn worth_releasing(&mut self, free: usize, threshold: usize) -> bool {
28        self.settled = self.settled.min(free);
29        free - self.settled >= threshold
30    }
31
32    /// Releases freed memory if enough has built up. Run it off the async
33    /// workers.
34    #[cfg(all(target_os = "linux", target_env = "gnu"))]
35    pub(crate) fn release(&mut self) {
36        // SAFETY: both only read and update the allocator's own state, under
37        // its own locks, and take no pointers from the caller.
38        unsafe {
39            if self.worth_releasing(libc::mallinfo2().fordblks, RELEASE_AFTER_FREED) {
40                libc::malloc_trim(0);
41                self.settled = libc::mallinfo2().fordblks;
42            }
43        }
44    }
45
46    #[cfg(not(all(target_os = "linux", target_env = "gnu")))]
47    pub(crate) fn release(&mut self) {}
48}

50#[cfg(test)]
51mod tests {
52    use super::FreedMemory;
53
54    #[test]
55    fn released_pages_that_still_count_as_free_do_not_release_again() {
56        let mut freed = FreedMemory { settled: 500 };
57        assert!(!freed.worth_releasing(500, 32));
58        assert!(!freed.worth_releasing(531, 32));
59        assert!(freed.worth_releasing(532, 32));
60    }
61
62    #[test]
63    fn allocating_from_free_memory_lowers_what_counts_as_settled() {
64        let mut freed = FreedMemory { settled: 500 };
65        assert!(!freed.worth_releasing(100, 32));
66        assert_eq!(freed.settled, 100);
67        assert!(freed.worth_releasing(132, 32));
68    }
69
70    #[test]
71    fn a_fresh_process_counts_all_its_free_memory() {
72        assert!(FreedMemory::default().worth_releasing(32, 32));
73        assert!(!FreedMemory::default().worth_releasing(31, 32));
74    }
75}
