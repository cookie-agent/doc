1//! Threads (`docs/site/specs/webhooks.md` sections 7.4, 7.5, and 8): the
2//! sessions an owner sends its fires or events into by key.
3//!
4//! An owner (a cron job with a new-session target now, a webhook later)
5//! keeps a [`ThreadMap`] beside its definition, in its own store. A
6//! delivery whose key has a live session goes into it; one whose key has
7//! none creates a session from the owner's template and records it
8//! ([`crate::runtime::threads`]). Every owner's sessions may get its
9//! thread prompt, a plain user message the engine submits at a session's
10//! start and again after each compaction; [`ThreadOwner`] names the owner
11//! a prompt and a producer message belong to.
12

18};

22/// The most threads an owner remembers. Recording one more forgets the
23/// thread used least recently; its session stays, and its key's next
24/// delivery starts a new one.
25pub(crate) const THREAD_LIMIT: usize = 1024;
26
27/// What threads belong to.
28#[derive(Clone, Debug, Eq, Hash, PartialEq)]
29pub(crate) enum ThreadOwner {
30    /// A cron job with a new-session target.
31    Cron(CronJobId),
32}
33
34impl ThreadOwner {
35    /// The owner the protocol names.
36    pub(crate) fn reference(&self) -> ThreadOwnerRef {
37        match self {
38            Self::Cron(job) => ThreadOwnerRef::Cron(job.clone()),
39        }
40    }
41
42    /// The origin of the thread prompts this owner sends.
43    pub(crate) fn origin(&self) -> EventOrigin {
44        thread_prompt_origin(&self.reference())
45    }
46
47    /// The owner of the thread prompt `origin` marks.
48    #[cfg_attr(not(test), allow(dead_code))]
49    pub(crate) fn from_origin(origin: &EventOrigin) -> Option<Self> {
50        match thread_prompt_owner(origin)? {
51            ThreadOwnerRef::Cron(job) => Some(Self::Cron(job)),
52            _ => None,
53        }
54    }
55
56    /// The owner whose fires or events `owner`'s producer messages carry.
57    pub(crate) fn of_producer(owner: &ProducerOwner) -> Option<Self> {
58        let job = owner.source()?.cron_job()?;
59        CronJobId::new(job.to_owned()).ok().map(Self::Cron)
60    }
61}

63/// One remembered thread.
64#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
65#[serde(deny_unknown_fields)]
66pub(crate) struct ThreadEntry {
67    pub(crate) key: String,
68    pub(crate) session: SessionId,
69    pub(crate) last_used: Timestamp,
70}
71
72/// An owner's threads, the most recently used last, at most
73/// [`THREAD_LIMIT`].
74#[derive(Clone, Debug, Default, Deserialize, Eq, PartialEq, Serialize)]
75#[serde(transparent)]
76pub(crate) struct ThreadMap(Vec<ThreadEntry>);
77
78impl ThreadMap {
79    pub(crate) fn is_empty(&self) -> bool {
80        self.0.is_empty()
81    }
82
83    /// The session `key`'s thread is.
84    pub(crate) fn session(&self, key: &str) -> Option<SessionId> {
85        self.0
86            .iter()
87            .find(|entry| entry.key == key)
88            .map(|entry| entry.session)
89    }
90
91    /// Records that `key`'s thread is `session`, used at `at`: the entry
92    /// moves to the end, and the least recently used one goes past
93    /// [`THREAD_LIMIT`].
94    pub(crate) fn record(&mut self, key: &str, session: SessionId, at: Timestamp) {
95        self.0.retain(|entry| entry.key != key);
96        self.0.push(ThreadEntry {
97            key: key.to_owned(),
98            session,
99            last_used: at,
100        });
101        let excess = self.0.len().saturating_sub(THREAD_LIMIT);
102        self.0.drain(..excess);
103    }
104
105    /// Forgets the threads whose sessions are among `deleted`. Whether any
106    /// was.
107    pub(crate) fn forget_sessions(&mut self, deleted: &HashSet<SessionId>) -> bool {
108        let before = self.0.len();
109        self.0.retain(|entry| !deleted.contains(&entry.session));
110        self.0.len() != before
111    }
112
113    /// Whether any thread is one of `sessions`.
114    pub(crate) fn names_any(&self, sessions: &HashSet<SessionId>) -> bool {
115        self.0.iter().any(|entry| sessions.contains(&entry.session))
116    }
117
118    /// The threads as clients read them.
119    pub(crate) fn wire(&self) -> Vec<CronThread> {
120        self.0
121            .iter()
122            .map(|entry| CronThread {
123                key: entry.key.clone(),
124                session_id: entry.session,
125                last_used: entry.last_used,
126            })
127            .collect()
128    }
129}

131/// Checks a cron job's `thread_key`: a time format of at most
132/// [`cookie_agent_protocol::CRON_THREAD_KEY_MAX_CHARS`] characters, without
133/// control characters, that formats a time into a key that is not blank.
134pub(crate) fn check_time_key_format(format: &str) -> Result<(), String> {
135    if format.trim().is_empty() {
136        return Err("must not be blank".into());
137    }
138    if format.chars().count() > cookie_agent_protocol::CRON_THREAD_KEY_MAX_CHARS {
139        return Err(format!(
140            "must be at most {} characters",
141            cookie_agent_protocol::CRON_THREAD_KEY_MAX_CHARS
142        ));
143    }
144    if format.chars().any(char::is_control) {
145        return Err("must not contain control characters".into());
146    }
147    time_key(format, Timestamp::UNIX_EPOCH, &TimeZone::UTC).map(drop)
148}
149
150/// The key `format` gives for `at` in `zone`: `strftime`'s text, trimmed.
151pub(crate) fn time_key(format: &str, at: Timestamp, zone: &TimeZone) -> Result<String, String> {
152    let zoned = at.to_zoned(zone.clone());
153    let key = jiff::fmt::strtime::format(format, &zoned)
154        .map_err(|error| format!("is not a time format: {error}"))?;
155    let key = key.trim();
156    if key.is_empty() {
157        return Err("formats a blank key".into());
158    }
159    if key.chars().count() > cookie_agent_protocol::THREAD_KEY_MAX_CHARS {
160        return Err(format!(
161            "formats a key longer than {} characters",
162            cookie_agent_protocol::THREAD_KEY_MAX_CHARS
163        ));
164    }
165    if key.chars().any(char::is_control) {
166        return Err("formats a key with control characters".into());
167    }
168    Ok(key.to_owned())
169}
170
171/// Checks a `thread_prompt`: not blank, at most
172/// [`cookie_agent_protocol::THREAD_PROMPT_MAX_BYTES`].
173pub(crate) fn check_thread_prompt(prompt: &str) -> Result<(), String> {
174    if prompt.trim().is_empty() {
175        return Err("must not be blank".into());
176    }
177    if prompt.len() > cookie_agent_protocol::THREAD_PROMPT_MAX_BYTES {
178        return Err(format!(
179            "must be at most {} bytes",
180            cookie_agent_protocol::THREAD_PROMPT_MAX_BYTES
181        ));
182    }
183    Ok(())
184}

198    #[test]
199    fn a_daily_key_changes_at_local_midnight() {
200        let berlin = zone("Europe/Berlin");
201        // 22:59 UTC is 23:59 in Berlin in winter; a minute later is the
202        // next day there, while UTC is still on the first.
203        assert_eq!(
204            time_key("%Y-%m-%d", at("2026-11-12T22:59:00Z"), &berlin).unwrap(),
205            "2026-11-12"
206        );
207        assert_eq!(
208            time_key("%Y-%m-%d", at("2026-11-12T23:00:00Z"), &berlin).unwrap(),
209            "2026-11-13"
210        );
211    }

216        // 2026-12-31 (Thursday) and 2027-01-03 (Sunday) are both in ISO
217        // week 53 of 2026; 2027-01-04 starts week 1 of 2027.
218        assert_eq!(

280    #[test]
281    fn recording_moves_a_thread_to_the_end_and_forgets_the_least_recent() {
282        let mut map = ThreadMap::default();
283        let (first, second) = (session(), session());
284        map.record("a", first, at("2026-11-12T00:00:00Z"));
285        map.record("b", second, at("2026-11-12T01:00:00Z"));
286        map.record("a", first, at("2026-11-12T02:00:00Z"));
287        assert_eq!(
288            map.wire()
289                .iter()
290                .map(|thread| thread.key.as_str())
291                .collect::<Vec<_>>(),
292            ["b", "a"]
293        );
294        for index in 0..THREAD_LIMIT {
295            map.record(&format!("k{index}"), session(), at("2026-11-13T00:00:00Z"));
296        }
297        assert_eq!(map.wire().len(), THREAD_LIMIT);
298        assert_eq!(map.session("b"), None);
299        assert_eq!(map.session("a"), None);
300    }

302    #[test]
303    fn deleted_sessions_leave_the_map() {
304        let mut map = ThreadMap::default();
305        let (kept, deleted) = (session(), session());
306        map.record("kept", kept, at("2026-11-12T00:00:00Z"));
307        map.record("gone", deleted, at("2026-11-12T00:00:00Z"));
308        assert!(map.forget_sessions(&HashSet::from([deleted])));
309        assert!(!map.forget_sessions(&HashSet::from([deleted])));
310        assert_eq!(map.session("gone"), None);
311        assert_eq!(map.session("kept"), Some(kept));
312    }
