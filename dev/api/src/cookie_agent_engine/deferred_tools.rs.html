1//! Deferred tools (`docs/site/specs/deferred-tools-cookie.md`): which tools a
2//! request defers, the tool round its eager tools come from, and the
3//! announcements that tell the model what it can load.
4//!
5//! Deferral is generic: [`policy_defers`] alone decides which composed tools
6//! are deferred. Everything else here asks only whether a tool is deferred.

25/// The loader tool: finds and loads deferred tools.
26pub(crate) const TOOL_SEARCH: &str = "tool_search";
27/// The retired loader that loaded deferred tools by name; history written
28/// before `tool_search` loaded holds its calls.
29pub(crate) const LEGACY_TOOL_LOAD: &str = "tool_load";
30
31/// Whether `name` is a loader tool, the retired one included.
32pub(crate) fn is_loader(name: &str) -> bool {
33    matches!(name, TOOL_SEARCH | LEGACY_TOOL_LOAD)
34}

36/// The deferral policy: whether a composed tool is deferred on a model that
37/// defers. It picks MCP and plugin tools (section 4), and the tools of the
38/// built-in groups in [`DEFERRED_BUILTIN_GROUPS`] (stable prefix spec,
39/// section 3), unless `config` sends their group up front.
40pub(crate) fn policy_defers(
41    source: &SourcedProvider,
42    tool: &str,
43    config: &DeferredToolsConfig,
44) -> bool {
45    match source.provider.tool_source(tool) {
46        Some(ProvidedToolSource::Mcp { .. } | ProvidedToolSource::Plugin { .. }) => {
47            !source.provider.always_load(tool)
48        }
49        None => source.provider.tool_group().is_some_and(|group| {
50            DEFERRED_BUILTIN_GROUPS.contains(&group.name) && !config.always_loads(group.name)
51        }),
52    }
53}

55/// The group a tool is announced under (section 7.1).
56#[derive(Clone, Debug, Eq, PartialEq)]
57pub(crate) struct Group {
58    pub(crate) name: String,
59    pub(crate) source: ToolGroupSource,
60    pub(crate) summary: Option<String>,
61    /// The summary is the source author's text, not cookie's.
62    pub(crate) untrusted: bool,
63}

65pub(crate) fn group_of(source: &SourcedProvider, tool: &str) -> Group {
66    match source.provider.tool_source(tool) {
67        Some(ProvidedToolSource::Mcp { server }) => Group {
68            name: server,
69            source: ToolGroupSource::Mcp,
70            summary: source.provider.source_summary(tool),
71            untrusted: true,
72        },
73        Some(ProvidedToolSource::Plugin { plugin }) => Group {
74            name: plugin,
75            source: ToolGroupSource::Plugin,
76            summary: source.provider.source_summary(tool),
77            untrusted: true,
78        },
79        None => match source.provider.tool_group() {
80            Some(group) => Group {
81                name: group.name.to_owned(),
82                source: ToolGroupSource::Builtin,
83                summary: Some(group.summary.to_owned()),
84                untrusted: false,
85            },
86            None => Group {
87                name: source.attachment.as_ref().map_or_else(
88                    || {
89                        let id = source.provider.provider_id();
90                        id.strip_prefix("builtin.").unwrap_or(id).to_owned()
91                    },
92                    ToString::to_string,
93                ),
94                source: ToolGroupSource::Builtin,
95                summary: None,
96                untrusted: false,
97            },
98        },
99    }
100}

106/// How a round records a provider in `prompt_groups`.

110        .map_or_else(|| id.clone(), |attachment| format!("{id}@{attachment}"))

113/// The providers whose prompt sections leave the system prompt: each has a
114/// tool, and none of its tools is eager (section 7.3).
115pub(crate) fn deferred_providers<'a>(
116    tools: impl IntoIterator<Item = (&'a str, &'a ProviderKey)>,
117    eager: &BTreeSet<String>,
118) -> BTreeSet<ProviderKey> {
119    let mut providers = BTreeMap::<&ProviderKey, bool>::new();
120    for (name, provider) in tools {
121        *providers.entry(provider).or_default() |= eager.contains(name);
122    }
123    providers
124        .into_iter()
125        .filter(|(_, has_eager)| !has_eager)
126        .map(|(provider, _)| provider.clone())
127        .collect()
128}

186fn round_definition(
187    definition: &ToolDefinition,
188    source: Option<&RoundToolSource>,
189) -> RoundToolDefinition {
190    RoundToolDefinition {
191        name: definition.name.clone(),
192        description: definition.description.clone(),
193        parameters: definition.input_schema.as_value().clone(),
194        provider_options: definition.provider_options.clone(),
195        source: source.cloned(),
196    }
197}

199fn declared(definition: &RoundToolDefinition) -> Option<ToolDefinition> {
200    let mut declared = ToolDefinition::new(
201        definition.name.clone(),
202        definition.description.clone(),
203        JsonSchema::new(definition.parameters.clone()).ok()?,
204    );
205    declared.provider_options = definition.provider_options.clone();
206    Some(declared)
207}

219/// Plans a request's tools on a model that defers (sections 4, 7 and 8).
220///
221/// `candidates` are the composed tools, the loader tools excluded;
222/// `loaders` are the loader tools' definitions. The tool round comes from
223/// `fold`, or starts here when the log has none since its checkpoint. The
224/// view takes each deferred tool's entry from `previous`, the view of the
225/// request before, while the tool is unchanged: a step copies no
226/// description or schema of a tool that stayed deferred.
227pub(crate) fn plan_reusing(
228    mut candidates: Vec<Candidate>,
229    loaders: &[ToolDefinition],
230    fold: &DeferralFold,
231    previous: &DeferralView,
232) -> DeferralPlan {
233    let mut pending = Vec::new();
234    // The round's record is borrowed, not copied, unless it changes here.
235    let mut round: Cow<'_, [RoundToolDefinition]> = match &fold.round {
236        Some(round) => Cow::Borrowed(&round.tools),
237        None => {
238            // A round after a checkpoint declares the tools it folded.
239            for candidate in candidates.iter_mut() {
240                candidate.deferred &= !fold.folded.contains(&candidate.definition.name);
241            }
242            let mut tools = candidates
243                .iter()
244                .filter(|candidate| !candidate.deferred)
245                .map(|candidate| round_definition(&candidate.definition, candidate.source.as_ref()))
246                .collect::<Vec<_>>();
247            if candidates.iter().any(|candidate| candidate.deferred) {
248                tools.extend(loaders.iter().map(|loader| round_definition(loader, None)));
249            }
250            tools.sort_by(|left, right| left.name.cmp(&right.name));
251            let mut prompt_groups = candidates
252                .iter()
253                .filter(|candidate| !candidate.deferred)
254                .map(|candidate| provider_label(&candidate.provider))
255                .collect::<Vec<_>>();
256            prompt_groups.sort();
257            prompt_groups.dedup();
258            pending.push(EventPayload::ToolRoundStarted {
259                tools: tools.clone(),
260                prompt_groups,
261            });
262            Cow::Owned(tools)
263        }
264    };
265    let eager_names = round
266        .iter()
267        .map(|tool| tool.name.clone())
268        .collect::<BTreeSet<_>>();
269    let present = candidates
270        .iter()
271        .map(|candidate| candidate.definition.name.clone())
272        .collect::<HashSet<_>>();
273    let deferred_providers = deferred_providers(
274        candidates
275            .iter()
276            .map(|candidate| (candidate.definition.name.as_str(), &candidate.provider)),
277        &eager_names,
278    );
279    // A tool not in the round is deferred: the policy defers it, or it
280    // arrived mid-round (section 8.2).
281    let deferred = candidates
282        .into_iter()
283        .filter(|candidate| !eager_names.contains(&candidate.definition.name))
284        .collect::<Vec<_>>();
285    if fold.round.is_some()
286        && !deferred.is_empty()
287        && !loaders.is_empty()
288        && !eager_names.contains(TOOL_SEARCH)
289    {
290        // The loader tool joins mid-round, once.
291        let tools = round.to_mut();
292        tools.extend(loaders.iter().map(|loader| round_definition(loader, None)));
293        tools.sort_by(|left, right| left.name.cmp(&right.name));
294        pending.push(EventPayload::ToolRoundStarted {
295            tools: round.to_vec(),
296            prompt_groups: fold
297                .round
298                .as_ref()
299                .map(|round| round.prompt_groups.clone())
300                .unwrap_or_default(),
301        });
302    }
303    let mut definitions = Vec::with_capacity(round.len() + deferred.len());
304    let mut departed = HashSet::new();
305    let mut eager = BTreeSet::new();
306    for tool in round.iter() {
307        if let Some(definition) = declared(tool) {
308            eager.insert(tool.name.clone());
309            definitions.push(definition);
310        }
311        if !present.contains(&tool.name) && !is_loader(&tool.name) {
312            departed.insert(tool.name.clone());
313        }
314    }
315    let previous = previous
316        .deferred
317        .iter()
318        .map(|entry| (entry.name.as_str(), entry))
319        .collect::<std::collections::HashMap<_, _>>();
320    let mut entries = Vec::with_capacity(deferred.len());
321    for candidate in deferred {
322        let definition = &candidate.definition;
323        let reused = previous
324            .get(definition.name.as_str())
325            .filter(|entry| {
326                entry.description == definition.description
327                    && entry.parameters == *definition.input_schema.as_value()
328                    && entry.group == candidate.group
329                    && entry.provider == candidate.provider
330            })
331            .map(|entry| std::sync::Arc::clone(entry));
332        entries.push(reused.unwrap_or_else(|| {
333            std::sync::Arc::new(DeferredEntry {
334                name: definition.name.clone(),
335                description: definition.description.clone(),
336                parameters: definition.input_schema.as_value().clone(),
337                group: candidate.group,
338                provider: candidate.provider,
339            })
340        }));
341        definitions.push(candidate.definition.deferred());
342    }
343    definitions.sort_by(|left, right| left.name.cmp(&right.name));
344    if let Some(announcement) = announcement(&entries, fold) {
345        pending.push(announcement);
346    }
347    DeferralPlan {
348        definitions,
349        departed,
350        pending,
351        view: DeferralView {
352            deferred: entries,
353            eager,
354            loaded: fold.loaded.clone(),
355            sections: BTreeMap::new(),
356        },
357        deferred_providers,
358    }
359}

366    /// ends; a call to one is refused.

369    pub(crate) pending: Vec<EventPayload>,

372/// Plans a request's tools on a model that does not defer (stable prefix
373/// spec, section 4). The round's tools are declared from its record, those
374/// that left included; a tool that joined is declared from its live
375/// definition. A round starts here when the log has none since its
376/// checkpoint, with every composed tool. A loader tool a deferring model's
377/// round declared is left out: nothing is loadable here.
378pub(crate) fn plan_eager(candidates: Vec<EagerCandidate>, fold: &DeferralFold) -> EagerPlan {
379    let mut pending = Vec::new();
380    // The round's record is borrowed, not copied.
381    let round: Cow<'_, [RoundToolDefinition]> = match &fold.round {
382        Some(round) => Cow::Borrowed(&round.tools),
383        None => {
384            let mut tools = candidates
385                .iter()
386                .map(|candidate| round_definition(&candidate.definition, candidate.source.as_ref()))
387                .collect::<Vec<_>>();
388            tools.sort_by(|left, right| left.name.cmp(&right.name));
389            let mut prompt_groups = candidates
390                .iter()
391                .map(|candidate| provider_label(&candidate.provider))
392                .collect::<Vec<_>>();
393            prompt_groups.sort();
394            prompt_groups.dedup();
395            pending.push(EventPayload::ToolRoundStarted {
396                tools: tools.clone(),
397                prompt_groups,
398            });
399            Cow::Owned(tools)
400        }
401    };
402    let present = candidates
403        .iter()
404        .map(|candidate| candidate.definition.name.clone())
405        .collect::<HashSet<_>>();
406    let mut in_round = HashSet::with_capacity(round.len());
407    let mut definitions = Vec::with_capacity(round.len() + candidates.len());
408    let mut departed = HashSet::new();
409    for tool in round.iter() {
410        in_round.insert(tool.name.clone());
411        if is_loader(&tool.name) {
412            continue;
413        }
414        if let Some(definition) = declared(tool) {
415            definitions.push(definition);
416        }
417        if !present.contains(&tool.name) {
418            departed.insert(tool.name.clone());
419        }
420    }
421    definitions.extend(
422        candidates
423            .into_iter()
424            .filter(|candidate| !in_round.contains(&candidate.definition.name))
425            .map(|candidate| candidate.definition),
426    );
427    definitions.sort_by(|left, right| left.name.cmp(&right.name));
428    EagerPlan {
429        definitions,
430        departed,
431        pending,
432    }
433}

435/// The announcement of what became loadable and what left since the
436/// announcements `fold` adds up to; `None` when nothing changed (section 7.2).
437fn announcement(
438    entries: &[std::sync::Arc<DeferredEntry>],
439    fold: &DeferralFold,
440) -> Option<EventPayload> {
441    let mut loadable = BTreeMap::<&str, (&Group, BTreeSet<&str>)>::new();
442    for entry in entries {
443        if !fold.loaded.contains(&entry.name) {
444            loadable
445                .entry(entry.group.name.as_str())
446                .or_insert_with(|| (&entry.group, BTreeSet::new()))
447                .1
448                .insert(entry.name.as_str());
449        }
450    }
451    let declared = entries
452        .iter()
453        .map(|entry| entry.name.as_str())
454        .collect::<HashSet<_>>();
455    let mut added = Vec::new();
456    for (name, (group, tools)) in &loadable {
457        let known = fold.announced.get(*name);
458        let new = tools
459            .iter()
460            .filter(|tool| known.is_none_or(|state| !state.tools.contains(**tool)))
461            .map(|tool| (*tool).to_owned())
462            .collect::<Vec<_>>();
463        if new.is_empty() {
464            continue;
465        }
466        let summary = group
467            .summary
468            .clone()
469            .filter(|summary| known.is_none_or(|state| state.summary.as_ref() != Some(summary)))
470            .map(|summary| bounded_summary(&summary));
471        added.push(AnnouncedToolGroup {
472            name: (*name).to_owned(),
473            source: group.source,
474            untrusted: group.untrusted && summary.is_some(),
475            summary,
476            tools: new,
477        });
478    }
479    // A loaded tool stays declared, so only a tool that left the request
480    // altogether is removed.
481    let mut removed = Vec::new();
482    for (name, state) in &fold.announced {
483        let gone = state
484            .tools
485            .iter()
486            .filter(|tool| !declared.contains(tool.as_str()))
487            .cloned()
488            .collect::<Vec<_>>();
489        if !gone.is_empty() {
490            removed.push(AnnouncedToolGroup {
491                name: name.clone(),
492                source: entries
493                    .iter()
494                    .find(|entry| &entry.group.name == name)
495                    .map_or(ToolGroupSource::Unknown, |entry| entry.group.source),
496                summary: None,
497                untrusted: false,
498                tools: gone,
499            });
500        }
501    }
502    (!added.is_empty() || !removed.is_empty())
503        .then_some(EventPayload::DeferredToolsAnnounced { added, removed })
504}

506/// A summary within its byte bound, cut at a character boundary.
507fn bounded_summary(summary: &str) -> String {
508    const MARKER: &str = " [truncated]";
509    let limit = AnnouncedToolGroup::MAX_SUMMARY_BYTES;
510    if summary.len() <= limit {
511        return summary.to_owned();
512    }
513    let mut end = limit - MARKER.len();
514    while !summary.is_char_boundary(end) {
515        end -= 1;
516    }
517    format!("{}{MARKER}", &summary[..end])
518}
