1//! Deferred tools (`docs/site/specs/deferred-tools-cookie.md`): which tools a
2//! request defers, the tool round its eager tools come from, and the
3//! announcements that tell the model what it can load.
4//!
5//! Deferral is generic: [`policy_defers`] alone decides which composed tools
6//! are deferred. Everything else here asks only whether a tool is deferred.

22/// The loader tool: finds and loads deferred tools.
23pub(crate) const TOOL_SEARCH: &str = "tool_search";
24/// The retired loader that loaded deferred tools by name; history written
25/// before `tool_search` loaded holds its calls.
26pub(crate) const LEGACY_TOOL_LOAD: &str = "tool_load";
27
28/// Whether `name` is a loader tool, the retired one included.
29pub(crate) fn is_loader(name: &str) -> bool {
30    matches!(name, TOOL_SEARCH | LEGACY_TOOL_LOAD)
31}

33/// The deferral policy: whether a composed tool is deferred on a model that
34/// defers. It picks MCP and plugin tools (section 4), and the tools of the
35/// built-in groups in [`DEFERRED_BUILTIN_GROUPS`] (stable prefix spec,
36/// section 3), unless `config` sends their group up front.
37pub(crate) fn policy_defers(
38    source: &SourcedProvider,
39    tool: &str,
40    config: &DeferredToolsConfig,
41) -> bool {
42    match source.provider.tool_source(tool) {
43        Some(ProvidedToolSource::Mcp { .. } | ProvidedToolSource::Plugin { .. }) => {
44            !source.provider.always_load(tool)
45        }
46        None => source.provider.tool_group().is_some_and(|group| {
47            DEFERRED_BUILTIN_GROUPS.contains(&group.name) && !config.always_loads(group.name)
48        }),
49    }
50}

52/// The group a tool is announced under (section 7.1).
53#[derive(Clone, Debug, Eq, PartialEq)]
54pub(crate) struct Group {
55    pub(crate) name: String,
56    pub(crate) source: ToolGroupSource,
57    pub(crate) summary: Option<String>,
58    /// The summary is the source author's text, not cookie's.
59    pub(crate) untrusted: bool,
60}

62pub(crate) fn group_of(source: &SourcedProvider, tool: &str) -> Group {
63    match source.provider.tool_source(tool) {
64        Some(ProvidedToolSource::Mcp { server }) => Group {
65            name: server,
66            source: ToolGroupSource::Mcp,
67            summary: source.provider.source_summary(tool),
68            untrusted: true,
69        },
70        Some(ProvidedToolSource::Plugin { plugin }) => Group {
71            name: plugin,
72            source: ToolGroupSource::Plugin,
73            summary: source.provider.source_summary(tool),
74            untrusted: true,
75        },
76        None => match source.provider.tool_group() {
77            Some(group) => Group {
78                name: group.name.to_owned(),
79                source: ToolGroupSource::Builtin,
80                summary: Some(group.summary.to_owned()),
81                untrusted: false,
82            },
83            None => Group {
84                name: source.attachment.as_ref().map_or_else(
85                    || {
86                        let id = source.provider.provider_id();
87                        id.strip_prefix("builtin.").unwrap_or(id).to_owned()
88                    },
89                    ToString::to_string,
90                ),
91                source: ToolGroupSource::Builtin,
92                summary: None,
93                untrusted: false,
94            },
95        },
96    }
97}

103/// How a round records a provider in `prompt_groups`.

107        .map_or_else(|| id.clone(), |attachment| format!("{id}@{attachment}"))

110/// The providers whose prompt sections leave the system prompt: each has a
111/// tool, and none of its tools is eager (section 7.3).
112pub(crate) fn deferred_providers<'a>(
113    tools: impl IntoIterator<Item = (&'a str, &'a ProviderKey)>,
114    eager: &BTreeSet<String>,
115) -> BTreeSet<ProviderKey> {
116    let mut providers = BTreeMap::<&ProviderKey, bool>::new();
117    for (name, provider) in tools {
118        *providers.entry(provider).or_default() |= eager.contains(name);
119    }
120    providers
121        .into_iter()
122        .filter(|(_, has_eager)| !has_eager)
123        .map(|(provider, _)| provider.clone())
124        .collect()
125}

181fn round_definition(
182    definition: &ToolDefinition,
183    source: Option<&RoundToolSource>,
184) -> RoundToolDefinition {
185    RoundToolDefinition {
186        name: definition.name.clone(),
187        description: definition.description.clone(),
188        parameters: definition.input_schema.as_value().clone(),
189        provider_options: definition.provider_options.clone(),
190        source: source.cloned(),
191    }
192}

194fn declared(definition: &RoundToolDefinition) -> Option<ToolDefinition> {
195    let mut declared = ToolDefinition::new(
196        definition.name.clone(),
197        definition.description.clone(),
198        JsonSchema::new(definition.parameters.clone()).ok()?,
199    );
200    declared.provider_options = definition.provider_options.clone();
201    Some(declared)
202}

204/// Plans a request's tools on a model that defers (sections 4, 7 and 8).
205///
206/// `candidates` are the composed tools, the loader tools excluded;
207/// `loaders` are the loader tools' definitions. The tool round comes from
208/// `fold`, or starts here when the log has none since its checkpoint.
209pub(crate) fn plan(
210    mut candidates: Vec<Candidate>,
211    loaders: &[ToolDefinition],
212    fold: &DeferralFold,
213) -> DeferralPlan {
214    let mut pending = Vec::new();
215    let mut round = match &fold.round {
216        Some(round) => round.tools.clone(),
217        None => {
218            // A round after a checkpoint declares the tools it folded.
219            for candidate in candidates.iter_mut() {
220                candidate.deferred &= !fold.folded.contains(&candidate.definition.name);
221            }
222            let mut tools = candidates
223                .iter()
224                .filter(|candidate| !candidate.deferred)
225                .map(|candidate| round_definition(&candidate.definition, candidate.source.as_ref()))
226                .collect::<Vec<_>>();
227            if candidates.iter().any(|candidate| candidate.deferred) {
228                tools.extend(loaders.iter().map(|loader| round_definition(loader, None)));
229            }
230            tools.sort_by(|left, right| left.name.cmp(&right.name));
231            let mut prompt_groups = candidates
232                .iter()
233                .filter(|candidate| !candidate.deferred)
234                .map(|candidate| provider_label(&candidate.provider))
235                .collect::<Vec<_>>();
236            prompt_groups.sort();
237            prompt_groups.dedup();
238            pending.push(EventPayload::ToolRoundStarted {
239                tools: tools.clone(),
240                prompt_groups,
241            });
242            tools
243        }
244    };
245    let eager_names = round
246        .iter()
247        .map(|tool| tool.name.clone())
248        .collect::<BTreeSet<_>>();
249    let present = candidates
250        .iter()
251        .map(|candidate| candidate.definition.name.clone())
252        .collect::<HashSet<_>>();
253    let deferred_providers = deferred_providers(
254        candidates
255            .iter()
256            .map(|candidate| (candidate.definition.name.as_str(), &candidate.provider)),
257        &eager_names,
258    );
259    // A tool not in the round is deferred: the policy defers it, or it
260    // arrived mid-round (section 8.2).
261    let deferred = candidates
262        .into_iter()
263        .filter(|candidate| !eager_names.contains(&candidate.definition.name))
264        .collect::<Vec<_>>();
265    if fold.round.is_some()
266        && !deferred.is_empty()
267        && !loaders.is_empty()
268        && !eager_names.contains(TOOL_SEARCH)
269    {
270        // The loader tool joins mid-round, once.
271        round.extend(loaders.iter().map(|loader| round_definition(loader, None)));
272        round.sort_by(|left, right| left.name.cmp(&right.name));
273        pending.push(EventPayload::ToolRoundStarted {
274            tools: round.clone(),
275            prompt_groups: fold
276                .round
277                .as_ref()
278                .map(|round| round.prompt_groups.clone())
279                .unwrap_or_default(),
280        });
281    }
282    let mut definitions = Vec::with_capacity(round.len() + deferred.len());
283    let mut departed = HashSet::new();
284    let mut eager = BTreeSet::new();
285    for tool in &round {
286        if let Some(definition) = declared(tool) {
287            eager.insert(tool.name.clone());
288            definitions.push(definition);
289        }
290        if !present.contains(&tool.name) && !is_loader(&tool.name) {
291            departed.insert(tool.name.clone());
292        }
293    }
294    let mut entries = Vec::with_capacity(deferred.len());
295    for candidate in deferred {
296        entries.push(DeferredEntry {
297            name: candidate.definition.name.clone(),
298            description: candidate.definition.description.clone(),
299            parameters: candidate.definition.input_schema.as_value().clone(),
300            group: candidate.group,
301            provider: candidate.provider,
302        });
303        definitions.push(candidate.definition.deferred());
304    }
305    definitions.sort_by(|left, right| left.name.cmp(&right.name));
306    if let Some(announcement) = announcement(&entries, fold) {
307        pending.push(announcement);
308    }
309    DeferralPlan {
310        definitions,
311        departed,
312        pending,
313        view: DeferralView {
314            deferred: entries,
315            eager,
316            loaded: fold.loaded.clone(),
317            sections: BTreeMap::new(),
318        },
319        deferred_providers,
320    }
321}

334/// Plans a request's tools on a model that does not defer (stable prefix
335/// spec, section 4). The round's tools are declared from its record, those
336/// that left included; a tool that joined is declared from its live
337/// definition. A round starts here when the log has none since its
338/// checkpoint, with every composed tool. A loader tool a deferring model's
339/// round declared is left out: nothing is loadable here.
340pub(crate) fn plan_eager(candidates: Vec<EagerCandidate>, fold: &DeferralFold) -> EagerPlan {
341    let mut pending = Vec::new();
342    let round = match &fold.round {
343        Some(round) => round.tools.clone(),
344        None => {
345            let mut tools = candidates
346                .iter()
347                .map(|candidate| round_definition(&candidate.definition, candidate.source.as_ref()))
348                .collect::<Vec<_>>();
349            tools.sort_by(|left, right| left.name.cmp(&right.name));
350            let mut prompt_groups = candidates
351                .iter()
352                .map(|candidate| provider_label(&candidate.provider))
353                .collect::<Vec<_>>();
354            prompt_groups.sort();
355            prompt_groups.dedup();
356            pending.push(EventPayload::ToolRoundStarted {
357                tools: tools.clone(),
358                prompt_groups,
359            });
360            tools
361        }
362    };
363    let present = candidates
364        .iter()
365        .map(|candidate| candidate.definition.name.clone())
366        .collect::<HashSet<_>>();
367    let mut in_round = HashSet::with_capacity(round.len());
368    let mut definitions = Vec::with_capacity(round.len() + candidates.len());
369    let mut departed = HashSet::new();
370    for tool in &round {
371        in_round.insert(tool.name.clone());
372        if is_loader(&tool.name) {
373            continue;
374        }
375        if let Some(definition) = declared(tool) {
376            definitions.push(definition);
377        }
378        if !present.contains(&tool.name) {
379            departed.insert(tool.name.clone());
380        }
381    }
382    definitions.extend(
383        candidates
384            .into_iter()
385            .filter(|candidate| !in_round.contains(&candidate.definition.name))
386            .map(|candidate| candidate.definition),
387    );
388    definitions.sort_by(|left, right| left.name.cmp(&right.name));
389    EagerPlan {
390        definitions,
391        departed,
392        pending,
393    }
394}

396/// The announcement of what became loadable and what left since the
397/// announcements `fold` adds up to; `None` when nothing changed (section 7.2).
398fn announcement(entries: &[DeferredEntry], fold: &DeferralFold) -> Option<EventPayload> {
399    let mut loadable = BTreeMap::<&str, (&Group, BTreeSet<&str>)>::new();
400    for entry in entries {
401        if !fold.loaded.contains(&entry.name) {
402            loadable
403                .entry(entry.group.name.as_str())
404                .or_insert_with(|| (&entry.group, BTreeSet::new()))
405                .1
406                .insert(entry.name.as_str());
407        }
408    }
409    let declared = entries
410        .iter()
411        .map(|entry| entry.name.as_str())
412        .collect::<HashSet<_>>();
413    let mut added = Vec::new();
414    for (name, (group, tools)) in &loadable {
415        let known = fold.announced.get(*name);
416        let new = tools
417            .iter()
418            .filter(|tool| known.is_none_or(|state| !state.tools.contains(**tool)))
419            .map(|tool| (*tool).to_owned())
420            .collect::<Vec<_>>();
421        if new.is_empty() {
422            continue;
423        }
424        let summary = group
425            .summary
426            .clone()
427            .filter(|summary| known.is_none_or(|state| state.summary.as_ref() != Some(summary)))
428            .map(|summary| bounded_summary(&summary));
429        added.push(AnnouncedToolGroup {
430            name: (*name).to_owned(),
431            source: group.source,
432            untrusted: group.untrusted && summary.is_some(),
433            summary,
434            tools: new,
435        });
436    }
437    // A loaded tool stays declared, so only a tool that left the request
438    // altogether is removed.
439    let mut removed = Vec::new();
440    for (name, state) in &fold.announced {
441        let gone = state
442            .tools
443            .iter()
444            .filter(|tool| !declared.contains(tool.as_str()))
445            .cloned()
446            .collect::<Vec<_>>();
447        if !gone.is_empty() {
448            removed.push(AnnouncedToolGroup {
449                name: name.clone(),
450                source: entries
451                    .iter()
452                    .find(|entry| &entry.group.name == name)
453                    .map_or(ToolGroupSource::Unknown, |entry| entry.group.source),
454                summary: None,
455                untrusted: false,
456                tools: gone,
457            });
458        }
459    }
460    (!added.is_empty() || !removed.is_empty())
461        .then_some(EventPayload::DeferredToolsAnnounced { added, removed })
462}

464/// A summary within its byte bound, cut at a character boundary.
465fn bounded_summary(summary: &str) -> String {
466    const MARKER: &str = " [truncated]";
467    let limit = AnnouncedToolGroup::MAX_SUMMARY_BYTES;
468    if summary.len() <= limit {
469        return summary.to_owned();
470    }
471    let mut end = limit - MARKER.len();
472    while !summary.is_char_boundary(end) {
473        end -= 1;
474    }
475    format!("{}{MARKER}", &summary[..end])
476}
