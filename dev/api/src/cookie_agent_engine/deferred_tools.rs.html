1//! Deferred tools (`docs/site/specs/deferred-tools-cookie.md`): which tools a
2//! request defers, the tool round its eager tools come from, and the
3//! announcements that tell the model what it can load.
4//!
5//! Deferral is generic: [`policy_defers`] alone decides which composed tools
6//! are deferred. Everything else here asks only whether a tool is deferred.

21/// The loader tool: finds and loads deferred tools.
22pub(crate) const TOOL_SEARCH: &str = "tool_search";
23/// The retired loader that loaded deferred tools by name; history written
24/// before `tool_search` loaded holds its calls.
25pub(crate) const LEGACY_TOOL_LOAD: &str = "tool_load";
26
27/// Whether `name` is a loader tool, the retired one included.
28pub(crate) fn is_loader(name: &str) -> bool {
29    matches!(name, TOOL_SEARCH | LEGACY_TOOL_LOAD)
30}

32/// The deferral policy: whether a composed tool is deferred on a model that
33/// defers. Today it picks MCP and plugin tools (section 4).
34pub(crate) fn policy_defers(source: &SourcedProvider, tool: &str) -> bool {
35    matches!(
36        source.provider.tool_source(tool),
37        Some(ProvidedToolSource::Mcp { .. } | ProvidedToolSource::Plugin { .. })
38    ) && !source.provider.always_load(tool)
39}

41/// The group a tool is announced under (section 7.1).
42#[derive(Clone, Debug, Eq, PartialEq)]
43pub(crate) struct Group {
44    pub(crate) name: String,
45    pub(crate) source: ToolGroupSource,
46    pub(crate) summary: Option<String>,
47    /// The summary is the source author's text, not cookie's.
48    pub(crate) untrusted: bool,
49}

51pub(crate) fn group_of(source: &SourcedProvider, tool: &str) -> Group {
52    match source.provider.tool_source(tool) {
53        Some(ProvidedToolSource::Mcp { server }) => Group {
54            name: server,
55            source: ToolGroupSource::Mcp,
56            summary: source.provider.source_summary(tool),
57            untrusted: true,
58        },
59        Some(ProvidedToolSource::Plugin { plugin }) => Group {
60            name: plugin,
61            source: ToolGroupSource::Plugin,
62            summary: source.provider.source_summary(tool),
63            untrusted: true,
64        },
65        None => match source.provider.tool_group() {
66            Some(group) => Group {
67                name: group.name.to_owned(),
68                source: ToolGroupSource::Builtin,
69                summary: Some(group.summary.to_owned()),
70                untrusted: false,
71            },
72            None => Group {
73                name: source.attachment.as_ref().map_or_else(
74                    || {
75                        let id = source.provider.provider_id();
76                        id.strip_prefix("builtin.").unwrap_or(id).to_owned()
77                    },
78                    ToString::to_string,
79                ),
80                source: ToolGroupSource::Builtin,
81                summary: None,
82                untrusted: false,
83            },
84        },
85    }
86}

88/// A tool provider as prompt sections know it: its ID and the attachment
89/// that offers it.
90pub(crate) type ProviderKey = (String, Option<String>);

92/// How a round records a provider in `prompt_groups`.
93pub(crate) fn provider_label((id, attachment): &ProviderKey) -> String {
94    attachment
95        .as_ref()
96        .map_or_else(|| id.clone(), |attachment| format!("{id}@{attachment}"))
97}

99/// The providers whose prompt sections leave the system prompt: each has a
100/// tool, and none of its tools is eager (section 7.3).
101pub(crate) fn deferred_providers<'a>(
102    tools: impl IntoIterator<Item = (&'a str, &'a ProviderKey)>,
103    eager: &BTreeSet<String>,
104) -> BTreeSet<ProviderKey> {
105    let mut providers = BTreeMap::<&ProviderKey, bool>::new();
106    for (name, provider) in tools {
107        *providers.entry(provider).or_default() |= eager.contains(name);
108    }
109    providers
110        .into_iter()
111        .filter(|(_, has_eager)| !has_eager)
112        .map(|(provider, _)| provider.clone())
113        .collect()
114}

147/// A request's tools under deferral.
148pub(crate) struct DeferralPlan {
149    /// Every tool the request declares, sorted by name.
150    pub(crate) definitions: Vec<ToolDefinition>,

152    /// ends; a call to one is refused.

158    pub(crate) deferred_providers: BTreeSet<ProviderKey>,

161fn round_definition(definition: &ToolDefinition) -> RoundToolDefinition {
162    RoundToolDefinition {
163        name: definition.name.clone(),
164        description: definition.description.clone(),
165        parameters: definition.input_schema.as_value().clone(),
166        provider_options: definition.provider_options.clone(),
167    }
168}

170fn declared(definition: &RoundToolDefinition) -> Option<ToolDefinition> {
171    let mut declared = ToolDefinition::new(
172        definition.name.clone(),
173        definition.description.clone(),
174        JsonSchema::new(definition.parameters.clone()).ok()?,
175    );
176    declared.provider_options = definition.provider_options.clone();
177    Some(declared)
178}

180/// Plans a request's tools on a model that defers (sections 4, 7 and 8).
181///
182/// `candidates` are the composed tools, the loader tools excluded;
183/// `loaders` are the loader tools' definitions. The tool round comes from
184/// `fold`, or starts here when the log has none since its checkpoint.
185pub(crate) fn plan(
186    mut candidates: Vec<Candidate>,
187    loaders: &[ToolDefinition],
188    fold: &DeferralFold,
189) -> DeferralPlan {
190    let mut pending = Vec::new();
191    let mut round = match &fold.round {
192        Some(round) => round.tools.clone(),
193        None => {
194            // A round after a checkpoint declares the tools it folded.
195            for candidate in candidates.iter_mut() {
196                candidate.deferred &= !fold.folded.contains(&candidate.definition.name);
197            }
198            let mut tools = candidates
199                .iter()
200                .filter(|candidate| !candidate.deferred)
201                .map(|candidate| round_definition(&candidate.definition))
202                .collect::<Vec<_>>();
203            if candidates.iter().any(|candidate| candidate.deferred) {
204                tools.extend(loaders.iter().map(round_definition));
205            }
206            tools.sort_by(|left, right| left.name.cmp(&right.name));
207            let mut prompt_groups = candidates
208                .iter()
209                .filter(|candidate| !candidate.deferred)
210                .map(|candidate| provider_label(&candidate.provider))
211                .collect::<Vec<_>>();
212            prompt_groups.sort();
213            prompt_groups.dedup();
214            pending.push(EventPayload::ToolRoundStarted {
215                tools: tools.clone(),
216                prompt_groups,
217            });
218            tools
219        }
220    };
221    let eager_names = round
222        .iter()
223        .map(|tool| tool.name.clone())
224        .collect::<BTreeSet<_>>();
225    let present = candidates
226        .iter()
227        .map(|candidate| candidate.definition.name.clone())
228        .collect::<HashSet<_>>();
229    let deferred_providers = deferred_providers(
230        candidates
231            .iter()
232            .map(|candidate| (candidate.definition.name.as_str(), &candidate.provider)),
233        &eager_names,
234    );
235    // A tool not in the round is deferred: the policy defers it, or it
236    // arrived mid-round (section 8.2).
237    let deferred = candidates
238        .into_iter()
239        .filter(|candidate| !eager_names.contains(&candidate.definition.name))
240        .collect::<Vec<_>>();
241    if fold.round.is_some()
242        && !deferred.is_empty()
243        && !loaders.is_empty()
244        && !eager_names.contains(TOOL_SEARCH)
245    {
246        // The loader tool joins mid-round, once.
247        round.extend(loaders.iter().map(round_definition));
248        round.sort_by(|left, right| left.name.cmp(&right.name));
249        pending.push(EventPayload::ToolRoundStarted {
250            tools: round.clone(),
251            prompt_groups: fold
252                .round
253                .as_ref()
254                .map(|round| round.prompt_groups.clone())
255                .unwrap_or_default(),
256        });
257    }
258    let mut definitions = Vec::with_capacity(round.len() + deferred.len());
259    let mut departed = HashSet::new();
260    let mut eager = BTreeSet::new();
261    for tool in &round {
262        if let Some(definition) = declared(tool) {
263            eager.insert(tool.name.clone());
264            definitions.push(definition);
265        }
266        if !present.contains(&tool.name) && !is_loader(&tool.name) {
267            departed.insert(tool.name.clone());
268        }
269    }
270    let mut entries = Vec::with_capacity(deferred.len());
271    for candidate in deferred {
272        entries.push(DeferredEntry {
273            name: candidate.definition.name.clone(),
274            description: candidate.definition.description.clone(),
275            parameters: candidate.definition.input_schema.as_value().clone(),
276            group: candidate.group,
277            provider: candidate.provider,
278        });
279        definitions.push(candidate.definition.deferred());
280    }
281    definitions.sort_by(|left, right| left.name.cmp(&right.name));
282    if let Some(announcement) = announcement(&entries, fold) {
283        pending.push(announcement);
284    }
285    DeferralPlan {
286        definitions,
287        departed,
288        pending,
289        view: DeferralView {
290            deferred: entries,
291            eager,
292            loaded: fold.loaded.clone(),
293            sections: BTreeMap::new(),
294        },
295        deferred_providers,
296    }
297}

299/// The announcement of what became loadable and what left since the
300/// announcements `fold` adds up to; `None` when nothing changed (section 7.2).
301fn announcement(entries: &[DeferredEntry], fold: &DeferralFold) -> Option<EventPayload> {
302    let mut loadable = BTreeMap::<&str, (&Group, BTreeSet<&str>)>::new();
303    for entry in entries {
304        if !fold.loaded.contains(&entry.name) {
305            loadable
306                .entry(entry.group.name.as_str())
307                .or_insert_with(|| (&entry.group, BTreeSet::new()))
308                .1
309                .insert(entry.name.as_str());
310        }
311    }
312    let declared = entries
313        .iter()
314        .map(|entry| entry.name.as_str())
315        .collect::<HashSet<_>>();
316    let mut added = Vec::new();
317    for (name, (group, tools)) in &loadable {
318        let known = fold.announced.get(*name);
319        let new = tools
320            .iter()
321            .filter(|tool| known.is_none_or(|state| !state.tools.contains(**tool)))
322            .map(|tool| (*tool).to_owned())
323            .collect::<Vec<_>>();
324        if new.is_empty() {
325            continue;
326        }
327        let summary = group
328            .summary
329            .clone()
330            .filter(|summary| known.is_none_or(|state| state.summary.as_ref() != Some(summary)))
331            .map(|summary| bounded_summary(&summary));
332        added.push(AnnouncedToolGroup {
333            name: (*name).to_owned(),
334            source: group.source,
335            untrusted: group.untrusted && summary.is_some(),
336            summary,
337            tools: new,
338        });
339    }
340    // A loaded tool stays declared, so only a tool that left the request
341    // altogether is removed.
342    let mut removed = Vec::new();
343    for (name, state) in &fold.announced {
344        let gone = state
345            .tools
346            .iter()
347            .filter(|tool| !declared.contains(tool.as_str()))
348            .cloned()
349            .collect::<Vec<_>>();
350        if !gone.is_empty() {
351            removed.push(AnnouncedToolGroup {
352                name: name.clone(),
353                source: entries
354                    .iter()
355                    .find(|entry| &entry.group.name == name)
356                    .map_or(ToolGroupSource::Unknown, |entry| entry.group.source),
357                summary: None,
358                untrusted: false,
359                tools: gone,
360            });
361        }
362    }
363    (!added.is_empty() || !removed.is_empty())
364        .then_some(EventPayload::DeferredToolsAnnounced { added, removed })
365}

367/// A summary within its byte bound, cut at a character boundary.
368fn bounded_summary(summary: &str) -> String {
369    const MARKER: &str = " [truncated]";
370    let limit = AnnouncedToolGroup::MAX_SUMMARY_BYTES;
371    if summary.len() <= limit {
372        return summary.to_owned();
373    }
374    let mut end = limit - MARKER.len();
375    while !summary.is_char_boundary(end) {
376        end -= 1;
377    }
378    format!("{}{MARKER}", &summary[..end])
379}
