466#[derive(Debug)]
467struct RuntimeRevisionIndex {
468    path: PathBuf,
469    mappings:
470        HashMap<cookie_agent_protocol::RuntimeRevision, cookie_agent_protocol::RuntimeRevision>,
471}
472
473impl RuntimeRevisionIndex {
474    fn open(path: PathBuf) -> Result<Self, EngineError> {
475        let mut mappings = HashMap::new();
476        for record in events::load_jsonl_shared::<StoredRuntimeRevisionMapping>(&path)? {
477            match mappings.insert(
478                record.runtime_revision.clone(),
479                record.model_runtime_revision.clone(),
480            ) {
481                Some(existing) if existing != record.model_runtime_revision => {
482                    return Err(EngineError::RuntimeCompileFailed);
483                }
484                _ => {}
485            }
486        }
487        Ok(Self { path, mappings })
488    }
489
490    fn record(
491        &mut self,
492        runtime_revision: cookie_agent_protocol::RuntimeRevision,
493        model_runtime_revision: cookie_agent_protocol::RuntimeRevision,
494    ) -> Result<(), EngineError> {
495        if let Some(existing) = self.mappings.get(&runtime_revision) {
496            return if existing == &model_runtime_revision {
497                Ok(())
498            } else {
499                Err(EngineError::RuntimeCompileFailed)
500            };
501        }
502        events::append_jsonl(
503            &self.path,
504            &StoredRuntimeRevisionMapping {
505                protocol_version: cookie_agent_protocol::ProtocolVersion::current(),
506                runtime_revision: runtime_revision.clone(),
507                model_runtime_revision: model_runtime_revision.clone(),
508            },
509        )?;
510        self.mappings
511            .insert(runtime_revision, model_runtime_revision);
512        Ok(())
513    }
514
515    fn resolve(
516        &self,
517        runtime_revision: &cookie_agent_protocol::RuntimeRevision,
518    ) -> Option<cookie_agent_protocol::RuntimeRevision> {
519        self.mappings.get(runtime_revision).cloned()
520    }
521}

696            Self::Start { .. } => Some(CompactionDeferredKind::Start),
697            Self::PromotePendingInputs { .. } => Some(CompactionDeferredKind::PromotePendingInputs),
698            Self::PromotePendingOrComplete { .. } => {
699                Some(CompactionDeferredKind::PromotePendingOrComplete)
700            }
701            Self::Resume { .. } => Some(CompactionDeferredKind::Resume),
702            _ => None,
703        }
704    }
705
706    fn reject_duplicate_during_compaction(self, session: SessionId) {
707        match self {
708            Self::Start { reply, .. } => {
709                let _ = reply.send(Err(EngineError::SessionRunning(session)));
710            }
711            Self::PromotePendingInputs { reply, .. } => {
712                let _ = reply.send(Err(EngineError::SessionRunning(session)));
713            }
714            Self::PromotePendingOrComplete { reply, .. } => {
715                let _ = reply.send(Ok(false));
716            }
717            Self::Resume { reply } => {
718                let _ = reply.send(Err(EngineError::SessionRunning(session)));
719            }
720            _ => unreachable!("only barrier-sensitive commands are superseded"),
721        }
722    }

863                tools: Mutex::new(tools),
864                provider_ids: Mutex::new(provider_ids),

866                plugins,
867                mcp_mutation: tokio::sync::Mutex::new(()),

918            engine.inner.plugins.start_eager(runtime);
919        }
920        engine.start_subagent_janitor();

922    }

1229    /// Startup manifest pass: root logs only (§4.2). Child logs are validated by
1230    /// the same logic when their tree is loaded, so the fail-closed shape is
1231    /// unchanged — only deferred to the access that needs the child.
1232    fn validate_root_manifests(&self) -> Result<(), EngineError> {
1233        let runtime = self.current_runtime();
1234        let mut bindings = Vec::new();
1235        for session in self.inner.store.root_snapshots() {
1236            for event in session.log.event_snapshot().iter() {
1237                match &event.payload {
1238                    Event::SessionCreated { creation_agent, .. } => bindings.extend(
1239                        creation_agent
1240                            .fallback_chain
1241                            .iter()
1242                            .map(|binding| (session.meta.session_id, binding.clone())),
1243                    ),
1244                    Event::RunStarted {
1245                        selected_suffix, ..
1246                    } => bindings.extend(
1247                        selected_suffix
1248                            .iter()
1249                            .map(|binding| (session.meta.session_id, binding.clone())),
1250                    ),
1251                    _ => {}
1252                }
1253            }
1254        }
1255        self.validate_manifest_bindings(&bindings)?;
1256        for entry in self.inner.delegation_events.entries() {
1257            let manifest = runtime
1258                .manifests
1259                .require(&entry.revisions.manifest_revision)?;
1260            if manifest.payload.catalog_revision != entry.revisions.catalog_revision
1261                || manifest.payload.provider_state_revision
1262                    != entry.revisions.provider_state_revision
1263                || manifest.payload.model_revision != entry.revisions.model_revision
1264                || manifest.payload.recipe_registry_revision
1265                    != entry.revisions.recipe_registry_revision
1266                || crate::runtime_snapshot::projection::runtime_revision(
1267                    &entry.revisions.recipe_registry_revision,
1268                    &entry.revisions.catalog_revision,
1269                    &entry.revisions.provider_state_revision,
1270                    &entry.revisions.model_revision,
1271                    &entry.revisions.agent_revision,
1272                )? != entry.revisions.runtime_revision
1273            {
1274                return Err(EngineError::RuntimeCompileFailed);
1275            }
1276            for binding in &entry.selected_suffix {
1277                self.validate_manifest_bindings(&[(
1278                    entry.reservation.child_session_id,
1279                    binding.clone(),
1280                )])?;
1281            }
1282        }
1283        Ok(())
1284    }

1286    /// The single acceptance rule for a referenced model binding: an unusable
1287    /// snapshot reference is tolerated, anything else fails.
1288    fn validate_manifest_bindings(
1289        &self,
1290        bindings: &[(SessionId, cookie_agent_protocol::FrozenModelBinding)],
1291    ) -> Result<(), EngineError> {
1292        let runtime = self.current_runtime();
1293        for (_session_id, binding) in bindings {
1294            let validation =
1295                validate_referenced_binding(&runtime.manifests, &runtime.models, binding);
1296            if !matches!(
1297                &validation,
1298                Ok(())
1299                    | Err(EngineError::SnapshotRehydration(
1300                        RehydrationError::SnapshotConfigMismatch

1310    /// Receives the products of a lazy tree load and folds them into the engine
1311    /// singletons (§3.3(d)). Validation runs first so a rejected load cannot
1312    /// leave half-applied state behind, and the artifact router is told the tree
1313    /// is loaded last: until this function returns `Ok` the load is not a
1314    /// completed one, and a sweep must still treat the child logs as unproven.
1315    fn apply_tree_load(
1316        &self,
1317        products: Arc<crate::session::TreeLoadProducts>,
1318    ) -> Result<(), EngineError> {
1319        if let Err(error) = self.validate_manifest_bindings(&products.bindings) {
1320            eprintln!("session tree {} rejected: {error}", products.root);
1321            return Err(error);
1322        }
1323        let extended = self
1324            .inner
1325            .delegation_events
1326            .extend_from_payloads(&products.delegations)?;
1327        if !extended.is_empty() {
1328            // The registry is a fold over the whole event store, and its rebuild
1329            // retains only what it is handed; rebuilding from all entries is now
1330            // cheap because no child log is read to do it (§4.1).
1331            self.rebuild_delegation_registry(&self.inner.delegation_events.entries(), false)?;
1332        }
1333        let invalidated = self.inner.grant_journal.invalidated_ids();
1334        for grant in &products.grants {
1335            if !invalidated.contains(&grant.grant_id) {
1336                self.inner.approvals.store.grant(grant.clone());
1337            }
1338        }
1339        self.reconcile_loaded_tree_producers(products.producer_projections.clone());
1340        // Its child logs are now part of the durable live set for artifact
1341        // collection, and its own directory may be collected: the set the one
1342        // fold harvested replaces any later child-log scan (§3.3(b), §5.2).
1343        self.inner.artifacts.note_tree_live_refs(
1344            products.root,
1345            products.artifact_refs.clone(),
1346            products.child_log_fingerprints.clone(),
1347        );
1348        self.inner.artifacts.note_tree_loaded(products.root);
1349        Ok(())
1350    }

1352    /// Installs the store-side hook and delivers everything a load that already
1353    /// completed was holding back (§3.3, D5).
1354    fn install_tree_load_observer(&self) -> Result<(), EngineError> {
1355        struct Observer(std::sync::Weak<Inner>);
1356
1357        impl crate::session::TreeLoadObserver for Observer {
1358            fn tree_loaded(
1359                &self,
1360                products: Arc<crate::session::TreeLoadProducts>,
1361            ) -> Result<(), EngineError> {
1362                let inner = self.0.upgrade().ok_or(EngineError::ActorStopped)?;
1363                Engine { inner }.apply_tree_load(products)
1364            }
1365        }
1366
1367        self.inner
1368            .store
1369            .set_tree_load_observer(Arc::new(Observer(Arc::downgrade(&self.inner))))?;
1370        Ok(())
1371    }

1382    /// Producer reconciliation for children surfaced by a tree load. Runs on the
1383    /// engine runtime when one is available; otherwise the periodic plugin
1384    /// producer scan picks them up, since loaded trees cache their hits (§4.4).
1385    fn reconcile_loaded_tree_producers(
1386        &self,
1387        projections: Vec<(SessionId, crate::goal_projection::GoalProducerProjection)>,
1388    ) {
1389        if projections.is_empty() {
1390            return;
1391        }
1392        let Some(handle) = self
1393            .inner
1394            .runtime
1395            .clone()
1396            .or_else(|| tokio::runtime::Handle::try_current().ok())
1397        else {
1398            return;
1399        };
1400        let weak = Arc::downgrade(&self.inner);
1401        handle.spawn(async move {
1402            for (session, projection) in projections {
1403                let Some(inner) = weak.upgrade() else {
1404                    return;
1405                };
1406                let _ = Engine { inner }
1407                    .request(session, |reply| {
1408                        SessionCommand::Producer(
1409                            crate::runtime::producers::ProducerCommand::Reconcile {
1410                                projection: Some(projection.clone()),
1411                                reply,
1412                            },
1413                        )
1414                    })
1415                    .await;
1416            }
1417        });
1418    }

1508    /// Stops new session mailbox traffic and cancels active work. Existing
1509    /// client clones may keep a session mailbox alive.
1510    pub async fn shutdown(&self) {
1511        self.inner
1512            .delegation
1513            .admission_tasks_closing
1514            .store(true, Ordering::Release);
1515        let janitor = self
1516            .inner
1517            .janitor_task
1518            .lock()
1519            .ok()
1520            .and_then(|mut task| task.take());
1521        if let Some(task) = janitor {
1522            task.abort();
1523            let _ = task.await;
1524        }
1525        let tasks = self
1526            .inner
1527            .delegation
1528            .admission_tasks
1529            .lock()
1530            .map(|mut tasks| tasks.drain(..).collect::<Vec<_>>())
1531            .unwrap_or_default();
1532        for task in &tasks {
1533            task.abort();
1534        }
1535        for task in tasks {
1536            if let Err(error) = task.await
1537                && !error.is_cancelled()
1538            {
1539                eprintln!("admission task stopped during shutdown: {error}");
1540            }
1541        }
1542        let blocking_tasks = self
1543            .inner
1544            .delegation
1545            .admission_blocking_tasks
1546            .lock()
1547            .map(|mut tasks| tasks.drain(..).collect::<Vec<_>>())
1548            .unwrap_or_default();
1549        for task in blocking_tasks {
1550            if let Err(error) = task.await {
1551                eprintln!("admission blocking task stopped during shutdown: {error}");
1552            }
1553        }
1554        let active: Vec<_> = self
1555            .inner
1556            .sessions
1557            .active
1558            .lock()
1559            .unwrap_or_else(|poisoned| poisoned.into_inner())
1560            .values()
1561            .cloned()
1562            .collect();
1563        for run in active {
1564            run.cancellation.cancel();
1565        }
1566        // Joined here, before the plugin and MCP transports go away and well
1567        // before the actors are cleared: a run task may be mid tool call, and
1568        // it needs live transports to unwind and a live store to append its
1569        // `RunCancelled`. Anything still running when the bound expires is
1570        // aborted, so shutdown stays bounded.
1571        self.join_run_tasks(RUN_TASK_SHUTDOWN_TIMEOUT).await;
1572        self.inner.plugins.shutdown().await;
1573        self.inner.mcp.shutdown().await;
1574        self.inner
1575            .plugin_diagnostics
1576            .accumulator
1577            .shutdown
1578            .store(true, Ordering::Release);
1579        self.inner
1580            .plugin_diagnostics
1581            .accumulator
1582            .notify
1583            .notify_one();
1584        let diagnostic_task = self
1585            .inner
1586            .plugin_diagnostics
1587            .task
1588            .lock()
1589            .ok()
1590            .and_then(|mut task| task.take());
1591        if let Some(mut task) = diagnostic_task
1592            && tokio::time::timeout(
1593                plugin_diagnostics::PLUGIN_DIAGNOSTIC_SHUTDOWN_TIMEOUT,
1594                &mut task,
1595            )
1596            .await
1597            .is_err()
1598        {
1599            for plugin in self.inner.plugin_diagnostics.accumulator.offenders() {
1600                self.inner.plugins.note_offender_diagnostic(
1601                    &plugin,
1602                    "plugin diagnostic drain incomplete: shutdown deadline exceeded".into(),
1603                );
1604            }
1605            task.abort();
1606            let _ = task.await;
1607        }
1608        self.inner
1609            .sessions
1610            .actors
1611            .lock()
1612            .unwrap_or_else(|poisoned| poisoned.into_inner())
1613            .clear();
1614        self.inner
1615            .tools
1616            .lock()
1617            .unwrap_or_else(|poisoned| poisoned.into_inner())
1618            .clear();
1619        self.inner.store.release_ownership();
1620    }
1621}
