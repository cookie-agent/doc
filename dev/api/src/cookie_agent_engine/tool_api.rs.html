129    sanitize_tool_display(text, cookie_agent_protocol::MAX_TOOL_DISPLAY_BYTES)

132pub(crate) fn sanitize_tool_display(text: &str, maximum: usize) -> String {
133    let mut display = String::new();
134    for character in text.chars() {
135        let character = if character.is_control() && !matches!(character, '\n' | '\t') {
136            ' '
137        } else {
138            character
139        };
140        if display.len() + character.len_utf8() > maximum {
141            break;
142        }
143        display.push(character);
144    }
145    display
146}

188impl ProgressSink {
189    #[cfg(feature = "test-support")]
190    pub async fn for_test(
191        sender: mpsc::Sender<ToolProgress>,
192        output: OutputHub,

194        declaration: cookie_agent_protocol::ToolOutputDeclaration,
195    ) -> Result<Self, ToolError> {
196        let store =
197            ArtifactStore::open(directory).map_err(|e| ToolError::execution(e.to_string()))?;
198        let capture = OutputCapture::new(store, declaration, 2_000, 16 * 1024).await?;
199        Ok(Self::with_capture(sender, output, capture))
200    }

204        Self {

206            output,
207            capture: None,
208            display_used: Arc::new(std::sync::Mutex::new(0)),
209        }

211    pub(crate) fn with_capture(
212        sender: mpsc::Sender<ToolProgress>,
213        output: OutputHub,
214        capture: OutputCapture,
215    ) -> Self {
216        Self {
217            sender,
218            output,
219            capture: Some(capture),
220            display_used: Arc::new(std::sync::Mutex::new(0)),
221        }
222    }
223    pub async fn send(&self, mut progress: ToolProgress) -> Result<(), ToolError> {
224        if !progress.output.is_empty() && self.capture.is_none() {
225            return Err(ToolError::execution("output capture is unavailable"));
226        }
227        if progress.message.len() > cookie_agent_protocol::SafeDisplayText::MAX_BYTES
228            || progress.display.as_ref().is_some_and(|display| {
229                display.len() > cookie_agent_protocol::SafeDisplayText::MAX_BYTES
230            })
231        {
232            return Err(ToolError::resource_limit(
233                "tool display delta exceeds the display event bound",
234            ));
235        }
236        if progress.output.len() > cookie_agent_protocol::MAX_TOOL_STREAMS
237            || progress.output.iter().fold(0_usize, |total, chunk| {
238                total.saturating_add(chunk.text.len())
239            }) > cookie_agent_protocol::MAX_TOOL_DELTA_BYTES
240        {
241            return Err(ToolError::resource_limit(
242                "tool output delta exceeds 64 KiB or 8 chunks",
243            ));
244        }
245        for chunk in &progress.output {
246            if let Some(name) = &chunk.stream {
247                cookie_agent_protocol::validate_tool_stream_name(name)
248                    .map_err(ToolError::execution)?;
249            }
250        }
251        {
252            let mut used = self.display_used.lock().unwrap_or_else(|p| p.into_inner());
253            let remaining = cookie_agent_protocol::MAX_TOOL_DISPLAY_BYTES.saturating_sub(*used);
254            progress.message = sanitize_tool_display(&progress.message, remaining);
255            *used += progress.message.len();
256            progress.display = progress
257                .display
258                .as_deref()
259                .map(|text| {
260                    sanitize_tool_display(
261                        text,
262                        cookie_agent_protocol::MAX_TOOL_DISPLAY_BYTES.saturating_sub(*used),
263                    )
264                })
265                .filter(|text| !text.is_empty());
266            *used += progress.display.as_ref().map_or(0, String::len);
267        }
268        let permit = if progress.message.is_empty() && progress.display.is_none() {
269            None
270        } else {
271            Some(
272                self.sender
273                    .clone()
274                    .reserve_owned()
275                    .await
276                    .map_err(|_| ToolError::ProgressSinkClosed)?,
277            )
278        };
279        if let Some(capture) = &self.capture {
280            return capture.emit(progress, self.output.clone(), permit).await;
281        }
282        if let Some(permit) = permit {
283            permit.send(progress);
284        }
285        Ok(())
286    }

422pub enum ToolError {

424    ProgressSinkClosed,
425    #[error("tool failed: {0}")]

427    #[error("prepared operation changed: {0}")]
428    OperationChanged(String),
429    #[error("unsupported prepared-operation security: {0}")]
430    UnsupportedSecurity(String),
431    #[error("prepared operation is unsupported on this platform: {0}")]
432    UnsupportedPlatform(String),
433    #[error("prepared capability resource limit exceeded: {0}")]
434    ResourceLimit(String),
435    #[error("invalid URL: {0}")]
436    InvalidUrl(String),
437    #[error("permission denied: {0}")]
438    PermissionDenied(String),
439    #[error("redirect error: {0}")]
440    RedirectError(String),
441    #[error("request timed out: {0}")]
442    Timeout(String),
443    #[error("transport error: {0}")]
444    TransportError(String),
445    #[error("unsupported content type: {0}")]
446    UnsupportedContentType(String),
447}

449impl ToolError {

452        Self::OperationChanged(message.into())

456    pub fn unsupported_security(message: impl Into<String>) -> Self {

465    #[must_use]
466    pub fn resource_limit(message: impl Into<String>) -> Self {
467        Self::ResourceLimit(message.into())
468    }
469
470    #[must_use]
471    pub fn execution(message: impl Into<String>) -> Self {
472        Self::Failed(message.into())
473    }
474
475    #[must_use]
476    pub(crate) const fn code(&self) -> ToolCallFailureCode {
477        match self {
478            Self::ProgressSinkClosed | Self::Failed(_) => ToolCallFailureCode::ExecutionFailed,
479            Self::OperationChanged(_) => ToolCallFailureCode::OperationChanged,
480            Self::UnsupportedSecurity(_) | Self::ResourceLimit(_) => {
481                ToolCallFailureCode::ExecutionFailed
482            }
483            Self::UnsupportedPlatform(_) => ToolCallFailureCode::UnsupportedPlatform,
484            Self::InvalidUrl(_) => ToolCallFailureCode::InvalidUrl,
485            Self::PermissionDenied(_) => ToolCallFailureCode::PermissionDenied,
486            Self::RedirectError(_) => ToolCallFailureCode::RedirectError,
487            Self::Timeout(_) => ToolCallFailureCode::Timeout,
488            Self::TransportError(_) => ToolCallFailureCode::TransportError,
489            Self::UnsupportedContentType(_) => ToolCallFailureCode::UnsupportedContentType,
490        }
491    }
492
493    #[must_use]
494    pub fn message(&self) -> String {
495        match self {
496            Self::ProgressSinkClosed => "tool progress sink closed".into(),
497            Self::Failed(message)
498            | Self::OperationChanged(message)
499            | Self::UnsupportedSecurity(message)
500            | Self::UnsupportedPlatform(message)
501            | Self::InvalidUrl(message)
502            | Self::PermissionDenied(message)
503            | Self::RedirectError(message)
504            | Self::Timeout(message)
505            | Self::TransportError(message)
506            | Self::UnsupportedContentType(message)
507            | Self::ResourceLimit(message) => message.clone(),
508        }
509    }

532pub struct PreparedTool {
533    pub(crate) operation: PreparedOperationIdentity,
534    pub(crate) policy_labels: Vec<Option<String>>,
535    pub(crate) normalized_arguments: serde_json::Value,
536    pub(crate) serialization_key: Option<PreparedSerializationKey>,
537    pub(crate) executor: PreparedExecutorCell,
538}
539
540pub(crate) type PreparedExecutorCell = Arc<tokio::sync::Mutex<Option<Box<dyn PreparedExecutor>>>>;
541
542impl PreparedTool {
543    pub fn new(
544        operation: PreparedOperationIdentity,
545        normalized_arguments: serde_json::Value,
546        serialization_key: Option<PreparedSerializationKey>,
547        executor: Box<dyn PreparedExecutor>,
548    ) -> Result<Self, ToolError> {
549        if operation.resources().is_empty() {
550            return Err(ToolError::execution(
551                "prepared tool requires at least one permission resource",
552            ));
553        }
554        if normalized_arguments.is_null() {
555            return Err(ToolError::execution(
556                "prepared normalized arguments must not be null",
557            ));
558        }
559        let policy_labels = operation
560            .resources()
561            .iter()
562            .map(|resource| Some(resource.canonical.as_str().to_owned()))
563            .collect();
564        Ok(Self {
565            operation,
566            policy_labels,
567            normalized_arguments,
568            serialization_key,
569            executor: Arc::new(tokio::sync::Mutex::new(Some(executor))),
570        })
571    }
572
573    #[must_use]
574    pub const fn operation(&self) -> &PreparedOperationIdentity {
575        &self.operation
576    }
577
578    pub fn with_policy_labels(mut self, labels: Vec<String>) -> Result<Self, ToolError> {
579        if labels.is_empty() || labels.len() != self.operation.resources().len() {
580            return Err(ToolError::execution(
581                "prepared policy labels must cover every resource",
582            ));
583        }
584        for (resource, label) in self.operation.resources().iter().zip(&labels) {
585            let expected = Sha256Digest::of_bytes(label.as_bytes());
586            if resource
587                .canonical
588                .as_str()
589                .rsplit_once(':')
590                .is_none_or(|(_, digest)| digest != expected.as_str())
591            {
592                return Err(ToolError::execution(
593                    "prepared policy label does not match its immutable resource identity",
594                ));
595            }
596        }
597        self.policy_labels = labels.into_iter().map(Some).collect();
598        Ok(self)
599    }
600
601    pub fn with_permission_resource(mut self, resource: Option<String>) -> Result<Self, ToolError> {
602        if resource.as_ref().is_some_and(String::is_empty) {
603            return Err(ToolError::execution(
604                "permission resource must not be empty",
605            ));
606        }
607        self.policy_labels.fill(resource);
608        Ok(self)
609    }
610
611    #[must_use]
612    pub const fn normalized_arguments(&self) -> &serde_json::Value {
613        &self.normalized_arguments
614    }
615
616    #[must_use]
617    pub fn policy_labels(&self) -> &[Option<String>] {
618        &self.policy_labels
619    }

621    #[cfg(feature = "test-support")]
622    pub async fn execute_for_test(
623        self,
624        context: ToolExecutionContext,
625    ) -> Result<ToolResult, ToolError> {
626        let executor = self
627            .executor
628            .lock()
629            .await
630            .take()
631            .ok_or_else(|| ToolError::execution("prepared executor was already consumed"))?;
632        executor.execute(context).await?.into_result_for_test()
633    }
