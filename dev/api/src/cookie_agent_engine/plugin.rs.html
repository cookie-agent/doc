172impl Drop for ProducerConnectionLease {
173    fn drop(&mut self) {
174        let mut connection = self
175            .0
176            .producer_connection
177            .lock()
178            .unwrap_or_else(|p| p.into_inner());
179        if connection.epoch != Some(self.1) {
180            return;
181        }
182        connection.epoch = None;
183        connection.live = false;
184        if connection
185            .recovery
186            .is_some_and(|status| status != PluginRecoveryStatus::Disabled)
187        {
188            connection.recovery = Some(PluginRecoveryStatus::Failed);
189        }
190        drop(connection);
191        self.0
192            .producer_changes
193            .send_modify(|revision| *revision = revision.wrapping_add(1));
194    }
195}

316impl PluginNotificationQueue {
317    fn new(capacity: usize) -> Self {
318        Self {
319            capacity: capacity.max(1),
320            queued: Mutex::new(VecDeque::new()),
321            notify: Notify::new(),
322        }
323    }
324
325    fn push(&self, control: Control, class: PluginDeliveryClass) -> Option<PluginDeliveryClass> {
326        let mut queued = self
327            .queued
328            .lock()
329            .unwrap_or_else(|poisoned| poisoned.into_inner());
330        let dropped = if queued.len() < self.capacity {
331            None
332        } else {
333            let index = match class {
334                PluginDeliveryClass::Chunk => queued
335                    .iter()
336                    .position(|item| item.class == PluginDeliveryClass::Chunk),
337                PluginDeliveryClass::Ordinary => queued
338                    .iter()
339                    .position(|item| item.class == PluginDeliveryClass::Chunk)
340                    .or_else(|| {
341                        queued
342                            .iter()
343                            .position(|item| item.class == PluginDeliveryClass::Ordinary)
344                    }),
345                PluginDeliveryClass::Terminal => queued
346                    .iter()
347                    .position(|item| item.class == PluginDeliveryClass::Chunk)
348                    .or_else(|| {
349                        queued
350                            .iter()
351                            .position(|item| item.class == PluginDeliveryClass::Ordinary)
352                    })
353                    .or_else(|| {
354                        queued
355                            .iter()
356                            .position(|item| item.class == PluginDeliveryClass::Terminal)
357                    }),
358            };
359            let Some(index) = index else {
360                return Some(class);
361            };
362            queued.remove(index).map(|item| item.class)
363        };
364        queued.push_back(QueuedPluginNotification { control, class });
365        drop(queued);
366        self.notify.notify_one();
367        dropped
368    }
369
370    #[cfg(test)]
371    fn pop(&self) -> Option<Control> {
372        self.queued
373            .lock()
374            .unwrap_or_else(|poisoned| poisoned.into_inner())
375            .pop_front()
376            .map(|item| item.control)
377    }
378
379    async fn notified(&self) {
380        self.notify.notified().await;
381    }
382
383    fn drain(&self) -> VecDeque<Control> {
384        std::mem::take(
385            &mut *self
386                .queued
387                .lock()
388                .unwrap_or_else(|poisoned| poisoned.into_inner()),
389        )
390        .into_iter()
391        .map(|item| item.control)
392        .collect()
393    }
394
395    fn clear(&self) {
396        self.queued
397            .lock()
398            .unwrap_or_else(|poisoned| poisoned.into_inner())
399            .clear();
400    }

510                            PluginRecoveryStatus::Starting
511                        } else {
512                            PluginRecoveryStatus::Disabled
513                        }),
514                        outcome: None,
515                    }),
516                    producer_handler: Arc::clone(&producer_handler),
517                    producer_changes: producer_changes.clone(),
518                    config,
519                    status: Mutex::new(PluginStatus {
520                        plugin: name.clone(),
521                        state,
522                        reason: None,
523                        tools: Vec::new(),
524                        dropped_events: 0,
525                    }),
526                    control: Mutex::new(None),
527                    notifications: Mutex::new(Arc::new(PluginNotificationQueue::new(
528                        PLUGIN_NOTIFICATION_CAPACITY,
529                    ))),
530                    declarations: Mutex::new(Vec::new()),
531                    capabilities: Mutex::new(None),
532                    dropped_events: AtomicU64::new(0),
533                    forced_failure: Mutex::new(None),
534                    forced_failure_notify: Notify::new(),
535                    ready: Arc::clone(&ready),
536                    mcp: Arc::clone(&mcp),
537                    emit_handler: Arc::clone(&emit_handler),
538                    contexts: Mutex::new(PluginContexts::default()),
539                    publish_quotas: Mutex::new(HashMap::new()),
540                });
541                (name, runtime)
542            })
543            .collect();
544        let inner = Arc::new(PluginRegistryInner {
545            plugins,
546            tasks: Mutex::new(Vec::new()),
547            ready,
548            emit_handler,
549            producer_handler,
550            producer_changes,
551        });
552        let weak = Arc::downgrade(&inner);
553        mcp.set_plugin_collision_handler(Arc::new(move |plugin, tool, replacement| {
554            let Some(inner) = weak.upgrade() else {
555                return;
556            };
557            let Some(runtime) = inner.plugins.get(plugin) else {
558                return;
559            };
560            if let Some(replacement) = replacement {
561                let mut status = runtime
562                    .status
563                    .lock()
564                    .unwrap_or_else(|poisoned| poisoned.into_inner());
565                status.tools.retain(|name| name != tool);
566                status.reason = Some(format!(
567                    "tool `{tool}` was replaced by later plugin `{replacement}`"
568                ));
569                return;
570            }
571            *runtime
572                .forced_failure
573                .lock()
574                .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(format!(
575                "plugin `{plugin}` tool `{tool}` was preempted by MCP"
576            ));
577            runtime.forced_failure_notify.notify_one();
578        }));
579        Self { inner }
580    }

728    ) -> Result<(), JsonRpcError> {
729        let plugin = self
730            .inner
731            .plugins
732            .get(&authority.plugin)
733            .ok_or_else(|| producer_error(-32600, "unknown plugin connection"))?;
734        let mut connection = plugin
735            .producer_connection
736            .lock()
737            .unwrap_or_else(|p| p.into_inner());
738        if !connection.live || connection.epoch != Some(authority.connection_epoch) {
739            return Err(producer_error(-32600, "stale plugin connection"));
740        }
741        if let Some(previous) = &connection.outcome {
742            return if previous == outcome {
743                Ok(())
744            } else {
745                Err(producer_error(-32600, "recovery outcome already reported"))
746            };
747        }
748        connection.recovery = Some(match outcome {
749            ExtensionRecoveryOutcome::Ready => PluginRecoveryStatus::Ready,
750            ExtensionRecoveryOutcome::Failed { .. } => PluginRecoveryStatus::Failed,
751        });
752        connection.outcome = Some(outcome.clone());
753        drop(connection);
754        if let ExtensionRecoveryOutcome::Failed { message } = outcome {
755            self.note_offender_diagnostic(&authority.plugin, message.to_string());
756        }
757        plugin
758            .producer_changes
759            .send_modify(|revision| *revision = revision.wrapping_add(1));
760        Ok(())
761    }

763    pub(crate) fn check_publish_quota(
764        &self,
765        plugin: &str,
766        session_id: cookie_agent_protocol::SessionId,
767        event_bytes: usize,
768    ) -> Result<(), &'static str> {
769        let runtime = self
770            .inner
771            .plugins
772            .get(plugin)
773            .ok_or("plugin is not registered")?;
774        let now = Instant::now();
775        let mut quotas = runtime
776            .publish_quotas
777            .lock()
778            .unwrap_or_else(|poisoned| poisoned.into_inner());
779        let quota = quotas.entry(session_id).or_insert(PublishQuota {
780            second_started: now,
781            minute_started: now,
782            events_this_second: 0,
783            bytes_this_minute: 0,
784        });
785        if now.duration_since(quota.second_started) >= Duration::from_secs(1) {
786            quota.second_started = now;
787            quota.events_this_second = 0;
788        }
789        if now.duration_since(quota.minute_started) >= Duration::from_secs(60) {
790            quota.minute_started = now;
791            quota.bytes_this_minute = 0;
792        }
793        if quota.events_this_second >= PLUGIN_EVENTS_PER_SECOND {
794            return Err("plugin event rate exceeds 40 events per second");
795        }
796        if quota.bytes_this_minute.saturating_add(event_bytes) > PLUGIN_BYTES_PER_MINUTE {
797            return Err("plugin event volume exceeds 4 MiB per minute");
798        }
799        quota.events_this_second += 1;
800        quota.bytes_this_minute += event_bytes;
801        Ok(())
802    }

913        notification: Notification,
914        session_id: cookie_agent_protocol::SessionId,
915        context_id: &str,

917        class: PluginDeliveryClass,
918        subscribed: impl Fn(&ExtensionPluginCapabilities) -> bool,

921        for runtime in self.inner.plugins.values() {
922            if source_plugin == Some(runtime.name.as_str()) {
923                continue;
924            }
925            let enabled = runtime
926                .capabilities
927                .lock()
928                .unwrap_or_else(|poisoned| poisoned.into_inner())
929                .as_ref()
930                .is_some_and(&subscribed);
931            if !enabled {
932                continue;
933            }
934            if runtime
935                .control
936                .lock()
937                .unwrap_or_else(|poisoned| poisoned.into_inner())
938                .is_none()

940                continue;
941            }

943                .notifications
944                .lock()

946                .clone();

1002        runtime: &Arc<PluginRuntime>,
1003        method: &'static str,
1004        params: Value,
1005        session_id: Option<cookie_agent_protocol::SessionId>,

1266    async fn run(
1267        self: Arc<Self>,
1268        receiver: mpsc::Receiver<Control>,
1269        notifications: Arc<PluginNotificationQueue>,
1270    ) {
1271        static NEXT_CONNECTION_EPOCH: AtomicU64 = AtomicU64::new(1);
1272        // A compare-exchange loop rather than `fetch_update`, which Rust 1.99
1273        // deprecates in favour of a `try_update` the 1.88 MSRV lacks.
1274        let mut epoch = NEXT_CONNECTION_EPOCH.load(Ordering::Acquire);
1275        loop {
1276            let next = epoch
1277                .checked_add(1)
1278                .expect("plugin connection epoch space exhausted");
1279            match NEXT_CONNECTION_EPOCH.compare_exchange_weak(
1280                epoch,
1281                next,
1282                Ordering::AcqRel,
1283                Ordering::Acquire,
1284            ) {
1285                Ok(_) => break,
1286                Err(actual) => epoch = actual,
1287            }
1288        }
1289        {
1290            let mut connection = self
1291                .producer_connection
1292                .lock()
1293                .unwrap_or_else(|p| p.into_inner());
1294            connection.epoch = Some(epoch);
1295            connection.live = false;
1296            connection.outcome = None;
1297        }
1298        let lease = ProducerConnectionLease(Arc::clone(&self), epoch);
1299        self.set_status(PluginState::Connecting, None, Vec::new());
1300        let result = self.spawn_and_supervise(receiver, &notifications).await;
1301        drop(lease);
1302        *self
1303            .control
1304            .lock()
1305            .unwrap_or_else(|poisoned| poisoned.into_inner()) = None;
1306        notifications.clear();
1307        *self
1308            .contexts
1309            .lock()
1310            .unwrap_or_else(|poisoned| poisoned.into_inner()) = PluginContexts::default();
1311        self.declarations
1312            .lock()
1313            .unwrap_or_else(|poisoned| poisoned.into_inner())
1314            .clear();
1315        *self
1316            .capabilities
1317            .lock()
1318            .unwrap_or_else(|poisoned| poisoned.into_inner()) = None;
1319        self.mcp.release_plugin_tools(&self.name);
1320        match result {
1321            Ok(()) => self.set_status(PluginState::Disconnected, None, Vec::new()),
1322            Err(reason) => self.set_status(PluginState::Failed, Some(reason), Vec::new()),
1323        }
1324    }

1326    async fn spawn_and_supervise(
1327        self: &Arc<Self>,
1328        receiver: mpsc::Receiver<Control>,
1329        notifications: &Arc<PluginNotificationQueue>,
1330    ) -> Result<(), String> {
1331        let epoch = self
1332            .producer_connection
1333            .lock()
1334            .unwrap_or_else(|p| p.into_inner())
1335            .epoch
1336            .expect("supervisor owns a connection epoch");
1337        let command = self
1338            .config
1339            .command
1340            .as_ref()
1341            .expect("enabled plugins are validated");
1342        let mut process = tokio::process::Command::new(command);
1343        process
1344            .args(&self.config.args)
1345            .env_clear()
1346            .envs(&self.config.env)
1347            .stdin(Stdio::piped())
1348            .stdout(Stdio::piped())
1349            .stderr(Stdio::inherit());
1350        if let Some(cwd) = &self.config.cwd {
1351            process.current_dir(cwd);
1352        }
1353        let mut wrapped = CommandWrap::from(process);
1354        #[cfg(unix)]
1355        wrapped.wrap(ProcessGroup::leader());
1356        #[cfg(windows)]
1357        wrapped.wrap(JobObject);
1358        wrapped.wrap(KillOnDrop);
1359        let mut child = wrapped
1360            .spawn()
1361            .map_err(|error| format!("spawn failure for command `{command}`: {error}"))?;
1362        let result = self
1363            .supervise_spawned(&mut child, receiver, notifications)
1364            .await;
1365        // Revoke before process cleanup, which may wait for the shutdown grace.
1366        drop(ProducerConnectionLease(Arc::clone(self), epoch));
1367        let cleanup = terminate_and_reap(&mut child).await;
1368        match (result, cleanup) {
1369            (Err(reason), Err(cleanup)) => Err(format!("{reason}; cleanup failed: {cleanup}")),
1370            (result, Ok(())) => result,
1371            (Ok(()), Err(cleanup)) => Err(format!("cleanup failed: {cleanup}")),
1372        }
1373    }

1399        self: &Arc<Self>,
1400        child: &mut Box<dyn ChildWrapper>,
1401        mut stdin: tokio::process::ChildStdin,
1402        mut control: mpsc::Receiver<Control>,
1403        notifications: &Arc<PluginNotificationQueue>,
1404        mut inbound: mpsc::Receiver<ReaderEvent>,
1405    ) -> Result<(), String> {
1406        let mut initialize = extension_initialize_request(env!("CARGO_PKG_VERSION"));
1407        let mut params: ExtensionInitializeParams =
1408            serde_json::from_value(initialize.params.take().expect("initialize has params"))
1409                .expect("initialize params are typed");
1410        params.capabilities.producer_messaging = self.config.producer_messaging
1411            && self
1412                .producer_handler
1413                .lock()
1414                .unwrap_or_else(|p| p.into_inner())
1415                .is_some();
1416        initialize.params =
1417            Some(serde_json::to_value(params).expect("initialize params serialize"));
1418        write_json(&mut stdin, &initialize).await?;
1419        let mut pending = HashMap::from([(
1420            1_i64,
1421            PendingRequest::Initialize {
1422                deadline: Instant::now() + Duration::from_millis(self.config.startup_timeout_ms),
1423            },
1424        )]);
1425        let mut next_request_id = 2_i64;
1426        let mut connected = false;
1427        let mut producer_tasks = JoinSet::new();
1428
1429        loop {
1430            if let Some(reason) = self
1431                .forced_failure
1432                .lock()
1433                .unwrap_or_else(|poisoned| poisoned.into_inner())
1434                .take()
1435            {
1436                return Err(reason);
1437            }
1438            let deadline = pending.values().map(PendingRequest::deadline).min();
1439            tokio::select! {
1440                status = child.wait() => {
1441                    return Err(match status {
1442                        Ok(status) => format!("process exited with {status}"),
1443                        Err(error) => format!("process wait failed: {error}"),
1444                    });
1445                }
1446                event = inbound.recv() => match event {
1447                    Some(ReaderEvent::Message(message)) => {
1448                        self.dispatch_message(
1449                            &message,
1450                            &mut stdin,
1451                            &mut pending,
1452                            &mut connected,
1453                            &mut producer_tasks,
1454                        ).await?;
1455                    }
1456                    Some(ReaderEvent::Failed(reason)) => return Err(reason),
1457                    Some(ReaderEvent::Eof) | None => return Err("plugin stdout reached EOF".into()),
1458                },
1459                () = notifications.notified() => {
1460                    for notification in notifications.drain() {
1461                        match notification {
1462                            Control::Notify { notification, session_id, context_id, context_lifetime } if connected => {
1463                                self.register_context(
1464                                    &context_id,
1465                                    session_id,
1466                                    Instant::now() + context_lifetime,
1467                                );
1468                                write_json(&mut stdin, &notification).await?;
1469                            }
1470                            Control::Notify { .. } => {}
1471                            _ => unreachable!("notification queue accepts only notifications"),
1472                        }
1473                    }
1474                },
1475                command = control.recv() => match command {
1476                    Some(Control::ReplyFrame(response)) => write_json(&mut stdin, &response).await?,
1477                    Some(Control::Ping(reply)) if connected => {
1478                        let id = next_request_id;
1479                        next_request_id = next_request_id.checked_add(1)
1480                            .ok_or_else(|| "plugin request id space exhausted".to_owned())?;
1481                        let request = Request::new(
1482                            JsonRpcId::Number(id),
1483                            PLUGIN_PING_METHOD,
1484                            Some(serde_json::to_value(ExtensionPingParams {})
1485                                .expect("ping params serialize")),
1486                        );
1487                        write_json(&mut stdin, &request).await?;
1488                        pending.insert(id, PendingRequest::Ping {
1489                            deadline: Instant::now()
1490                                + Duration::from_millis(self.config.interception_timeout_ms),
1491                            reply,
1492                        });
1493                    }
1494                    Some(Control::Ping(reply)) => {
1495                        let _ = reply.send(Err(format!("plugin `{}` is not connected", self.name)));
1496                    }
1497                    Some(Control::ToolCall { params, context_lifetime, reply }) if connected => {
1498                        self.register_context(
1499                            &params.context_id,
1500                            params.session_id,
1501                            Instant::now() + context_lifetime,
1502                        );
1503                        let id = next_request_id;
1504                        next_request_id = next_request_id.checked_add(1)
1505                            .ok_or_else(|| "plugin request id space exhausted".to_owned())?;
1506                        let request = Request::new(
1507                            JsonRpcId::Number(id),
1508                            PLUGIN_TOOLS_CALL_METHOD,
1509                            Some(serde_json::to_value(params)
1510                                .expect("plugin tool call params serialize")),
1511                        );
1512                        write_json(&mut stdin, &request).await?;
1513                        pending.insert(id, PendingRequest::ToolCall {
1514                            deadline: Instant::now()
1515                                + Duration::from_millis(self.config.tool_timeout_ms),
1516                            reply,
1517                        });
1518                    }
1519                    Some(Control::ToolCall { reply, .. }) => {
1520                        let _ = reply.send(Err(format!("plugin `{}` is not connected", self.name)));
1521                    }
1522                    Some(Control::Notify { notification, session_id, context_id, context_lifetime }) if connected => {
1523                        self.register_context(
1524                            &context_id,
1525                            session_id,
1526                            Instant::now() + context_lifetime,
1527                        );
1528                        write_json(&mut stdin, &notification).await?;
1529                    }
1530                    Some(Control::Notify { .. }) => {}
1531                    Some(Control::Intercept { method, params, session_id, context_id, context_lifetime, reply }) if connected => {
1532                        if let (Some(session_id), Some(context_id), Some(context_lifetime)) =
1533                            (session_id, context_id.as_deref(), context_lifetime)
1534                        {
1535                            self.register_context(
1536                                context_id,
1537                                session_id,
1538                                Instant::now() + context_lifetime,
1539                            );
1540                        }
1541                        let id = next_request_id;
1542                        next_request_id = next_request_id.checked_add(1)
1543                            .ok_or_else(|| "plugin request id space exhausted".to_owned())?;
1544                        write_json(&mut stdin, &Request::new(
1545                            JsonRpcId::Number(id), method, Some(params),
1546                        )).await?;
1547                        pending.insert(id, PendingRequest::Intercept {
1548                            deadline: Instant::now()
1549                                + Duration::from_millis(self.config.interception_timeout_ms),
1550                            reply,
1551                        });
1552                    }
1553                    Some(Control::Intercept { reply, .. }) => {
1554                        let _ = reply.send(Err(format!("plugin `{}` is not connected", self.name)));
1555                    }
1556                    Some(Control::Shutdown) | None => {
1557                        return graceful_shutdown(
1558                            child,
1559                            &mut stdin,
1560                            self.config.shutdown_grace_ms,
1561                        ).await;
1562                    }
1563                },
1564                () = self.forced_failure_notify.notified() => {}
1565                completed = producer_tasks.join_next(), if !producer_tasks.is_empty() => {
1566                    completed.expect("task set is nonempty")
1567                        .map_err(|error| format!("plugin producer handler failed: {error}"))?;
1568                }
1569                () = wait_for_deadline(deadline) => {
1570                    let now = Instant::now();
1571                    let expired = pending
1572                        .iter()
1573                        .filter_map(|(id, request)| (request.deadline() <= now).then_some(*id))
1574                        .collect::<Vec<_>>();
1575                    for id in expired {
1576                        match pending.remove(&id).expect("expired pending request exists") {
1577                            PendingRequest::Initialize { .. } => return Err("handshake timeout".into()),
1578                            PendingRequest::Ping { reply, .. } => {
1579                                let _ = reply.send(Err("plugin ping timed out".into()));
1580                            }
1581                            PendingRequest::ToolCall { reply, .. } => {
1582                                let _ = reply.send(Err("plugin tool call timed out".into()));
1583                            }
1584                            PendingRequest::Intercept { reply, .. } => {
1585                                let _ = reply.send(Err("plugin interception timed out".into()));
1586                            }
1587                        }
1588                    }
1589                }
1590            }
1591        }
1592    }

1594    async fn dispatch_message(
1595        self: &Arc<Self>,
1596        message: &str,
1597        stdin: &mut tokio::process::ChildStdin,
1598        pending: &mut HashMap<i64, PendingRequest>,
1599        connected: &mut bool,
1600        producer_tasks: &mut JoinSet<()>,
1601    ) -> Result<(), String> {
1602        let value: Value = serde_json::from_str(message)
1603            .map_err(|error| format!("malformed plugin message: {error}"))?;
1604        if value.get("method").is_some() {
1605            if value.get("id").is_some() {
1606                let request: Request = serde_json::from_value(value)
1607                    .map_err(|error| format!("malformed plugin request: {error}"))?;
1608                if let Some(response) = self.dispatch_producer_request(request, producer_tasks) {
1609                    write_json(stdin, &response).await?;
1610                }
1611            } else {
1612                let notification = serde_json::from_value::<Notification>(value)
1613                    .map_err(|error| format!("malformed plugin notification: {error}"))?;
1614                if notification.method == PLUGIN_EMIT_METHOD {
1615                    let params: ExtensionEmitParams =
1616                        serde_json::from_value(notification.params.unwrap_or(Value::Null))
1617                            .map_err(|error| {
1618                                format!("malformed plugin emit notification: {error}")
1619                            })?;
1620                    let capabilities = self
1621                        .capabilities
1622                        .lock()
1623                        .unwrap_or_else(|poisoned| poisoned.into_inner())
1624                        .clone();
1625                    let handler = self
1626                        .emit_handler()
1627                        .ok_or_else(|| "plugin event publisher is unavailable".to_owned())?;
1628                    let outcome = if let Some(capabilities) = capabilities {
1629                        let context = self.consume_context(&params.context_id, params.session_id);
1630                        handler(PluginEmitRequest {
1631                            plugin: self.name.clone(),
1632                            session_id: params.session_id,
1633                            context,
1634                            name: params.name.clone(),
1635                            payload: params.payload,
1636                            publish_bus: capabilities.publish_bus,
1637                            publish_session_events: capabilities.publish_session_events,
1638                        })
1639                        .await
1640                    } else {
1641                        PluginEmitOutcome {
1642                            bus: ExtensionEmitStatus::Rejected,
1643                            durable: ExtensionEmitStatus::Rejected,
1644                            reason: Some("plugin is not initialized".into()),
1645                        }
1646                    };
1647                    write_json(
1648                        stdin,
1649                        &Notification::new(
1650                            PLUGIN_EMIT_RESULT_METHOD,
1651                            Some(
1652                                serde_json::to_value(ExtensionEmitResultParams {
1653                                    name: params.name,
1654                                    bus: outcome.bus,
1655                                    durable: outcome.durable,
1656                                    reason: outcome.reason,
1657                                })
1658                                .expect("emit result params serialize"),
1659                            ),
1660                        ),
1661                    )
1662                    .await?;
1663                }
1664            }
1665            return Ok(());
1666        }
1667
1668        let response: Response = serde_json::from_value(value)
1669            .map_err(|error| format!("malformed plugin response: {error}"))?;
1670        let id = match &response {
1671            Response::Success(response) => &response.id,
1672            Response::Error(response) => &response.id,
1673        };
1674        let JsonRpcId::Number(id) = id else {
1675            return Ok(());
1676        };
1677        let Some(request) = pending.remove(id) else {
1678            return Ok(());
1679        };
1680        match request {
1681            PendingRequest::Initialize { .. } => {
1682                let initialize = parse_initialize(response, &self.name)?;
1683                validate_tools(&initialize)?;
1684                let names = initialize
1685                    .tools
1686                    .iter()
1687                    .map(|tool| tool.name.clone())
1688                    .collect::<Vec<_>>();
1689                self.mcp
1690                    .claim_plugin_tools(&self.name, &names)
1691                    .map_err(|error| error.to_string())?;
1692                *self
1693                    .declarations
1694                    .lock()
1695                    .unwrap_or_else(|poisoned| poisoned.into_inner()) = initialize.tools;
1696                *self
1697                    .capabilities
1698                    .lock()
1699                    .unwrap_or_else(|poisoned| poisoned.into_inner()) =
1700                    Some(initialize.capabilities);
1701                self.set_status(PluginState::Connected, None, names);
1702                *connected = true;
1703                let capable = self
1704                    .capabilities
1705                    .lock()
1706                    .unwrap_or_else(|p| p.into_inner())
1707                    .as_ref()
1708                    .is_some_and(|capabilities| capabilities.producer_messaging);
1709                let handler_available = self
1710                    .producer_handler
1711                    .lock()
1712                    .unwrap_or_else(|p| p.into_inner())
1713                    .is_some();
1714                {
1715                    let mut connection = self
1716                        .producer_connection
1717                        .lock()
1718                        .unwrap_or_else(|p| p.into_inner());
1719                    connection.live =
1720                        capable && self.config.producer_messaging && handler_available;
1721                    if capable || self.config.producer_messaging {
1722                        connection.recovery = Some(if connection.live {
1723                            PluginRecoveryStatus::Starting
1724                        } else if capable && self.config.producer_messaging {
1725                            PluginRecoveryStatus::Failed
1726                        } else {
1727                            PluginRecoveryStatus::Disabled
1728                        });
1729                    }
1730                }
1731                self.producer_changes
1732                    .send_modify(|revision| *revision = revision.wrapping_add(1));
1733                if capable && self.config.producer_messaging && handler_available {
1734                    write_json(stdin, &extension_recovery_start_notification()).await?;
1735                }
1736                Ok(())
1737            }
1738            PendingRequest::Ping { reply, .. } => {
1739                let result = parse_ping(response);
1740                let _ = reply.send(result);
1741                Ok(())
1742            }
1743            PendingRequest::ToolCall { reply, .. } => {
1744                let result = parse_tool_call(response);
1745                let _ = reply.send(result);
1746                Ok(())
1747            }
1748            PendingRequest::Intercept { reply, .. } => {
1749                let result = match response {
1750                    Response::Success(success) => Ok(success.result),
1751                    Response::Error(error) => Err(format!(
1752                        "plugin interception rejected: {}",
1753                        error.error.message
1754                    )),
1755                };
1756                let _ = reply.send(result);
1757                Ok(())
1758            }
1759        }
1760    }

1762    fn dispatch_producer_request(
1763        self: &Arc<Self>,
1764        request: Request,
1765        tasks: &mut JoinSet<()>,
1766    ) -> Option<Response> {
1767        let id = request.id;
1768        let reject = |error| Some(producer_response(id.clone(), Err(error)));
1769        let decoded =
1770            match decode_producer_request(&request.method, request.params.unwrap_or(Value::Null)) {
1771                Ok(decoded) => decoded,
1772                Err(error) => return reject(error),
1773            };
1774        let authority = {
1775            let connection = self
1776                .producer_connection
1777                .lock()
1778                .unwrap_or_else(|p| p.into_inner());
1779            if !connection.live {
1780                return reject(producer_error(
1781                    -32600,
1782                    "producer messaging is not enabled on this connection",
1783                ));
1784            }
1785            PluginConnectionAuthority {
1786                plugin: self.name.clone(),
1787                connection_epoch: connection.epoch.expect("live connection has an epoch"),
1788            }
1789        };
1790        let Some(handler) = self
1791            .producer_handler
1792            .lock()
1793            .unwrap_or_else(|p| p.into_inner())
1794            .clone()
1795        else {
1796            return reject(producer_error(
1797                -32601,
1798                "plugin producer handler is unavailable",
1799            ));
1800        };
1801        // Never wait for capacity on the reader: callbacks may themselves need
1802        // replies from this connection. Overload is an explicit rejection.
1803        if tasks.len() >= PLUGIN_PRODUCER_CONCURRENCY {
1804            return reject(producer_error(
1805                -32000,
1806                "plugin producer handler capacity exceeded",
1807            ));
1808        }
1809        let Some(control) = self
1810            .control
1811            .lock()
1812            .unwrap_or_else(|p| p.into_inner())
1813            .clone()
1814        else {
1815            return reject(producer_error(-32600, "plugin connection is closed"));
1816        };
1817        let runtime = Arc::clone(self);
1818        tasks.spawn(async move {
1819            let current = || {
1820                let connection = runtime
1821                    .producer_connection
1822                    .lock()
1823                    .unwrap_or_else(|p| p.into_inner());
1824                connection.live && connection.epoch == Some(authority.connection_epoch)
1825            };
1826            let result = if current() {
1827                handler(authority.clone(), decoded)
1828                    .await
1829                    .and_then(|response| producer_result(&request.method, response))
1830            } else {
1831                Err(producer_error(-32600, "stale plugin connection"))
1832            };
1833            let result = if current() {
1834                result
1835            } else {
1836                Err(producer_error(-32600, "stale plugin connection"))
1837            };
1838            // A closed receiver means the transport disconnected; the caller sees
1839            // EOF, not a fabricated ACK. A full queue is awaited without dropping.
1840            let _ = control
1841                .send(Control::ReplyFrame(producer_response(id, result)))
1842                .await;
1843        });
1844        None
1845    }

1895            serde_json::from_value(params).map(PluginProducerRequest::RecoveryComplete)
1896        }
1897        _ => return Err(producer_error(-32601, "unknown plugin request method")),
1898    };

1900        .map_err(|error| producer_error(-32602, &format!("invalid producer parameters: {error}")))
1901}

1945fn prune_plugin_contexts(contexts: &mut PluginContexts, now: Instant) {
1946    let mut active = VecDeque::with_capacity(contexts.active.len());
1947    while let Some(context) = contexts.active.pop_front() {
1948        if context.expires_at <= now {
1949            let expired = context;
1950            remember_spent_context(contexts, expired.id, expired.session_id, now);
1951        } else {
1952            active.push_back(context);
1953        }
1954    }
1955    contexts.active = active;
1956    contexts.spent.retain(|context| context.expires_at > now);
1957}

1959fn remember_spent_context(
1960    contexts: &mut PluginContexts,
1961    id: String,
1962    session_id: cookie_agent_protocol::SessionId,
1963    now: Instant,
1964) {
1965    contexts.spent.push_back(PluginSpentContext {
1966        id,
1967        session_id,
1968        expires_at: now + PLUGIN_SPENT_CONTEXT_LIFETIME,
1969    });
1970    if contexts.spent.len() > PLUGIN_SPENT_CONTEXT_CAPACITY {
1971        contexts.spent.pop_front();
1972    }
1973}

1979fn spawn_reader(
1980    stdout: tokio::process::ChildStdout,
1981) -> (JoinHandle<()>, mpsc::Receiver<ReaderEvent>) {
1982    let (sender, receiver) = mpsc::channel(32);
1983    let task = tokio::spawn(async move {
1984        let mut frames = FramedRead::new(stdout, LinesCodec::new_with_max_length(MAX_FRAME_BYTES));
1985        while let Some(frame) = frames.next().await {
1986            let event = match frame {
1987                Ok(message) => ReaderEvent::Message(message),
1988                Err(error) => ReaderEvent::Failed(format!(
1989                    "plugin stdout frame is invalid UTF-8 or exceeds {MAX_FRAME_BYTES} bytes: {error}"
1990                )),
1991            };
1992            let failed = matches!(event, ReaderEvent::Failed(_));
1993            if sender.send(event).await.is_err() || failed {
1994                return;
1995            }
1996        }
1997        let _ = sender.send(ReaderEvent::Eof).await;
1998    });
1999    (task, receiver)
2000}

2002fn parse_initialize(
2003    response: Response,
2004    expected_name: &str,
2005) -> Result<ExtensionInitializeResult, String> {
2006    let success = match response {
2007        Response::Success(success) => success,
2008        Response::Error(error) => {
2009            return Err(format!(
2010                "handshake rejected: {}",
2011                cookie_agent_protocol::diagnostics::rpc(&error.error)
2012            ));
2013        }
2014    };
2015    let found_version = success
2016        .result
2017        .get("protocol_version")

2302    let mut names = HashSet::new();
2303    for tool in &initialize.tools {

2320            format!(

2326            .as_value()

2332        {
2333            return Err(format!(
