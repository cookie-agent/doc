38/// How clients show a call of a base tool: `title`, `hint`, `primary` beside
39/// the title, and the call's arguments as `entries`.
40pub(crate) fn base_presentation(
41    title: &str,
42    hint: &str,
43    primary: &str,
44    entries: Vec<(&str, String)>,
45) -> cookie_agent_protocol::PreparedPresentation {
46    let presentation = cookie_agent_runtime_api::tool_presentation(title, primary).with_hint(hint);
47    if entries.is_empty() {
48        presentation
49    } else {
50        presentation.with_arguments([cookie_agent_protocol::DisplayBlock::key_value(entries)])
51    }
52}

54/// The longest text a base tool's `markdown` result holds before the display
55/// budget cuts it.
56const MAX_MARKDOWN_DISPLAY_BYTES: usize = 48 * 1024;
57
58/// `text` as a result's `markdown` display.
59pub(crate) fn markdown_display(
60    text: &str,
61) -> Vec<cookie_agent_protocol::Open<cookie_agent_protocol::DisplayBlock>> {
62    if text.is_empty() {
63        return Vec::new();
64    }
65    let (head, omitted_lines) =
66        cookie_agent_runtime_api::text_head(text, MAX_MARKDOWN_DISPLAY_BYTES);
67    cookie_agent_protocol::fit_result_display(vec![cookie_agent_protocol::Open::Known(
68        cookie_agent_protocol::DisplayBlock::Markdown {
69            text: head.to_owned(),
70            omitted_lines,
71        },
72    )])
73}

75impl Engine {
76    /// Registers the base layer's providers, in composition order.
77    pub(crate) fn register_base_layer(&self) -> Result<(), EngineError> {
78        let providers: [Arc<dyn ToolProvider>; 7] = [
79            Arc::new(DelegateToolProvider::new(self.clone())),
80            Arc::new(MessageToolProvider::new(self.clone())),
81            Arc::new(SkillTool::new(self.clone())),
82            Arc::new(ReadArtifactTool),
83            Arc::new(GoalTools::new(self.clone())),
84            Arc::new(StopBackgroundTool::new(self.clone())),
85            Arc::new(LoaderTools::new(self.clone())),
86        ];
87        for provider in providers {
88            self.try_register_tool_provider(ToolLayer::Base, provider)?;
89        }
90        Ok(())
91    }
92}

94/// Names of every base-layer tool. A runtime tool never takes one: it is
95/// published as `<attachment>_<name>` instead, even while the base tool is
96/// not offered.
97pub(crate) fn reserved_tool_names() -> &'static std::collections::HashSet<String> {
98    static NAMES: std::sync::OnceLock<std::collections::HashSet<String>> =
99        std::sync::OnceLock::new();
100    NAMES.get_or_init(|| {
101        base_layer_tool_specs("reserved")
102            .into_iter()
103            .map(|spec| spec.name)
104            .collect()
105    })
106}

108/// Every base-layer tool as the engine publishes it, for the tool reference.
109/// `delegate_subagent` lists `delegate_target` as its only subagent and
110/// shows `capabilities` as a root with `global` and a workspace sees it.
111#[must_use]
112pub fn base_layer_tool_specs(delegate_target: &str) -> Vec<ToolSpec> {
113    let mut specs = vec![SkillTool::spec()];
114    specs.extend(DelegateToolProvider::specs(
115        &[delegate_target],
116        crate::runtime::runtime_delegation::CapabilitiesParam::Optional,
117    ));
118    specs.push(MessageToolProvider::spec());
119    specs.push(GoalTools::get_spec());
120    specs.push(GoalTools::set_spec());
121    specs.push(GoalTools::update_spec());
122    specs.push(ReadArtifactTool::spec());
123    specs.push(StopBackgroundTool::spec());
124    specs.push(LoaderTools::search_spec());
125    specs
126}
