3//! never as the user, with the request as untrusted data.

9/// Headers the model never sees, even when a webhook lists them.
10pub const ALWAYS_HIDDEN_HEADERS: &[&str] = &[
11    "authorization",
12    "proxy-authorization",
13    "cookie",
14    "set-cookie",
15    "x-hub-signature",
16    "x-hub-signature-256",
17    "webhook-signature",
18];
19/// The longest header value shown; a longer one is cut.
20const MAX_HEADER_VALUE_CHARS: usize = 1000;
21/// The elements of the message, whose closing tags inside the request's
22/// text are defused so the request cannot end its own block.
23const ELEMENTS: &[&str] = &["webhook_event", "instructions", "request", "body"];
24
25/// The body as the message shows it, decided before the message is
26/// rendered so the caller can store the body as an artifact first.
27#[derive(Clone, Debug, Eq, PartialEq)]
28pub struct ShownBody {
29    /// The body's size in bytes, as received.
30    pub bytes: usize,
31    /// The text shown inline, cut to the inline limit; `None` for an empty
32    /// or binary body.
33    pub text: Option<String>,
34    /// Whether `text` is only the start of the body.
35    pub truncated: bool,
36    /// Whether the body is binary.
37    pub binary: bool,
38}
39
40impl ShownBody {
41    /// Whether the caller should store the whole body as an artifact and
42    /// pass its reference to [`render_event`]: a binary body, or one cut to
43    /// the inline limit.
44    #[must_use]
45    pub const fn wants_artifact(&self) -> bool {
46        self.binary || self.truncated
47    }
48}
49
50/// How `body` is shown, with at most `inline_limit` bytes inline: JSON and
51/// form bodies pretty-printed, text as it is, cut at a line break in the
52/// last half of the limit or else at a character boundary.
53#[must_use]
54pub fn shown_body(request: &WebhookRequest, body: &ParsedBody, inline_limit: usize) -> ShownBody {
55    let bytes = request.body.len();
56    let full = match body {
57        ParsedBody::Empty | ParsedBody::Binary => None,
58        ParsedBody::Json(value) | ParsedBody::Form(value) => {
59            Some(serde_json::to_string_pretty(value).unwrap_or_else(|_| value.to_string()))
60        }
61        ParsedBody::Text(text) => Some(text.clone()),
62    };
63    let (text, truncated) = match full {
64        Some(full) if full.len() > inline_limit => {
65            (Some(cut(&full, inline_limit).to_owned()), true)
66        }
67        other => (other, false),
68    };
69    ShownBody {
70        bytes,
71        text,
72        truncated,
73        binary: matches!(body, ParsedBody::Binary),
74    }
75}
76
77fn cut(text: &str, limit: usize) -> &str {
78    let mut end = limit.min(text.len());
79    while !text.is_char_boundary(end) {
80        end -= 1;
81    }
82    let head = &text[..end];
83    match head.rfind('\n') {
84        Some(at) if at >= end / 2 => &head[..at],
85        _ => head,
86    }
87}

89/// The attributes of the `<webhook_event>` element.
90#[derive(Clone, Copy, Debug)]
91pub struct EventAttributes<'a> {
92    pub webhook_id: &'a str,
93    pub name: Option<&'a str>,
94    pub event_id: &'a str,
95    /// When the request arrived, in the host's local time.
96    pub received_at: &'a jiff::Zoned,
97    pub delivery: Option<&'a str>,
98    pub thread: Option<&'a str>,
99}
100
101/// Everything one event's message is made of.
102#[derive(Clone, Copy, Debug)]
103pub struct EventMessage<'a> {
104    pub attributes: EventAttributes<'a>,
105    /// The webhook's `prompt`, if it has one.
106    pub instructions: Option<&'a str>,
107    pub request: &'a WebhookRequest,
108    /// The webhook's `headers`: extra headers the model sees.
109    pub extra_headers: &'a [String],
110    /// The check's own secret headers ([`super::verify::Verifier::secret_headers`]).
111    pub secret_headers: &'a [String],
112    pub body: &'a ShownBody,
113    /// The stored body's reference, such as `artifact://sha256/…`, when the
114    /// caller stored it.
115    pub artifact: Option<&'a str>,
116}
117
118/// The message, as the model reads it.
119#[must_use]
120pub fn render_event(message: &EventMessage<'_>) -> String {
121    let EventAttributes {
122        webhook_id,
123        name,
124        event_id,
125        received_at,
126        delivery,
127        thread,
128    } = message.attributes;
129    let mut out = format!(
130        "<webhook_event webhook=\"{}\"",
131        escape_xml_attribute(webhook_id)
132    );
133    push_attribute(&mut out, "name", name);
134    push_attribute(&mut out, "event", Some(event_id));
135    let received = received_at.strftime("%Y-%m-%dT%H:%M:%S%:z").to_string();
136    push_attribute(&mut out, "received_at", Some(&received));
137    push_attribute(&mut out, "delivery", delivery);
138    push_attribute(&mut out, "thread", thread);
139    out.push_str(">\n");
140    if let Some(instructions) = message.instructions.filter(|text| !text.trim().is_empty()) {
141        out.push_str("<instructions>\n");
142        out.push_str(&defuse(instructions.trim_end()));
143        out.push_str("\n</instructions>\n");
144    }
145    render_request(&mut out, message);
146    render_body(&mut out, message.body, message.artifact);
147    out.push_str("</webhook_event>");
148    out
149}
150
151fn push_attribute(out: &mut String, name: &str, value: Option<&str>) {
152    if let Some(value) = value {
153        out.push_str(&format!(
154            " {name}=\"{}\"",
155            escape_xml_attribute(&plain(value))
156        ));
157    }
158}
159
160fn render_request(out: &mut String, message: &EventMessage<'_>) {
161    let request = message.request;
162    out.push_str("<request");
163    push_attribute(out, "method", Some(&request.method));
164    push_attribute(out, "content_type", request.header("content-type"));
165    push_attribute(out, "user_agent", request.header("user-agent"));
166    let lines = shown_headers(request, message.extra_headers, message.secret_headers)
167        .map(|(name, value)| format!("{}: {}", plain(name), defuse(&shorten(&plain(value)))))
168        .collect::<Vec<_>>();
169    if lines.is_empty() {
170        out.push_str("/>\n");
171    } else {
172        out.push_str(">\n");
173        out.push_str(&lines.join("\n"));
174        out.push_str("\n</request>\n");
175    }
176}

178/// The headers shown as lines, in the order received: the webhook's
179/// `headers` and any `X-<anything>-Event`, never `Content-Type` or
180/// `User-Agent` (attributes already), [`ALWAYS_HIDDEN_HEADERS`], or the
181/// check's own.
182fn shown_headers<'a>(
183    request: &'a WebhookRequest,
184    extra: &'a [String],
185    secret: &'a [String],
186) -> impl Iterator<Item = (&'a str, &'a str)> + 'a {
187    request.headers.iter().filter_map(move |(name, value)| {
188        let lower = name.to_ascii_lowercase();
189        let hidden = ALWAYS_HIDDEN_HEADERS.contains(&lower.as_str())
190            || secret
191                .iter()
192                .any(|header| header.eq_ignore_ascii_case(&lower))
193            || lower == "content-type"
194            || lower == "user-agent";
195        let wanted = extra
196            .iter()
197            .any(|header| header.eq_ignore_ascii_case(&lower))
198            || is_event_header(&lower);
199        (wanted && !hidden).then_some((name.as_str(), value.as_str()))
200    })
201}
202
203fn is_event_header(lower: &str) -> bool {
204    lower
205        .strip_prefix("x-")
206        .and_then(|rest| rest.strip_suffix("-event"))
207        .is_some_and(|middle| !middle.is_empty())
208}

210fn render_body(out: &mut String, body: &ShownBody, artifact: Option<&str>) {
211    if body.bytes == 0 {
212        return;
213    }
214    out.push_str(&format!("<body bytes=\"{}\"", body.bytes));
215    if body.binary {
216        out.push_str(" binary=\"true\"");
217    }
218    if body.truncated {
219        out.push_str(" truncated=\"true\"");
220    }
221    push_attribute(out, "artifact", artifact);
222    match &body.text {
223        Some(text) => {
224            out.push_str(">\n");
225            out.push_str(&defuse(text));
226            out.push_str("\n</body>\n");
227        }
228        None => out.push_str("/>\n"),
229    }
230}
231
232/// `text` with every control character made a space, for attributes and
233/// header lines.
234fn plain(text: &str) -> String {
235    text.chars()
236        .map(|character| {
237            if character.is_control() {
238                ' '
239            } else {
240                character
241            }
242        })
243        .collect()
244}
245
246fn shorten(text: &str) -> String {
247    match text.char_indices().nth(MAX_HEADER_VALUE_CHARS) {
248        Some((at, _)) => format!("{}…", &text[..at]),
249        None => text.to_owned(),
250    }
251}
252
253/// `text` with every closing tag of the message's own elements written
254/// `<\/name`, so a request cannot close its block and speak outside it.
255/// Everything else stays as sent.
256fn defuse(text: &str) -> String {
257    let mut out = String::with_capacity(text.len());
258    let mut rest = text;
259    while let Some(at) = rest.find("</") {
260        let (before, from) = rest.split_at(at);
261        out.push_str(before);
262        let after = &from[2..];
263        let closes_ours = ELEMENTS.iter().any(|element| {
264            after
265                .get(..element.len())
266                .is_some_and(|head| head.eq_ignore_ascii_case(element))
267        });
268        out.push_str(if closes_ours { "<\\/" } else { "</" });
269        rest = after;
270    }
271    out.push_str(rest);
272    out
273}

304    fn render(
305        request: &WebhookRequest,
306        extra: &[String],
307        inline: usize,
308        artifact: Option<&str>,
309    ) -> (ShownBody, String) {
310        let received_at = zoned();
311        let parsed = parse_body(request);
312        let body = shown_body(request, &parsed, inline);
313        let text = render_event(&EventMessage {
314            attributes: EventAttributes {
315                webhook_id: "w7k2m9qa",
316                name: Some("GitHub \"issues\""),
317                event_id: "e3j8x0pq",
318                received_at: &received_at,
319                delivery: Some("7c1e"),
320                thread: Some("1234"),
321            },
322            instructions: Some("Act on it."),
323            request,
324            extra_headers: extra,
325            secret_headers: &["x-hub-signature-256".to_owned()],
326            body: &body,
327            artifact,
328        });
329        (body, text)
330    }

395    #[test]
396    fn a_large_body_is_cut_and_asks_for_an_artifact() {
397        let lines = (0..100)
398            .map(|line| format!("line {line:03}"))
399            .collect::<Vec<_>>()
400            .join("\n");
401        let request = WebhookRequest {
402            method: "POST".to_owned(),
403            headers: vec![("Content-Type".to_owned(), "text/plain".to_owned())],
404            body: lines.clone().into_bytes(),
405        };
406        let (body, text) = render(&request, &[], 100, Some("artifact://sha256/abc"));
407        assert!(body.truncated && body.wants_artifact());
408        let shown = body.text.as_deref().unwrap();
409        assert!(
410            shown.len() <= 100 && lines.starts_with(shown) && !shown.ends_with('\n'),
411            "{shown:?}"
412        );
413        assert!(
414            shown.ends_with("line 010") || shown.ends_with("line 011"),
415            "{shown:?}"
416        );
417        assert!(
418            text.contains(&format!(
419                "<body bytes=\"{}\" truncated=\"true\" artifact=\"artifact://sha256/abc\">\n",
420                lines.len()
421            )),
422            "{text}"
423        );
424        let (body, _) = render(&request, &[], lines.len(), None);
425        assert!(!body.truncated && !body.wants_artifact());

429    fn cuts_keep_whole_characters() {
430        assert_eq!(cut("ééé", 3), "é");
431        assert_eq!(cut("ééé", 4), "éé");
432        assert_eq!(cut("abc", 10), "abc");
433        // A line break early in the limit is not worth losing half of it.
434        assert_eq!(cut("a\nbcdefghij", 8), "a\nbcdefg");
