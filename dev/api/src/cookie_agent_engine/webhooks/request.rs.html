1//! A webhook request as the HTTP layer read it, its body, and the values a
2//! filter or a thread key reads (`docs/site/specs/webhooks.md` sections
3//! 5.2, 5.3, 7.1, and 7.4).

5use serde_json::Value;
6
7/// The longest delivery ID the daemon keeps; a longer one counts as none.
8pub const MAX_DELIVERY_ID_CHARS: usize = 200;
9/// The longest thread key; a longer value counts as none (section 7.4).
10pub const MAX_THREAD_KEY_CHARS: usize = 200;
11
12/// One request, read in full (the body limit is the HTTP layer's).
13#[derive(Clone, Debug, Default, Eq, PartialEq)]
14pub struct WebhookRequest {
15    /// The method, such as `POST`.
16    pub method: String,
17    /// The headers in the order received, with their names as sent. A
18    /// value that is not UTF-8 is kept lossily.
19    pub headers: Vec<(String, String)>,
20    /// The body's bytes.
21    pub body: Vec<u8>,
22}
23
24impl WebhookRequest {
25    /// The first value of the header `name`, matched case-insensitively.
26    #[must_use]
27    pub fn header(&self, name: &str) -> Option<&str> {
28        self.headers
29            .iter()
30            .find(|(header, _)| header.eq_ignore_ascii_case(name))
31            .map(|(_, value)| value.as_str())
32    }
33
34    /// The media type of `Content-Type` in lowercase, without parameters.
35    #[must_use]
36    pub fn media_type(&self) -> Option<String> {
37        let value = self.header("content-type")?;
38        let media = value.split(';').next().unwrap_or_default().trim();
39        (!media.is_empty()).then(|| media.to_ascii_lowercase())
40    }
41}

43/// What a request's body is, as the filter, a thread key, and the model
44/// read it.
45#[derive(Clone, Debug, PartialEq)]
46pub enum ParsedBody {
47    /// No body.
48    Empty,
49    /// A JSON body.
50    Json(Value),
51    /// A form body (`application/x-www-form-urlencoded`) as a JSON object
52    /// of its fields: a field sent once is a string, one sent several
53    /// times an array of strings, and a `payload` field holding JSON (as
54    /// GitHub's form mode sends) that JSON.
55    Form(Value),
56    /// UTF-8 text that is neither.
57    Text(String),
58    /// Anything else, shown only as an artifact.
59    Binary,
60}
61
62impl ParsedBody {
63    /// The JSON a pointer reads: a JSON body, or a form body's object.
64    #[must_use]
65    pub fn json(&self) -> Option<&Value> {
66        match self {
67            Self::Json(value) | Self::Form(value) => Some(value),
68            Self::Empty | Self::Text(_) | Self::Binary => None,
69        }
70    }
71}
72
73/// Reads `request`'s body by its media type. A body labelled JSON (or
74/// `+json`) that does not parse reads as text or binary; an unlabelled
75/// body that parses as a JSON object or array reads as JSON.
76#[must_use]
77pub fn parse_body(request: &WebhookRequest) -> ParsedBody {
78    let body = request.body.as_slice();
79    if body.is_empty() {
80        return ParsedBody::Empty;
81    }
82    let media = request.media_type();
83    let labelled_json = media
84        .as_deref()
85        .is_some_and(|media| media == "application/json" || media.ends_with("+json"));
86    if media.as_deref() == Some("application/x-www-form-urlencoded") {
87        return ParsedBody::Form(parse_form(body));
88    }
89    let unlabelled = media.is_none() && starts_like_json(body);
90    if (labelled_json || unlabelled)
91        && let Ok(value) = serde_json::from_slice::<Value>(body)
92    {
93        return ParsedBody::Json(value);
94    }
95    match std::str::from_utf8(body) {
96        Ok(text) if !text.contains('\0') => ParsedBody::Text(text.to_owned()),
97        _ => ParsedBody::Binary,
98    }
99}
100
101fn starts_like_json(body: &[u8]) -> bool {
102    body.iter()
103        .find(|byte| !byte.is_ascii_whitespace())
104        .is_some_and(|byte| matches!(byte, b'{' | b'['))
105}
106
107fn parse_form(body: &[u8]) -> Value {
108    let mut object = serde_json::Map::new();
109    for (name, value) in form_urlencoded::parse(body) {
110        let value = Value::String(value.into_owned());
111        match object.get_mut(name.as_ref()) {
112            None => {
113                object.insert(name.into_owned(), value);
114            }
115            Some(Value::Array(values)) => values.push(value),
116            Some(first) => {
117                let first = std::mem::take(first);
118                object.insert(name.into_owned(), Value::Array(vec![first, value]));
119            }
120        }
121    }
122    if let Some(Value::String(payload)) = object.get("payload")
123        && let Ok(parsed) = serde_json::from_str::<Value>(payload)
124    {
125        object.insert("payload".to_owned(), parsed);
126    }
127    Value::Object(object)
128}

130/// Where a filter condition or a thread key reads its value.
131#[derive(Clone, Debug, Eq, PartialEq)]
132pub enum ValueSource {
133    /// A header, by name, case-insensitively.
134    Header(String),
135    /// A JSON Pointer (RFC 6901) into a JSON or form body.
136    Json(String),
137}
138
139/// Why a value source is refused when a webhook is saved.
140#[derive(Debug, thiserror::Error, Eq, PartialEq)]
141pub enum ValueSourceError {
142    #[error("a header name must be 1 to 128 visible ASCII characters without ':'")]
143    HeaderName,
144    #[error("a JSON pointer must be empty or start with '/', and be at most 512 characters")]
145    Pointer,
146}
147
148impl ValueSource {
149    /// Checks the source's shape.
150    pub fn validate(&self) -> Result<(), ValueSourceError> {
151        match self {
152            Self::Header(name) => {
153                let valid = (1..=128).contains(&name.len())
154                    && name
155                        .bytes()
156                        .all(|byte| byte.is_ascii_graphic() && byte != b':');
157                if valid {
158                    Ok(())
159                } else {
160                    Err(ValueSourceError::HeaderName)
161                }
162            }
163            Self::Json(pointer) => {
164                if pointer.len() <= 512 && (pointer.is_empty() || pointer.starts_with('/')) {
165                    Ok(())
166                } else {
167                    Err(ValueSourceError::Pointer)
168                }
169            }
170        }
171    }
172}

174/// The value `source` reads from a request, as a string: a header's
175/// value, or the JSON a pointer names when it is a scalar (a string as
176/// itself; a number, boolean, or null as its JSON text). An object, an
177/// array, or nothing reads as `None`.
178#[must_use]
179pub fn read_value(
180    request: &WebhookRequest,
181    body: &ParsedBody,
182    source: &ValueSource,
183) -> Option<String> {
184    match source {
185        ValueSource::Header(name) => request.header(name).map(str::to_owned),
186        ValueSource::Json(pointer) => scalar_text(body.json()?.pointer(pointer)?),
187    }
188}
189
190fn scalar_text(value: &Value) -> Option<String> {
191    match value {
192        Value::String(text) => Some(text.clone()),
193        Value::Number(_) | Value::Bool(_) | Value::Null => Some(value.to_string()),
194        Value::Array(_) | Value::Object(_) => None,
195    }
196}

198/// A request's thread key (section 7.4): the value `source` reads, unless
199/// it is empty, longer than [`MAX_THREAD_KEY_CHARS`], or holds a control
200/// character, when the event has none and gets a session of its own.
201#[must_use]
202pub fn thread_key(
203    request: &WebhookRequest,
204    body: &ParsedBody,
205    source: &ValueSource,
206) -> Option<String> {
207    read_value(request, body, source).filter(|key| plain_short(key, MAX_THREAD_KEY_CHARS))
208}
209
210fn plain_short(value: &str, max_chars: usize) -> bool {
211    !value.is_empty() && value.chars().count() <= max_chars && !value.chars().any(char::is_control)
212}

214/// The sender's own header naming a delivery, which a signature check
215/// implies (section 5.3).
216#[derive(Clone, Copy, Debug, Eq, PartialEq)]
217pub enum DeliveryHeader {
218    /// `X-GitHub-Delivery`, for the `github` check.
219    GitHub,
220    /// `webhook-id`, for the `standard` check.
221    Standard,
222}
223
224impl DeliveryHeader {
225    const fn name(self) -> &'static str {
226        match self {
227            Self::GitHub => "x-github-delivery",
228            Self::Standard => "webhook-id",
229        }
230    }
231}
232
233/// A request's delivery ID (section 5.3): the check's own header, then
234/// `Idempotency-Key`. A value that is empty, longer than
235/// [`MAX_DELIVERY_ID_CHARS`], or holds a control character is skipped.
236#[must_use]
237pub fn delivery_id(request: &WebhookRequest, own: Option<DeliveryHeader>) -> Option<String> {
238    own.map(DeliveryHeader::name)
239        .into_iter()
240        .chain(["idempotency-key"])
241        .filter_map(|name| request.header(name))
242        .map(str::trim)
243        .find(|value| plain_short(value, MAX_DELIVERY_ID_CHARS))
244        .map(str::to_owned)
245}

263    #[test]
264    fn bodies_read_by_media_type() {
265        assert_eq!(
266            parse_body(&request(
267                Some("application/json; charset=utf-8"),
268                "{\"a\":1}"
269            )),
270            ParsedBody::Json(json!({"a": 1}))
271        );
272        assert_eq!(
273            parse_body(&request(Some("application/vnd.github+json"), "[1]")),
274            ParsedBody::Json(json!([1]))
275        );
276        assert_eq!(
277            parse_body(&request(None, "  {\"a\":true}")),
278            ParsedBody::Json(json!({"a": true}))
279        );
280        // Unlabelled scalars stay text; a broken JSON label falls back.
281        assert_eq!(
282            parse_body(&request(None, "42")),
283            ParsedBody::Text("42".to_owned())
284        );
285        assert_eq!(
286            parse_body(&request(Some("application/json"), "{oops")),
287            ParsedBody::Text("{oops".to_owned())
288        );
289        assert_eq!(
290            parse_body(&request(Some("text/plain"), "{\"a\":1}")),
291            ParsedBody::Text("{\"a\":1}".to_owned())
292        );
293        assert_eq!(
294            parse_body(&request(Some("text/plain"), "")),
295            ParsedBody::Empty
296        );
297        let binary = WebhookRequest {
298            body: vec![0xff, 0x00, 0x10],
299            ..WebhookRequest::default()
300        };
301        assert_eq!(parse_body(&binary), ParsedBody::Binary);
302        let nul = WebhookRequest {
303            body: b"a\0b".to_vec(),
304            ..WebhookRequest::default()
305        };
306        assert_eq!(parse_body(&nul), ParsedBody::Binary);
307    }

328    #[test]
329    fn values_read_scalars_as_strings() {
330        let request = WebhookRequest {
331            headers: vec![("X-GitHub-Event".to_owned(), "issues".to_owned())],
332            ..WebhookRequest::default()
333        };
334        let body = ParsedBody::Json(json!({
335            "s": "text", "n": 12, "f": 1.5, "b": false, "z": null,
336            "o": {"k": 1}, "a": [1], "a~b": {"c/d": "escaped"},
337        }));
338        let read = |source: ValueSource| read_value(&request, &body, &source);
339        assert_eq!(
340            read(ValueSource::Header("x-github-event".into())).as_deref(),
341            Some("issues")
342        );
343        assert_eq!(read(ValueSource::Header("X-Missing".into())), None);
344        for (pointer, expected) in [
345            ("/s", Some("text")),
346            ("/n", Some("12")),
347            ("/f", Some("1.5")),
348            ("/b", Some("false")),
349            ("/z", Some("null")),
350            ("/o", None),
351            ("/a", None),
352            ("/a/0", Some("1")),
353            ("/o/k", Some("1")),
354            ("/a~0b/c~1d", Some("escaped")),
355            ("/missing", None),
356            ("", None),
357        ] {
358            assert_eq!(
359                read(ValueSource::Json(pointer.into())).as_deref(),
360                expected,
361                "{pointer}"
362            );
363        }
364        let text = ParsedBody::Text("x".into());
365        assert_eq!(
366            read_value(&request, &text, &ValueSource::Json("/s".into())),
367            None
368        );
369    }

425    #[test]
426    fn value_sources_are_checked() {
427        assert!(
428            ValueSource::Header("X-GitHub-Event".into())
429                .validate()
430                .is_ok()
431        );
432        assert!(ValueSource::Header(String::new()).validate().is_err());
433        assert!(ValueSource::Header("a:b".into()).validate().is_err());
434        assert!(ValueSource::Header("a b".into()).validate().is_err());
435        assert!(ValueSource::Json(String::new()).validate().is_ok());
436        assert!(ValueSource::Json("/issue/number".into()).validate().is_ok());
437        assert!(ValueSource::Json("issue".into()).validate().is_err());
438        assert!(
439            ValueSource::Json(format!("/{}", "a".repeat(512)))
440                .validate()
441                .is_err()
442        );
443    }
