1//! A webhook's filter (`docs/site/specs/webhooks.md` section 5.2): every
2//! condition must hold for a request to become an event.
3
4use super::request::{ParsedBody, ValueSource, ValueSourceError, WebhookRequest, read_value};
5
6/// The most conditions a filter holds.
7pub const MAX_CONDITIONS: usize = 32;
8/// The most strings one condition's `one_of` holds.
9pub const MAX_ONE_OF: usize = 64;
10/// The longest string in a `one_of`.
11pub const MAX_VALUE_CHARS: usize = 512;
12
13/// One condition: the value `source` reads is one of `one_of`.
14#[derive(Clone, Debug, Eq, PartialEq)]
15pub struct Condition {
16    pub source: ValueSource,
17    pub one_of: Vec<String>,
18}
19
20/// Why a filter is refused when a webhook is saved.
21#[derive(Debug, thiserror::Error, Eq, PartialEq)]
22pub enum FilterError {
23    #[error("a filter holds at most 32 conditions")]
24    TooManyConditions,
25    #[error("a condition's one_of holds 1 to 64 strings of at most 512 characters")]
26    OneOf,
27    #[error(transparent)]
28    Source(#[from] ValueSourceError),
29}
30
31/// Checks a filter's shape.
32pub fn validate(conditions: &[Condition]) -> Result<(), FilterError> {
33    if conditions.len() > MAX_CONDITIONS {
34        return Err(FilterError::TooManyConditions);
35    }
36    for condition in conditions {
37        condition.source.validate()?;
38        let valid = (1..=MAX_ONE_OF).contains(&condition.one_of.len())
39            && condition
40                .one_of
41                .iter()
42                .all(|value| value.chars().count() <= MAX_VALUE_CHARS);
43        if !valid {
44            return Err(FilterError::OneOf);
45        }
46    }
47    Ok(())
48}
49
50/// Whether `request` passes every condition; an empty filter passes all.
51/// Values compare exactly, case included; a missing value, an object, or
52/// an array matches nothing.
53#[must_use]
54pub fn matches(conditions: &[Condition], request: &WebhookRequest, body: &ParsedBody) -> bool {
55    conditions.iter().all(|condition| {
56        read_value(request, body, &condition.source)
57            .is_some_and(|value| condition.one_of.contains(&value))
58    })
59}

61#[cfg(test)]
62mod tests {
63    use serde_json::json;
64
65    use super::*;
66
67    fn condition(source: ValueSource, one_of: &[&str]) -> Condition {
68        Condition {
69            source,
70            one_of: one_of.iter().map(|value| (*value).to_owned()).collect(),
71        }
72    }
73
74    #[test]
75    fn every_condition_must_hold() {
76        let request = WebhookRequest {
77            headers: vec![("X-GitHub-Event".to_owned(), "issues".to_owned())],
78            ..WebhookRequest::default()
79        };
80        let body = ParsedBody::Json(json!({
81            "action": "opened", "pull_request": {"draft": false}, "number": 7, "labels": ["a"],
82        }));
83        let event = condition(
84            ValueSource::Header("x-github-event".into()),
85            &["issues", "issue_comment"],
86        );
87        let action = condition(ValueSource::Json("/action".into()), &["opened"]);
88        assert!(matches(&[], &request, &body));
89        assert!(matches(&[event.clone(), action.clone()], &request, &body));
90        let closed = condition(ValueSource::Json("/action".into()), &["closed"]);
91        assert!(!matches(&[event, closed], &request, &body));
92        for (pointer, one_of, expected) in [
93            ("/pull_request/draft", "false", true),
94            ("/number", "7", true),
95            ("/number", "07", false),
96            ("/action", "Opened", false),
97            ("/labels", "a", false),
98            ("/missing", "null", false),
99        ] {
100            let filter = [condition(ValueSource::Json(pointer.into()), &[one_of])];
101            assert_eq!(
102                matches(&filter, &request, &body),
103                expected,
104                "{pointer} {one_of}"
105            );
106        }
107        let text = ParsedBody::Text("action=opened".into());
108        assert!(!matches(&[action], &request, &text));
109    }
110
111    #[test]
112    fn filters_are_checked_when_saved() {
113        let ok = condition(ValueSource::Json("/action".into()), &["opened"]);
114        assert_eq!(validate(std::slice::from_ref(&ok)), Ok(()));
115        assert_eq!(
116            validate(&vec![ok; MAX_CONDITIONS + 1]),
117            Err(FilterError::TooManyConditions)
118        );
119        assert_eq!(
120            validate(&[condition(ValueSource::Json("/a".into()), &[])]),
121            Err(FilterError::OneOf)
122        );
123        let many = vec!["x"; MAX_ONE_OF + 1];
124        assert_eq!(
125            validate(&[condition(ValueSource::Json("/a".into()), &many)]),
126            Err(FilterError::OneOf)
127        );
128        let long = "x".repeat(MAX_VALUE_CHARS + 1);
129        assert_eq!(
130            validate(&[condition(ValueSource::Json("/a".into()), &[&long])]),
131            Err(FilterError::OneOf)
132        );
133        assert_eq!(
134            validate(&[condition(ValueSource::Json("a".into()), &["x"])]),
135            Err(FilterError::Source(ValueSourceError::Pointer))
136        );
137    }
