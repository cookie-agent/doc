1//! Converting a session tree's history to this build's format on disk
2//! (`docs/site/specs/compatibility-v1.md` section 8.7).

8/// What converting one session tree did.
9#[derive(Clone, Debug, Default, Eq, PartialEq)]
10pub struct TreeConversion {
11    /// Each log converted, with the history format it had.
12    pub converted: Vec<(SessionId, u32)>,
13    /// Each log left as it is because no conversion from its format exists:
14    /// `None` for one written before history formats, else its format,
15    /// such as a newer build's.
16    pub left: Vec<(SessionId, Option<u32>)>,
17}
18
19/// What one member's log needs.
20enum Stage {
21    Current,
22    Staged(Staged),
23    Left(Option<u32>),
24}
25
26/// One log converted and written beside its original, not yet in place.
27struct Staged {
28    dir: PathBuf,
29    format: u32,
30    log: PathBuf,
31    metadata: PathBuf,
32}
33
34const STAGED_SUFFIX: &str = "convert.tmp";

36/// Converts every log of the tree rooted at `root_dir` (the root's, and
37/// each child's under `subagents/`) whose history format is older than
38/// this build's, together with each one's `metadata` cache. The caller holds
39/// the tree's lock.
40///
41/// Each converted log and its `metadata`, folded from it, are first written
42/// beside the originals and synced. Only once every log converted does each
43/// original become `events.jsonl.v<N>.bak` and the new file take its place;
44/// `subagents/index.json`, a cache of the old summaries, is removed and
45/// rebuilt by the next tree load. A failure before that leaves every
46/// original untouched. A log already current, of a newer format, or of one
47/// no conversion exists from is left for its open to judge.
48pub(crate) fn convert_tree(
49    root: SessionId,
50    root_dir: &Path,
51) -> Result<TreeConversion, SessionError> {
52    let mut members = vec![(root, root_dir.to_owned())];
53    if let Ok(entries) = fs::read_dir(root_dir.join(SUBAGENTS_DIR)) {
54        for entry in entries.flatten() {
55            if let Some(id) = entry
56                .file_name()
57                .to_str()
58                .and_then(|name| name.parse::<SessionId>().ok())
59                && entry.path().is_dir()
60            {
61                members.push((id, entry.path()));
62            }
63        }
64    }
65    let mut staged = Vec::new();
66    let mut conversion = TreeConversion::default();
67    for (id, dir) in &members {
68        match stage(*id, dir) {
69            Ok(Stage::Staged(log)) => {
70                conversion.converted.push((*id, log.format));
71                staged.push(log);
72            }
73            Ok(Stage::Current) => {}
74            Ok(Stage::Left(format)) => conversion.left.push((*id, format)),
75            Err(error) => {
76                for log in &staged {
77                    let _ = fs::remove_file(&log.log);
78                    let _ = fs::remove_file(&log.metadata);
79                }
80                return Err(error);
81            }
82        }
83    }
84    for log in &staged {
85        commit(log)?;
86    }
87    if !staged.is_empty() {
88        let index = root_dir.join(SUBAGENTS_DIR).join(SUBAGENT_INDEX_FILE);
89        match fs::remove_file(&index) {
90            Ok(()) => fsync_directory(&root_dir.join(SUBAGENTS_DIR))?,
91            Err(source) if source.kind() == std::io::ErrorKind::NotFound => {}
92            Err(source) => {
93                return Err(SessionError::Io {
94                    path: index,
95                    source,
96                });
97            }
98        }
99    }
100    Ok(conversion)
101}

103/// The history format the log at `path` declares on its first line: `None`
104/// when there is no log, `Some(None)` when it declares none.
105fn log_format(path: &Path) -> Result<Option<Option<u32>>, SessionError> {
106    use std::io::BufRead as _;
107    let file = match fs::File::open(path) {
108        Ok(file) => file,
109        Err(source) if source.kind() == std::io::ErrorKind::NotFound => return Ok(None),
110        Err(source) => {
111            return Err(SessionError::Io {
112                path: path.to_owned(),
113                source,
114            });
115        }
116    };
117    let mut first = Vec::new();
118    std::io::BufReader::new(file)
119        .read_until(b'\n', &mut first)
120        .map_err(|source| SessionError::Io {
121            path: path.to_owned(),
122            source,
123        })?;
124    Ok(Some(
125        serde_json::from_slice::<serde_json::Value>(&first)
126            .ok()
127            .as_ref()
128            .and_then(history::history_format),
129    ))
130}

132/// Converts one member's log beside the original, when it needs it.
133fn stage(id: SessionId, dir: &Path) -> Result<Stage, SessionError> {
134    let path = dir.join(EVENTS_FILE);
135    let format = match log_format(&path)? {
136        None => return Ok(Stage::Current),
137        Some(Some(format)) if format == cookie_agent_protocol::HISTORY_FORMAT => {
138            return Ok(Stage::Current);
139        }
140        Some(Some(format)) if history::convertible(format) => format,
141        Some(format) => return Ok(Stage::Left(format)),
142    };
143    let bytes = fs::read(&path).map_err(|source| SessionError::Io {
144        path: path.clone(),
145        source,
146    })?;
147    // A torn final record stays in the backup only.
148    let complete = bytes
149        .iter()
150        .rposition(|byte| *byte == b'\n')
151        .map_or(0, |at| at + 1);
152    let converted =
153        history::convert_bytes(format, &bytes[..complete]).map_err(|error: ConvertError| {
154            SessionError::Event(EventLogError::Corrupt {
155                path: path.clone(),
156                message: error.to_string(),
157            })
158        })?;
159    let log = dir.join(format!("{EVENTS_FILE}.{STAGED_SUFFIX}"));
160    let metadata = dir.join(format!("{SESSION_META_FILE}.{STAGED_SUFFIX}"));
161    let staged = Staged {
162        dir: dir.to_owned(),
163        format,
164        log,
165        metadata,
166    };
167    let result = (|| {
168        write_staged(&staged.log, &converted)?;
169        let meta = projection(EventLog::open_read_only(staged.log.clone(), id)?)?.meta;
170        let meta = serde_json::to_vec_pretty(&meta).map_err(|source| SessionError::Json {
171            path: staged.metadata.clone(),
172            source,
173        })?;
174        write_staged(&staged.metadata, &meta)
175    })();
176    match result {
177        Ok(()) => Ok(Stage::Staged(staged)),
178        Err(error) => {
179            let _ = fs::remove_file(&staged.log);
180            let _ = fs::remove_file(&staged.metadata);
181            Err(error)
182        }
183    }
184}

186/// Writes `bytes` as the private file at `path`, replacing what a crash left
187/// there, and syncs it.
188fn write_staged(path: &Path, bytes: &[u8]) -> Result<(), SessionError> {
189    let io_error = |source| SessionError::Io {
190        path: path.to_owned(),
191        source,
192    };
193    match fs::remove_file(path) {
194        Ok(()) => {}
195        Err(source) if source.kind() == std::io::ErrorKind::NotFound => {}
196        Err(source) => return Err(io_error(source)),
197    }
198    #[cfg(unix)]
199    let mut file = OpenOptions::new()
200        .write(true)
201        .create_new(true)
202        .mode(0o600)
203        .open(path)
204        .map_err(io_error)?;
205    #[cfg(windows)]
206    let mut file =
207        cookie_agent_models::secure_store::create_windows_private_file(path).map_err(io_error)?;
208    file.write_all(bytes)
209        .and_then(|()| file.sync_all())
210        .map_err(io_error)
211}

213/// Puts one staged log in place: the original stays as its backup.
214fn commit(staged: &Staged) -> Result<(), SessionError> {
215    let log = staged.dir.join(EVENTS_FILE);
216    let backup = staged
217        .dir
218        .join(format!("{EVENTS_FILE}.v{}.bak", staged.format));
219    let backup_error = |source| SessionError::Io {
220        path: backup.clone(),
221        source,
222    };
223    match fs::remove_file(&backup) {
224        Ok(()) => {}
225        Err(source) if source.kind() == std::io::ErrorKind::NotFound => {}
226        Err(source) => return Err(backup_error(source)),
227    }
228    // The backup is a second name for the original, so `events.jsonl` names
229    // a complete log at every moment; a copy where links do not work.
230    if fs::hard_link(&log, &backup).is_err() {
231        fs::copy(&log, &backup)
232            .and_then(|_| fs::File::open(&backup)?.sync_all())
233            .map_err(backup_error)?;
234    }
235    fsync_directory(&staged.dir)?;
236    replace(&staged.log, &log)?;
237    replace(&staged.metadata, &meta_path(&staged.dir))?;
238    fsync_directory(&staged.dir)?;
239    Ok(())
240}
241
242fn replace(source: &Path, target: &Path) -> Result<(), SessionError> {
243    #[cfg(unix)]
244    let result = fs::rename(source, target);
245    #[cfg(windows)]
246    let result = replace_windows_path_with_retry(source, target);
247    result.map_err(|source| SessionError::Io {
248        path: target.to_owned(),
249        source,
250    })
251}

253/// What upgrading one session tree did.
254#[derive(Debug)]
255pub enum TreeUpgrade {
256    /// Every log was already in this build's history format.
257    Current,
258    /// Logs were converted, or left as they are (see
259    /// [`TreeConversion::left`]).
260    Converted(TreeConversion),
261    /// Another cookie process holds the tree; it converts it itself.
262    InUse,
263    Failed(SessionError),
264}

266/// Converts every session tree of the work dir of `cwd`, or of every work
267/// dir when `cwd` is `None`, to this build's history format, as an owned
268/// open would (`cookie sessions upgrade`). Returns each tree's root and work
269/// dir with what happened to it.
270pub fn upgrade_sessions(
271    data_root: &Path,
272    cwd: Option<&Path>,
273) -> Result<Vec<(PathBuf, SessionId, TreeUpgrade)>, SessionError> {
274    let work_dirs = match cwd {
275        Some(cwd) => vec![SessionStore::resolve_workdir_dir(data_root, cwd)],
276        None => {
277            let sessions = data_root.join(SESSIONS_ROOT_DIR);
278            match fs::read_dir(&sessions) {
279                Ok(entries) => {
280                    let mut dirs = entries
281                        .flatten()
282                        .map(|entry| entry.path())
283                        .filter(|path| path.is_dir())
284                        .collect::<Vec<_>>();
285                    dirs.sort();
286                    dirs
287                }
288                Err(source) if source.kind() == std::io::ErrorKind::NotFound => Vec::new(),
289                Err(source) => {
290                    return Err(SessionError::Io {
291                        path: sessions,
292                        source,
293                    });
294                }
295            }
296        }
297    };
298    let mut upgrades = Vec::new();
299    for work_dir in work_dirs {
300        let Ok(entries) = fs::read_dir(&work_dir) else {
301            continue;
302        };
303        let mut roots = entries
304            .flatten()
305            .filter_map(|entry| {
306                let id = entry.file_name().to_str()?.parse::<SessionId>().ok()?;
307                entry.path().join(EVENTS_FILE).is_file().then_some(id)
308            })
309            .collect::<Vec<_>>();
310        roots.sort_by_key(ToString::to_string);
311        for root in roots {
312            let root_dir = work_dir.join(root.to_string());
313            let upgrade = match try_acquire(&root_dir) {
314                Ok(SessionOwnership::Owned(_lock)) => match convert_tree(root, &root_dir) {
315                    Ok(conversion) if conversion == TreeConversion::default() => {
316                        TreeUpgrade::Current
317                    }
318                    Ok(conversion) => TreeUpgrade::Converted(conversion),
319                    Err(error) => TreeUpgrade::Failed(error),
320                },
321                Ok(SessionOwnership::Foreign) => TreeUpgrade::InUse,
322                Err(source) => TreeUpgrade::Failed(SessionError::Io {
323                    path: root_dir.clone(),
324                    source,
325                }),
326            };
327            upgrades.push((work_dir.clone(), root, upgrade));
328        }
329    }
330    Ok(upgrades)
331}
