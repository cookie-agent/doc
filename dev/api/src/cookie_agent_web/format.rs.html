6/// The last component of `path`, `/` or `\` separated, ignoring trailing
7/// separators; the path itself when it has no other component (`/`).
8pub fn basename(path: &str) -> &str {
9    let trimmed = path.trim_end_matches(['/', '\\']);
10    match trimmed.rsplit(['/', '\\']).next() {
11        Some(last) if !last.is_empty() => last,
12        _ => path,
13    }
14}

16/// How long before `now` `then` was: "just now", minutes, hours, or days
17/// ago, else its date. A time after `now` (another clock) is "just now".
18pub fn relative_time(now: Timestamp, then: Timestamp) -> String {
19    let seconds = now.as_second().saturating_sub(then.as_second());
20    match seconds {
21        ..60 => "just now".to_owned(),
22        60..3_600 => format!("{}m ago", seconds / 60),
23        3_600..86_400 => format!("{}h ago", seconds / 3_600),
24        86_400..604_800 => format!("{}d ago", seconds / 86_400),
25        // Formatted by hand: jiff's strftime would add 20 KB to the page.
26        _ => {
27            let date = jiff::tz::TimeZone::UTC.to_datetime(then).date();
28            format!("{}-{:02}-{:02}", date.year(), date.month(), date.day())
29        }
30    }
31}

33/// `count` in at most four characters: as it is below a thousand, else in
34/// thousands or millions with one decimal while that fits.
35pub fn compact_count(count: u64) -> String {
36    match count {
37        ..1_000 => count.to_string(),
38        1_000..10_000 => tenths(count, 1_000, "k"),
39        10_000..1_000_000 => format!("{}k", count / 1_000),
40        1_000_000..10_000_000 => tenths(count, 1_000_000, "M"),
41        _ => format!("{}M", count / 1_000_000),
42    }
43}
44
45/// `count / unit` with one decimal, rounded down, and no `.0`.
46fn tenths(count: u64, unit: u64, suffix: &str) -> String {
47    let tenths = count / (unit / 10);
48    match tenths % 10 {
49        0 => format!("{}{suffix}", tenths / 10),
50        decimal => format!("{}.{decimal}{suffix}", tenths / 10),
51    }
52}

54#[cfg(test)]
55mod tests {
56    use jiff::{SignedDuration, Timestamp};
57
58    use super::{basename, compact_count, relative_time};
59
60    #[test]
61    fn the_basename_is_the_last_component() {
62        assert_eq!(basename("/tmp/tmp.x1/greeter"), "greeter");
63        assert_eq!(basename("/home/ada/project/"), "project");
64        assert_eq!(basename(r"C:\Users\ada\project"), "project");
65        assert_eq!(basename("greeter"), "greeter");
66        assert_eq!(basename("/"), "/");
67        assert_eq!(basename(""), "");
68    }
69
70    #[test]
71    fn relative_times_step_from_minutes_to_a_date() {
72        let now = Timestamp::from_second(1_800_000_000).expect("timestamp");
73        let ago = |seconds: i64| {
74            relative_time(
75                now,
76                now.checked_sub(SignedDuration::from_secs(seconds))
77                    .expect("timestamp"),
78            )
79        };
80        assert_eq!(ago(59), "just now");
81        assert_eq!(ago(-30), "just now");
82        assert_eq!(ago(60), "1m ago");
83        assert_eq!(ago(3_599), "59m ago");
84        assert_eq!(ago(3_600), "1h ago");
85        assert_eq!(ago(86_399), "23h ago");
86        assert_eq!(ago(86_400), "1d ago");
87        assert_eq!(ago(604_799), "6d ago");
88        assert_eq!(ago(604_800), "2027-01-08");
89    }
90
91    #[test]
92    fn counts_shorten_at_each_boundary() {
93        assert_eq!(compact_count(999), "999");
94        assert_eq!(compact_count(1_000), "1k");
95        assert_eq!(compact_count(1_250), "1.2k");
96        assert_eq!(compact_count(9_999), "9.9k");
97        assert_eq!(compact_count(10_000), "10k");
98        assert_eq!(compact_count(999_999), "999k");
99        assert_eq!(compact_count(1_000_000), "1M");
100        assert_eq!(compact_count(1_500_000), "1.5M");
101        assert_eq!(compact_count(12_000_000), "12M");
102    }
