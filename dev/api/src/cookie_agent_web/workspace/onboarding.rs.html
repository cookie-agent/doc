1//! Onboarding (`docs/site/specs/default-models-and-onboarding.md` section
2//! 9): the daemon's `onboarding.get` answer, the first-run sheet it opens
3//! on a daemon that is not set up, and the calls the sheet makes. "Not now"
4//! is this browser's choice; the daemon's answer stays what its
5//! configuration says.
6
7use cookie_agent_protocol::{OnboardingRunParams, OnboardingRunResult, OnboardingStatus};
8use leptos::{prelude::*, task::spawn_local};
9
10use super::Workspace;
11use crate::browser;
12
13/// Where this browser remembers "Not now": the page's origin is the
14/// daemon's, so it is one choice per daemon.
15pub const NOT_NOW_KEY: &str = "cookie.onboarding_not_now";
16
17/// The first-run sheet's state that outlives a render.
18#[derive(Clone, Copy)]
19pub struct Onboarding {
20    /// `onboarding.get`'s answer; `None` before it or from a daemon
21    /// without onboarding.
22    pub status: RwSignal<Option<OnboardingStatus>>,
23    pub open: RwSignal<bool>,
24    /// `onboarding.run` is in flight.
25    pub running: RwSignal<bool>,
26    /// What the last run did, for the summary that ends the sheet.
27    pub result: RwSignal<Option<OnboardingRunResult>>,
28    /// Why the last run was refused.
29    pub error: RwSignal<Option<String>>,
30}
31
32impl Default for Onboarding {
33    fn default() -> Self {
34        Self {
35            status: RwSignal::new(None),
36            open: RwSignal::new(false),
37            running: RwSignal::new(false),
38            result: RwSignal::new(None),
39            error: RwSignal::new(None),
40        }
41    }
42}

44/// Whether the sheet opens by itself for `status` on connecting: the daemon
45/// is not set up and this browser did not say "Not now".
46#[must_use]
47pub fn opens_on_connect(status: &OnboardingStatus, not_now: bool) -> bool {
48    !status.ready && !not_now
49}
50
51/// Whether an `onboarding.changed` closes an open sheet: the daemon is set
52/// up now and this page is not showing its own run's summary.
53#[must_use]
54pub fn closes_on_change(status: &OnboardingStatus, running: bool, summary: bool) -> bool {
55    status.ready && !running && !summary
56}
57
58impl Workspace {
59    /// Reads `onboarding.get` on connecting and opens the sheet when the
60    /// daemon is not set up, unless this browser said "Not now". A daemon
61    /// without onboarding answers with an error, which leaves it closed.
62    pub(super) fn refresh_onboarding(self, on_connect: bool) {
63        let Some(client) = self.client() else {
64            return;
65        };
66        let read = self.begin_read(|reads| &mut reads.onboarding);
67        spawn_local(async move {
68            let answer = client.get_onboarding().await;
69            // An `onboarding.changed` since the read was asked for is newer
70            // than its answer: on connecting, the sheet opens by that.
71            let answer = if self.read_is_current(|reads| &mut reads.onboarding, read) {
72                answer
73            } else if let Some(status) = self.onboarding.status.get_untracked() {
74                Ok(status)
75            } else {
76                return;
77            };
78            match answer {
79                Ok(status) => {
80                    let opens = on_connect
81                        && opens_on_connect(&status, browser::load(NOT_NOW_KEY).is_some());
82                    self.onboarding.status.set(Some(status));
83                    if opens {
84                        self.onboarding.result.set(None);
85                        self.onboarding.open.set(true);
86                    }
87                }
88                Err(_) => self.onboarding.status.set(None),
89            }
90        });
91    }
92
93    /// `onboarding.changed`: the new answer. Finishing in another tab closes
94    /// the sheet here.
95    pub(super) fn apply_onboarding_change(self, status: OnboardingStatus) {
96        self.supersede_reads(|reads| &mut reads.onboarding);
97        let closes = closes_on_change(
98            &status,
99            self.onboarding.running.get_untracked(),
100            self.onboarding.result.with_untracked(Option::is_some),
101        );
102        self.onboarding.status.set(Some(status));
103        if closes {
104            self.onboarding.open.set(false);
105        }
106    }
107
108    /// Opens the sheet: Settings' "Set up Cookie".
109    pub fn open_onboarding(self) {
110        self.onboarding.result.set(None);
111        self.onboarding.error.set(None);
112        self.onboarding.open.set(true);
113        self.refresh_onboarding(false);
114    }
115
116    /// "Not now": closes the sheet and keeps it closed on later visits.
117    pub fn dismiss_onboarding(self) {
118        browser::store(NOT_NOW_KEY, "1");
119        self.onboarding.open.set(false);
120    }
121
122    /// Closes the sheet, as its summary's Done does.
123    pub fn close_onboarding(self) {
124        self.onboarding.open.set(false);
125        self.onboarding.result.set(None);
126    }
127
128    /// `onboarding.run`; the summary on success, the reason in the sheet
129    /// otherwise.
130    pub fn run_onboarding(self, params: OnboardingRunParams) {
131        let Some(client) = self.client() else {
132            return;
133        };
134        self.onboarding.running.set(true);
135        self.onboarding.error.set(None);
136        spawn_local(async move {
137            match client.run_onboarding(params).await {
138                Ok(result) => {
139                    self.onboarding.status.set(Some(result.status.clone()));
140                    self.onboarding.result.set(Some(result));
141                    browser::forget(NOT_NOW_KEY);
142                }
143                Err(error) => self
144                    .onboarding
145                    .error
146                    .set(Some(crate::errors::describe(&error))),
147            }
148            self.onboarding.running.set(false);
149        });
150    }
151
152    /// "Start your first session": a new-session draft, on the daemon's
153    /// default agent when it has it.
154    pub fn start_first_session(self) {
155        self.close_onboarding();
156        self.start_new_session();
157        let default = untrack(|| self.default_agent());
158        if let Some(agent) = untrack(|| self.selectable_agents())
159            .into_iter()
160            .find(|agent| agent.id == default)
161        {
162            self.pick_agent(&agent);
163        }
164    }
165}

167#[cfg(test)]
168mod tests {
169    use cookie_agent_protocol::{OnboardingRequirement, OnboardingStatus};
170
171    use super::{closes_on_change, opens_on_connect};
172
173    fn status(ready: bool) -> OnboardingStatus {
174        OnboardingStatus {
175            ready,
176            missing: if ready {
177                Vec::new()
178            } else {
179                vec![OnboardingRequirement::Provider]
180            },
181            providers_connected: Vec::new(),
182            default_model: None,
183            small_model: None,
184            utility_model: None,
185        }
186    }
187
188    #[test]
189    fn the_sheet_opens_on_a_daemon_not_set_up_unless_not_now() {
190        assert!(opens_on_connect(&status(false), false));
191        assert!(!opens_on_connect(&status(false), true));
192        assert!(!opens_on_connect(&status(true), false));
193    }
194
195    #[test]
196    fn finishing_elsewhere_closes_the_sheet_but_not_this_page_s_summary() {
197        assert!(closes_on_change(&status(true), false, false));
198        assert!(!closes_on_change(&status(true), true, false));
199        assert!(!closes_on_change(&status(true), false, true));
200        assert!(!closes_on_change(&status(false), false, false));
201    }
