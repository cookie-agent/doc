1//! A provider's browser sign-in from the connect sheet. The daemon always
2//! listens for the redirect itself, so a browser on the daemon's machine
3//! finishes on its own. A page served from loopback is on that machine; any
4//! other page also offers to paste the address the sign-in tab could not
5//! load, for a browser elsewhere.
6
7use cookie_agent_protocol::{
8    ProviderAuthBeginParams, ProviderAuthCancelParams, ProviderAuthCompleteParams,
9    ProviderDescriptor, ProviderId,
10};
11use leptos::{prelude::*, task::spawn_local};
12
13use super::Workspace;
14use crate::{browser, oauth};
15
16/// Where one provider's sign-in stands in the connect sheet.
17#[derive(Clone, Debug, Default, Eq, PartialEq)]
18pub enum ProviderSignIn {
19    #[default]
20    Idle,
21    /// Waiting for the user to finish in the sign-in tab.
22    Waiting {
23        /// The address the sign-in tab shows, to open it again.
24        authorization_url: String,
25        /// The browser shares the daemon's machine, so the daemon surely
26        /// catches the redirect; nothing to paste.
27        caught_by_daemon: bool,
28    },
29    Failed(String),
30}
31
32/// Whether the page is served from this machine's loopback, the same
33/// machine as a local daemon.
34fn page_on_loopback() -> bool {
35    browser::page_href()
36        .and_then(|href| url::Url::parse(&href).ok())
37        .and_then(|url| url.host_str().map(str::to_owned))
38        .is_some_and(|host| matches!(host.as_str(), "127.0.0.1" | "localhost" | "[::1]"))
39}
40
41impl Workspace {
42    /// Starts signing in to `provider`. The sign-in tab opens now, while the
43    /// click counts, and is sent to the authorization page once the daemon
44    /// answers.
45    pub fn sign_in_provider(self, provider: ProviderDescriptor, state: RwSignal<ProviderSignIn>) {
46        let Some(client) = self.client() else {
47            return;
48        };
49        let Some(method) = cookie_agent_client_state::providers::sign_in_method(&provider) else {
50            return;
51        };
52        let caught_by_daemon = page_on_loopback();
53        let params = ProviderAuthBeginParams {
54            provider_id: provider.id.clone(),
55            method: method.id.clone(),
56            redirect_uri: None,
57        };
58        let tab = oauth::open_blank_tab();
59        state.set(ProviderSignIn::Idle);
60        spawn_local(async move {
61            match client.begin_provider_auth(params).await {
62                Ok(began) => {
63                    self.status.set(format!(
64                        "signing in to {} in the browser",
65                        began.provider_id
66                    ));
67                    oauth::go_to(tab, &began.authorization_url);
68                    state.set(ProviderSignIn::Waiting {
69                        authorization_url: began.authorization_url,
70                        caught_by_daemon,
71                    });
72                }
73                Err(error) => {
74                    if let Some(tab) = tab {
75                        let _ = tab.close();
76                    }
77                    state.set(ProviderSignIn::Failed(crate::errors::describe_full(&error)));
78                }
79            }
80        });
81    }
82
83    /// Finishes the sign-in with the address the sign-in tab could not load,
84    /// then calls `finished`.
85    pub fn finish_provider_sign_in(
86        self,
87        provider_id: ProviderId,
88        callback_url: String,
89        state: RwSignal<ProviderSignIn>,
90        finished: impl FnOnce() + 'static,
91    ) {
92        let Some(client) = self.client() else {
93            return;
94        };
95        spawn_local(async move {
96            match client
97                .complete_provider_auth(ProviderAuthCompleteParams {
98                    provider_id: provider_id.clone(),
99                    callback_url,
100                })
101                .await
102            {
103                Ok(completed) => {
104                    state.set(ProviderSignIn::Idle);
105                    finished();
106                    self.status
107                        .set(completed.account.email.as_ref().map_or_else(
108                            || format!("signed in to {provider_id}"),
109                            |email| format!("signed in to {provider_id} as {email}"),
110                        ));
111                }
112                Err(error) => {
113                    state.set(ProviderSignIn::Failed(crate::errors::describe_full(&error)));
114                }
115            }
116        });
117    }
118
119    /// Ends a sign-in in progress.
120    pub fn cancel_provider_sign_in(self, provider_id: ProviderId, state: RwSignal<ProviderSignIn>) {
121        state.set(ProviderSignIn::Idle);
122        let Some(client) = self.client() else {
123            return;
124        };
125        spawn_local(async move {
126            let _ = client
127                .cancel_provider_auth(ProviderAuthCancelParams { provider_id })
128                .await;
129        });
130    }
131}
