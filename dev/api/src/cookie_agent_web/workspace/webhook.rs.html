18/// The webhook form: what it edits and the targets it switches between.
19#[derive(Clone, Debug, PartialEq)]
20pub struct WebhookForm {
21    /// The webhook an edit changes; `None` for a new one.
22    pub editing: Option<Webhook>,
23    /// The form as it opened.
24    pub draft: WebhookDraft,
25    /// The session "this session" is, and whether it is a subagent's;
26    /// `None` when the form opened without one.
27    pub session: Option<(SessionId, bool)>,
28    /// The template "a new session per event" starts from.
29    pub template: WebhookTemplate,
30}

32/// A secret or URL the daemon returned once, shown until dismissed.
33#[derive(Clone, Debug, Eq, PartialEq)]
34pub struct WebhookRevealed {
35    pub webhook: WebhookId,
36    /// The webhook's title, for the note.
37    pub title: String,

40    /// Whether the URL is new (a create, or a rotated URL).

53    /// Reads `webhooks.list`, when the daemon keeps webhooks.
54    pub fn poll_webhooks(self) {
55        if self.is_guest() || !self.has_webhooks.get_untracked() {
56            return;
57        }
58        let Some(client) = self.client() else {
59            return;
60        };
61        let read = self.begin_read(|reads| &mut reads.webhooks);
62        spawn_local(async move {
63            let result = client
64                .list_webhooks()
65                .await
66                .map_err(|error| crate::errors::describe_full(&error));
67            if self.read_is_current(|reads| &mut reads.webhooks, read) {
68                self.webhooks.set(Some(result));
69            }
70        });
71    }

87    /// The webhooks listed, if they are.
88    pub fn webhook_list(self) -> Vec<Webhook> {
89        self.webhooks.with(|webhooks| {
90            webhooks
91                .as_ref()
92                .and_then(|webhooks| webhooks.as_ref().ok())
93                .map(|webhooks| webhooks.webhooks.clone())
94                .unwrap_or_default()
95        })
96    }

136    /// The ⋮ menu's "Add a webhook": the form, for a webhook that delivers
137    /// into the open session.
138    pub fn add_webhook(self) {
139        let Some((session, subagent)) = self.cron_session() else {
140            self.status
141                .set("open a session before adding a webhook to it".into());
142            return;
143        };
144        self.open_view(View::Webhooks, None);
145        self.webhook_form.set(Some(WebhookForm {
146            editing: None,
147            draft: WebhookDraft::for_session(session, subagent),
148            session: Some((session, subagent)),
149            template: self.webhook_template(),
150        }));
151    }

153    /// The view's "New": the form, for a webhook that delivers into the
154    /// open session, else one that starts a new session per event.
155    pub fn new_webhook(self) {
156        let session = self.cron_session();
157        let template = self.webhook_template();
158        let draft = match session {
159            Some((session, subagent)) => WebhookDraft::for_session(session, subagent),
160            None => WebhookDraft::for_new_session(template.clone()),
161        };
162        self.webhook_error.set(None);
163        self.webhook_form.set(Some(WebhookForm {
164            editing: None,
165            draft,
166            session,
167            template,
168        }));
169    }

171    /// The form, filled from `webhook`.
172    pub fn edit_webhook(self, webhook: Webhook) {
173        let session = webhook.target.session().map(|session| {
174            (
175                session,
176                matches!(
177                    webhook.target,
178                    cookie_agent_protocol::WebhookTarget::Subagent(_)
179                ),
180            )
181        });
182        let template = match &webhook.target {
183            cookie_agent_protocol::WebhookTarget::New(template) => template.clone(),
184            _ => self.webhook_template(),
185        };
186        self.webhook_error.set(None);
187        self.webhook_form.set(Some(WebhookForm {
188            draft: WebhookDraft::of_webhook(&webhook),
189            editing: Some(webhook),
190            session,
191            template,
192        }));
193    }

201    /// Sends the form: `webhooks.create` for a new webhook,
202    /// `webhooks.update` with what changed for an edit.
203    pub fn save_webhook(self, draft: WebhookDraft) {
204        let Some(client) = self.client() else {
205            return;
206        };
207        let Some(form) = self.webhook_form.get_untracked() else {
208            return;
209        };
210        let request = match &form.editing {
211            None => draft.create_params().map(Ok),
212            Some(webhook) => draft.update_params(webhook).map(Err),
213        };
214        let request = match request {
215            Ok(request) => request,
216            Err(message) => {
217                self.webhook_error.set(Some(message));
218                return;
219            }
220        };
221        if let Err(update) = &request
222            && !update_changes_anything(update)
223        {
224            self.close_webhook_form();
225            return;
226        }
227        self.webhook_error.set(None);
228        spawn_local(async move {
229            let created = request.is_ok();
230            let (verb, result) = match request {
231                Ok(create) => ("Added", client.create_webhook(create).await),
232                Err(update) => ("Saved", client.update_webhook(update).await),
233            };
234            match result {
235                Ok(result) => {
236                    let title = webhook_title(&result.webhook);
237                    self.status.set(format!("{verb} {title}."));
238                    self.webhook_selected.set(Some(result.webhook.id.clone()));
239                    if created || result.secret.is_some() {
240                        self.webhook_revealed.set(Some(WebhookRevealed {
241                            webhook: result.webhook.id.clone(),
242                            title,
243                            secret: result.secret,
244                            new_url: created,
245                        }));
246                    }
247                    self.webhook_form.set(None);
248                    self.poll_webhooks();
249                }
250                Err(error) => self
251                    .webhook_error
252                    .set(Some(crate::errors::describe_full(&error))),
253            }
254        });
255    }

257    /// Turns webhook `id` on or off.
258    pub fn set_webhook_enabled(self, id: WebhookId, enabled: bool) {
259        let Some(client) = self.client() else {
260            return;
261        };

263        update.enabled = Some(enabled);
264        spawn_local(async move {
265            match client.update_webhook(update).await {
266                Ok(result) => {
267                    let verb = if enabled { "Enabled" } else { "Disabled" };
268                    self.status
269                        .set(format!("{verb} {}.", webhook_title(&result.webhook)));
270                    self.webhook_error.set(None);
271                    self.poll_webhooks();
272                }
273                Err(error) => self
274                    .webhook_error
275                    .set(Some(crate::errors::describe(&error))),
276            }
277        });

308    /// Replaces webhook `id`'s URL or secret; a new secret is shown once.
309    pub fn rotate_webhook(self, id: WebhookId, what: WebhookRotate) {
310        let Some(client) = self.client() else {
311            return;
312        };
313        spawn_local(async move {
314            match client.rotate_webhook(id, what).await {
315                Ok(result) => {
316                    let title = webhook_title(&result.webhook);
317                    self.status.set(match what {
318                        WebhookRotate::Url => format!("{title} has a new URL."),
319                        WebhookRotate::Secret => format!("{title} has a new secret."),
320                    });
321                    self.webhook_revealed.set(Some(WebhookRevealed {
322                        webhook: result.webhook.id.clone(),
323                        title,
324                        secret: result.secret,
325                        new_url: what == WebhookRotate::Url,
326                    }));
327                    self.webhook_error.set(None);
328                    self.poll_webhooks();
329                }
330                Err(error) => self
331                    .webhook_error
332                    .set(Some(crate::errors::describe(&error))),
333            }
334        });
335    }

337    /// Shows webhook `id`'s signing secret.
338    pub fn reveal_webhook_secret(self, id: WebhookId, title: String) {
339        let Some(client) = self.client() else {
340            return;
341        };
342        spawn_local(async move {
343            match client.webhook_secret(id.clone()).await {
344                Ok(result) => {
345                    self.webhook_error.set(None);
346                    match result.secret {
347                        Some(secret) => self.webhook_revealed.set(Some(WebhookRevealed {
348                            webhook: id,
349                            title,
350                            secret: Some(secret),
351                            new_url: false,
352                        })),
353                        None => self.status.set(format!("{title} has no secret.")),
354                    }
355                }
356                Err(error) => self
357                    .webhook_error
358                    .set(Some(crate::errors::describe(&error))),
359            }
360        });
361    }

363    /// Deletes webhook `id`; its sessions stay.
364    pub fn delete_webhook(self, id: WebhookId, title: String) {
365        let Some(client) = self.client() else {
366            return;
367        };
368        spawn_local(async move {
369            match client.delete_webhook(id.clone()).await {
370                Ok(_) => {
371                    self.status.set(format!("Deleted {title}."));
372                    if self.webhook_selected.get_untracked() == Some(id.clone()) {
373                        self.webhook_selected.set(None);
374                    }
375                    if self.webhook_revealed.with_untracked(|shown| {
376                        shown.as_ref().is_some_and(|shown| shown.webhook == id)
377                    }) {
378                        self.webhook_revealed.set(None);
379                    }
380                    self.webhook_error.set(None);
381                    self.poll_webhooks();
382                }
383                Err(error) => self
384                    .webhook_error
385                    .set(Some(crate::errors::describe(&error))),
386            }
387        });
388    }

390    /// Opens event `event` of webhook `id`, reading its request; a second
391    /// press on the same event closes it.
392    pub fn toggle_webhook_event(self, id: WebhookId, event: String) {
393        let open = self.webhook_event.with_untracked(|open| {
394            open.as_ref()
395                .is_some_and(|open| open.webhook == id && open.event == event)
396        });
397        if open {
398            self.webhook_event.set(None);
399            return;
400        }
401        let Some(client) = self.client() else {
402            return;
403        };
404        self.webhook_event.set(Some(WebhookEventOpen {
405            webhook: id.clone(),
406            event: event.clone(),
407            request: None,
408        }));
409        spawn_local(async move {
410            let result = client
411                .webhook_event(id.clone(), event.clone())
412                .await
413                .map_err(|error| crate::errors::describe(&error));
414            self.webhook_event.update(|open| {
415                if let Some(open) = open
416                    && open.webhook == id
417                    && open.event == event
418                {
419                    open.request = Some(result);
420                }
421            });
422        });
423    }
