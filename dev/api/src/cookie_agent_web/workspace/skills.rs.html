1//! `/skills`: the session's skills (`skills.list`), searched, and run with
2//! their arguments: `skills.get` checks the skill, then the encoded skill
3//! submission is sent as the TUI sends it.
4
5use cookie_agent_protocol::{InputDelivery, SkillsGetParams, SkillsListParams};
6use leptos::{prelude::*, task::spawn_local};
7
8use super::{Panel, Workspace};
9
10impl Workspace {
11    /// Opens `/skills`, searching for `query`.
12    pub(super) fn open_skills(self, query: String) {
13        let (Some(client), Some(session_id)) = (self.client(), self.selected.get_untracked())
14        else {
15            self.status
16                .set("select a session before running a skill".into());
17            return;
18        };
19        self.skill_query.set(query);
20        self.skill_pick.set(None);
21        self.skill_args.set(String::new());
22        self.skills.set(None);
23        self.panel.set(Some(Panel::Skills));
24        spawn_local(async move {
25            let result = client
26                .list_skills(SkillsListParams {
27                    session_id: Some(session_id),
28                })
29                .await
30                .map(|result| result.skills)
31                .map_err(|error| crate::errors::describe_full(&error));
32            self.skills.set(Some(result));
33        });
34    }
35
36    /// Runs the picked skill with the typed arguments.
37    pub fn run_skill(self) {
38        let (Some(client), Some(session_id), Some(skill)) = (
39            self.client(),
40            self.selected.get_untracked(),
41            self.skill_pick.get_untracked(),
42        ) else {
43            return;
44        };
45        if !self.writable() {
46            self.status
47                .set("cannot run a skill in a read-only session".into());
48            return;
49        }
50        let name = skill.name;
51        let args = self.skill_args.get_untracked();
52        self.panel.set(None);
53        spawn_local(async move {
54            match client
55                .get_skill(SkillsGetParams {
56                    session_id,
57                    name: name.clone(),
58                    args: args.clone(),
59                })
60                .await
61            {
62                Ok(result) if result.skill.user_invocable => {
63                    let text = cookie_agent_protocol::encode_skill_submission(&name, &args);
64                    let content = cookie_agent_client_state::ComposerInput::text(text.clone())
65                        .content()
66                        .expect("a skill submission is text");
67                    self.send_content(content, InputDelivery::Steer, String::new(), Vec::new());
68                }
69                Ok(_) => self.status.set("skill is not user-invocable".into()),
70                Err(error) => self.status.set(format!(
71                    "load skill failed: {}",
72                    crate::errors::describe(&error)
73                )),
74            }
75        });
76    }
77}
