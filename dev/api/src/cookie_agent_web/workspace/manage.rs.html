76    /// Reads again what the open panel shows after a `*.changed`
77    /// notification.
78    pub(super) fn refresh_open_panel(self, delivery: &ClientDelivery) {
79        let runtimes = self.panel.get_untracked() == Some(Panel::Runtimes)
80            || self.showing(View::Machines, None);
81        match delivery {
82            ClientDelivery::RuntimesChanged(_) if runtimes => self.poll_runtimes(),
83            ClientDelivery::MachinesChanged(_) => {
84                if runtimes {
85                    self.poll_runtimes();
86                }
87                // Renaming or merging a machine moves its memories too.
88                if self.showing(View::Memory, None) {
89                    self.poll_memory();
90                }
91            }
92            ClientDelivery::McpChanged(_)
93                if self.showing(View::Settings, Some(SettingsSection::Mcp)) =>
94            {
95                self.poll_mcp();
96            }
97            ClientDelivery::MemoryChanged(_) if self.showing(View::Memory, None) => {
98                self.poll_memory();
99            }
100            ClientDelivery::PluginsChanged(_)
101                if self.showing(View::Settings, Some(SettingsSection::Plugins)) =>
102            {
103                self.poll_plugins();
104            }
105            _ => {}
106        }
107    }

116    /// Reads `runtimes.list` and `machines.list`; a new session's draft
117    /// gets its runtimes once the listing says where `local` works.
118    pub fn poll_runtimes(self) {
119        let Some(client) = self.client() else {
120            return;
121        };
122        spawn_local(async move {
123            match client.list_runtimes().await {
124                Ok(listing) => {
125                    self.runtimes.set(Some(listing));
126                    if self.new_session.get_untracked()
127                        && self.panel.get_untracked() == Some(Panel::Runtimes)
128                    {
129                        self.materialize_draft_runtimes();
130                    }
131                }
132                Err(error) => self.status.set(format!(
133                    "runtimes unavailable: {}",
134                    crate::errors::describe(&error)
135                )),
136            }
137            let machines = client
138                .list_machines()
139                .await
140                .map_err(|error| crate::errors::describe_full(&error));
141            self.machines.set(Some(machines));
142        });
143    }

147    pub fn runtime_rows(self) -> Vec<RuntimeRow> {
148        let listing = self.runtimes.get();
149        if self.new_session.get() {
150            return self.draft_runtimes.with(|entries| {
151                draft_runtime_rows(entries.as_deref().unwrap_or_default(), listing.as_ref())
152            });
153        }
154        let Some(session_id) = self.selected.get() else {
155            return Vec::new();
156        };
157        self.with_store(|store| {
158            store
159                .sessions
160                .get(&session_id)
161                .map(|state| {
162                    runtime_rows(state.runtime_attachments.attachments(), listing.as_ref())
163                })
164                .unwrap_or_default()
165        })
166    }

168    /// Gives the new-session draft its own runtimes, starting from what
169    /// `session.create` attaches by default: `global`, and `local`'s
170    /// `coding` at the chosen directory (else where the daemon's own
171    /// instance works); their instances open at once.
172    fn materialize_draft_runtimes(self) {
173        if self.draft_runtimes.with_untracked(Option::is_some) {
174            return;
175        }
176        let directory = self.new_directory.get_untracked().trim().to_owned();
177        let workspace = if directory.is_empty() {
178            self.runtimes.with_untracked(|listing| {
179                listing.as_ref().and_then(|listing| {
180                    listing
181                        .instances
182                        .iter()
183                        .find(|instance| {
184                            instance.runtime_id.as_str() == RuntimeId::LOCAL
185                                && instance.capability == CapabilityName::coding()
186                        })
187                        .and_then(|instance| instance.workspace.clone())
188                })
189            })
190        } else {
191            Some(directory)
192        };
193        self.draft_runtimes.set(Some(Vec::new()));
194        let global =
195            RuntimeId::new(RuntimeId::GLOBAL)
196                .ok()
197                .map(|runtime_id| RuntimeAttachRequest {
198                    capability: Some(cookie_agent_protocol::default_capability(&runtime_id)),
199                    runtime_id,
200                    workspace: None,
201                });
202        let coding = RuntimeId::new(RuntimeId::LOCAL)
203            .ok()
204            .map(|runtime_id| RuntimeAttachRequest {
205                runtime_id,
206                capability: Some(CapabilityName::coding()),
207                workspace,
208            });
209        for request in global.into_iter().chain(coding) {
210            let entry = self.push_draft_runtime(request);
211            // `global` is always running; opening it would only retry its
212            // MCP servers.
213            if !entry.is_global() {
214                self.open_draft_runtime(&entry);
215            }
216        }
217    }

219    fn push_draft_runtime(self, request: RuntimeAttachRequest) -> DraftRuntime {
220        self.next_draft_runtime.update_value(|next| *next += 1);
221        let entry = DraftRuntime {
222            id: self.next_draft_runtime.get_value(),
223            request,
224            opened: None,
225        };
226        self.draft_runtimes
227            .update(|entries| entries.get_or_insert_with(Vec::new).push(entry.clone()));
228        entry
229    }

231    /// Opens (or retries) a draft runtime's instance, so the panel shows its
232    /// status before the session attaches it.
233    fn open_draft_runtime(self, entry: &DraftRuntime) {
234        let Some(client) = self.client() else {
235            return;
236        };
237        let (id, request) = (entry.id, entry.request.clone());
238        spawn_local(async move {
239            let result = client
240                .reconnect_runtime(RuntimesReconnectParams {
241                    runtime_id: request.runtime_id.clone(),
242                    capability: request.capability,
243                    workspace: request.workspace,
244                })
245                .await
246                .map(|result| result.instance)
247                .map_err(|error| crate::errors::describe_full(&error));
248            self.install_draft_runtime(id, result);
249        });
250    }

252    /// Records what opening draft runtime `id` answered. An instance the
253    /// draft already attaches under another spelling of its workspace
254    /// leaves the draft.
255    fn install_draft_runtime(
256        self,
257        id: u64,
258        result: Result<cookie_agent_protocol::RuntimeInstanceInfo, String>,
259    ) {
260        let mut status = None;
261        self.draft_runtimes.update(|entries| {
262            let Some(entries) = entries.as_mut() else {
263                return;
264            };
265            let Some(index) = entries.iter().position(|entry| entry.id == id) else {
266                return;
267            };
268            let runtime_id = entries[index].request.runtime_id.clone();
269            if let Ok(instance) = &result {
270                let request = RuntimeAttachRequest {
271                    runtime_id: instance.runtime_id.clone(),
272                    capability: Some(instance.capability.clone()),
273                    workspace: instance.workspace.clone(),
274                };
275                if entries
276                    .iter()
277                    .enumerate()
278                    .any(|(other, entry)| other != index && entry.same_instance(&request))
279                {
280                    entries.remove(index);
281                    status = Some(format!(
282                        "the new session already attaches runtime `{runtime_id}` there"
283                    ));
284                    return;
285                }
286            }
287            if let Err(error) = &result {
288                status = Some(format!("runtime `{runtime_id}` cannot start: {error}"));
289            }
290            entries[index].opened = Some(result);
291        });
292        if let Some(status) = status {
293            self.status.set(status);
294        }
295    }

297    /// Attaches `runtime_id`'s `capability` at `workspace`: to the viewed
298    /// session, or to the new session's draft.
299    pub fn attach_runtime(
300        self,
301        runtime_id: RuntimeId,
302        capability: Option<CapabilityName>,
303        workspace: Option<String>,
304    ) {
305        let request = RuntimeAttachRequest {
306            runtime_id: runtime_id.clone(),
307            capability: capability.clone(),
308            workspace: workspace.clone(),
309        };
310        if self.new_session.get_untracked() {
311            self.materialize_draft_runtimes();
312            if self.draft_runtimes.with_untracked(|entries| {
313                entries
314                    .iter()
315                    .flatten()
316                    .any(|entry| entry.same_instance(&request))
317            }) {
318                self.status.set(format!(
319                    "the new session already attaches runtime `{runtime_id}` there"
320                ));
321                return;
322            }
323            self.status.set(format!("opening runtime `{runtime_id}`…"));
324            let entry = self.push_draft_runtime(request);
325            self.open_draft_runtime(&entry);
326            return;
327        }
328        let (Some(client), Some(session_id)) = (self.client(), self.selected.get_untracked())
329        else {
330            self.status
331                .set("select a session before attaching a runtime".into());
332            return;
333        };
334        self.status
335            .set(format!("attaching runtime `{runtime_id}`…"));
336        spawn_local(async move {
337            let status = match client
338                .attach_runtime(SessionRuntimeAttachParams {
339                    session_id,
340                    runtime_id,
341                    capability,
342                    workspace,
343                })
344                .await
345            {
346                Ok(result) => format!("attached runtime as `{}`", result.attachment.name),
347                Err(error) => format!("attach failed: {}", crate::errors::describe(&error)),
348            };
349            self.status.set(status);
350            self.poll_runtimes();
351        });
352    }

354    /// Detaches the viewed session's attachment `name`, or takes the
355    /// draft's entry `index` out.
356    pub fn detach_runtime(self, name: AttachmentName, index: usize) {
357        if self.new_session.get_untracked() {
358            let mut removed = None;
359            self.draft_runtimes.update(|entries| {
360                if let Some(entries) = entries.as_mut()
361                    && index < entries.len()
362                {
363                    removed = Some(entries.remove(index));
364                }
365            });
366            if let Some(entry) = removed {
367                self.status.set(format!(
368                    "the new session will not attach runtime `{}`",
369                    entry.request.runtime_id
370                ));
371            }
372            return;
373        }
374        let (Some(client), Some(session_id)) = (self.client(), self.selected.get_untracked())
375        else {
376            return;
377        };
378        spawn_local(async move {
379            let status = match client
380                .detach_runtime(SessionRuntimeDetachParams {
381                    session_id,
382                    name: name.clone(),
383                })
384                .await
385            {
386                Ok(_) => format!("detached runtime `{name}`"),
387                Err(error) => format!("detach failed: {}", crate::errors::describe(&error)),
388            };
389            self.status.set(status);
390            self.poll_runtimes();
391        });
392    }

394    /// Retries row `index`'s instance (`runtimes.reconnect`).
395    pub fn retry_runtime(self, row: RuntimeRow, index: usize) {
396        if self.new_session.get_untracked() {
397            let mut entry = None;
398            self.draft_runtimes.update(|entries| {
399                if let Some(found) = entries.as_mut().and_then(|entries| entries.get_mut(index)) {
400                    found.opened = None;
401                    entry = Some(found.clone());
402                }
403            });
404            if let Some(entry) = entry {
405                self.status
406                    .set(format!("retrying runtime `{}`…", entry.request.runtime_id));
407                self.open_draft_runtime(&entry);
408            }
409            return;
410        }
411        let Some(client) = self.client() else {
412            return;
413        };
414        let runtimes = self
415            .runtimes
416            .with_untracked(|listing| listing.as_ref().map(|listing| listing.runtimes.clone()))
417            .unwrap_or_default();
418        let (runtime_id, capability, workspace) =
419            match cookie_agent_client_state::runtimes::reconnect_request(&runtimes, &row) {
420                Ok(request) => request,
421                Err(error) => {
422                    self.status.set(error);
423                    return;
424                }
425            };
426        self.status.set(format!("retrying runtime `{runtime_id}`…"));
427        spawn_local(async move {
428            let status = match client
429                .reconnect_runtime(RuntimesReconnectParams {
430                    runtime_id: runtime_id.clone(),
431                    capability: Some(capability),
432                    workspace,
433                })
434                .await
435            {
436                Ok(result) => reconnect_status(&runtime_id, &result.instance),
437                Err(error) => format!("retry failed: {}", crate::errors::describe(&error)),
438            };
439            self.status.set(status);
440            self.poll_runtimes();
441        });
442    }

444    /// Sends a machine change; the error reads as the daemon's registry
445    /// error says.
446    pub fn change_machines(self, request: MachineRequest) {
447        let Some(client) = self.client() else {
448            return;
449        };
450        self.status.set("changing machines…".into());
451        spawn_local(async move {
452            let done = request.done();
453            let result = match request {
454                MachineRequest::Rename { machine, name } => {
455                    client.rename_machine(machine, name).await
456                }
457                MachineRequest::Assign { runtime, machine } => {
458                    client.assign_machine(runtime, machine).await
459                }
460                MachineRequest::Unassign { runtime } => client.unassign_machine(runtime).await,
461                MachineRequest::Merge { from, into } => client.merge_machines(from, into).await,
462                MachineRequest::Forget { machine } => client.forget_machine(machine).await,
463            };
464            match result {
465                Ok(machines) => {
466                    self.status.set(done);
467                    self.machines.set(Some(Ok(machines)));
468                }
469                Err(error) => self.status.set(format!(
470                    "machine change failed: {}",
471                    crate::errors::describe(&error)
472                )),
473            }
474            self.poll_runtimes();
475        });
476    }

478    /// Writes `template` into `config.toml` with a `config.apply` set of
479    /// its `[runtimes.<id>]` table, unless the daemon lists a runtime of
480    /// that ID or the file has a template of it. A set does not depend on
481    /// the rest of the file, so a revision that moved meanwhile is read
482    /// again once. `outcome` gets what the form says.
483    pub fn add_runtime_template(
484        self,
485        template: SshTemplate,
486        outcome: RwSignal<Option<Result<String, String>>>,
487    ) {
488        let Some(client) = self.client() else {
489            return;
490        };
491        let id = template.id.clone();
492        let listed = self.runtimes.with_untracked(|listing| {
493            listing.as_ref().is_some_and(|listing| {
494                listing
495                    .runtimes
496                    .iter()
497                    .any(|runtime| runtime.runtime_id == id)
498            })
499        });
500        if listed {
501            outcome.set(Some(Err(format!(
502                "The daemon has a runtime `{id}` already."
503            ))));
504            return;
505        }
506        outcome.set(None);
507        self.status.set(format!("adding runtime {id}…"));
508        spawn_local(async move {
509            let mut retried = false;
510            let result = loop {
511                let config = match client.get_config().await {
512                    Ok(config) => config,
513                    Err(error) => break Err(crate::errors::describe(&error)),
514                };
515                let text = config
516                    .documents
517                    .iter()
518                    .find(|document| document.path == "config.toml")
519                    .map_or("", |document| document.text.as_str());
520                if has_template(text, &id) {
521                    break Err(format!(
522                        "config.toml has a [runtimes.{id}] template already."
523                    ));
524                }
525                match ConfigApplyOutcome::from_result(
526                    client
527                        .apply_config(ConfigApplyParams {
528                            expected_revision: config.revision,
529                            changes: vec![template.change()],
530                        })
531                        .await,
532                ) {
533                    ConfigApplyOutcome::Refused {
534                        reason: cookie_agent_protocol::ConfigApplyErrorCode::Conflict,
535                        ..
536                    } if !retried => retried = true,
537                    ConfigApplyOutcome::Applied(status) => {
538                        break Ok(format!(
539                            "Added [runtimes.{id}] to config.toml.{}",
540                            restart_note(&status)
541                        ));
542                    }
543                    ConfigApplyOutcome::Refused { detail, .. } => {
544                        break Err(format!("config.toml was not changed: {detail}"));
545                    }
546                    ConfigApplyOutcome::Failed(error) => break Err(error),
547                }
548            };
549            self.status.set(match &result {
550                Ok(done) | Err(done) => done.clone(),
551            });
552            outcome.set(Some(result));
553            if self.config.with_untracked(Option::is_some) {
554                self.poll_config();
555            }
556        });
557    }

584    /// Adds or edits a server from its form, then keeps it where `persist`
585    /// says. `editing` names the edited server, which an edit cannot rename.
586    pub fn submit_mcp(
587        self,
588        fields: McpFields,
589        editing: Option<McpServerInfo>,
590        persist: PersistChoice,
591    ) -> Result<(), String> {
592        let base = editing
593            .as_ref()
594            .map_or_else(new_definition, |server| server.definition.clone());
595        let (name, definition) = fields.definition(&base)?;
596        if editing.as_ref().is_some_and(|server| server.name != name) {
597            return Err("editing cannot rename a server; remove it and add the new name".into());
598        }
599        let Some(client) = self.client() else {
600            return Err("not connected".into());
601        };
602        let target = persist.target();
603        spawn_local(async move {
604            let result = async {
605                let mutation = if editing.is_some() {
606                    client
607                        .edit_mcp_server(McpServerEditParams {
608                            name: name.clone(),
609                            definition,
610                        })
611                        .await
612                } else {
613                    client
614                        .add_mcp_server(McpServerAddParams {
615                            name: name.clone(),
616                            definition,
617                        })
618                        .await
619                }
620                .map_err(|error| crate::errors::describe_full(&error))?;
621                if let Some(target) = target {
622                    return client
623                        .persist_mcp_server(McpServerPersistParams { name, target })
624                        .await
625                        .map(|result| result.server)
626                        .map_err(|error| crate::errors::describe_full(&error));
627                }
628                Ok(mutation.server)
629            }
630            .await;
631            self.finish_mcp(result);
632        });
633        Ok(())
634    }

685    fn finish_mcp(self, result: Result<Option<McpServerInfo>, String>) {
686        self.status.set(match result {
687            Ok(Some(server)) => format!(

710    /// Connects `provider` with what its form holds; the credentials are
711    /// read from the form once and only travel in the request.
712    pub fn connect_provider(
713        self,
714        provider: ProviderDescriptor,
715        auth_method: cookie_agent_protocol::AuthMethodId,
716        setup_values: std::collections::BTreeMap<
717            cookie_agent_protocol::SetupFieldId,
718            cookie_agent_protocol::SafeSetupValue,
719        >,
720        auth_values: cookie_agent_protocol::ProviderCredentialValues,
721        reconnect: bool,
722    ) -> Result<(), String> {
723        let Some(client) = self.client() else {
724            return Err("not connected".into());
725        };
726        let Some(expected_revision) = self
727            .runtime
728            .with_value(|runtime| runtime.revision().cloned())
729        else {
730            return Err("Catalog revision is unavailable.".into());
731        };
732        let action = if reconnect {
733            ProviderAction::Reconnect
734        } else {
735            ProviderAction::Connect
736        };
737        let provider_id = provider.id.clone();
738        self.provider_operations.update(|operations| {
739            operations.insert(provider_id.clone(), ProviderOperation::InProgress(action));
740        });
741        self.status
742            .set(format!("{} provider {provider_id}…", action_name(action)));
743        let params = ProviderConnectParams {
744            client_connect_id: ClientConnectId::new(uuid::Uuid::now_v7().to_string())
745                .expect("uuid-derived client connect id"),
746            provider_id: provider_id.clone(),
747            expected_revision,
748            setup_values,
749            auth_method,
750            auth_values,
751        };
752        spawn_local(async move {
753            let result = client.connect_provider_retrying(params, &provider).await;
754            self.finish_provider(provider_id, action, result.map(|_| ()));
755        });
756        Ok(())
757    }

759    /// Disconnects `provider`'s stored connection.
760    pub fn disconnect_provider(self, provider: ProviderDescriptor) {
761        let Some(client) = self.client() else {
762            return;
763        };
764        let Some(expected_revision) = self
765            .runtime
766            .with_value(|runtime| runtime.revision().cloned())
767        else {
768            self.status
769                .set("Runtime snapshot unavailable; retry before disconnecting.".into());
770            return;
771        };
772        let params = ProviderDisconnectParams {
773            provider_id: provider.id.clone(),
774            expected_revision,
775            expected_connection_generation: provider
776                .durable_connection
777                .as_ref()
778                .map(|connection| connection.connection_generation),
779            client_request_id: ClientRequestId::new(uuid::Uuid::now_v7().to_string())
780                .expect("uuid-derived client request id"),
781        };
782        let provider_id = provider.id.clone();
783        self.provider_operations.update(|operations| {
784            operations.insert(
785                provider_id.clone(),
786                ProviderOperation::InProgress(ProviderAction::Disconnect),
787            );
788        });
789        self.status
790            .set(format!("disconnect provider {provider_id}…"));
791        spawn_local(async move {
792            let mut attempt = 0;
793            let result = loop {
794                match client.disconnect_provider(params.clone()).await {
795                    Ok(_) => break Ok(()),
796                    Err(error) if retry_store_contention_once(attempt, &error) => attempt += 1,
797                    Err(error) => break Err(error),
798                }
799            };
800            self.finish_provider(provider_id, ProviderAction::Disconnect, result);
801        });
802    }
803
804    fn finish_provider(
805        self,
806        provider_id: cookie_agent_protocol::ProviderId,
807        action: ProviderAction,
808        result: Result<(), cookie_agent_protocol::ClientError>,
809    ) {
810        match result {
811            Ok(()) => {
812                self.provider_operations.update(|operations| {
813                    operations.remove(&provider_id);
814                });
815                self.status.set(match action {
816                    ProviderAction::Disconnect => format!("Disconnected provider {provider_id}."),
817                    _ => format!("Connected provider {provider_id}."),
818                });
819                self.resync_runtime();
820            }
821            Err(error) => {
822                let message = store_contention_message(&error, "providers");
823                self.provider_operations.update(|operations| {
824                    operations.insert(
825                        provider_id.clone(),
826                        ProviderOperation::Error {
827                            action,
828                            message: message.clone(),
829                        },
830                    );
831                });
832                self.status.set(format!(
833                    "{} {provider_id} failed: {message}",
834                    action_name(action)
835                ));
836            }
837        }
838    }

906    /// Applies the editor's `text` with `config.apply` at the revision the
907    /// document was read at.
908    pub fn apply_config(self, text: String) {
909        let (Some(client), Some(edit)) = (self.client(), self.config_edit.get_untracked()) else {
910            return;
911        };
912        if edit.original.as_deref() == Some(text.as_str()) {
913            self.config_drafts.update(|drafts| {
914                drafts.remove(&edit.path);
915            });
916            self.config_edit.set(None);
917            self.status.set(format!("No changes to {}.", edit.path));
918            return;
919        }
920        let edit = ConfigEdit { text, ..edit };
921        self.status.set(format!("applying {}…", edit.path));
922        let change = ConfigChange::Put {
923            path: edit.path.clone(),
924            text: edit.text.clone(),
925            expected_fingerprint: edit.expected_fingerprint.clone(),
926        };
927        spawn_local(async move {
928            let outcome = ConfigApplyOutcome::from_result(
929                client
930                    .apply_config(ConfigApplyParams {
931                        expected_revision: edit.revision.clone(),
932                        changes: vec![change],
933                    })
934                    .await,
935            );
936            self.finish_config(edit.path.clone(), Some(edit), outcome);
937        });
938    }

940    /// Deletes the agent document `path`.
941    pub fn delete_config(self, path: String) {
942        let (Some(client), Some(Ok(config))) = (self.client(), self.config.get_untracked()) else {
943            return;
944        };
945        let Some(document) = config
946            .documents
947            .iter()
948            .find(|document| document.path == path)
949        else {
950            return;
951        };
952        self.status.set(format!("deleting {path}…"));
953        let change = ConfigChange::Delete {
954            path: path.clone(),
955            expected_fingerprint: document.fingerprint.clone(),
956        };
957        let revision = config.revision.clone();
958        spawn_local(async move {
959            let outcome = ConfigApplyOutcome::from_result(
960                client
961                    .apply_config(ConfigApplyParams {
962                        expected_revision: revision,
963                        changes: vec![change],
964                    })
965                    .await,
966            );
967            self.finish_config(path, None, outcome);
968        });
969    }

971    fn finish_config(self, path: String, edit: Option<ConfigEdit>, outcome: ConfigApplyOutcome) {
972        let report = outcome.report(&path);
973        let failed = matches!(outcome, ConfigApplyOutcome::Failed(_));
974        if report.keep_draft {
975            if let Some(edit) = edit.filter(|_| !failed) {
976                self.config_drafts.update(|drafts| {
977                    drafts.insert(path.clone(), edit.clone());
978                });
979                // The editor stays open on the refused text, to be fixed.
980                self.config_edit.set(Some(edit));
981            }
982        } else {
983            self.config_drafts.update(|drafts| {
984                drafts.remove(&path);
985            });
986            if self
987                .config_edit
988                .with_untracked(|open| open.as_ref().is_some_and(|open| open.path == path))
989            {
990                self.config_edit.set(None);
991            }
992        }
993        if !failed {
994            self.config_error.set(report.error);
995        }
996        self.status.set(report.status);
997        self.poll_config();
998    }

1097    fn write_memory(self, edit: MemoryEdit) {
1098        let Some(client) = self.client() else {
1099            return;
1100        };
1101        self.status.set(format!("saving {}…", edit.url));
1102        spawn_local(async move {
1103            let outcome = MemoryWriteOutcome::from_result(
1104                client
1105                    .write_memory(edit.url.clone(), edit.text.clone(), edit.revision.clone())
1106                    .await,
1107            );
1108            match outcome {
1109                MemoryWriteOutcome::Saved { notice } => {
1110                    self.memory_pending.set(None);
1111                    self.memory_edit.set(None);
1112                    self.status.set(saved_status(
1113                        &edit.url,
1114                        edit.revision.is_none(),
1115                        notice.as_deref(),
1116                    ));
1117                }
1118                MemoryWriteOutcome::Invalid(message) => {
1119                    self.status.set(format!("Not saved: {message}"));
1120                    self.memory_edit.set(None);
1121                    self.memory_pending
1122                        .set(Some(MemoryPending::Invalid { edit, message }));
1123                }
1124                MemoryWriteOutcome::Changed => {
1125                    self.status.set(format!(
1126                        "Not saved: {} changed since it was opened.",
1127                        edit.url
1128                    ));
1129                    self.memory_edit.set(None);
1130                    self.memory_pending
1131                        .set(Some(MemoryPending::Changed { edit }));
1132                }
1133                MemoryWriteOutcome::Failed(error) => {
1134                    self.status
1135                        .set(format!("{} was not saved: {error}", edit.url));
1136                }
1137            }
1138            self.poll_memory();
1139        });
1140    }

1142    /// Answers a pending write: edit it again, reload the current memory,
1143    /// overwrite it with this text, or drop it.
1144    pub fn resolve_memory(self, choice: MemoryChoice) {
1145        let Some(pending) = self.memory_pending.get_untracked() else {
1146            return;
1147        };
1148        self.memory_pending.set(None);
1149        match (choice, pending) {
1150            (MemoryChoice::EditAgain, MemoryPending::Invalid { edit, .. }) => {
1151                self.memory_edit.set(Some(edit));
1152            }
1153            (MemoryChoice::Reload, MemoryPending::Changed { edit }) => self.edit_memory(edit.url),
1154            (MemoryChoice::Overwrite, MemoryPending::Changed { edit }) => {
1155                let Some(client) = self.client() else {
1156                    return;
1157                };
1158                spawn_local(async move {
1159                    let revision = client
1160                        .read_memory(edit.url.clone())
1161                        .await
1162                        .ok()
1163                        .map(|read| read.revision);
1164                    self.write_memory(MemoryEdit { revision, ..edit });
1165                });
1166            }
1167            (_, MemoryPending::Invalid { edit, .. } | MemoryPending::Changed { edit }) => {
1168                self.status
1169                    .set(format!("Your edit of {} was dropped.", edit.url));
1170            }
1171        }
1172    }

1189    /// Deletes every memory of `project` with `memory.delete`, each at the
1190    /// revision listed; the folder's other files stay, and so does the
1191    /// folder when there are any. A memory that is refused does not stop
1192    /// the others.
1193    pub fn delete_project(self, project: String) {
1194        let (Some(client), Some(Ok(listing))) = (self.client(), self.memory.get_untracked()) else {
1195            return;
1196        };
1197        let memories = project_memories(&listing, &project);
1198        let strays = project_strays(&listing, &project);
1199        self.status.set(format!("deleting project {project}…"));
1200        spawn_local(async move {
1201            let mut deleted = 0;
1202            let mut failure = None;
1203            for (url, revision) in &memories {
1204                match client.delete_memory(url.clone(), revision.clone()).await {
1205                    Ok(_) => deleted += 1,
1206                    Err(error) => {
1207                        failure.get_or_insert_with(|| delete_refusal(url, &error));
1208                    }
1209                }
1210            }
1211            let mut status =
1212                project_delete_status(&project, deleted, memories.len(), failure.as_deref());
1213            if strays > 0 {
1214                status.push_str(&format!(
1215                    " Its folder keeps {} that {} not memories.",
1216                    if strays == 1 {
1217                        "1 file".to_owned()
1218                    } else {
1219                        format!("{strays} files")
1220                    },
1221                    if strays == 1 { "is" } else { "are" }
1222                ));
1223            } else if failure.is_none() {
1224                crate::browser::forget(&folder_key(&project));
1225            }
1226            self.status.set(status);
1227            self.poll_memory();
1228        });
1229    }
