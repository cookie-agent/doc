53/// What renaming declaration `old` to `new` did to its documents.
54fn rename_status(old: &AgentId, new: &AgentId, renamed: &RuntimesAgentsRenameResult) -> String {
55    let mut status = format!("ACP agent `{old}` is `{new}` now");
56    if !renamed.moved.is_empty() {
57        status.push_str(&format!("; moved {}", renamed.moved.join(", ")));
58    }
59    if !renamed.stale_templates.is_empty() {
60        status.push_str(&format!(
61            "; {} still name `{old}` in acp.agent",
62            renamed.stale_templates.join(", ")
63        ));
64    }
65    status
66}

112    /// Reads again what the open panel shows after a `*.changed`
113    /// notification.
114    pub(super) fn refresh_open_panel(self, delivery: &ClientDelivery) {
115        let runtimes = self.panel.get_untracked() == Some(Panel::Runtimes)
116            || self.showing(View::Machines, None)
117            || (self.new_session.get_untracked() && self.showing(View::Chat, None));
118        match delivery {
119            ClientDelivery::RuntimesChanged(_) if runtimes => self.poll_runtimes(),
120            ClientDelivery::MachinesChanged(_) => {
121                if runtimes {
122                    self.poll_runtimes();
123                }
124                // Renaming or merging a machine moves its memories too.
125                if self.showing(View::Memory, None) {
126                    self.poll_memory();
127                }
128            }
129            ClientDelivery::McpChanged(_)
130                if runtimes || self.showing(View::Settings, Some(SettingsSection::Mcp)) =>
131            {
132                self.poll_mcp();
133            }
134            ClientDelivery::MemoryChanged(_) if self.showing(View::Memory, None) => {
135                self.poll_memory();
136            }
137            ClientDelivery::PluginsChanged(_)
138                if self.showing(View::Settings, Some(SettingsSection::Plugins)) =>
139            {
140                self.poll_plugins();
141            }
142            _ => {}
143        }
144    }

162    /// Reads `runtimes.list`, `machines.list`, and `mcp.server.list`.
163    pub fn poll_runtimes(self) {
164        let Some(client) = self.client() else {
165            return;
166        };
167        let read = self.begin_read(|reads| &mut reads.runtimes);
168        spawn_local(async move {
169            let runtimes = client.list_runtimes().await;
170            let machines = client
171                .list_machines()
172                .await
173                .map_err(|error| crate::errors::describe_full(&error));
174            if !self.read_is_current(|reads| &mut reads.runtimes, read) {
175                return;
176            }
177            match runtimes {
178                Ok(listing) => self.runtimes.set(Some(listing)),
179                Err(error) => self.status.set(format!(
180                    "runtimes unavailable: {}",
181                    crate::errors::describe(&error)
182                )),
183            }
184            self.machines.set(Some(machines));
185        });
186        // The MCP servers each runtime hosts, which its rows list.
187        self.poll_mcp();
188    }

266    /// Retries `row`'s instance (`runtimes.reconnect`).
267    pub fn retry_runtime(self, row: RuntimeRow) {
268        let Some(client) = self.client() else {
269            return;
270        };
271        let runtimes = self
272            .runtimes
273            .with_untracked(|listing| listing.as_ref().map(|listing| listing.runtimes.clone()))
274            .unwrap_or_default();
275        let (runtime_id, capability, workspace) =
276            match cookie_agent_client_state::runtimes::reconnect_request(&runtimes, &row) {
277                Ok(request) => request,
278                Err(error) => {
279                    self.status.set(error);
280                    return;
281                }
282            };
283        self.status.set(format!("retrying runtime `{runtime_id}`…"));
284        spawn_local(async move {
285            let status = match client
286                .reconnect_runtime(RuntimesReconnectParams {
287                    runtime_id: runtime_id.clone(),
288                    capability: Some(capability),
289                    workspace,
290                })
291                .await
292            {
293                Ok(result) => reconnect_status(&runtime_id, &result.instance),
294                Err(error) => format!("retry failed: {}", crate::errors::describe(&error)),
295            };
296            self.status.set(status);
297            self.poll_runtimes();
298        });
299    }

301    /// Sends a machine change; the error reads as the daemon's registry
302    /// error says.
303    pub fn change_machines(self, request: MachineRequest) {
304        let Some(client) = self.client() else {
305            return;
306        };
307        spawn_local(async move {
308            let done = request.done();
309            let result = match request {
310                MachineRequest::Rename { machine, name } => {
311                    client.rename_machine(machine, name).await
312                }
313                MachineRequest::Assign { runtime, machine } => {
314                    client.assign_machine(runtime, machine).await
315                }
316                MachineRequest::Unassign { runtime } => client.unassign_machine(runtime).await,
317                MachineRequest::Merge { from, into } => client.merge_machines(from, into).await,
318                MachineRequest::Forget { machine } => client.forget_machine(machine).await,
319                MachineRequest::ForgetRuntime { runtime } => {
320                    client.forget_machine_runtime(runtime).await
321                }
322            };
323            match result {
324                Ok(machines) => {
325                    self.status.set(done);
326                    self.machines.set(Some(Ok(machines)));
327                }
328                Err(error) => self.status.set(format!(
329                    "machine change failed: {}",
330                    crate::errors::describe(&error)
331                )),
332            }
333            self.poll_runtimes();
334        });
335    }

337    /// Writes `template` into `config.toml` with a `config.apply` set of
338    /// its `[runtimes.<id>]` table, unless the daemon lists a runtime of
339    /// that ID or the file has a template of it. A set does not depend on
340    /// the rest of the file, so a revision that moved meanwhile is read
341    /// again once. `outcome` gets what the form says.
342    pub fn add_runtime_template(
343        self,
344        template: SshTemplate,
345        outcome: RwSignal<Option<Result<String, String>>>,
346    ) {
347        let Some(client) = self.client() else {
348            return;
349        };
350        let id = template.id.clone();
351        let listed = self.runtimes.with_untracked(|listing| {
352            listing.as_ref().is_some_and(|listing| {
353                listing
354                    .runtimes
355                    .iter()
356                    .any(|runtime| runtime.runtime_id == id)
357            })
358        });
359        if listed {
360            outcome.set(Some(Err(format!(
361                "The daemon has a runtime `{id}` already."
362            ))));
363            return;
364        }
365        outcome.set(None);
366        self.status.set(format!("adding runtime {id}…"));
367        spawn_local(async move {
368            let mut retried = false;
369            let result = loop {
370                let config = match client.get_config().await {
371                    Ok(config) => config,
372                    Err(error) => break Err(crate::errors::describe(&error)),
373                };
374                let text = config
375                    .documents
376                    .iter()
377                    .find(|document| document.path == "config.toml")
378                    .map_or("", |document| document.text.as_str());
379                if has_template(text, &id) {
380                    break Err(format!(
381                        "config.toml has a [runtimes.{id}] template already."
382                    ));
383                }
384                match ConfigApplyOutcome::from_result(
385                    client
386                        .apply_config(ConfigApplyParams {
387                            expected_revision: config.revision,
388                            changes: vec![template.change()],
389                        })
390                        .await,
391                ) {
392                    ConfigApplyOutcome::Refused {
393                        reason: cookie_agent_protocol::ConfigApplyErrorCode::Conflict,
394                        ..
395                    } if !retried => retried = true,
396                    ConfigApplyOutcome::Applied(status) => {
397                        break Ok(format!(
398                            "Added [runtimes.{id}] to config.toml.{}",
399                            restart_note(&status)
400                        ));
401                    }
402                    ConfigApplyOutcome::Refused { detail, .. } => {
403                        break Err(format!("config.toml was not changed: {detail}"));
404                    }
405                    ConfigApplyOutcome::Failed(error) => break Err(error),
406                }
407            };
408            self.status.set(match &result {
409                Ok(done) | Err(done) => done.clone(),
410            });
411            outcome.set(Some(result));
412            if self.config.with_untracked(Option::is_some) {
413                self.poll_config();
414            }
415        });
416    }

446    /// Adds or edits a server from its form. `editing` names the edited
447    /// server, whose runtime the change goes to; a new one goes to
448    /// `runtime`, `global` when `None`. A changed name renames the server
449    /// first. `scope` is a runtime server's, and `persist` says where a
450    /// `global` one is kept.
451    pub fn submit_mcp(
452        self,
453        fields: McpFields,
454        editing: Option<McpServerInfo>,
455        persist: PersistChoice,
456        runtime: Option<RuntimeId>,
457        scope: Option<McpServerScope>,
458    ) -> Result<(), String> {
459        let base = editing
460            .as_ref()
461            .map_or_else(new_definition, |server| server.definition.clone());
462        let (name, mut definition) = fields.definition(&base)?;
463        let runtime_id = match &editing {
464            Some(server) => mutation_runtime(server),
465            None => runtime.filter(|runtime| runtime.as_str() != RuntimeId::GLOBAL),
466        };
467        if runtime_id.is_some() {
468            definition.scope = scope;
469        }
470        let renamed = editing
471            .as_ref()
472            .map(|server| server.name.clone())
473            .filter(|old| *old != name);
474        let Some(client) = self.client() else {
475            return Err("not connected".into());
476        };
477        let target = persist.target().filter(|_| runtime_id.is_none());
478        spawn_local(async move {
479            let result = async {
480                if let Some(old) = renamed {
481                    client
482                        .rename_mcp_server(McpServerRenameParams {
483                            runtime_id: runtime_id.clone(),
484                            name: old,
485                            new_name: name.clone(),
486                        })
487                        .await
488                        .map_err(|error| crate::errors::describe_full(&error))?;
489                }
490                let mutation = if editing.is_some() {
491                    client
492                        .edit_mcp_server(McpServerEditParams {
493                            runtime_id,
494                            name: name.clone(),
495                            definition,
496                        })
497                        .await
498                } else {
499                    client
500                        .add_mcp_server(McpServerAddParams {
501                            runtime_id,
502                            name: name.clone(),
503                            definition,
504                        })
505                        .await
506                }
507                .map_err(|error| crate::errors::describe_full(&error))?;
508                if let Some(target) = target {
509                    return client
510                        .persist_mcp_server(McpServerPersistParams { name, target })
511                        .await
512                        .map(|result| result.server)
513                        .map_err(|error| crate::errors::describe_full(&error));
514                }
515                Ok(mutation.server)
516            }
517            .await;
518            self.finish_mcp(result);
519        });
520        Ok(())
521    }

609    /// Adds or edits one of `runtime`'s ACP agent declarations from its form;
610    /// `editing` names the edited one, which a changed key renames first.
611    pub fn submit_runtime_agent(
612        self,
613        runtime: RuntimeId,
614        fields: &AcpAgentFields,
615        editing: Option<AgentId>,
616        into: RuntimeAgentsSignal,
617    ) -> Result<(), String> {
618        let (agent, definition) = fields.definition()?;
619        let renamed = editing.filter(|old| *old != agent);
620        let Some(client) = self.client() else {
621            return Err("not connected".into());
622        };
623        spawn_local(async move {
624            let result = async {
625                if let Some(old) = renamed {
626                    let renamed = client
627                        .rename_runtime_agent(RuntimesAgentsRenameParams {
628                            runtime_id: runtime.clone(),
629                            agent: old.clone(),
630                            new_agent: agent.clone(),
631                        })
632                        .await
633                        .map_err(|error| crate::errors::describe_full(&error))?;
634                    self.status.set(rename_status(&old, &agent, &renamed));
635                }
636                client
637                    .set_runtime_agent(RuntimesAgentsSetParams {
638                        runtime_id: runtime.clone(),
639                        agent,
640                        definition,
641                    })
642                    .await
643                    .map(|result| result.agents)
644                    .map_err(|error| crate::errors::describe_full(&error))
645            }
646            .await;
647            self.finish_runtime_agents(runtime, result, into);
648        });
649        Ok(())
650    }

675    fn finish_runtime_agents(
676        self,
677        runtime: RuntimeId,
678        result: Result<Vec<RuntimeAgentDeclaration>, String>,
679        into: RuntimeAgentsSignal,

762    /// Connects `provider` with what its form holds; the credentials are
763    /// read from the form once and only travel in the request.
764    pub fn connect_provider(
765        self,
766        provider: ProviderDescriptor,
767        auth_method: cookie_agent_protocol::AuthMethodId,
768        setup_values: std::collections::BTreeMap<
769            cookie_agent_protocol::SetupFieldId,
770            cookie_agent_protocol::SafeSetupValue,
771        >,
772        auth_values: cookie_agent_protocol::ProviderCredentialValues,
773        reconnect: bool,
774    ) -> Result<(), String> {
775        let Some(client) = self.client() else {
776            return Err("not connected".into());
777        };
778        let Some(expected_revision) = self
779            .runtime
780            .with_value(|runtime| runtime.revision().cloned())
781        else {
782            return Err("Catalog revision is unavailable.".into());
783        };
784        let action = if reconnect {
785            ProviderAction::Reconnect
786        } else {
787            ProviderAction::Connect
788        };
789        let provider_id = provider.id.clone();
790        self.provider_operations.update(|operations| {
791            operations.insert(provider_id.clone(), ProviderOperation::InProgress(action));
792        });
793        self.status
794            .set(format!("{} provider {provider_id}…", action_name(action)));
795        let params = ProviderConnectParams {
796            client_connect_id: ClientConnectId::new(uuid::Uuid::now_v7().to_string())
797                .expect("uuid-derived client connect id"),
798            provider_id: provider_id.clone(),
799            expected_revision,
800            setup_values,
801            auth_method,
802            auth_values,
803        };
804        spawn_local(async move {
805            let result = client.connect_provider_retrying(params, &provider).await;
806            self.finish_provider(provider_id, action, result.map(|_| ()));
807        });
808        Ok(())
809    }

811    /// Disconnects `provider`'s stored connection.
812    pub fn disconnect_provider(self, provider: ProviderDescriptor) {
813        let Some(client) = self.client() else {
814            return;
815        };
816        let Some(expected_revision) = self
817            .runtime
818            .with_value(|runtime| runtime.revision().cloned())
819        else {
820            self.status
821                .set("Runtime snapshot unavailable; retry before disconnecting.".into());
822            return;
823        };
824        let params = ProviderDisconnectParams {
825            provider_id: provider.id.clone(),
826            expected_revision,
827            expected_connection_generation: provider
828                .durable_connection
829                .as_ref()
830                .map(|connection| connection.connection_generation),
831            client_request_id: ClientRequestId::new(uuid::Uuid::now_v7().to_string())
832                .expect("uuid-derived client request id"),
833        };
834        let provider_id = provider.id.clone();
835        self.provider_operations.update(|operations| {
836            operations.insert(
837                provider_id.clone(),
838                ProviderOperation::InProgress(ProviderAction::Disconnect),
839            );
840        });
841        self.status
842            .set(format!("disconnect provider {provider_id}…"));
843        spawn_local(async move {
844            let mut attempt = 0;
845            let result = loop {
846                match client.disconnect_provider(params.clone()).await {
847                    Ok(_) => break Ok(()),
848                    Err(error) if retry_store_contention_once(attempt, &error) => attempt += 1,
849                    Err(error) => break Err(error),
850                }
851            };
852            self.finish_provider(provider_id, ProviderAction::Disconnect, result);
853        });
854    }
855
856    fn finish_provider(
857        self,
858        provider_id: cookie_agent_protocol::ProviderId,
859        action: ProviderAction,
860        result: Result<(), cookie_agent_protocol::ClientError>,
861    ) {
862        match result {
863            Ok(()) => {
864                self.provider_operations.update(|operations| {
865                    operations.remove(&provider_id);
866                });
867                self.status.set(match action {
868                    ProviderAction::Disconnect => format!("Disconnected provider {provider_id}."),
869                    _ => format!("Connected provider {provider_id}."),
870                });
871                self.resync_runtime();
872            }
873            Err(error) => {
874                let message = store_contention_message(&error, "providers");
875                self.provider_operations.update(|operations| {
876                    operations.insert(
877                        provider_id.clone(),
878                        ProviderOperation::Error {
879                            action,
880                            message: message.clone(),
881                        },
882                    );
883                });
884                self.status.set(format!(
885                    "{} {provider_id} failed: {message}",
886                    action_name(action)
887                ));
888            }
889        }
890    }

968    /// Applies the editor's `text` with `config.apply` at the revision the
969    /// document was read at.
970    pub fn apply_config(self, text: String) {
971        let (Some(client), Some(edit)) = (self.client(), self.config_edit.get_untracked()) else {
972            return;
973        };
974        if edit.original.as_deref() == Some(text.as_str()) {
975            self.config_drafts.update(|drafts| {
976                drafts.remove(&edit.path);
977            });
978            self.config_edit.set(None);
979            self.status.set(format!("No changes to {}.", edit.path));
980            return;
981        }
982        let edit = ConfigEdit { text, ..edit };
983        self.status.set(format!("applying {}…", edit.path));
984        let change = ConfigChange::Put {
985            path: edit.path.clone(),
986            text: edit.text.clone(),
987            expected_fingerprint: edit.expected_fingerprint.clone(),
988        };
989        spawn_local(async move {
990            let outcome = ConfigApplyOutcome::from_result(
991                client
992                    .apply_config(ConfigApplyParams {
993                        expected_revision: edit.revision.clone(),
994                        changes: vec![change],
995                    })
996                    .await,
997            );
998            self.finish_config(edit.path.clone(), Some(edit), outcome);
999        });
1000    }

1033    fn finish_config(self, path: String, edit: Option<ConfigEdit>, outcome: ConfigApplyOutcome) {
1034        let report = outcome.report(&path);
1035        let failed = matches!(outcome, ConfigApplyOutcome::Failed(_));
1036        if report.keep_draft {
1037            if let Some(edit) = edit.filter(|_| !failed) {
1038                self.config_drafts.update(|drafts| {
1039                    drafts.insert(path.clone(), edit.clone());
1040                });
1041                // The editor stays open on the refused text, to be fixed.
1042                self.config_edit.set(Some(edit));
1043            }
1044        } else {
1045            self.config_drafts.update(|drafts| {
1046                drafts.remove(&path);
1047            });
1048            if self
1049                .config_edit
1050                .with_untracked(|open| open.as_ref().is_some_and(|open| open.path == path))
1051            {
1052                self.config_edit.set(None);
1053            }
1054        }
1055        if !failed {
1056            self.config_error.set(report.error);
1057        }
1058        self.status.set(report.status);
1059        self.poll_config();
1060    }

1112    /// Reads `url` into the editor; a missing memory opens the template.
1113    pub fn edit_memory(self, url: String) {
1114        let Some(client) = self.client() else {
1115            return;
1116        };
1117        self.memory_pending.set(None);
1118        self.status.set(format!("reading {url}…"));
1119        spawn_local(async move {
1120            let version = client.daemon_version();
1121            match client.read_memory(url.clone()).await {
1122                Ok(read) => {
1123                    self.status.set(match read.invalid.as_deref() {
1124                        Some(reason) => format!("{url} is not a valid memory: {reason}"),
1125                        None => format!("editing {url}"),
1126                    });
1127                    let last_read = self.memory.with_untracked(|listing| {
1128                        listing
1129                            .as_ref()
1130                            .and_then(|listing| listing.as_ref().ok())
1131                            .and_then(|listing| {
1132                                listing
1133                                    .memories
1134                                    .iter()
1135                                    .find(|memory| memory.url == read.url)
1136                            })
1137                            .and_then(|memory| memory.last_read)
1138                    });
1139                    // The daemon owns the stamp lines and writes its own on
1140                    // every save: the editor leaves them out.
1141                    let text = editor_text(&read.text, version);
1142                    let parts = self
1143                        .listed_parts(&client, &read.url, &read.revision, &text)
1144                        .await;
1145                    self.memory_edit.set(Some(MemoryEdit {
1146                        stamps: Some(MemoryStamps::of_read(&read, last_read)),
1147                        url: read.url,
1148                        text: text.clone(),
1149                        original: text,
1150                        revision: Some(read.revision),
1151                        read_parts: parts.clone(),
1152                        parts,
1153                    }));
1154                }
1155                Err(error) if error.is(&errors::MEMORY_NOT_FOUND) => {
1156                    self.memory_edit.set(Some(new_memory(url)));
1157                }
1158                Err(error) => self.status.set(format!(
1159                    "{url} could not be read: {}",
1160                    crate::errors::describe(&error)
1161                )),
1162            }
1163        });
1164    }

1166    /// The parts of `url`, read at `revision` as `text`, from the listing:
1167    /// the panel's when it lists that revision, else a fresh one, since a
1168    /// save's refresh may not have landed yet.
1169    async fn listed_parts(
1170        self,
1171        client: &cookie_agent_protocol::Client,
1172        url: &str,
1173        revision: &str,
1174        text: &str,
1175    ) -> Option<MemoryParts> {
1176        let listed = self.memory.with_untracked(|listing| {
1177            let listing = listing.as_ref()?.as_ref().ok()?;
1178            let memory = listing.memories.iter().find(|memory| memory.url == url)?;
1179            (memory.revision == revision).then(|| memory_parts(memory, revision, text))
1180        });
1181        if let Some(parts) = listed {
1182            return parts;
1183        }
1184        let listing = client.list_memory(MemoryListParams::default()).await.ok()?;
1185        let memory = listing.memories.iter().find(|memory| memory.url == url)?;
1186        memory_parts(memory, revision, text)
1187    }

1224    fn write_memory(self, edit: MemoryEdit) {
1225        let Some(client) = self.client() else {
1226            return;
1227        };
1228        self.status.set(format!("saving {}…", edit.url));
1229        spawn_local(async move {
1230            let outcome = MemoryWriteOutcome::from_result(
1231                client
1232                    .write_memory(edit.url.clone(), edit.text.clone(), edit.revision.clone())
1233                    .await,
1234            );
1235            match outcome {
1236                MemoryWriteOutcome::Saved { notice } => {
1237                    self.memory_pending.set(None);
1238                    self.memory_edit.set(None);
1239                    self.status.set(saved_status(
1240                        &edit.url,
1241                        edit.revision.is_none(),
1242                        notice.as_deref(),

1244                }
1245                MemoryWriteOutcome::Invalid(message) => {
1246                    self.status.set(format!("Not saved: {message}"));
1247                    self.memory_edit.set(None);
1248                    self.memory_pending
1249                        .set(Some(MemoryPending::Invalid { edit, message }));
1250                }
1251                MemoryWriteOutcome::Changed => {

1253                        "Not saved: {} changed since it was opened.",
1254                        edit.url
1255                    ));
1256                    self.memory_edit.set(None);
1257                    self.memory_pending
1258                        .set(Some(MemoryPending::Changed { edit }));
1259                }
1260                MemoryWriteOutcome::Failed(error) => {
1261                    self.status
1262                        .set(format!("{} was not saved: {error}", edit.url));
1263                }
1264            }
1265            self.poll_memory();
1266        });
1267    }

1269    /// Answers a pending write: edit it again, reload the current memory,
1270    /// overwrite it with this text, or drop it.
1271    pub fn resolve_memory(self, choice: MemoryChoice) {
1272        let Some(pending) = self.memory_pending.get_untracked() else {
1273            return;
1274        };
1275        self.memory_pending.set(None);
1276        match (choice, pending) {
1277            (MemoryChoice::EditAgain, MemoryPending::Invalid { edit, .. }) => {
1278                self.memory_edit.set(Some(edit));
1279            }
1280            (MemoryChoice::Reload, MemoryPending::Changed { edit }) => self.edit_memory(edit.url),
1281            (MemoryChoice::Overwrite, MemoryPending::Changed { edit }) => {
1282                let Some(client) = self.client() else {
1283                    return;
1284                };
1285                spawn_local(async move {
1286                    let revision = client
1287                        .read_memory(edit.url.clone())
1288                        .await
1289                        .ok()
1290                        .map(|read| read.revision);
1291                    self.write_memory(MemoryEdit { revision, ..edit });
1292                });
1293            }
1294            (_, MemoryPending::Invalid { edit, .. } | MemoryPending::Changed { edit }) => {
1295                self.status
1296                    .set(format!("Your edit of {} was dropped.", edit.url));
1297            }
1298        }
1299    }

1316    /// Deletes every memory of `project` with `memory.delete`, each at the
1317    /// revision listed; the folder's other files stay, and so does the
1318    /// folder when there are any. A memory that is refused does not stop
1319    /// the others.
1320    pub fn delete_project(self, project: String) {

1322            return;
1323        };
1324        let memories = project_memories(&listing, &project);
1325        let strays = project_strays(&listing, &project);
1326        self.status.set(format!("deleting project {project}…"));
1327        spawn_local(async move {
1328            let mut deleted = 0;
1329            let mut failure = None;
1330            for (url, revision) in &memories {
1331                match client.delete_memory(url.clone(), revision.clone()).await {
1332                    Ok(_) => deleted += 1,
1333                    Err(error) => {

1337            }
1338            let mut status =
1339                project_delete_status(&project, deleted, memories.len(), failure.as_deref());
1340            if strays > 0 {
1341                status.push_str(&format!(
1342                    " Its folder keeps {} that {} not memories.",
1343                    if strays == 1 {
1344                        "1 file".to_owned()
1345                    } else {
1346                        format!("{strays} files")
1347                    },
1348                    if strays == 1 { "is" } else { "are" }
1349                ));
1350            } else if failure.is_none() {
1351                crate::browser::forget(&folder_key(&project));
1352            }
1353            self.status.set(status);
1354            self.poll_memory();
1355        });
1356    }
1357
1358    // ---- /plugins ---------------------------------------------------------
1359
1360    pub fn poll_plugins(self) {
1361        let Some(client) = self.client() else {
1362            return;
1363        };
1364        let read = self.begin_read(|reads| &mut reads.plugins);
1365        spawn_local(async move {
1366            let result = client
1367                .list_plugins()
1368                .await
1369                .map(|listing| listing.plugins)
1370                .map_err(|error| crate::errors::describe_full(&error));
1371            if !self.read_is_current(|reads| &mut reads.plugins, read) {
1372                return;
1373            }
1374            if let Err(error) = &result {
1375                self.status.set(format!("plugin refresh failed: {error}"));
1376            }
1377            self.plugins.set(Some(result));
1378        });
1379    }
1380
1381    pub fn restart_plugin(self, name: String) {
1382        let Some(client) = self.client() else {
1383            return;
1384        };
1385        self.status.set(format!("restarting plugin `{name}`…"));
1386        spawn_local(async move {
1387            self.status
1388                .set(match client.restart_plugin(name.clone()).await {
1389                    Ok(_) => format!("plugin `{name}` is restarting"),
1390                    Err(error) => crate::errors::describe(&error),
1391                });
1392            self.poll_plugins();
1393        });
1394    }
