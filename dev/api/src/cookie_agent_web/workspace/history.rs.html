1//! History edits, as the TUI makes them: a user message's menu (copy,
2//! revert, fork), and `/compact` with its cancel.
3
4use cookie_agent_client_state::TranscriptItem;
5use cookie_agent_protocol::{
6    ClientForkId, SafeDisplayText, SessionCompactCancelParams, SessionCompactParams,
7    SessionForkParams, SessionId, SessionRevertParams,
8};
9use leptos::{prelude::*, task::spawn_local};
10
11use super::{Panel, Workspace, client_run_id};

13/// The message a menu was opened on. Its text is captured on opening: a
14/// rebuild (a concurrent revert) can change the transcript before the
15/// action runs.
16#[derive(Clone, Debug, PartialEq)]
17pub struct UserMenu {
18    pub session_id: SessionId,
19    /// The physical sequence of the message's `user_input_submitted`.
20    pub seq: u64,
21    pub text: String,
22    /// The revert is waiting for its confirmation.
23    pub confirming_revert: bool,
24}
25
26impl Workspace {
27    /// Opens the menu of the viewed session's user message `seq`.
28    pub fn open_user_menu(self, seq: u64) {
29        let Some(session_id) = self.selected.get_untracked() else {
30            return;
31        };
32        let text = self.store.with_value(|store| {
33            store.sessions.get(&session_id).and_then(|state| {
34                state.transcript.iter().find_map(|item| match item {
35                    TranscriptItem::User {
36                        seq: item_seq,
37                        markdown,
38                        ..
39                    } if *item_seq == seq => Some(markdown.as_str().to_owned()),
40                    _ => None,
41                })
42            })
43        });
44        let Some(text) = text else {
45            self.status
46                .set("message is no longer in the visible branch".into());
47            return;
48        };
49        self.user_menu.set(Some(UserMenu {
50            session_id,
51            seq,
52            text,
53            confirming_revert: false,
54        }));
55        self.panel.set(Some(Panel::Message));
56    }
57
58    /// Asks before a revert, which removes the message and every later
59    /// turn.
60    pub fn confirm_revert(self, confirming: bool) {
61        self.user_menu.update(|menu| {
62            if let Some(menu) = menu {
63                menu.confirming_revert = confirming;
64            }
65        });
66    }
67
68    /// Reverts the session to just before the menu's message
69    /// (`through_seq = seq - 1`), naming the tip this view shows so the
70    /// daemon refuses instead of deleting turns another client added. The
71    /// `rewound` notification rebuilds the transcript; the message text
72    /// comes back into the composer.
73    pub fn revert(self) {
74        let (Some(client), Some(menu)) = (self.client(), self.user_menu.get_untracked()) else {
75            return;
76        };
77        self.panel.set(None);
78        self.user_menu.set(None);
79        let Some(expected_last_seq) = self.store.with_value(|store| {
80            store
81                .sessions
82                .get(&menu.session_id)
83                .map(|state| state.last_seq)
84        }) else {
85            self.status.set("session is not loaded".into());
86            return;
87        };
88        // User messages always follow `session_created` (sequence 1).
89        let through_seq = menu.seq.saturating_sub(1).max(1);
90        self.status.set("reverting to before the message…".into());
91        spawn_local(async move {
92            match client
93                .revert_session(SessionRevertParams {
94                    session_id: menu.session_id,
95                    through_seq,
96                    expected_last_seq,
97                })
98                .await
99            {
100                Ok(result) => {
101                    let text = result.instructions_override.unwrap_or(menu.text);
102                    if self.selected.get_untracked() == Some(menu.session_id) {
103                        self.composer.update(|composer| {
104                            if composer.is_empty() {
105                                *composer = text;
106                            } else {
107                                composer.push('\n');
108                                composer.push_str(&text);
109                            }
110                        });
111                        self.status
112                            .set("reverted; the message text is back in the composer".into());
113                    }
114                    self.refresh_tree();
115                }
116                Err(error) if error.is(&cookie_agent_protocol::errors::STALE_REVERT) => {
117                    self.status.set(
118                        "revert refused: the session changed since you opened this message; \
119                         review the new turns and revert again"
120                            .into(),
121                    );
122                }
123                Err(error) => self.status.set(format!("revert failed: {error}")),
124            }
125        });
126    }
127
128    /// Forks the session at the menu's message (`through_seq = seq`,
129    /// keeping the message), then opens the fork.
130    pub fn fork(self) {
131        let (Some(client), Some(menu)) = (self.client(), self.user_menu.get_untracked()) else {
132            return;
133        };
134        self.panel.set(None);
135        self.user_menu.set(None);
136        self.status
137            .set("forking the session from the message…".into());
138        spawn_local(async move {
139            match client
140                .fork_session(SessionForkParams {
141                    session_id: menu.session_id,
142                    through_seq: menu.seq,
143                    client_fork_id: ClientForkId::new(client_run_id().as_str())
144                        .expect("a client run id is a valid fork id"),
145                })
146                .await
147            {
148                Ok(result) => {
149                    let forked = result.session_id;
150                    self.owned.update_value(|owned| {
151                        owned.insert(forked);
152                    });
153                    self.read_only.update(|read_only| {
154                        read_only.remove(&forked);
155                    });
156                    self.refresh_lists().await;
157                    self.reroot_tree(forked);
158                    self.open_session(forked);
159                    self.status.set("forked the session; switched to it".into());
160                }
161                Err(error) => self.status.set(format!("fork failed: {error}")),
162            }
163        });
164    }
165
166    /// Copies the menu's message with `navigator.clipboard`; elsewhere the
167    /// sheet's text is selected for the user to copy.
168    pub fn copy_message(self) {
169        let Some(menu) = self.user_menu.get_untracked() else {
170            return;
171        };
172        spawn_local(async move {
173            let copied = crate::browser::copy_text(&menu.text).await;
174            self.status.set(if copied {
175                "message copied".into()
176            } else {
177                crate::browser::select_text("message-text");
178                "copying needs a secure page; the text is selected, copy it with the keyboard"
179                    .into()
180            });
181        });
182    }
183
184    /// Whether the viewed session compacts outside any run, which the
185    /// gestures that cancel a run cancel.
186    pub(super) fn compacting(self) -> bool {
187        let Some(session_id) = self.selected.get_untracked() else {
188            return false;
189        };
190        self.store.with_value(|store| {
191            store
192                .sessions
193                .get(&session_id)
194                .is_some_and(|state| state.active_run.is_none() && state.is_compacting())
195        })
196    }
197
198    /// Compacts the viewed session's context (`/compact [focus]`).
199    pub fn compact(self, focus: String) {
200        let (Some(client), Some(session_id)) = (self.client(), self.selected.get_untracked())
201        else {
202            self.status
203                .set("select a session before compacting context".into());
204            return;
205        };
206        if !self.writable() {
207            self.status.set("cannot compact a read-only session".into());
208            return;
209        }
210        let focus = match (!focus.trim().is_empty())
211            .then(|| SafeDisplayText::new(focus.trim()))
212            .transpose()
213        {
214            Ok(focus) => focus,
215            Err(_) => {
216                self.status
217                    .set("compaction focus must be control-free and at most 1024 bytes".into());
218                return;
219            }
220        };
221        self.status.set("compacting context…".into());
222        spawn_local(async move {
223            let status = match client
224                .compact_session(SessionCompactParams { session_id, focus })
225                .await
226            {
227                Ok(result) if result.compacted => "context compacted".into(),
228                Ok(result) => result.cancellation_reason.unwrap_or_else(|| {
229                    "context did not require or could not produce a smaller checkpoint".into()
230                }),
231                Err(error) => format!("context compaction failed: {error}"),
232            };
233            if self.selected.get_untracked() == Some(session_id) {
234                self.status.set(status);
235            }
236        });
237    }
238
239    /// Cancels the viewed session's compaction.
240    pub(super) fn cancel_compaction(self) {
241        let (Some(client), Some(session_id)) = (self.client(), self.selected.get_untracked())
242        else {
243            return;
244        };
245        spawn_local(async move {
246            let status = match client
247                .cancel_compaction(SessionCompactCancelParams { session_id })
248                .await
249            {
250                Ok(result) if result.cancelled => "compaction cancellation requested".into(),
251                Ok(_) => "no compaction to cancel".into(),
252                Err(error) => error.to_string(),
253            };
254            self.status.set(status);
255        });
256    }
