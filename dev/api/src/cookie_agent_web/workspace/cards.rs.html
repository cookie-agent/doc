1//! The new session's runtime cards, above its composer: what they offer
2//! from `runtimes.list`, the machine groups the user opened, and where the
3//! draft will work. The first message creates the session with the
4//! selected cards (`Workspace::submit`). A draft starts with `global`
5//! alone, or its project's defaults; the browser remembers no selection.

22/// A set of runtime cards and what the user picked on them: the new
23/// session's, or the project editor's default attachments.
24#[derive(Clone, Copy)]
25pub struct CardSet {
26    pub selection: RwSignal<Selection>,
27    /// Refusals by card key and workspace field (the new session's only).
28    pub errors: RwSignal<HashMap<(String, usize), String>>,
29}
30
31impl Workspace {
32    /// The new session's cards.
33    pub fn new_cards(self) -> CardSet {
34        CardSet {
35            selection: self.selection,
36            errors: self.card_errors,
37        }
38    }

40    /// Every runtime's home, from `runtimes.list` (none before it answered,
41    /// or from a daemon older than client protocol 1.7).
42    pub fn homes(self) -> RuntimeHomes {
43        self.runtimes
44            .with(|listing| listing.as_ref().map(RuntimeHomes::from_listing))
45            .unwrap_or_default()
46    }
47
48    /// The attachments of `session_id`'s root session, as the session list
49    /// has them.
50    pub(super) fn root_attachments(self, session_id: SessionId) -> Option<Vec<RuntimeAttachment>> {
51        let root = self.tree_root.get_untracked().unwrap_or(session_id);
52        self.sessions.with_untracked(|sessions| {
53            sessions
54                .iter()
55                .find(|session| session.session_id == root)
56                .map(|session| session.attachments.clone())
57                .filter(|attachments| !attachments.is_empty())
58        })
59    }

61    /// The cards `runtimes.list` offers, in the groups' order.
62    pub fn cards(self) -> Vec<Card> {
63        self.card_groups()
64            .into_iter()
65            .flat_map(|group| group.cards)
66            .collect()
67    }
68
69    /// The cards grouped by machine, this daemon's first.
70    pub fn card_groups(self) -> Vec<Group> {
71        self.runtimes
72            .with(|listing| {
73                listing
74                    .as_ref()
75                    .map(|listing| groups(&cards(listing), daemon_machine(listing).as_ref()))
76            })
77            .unwrap_or_default()
78    }

80    /// Whether `group` shows its cards in `set`.
81    pub fn group_shown(self, group: &Group, set: CardSet) -> bool {
82        let name = group.name();
83        let stored = self
84            .group_open
85            .with(|open| open.get(&name).copied())
86            .map(|open| if open { "open" } else { "closed" }.to_owned())
87            .or_else(|| browser::load(&group_key(&name)));
88        set.selection
89            .with(|selection| group.open(stored.as_deref(), selection))
90    }
91
92    /// Opens or closes `group`, as the browser remembers.
93    pub fn toggle_group(self, group: &Group, set: CardSet) {
94        let open = !untrack(|| self.group_shown(group, set));
95        let name = group.name();
96        browser::store(&group_key(&name), if open { "open" } else { "closed" });
97        self.group_open.update(|groups| {
98            groups.insert(name, open);
99        });
100    }
101
102    /// Selects or deselects `card` in `set`.
103    pub fn toggle_card(self, card: &Card, set: CardSet) {
104        set.selection.update(|selection| {
105            selection.toggle(card);
106        });
107        let key = card.key();
108        set.errors
109            .update(|errors| errors.retain(|(failed, _), _| *failed != key));
110        if set.selection == self.selection {
111            self.draft_error.set(None);
112        }
113    }

115    /// The workspaces a field of `runtime_id` suggests, as the fields show
116    /// them: this browser's recent ones, then the sessions'.
117    pub fn workspace_suggestions(self, runtime_id: &RuntimeId) -> Vec<String> {
118        let recent =
119            recent_from_storage(browser::load(&recent_key(runtime_id.as_str())).as_deref());
120        let home = self.homes();
121        self.sessions.with(|sessions| {
122            suggestions(runtime_id.as_str(), &recent, sessions, home.get(runtime_id))
123        })
124    }

126    /// What the new session will attach besides `global`, as typed on its
127    /// cards (or its project's defaults while the draft keeps them): each
128    /// runtime and workspace, in pick order, with fields left empty out.
129    fn draft_places(self) -> Vec<(RuntimeId, Option<String>)> {
130        self.selection.with(|selection| {
131            selection
132                .project_requests()
133                .1
134                .into_iter()
135                .map(|request| {
136                    let workspace = request.workspace.filter(|workspace| !workspace.is_empty());
137                    (request.runtime_id, workspace)
138                })
139                .collect()
140        })
141    }

143    /// The runtimes the bottom bar's workspaces are on, each once, in
144    /// attach order: their colours mark the label.
145    pub fn places_runtimes(self) -> Vec<String> {
146        let mut runtimes = Vec::<String>::new();
147        let mut add = |runtime_id: &RuntimeId| {
148            if !runtimes.iter().any(|known| known == runtime_id.as_str()) {
149                runtimes.push(runtime_id.to_string());
150            }
151        };
152        if self.new_session.get() {
153            for (runtime_id, workspace) in self.draft_places() {
154                if workspace.is_some() {
155                    add(&runtime_id);
156                }
157            }
158            return runtimes;
159        }
160        let Some(session_id) = self.tree_root.get().or(self.selected.get()) else {
161            return runtimes;
162        };
163        self.sessions.with(|sessions| {
164            let attachments = sessions
165                .iter()
166                .find(|session| session.session_id == session_id)
167                .map(|session| session.attachments.as_slice())
168                .unwrap_or_default();
169            for attachment in attachments {
170                if attachment.workspace.is_some() {
171                    add(&attachment.runtime_id);
172                }
173            }
174        });
175        runtimes
176    }

178    /// The runtimes an agent picked now would run with: the draft's (as
179    /// its cards stand) for a new session, else the viewed session's. An
180    /// ACP agent runs on the one of them that offers it.
181    pub fn agent_runtimes(self) -> Vec<RuntimeId> {
182        let mut runtimes = Vec::<RuntimeId>::new();
183        let mut add = |runtime_id: &RuntimeId| {
184            if !runtimes.contains(runtime_id) {
185                runtimes.push(runtime_id.clone());
186            }
187        };
188        if self.new_session.get() {
189            for (runtime_id, _) in self.draft_places() {
190                add(&runtime_id);
191            }
192            return runtimes;
193        }
194        let Some(session_id) = self.selected.get() else {
195            return runtimes;
196        };
197        self.sessions.with(|sessions| {
198            if let Some(session) = sessions
199                .iter()
200                .find(|session| session.session_id == session_id)
201            {
202                for attachment in &session.attachments {
203                    add(&attachment.runtime_id);
204                }
205            }
206        });
207        runtimes
208    }

210    /// What the bottom bar says of where the viewed session (or the draft)
211    /// works: its workspaces under their runtimes' homes; `None` before it
212    /// has any runtimes.
213    pub fn places_label(self) -> Option<String> {
214        let homes = self.homes();
215        if self.new_session.get() {
216            let shown = self
217                .draft_places()
218                .into_iter()
219                .filter_map(|(runtime_id, workspace)| {
220                    let path = homes.display(&runtime_id, &workspace?);
221                    Some(if runtime_id.as_str() == RuntimeId::LOCAL {
222                        path
223                    } else {
224                        format!("{runtime_id}:{path}")
225                    })
226                })
227                .collect::<Vec<_>>();
228            return Some(if !shown.is_empty() {
229                shown.join(", ")
230            } else if self.selection.with(|selection| selection.global) {
231                "global only: no workspace".to_owned()
232            } else {
233                "no workspace".to_owned()
234            });
235        }
236        let session_id = self.tree_root.get().or(self.selected.get())?;
237        self.sessions.with(|sessions| {
238            sessions
239                .iter()
240                .find(|session| session.session_id == session_id)
241                .map(|session| {
242                    crate::places::places_label(session, &homes, self.lists_attachments.get())
243                })
244        })
245    }
