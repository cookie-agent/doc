574    fn update_store<T>(self, update: impl FnOnce(&mut StateStore) -> T) -> T {
575        let result = self
576            .store
577            .try_update_value(update)
578            .expect("the store lives");
579        let deferred = self
580            .deferred
581            .try_update_value(|deferred| deferred.as_mut().map(|changed| *changed = true))
582            .flatten();
583        if deferred.is_none() {
584            self.revision
585                .update(|revision| *revision = revision.wrapping_add(1));
586        }
587        result
588    }

615    /// Adopts a connection: the first one, or a redialed one after a drop.
616    /// A redial rebuilds what the old connection fed, as the TUI does: the
617    /// viewed session from a full replay, the lists, approvals, and mode.
618    pub fn adopt(self, client: Client) {
619        let reconnected = self.connection.get_value() > 0;
620        self.connection.update_value(|connection| *connection += 1);
621        self.can_delete.set(
622            cookie_agent_client_state::session_delete::daemon_can_delete_sessions(
623                client.daemon_version(),
624            ),
625        );
626        self.lists_attachments.set(
627            cookie_agent_client_state::paths::daemon_reports_runtime_homes(client.daemon_version()),
628        );
629        self.has_projects
630            .set(cookie_agent_client_state::projects::daemon_has_projects(
631                client.daemon_version(),
632            ));
633        self.can_sign_in.set(
634            cookie_agent_client_state::mcp::daemon_accepts_page_redirects(client.daemon_version()),
635        );
636        self.shares_acp_agents.set(
637            cookie_agent_client_state::runtimes::daemon_shares_acp_agents(client.daemon_version()),
638        );
639        self.has_crons
640            .set(cookie_agent_client_state::cron::daemon_has_crons(
641                client.daemon_version(),
642            ));
643        self.can_restart
644            .set(cookie_agent_client_state::restart::daemon_can_restart(
645                client.daemon_version(),
646            ));
647        self.has_push
648            .set(crate::push::daemon_has_push(client.daemon_version()));
649        self.keeps_unread
650            .set(cookie_agent_client_state::unread::daemon_keeps_unread(
651                client.daemon_version(),
652            ));
653        // A new connection has heard nothing of what the page shows.
654        self.presence_sent.set_value(None);
655        self.restarting.set(None);
656        self.restart_confirm.set(false);
657        self.client.set_value(Some(client));
658        self.refresh_config_status();
659        self.subscriptions
660            .update_value(|subscriptions| subscriptions.reset());
661        self.finish_waiting_sign_in();
662        let selected = self.selected.get_untracked();
663        self.refresh_onboarding(!reconnected);
664        // A phone opens on its sessions, over a new session: what it is
665        // opened for is mostly a session already going. A notification's
666        // session, opened next, closes them.
667        if !reconnected
668            && self.narrow.get_untracked()
669            && self.new_session.get_untracked()
670            && self.view.get_untracked() == crate::layout::View::Chat
671        {
672            self.sidebar_open.set(true);
673        }
674        self.report_presence();
675        if reconnected {
676            self.reread_open();
677            if self.status.with_untracked(|status| status == RECONNECTING) {
678                self.status.set(RECONNECTED.into());
679            }
680        }
681        if reconnected {
682            // The daemon may be a new process, which owns none of the
683            // sessions the last one did: a session it does not own answers
684            // `events.subscribe` with a snapshot that no tail keeps current.
685            // Each is classified (taken) again, as opening it does.
686            self.owned.set_value(HashSet::new());
687        }
688        spawn_local(async move {
689            self.refresh_lists().await;
690            if reconnected && let Some(session_id) = selected {
691                let subscribe = self
692                    .subscriptions
693                    .try_update_value(|subscriptions| {
694                        subscriptions.subscribe_from_scratch(session_id, false)
695                    })
696                    .expect("the subscriptions live");
697                self.send_subscribe(subscribe);
698                // Owned again, the snapshot subscription becomes live.
699                self.classify_again(session_id);
700                self.refresh_approvals(session_id);
701                self.load_permission_mode(session_id);
702                self.refresh_cost(session_id);
703                self.status.set(RECONNECTED.into());
704            }
705        });
706    }

716    /// Reads the session list and the runtime snapshot again.
717    pub async fn refresh_lists(self) {
718        let Some(client) = self.client() else {
719            return;
720        };
721        let read = self.session_list.try_update_value(|list| list.begin());
722        let answer = client.list_sessions(SessionListParams::default()).await;
723        if let Some(read) = read {
724            match answer {
725                Ok(result) => {
726                    if let Some(sessions) = self
727                        .session_list
728                        .try_update_value(|list| list.land(read, result.sessions))
729                        .flatten()
730                    {
731                        self.sessions.set(sessions);
732                    }
733                }
734                Err(error) => {
735                    self.session_list.update_value(|list| list.fail(read));
736                    self.status.set(crate::errors::describe(&error));
737                }
738            }
739        }
740        self.refresh_projects();
741        self.poll_runtimes();
742        self.poll_crons();

744            Ok(result) => {

748                if self.draft.get_untracked().is_none() {
749                    self.draft.set(self.default_draft());
750                }
751            }
752            Err(error) => self.status.set(format!(
753                "runtime snapshot unavailable: {}",
754                crate::errors::describe(&error)
755            )),
756        }

759    /// Applies one delivery of the connection's stream.
760    pub fn handle(self, delivery: ClientDelivery) {
761        match &delivery {
762            ClientDelivery::SessionsChanged(change) => {
763                self.apply_sessions_change((**change).clone());
764                return;
765            }
766            ClientDelivery::RuntimeChanged(changed) => {
767                let applied = self
768                    .runtime
769                    .try_update_value(|runtime| {
770                        let duplicate = runtime.revision() == Some(&changed.snapshot.revision);
771                        runtime.apply_notification((**changed).clone()) || duplicate
772                    })
773                    .unwrap_or(true);
774                self.runtime_revision.update(|revision| *revision += 1);
775                if !applied {
776                    // Outside the predecessor chain: fetch the current one.
777                    self.resync_runtime();
778                }
779                self.config_may_have_changed(&changed.reasons);
780                return;
781            }
782            ClientDelivery::RecoveryFailed { session_id, error } => {
783                self.subscriptions
784                    .update_value(|subscriptions| subscriptions.recovery_failed(*session_id));
785                self.schedule_subscription_retry(*session_id);
786                self.status.set(format!("recovery failed: {error}"));
787                return;
788            }
789            ClientDelivery::ProjectsChanged(_) => {
790                self.refresh_projects();
791                return;
792            }
793            ClientDelivery::CronsChanged(_) => {
794                self.poll_crons();
795                return;
796            }
797            ClientDelivery::NotificationsChanged(_) => {
798                if self.notifications.with_untracked(Option::is_some) {
799                    self.poll_notifications();
800                }
801                return;
802            }
803            ClientDelivery::DaemonRestarting(restarting) => {
804                self.note_restarting(restarting);
805                return;
806            }
807            ClientDelivery::OnboardingChanged(status) => {
808                self.apply_onboarding_change(status.clone());
809                return;
810            }
811            ClientDelivery::PluginEvent(event) => {
812                if self.selected.get_untracked() == Some(event.session_id) {
813                    self.status
814                        .set(cookie_agent_client_state::plugins::plugin_event_status(
815                            event,
816                        ));
817                }
818                return;
819            }
820            ClientDelivery::MemoryChanged(_) if self.has_projects.get_untracked() => {
821                // A project's memory count follows its memories.
822                self.refresh_projects();
823                self.refresh_open_panel(&delivery);
824                return;
825            }
826            ClientDelivery::McpChanged(_)
827            | ClientDelivery::RuntimesChanged(_)
828            | ClientDelivery::PluginsChanged(_)
829            | ClientDelivery::MachinesChanged(_)
830            | ClientDelivery::MemoryChanged(_) => {
831                // The cards and every path's home follow the runtimes.
832                if matches!(
833                    delivery,
834                    ClientDelivery::RuntimesChanged(_) | ClientDelivery::MachinesChanged(_)
835                ) && self.panel.get_untracked() != Some(Panel::Runtimes)
836                {
837                    self.poll_runtimes();
838                }
839                // An open panel refreshes now instead of at its next poll.
840                self.refresh_open_panel(&delivery);
841                return;
842            }
843            ClientDelivery::Disconnected { .. } | ClientDelivery::Diagnostic { .. } => return,
844            _ => {}
845        }
846        let own_origin = self.client().and_then(|client| client.origin());
847        let selected = self.selected.get_untracked();
848        let foreign_pick = is_foreign_model_pick(&delivery, selected, own_origin.as_ref());
849        // What the rest needs to know of the event, read before the store
850        // takes it: the store reduces the event it is handed without a copy.
851        let event = match &delivery {
852            ClientDelivery::Live { message, .. } => match message.as_ref() {
853                cookie_agent_protocol::EventSubscriptionMessage::Event { event } => {
854                    Some(event.as_ref())
855                }
856                _ => None,
857            },
858            ClientDelivery::ReplayEvent { event, .. } => Some(event.as_ref()),
859            _ => None,
860        };
861        let live = matches!(&delivery, ClientDelivery::Live { .. });
862        let run_ended = event.and_then(|event| {
863            (live && matches!(&event.payload, EventPayload::RunFinished { .. }))
864                .then_some(event.session_id)
865        });
866        let cost_changed = event.and_then(|event| {
867            matches!(
868                event.payload,
869                EventPayload::ModelUsageRecorded { .. }
870                    | EventPayload::InternalAgentUsageRecorded { .. }
871            )
872            .then_some(event.session_id)
873        });
874        let replay_ended = match &delivery {
875            ClientDelivery::ReplayEnd { session_id, .. } => Some(*session_id),
876            _ => None,
877        };
878        // A delegation's lifecycle is how a child's status reaches the
879        // tree; a link, a run end, or a finished replay can change it too.
880        let tree_changed = replay_ended.is_some_and(|session_id| self.in_viewed_tree(session_id))
881            || event.is_some_and(|event| {
882                matches!(
883                    &event.payload,
884                    EventPayload::ToolCallLinked { .. }
885                        | EventPayload::RunFinished { .. }
886                        | EventPayload::DelegationProgressed { .. }
887                        | EventPayload::DelegationFinished { .. }
888                        | EventPayload::DelegateChildTerminated { .. }
889                ) && self.in_viewed_tree(event.session_id)
890            });
891        let outcome = self.update_store(|store| store.apply_delivery(delivery));
892        if foreign_pick {
893            self.draft_publish
894                .update_value(DraftPublish::yield_to_foreign_pick);
895            self.sync_draft();
896            self.status
897                .set("the model was picked by another client".into());
898        }
899        if let Some(session_id) = run_ended
900            && !self.is_recent(session_id)
901            && self.is_settled(session_id)
902            && self
903                .subscriptions
904                .try_update_value(|subscriptions| subscriptions.release(session_id))
905                .unwrap_or(false)
906        {
907            self.release(session_id);
908        }
909        if let Some(session_id) = replay_ended {
910            let owned = self.owned(session_id);
911            if let Some(subscribe) = self
912                .subscriptions
913                .try_update_value(|subscriptions| subscriptions.note_replay_end(session_id, owned))
914                .flatten()
915            {
916                self.send_subscribe(subscribe);
917            }
918        }
919        if tree_changed {
920            self.refresh_tree();
921        }
922        // The background sheet closes with its last call, as in the TUI.
923        if self.panel.get_untracked() == Some(Panel::Background)
924            && untrack(|| self.background_calls()).is_empty()
925        {
926            self.panel.set(None);
927        }
928        match outcome {
929            DeliveryOutcome::Applied => {
930                self.sync_draft();
931                self.restore_voided_inputs();
932                if let Some(session_id) = cost_changed {
933                    self.refresh_cost(session_id);
934                }
935            }
936            DeliveryOutcome::Gap { cursor, .. } => {
937                self.status
938                    .set(format!("event gap after sequence {cursor}; replaying"));
939            }
940            DeliveryOutcome::ReplayFailed { session_id } => {
941                self.status
942                    .set("incomplete replay; retrying recovery".into());
943                if let Some(client) = self.client() {
944                    client.recover_session(session_id, true);
945                }
946                let owned = self.owned(session_id);
947                self.subscriptions
948                    .update_value(|subscriptions| subscriptions.recover(session_id, owned));
949            }
950        }
951    }

953    fn apply_sessions_change(self, change: SessionsChangedNotification) {
954        // A `session.list` answer on its way may predate this change.
955        self.session_list.update_value(|list| list.note(&change));
956        // A child that joined the shown tree with no event of its parent's,
957        // as an ACP agent's own subagent does, shows once the tree is read
958        // again.
959        if let SessionsChangedNotification::Upserted { session } = &change
960            && self.tree.with_untracked(|tree| {
961                cookie_agent_client_state::tree::joins(
962                    self.tree_root.get_untracked(),
963                    tree.as_ref(),
964                    session,
965                )
966            })
967        {
968            self.refresh_tree();
969        }
970        match change {
971            SessionsChangedNotification::Upserted { .. } => {
972                // A project's session count follows its sessions, which
973                // `projects.changed` does not announce.
974                let mut moved = false;
975                self.sessions.update(|sessions| {
976                    moved = crate::session_list::apply_change(sessions, &change);
977                });
978                if moved {
979                    self.refresh_projects();
980                }
981            }
982            SessionsChangedNotification::Removed { session_id } => {
983                self.forget_removed(session_id);
984                self.refresh_projects();
985            }
986            SessionsChangedNotification::ApprovalsChanged {
987                root_session_id, ..
988            } => {
989                if self.selected.get_untracked() == Some(root_session_id) {
990                    self.refresh_approvals(root_session_id);
991                }
992            }
993            SessionsChangedNotification::PermissionModeChanged {
994                root_session_id,
995                mode,
996                ..
997            } => {
998                // Newer than a `session.permission.get` on its way.
999                self.supersede_reads(|reads| {
1000                    reads.permission_modes.entry(root_session_id).or_default()
1001                });
1002                self.permission_modes.update(|modes| {
1003                    modes.insert(root_session_id, mode);
1004                });
1005            }
1006            SessionsChangedNotification::Resync => {
1007                let selected = self.selected.get_untracked();
1008                spawn_local(async move {
1009                    self.refresh_lists().await;
1010                    if let Some(session_id) = selected {
1011                        self.refresh_approvals(session_id);
1012                        self.load_permission_mode(session_id);
1013                    }
1014                });
1015            }
1016        }
1017    }

1021    fn forget_removed(self, session_id: SessionId) {

1030        self.unpick(session_id);

1037            error
1038                .as_ref()
1039                .is_some_and(|(failed, _)| *failed == session_id)
1040        }) {
1041            self.delete_error.set(None);
1042        }

1045    /// Deletes the root session `session_id` and its subagents, as the
1046    /// list showed it when the user confirmed. The list loses it when the
1047    /// daemon says `sessions.changed` `removed`; a refusal shows the
1048    /// daemon's message.
1049    pub fn delete_session(self, session_id: SessionId) {
1050        use cookie_agent_client_state::session_delete::{
1051            DAEMON_TOO_OLD_TO_DELETE, SessionDeleteOutcome, delete_session,
1052        };
1053        let Some(session) = self.sessions.with_untracked(|sessions| {
1054            sessions
1055                .iter()
1056                .find(|session| session.session_id == session_id)
1057                .cloned()
1058        }) else {
1059            return;
1060        };
1061        let Some(client) = self.client() else {
1062            return;
1063        };
1064        self.delete_error.set(None);
1065        spawn_local(async move {
1066            let message = match delete_session(&client, &session).await {
1067                SessionDeleteOutcome::Deleted(deleted) => {
1068                    let subagents = deleted.len().saturating_sub(1);
1069                    self.status.set(match subagents {
1070                        0 => "deleted the session".to_owned(),
1071                        1 => "deleted the session and its subagent".to_owned(),
1072                        count => format!("deleted the session and its {count} subagents"),
1073                    });
1074                    return;
1075                }
1076                SessionDeleteOutcome::DaemonTooOld => DAEMON_TOO_OLD_TO_DELETE.to_owned(),
1077                SessionDeleteOutcome::Failed(error) => crate::errors::describe(&error),
1078            };
1079            self.delete_error.set(Some((session_id, message)));
1080        });
1081    }

1085            return;

1094                    runtime.install_current(result.snapshot);
1095                });
1096                self.runtime_revision.update(|revision| *revision += 1);
1097            }
1098        });

1152    /// The draft a fresh session starts from: the `primary` agent (else the
1153    /// first), its first live model (else the first model), as the TUI picks.
1154    fn default_draft(self) -> Option<RunSelection> {

1156        let agent = agents
1157            .iter()
1158            .find(|agent| agent.id.as_str() == "primary")
1159            .or_else(|| agents.first())?;
1160        let models = self.models();
1161        let live = |selection: &ModelSelection| {

1427    fn apply_ownership_classification(
1428        self,
1429        session_id: SessionId,
1430        generation: u64,
1431        outcome: SessionOwnershipOutcome,
1432    ) {
1433        let current = self
1434            .ownership_classifications
1435            .with_value(|classifications| classifications.get(&session_id) == Some(&generation));
1436        if !current {
1437            return;
1438        }
1439        self.ownership_classifications
1440            .update_value(|classifications| {
1441                classifications.remove(&session_id);
1442            });
1443        let owned = match outcome {
1444            SessionOwnershipOutcome::Owned(session) => {
1445                self.sessions.update(|sessions| {
1446                    if let Some(existing) = sessions
1447                        .iter_mut()
1448                        .find(|existing| existing.session_id == session_id)
1449                    {
1450                        *existing = *session;
1451                    }
1452                });
1453                true
1454            }
1455            SessionOwnershipOutcome::Foreign => {
1456                if self.selected.get_untracked() == Some(session_id) {
1457                    self.status.set(
1458                        "This session is owned by another cookie process; read-only snapshot."
1459                            .into(),
1460                    );
1461                }
1462                false
1463            }
1464            SessionOwnershipOutcome::Failed(error) => {
1465                if self.selected.get_untracked() == Some(session_id) {
1466                    self.status.set(error);
1467                }
1468                false
1469            }
1470        };
1471        self.owned.update_value(|owned_sessions| {
1472            if owned {
1473                owned_sessions.insert(session_id);
1474            } else {
1475                owned_sessions.remove(&session_id);
1476            }
1477        });
1478        self.read_only.update(|read_only| {
1479            if owned {
1480                read_only.remove(&session_id);
1481            } else {
1482                read_only.insert(session_id);
1483            }
1484        });
1485        // Owned now: subscribe, or upgrade a snapshot subscription made
1486        // before ownership to a live one.
1487        if let Some(subscribe) = self
1488            .subscriptions
1489            .try_update_value(|subscriptions| {
1490                subscriptions.subscribe(session_id, owned, SubscribeIntent::Follow)
1491            })
1492            .flatten()
1493        {
1494            self.send_subscribe(subscribe);
1495        }
1496    }

1498    fn send_subscribe(self, subscribe: Subscribe) {
1499        let Some(client) = self.client() else {
1500            return;
1501        };
1502        let Subscribe {
1503            session_id,
1504            attempt,
1505            from_scratch,
1506        } = subscribe;
1507        let cursor = if from_scratch {
1508            None
1509        } else {
1510            self.store
1511                .with_value(|store| store.sessions.get(&session_id).map(|state| state.last_seq))
1512        };
1513        let connection = self.connection.get_value();
1514        spawn_local(async move {
1515            let outcome = match client.subscribe_events(session_id, cursor).await {
1516                Ok(()) => {
1517                    // Background calls are not in the event log: read them
1518                    // once subscribed; `background.changed` follows, and one
1519                    // that lands first is newer than this answer.
1520                    let mark = self
1521                        .store
1522                        .with_value(|store| store.background_mark(session_id));
1523                    if let Ok(listed) = client.list_background(session_id).await {
1524                        self.update_store(|store| {
1525                            store.install_background_listing(session_id, mark, listed.calls)
1526                        });
1527                    }
1528                    SubscriptionOutcome::Established
1529                }
1530                Err(ClientError::ReplayInProgress) => SubscriptionOutcome::ReplayInProgress,
1531                Err(error) => SubscriptionOutcome::Failed(crate::errors::describe(&error)),
1532            };
1533            if !self.is_current(connection) {
1534                return;
1535            }
1536            if let SubscriptionOutcome::Failed(error) = &outcome
1537                && self.selected.get_untracked() == Some(session_id)
1538            {
1539                self.status.set(error.clone());
1540            }
1541            let owned = self.owned(session_id);
1542            if let Some(next) = self
1543                .subscriptions
1544                .try_update_value(|subscriptions| {
1545                    subscriptions.finish(session_id, attempt, &outcome, owned)
1546                })
1547                .flatten()
1548            {
1549                self.send_subscribe(next);
1550            }
1551        });
1552    }

1634            return;

1645                    .get(&session_id)

1657            files: files.clone(),

1667    /// Sends `content` to the viewed session, or creates the new session
1668    /// with it. On failure `text` and `files` go back to an empty composer.
1669    pub fn send_content(
1670        self,
1671        content: cookie_agent_protocol::UserContent,
1672        delivery: InputDelivery,
1673        text: String,
1674        files: Vec<cookie_agent_protocol::ArtifactRef>,
1675    ) {
1676        let Some(client) = self.client() else {
1677            return;
1678        };
1679        let new_session = self.new_session.get_untracked();
1680        if new_session && !self.may_create() {
1681            self.return_to_draft(ComposerSlot::NewSession, text, files);
1682            return;
1683        }
1684        self.sent.notify();
1685        let selection = self.draft.get_untracked();
1686        spawn_local(async move {
1687            let mut content = content;
1688            let mut files = files;
1689            let session_id = if new_session {
1690                self.creating.set(true);
1691                self.draft_error.set(None);
1692                self.card_errors.set(HashMap::new());
1693                let created = self.create_session(&client, selection.clone()).await;
1694                self.creating.set(false);
1695                let session_id = match created {
1696                    Ok(session_id) => session_id,
1697                    Err(refusal) => {
1698                        self.refuse_draft(refusal, text, files);
1699                        return;
1700                    }
1701                };
1702                // The draft's files go into the session now, and the
1703                // message names them by their own references.
1704                if self.holds_draft_files(&files) {
1705                    match self.upload_draft_files(&client, session_id, files).await {
1706                        Ok(uploaded) => {
1707                            files = uploaded;
1708                            let rebuilt = ComposerInput {
1709                                text: text.clone(),
1710                                files: files.clone(),
1711                            }
1712                            .content();
1713                            if let Some(rebuilt) = rebuilt {
1714                                content = rebuilt;
1715                            }
1716                        }
1717                        Err(uploaded) => {
1718                            // As a failed submit: the message goes back to
1719                            // the session's draft.
1720                            self.return_to_draft(ComposerSlot::Session(session_id), text, uploaded);
1721                            return;
1722                        }
1723                    }
1724                }
1725                session_id
1726            } else {
1727                match self.selected.get_untracked() {
1728                    Some(session_id) => session_id,
1729                    None => return,
1730                }
1731            };
1732            let result = client
1733                .submit_input(SessionSubmitInputParams {
1734                    session_id,
1735                    content,
1736                    delivery,
1737                    selection,
1738                    client_run_id: client_run_id(),
1739                    reset_fallback: false,
1740                })
1741                .await;
1742            match result {
1743                Ok(SessionSubmitInputResult::Handled { reason }) => self.status.set(reason),
1744                Ok(SessionSubmitInputResult::Queued { .. }) => {
1745                    self.status.set("queued behind the running run".into());
1746                }
1747                Ok(_) => {}
1748                Err(error) => {
1749                    // The text goes back to the session's draft, as in the
1750                    // TUI.
1751                    self.return_to_draft(ComposerSlot::Session(session_id), text, files);
1752                    self.status.set(crate::errors::describe(&error));
1753                }
1754            }
1755        });
1756    }

1780    /// The daemon did not create the new session: the draft stays as it
1781    /// was, the message goes back into the composer, and the error shows on
1782    /// the card of the runtime it names, else beside the composer.
1783    fn refuse_draft(
1784        self,
1785        refusal: DraftRefusal,
1786        text: String,
1787        files: Vec<cookie_agent_protocol::ArtifactRef>,
1788    ) {
1789        self.return_to_draft(ComposerSlot::NewSession, text, files);
1790        // The page left the draft meanwhile: the message waits in it, and
1791        // the status line says why.
1792        if !self.new_session.get_untracked() {
1793            self.status.set(format!(
1794                "the new session was not created: {}",
1795                refusal.message
1796            ));
1797            return;
1798        }
1799        let card = crate::cards::failing_request(&refusal.message, &refusal.requests);
1800        match card {
1801            Some((key, index)) => {
1802                let label = untrack(|| self.cards())
1803                    .into_iter()
1804                    .find(|card| card.key() == key)
1805                    .map_or_else(|| key.clone(), |card| card.label());
1806                let notice =
1807                    format!("No session was created: {label} did not start; see its card.");
1808                self.status.set(notice.clone());
1809                self.draft_error.set(Some(notice));

1841        client: &Client,
1842        selection: Option<RunSelection>,

1858            .get_untracked()

1873                project,

1881                line: crate::errors::describe(&error),

1891                    serde_json::to_string(&crate::cards::remember(&recent, workspace))

1894                }
1895            }
1896        }

1898        if created.session.project.is_some() {
1899            self.refresh_projects();
1900        }
1901        self.owned.update_value(|owned| {
1902            owned.insert(session_id);
1903        });
1904        self.sessions.update(|sessions| {

1906                .iter()
1907                .any(|session| session.session_id == session_id)

1910            }
1911        });
1912        if let Some(mode) = self.new_session_mode.get_untracked() {
1913            self.permission_modes.update(|modes| {
1914                modes.insert(session_id, mode);
1915            });
1916        }
1917        self.new_session_mode.set(None);
1918        self.draft_rules.set(Vec::new());
1919        self.draft_agent_options.set(Default::default());

1921        self.card_errors.set(HashMap::new());

1931            })

1935        }

2443            store
2444                .sessions
2445                .get_mut(&session_id)
2446                .and_then(|state| state.toggle_full_content(call_id, block))
2447        });
2448        let (Some(read), Some(client)) = (read, self.client()) else {
2449            return;
2450        };
2451        spawn_local(async move {

2453                client
2454                    .read_artifact(ArtifactReadParams {
2455                        session_id,
2456                        sha256: read.artifact.sha256.clone(),
2457                        offset: 0,
2458                        length: read.length,
2459                    })
2460                    .await
2461                    .map_err(|error| crate::errors::describe_full(&error)),
2462            );
2463            self.update_store(|store| {
2464                if let Some(session) = store.sessions.get_mut(&session_id) {
2465                    session.finish_full_content(call_id, block, state);
2466                }
2467            });
2468        });
2469    }

2506    /// A media block's bytes, read in chunks, with its media type.
2507    pub async fn read_media(self, sha256: String) -> Result<(Vec<u8>, String), String> {
2508        let (Some(client), Some(session_id)) = (self.client(), self.selected.get_untracked())
2509        else {
2510            return Err("not connected".into());
2511        };
2512        let mut bytes = Vec::new();
2513        loop {
2514            let read = client
2515                .read_artifact(ArtifactReadParams {
2516                    session_id,
2517                    sha256: sha256.clone(),
2518                    offset: bytes.len() as u64,
2519                    length: cookie_agent_protocol::ARTIFACT_CHUNK_BYTES,
2520                })
2521                .await
2522                .map_err(|error| crate::errors::describe_full(&error))?;
2523            use base64::Engine as _;
2524            let chunk = base64::engine::general_purpose::STANDARD
2525                .decode(read.data)
2526                .map_err(|error| error.to_string())?;
2527            let done = chunk.is_empty() || bytes.len() + chunk.len() >= read.total_bytes as usize;
2528            bytes.extend(chunk);
2529            if done {
2530                return Ok((bytes, read.media_type));
2531            }
2532        }
2533    }
