1//! Code highlighting with highlight.js, shipped in the bundle
2//! (`highlight.min.js`, no CDN) and loaded the first time a code block that
3//! names its language is on the page.
4
5use std::cell::Cell;
6
7use wasm_bindgen::{JsCast as _, JsValue, closure::Closure};
8
9thread_local! {
10    static LOADING: Cell<bool> = const { Cell::new(false) };
11}
12
13const PENDING: &str = "pre code[class*=\"language-\"]:not([data-highlighted])";
14
15fn highlighter() -> Option<js_sys::Function> {
16    let window = web_sys::window()?;
17    let hljs = js_sys::Reflect::get(&window, &JsValue::from_str("hljs")).ok()?;
18    if hljs.is_undefined() {
19        return None;
20    }
21    js_sys::Reflect::get(&hljs, &JsValue::from_str("highlightElement"))
22        .ok()?
23        .dyn_into()
24        .ok()
25}
26
27/// Highlights the code blocks under `root` that name a language and are not
28/// highlighted yet, loading the highlighter first when it is not.
29pub fn highlight_within(root: &web_sys::Element) {
30    let Ok(pending) = root.query_selector_all(PENDING) else {
31        return;
32    };
33    if pending.length() == 0 {
34        return;
35    }
36    let Some(highlight) = highlighter() else {
37        load();
38        return;
39    };
40    for index in 0..pending.length() {
41        if let Some(element) = pending.item(index) {
42            let _ = highlight.call1(&JsValue::NULL, &element);
43        }
44    }
45}
46
47/// Adds the highlighter's script once; when it has loaded, the whole page's
48/// pending blocks are highlighted.
49fn load() {
50    if LOADING.with(|loading| loading.replace(true)) {
51        return;
52    }
53    let Some(document) = web_sys::window().and_then(|window| window.document()) else {
54        return;
55    };
56    let Ok(script) = document.create_element("script") else {
57        return;
58    };
59    let _ = script.set_attribute("src", "highlight.min.js");
60    let loaded = Closure::<dyn Fn()>::new(move || {
61        if let Some(body) = web_sys::window()
62            .and_then(|window| window.document())
63            .and_then(|document| document.body())
64        {
65            highlight_within(&body);
66        }
67    });
68    let _ = script.add_event_listener_with_callback("load", loaded.as_ref().unchecked_ref());
69    loaded.forget();
70    if let Some(head) = document.head() {
71        let _ = head.append_child(&script);
72    }
73}
