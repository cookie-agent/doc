1//! Code highlighting with highlight.js, shipped in the bundle
2//! (`highlight.min.js`, no CDN) and loaded the first time a code block that
3//! names its language is on the page.
4
5use std::cell::Cell;
6
7use wasm_bindgen::{JsCast as _, JsValue, closure::Closure};
8
9thread_local! {
10    static LOADING: Cell<bool> = const { Cell::new(false) };
11}
12
13const PENDING: &str = "pre code[class*=\"language-\"]:not([data-highlighted])";
14
15fn highlighter() -> Option<js_sys::Function> {
16    let window = web_sys::window()?;
17    let hljs = js_sys::Reflect::get(&window, &JsValue::from_str("hljs")).ok()?;
18    if hljs.is_undefined() {
19        return None;
20    }
21    js_sys::Reflect::get(&hljs, &JsValue::from_str("highlightElement"))
22        .ok()?
23        .dyn_into()
24        .ok()
25}
26
27/// Highlights the code blocks under `root` that name a language and are not
28/// highlighted yet, loading the highlighter first when it is not.
29pub fn highlight_within(root: &web_sys::Element) {
30    let Ok(pending) = root.query_selector_all(PENDING) else {
31        return;
32    };
33    let is_pending = root.matches(PENDING).unwrap_or(false);
34    if pending.length() == 0 && !is_pending {
35        return;
36    }
37    let Some(highlight) = highlighter() else {
38        load();
39        return;
40    };
41    if is_pending {
42        let _ = highlight.call1(&JsValue::NULL, root);
43    }
44    for index in 0..pending.length() {
45        if let Some(element) = pending.item(index) {
46            let _ = highlight.call1(&JsValue::NULL, &element);
47        }
48    }
49}
50
51/// What waited for the highlighter to load: blocks of the transcript as they
52/// near the view, as their rows would have been, and any others now.
53fn highlight_pending() {
54    let Some(pending) = web_sys::window()
55        .and_then(|window| window.document())
56        .and_then(|document| document.query_selector_all(PENDING).ok())
57    else {
58        return;
59    };
60    let Some(highlight) = highlighter() else {
61        return;
62    };
63    for index in 0..pending.length() {
64        let Some(block) = pending
65            .item(index)
66            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
67        else {
68            continue;
69        };
70        if !crate::near_view::watch(&block) {
71            let _ = highlight.call1(&JsValue::NULL, &block);
72        }
73    }
74}
75
76/// Adds the highlighter's script once; when it has loaded, the page's pending
77/// blocks are highlighted.
78fn load() {
79    if LOADING.with(|loading| loading.replace(true)) {
80        return;
81    }
82    let Some(document) = web_sys::window().and_then(|window| window.document()) else {
83        return;
84    };
85    let Ok(script) = document.create_element("script") else {
86        return;
87    };
88    let _ = script.set_attribute("src", "highlight.min.js");
89    let loaded = Closure::<dyn Fn()>::new(highlight_pending);
90    let _ = script.add_event_listener_with_callback("load", loaded.as_ref().unchecked_ref());
91    // Fires once, when the script loads, and the page never unloads it.
92    loaded.forget();
93    if let Some(head) = document.head() {
94        let _ = head.append_child(&script);
95    }
96}
