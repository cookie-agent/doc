34/// What the page shows besides the projection itself.
35pub struct TranscriptView<'a> {
36    /// The session the projection is of: item IDs repeat across sessions.
37    pub session_id: cookie_agent_protocol::SessionId,
38    pub state: &'a SessionState,
39    pub names: &'a ModelDisplayNames,
40    /// Diagnostic rows below this level are hidden (`/events`).
41    pub minimum_level: EventLevel,
42    /// What the user opened.
43    pub expanded: &'a Expanded,
44    /// The subagents this session's delegate calls started, by call.
45    pub delegates: &'a HashMap<ToolCallId, Delegate>,
46    /// Every runtime's home: a tool row shows its runtime's paths as
47    /// `~/…`.
48    pub homes: &'a cookie_agent_client_state::paths::RuntimeHomes,
49}

56/// What the user opened in one session's transcript: tool calls, the full
57/// output of calls, thinking blocks, and other rows' blocks, each under its
58/// own kind of key, so a pass looks a call up by its ID.
59#[derive(Clone, Debug, Default, Eq, PartialEq)]
60pub struct Expanded {
61    tools: FastSet<ToolCallId>,
62    all_output: FastSet<ToolCallId>,
63    thinking: FastSet<u64>,
64    blocks: FastSet<String>,
65}
66
67impl Expanded {
68    /// Opens or closes what `key`, a row's `data-call` or
69    /// `data-block-key`, names: a call, a call's full output (its ID and
70    /// `:all`), a thinking block (`thinking:` and its segment), or another
71    /// row's block.
72    pub fn toggle(&mut self, key: &str) {
73        fn flip<T: Eq + Hash>(set: &mut FastSet<T>, key: T) {
74            if !set.remove(&key) {
75                set.insert(key);
76            }
77        }
78        if let Some(id) = key.strip_prefix("thinking:").and_then(|id| id.parse().ok()) {
79            flip(&mut self.thinking, id);
80        } else if let Some(call) = key.strip_suffix(":all").and_then(|call| call.parse().ok()) {
81            flip(&mut self.all_output, call);
82        } else if let Ok(call) = key.parse() {
83            flip(&mut self.tools, call);
84        } else {
85            flip(&mut self.blocks, key.to_owned());
86        }
87    }
88
89    /// Whether the call is open.
90    pub fn tool(&self, call: &ToolCallId) -> bool {
91        self.tools.contains(call)
92    }
93
94    /// Whether the call shows its output past the row budget.
95    pub fn all_output(&self, call: &ToolCallId) -> bool {
96        self.all_output.contains(call)
97    }
98
99    /// Whether the thinking segment that starts at `id` is open.
100    pub fn thinking(&self, id: u64) -> bool {
101        self.thinking.contains(&id)
102    }
103
104    /// Whether the row whose block key is `key` is open.
105    pub fn block(&self, key: &str) -> bool {
106        self.blocks.contains(key)
107    }
108}

110/// The keys opened, each as [`Expanded::toggle`] takes it.
111impl<S: AsRef<str>> FromIterator<S> for Expanded {
112    fn from_iter<I: IntoIterator<Item = S>>(keys: I) -> Self {
113        let mut expanded = Self::default();
114        for key in keys {
115            expanded.toggle(key.as_ref());
116        }
117        expanded
118    }
119}

121/// What every key of one pass over the transcript depends on, worked out
122/// once for the pass.
123struct Pass<'a, 'v> {
124    view: &'a TranscriptView<'v>,
125    homes: u64,
126    /// The calls waiting for the user's approval.
127    waiting: FastSet<ToolCallId>,
128}
129
130impl<'a, 'v> Pass<'a, 'v> {
131    fn new(view: &'a TranscriptView<'v>) -> Self {
132        Self {
133            view,
134            homes: homes_key(view.homes),
135            waiting: view
136                .state
137                .approvals
138                .iter()
139                .filter(|approval| approval.is_visible_user_escalation())
140                .map(|approval| approval.tool_call_id)
141                .collect(),
142        }
143    }
144
145    /// Hashes what a call's row shows besides the call's own revision.
146    fn hash_tool(&self, call_id: &ToolCallId, hasher: &mut impl std::hash::Hasher) {
147        let view = self.view;
148        call_id.hash(hasher);
149        view.expanded.tool(call_id).hash(hasher);
150        view.expanded.all_output(call_id).hash(hasher);
151        // Raw arguments show only at `info` and below.
152        (view.minimum_level <= EventLevel::Info).hash(hasher);
153        view.state
154            .tools
155            .get(call_id)
156            .map(|tool| tool.revision)
157            .hash(hasher);
158        self.waiting.contains(call_id).hash(hasher);
159        view.delegates.get(call_id).hash(hasher);
160    }
161}

166        self.state.approvals.iter().any(|approval| {
167            approval.tool_call_id == tool.id && approval.is_visible_user_escalation()
168        })

184/// The maps and sets a pass fills, under a fast fixed hash: their keys are
185/// hashes and IDs already, and a pass looks up every part of every row.
186pub type FastMap<K, V> = HashMap<K, V, foldhash::fast::FixedState>;
187type FastSet<T> = HashSet<T, foldhash::fast::FixedState>;
188
189fn fast_map<K, V>(capacity: usize) -> FastMap<K, V> {
190    HashMap::with_capacity_and_hasher(capacity, foldhash::fast::FixedState::default())
191}

193/// A key that changes whenever the item's HTML would: its version, the
194/// revisions of the tool calls it shows, and their expansion.
195pub fn item_key(view: &TranscriptView<'_>, item: &TranscriptItem) -> u64 {
196    item_key_in(&Pass::new(view), item)
197}
198
199fn item_key_in(pass: &Pass<'_, '_>, item: &TranscriptItem) -> u64 {
200    let view = pass.view;
201    let mut hasher = foldhash::fast::FixedState::default().build_hasher();
202    view.session_id.hash(&mut hasher);
203    pass.homes.hash(&mut hasher);
204    item.id().hash(&mut hasher);
205    item.version().hash(&mut hasher);
206    std::mem::discriminant(item).hash(&mut hasher);
207    if let TranscriptItem::Assistant { id, children, .. } = item {
208        // The footer's inputs: metrics can land after the reply settles.
209        view.state.assistant_metrics.get(id).hash(&mut hasher);
210        view.state
211            .interrupted_assistant_items
212            .contains(id)
213            .hash(&mut hasher);
214        children.len().hash(&mut hasher);
215        for child in children {
216            child.version().hash(&mut hasher);
217            match child {
218                AssistantChild::Thinking { id: part, .. } => {
219                    view.expanded.thinking(*part).hash(&mut hasher);
220                    hash_thinking_status(view.state, *id, *part, &mut hasher);
221                }
222                AssistantChild::Tool { call_id } => pass.hash_tool(call_id, &mut hasher),
223                _ => {}
224            }
225        }
226    }
227    if let Some(key) = block_key(item) {
228        view.expanded.block(&key).hash(&mut hasher);
229    }
230    if let TranscriptItem::ProducerMessage { .. } = item {
231        (view.minimum_level == EventLevel::Debug).hash(&mut hasher);
232    }
233    if let TranscriptItem::Event { level, .. } = item {
234        (*level >= view.minimum_level).hash(&mut hasher);
235    }
236    hasher.finish()
237}

239/// The item as HTML; `None` for a diagnostic row the level filter hides.
240pub fn item_html(view: &TranscriptView<'_>, item: &TranscriptItem) -> Option<String> {
241    Some(match item {
242        TranscriptItem::User { markdown, seq, .. } => {
243            let files = view
244                .state
245                .user_files
246                .get(seq)
247                .map(|files| {
248                    files
249                        .iter()
250                        .map(|file| {
251                            media_html(
252                                &file.sha256,
253                                &file.media_type,
254                                file.byte_length,
255                                file.filename.as_deref(),
256                            )
257                        })
258                        .collect::<String>()
259                })
260                .unwrap_or_default();
261            format!(
262                // The row is the whole band at the bubble's height: a mouse
263                // over it shows the ⋯ (style/main.css), whose room stays;
264                // a touch screen long-presses the bubble instead.
265                "<div class=\"user-row flex justify-end\" data-seq=\"{seq}\"><div class=\"user-message group ml-auto flex w-fit max-w-[80%] items-start justify-end gap-1\" data-seq=\"{seq}\"><button type=\"button\" data-action=\"user-menu\" data-seq=\"{seq}\" aria-label=\"Message actions\" title=\"Message actions\" class=\"inline-flex min-h-11 min-w-11 shrink-0 items-center justify-center rounded-md text-ink hover:bg-toasted active:bg-crust dark:text-sand dark:hover:bg-ember dark:active:bg-smoke\">{}</button><div class=\"min-w-0 rounded-2xl rounded-tr-sm bg-toasted px-3 py-2 dark:bg-ember\" data-bubble><div class=\"markdown\">{}</div>{files}</div></div></div>",
266                svg(icondata::FaEllipsisSolid, ""),
267                document_html(markdown)
268            )
269        }
270        TranscriptItem::Assistant {
271            id,
272            attribution,
273            children,
274            ..
275        } => {
276            let mut html = format!(
277                "<div class=\"{NOTE} font-mono\">{}</div>",
278                escape(&attribution.header(view.names))
279            );
280            for child in children {
281                html.push_str(&child_html(view, *id, child));
282            }
283            html.push_str(&reply_footer_html(view.state, *id).unwrap_or_default());
284            format!("<div class=\"flex flex-col gap-1\">{html}</div>")
285        }
286        TranscriptItem::Event {
287            level,
288            text,
289            repeat,
290            ..
291        } => {
292            if *level < view.minimum_level {
293                return None;
294            }
295            let class = match level {
296                EventLevel::Error => "text-red-700 dark:text-red-400",
297                EventLevel::Warning => "text-caramel dark:text-maple",
298                EventLevel::Info | EventLevel::Debug => "text-ink dark:text-sand",
299            };
300            let repeat = if *repeat > 1 {
301                format!(" ×{repeat}")
302            } else {
303                String::new()
304            };
305            format!(
306                "<div class=\"font-mono text-xs whitespace-pre-wrap {class}\">{} {}{repeat}</div>",
307                level.badge(),
308                escape(text)
309            )
310        }
311        TranscriptItem::Compacting { source, .. } => {
312            let key = block_key(item).unwrap_or_default();
313            let body = match source {
314                CompactingSource::Summary { text, .. } if text.is_empty() => {
315                    "waiting for the summary".to_owned()
316                }
317                CompactingSource::Summary { text, .. } => text.clone(),
318                CompactingSource::Native { model } => {
319                    format!("calling native compaction endpoint ({model})")
320                }
321            };
322            collapsible(
323                view.expanded.block(&key),
324                &key,
325                icondata::FaBroomSolid,
326                "compacting context…",
327                None,
328                &body,
329            )
330        }
331        TranscriptItem::Compaction { commit, .. } => {
332            let key = block_key(item).unwrap_or_default();
333            let (kind, body) = match &commit.checkpoint {
334                ContextCheckpoint::InternalSummary { checkpoint, .. } => {
335                    ("internal summary", checkpoint.summary().to_owned())
336                }
337                ContextCheckpoint::NativeWindow { window } => {
338                    let scope = window.scope();

356                view.expanded.block(&key),

359                &header,
360                None,
361                &body,
362            )
363        }
364        TranscriptItem::PluginMessage { role, input, .. } => {
365            let key = block_key(item).unwrap_or_default();

367                ExtensionMessageRole::System => "system",
368                ExtensionMessageRole::User => "user",
369                ExtensionMessageRole::Assistant => "assistant",

371            };
372            let header = format!(
373                "plugin message ({role}, {} lines)",

569                    body,

571            }
572            ProducerMessageStatus::Discarded if view.minimum_level == EventLevel::Debug => {
573                format!(
574                    "<div class=\"{NOTE} font-mono\">{}</div>",
575                    escape(&format!(
576                        "producer message discarded · {} · {}",
577                        producer_owner_label(producer_owner),
578                        producer_mode_label(*mode)
579                    ))
580                )
581            }
582            ProducerMessageStatus::Discarded => return None,
583        },
584    })

630/// Whether HTML holds what the page fills in after drawing it: a code
631/// block to highlight, or media to read.
632pub fn fills_in(html: &str) -> bool {
633    html.contains("language-") || html.contains("data-media")
634}

636/// The DOM IDs of the rows, and of the parts of replies, whose HTML changed
637/// since `old` and holds something the page fills in after drawing it:
638/// only those need looking into. What kept its HTML kept its nodes, filled
639/// in already (the page compares HTML as it draws: by content, the same
640/// text first by pointer). Text is left out: the page fills in each of its
641/// markdown blocks as it adds it.
642pub fn to_fill_in(old: &Rows, new: &Rows) -> Vec<String> {
643    let mut found = Vec::new();
644    for id in new.order.iter().copied() {
645        match (old.html.get(&id), new.html.get(&id)) {
646            (Some(old), Some(new)) if old == new => {}
647            (_, Some(RowHtml::Html(html))) => {
648                if fills_in(html) {
649                    found.push(row_element_id(id));
650                }
651            }
652            (old, Some(RowHtml::Reply { parts, .. })) => {
653                let old = match old {
654                    Some(RowHtml::Reply { parts, .. }) => parts
655                        .iter()
656                        .map(|(part, html)| (*part, html))
657                        .collect::<FastMap<_, _>>(),
658                    _ => FastMap::default(),
659                };
660                for (part, html) in parts.iter() {
661                    if let PartHtml::Html(html) = html
662                        && !matches!(old.get(part), Some(PartHtml::Html(old)) if old == html)
663                        && fills_in(html)
664                    {
665                        found.push(part_element_id(id, *part));
666                    }
667                }
668            }
669            (_, None) => {}
670        }
671    }
672    found
673}

707/// A reply part's ID within its reply: its segment, call, or committed
708/// content index; the rest by position.
709fn part_id(child: &AssistantChild, index: usize) -> u64 {
710    let state = foldhash::fast::FixedState::default();
711    match child {
712        AssistantChild::Text { id, .. } => state.hash_one((0u8, id)),
713        AssistantChild::Thinking { id, .. } => state.hash_one((1u8, id)),
714        AssistantChild::Tool { call_id } => state.hash_one((2u8, call_id)),
715        AssistantChild::CommittedTool {

722            content_index,
723            ..
724        } => state.hash_one((4u8, turn_seq, content_index)),

730/// A key that changes whenever the part's HTML would.
731fn part_key(pass: &Pass<'_, '_>, item_id: u64, child: &AssistantChild) -> u64 {
732    let view = pass.view;
733    let mut hasher = foldhash::fast::FixedState::default().build_hasher();
734    // Segment and turn numbers repeat across sessions.
735    view.session_id.hash(&mut hasher);
736    pass.homes.hash(&mut hasher);
737    std::mem::discriminant(child).hash(&mut hasher);
738    match child {
739        // A folded thinking block shows only its header, whatever streams
740        // into it.
741        AssistantChild::Thinking { id, version, .. } => {
742            let open = view.expanded.thinking(*id);
743            (id, open).hash(&mut hasher);
744            hash_thinking_status(view.state, item_id, *id, &mut hasher);
745            if open {
746                version.hash(&mut hasher);
747            }
748        }
749        AssistantChild::Tool { call_id } => pass.hash_tool(call_id, &mut hasher),
750        // The rest show what they hold, which does not change.
751        AssistantChild::Text { id, version, .. } => (id, version).hash(&mut hasher),
752        AssistantChild::CommittedTool {
753            turn_seq,
754            content_index,
755            name,
756        } => (turn_seq, content_index, name.as_str()).hash(&mut hasher),
757        AssistantChild::MediaFile {
758            turn_seq,
759            content_index,
760            ..
761        } => (turn_seq, content_index).hash(&mut hasher),
762        AssistantChild::Attribution { resolved_model } => {
763            view.names
764                .selection_label(&resolved_model.selection)
765                .hash(&mut hasher);
766        }
767        AssistantChild::Notice { text } => text.hash(&mut hasher),
768    }
769    hasher.finish()
770}

772/// The transcript's rows: the pinned row (`pinned`, from [`pinned_rows`])
773/// first, then every item the view shows. Rows and parts whose content
774/// key is in `cache` reuse its HTML.
775pub fn transcript_rows(
776    view: &TranscriptView<'_>,
777    pinned: Option<String>,
778    cache: &mut RowCache,
779) -> Rows {
780    let pass = Pass::new(view);
781    let mut rows = fast_map(cache.rows.len());
782    let mut parts = fast_map(cache.parts.len());
783    let mut order = Vec::with_capacity(view.state.transcript.len() + 1);
784    let mut html = fast_map(view.state.transcript.len() + 1);
785    if let Some(pinned) = pinned {
786        let id = pinned_row_id(view.session_id);
787        order.push(id);
788        html.insert(id, RowHtml::Html(Arc::from(pinned)));
789    }
790    for item in &view.state.transcript {
791        let key = item_key_in(&pass, item);
792        let row = match cache.rows.remove(&key) {
793            Some(row) => {
794                for key in &row.parts {
795                    if let Some(part) = cache.parts.get(key) {
796                        parts.insert(*key, part.clone());
797                    }
798                }
799                row
800            }
801            None => {
802                let row = match item {
803                    TranscriptItem::Assistant {
804                        id,
805                        attribution,
806                        children,
807                        ..
808                    } => {
809                        let (reply, keys) = reply_parts(&pass, *id, children, cache, &mut parts);
810                        CachedRow {
811                            html: RowHtml::Reply {
812                                header: Arc::from(format!(
813                                    "<div class=\"{NOTE} font-mono\">{}</div>",
814                                    escape(&attribution.header(view.names))
815                                )),
816                                parts: reply,
817                            },
818                            parts: keys,
819                        }
820                    }
821                    _ => match item_html(view, item) {
822                        Some(item) => CachedRow {
823                            html: RowHtml::Html(Arc::from(item)),
824                            parts: Vec::new(),
825                        },
826                        None => continue,
827                    },
828                };
829                cache.drawn += 1;
830                row
831            }
832        };
833        let id = row_id(view, item);
834        if html.insert(id, row.html.clone()).is_none() {
835            order.push(id);
836        }
837        rows.insert(key, row);
838    }
839    cache.rows = rows;
840    cache.parts = parts;
841    Rows {
842        order: order.into(),
843        html: Arc::new(html),
844    }
845}

847/// A reply's parts, each under its ID, and their content keys, then its
848/// footer when it has one. Parts whose key is in `cache` reuse its HTML;
849/// every part drawn lands in `used`.
850fn reply_parts(
851    pass: &Pass<'_, '_>,
852    item_id: u64,
853    children: &[AssistantChild],
854    cache: &mut RowCache,
855    used: &mut FastMap<u64, PartHtml>,
856) -> (Arc<[(u64, PartHtml)]>, Vec<u64>) {
857    let mut seen: FastSet<u64> =
858        HashSet::with_capacity_and_hasher(children.len(), foldhash::fast::FixedState::default());
859    let mut keys = Vec::with_capacity(children.len());
860    let mut parts: Vec<_> = children
861        .iter()
862        .enumerate()
863        .map(|(index, child)| {
864            let mut id = part_id(child, index);
865            if !seen.insert(id) {
866                id = foldhash::fast::FixedState::default().hash_one((id, index));
867                seen.insert(id);
868            }
869            let key = part_key(pass, item_id, child);
870            let html = match used.get(&key).or_else(|| cache.parts.get(&key)) {
871                Some(html) => html.clone(),
872                None => match child {
873                    // Each block is drawn once, and kept with its block.
874                    AssistantChild::Text { markdown, .. } => {
875                        PartHtml::Markdown(markdown.blocks().map(block_html).collect())
876                    }
877                    _ => {
878                        cache.drawn += 1;
879                        PartHtml::Html(Arc::from(child_html(pass.view, item_id, child)))
880                    }
881                },
882            };
883            used.insert(key, html.clone());
884            keys.push(key);
885            (id, html)
886        })
887        .collect();
888    // Drawn with its row, whose key holds what it shows.
889    if let Some(footer) = reply_footer_html(pass.view.state, item_id) {
890        parts.push((
891            foldhash::fast::FixedState::default().hash_one(7u8),
892            PartHtml::Html(Arc::from(footer)),
893        ));
894    }
895    (parts.into(), keys)
896}

1907            version, children, ..
1908        } = &mut state.transcript[0]
1909        else {
1910            unreachable!()
1911        };

1916                    version, markdown, ..

1922                    version, text: own, ..
1923                } => {
1924                    *version += 1;
1925                    own.push_str(thinking);
1926                }
1927                _ => {}
