27/// What the page shows besides the projection itself.
28pub struct TranscriptView<'a> {
29    /// The session the projection is of: item IDs repeat across sessions.
30    pub session_id: cookie_agent_protocol::SessionId,
31    pub state: &'a SessionState,
32    pub names: &'a ModelDisplayNames,
33    /// Diagnostic rows below this level are hidden (`/events`).
34    pub minimum_level: EventLevel,
35    /// Tool calls the user expanded.
36    pub expanded: &'a HashSet<String>,
37}

41/// A key that changes whenever the item's HTML would: its version, the
42/// revisions of the tool calls it shows, and their expansion.
43pub fn item_key(view: &TranscriptView<'_>, item: &TranscriptItem) -> u64 {
44    let mut hasher = foldhash::fast::FixedState::default().build_hasher();
45    view.session_id.hash(&mut hasher);
46    item.id().hash(&mut hasher);
47    item.version().hash(&mut hasher);
48    std::mem::discriminant(item).hash(&mut hasher);
49    if let TranscriptItem::Assistant { children, .. } = item {
50        children.len().hash(&mut hasher);
51        for child in children {
52            child.version().hash(&mut hasher);
53            if let AssistantChild::Tool { call_id } = child {
54                let id = call_id.to_string();
55                view.expanded.contains(&id).hash(&mut hasher);
56                // Raw arguments show only at `info` and below.
57                (view.minimum_level <= EventLevel::Info).hash(&mut hasher);
58                if let Some(tool) = view.state.tools.get(call_id) {
59                    tool.revision.hash(&mut hasher);
60                }
61            }
62        }
63    }
64    if let Some(key) = block_key(item) {
65        view.expanded.contains(&key).hash(&mut hasher);
66    }
67    if let TranscriptItem::ProducerMessage { .. } = item {
68        (view.minimum_level == EventLevel::Debug).hash(&mut hasher);
69    }
70    if let TranscriptItem::Event { level, .. } = item {
71        (*level >= view.minimum_level).hash(&mut hasher);
72    }
73    hasher.finish()
74}

76/// The item as HTML; `None` for a diagnostic row the level filter hides.
77pub fn item_html(view: &TranscriptView<'_>, item: &TranscriptItem) -> Option<String> {
78    Some(match item {
79        TranscriptItem::User { markdown, seq, .. } => {
80            let files = view
81                .state
82                .user_files
83                .get(seq)
84                .map(|files| {
85                    files
86                        .iter()
87                        .map(|file| {
88                            media_html(
89                                &file.sha256,
90                                &file.media_type,
91                                file.byte_length,
92                                file.filename.as_deref(),
93                            )
94                        })
95                        .collect::<String>()
96                })
97                .unwrap_or_default();
98            format!(
99                "<div class=\"user-message group ml-auto flex max-w-[90%] items-start gap-1\" data-seq=\"{seq}\"><button type=\"button\" data-action=\"user-menu\" data-seq=\"{seq}\" aria-label=\"Message actions\" class=\"min-h-11 min-w-11 rounded text-ink hover:bg-toasted active:bg-crust dark:text-sand dark:hover:bg-ember dark:active:bg-smoke\">⋯</button><div class=\"min-w-0 rounded-lg bg-toasted px-3 py-2 dark:bg-ember\"><div class=\"markdown\">{}</div>{files}</div></div>",
100                document_html(markdown)
101            )
102        }
103        TranscriptItem::Assistant {
104            attribution,
105            children,
106            ..
107        } => {
108            let mut html = format!(
109                "<div class=\"{NOTE} font-mono\">{}</div>",
110                escape(&attribution.header(view.names))
111            );
112            for child in children {
113                html.push_str(&child_html(view, child));
114            }
115            format!("<div class=\"flex flex-col gap-1\">{html}</div>")
116        }
117        TranscriptItem::Event {
118            level,
119            text,
120            repeat,
121            ..
122        } => {
123            if *level < view.minimum_level {
124                return None;
125            }
126            let class = match level {
127                EventLevel::Error => "text-red-700 dark:text-red-400",
128                EventLevel::Warning => "text-caramel dark:text-maple",
129                EventLevel::Info | EventLevel::Debug => "text-ink dark:text-sand",
130            };
131            let repeat = if *repeat > 1 {
132                format!(" ×{repeat}")
133            } else {
134                String::new()
135            };
136            format!(
137                "<div class=\"font-mono text-xs whitespace-pre-wrap {class}\">{} {}{repeat}</div>",
138                level.badge(),
139                escape(text)
140            )
141        }
142        TranscriptItem::Compacting { source, .. } => {
143            let key = block_key(item).unwrap_or_default();
144            let body = match source {
145                CompactingSource::Summary { text, .. } if text.is_empty() => {
146                    "waiting for the summary".to_owned()
147                }
148                CompactingSource::Summary { text, .. } => text.clone(),
149                CompactingSource::Native { model } => {
150                    format!("calling native compaction endpoint ({model})")
151                }
152            };
153            collapsible(view, &key, "🧹", "compacting context…", None, &body)
154        }
155        TranscriptItem::Compaction { commit, .. } => {
156            let key = block_key(item).unwrap_or_default();
157            let (kind, body) = match &commit.checkpoint {
158                ContextCheckpoint::InternalSummary { checkpoint } => {
159                    ("internal summary", checkpoint.summary().to_owned())
160                }
161                ContextCheckpoint::NativeWindow { window } => {
162                    let scope = window.scope();
163                    (
164                        "native window",
165                        format!(
166                            "adapter: {}\nmodel: {}/{}\nresource: {}",
167                            window.adapter_id(),
168                            scope.provider_id,
169                            scope.model_id,
170                            scope.resource_id
171                        ),
172                    )
173                }
174            };
175            let header = format!(
176                "context compacted ({kind}, {}→{} tokens)",
177                commit.budgets.input_tokens_before, commit.budgets.input_tokens_after
178            );
179            collapsible(view, &key, "🧹", &header, None, &body)
180        }
181        TranscriptItem::PluginMessage { role, input, .. } => {
182            let key = block_key(item).unwrap_or_default();
183            let role = match role {
184                ExtensionMessageRole::System => "system",
185                ExtensionMessageRole::User => "user",
186                ExtensionMessageRole::Assistant => "assistant",
187                ExtensionMessageRole::Tool => "tool",
188            };
189            let header = format!(
190                "plugin message ({role}, {} lines)",
191                display_line_count(input)
192            );
193            collapsible(view, &key, "🧩", &header, None, input)
194        }
195        TranscriptItem::AgentMd { entries, .. } => {
196            let key = block_key(item).unwrap_or_default();
197            let lines: usize = entries
198                .iter()
199                .map(|entry| display_line_count(&entry.content))
200                .sum();
201            let files = match entries.as_slice() {
202                [entry] => entry.source.as_str().to_owned(),
203                _ => format!("{} files", entries.len()),
204            };
205            let body = entries
206                .iter()
207                .map(|entry| {
208                    if entries.len() > 1 {
209                        format!("── {}\n{}", entry.source.as_str(), entry.content)
210                    } else {
211                        entry.content.clone()
212                    }
213                })
214                .collect::<Vec<_>>()
215                .join("\n");
216            let header = format!("AGENTS.md · {files} ({lines} lines)");
217            collapsible(view, &key, "📘", &header, None, &body)
218        }
219        TranscriptItem::SkillLoaded {
220            name,
221            source_path,
222            args,
223            rendered_body,
224            ..
225        } => {
226            let key = block_key(item).unwrap_or_default();
227            let header = format!(
228                "skill loaded · {name} ({} lines)",
229                display_line_count(rendered_body)
230            );
231            let mut detail = format!("source: {source_path}");
232            if !args.is_empty() {
233                detail.push_str(&format!("\narguments: {args}"));
234            }
235            collapsible(view, &key, "📚", &header, Some(&detail), rendered_body)
236        }
237        TranscriptItem::Goal {
238            goal, activation, ..
239        } => {
240            if *activation {
241                // The canonical `/goal` action, rebuilt from the activation
242                // event: shown like a message, with no message menu.
243                format!(
244                    "<div class=\"goal-activation ml-auto max-w-[90%] rounded-lg border border-caramel px-3 py-2 font-mono text-sm whitespace-pre-wrap dark:border-maple\">{}</div>",
245                    escape(&format!("/goal {}", goal.objective))
246                )
247            } else {
248                let finished = goal.items.iter().filter(|item| item.finished).count();
249                let checklist = if goal.items.is_empty() {
250                    format!("<div class=\"{NOTE}\">checklist: empty</div>")
251                } else {
252                    goal.items
253                        .iter()
254                        .map(|item| {
255                            format!(
256                                "<div class=\"{NOTE}\">{} {}</div>",
257                                if item.finished { "[x]" } else { "[ ]" },
258                                escape(&item.description)
259                            )
260                        })
261                        .collect()
262                };
263                format!(
264                    "<div class=\"goal-state rounded border-l-4 border-caramel px-3 py-1 dark:border-maple\"><div class=\"text-sm whitespace-pre-wrap\">🎯 {}</div><div class=\"{NOTE}\">status: {} · {finished}/{} finished</div>{checklist}</div>",
265                    escape(&goal.objective),
266                    goal_status_label(goal.status),
267                    goal.items.len()
268                )
269            }
270        }
271        TranscriptItem::ProducerMessage {
272            producer_owner,
273            mode,
274            body,
275            summary,
276            status,
277            ..
278        } => match status {
279            // A claimed message is already in the running request, so it
280            // shows before the response; pending and admitted ones do not
281            // show yet, as in the TUI.
282            ProducerMessageStatus::Pending | ProducerMessageStatus::Admitted => return None,
283            ProducerMessageStatus::Claimed | ProducerMessageStatus::Consumed => {
284                let key = block_key(item).unwrap_or_default();
285                let state = if *status == ProducerMessageStatus::Claimed {
286                    "claimed"
287                } else {
288                    "consumed"
289                };
290                let detail = format!(
291                    "{} · {} · {state}",
292                    producer_owner_label(producer_owner),
293                    producer_mode_label(*mode)
294                );
295                collapsible(
296                    view,
297                    &key,
298                    "◇",
299                    &producer_summary(producer_owner, *mode, summary.as_deref()),
300                    Some(&detail),
301                    body,
302                )
303            }
304            ProducerMessageStatus::Discarded if view.minimum_level == EventLevel::Debug => {
305                format!(
306                    "<div class=\"{NOTE} font-mono\">{}</div>",
307                    escape(&format!(
308                        "producer message discarded · {} · {}",
309                        producer_owner_label(producer_owner),
310                        producer_mode_label(*mode)
311                    ))
312                )
313            }
314            ProducerMessageStatus::Discarded => return None,
315        },
316    })
317}

319/// The key under which a collapsible row's expansion is kept, shared with
320/// the tool calls' in the page's expanded set.
321fn block_key(item: &TranscriptItem) -> Option<String> {
322    Some(match item {
323        TranscriptItem::Compacting { seq, .. } => format!("compacting:{seq}"),
324        TranscriptItem::Compaction { seq, .. } => format!("compaction:{seq}"),
325        TranscriptItem::PluginMessage { seq, .. } => format!("plugin:{seq}"),
326        TranscriptItem::AgentMd { seq, .. } => format!("agent-md:{seq}"),
327        TranscriptItem::SkillLoaded { seq, .. } => format!("skill:{seq}"),
328        TranscriptItem::ProducerMessage { message_id, .. } => format!("producer:{message_id}"),
329        _ => return None,
330    })
331}

333/// A row that shows `header` and, once clicked open, `detail` and `body`.
334fn collapsible(
335    view: &TranscriptView<'_>,
336    key: &str,
337    icon: &str,
338    header: &str,
339    detail: Option<&str>,
340    body: &str,
341) -> String {
342    let expanded = view.expanded.contains(key);
343    let chevron = if expanded { "▾" } else { "▸" };
344    let mut html = format!(
345        "<div class=\"collapsible-row\" data-block-key=\"{}\"><button type=\"button\" data-action=\"toggle-block\" data-block-key=\"{}\" aria-expanded=\"{expanded}\" class=\"flex min-h-11 w-full items-center gap-2 rounded px-2 text-left {NOTE} hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke\"><span>{icon} {chevron}</span><span class=\"min-w-0 flex-1 truncate\">{}</span></button>",
346        escape(key),
347        escape(key),
348        escape(header)
349    );
350    if expanded {
351        html.push_str("<div class=\"border-l border-toasted pl-3 dark:border-ember\">");
352        if let Some(detail) = detail {
353            html.push_str(&format!(
354                "<div class=\"{NOTE} whitespace-pre-wrap\">{}</div>",
355                escape(detail)
356            ));
357        }
358        html.push_str(&format!(
359            "<div class=\"{NOTE} whitespace-pre-wrap break-words\">{}</div>",
360            escape(body)
361        ));
362        html.push_str("</div>");
363    }
364    html.push_str("</div>");
365    html
366}

368fn child_html(view: &TranscriptView<'_>, child: &AssistantChild) -> String {
369    match child {
370        AssistantChild::Text { markdown, .. } => {
371            format!("<div class=\"markdown\">{}</div>", document_html(markdown))
372        }
373        AssistantChild::Thinking { text, .. } => format!(
374            "<details class=\"{NOTE}\"><summary class=\"cursor-pointer hover:text-espresso dark:hover:text-latte\">thinking</summary><div class=\"whitespace-pre-wrap\">{}</div></details>",
375            escape(text)
376        ),
377        AssistantChild::Tool { call_id } => match view.state.tools.get(call_id) {
378            Some(tool) => tool_html(view, tool),
379            None => String::new(),
380        },
381        AssistantChild::CommittedTool { name, .. } => format!(
382            "<div class=\"{NOTE} font-mono\">🔨 {} (not started)</div>",
383            escape(name.as_str())
384        ),
385        AssistantChild::MediaFile { file, .. } => match &file.source {
386            PersistedFileSource::Artifact {
387                byte_length,
388                sha256,
389                ..
390            } => media_html(
391                sha256.as_str(),
392                file.media_type.as_str(),
393                *byte_length,
394                file.filename.as_deref(),
395            ),
396            PersistedFileSource::Url { url } if super::safe_link(url) => format!(
397                "<a href=\"{}\" target=\"_blank\" rel=\"noopener noreferrer\" class=\"text-caramel underline hover:text-espresso dark:text-maple dark:hover:text-latte\">{}</a>",
398                escape(url),
399                escape(file.filename.as_deref().unwrap_or(url))
400            ),
401            _ => format!(
402                "<div class=\"{NOTE}\">{}</div>",
403                escape(&format!("media: {}", file.media_type))
404            ),
405        },
406        AssistantChild::Attribution { resolved_model } => format!(
407            "<div class=\"{NOTE} font-mono\">→ {}</div>",
408            escape(&view.names.selection_label(&resolved_model.selection))
409        ),
410        AssistantChild::Notice { text } => {
411            format!("<div class=\"{NOTE} italic\">{}</div>", escape(text))
412        }
413    }
414}

416fn tool_html(view: &TranscriptView<'_>, tool: &ToolCallState) -> String {
417    let id = tool.id.to_string();
418    let (status, class) = match tool.status {
419        ToolStatus::Running if tool.backgrounded => {
420            ("in the background", "text-ink dark:text-sand")
421        }
422        ToolStatus::Running => ("running", "text-caramel dark:text-maple"),
423        ToolStatus::Completed => ("done", "text-green-700 dark:text-green-400"),
424        ToolStatus::Failed => ("failed", "text-red-700 dark:text-red-400"),
425        ToolStatus::Cancelled => ("cancelled", "text-ink dark:text-sand"),
426        ToolStatus::Interrupted => ("interrupted", "text-ink dark:text-sand"),
427    };
428    let expanded = view.expanded.contains(&id) || tool.status == ToolStatus::Running;
429    // A running call the daemon can move to the background offers it, as
430    // the TUI's title-row button does.
431    let background = if tool.status == ToolStatus::Running
432        && tool.backgroundable
433        && !tool.backgrounded
434    {
435        format!(
436            "<button type=\"button\" data-action=\"background\" data-call=\"{}\" title=\"Move to the background\" class=\"min-h-11 shrink-0 rounded px-2 text-xs hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke\">⇲ background</button>",
437            escape(&id)
438        )
439    } else {
440        String::new()
441    };
442    let mut html = format!(
443        "<div class=\"rounded border border-toasted dark:border-ember\" data-tool=\"{}\"><div class=\"flex items-center\"><button type=\"button\" data-action=\"toggle-tool\" data-call=\"{}\" aria-expanded=\"{expanded}\" class=\"flex min-h-11 min-w-0 flex-1 items-center gap-2 px-2 py-1 text-left font-mono text-sm hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke\"><span>{}</span><span class=\"flex-1 truncate\">{}</span><span class=\"{class}\">{status}</span></button>{background}</div>",
444        escape(&id),
445        escape(&id),
446        hint_icon(tool.presentation.hint.as_ref().map(|hint| hint.rendered())),
447        escape(&tool.header_text()),
448    );
449    if let Some(progress) = tool
450        .progress
451        .as_ref()
452        .filter(|_| tool.status == ToolStatus::Running)
453    {
454        html.push_str(&format!(
455            "<div class=\"px-2 {NOTE}\">{}</div>",
456            escape(progress)
457        ));
458    }
459    if expanded {
460        html.push_str("<div class=\"flex flex-col gap-1 px-2 pb-2\">");
461        if tool.presentation.arguments.is_empty() {
462            // As in the TUI: a presentation with a preview is complete as
463            // given, and raw arguments are diagnostic detail at `info` and
464            // below.
465            if !tool.arguments.is_empty()
466                && tool.presentation.preview.is_empty()
467                && view.minimum_level <= EventLevel::Info
468            {
469                html.push_str(&format!(
470                    "<pre class=\"overflow-x-auto whitespace-pre-wrap font-mono text-xs\">{}</pre>",
471                    escape(&tool.arguments)
472                ));
473            }
474        } else {
475            html.push_str(&display_html(
476                &tool.presentation.arguments,
477                BlockPlace::plain,
478            ));
479        }
480        if let Some(failure) = &tool.failure {
481            html.push_str(&format!(
482                "<div class=\"text-sm text-red-700 dark:text-red-400\">{}</div>",
483                escape(failure)
484            ));
485        }
486        html.push_str(&display_html(tool.shown_display(), |index| BlockPlace {
487            call_id: Some(&id),
488            index,
489            full: tool.full_content(index),
490        }));
491        html.push_str("</div>");
492    }
493    html.push_str("</div>");
494    html
495}

508    fn tool(

510        arguments: &str,
511    ) -> cookie_agent_client_state::ToolCallState {
512        let name = presentation.title.to_string();
513        let presentation = cookie_agent_protocol::ToolPresentation::new(
514            presentation,
515            cookie_agent_protocol::ToolIdentity {
516                name: name.clone(),
517                published_name: name,
518                source: None,
519            },

521        let owner = cookie_agent_protocol::AssistantToolCallRef {
522            model_turn_seq: 1,
523            content_index: 0,
524            model_call_id: cookie_agent_protocol::ModelCallId::new("call-1").expect("call ID"),
525            provider_item_id: None,
526        };
527        let mut tool = cookie_agent_client_state::ToolCallState::started(
528            cookie_agent_protocol::ToolCallId::new_v7(),
529            owner,
530            presentation,
531            arguments.into(),
532        );
533        tool.status = cookie_agent_client_state::ToolStatus::Completed;
534        tool
535    }

537    #[test]
538    fn raw_arguments_show_only_without_blocks_or_a_preview_at_info() {
539        let state = SessionState::default();
540        let names = ModelDisplayNames::default();
541        let plain = tool(
542            cookie_agent_protocol::PreparedPresentation::new(
543                cookie_agent_protocol::SafeDisplayText::new("Lookup").expect("title"),
544                None,
545            ),
546            r#"{"query":"raw-marker"}"#,
547        );
548        let previewed = tool(
549            cookie_agent_protocol::PreparedPresentation::new(
550                cookie_agent_protocol::SafeDisplayText::new("Edit a.rs").expect("title"),
551                None,
552            )
553            .with_preview([cookie_agent_protocol::DisplayBlock::text("the preview")]),
554            r#"{"filePath":"a.rs","newString":"raw-marker"}"#,
555        );
556        let expanded = HashSet::from([plain.id.to_string(), previewed.id.to_string()]);
557        let view = |minimum_level| TranscriptView {
558            session_id: cookie_agent_protocol::SessionId::new_v7(),
559            state: &state,
560            names: &names,
561            minimum_level,
562            expanded: &expanded,
563        };
564        assert!(tool_html(&view(EventLevel::Info), &plain).contains("raw-marker"));
565        assert!(!tool_html(&view(EventLevel::Warning), &plain).contains("raw-marker"));
566        let previewed_html = tool_html(&view(EventLevel::Debug), &previewed);
567        assert!(!previewed_html.contains("raw-marker"), "{previewed_html}");
568    }

570    #[test]
571    fn rows_below_the_event_level_are_hidden_and_text_is_escaped() {
572        let state = SessionState::default();
573        let names = ModelDisplayNames::default();
574        let expanded = HashSet::new();
575        let session_id = cookie_agent_protocol::SessionId::new_v7();
576        let view = |minimum_level| TranscriptView {
577            session_id,
578            state: &state,
579            names: &names,
580            minimum_level,
581            expanded: &expanded,
582        };
583        let row = TranscriptItem::internal("<careful>");
584        let shown = item_html(&view(EventLevel::Info), &row).expect("shown");
585        assert!(shown.contains("[I] &lt;careful&gt;"), "{shown}");
586        assert_eq!(item_html(&view(EventLevel::Warning), &row), None);
587        assert_ne!(
588            item_key(&view(EventLevel::Info), &row),
589            item_key(&view(EventLevel::Warning), &row)
590        );
591        let user =
592            item_html(&view(EventLevel::Info), &TranscriptItem::user("hi <b>")).expect("user");
593        assert!(user.contains("hi &lt;b&gt;"), "{user}");
594        // Item IDs repeat across sessions: the same item of another session
595        // keys apart, so switching sessions never shows a stale row.
596        let other = TranscriptView {
597            session_id: cookie_agent_protocol::SessionId::new_v7(),
598            ..view(EventLevel::Info)
599        };
600        assert_ne!(
601            item_key(&view(EventLevel::Info), &row),
602            item_key(&other, &row)
603        );
604    }

606    #[test]
607    fn producer_messages_fold_under_their_summary_and_open_on_a_click() {
608        let state = SessionState::default();
609        let names = ModelDisplayNames::default();
610        let message_id = cookie_agent_protocol::ProducerMessageId::new_v7();
611        let row = |status| TranscriptItem::ProducerMessage {
612            id: 1,
613            seq: 4,
614            accepted_at: jiff::Timestamp::UNIX_EPOCH,
615            message_id,
616            producer_owner: cookie_agent_protocol::ProducerOwner::GoalControl {
617                goal_id: cookie_agent_protocol::GoalId::new_v7(),
618            },
619            mode: cookie_agent_protocol::ProducerDeliveryMode::Steer,
620            body: "[engine goal control]\nThe goal <x> was paused.".into(),
621            summary: None,
622            reminder: None,
623            status,
624        };
625        let collapsed = HashSet::new();
626        let opened = HashSet::from([format!("producer:{message_id}")]);
627        let view = |expanded| TranscriptView {
628            session_id: cookie_agent_protocol::SessionId::new_v7(),
629            state: &state,
630            names: &names,
631            minimum_level: EventLevel::Warning,
632            expanded,
633        };
634        let consumed = row(ProducerMessageStatus::Consumed);
635        let folded = item_html(&view(&collapsed), &consumed).expect("shown");
636        assert!(folded.contains("goal control · steer"), "{folded}");
637        assert!(!folded.contains("engine goal control"), "{folded}");
638        let open = item_html(&view(&opened), &consumed).expect("shown");
639        assert!(open.contains("The goal &lt;x&gt; was paused."), "{open}");
640        assert_eq!(
641            item_html(&view(&collapsed), &row(ProducerMessageStatus::Pending)),
642            None
643        );
644        assert_eq!(
645            item_html(&view(&collapsed), &row(ProducerMessageStatus::Discarded)),
646            None
647        );
648    }
