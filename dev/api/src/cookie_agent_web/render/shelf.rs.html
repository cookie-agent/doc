1//! A run's shelf as HTML (`docs/site/specs/assets.md` section 8.1): its
2//! assets as cards in one wrapping row, whose widths let them share rows.
3//! Images and video are one deck, a page and a document cards that open in
4//! the side panel, audio a player, and any other file a chip with a
5//! download. Each card is an asset placeholder (`data-asset`) the page fills
6//! in after drawing it (`crate::assets`), in the way its `data-asset-view`
7//! names. Handles the reply names become chips that open their card.

26/// How a shelf entry shows.
27#[derive(Clone, Copy, Debug, Eq, PartialEq)]
28pub enum Card {
29    /// In the deck: an image, an SVG, or a video.
30    Deck,
31    Page,
32    Document,
33    Audio,
34    File,
35}
36
37impl Card {
38    #[must_use]
39    pub fn of(media_type: &str) -> Self {
40        if media_type.starts_with("image/") || media_type.starts_with("video/") {
41            Self::Deck
42        } else if media_type.starts_with("audio/") {
43            Self::Audio
44        } else {
45            match AssetKind::of(media_type) {
46                AssetKind::Page => Self::Page,
47                AssetKind::Document => Self::Document,
48                _ => Self::File,
49            }
50        }
51    }
52}

54/// What a shelf shows, in order: one deck where its first image or video
55/// is, holding every one of them, then each other asset where it is.
56#[must_use]
57pub fn arrange(shelf: &Shelf) -> Vec<(Card, Vec<&ShelfEntry>)> {
58    let mut groups: Vec<(Card, Vec<&ShelfEntry>)> = Vec::new();
59    let mut deck: Option<usize> = None;
60    for entry in &shelf.entries {
61        let card = Card::of(&entry.block.media_type);
62        match (card, deck) {
63            (Card::Deck, Some(at)) => groups[at].1.push(entry),
64            (Card::Deck, None) => {
65                deck = Some(groups.len());
66                groups.push((card, vec![entry]));
67            }
68            _ => groups.push((card, vec![entry])),
69        }
70    }
71    groups
72}

74/// The shelf as HTML.
75#[must_use]
76pub fn shelf_html(shelf: &Shelf) -> String {
77    let mut html = String::from(
78        "<div class=\"flex flex-col gap-1.5 pt-1\" data-shelf><div class=\"text-[0.6875rem] font-semibold tracking-wider text-ink uppercase dark:text-sand\">From this reply</div><div class=\"flex flex-wrap items-start gap-3.5\" data-shelf-row>",
79    );
80    for (card, entries) in arrange(shelf) {
81        let caption = entries
82            .iter()
83            .find_map(|entry| entry.caption.as_deref())
84            .map(|caption| {
85                format!(
86                    "<p class=\"line-clamp-2 {NOTE}\" data-shelf-caption>{}</p>",
87                    escape(caption)
88                )
89            })
90            .unwrap_or_default();
91        let (width, body) = match card {
92            Card::Deck => (
93                if entries.len() == 1 {
94                    "w-[300px]"
95                } else {
96                    "w-[220px]"
97                },
98                deck_html(&entries),
99            ),
100            Card::Page => ("w-[240px]", page_html(&entries[0].block)),
101            Card::Document => ("w-[240px]", document_html(&entries[0].block)),
102            Card::Audio => ("w-[240px]", audio_html(&entries[0].block)),
103            Card::File => ("w-[240px]", file_html(&entries[0].block)),
104        };
105        html.push_str(&format!(
106            "<div class=\"flex max-w-full {width} flex-col gap-1.5\" data-shelf-item=\"{}\">{body}{caption}</div>",
107            match card {
108                Card::Deck => "deck",
109                Card::Page => "page",
110                Card::Document => "document",
111                Card::Audio => "audio",
112                Card::File => "file",
113            }
114        ));
115    }
116    html.push_str("</div></div>");
117    html
118}

143/// What the type pill of `block` says: its file's extension, else its
144/// media subtype, at most five characters.
145#[must_use]
146pub fn type_label(block: &AssetBlock) -> String {
147    let extension = block
148        .filename
149        .as_deref()
150        .and_then(|name| name.rsplit_once('.'))
151        .map(|(_, extension)| extension)
152        .filter(|extension| !extension.is_empty() && extension.len() <= 5);
153    let label = extension.unwrap_or_else(|| {
154        let subtype = block.media_type.split('/').nth(1).unwrap_or("file");
155        subtype
156            .split(['+', ';', '.', '-'])
157            .next()
158            .unwrap_or(subtype)
159    });
160    label.chars().take(5).collect::<String>().to_uppercase()
161}

163/// The deck: its first three as stacked cards, the first on top, and every
164/// one of them filled in, for the media viewer to step through.
165fn deck_html(entries: &[&ShelfEntry]) -> String {
166    let count = entries.len();
167    let mut cards = String::new();
168    // Drawn back to front: the first shown is the last drawn, on top.
169    for (index, entry) in entries.iter().enumerate().rev() {
170        let layer = if index < 3 {
171            format!(" data-layer=\"{index}\"")
172        } else {
173            " hidden".to_owned()
174        };
175        cards.push_str(&format!(
176            "<div class=\"asset-deck-card absolute inset-0 flex items-center justify-center overflow-hidden rounded-xl border border-toasted bg-cream shadow-md dark:border-ember dark:bg-char\" {} data-index=\"{index}\"{layer}><span class=\"px-3 text-center {NOTE}\" data-asset-body>{}</span></div>",
177            placeholder(&entry.block, "thumb"),
178            escape(&entry.block.title),
179        ));
180    }
181    let what = if entries
182        .iter()
183        .all(|entry| entry.block.media_type.starts_with("video/"))
184    {
185        "videos"
186    } else {
187        "images"
188    };
189    let count_badge = if count > 1 {
190        format!(
191            "<span class=\"pointer-events-none absolute right-2 bottom-2 z-10 rounded-full bg-black/60 px-2 text-[0.6875rem] text-white\" data-deck-count>{count} {what}</span>"
192        )
193    } else {
194        String::new()
195    };
196    let label = if count > 1 {
197        format!("Open the {count} {what}")
198    } else {
199        format!("Open {}", entries[0].block.title)
200    };
201    let height = if count == 1 { "h-[200px]" } else { "h-[162px]" };
202    format!(
203        "<div role=\"button\" tabindex=\"0\" data-action=\"asset-deck\" aria-label=\"{}\" title=\"{}\" class=\"asset-deck relative {height} w-full cursor-zoom-in rounded-xl\" data-deck=\"{count}\">{cards}{count_badge}</div>",
204        escape(&label),
205        escape(&label),
206    )
207}

286/// Every asset on a shelf of `shelves`, by handle, as the newest shelf
287/// shows it.
288#[must_use]
289pub fn shelved_handles<'a>(
290    shelves: impl IntoIterator<Item = &'a Shelf>,
291) -> HashMap<String, AssetBlock> {
292    let mut handles = HashMap::new();
293    for shelf in shelves {
294        for entry in &shelf.entries {
295            handles.insert(entry.block.handle.to_string(), entry.block.clone());
296        }
297    }
298    handles
299}

301const HANDLE_PREFIX: &str = "asset_";
302const HANDLE_LEN: usize = HANDLE_PREFIX.len() + 8;
303
304/// The handle `text` starts with: `asset_` and eight lowercase hex digits,
305/// not followed by more of a word.
306fn handle_at(text: &str) -> Option<&str> {
307    let handle = text.get(..HANDLE_LEN)?;
308    let digits = handle.strip_prefix(HANDLE_PREFIX)?;
309    let hex = digits
310        .bytes()
311        .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte));
312    let ends = text[HANDLE_LEN..]
313        .chars()
314        .next()
315        .is_none_or(|next| !next.is_alphanumeric() && next != '_');
316    (hex && ends).then_some(handle)
317}

319/// `html`, a reply's text, with each handle of `known` it names as a chip:
320/// in its text, or alone in inline code, but not inside a link or a code
321/// block. `None` when it names none.
322#[must_use]
323pub fn with_chips(html: &str, known: &HashMap<String, AssetBlock>) -> Option<String> {
324    if known.is_empty() || !html.contains(HANDLE_PREFIX) {
325        return None;
326    }
327    let mut out = String::with_capacity(html.len());
328    let mut changed = false;
329    // How deep in `<a>`, `<pre>`, and `<code>` the text is.
330    let mut closed = 0usize;
331    let mut rest = html;
332    while !rest.is_empty() {
333        if rest.starts_with('<') {
334            // Text is escaped, so `<` always starts a tag.
335            if let Some(code) = rest.strip_prefix("<code>")
336                && closed == 0
337                && let Some(handle) = handle_at(code)
338                && code[HANDLE_LEN..].starts_with("</code>")
339                && let Some(block) = known.get(handle)
340            {
341                out.push_str(&chip_html(block));
342                rest = &code[HANDLE_LEN + "</code>".len()..];
343                changed = true;
344                continue;
345            }
346            let end = rest.find('>').map_or(rest.len(), |end| end + 1);
347            let tag = &rest[..end];
348            let name = tag
349                .trim_start_matches(['<', '/'])
350                .split(|char: char| char.is_whitespace() || char == '>')
351                .next()
352                .unwrap_or("");
353            if matches!(name, "a" | "pre" | "code") {
354                if tag.starts_with("</") {
355                    closed = closed.saturating_sub(1);
356                } else {
357                    closed += 1;
358                }
359            }
360            out.push_str(tag);
361            rest = &rest[end..];
362            continue;
363        }
364        let end = rest.find('<').unwrap_or(rest.len());
365        let mut text = &rest[..end];
366        rest = &rest[end..];
367        if closed > 0 {
368            out.push_str(text);
369            continue;
370        }
371        while let Some(at) = text.find(HANDLE_PREFIX) {
372            let word_before = text[..at]
373                .chars()
374                .next_back()
375                .is_some_and(|before| before.is_alphanumeric() || before == '_');
376            let block = (!word_before)
377                .then(|| handle_at(&text[at..]))
378                .flatten()
379                .and_then(|handle| known.get(handle));
380            match block {
381                Some(block) => {
382                    out.push_str(&text[..at]);
383                    out.push_str(&chip_html(block));
384                    text = &text[at + HANDLE_LEN..];
385                    changed = true;
386                }
387                None => {
388                    out.push_str(&text[..at + HANDLE_PREFIX.len()]);
389                    text = &text[at + HANDLE_PREFIX.len()..];
390                }
391            }
392        }
393        out.push_str(text);
394    }
395    changed.then_some(out)
396}

490    #[test]
491    fn handles_a_shelf_shows_become_chips_outside_links_and_code_blocks() {
492        let known = shelved_handles([&Shelf {
493            after: None,
494            entries: vec![entry(block("asset_3f9a1c2e", "text/html", None))],
495        }]);
496        let chips = |html: &str| with_chips(html, &known);
497        // Nothing to do: no handle, or one no shelf shows.
498        assert_eq!(chips("<p>plain</p>"), None);
499        assert_eq!(chips("<p>see asset_00000000</p>"), None);
500        // In text and alone in inline code.
501        for html in [
502            "<p>see asset_3f9a1c2e.</p>",
503            "<p>see <code>asset_3f9a1c2e</code>.</p>",
504        ] {
505            let shown = chips(html).expect("a chip");
506            assert_eq!(shown.matches("data-action=\"asset-chip\"").count(), 1);
507            assert!(shown.starts_with("<p>see <span"));
508            assert!(shown.ends_with("</span>.</p>"));
509            assert!(!shown.contains("<code>"));
510        }
511        // Not in a link, a code block, longer code, or a longer word.
512        for html in [
513            "<p><a href=\"x\">asset_3f9a1c2e</a></p>",
514            "<pre><code class=\"language-sh\">asset_3f9a1c2e</code></pre>",
515            "<p><code>read asset_3f9a1c2e</code></p>",
516            "<p>my_asset_3f9a1c2e asset_3f9a1c2ef</p>",
517        ] {
518            assert_eq!(chips(html), None, "{html}");
519        }
520        // Text after a link or a code block is text again.
521        let after = chips("<pre>x</pre><p><a href=\"x\">y</a> asset_3f9a1c2e</p>").expect("a chip");
522        assert!(after.contains("</a> <span"));
523    }
