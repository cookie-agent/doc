14/// Where a block sits, for the actions it offers.
15#[derive(Clone, Copy, Debug)]
16pub struct BlockPlace<'a> {
17    /// The tool call whose display holds the block; `None` for blocks that
18    /// are no call's result, such as an approval's preview.
19    pub call_id: Option<&'a str>,
20    /// The block's index in its display.
21    pub index: usize,
22    /// The block's loaded complete content, when it was cut and asked for.
23    pub full: Option<&'a FullContentState>,
24}
25
26impl BlockPlace<'_> {
27    /// A place with no call and no complete content.
28    pub const fn plain(index: usize) -> Self {
29        Self {
30            call_id: None,
31            index,
32            full: None,
33        }
34    }
35}

40/// A stored file shown inline: a placeholder the page fills with the
41/// artifact's image, video, audio, or download link once it is read.
42pub fn media_html(sha256: &str, media_type: &str, byte_length: u64, label: Option<&str>) -> String {
43    format!(
44        "<div class=\"media\" data-media=\"{}\" data-media-type=\"{}\" data-alt=\"{}\"><span class=\"{NOTE}\">{} ({}, {})</span></div>",
45        escape(sha256),
46        escape(media_type),
47        escape(label.unwrap_or_default()),
48        escape(label.unwrap_or("media")),
49        escape(media_type),
50        escape(&cookie_agent_protocol::human_bytes(byte_length)),
51    )
52}

54/// A plan as a checklist: each step with its status (done, in progress,
55/// pending) and a high or low priority.
56pub fn plan_html(entries: &[cookie_agent_protocol::PlanEntry]) -> String {
57    use cookie_agent_protocol::{PlanEntryPriority, PlanEntryStatus};
58    if entries.is_empty() {
59        return format!("<div class=\"plan {NOTE}\">no steps</div>");
60    }
61    let mut html = String::from("<ul class=\"plan flex flex-col gap-0.5 text-sm\">");
62    for entry in entries {
63        let (icon, state, text) = match entry.status {
64            PlanEntryStatus::Completed => (
65                icondata::FaSquareCheckRegular,
66                "completed",
67                "line-through opacity-70",
68            ),
69            PlanEntryStatus::InProgress => (
70                icondata::FaSquareCaretRightRegular,
71                "in_progress",
72                "font-semibold",
73            ),
74            PlanEntryStatus::Pending | PlanEntryStatus::Unknown => {
75                (icondata::FaSquareRegular, "pending", "")
76            }
77        };
78        let priority = match entry.priority {
79            Some(PlanEntryPriority::High) => {
80                format!(" <span class=\"{NOTE} text-caramel dark:text-maple\">high</span>")
81            }
82            Some(PlanEntryPriority::Low) => format!(" <span class=\"{NOTE}\">low</span>"),
83            Some(PlanEntryPriority::Medium | PlanEntryPriority::Unknown) | None => String::new(),
84        };
85        html.push_str(&format!(
86            "<li class=\"flex items-start gap-2\" data-plan-status=\"{state}\">{}<span class=\"sr-only\">{}:</span><span class=\"min-w-0 break-words {text}\">{}</span>{priority}</li>",
87            crate::icons::svg(icon, "mt-0.5 shrink-0"),
88            state.replace('_', " "),
89            escape(&entry.content),
90        ));
91    }
92    html.push_str("</ul>");
93    html
94}

96/// Every block of a display, in order.
97pub fn display_html<'a>(
98    blocks: &[Open<DisplayBlock>],
99    place: impl Fn(usize) -> BlockPlace<'a>,
100) -> String {
101    let mut html = String::new();
102    for (index, block) in blocks.iter().enumerate() {
103        if let Open::Known(block) = block {
104            html.push_str(&block_html(block, place(index)));
105        }
106    }
107    html
108}

110/// One block.
111pub fn block_html(block: &DisplayBlock, place: BlockPlace<'_>) -> String {
112    let body = match block {
113        DisplayBlock::Text {
114            text,
115            omitted_lines,
116        } => format!(
117            "<pre class=\"{}\">{}</pre>{}",
118            PRE.replace("whitespace-pre ", "whitespace-pre-wrap "),
119            escape(text),
120            omitted_note(*omitted_lines, "lines"),
121        ),
122        DisplayBlock::Markdown {
123            text,
124            omitted_lines,
125        } => format!(
126            "<div class=\"markdown\">{}</div>{}",
127            markdown_html(text),
128            omitted_note(*omitted_lines, "lines")
129        ),
130        DisplayBlock::Code {
131            text,
132            language,
133            start_line,
134            omitted_lines,
135        } => format!(
136            "{}{}",
137            numbered_code(
138                text.lines(),
139                start_line.unwrap_or(1),
140                start_line.is_some(),
141                language.as_deref()
142            ),
143            omitted_note(*omitted_lines, "lines"),
144        ),
145        DisplayBlock::FileExcerpt {
146            path,
147            language,
148            start_line,
149            lines,
150            total_lines,
151            omitted_lines,
152            ..
153        } => {
154            let end = start_line + lines.len().saturating_sub(1) as u64;
155            let range = match total_lines {
156                Some(total) => format!("lines {start_line}–{end} of {total}"),
157                None => format!("lines {start_line}–{end}"),
158            };
159            format!(
160                "<div class=\"{NOTE} font-mono\">{} · {range}</div>{}{}",
161                escape(path),
162                numbered_code(
163                    lines.iter().map(String::as_str),
164                    *start_line,
165                    true,
166                    language.as_deref().or_else(|| extension(path))
167                ),
168                omitted_note(*omitted_lines, "lines"),
169            )
170        }
171        DisplayBlock::Diff(diff) => diff_html(diff),
172        DisplayBlock::Terminal {
173            segments,
174            exit_code,
175            omitted_lines,
176            ..
177        } => terminal_html(segments, *exit_code, *omitted_lines),
178        DisplayBlock::KeyValue {
179            entries,
180            omitted_entries,
181        } => key_value_html(entries, *omitted_entries),
182        DisplayBlock::Table {
183            columns,
184            rows,
185            omitted_rows,
186        } => table_html(columns, rows, *omitted_rows),
187        DisplayBlock::Plan { entries } => plan_html(entries),
188        DisplayBlock::Json { value } => format!(
189            "<pre class=\"{PRE}\"><code class=\"language-json\">{}</code></pre>",
190            escape(&serde_json::to_string_pretty(value).unwrap_or_default())
191        ),
192        DisplayBlock::Media { artifact, alt } => media_html(
193            &artifact.sha256,
194            &artifact.media_type,
195            artifact.byte_length,
196            alt.as_deref().or(artifact.filename.as_deref()),
197        ),
198        // A memory URL opens the memory in the Memory view.
199        DisplayBlock::Link { url, label }
200            if cookie_agent_client_state::tool_header::is_memory_url(url) =>
201        {
202            let label = label
203                .as_deref()
204                .filter(|label| !label.is_empty())
205                .map(|label| {
206                    format!(
207                        "<span class=\"min-w-0 truncate text-ink dark:text-sand\">— {}</span>",
208                        escape(label)
209                    )
210                })
211                .unwrap_or_default();
212            format!(
213                "<button type=\"button\" data-action=\"open-memory\" data-memory-url=\"{url}\" title=\"Open it in Memory\" class=\"inline-flex min-h-11 max-w-full items-center gap-2 rounded-md px-2 text-left text-sm hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke\"><span class=\"shrink-0 font-mono underline decoration-dotted underline-offset-2\">{url}</span>{label}</button>",
214                url = escape(url),
215            )
216        }
217        DisplayBlock::Link { url, label } => {
218            let text = escape(label.as_deref().unwrap_or(url));
219            if safe_link(url) {
220                format!(
221                    "<a href=\"{}\" target=\"_blank\" rel=\"noopener noreferrer\" class=\"text-caramel underline hover:text-espresso dark:text-maple dark:hover:text-latte\">{text}</a>",
222                    escape(url)
223                )
224            } else {
225                format!("<span>{text} ({})</span>", escape(url))
226            }
227        }
228        DisplayBlock::SessionRef {
229            session_id,
230            short_id,
231            label,
232        } => format!(
233            "<button type=\"button\" data-action=\"open-session\" data-session=\"{session_id}\" class=\"rounded border border-crust px-2 py-1 font-mono text-sm hover:bg-toasted active:bg-crust dark:border-smoke dark:hover:bg-ember dark:active:bg-smoke\">{}</button>",
234            escape(
235                label
236                    .as_deref()
237                    .or(short_id.as_deref())
238                    .unwrap_or("open session")
239            )
240        ),
241    };
242    format!(
243        "<div class=\"block my-1\" data-kind=\"{}\">{body}{}</div>",
244        block.kind(),
245        full_content_html(block, place)
246    )
247}

249/// What a cut block with a `full` artifact offers: a button for its complete
250/// content, and that content once loaded.
251fn full_content_html(block: &DisplayBlock, place: BlockPlace<'_>) -> String {
252    let (Some(call_id), Some(_)) = (place.call_id, block_full(block)) else {
253        return String::new();
254    };
255    let action = |label: &str| {
256        format!(
257            "<button type=\"button\" data-action=\"full\" data-call=\"{}\" data-block=\"{}\" class=\"mt-1 min-h-11 rounded border border-crust px-3 py-1 text-sm hover:bg-toasted active:bg-crust dark:border-smoke dark:hover:bg-ember dark:active:bg-smoke\">{label}</button>",
258            escape(call_id),
259            place.index
260        )
261    };
262    match place.full {
263        None => action("Show the complete output"),
264        Some(FullContentState::Loading) => {
265            format!("<div class=\"{NOTE}\">loading the complete output…</div>")
266        }
267        Some(FullContentState::Failed(error)) => format!(
268            "<div class=\"{NOTE}\">could not read the complete output: {}</div>{}",
269            escape(error),
270            action("Try again")
271        ),
272        Some(FullContentState::Loaded { text, truncated }) => format!(
273            "<pre class=\"{PRE}\">{}</pre>{}{}",
274            escape(text),
275            if *truncated {
276                format!("<div class=\"{NOTE}\">only the first part is shown</div>")
277            } else {
278                String::new()
279            },
280            action("Hide the complete output")
281        ),
282    }
283}

288        _ => String::new(),

292fn extension(path: &str) -> Option<&str> {
293    path.rsplit_once('.')
294        .map(|(_, extension)| extension)
295        .filter(|extension| !extension.contains('/'))
296}

298/// Code lines in one highlightable `<code>`, behind a gutter of line
299/// numbers from `first` when `numbered`.
300fn numbered_code<'a>(
301    lines: impl Iterator<Item = &'a str>,
302    first: u64,
303    numbered: bool,
304    language: Option<&str>,
305) -> String {
306    let lines = lines.collect::<Vec<_>>();
307    let class = language
308        .map(|language| {
309            format!(
310                " class=\"language-{}\"",
311                escape(&language.to_ascii_lowercase())
312            )
313        })
314        .unwrap_or_default();
315    let code = escape(&lines.join("\n"));
316    if !numbered {
317        return format!("<pre class=\"{PRE}\"><code{class}>{code}</code></pre>");
318    }
319    let gutter = (0..lines.len() as u64)
320        .map(|offset| (first + offset).to_string())
321        .collect::<Vec<_>>()
322        .join("\n");
323    format!(
324        "<div class=\"flex overflow-x-auto rounded border border-toasted bg-cream font-mono text-sm dark:border-ember dark:bg-night\"><pre class=\"select-none border-r border-toasted p-2 text-right text-ink dark:border-ember dark:text-sand\">{gutter}</pre><pre class=\"flex-1 whitespace-pre p-2\"><code{class}>{code}</code></pre></div>"
325    )
326}

328/// A diff with the file's real line numbers: old on the left, new on the
329/// right, and the hunks' context as the producer kept it.
330pub fn diff_html(diff: &DiffBlock) -> String {
331    let change = match diff.change {
332        DiffChange::Created => "created",
333        DiffChange::Deleted => "deleted",
334        DiffChange::Renamed => "renamed",
335        DiffChange::Modified | DiffChange::Unknown => "modified",
336    };
337    let path = match &diff.old_path {
338        Some(old) => format!("{} → {}", escape(old), escape(&diff.path)),
339        None => escape(&diff.path),
340    };
341    let mut html = format!(
342        "<div class=\"{NOTE} font-mono\">{path} · {change} · <span class=\"text-green-700 dark:text-green-400\">+{}</span> <span class=\"text-red-700 dark:text-red-400\">−{}</span></div>",
343        diff.added, diff.removed
344    );
345    if diff.hunks.is_empty() {
346        html.push_str(&format!(
347            "<div class=\"{NOTE}\">no line changes to show</div>"
348        ));
349    }
350    html.push_str("<div class=\"overflow-x-auto rounded border border-toasted font-mono text-sm dark:border-ember\"><table class=\"diff w-full border-collapse\">");
351    for hunk in &diff.hunks {
352        html.push_str(&format!(
353            "<tr class=\"bg-parchment text-ink dark:bg-char dark:text-sand\"><td colspan=\"3\" class=\"px-2\">@@ -{},{} +{},{} @@</td></tr>",
354            hunk.old_start, hunk.old_lines, hunk.new_start, hunk.new_lines
355        ));
356        let (mut old, mut new) = (hunk.old_start, hunk.new_start);
357        for line in &hunk.lines {
358            let (old_number, new_number, sign, class) = match line.op {
359                DiffLineOp::Context => {
360                    let numbers = (Some(old), Some(new));
361                    old += 1;
362                    new += 1;
363                    (numbers.0, numbers.1, ' ', "")
364                }
365                DiffLineOp::Remove => {
366                    old += 1;
367                    (
368                        Some(old - 1),
369                        None,
370                        '-',
371                        " class=\"bg-red-100 dark:bg-red-950\"",
372                    )
373                }
374                DiffLineOp::Add => {
375                    new += 1;
376                    (
377                        None,
378                        Some(new - 1),
379                        '+',
380                        " class=\"bg-green-100 dark:bg-green-950\"",
381                    )
382                }
383            };
384            let number =
385                |number: Option<u64>| number.map(|number| number.to_string()).unwrap_or_default();
386            html.push_str(&format!(
387                "<tr{class}><td class=\"select-none px-2 text-right text-ink dark:text-sand\">{}</td><td class=\"select-none px-2 text-right text-ink dark:text-sand\">{}</td><td class=\"whitespace-pre px-2\">{sign}{}</td></tr>",
388                number(old_number),
389                number(new_number),
390                escape(&line.text)
391            ));
392        }
393    }
394    html.push_str("</table></div>");
395    if let Some(omitted) = diff.omitted {
396        html.push_str(&format!(
397            "<div class=\"{NOTE}\">… {} hunks, {} lines omitted</div>",
398            omitted.hunks, omitted.lines
399        ));
400    }
401    html
402}

404fn terminal_html(
405    segments: &[TerminalSegment],
406    exit_code: Option<i32>,
407    omitted_lines: Option<u64>,
408) -> String {
409    let mut out = format!("<pre class=\"terminal {PRE}\">");
410    let mut stdout = sgr::Rendition::default();
411    let mut stderr = sgr::Rendition::default();
412    for segment in segments {
413        match segment.stream {
414            TerminalStream::Stderr => {
415                out.push_str("<span class=\"stderr\">");
416                sgr::push_sgr_html(&mut out, &segment.text, &mut stderr);
417                out.push_str("</span>");
418            }
419            TerminalStream::Stdout | TerminalStream::Unknown => {
420                sgr::push_sgr_html(&mut out, &segment.text, &mut stdout);
421            }
422        }
423    }
424    out.push_str("</pre>");
425    out.push_str(&omitted_note(omitted_lines, "lines"));
426    if let Some(code) = exit_code {
427        out.push_str(&format!("<div class=\"{NOTE}\">exit code {code}</div>"));
428    }
429    out
430}

432fn key_value_html(entries: &[KeyValueEntry], omitted: Option<u64>) -> String {
433    let mut html = String::from("<dl class=\"grid grid-cols-[auto_1fr] gap-x-3 text-sm\">");
434    for entry in entries {
435        html.push_str("<dt class=\"font-bold\">");
436        push_escaped(&mut html, &entry.key);
437        html.push_str("</dt><dd class=\"break-all\">");
438        push_escaped(&mut html, &entry.value);
439        html.push_str("</dd>");
440    }
441    html.push_str("</dl>");
442    html.push_str(&omitted_note(omitted, "entries"));
443    html
444}

446fn table_html(columns: &[String], rows: &[Vec<String>], omitted: Option<u64>) -> String {
447    let mut html = String::from(
448        "<div class=\"overflow-x-auto\"><table class=\"border-collapse text-sm\"><thead><tr>",
449    );
450    for column in columns {
451        html.push_str("<th class=\"border border-toasted px-2 text-left dark:border-ember\">");
452        push_escaped(&mut html, column);
453        html.push_str("</th>");
454    }
455    html.push_str("</tr></thead><tbody>");
456    for row in rows {
457        html.push_str("<tr>");
458        for cell in row {
459            html.push_str("<td class=\"border border-toasted px-2 dark:border-ember\">");
460            push_escaped(&mut html, cell);
461            html.push_str("</td>");
462        }
463        html.push_str("</tr>");
464    }
465    html.push_str("</tbody></table></div>");
466    html.push_str(&omitted_note(omitted, "rows"));
467    html
468}
