1//! The connect screen's decisions, independent of the browser (spec 7.2,
2//! 7.3): the daemon URL a page fills in from its own address, the token a
3//! `#token=` fragment carries, whether to connect without a click, and the
4//! hint for a connection refused before its handshake.
5
6use url::Url;
7
8/// The `localStorage` key of the daemon URL the user last connected to.
9pub const DAEMON_URL_KEY: &str = "cookie.daemon_url";
10
11/// The `localStorage` key of the theme preference.
12pub const THEME_KEY: &str = "cookie.theme";
13
14/// The `sessionStorage` key of this tab's client identity: each tab is its
15/// own client, kept across reloads and reconnects.
16pub const IDENTITY_KEY: &str = "cookie.identity";
17
18/// The `localStorage` key of the token stored for the daemon at
19/// `daemon_url`.
20pub fn token_key(daemon_url: &str) -> String {
21    format!("cookie.token:{daemon_url}")
22}
23
24/// The daemon URL a page at `page` (its `location.href`) fills in: the
25/// page's scheme as a WebSocket scheme (`http`→`ws`, `https`→`wss`), its
26/// host and port, and `ws` in the page's directory. `None` for a page that
27/// is not served over HTTP, such as a `file:` page.
28pub fn daemon_url_for_page(page: &str) -> Option<String> {
29    let page = Url::parse(page).ok()?;
30    let scheme = match page.scheme() {
31        "http" => "ws",
32        "https" => "wss",
33        _ => return None,
34    };
35    let host = page.host_str()?;
36    let authority = match page.port() {
37        Some(port) => format!("{host}:{port}"),
38        None => host.to_owned(),
39    };
40    let path = page.path();
41    let directory = &path[..=path.rfind('/').unwrap_or(0)];
42    let directory = if directory.is_empty() { "/" } else { directory };
43    Some(format!("{scheme}://{authority}{directory}ws"))
44}

46/// The token a page's `location.hash` carries as `#token=<token>`, among
47/// other `&`-separated parameters; `None` without a non-empty one.
48pub fn token_from_fragment(fragment: &str) -> Option<String> {
49    let fragment = fragment.strip_prefix('#').unwrap_or(fragment);
50    url::form_urlencoded::parse(fragment.as_bytes())
51        .find(|(name, _)| name == "token")
52        .map(|(_, value)| value.trim().to_owned())
53        .filter(|token| !token.is_empty())
54}
55
56/// The fragment left after taking the token out: the other parameters, in
57/// their order, or nothing.
58pub fn fragment_without_token(fragment: &str) -> String {
59    let fragment = fragment.strip_prefix('#').unwrap_or(fragment);
60    let rest = url::form_urlencoded::parse(fragment.as_bytes())
61        .filter(|(name, _)| name != "token")
62        .collect::<Vec<_>>();
63    if rest.is_empty() {
64        return String::new();
65    }
66    let mut serializer = url::form_urlencoded::Serializer::for_suffix(String::from("#"), 1);
67    for (name, value) in &rest {
68        serializer.append_pair(name, value);
69    }
70    serializer.finish()
71}
72
73/// Whether the page connects without a click: it has a daemon URL (filled
74/// in or remembered) and a token (from the fragment or stored).
75pub fn connects_by_itself(daemon_url: Option<&str>, token: Option<&str>) -> bool {
76    daemon_url.is_some_and(|url| !url.trim().is_empty())
77        && token.is_some_and(|token| !token.trim().is_empty())
78}
79
80/// What the connect screen says when the daemon closed the connection before
81/// its handshake. A browser shows the page no reason, and the usual one is
82/// an origin the daemon does not allow, so it names the entry to add.
83pub fn refused_before_handshake_hint(page_origin: &str) -> String {
84    format!(
85        "The daemon closed the connection before its handshake. The likely cause: \
86         this page's origin is not allowed. Add \"{page_origin}\" to [server] \
87         allowed_origins in the daemon's config.toml and restart it."
88    )
89}

91#[cfg(test)]
92mod tests {
93    use super::*;
94
95    #[test]
96    fn the_daemon_url_follows_the_page_scheme_host_and_directory() {
97        for (page, daemon) in [
98            ("http://127.0.0.1:7419/", "ws://127.0.0.1:7419/ws"),
99            ("http://127.0.0.1:7419/index.html", "ws://127.0.0.1:7419/ws"),
100            ("https://abc.efg/", "wss://abc.efg/ws"),
101            ("https://abc.efg/cookie/", "wss://abc.efg/cookie/ws"),
102            (
103                "https://abc.efg/cookie/index.html?x=1#token=abc",
104                "wss://abc.efg/cookie/ws",
105            ),
106            ("http://[::1]:8080/a/b/", "ws://[::1]:8080/a/b/ws"),
107        ] {
108            assert_eq!(daemon_url_for_page(page).as_deref(), Some(daemon), "{page}");
109        }
110        assert_eq!(daemon_url_for_page("file:///home/me/index.html"), None);
111        assert_eq!(daemon_url_for_page("not a url"), None);
112    }
113
114    #[test]
115    fn the_token_comes_from_the_fragment_and_leaves_the_rest() {
116        assert_eq!(
117            token_from_fragment("#token=abc_-1").as_deref(),
118            Some("abc_-1")
119        );
120        assert_eq!(
121            token_from_fragment("#view=x&token=abc").as_deref(),
122            Some("abc")
123        );
124        assert_eq!(token_from_fragment("#token="), None);
125        assert_eq!(token_from_fragment(""), None);
126        assert_eq!(token_from_fragment("#other=1"), None);
127        assert_eq!(fragment_without_token("#token=abc"), "");
128        assert_eq!(fragment_without_token("#view=x&token=abc"), "#view=x");
129    }
130
131    #[test]
132    fn the_page_connects_by_itself_only_with_a_url_and_a_token() {
133        assert!(connects_by_itself(Some("ws://h/ws"), Some("t")));
134        assert!(!connects_by_itself(Some("ws://h/ws"), None));
135        assert!(!connects_by_itself(None, Some("t")));
136        assert!(!connects_by_itself(Some(" "), Some("t")));
137        assert!(!connects_by_itself(Some("ws://h/ws"), Some("")));
138    }
139
140    #[test]
141    fn the_hint_quotes_the_page_origin_exactly() {
142        let hint = refused_before_handshake_hint("http://192.168.1.5:7419");
143        assert!(hint.contains("\"http://192.168.1.5:7419\""), "{hint}");
144        assert!(hint.contains("allowed_origins"), "{hint}");
145        assert_eq!(token_key("ws://h/ws"), "cookie.token:ws://h/ws");
146    }
