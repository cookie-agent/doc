1//! The open session's ID, small and quiet, which copies on a click: in the
2//! title bar after the title and beside the session panel's heading, and,
3//! for a subagent, its handle beside it. Other sessions and the CLI reach a
4//! root session by its full ID; agents in a tree message a subagent by its
5//! handle.
6
7use cookie_agent_protocol::{SessionId, SessionOrigin};
8use leptos::prelude::*;
9
10use crate::{icons::Glyph, workspace::Workspace};

12/// The open session's ID, and its handle when it is a subagent's.
13pub(super) fn open_ids(workspace: Workspace) -> Option<(SessionId, Option<String>)> {
14    if workspace.new_session.get() {
15        return None;
16    }
17    let id = workspace.selected.get()?;
18    let handle = workspace.tree.with(|tree| {
19        let tree = tree.as_ref()?;
20        let open = *crate::tree::path(tree, id).last()?;
21        matches!(open.origin, SessionOrigin::Delegated { .. })
22            .then(|| open.short_id.clone())
23            .flatten()
24    });
25    Some((id, handle))
26}

28/// Copies `text`, saying `what` was copied.
29fn copy(workspace: Workspace, what: &'static str, text: String, done: RwSignal<bool>) {
30    leptos::task::spawn_local(async move {
31        if crate::browser::copy_text(&text).await {
32            workspace.status.set(format!("{what} copied."));
33            done.set(true);
34            set_timeout(
35                move || done.set(false),
36                std::time::Duration::from_millis(1400),
37            );
38        } else {
39            workspace.status.set(
40                "This browser does not let the page copy; select the text instead.".to_owned(),
41            );
42        }
43    });
44}

46/// One value that copies on a click: mono, muted, a copy icon on hover, and
47/// a check for a moment once copied. A long one gives way in its middle
48/// (`01a1e57f…dfaf8acc`), its ends kept. `display` is its `display` class,
49/// which may show it only where it fits.
50fn copy_chip(
51    workspace: Workspace,
52    what: &'static str,
53    value: String,
54    attr: &'static str,
55    display: &'static str,
56) -> impl IntoView {
57    let done = RwSignal::new(false);
58    // The ends stay; the middle shrinks to an ellipsis first.
59    let chars: Vec<char> = value.chars().collect();
60    let (head, middle, tail) = if chars.len() > 20 {
61        (
62            chars[..8].iter().collect::<String>(),
63            chars[8..chars.len() - 8].iter().collect::<String>(),
64            chars[chars.len() - 8..].iter().collect::<String>(),
65        )
66    } else {
67        (value.clone(), String::new(), String::new())
68    };
69    let copied = value.clone();
70    view! {
71        <button
72            type="button"
73            class=format!("group/id hit-area {display} h-6 min-w-0 shrink items-center gap-1 rounded-md px-1.5 font-mono text-[0.6875rem] text-muted transition-colors hover:bg-hover hover:text-fg")
74            class:text-ok=move || done.get()
75            title=format!("Click to copy the {}: {value}", what.to_lowercase())
76            aria-label=format!("Copy the {}", what.to_lowercase())
77            data-copy-id=attr
78            data-copied=move || done.get().to_string()
79            on:click=move |_| copy(workspace, what, copied.clone(), done)
80        >
81            <span class=move || if done.get() {
82                "inline-flex shrink-0"
83            } else {
84                "inline-flex shrink-0 opacity-0 transition-opacity group-hover/id:opacity-100 group-focus-visible/id:opacity-100"
85            }>
86                <Glyph
87                    icon=Signal::derive(move || if done.get() { icondata::LuCheck } else { icondata::LuCopy })
88                    class="text-[0.75rem]"
89                />
90            </span>
91            <span class="flex min-w-0">
92                <span class="shrink-0">{head}</span>
93                <span class="min-w-[1ch] shrink truncate">{middle}</span>
94                <span class="shrink-0">{tail}</span>
95            </span>
96        </button>
97    }
98}

100/// Where an ID chip shows inside a [`fitted_session_ids`] slot: from the
101/// slot width its narrowest form takes (17 characters of 11 px JetBrains
102/// Mono, its padding, and its icon: about 8.8rem), and the handle beside it
103/// from the width both take with a handle of up to 20 characters (about
104/// 18.9rem). The ID gives way before the handle does not, so a slot with
105/// room for one keeps the ID.
106const ID_FITS: &str = "hidden @min-[9rem]:inline-flex";
107const HANDLE_FITS: &str = "hidden @min-[19rem]:inline-flex";
108
109/// The open session's ID (and a subagent's handle before it), each copying
110/// itself; nothing while no session is open. `class` places the group.
111pub(super) fn session_ids(workspace: Workspace, class: &'static str) -> impl IntoView {
112    ids_group(workspace, class, "inline-flex", "inline-flex")
113}
114
115/// [`session_ids`] in a slot of the room a row leaves it (a title bar's,
116/// after the title): each chip shows only where it fits in its narrowest
117/// form, and the slot clips, so the IDs never run under what follows them,
118/// however crowded the row. `class` places the slot.
119pub(super) fn fitted_session_ids(workspace: Workspace, class: &'static str) -> impl IntoView {
120    view! {
121        // A size container: its width comes from the row, not the IDs, and
122        // the chips ask it whether they fit. The vertical padding keeps the
123        // chips' 44 px touch inside what it clips.
124        <div class=format!("@container -my-2.5 flex min-w-0 flex-1 basis-0 items-center overflow-hidden py-2.5 {class}")>
125            {ids_group(workspace, "", HANDLE_FITS, ID_FITS)}
126        </div>
127    }
128}
129
130fn ids_group(
131    workspace: Workspace,
132    class: &'static str,
133    handle_display: &'static str,
134    id_display: &'static str,
135) -> impl IntoView {
136    // Drawn again only when the ID or the handle changes, not on every
137    // change of the tree a running session makes.
138    let ids = Memo::new(move |_| open_ids(workspace));
139    move || {
140        ids.get().map(|(id, handle)| {
141            view! {
142                <span class=format!("flex min-w-0 shrink-[4] items-center gap-0.5 {class}") data-session-ids>
143                    {handle.map(|handle| copy_chip(workspace, "Handle", handle, "handle", handle_display))}
144                    {copy_chip(workspace, "Session ID", id.to_string(), "session", id_display)}
145                </span>
146            }
147        })
148    }
