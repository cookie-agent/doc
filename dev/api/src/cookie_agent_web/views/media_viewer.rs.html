1//! The media viewer: an image or video from the transcript over the whole
2//! page, to inspect it. The wheel, a pinch, the zoom buttons, and `+`, `-`,
3//! `0` zoom; a drag moves the zoomed file; a double click zooms in, or back
4//! to fit. A video plays with its own bar: play and pause (Space or `k`),
5//! a frame back or forward (`,` and `.`), five seconds back or forward
6//! (Left and Right), a seek bar, its speed, and its sound. The bar on top
7//! names the file and its size in pixels. Escape, ✕, or a click beside the
8//! file closes it. A shelf's deck opens as a group the viewer steps
9//! through: its arrows, Left and Right on an image, and its filmstrip.
10

22const BAR_BUTTON: &str = "inline-flex min-h-11 min-w-11 shrink-0 items-center justify-center gap-1 rounded-md px-2 text-sm hover:bg-white/15 active:bg-white/25 disabled:opacity-40";

24/// The step of the frame buttons: a frame at 30 frames a second, since a
25/// browser does not tell a video's frame rate.
26const FRAME: f64 = 1.0 / 30.0;
27
28/// The speeds a video plays at.
29const SPEEDS: [f64; 6] = [0.1, 0.25, 0.5, 1.0, 1.5, 2.0];

31/// The viewer, open while `workspace.media_viewer` holds a file.
32#[component]
33pub(crate) fn MediaViewer(workspace: Workspace) -> impl IntoView {
34    // Back closes the viewer before anything under it, however many items
35    // it stepped through.
36    Effect::new(move |was: Option<bool>| {
37        let open = workspace.media_viewer.with(Option::is_some);
38        if open && was != Some(true) {
39            workspace.open_transient(Callback::new(move |()| workspace.media_viewer.set(None)));
40        } else if !open && was == Some(true) {
41            workspace.transient_closed();
42        }
43        open
44    });
45    move || {
46        workspace
47            .media_viewer
48            .get()
49            .map(|viewing| view! { <Viewer workspace viewing /> })
50    }
51}

53#[component]
54fn Viewer(workspace: Workspace, viewing: Viewing) -> impl IntoView {
55    let close = move || workspace.media_viewer.set(None);
56    // The deck it steps through, and where it is in it.
57    let group = workspace.media_group.get_untracked();
58    let at = group.iter().position(|shown| *shown == viewing);
59    let stepping = group.len() > 1 && at.is_some();
60    let count = group.len();
61    let go = move |to: usize| {
62        let Some(next) = workspace
63            .media_group
64            .with_untracked(|group| group.get(to).cloned())
65        else {
66            return;
67        };
68        workspace.media_viewer.set(Some(next));
69        // The viewer drawn anew takes the focus, for its keys.
70        request_animation_frame(|| {
71            if let Some(viewer) = web_sys::window()
72                .and_then(|window| window.document())
73                .and_then(|document| document.get_element_by_id("media-viewer"))
74                .and_then(|viewer| viewer.dyn_into::<web_sys::HtmlElement>().ok())
75            {
76                let _ = viewer.focus();
77            }
78        });
79    };
80    let turn = move |by: isize| {
81        if let Some(at) = at.filter(|_| stepping) {
82            go((at as isize + by).rem_euclid(count as isize) as usize);
83        }
84    };
85    let zoom = RwSignal::new(Zoom::default());
86    let root = NodeRef::<html::Div>::new();
87    let stage = NodeRef::<html::Div>::new();
88    let video = NodeRef::<html::Video>::new();
89    let is_video = viewing.is_video();
90    let title = viewing.title().to_owned();
91    let label = title.clone();
92    let download = if viewing.alt.is_empty() {
93        "media".to_owned()
94    } else {
95        viewing.alt.clone()
96    };
97
98    // The focus goes to the viewer, for its keys, and back where it was
99    // once the viewer closes.
100    let before = StoredValue::new_local(
101        web_sys::window()
102            .and_then(|window| window.document())
103            .and_then(|document| document.active_element()),
104    );
105    Effect::new(move |_| {
106        if let Some(root) = root.get() {
107            let _ = root.focus();
108        }
109    });
110    on_cleanup(move || {
111        if let Some(Some(before)) = before.try_get_value()
112            && before.is_connected()
113            && let Ok(before) = before.dyn_into::<web_sys::HtmlElement>()
114        {
115            let _ = before.focus();
116        }
117    });
118
119    // Where a point of the page is, from the stage's centre.
120    let from_centre = move |client_x: i32, client_y: i32| {
121        stage.get_untracked().map_or((0.0, 0.0), |stage| {
122            let rect = stage.get_bounding_client_rect();
123            (
124                f64::from(client_x) - rect.left() - rect.width() / 2.0,
125                f64::from(client_y) - rect.top() - rect.height() / 2.0,
126            )
127        })
128    };
129
130    // The wheel zooms. The page gets it through a listener that is not
131    // passive, so that the browser's own zoom (Ctrl and the wheel, or a
132    // trackpad's pinch) does not zoom the page instead.
133    let wheel = StoredValue::new_local(None::<Closure<dyn Fn(web_sys::WheelEvent)>>);
134    Effect::new(move |_| {
135        let Some(element) = stage.get() else {
136            return;
137        };
138        let listener =
139            Closure::<dyn Fn(web_sys::WheelEvent)>::new(move |event: web_sys::WheelEvent| {
140                event.prevent_default();
141                let delta = match event.delta_mode() {
142                    web_sys::WheelEvent::DOM_DELTA_LINE => event.delta_y() * 16.0,
143                    web_sys::WheelEvent::DOM_DELTA_PAGE => event.delta_y() * 400.0,
144                    _ => event.delta_y(),
145                };
146                // A pinch arrives as small steps with Ctrl held.
147                let rate = if event.ctrl_key() { 0.01 } else { 0.002 };
148                let (x, y) = from_centre(event.client_x(), event.client_y());
149                zoom.update(|zoom| *zoom = zoom.zoom_at((-delta * rate).exp(), x, y));
150            });
151        let options = web_sys::AddEventListenerOptions::new();
152        options.set_passive(false);
153        let _ = element.add_event_listener_with_callback_and_add_event_listener_options(
154            "wheel",
155            listener.as_ref().unchecked_ref(),
156            &options,
157        );
158        wheel.set_value(Some(listener));
159    });
160
161    // The pointers down on the stage, by ID: one drags, two pinch.
162    let pointers = StoredValue::new(HashMap::<i32, (f64, f64)>::new());
163    // How far the pointers moved since the press, to tell a drag from a
164    // click, and whether the press was on the file.
165    let travel = StoredValue::new(0.0_f64);
166    let on_file = StoredValue::new(false);
167    let pressed_on_file = |event: &ev::PointerEvent| {
168        event
169            .target()
170            .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
171            .and_then(|target| target.closest("[data-viewer-file]").ok().flatten())
172            .is_some()
173    };
174    let on_pointerdown = move |event: ev::PointerEvent| {
175        if event.pointer_type() == "mouse" && event.button() != 0 {
176            return;
177        }
178        if event.pointer_type() == "mouse" {
179            // No text or image selection under a drag.
180            event.prevent_default();
181        }
182        if pointers.with_value(HashMap::is_empty) {
183            travel.set_value(0.0);
184            on_file.set_value(pressed_on_file(&event));
185        }
186        if let Some(stage) = stage.get_untracked() {
187            let _ = stage.set_pointer_capture(event.pointer_id());
188        }
189        pointers.update_value(|pointers| {
190            pointers.insert(
191                event.pointer_id(),
192                (f64::from(event.client_x()), f64::from(event.client_y())),
193            );
194        });
195    };
196    let on_pointermove = move |event: ev::PointerEvent| {
197        let id = event.pointer_id();
198        let now = (f64::from(event.client_x()), f64::from(event.client_y()));
199        let Some(was) = pointers.with_value(|pointers| pointers.get(&id).copied()) else {
200            return;
201        };
202        let other = pointers.with_value(|pointers| {
203            pointers
204                .iter()
205                .find(|(other, _)| **other != id)
206                .map(|(_, at)| *at)
207        });
208        pointers.update_value(|pointers| {
209            pointers.insert(id, now);
210        });
211        travel.update_value(|travel| *travel += (now.0 - was.0).hypot(now.1 - was.1));
212        match other {
213            // Two fingers: the file follows their midpoint and zooms with
214            // their spread.
215            Some(other) => {
216                let before = was.0 - other.0;
217                let before = before.hypot(was.1 - other.1);
218                let after = (now.0 - other.0).hypot(now.1 - other.1);
219                let mid = |a: (f64, f64)| ((a.0 + other.0) / 2.0, (a.1 + other.1) / 2.0);
220                let (from, to) = (mid(was), mid(now));
221                let (x, y) = from_centre(to.0 as i32, to.1 as i32);
222                zoom.update(|zoom| {
223                    let panned = zoom.panned(to.0 - from.0, to.1 - from.1);
224                    *zoom = if before > 0.0 {
225                        panned.zoom_at(after / before, x, y)
226                    } else {
227                        panned
228                    };
229                });
230            }
231            None => zoom.update(|zoom| *zoom = zoom.panned(now.0 - was.0, now.1 - was.1)),
232        }
233    };
234    let on_pointerup = move |event: ev::PointerEvent| {
235        pointers.update_value(|pointers| {
236            pointers.remove(&event.pointer_id());
237        });
238    };
239
240    // A click on a video plays or pauses it, a click beside the file
241    // closes the viewer; a drag does neither.
242    let toggle_play = move || {
243        if let Some(video) = video.get_untracked() {
244            if video.paused() {
245                let _ = video.play();
246            } else {
247                let _ = video.pause();
248            }
249        }
250    };
251    let on_click = move |_: ev::MouseEvent| {
252        if travel.get_value() > 5.0 {
253            return;
254        }
255        if !on_file.get_value() {
256            close();
257        } else if is_video {
258            toggle_play();
259        }
260    };
261    let on_dblclick = move |event: ev::MouseEvent| {
262        if !on_file.get_value() {
263            return;
264        }
265        let (x, y) = from_centre(event.client_x(), event.client_y());
266        zoom.update(|zoom| {
267            *zoom = if *zoom == Zoom::default() {
268                zoom.zoom_at(2.5, x, y)
269            } else {
270                Zoom::default()
271            };
272        });
273    };
274    let zoom_by = move |factor: f64| zoom.update(|zoom| *zoom = zoom.zoom_at(factor, 0.0, 0.0));
275
276    // The video's state, as its events tell it.
277    let playing = RwSignal::new(false);
278    let position = RwSignal::new(0.0_f64);
279    let duration = RwSignal::new(0.0_f64);
280    let speed = RwSignal::new(1.0_f64);
281    let muted = RwSignal::new(false);
282    let seek = move |to: f64| {
283        if let Some(video) = video.get_untracked() {
284            let end = video.duration();
285            let to = if end.is_finite() {
286                to.clamp(0.0, end)
287            } else {
288                to.max(0.0)
289            };
290            video.set_current_time(to);
291            position.set(to);
292        }
293    };
294    let step = move |by: f64| {
295        if let Some(video) = video.get_untracked() {
296            let _ = video.pause();
297            seek(video.current_time() + by);
298        }
299    };
300
301    let on_keydown = move |event: ev::KeyboardEvent| {
302        if event.ctrl_key() || event.meta_key() || event.alt_key() {
303            return;
304        }
305        let in_field = event
306            .target()
307            .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
308            .is_some_and(|target| target.tag_name() == "SELECT");
309        let handled = match event.key().as_str() {
310            "Escape" => {
311                close();
312                true
313            }
314            "+" | "=" => {
315                zoom_by(1.25);
316                true
317            }
318            "-" | "_" => {
319                zoom_by(0.8);
320                true
321            }
322            "0" => {
323                zoom.set(Zoom::default());
324                true
325            }
326            " " | "k" if is_video && !in_field => {
327                toggle_play();
328                true
329            }
330            "," if is_video => {
331                step(-FRAME);
332                true
333            }
334            "." if is_video => {
335                step(FRAME);
336                true
337            }
338            "ArrowLeft" if stepping && !is_video => {
339                turn(-1);
340                true
341            }
342            "ArrowRight" if stepping && !is_video => {
343                turn(1);
344                true
345            }
346            "ArrowLeft" if is_video && !in_field => {
347                seek(position.get_untracked() - 5.0);
348                true
349            }
350            "ArrowRight" if is_video && !in_field => {
351                seek(position.get_untracked() + 5.0);
352                true
353            }
354            _ => false,
355        };
356        if handled {
357            event.prevent_default();
358            event.stop_propagation();
359        }
360    };
361
362    // The file's own size in pixels, and how large its pixels show when it
363    // fits the window, measured once it loads and whenever the window
364    // resizes. Pixels shown three times their size or more show crisp
365    // rather than blurred.
366    let size = RwSignal::new(None::<(u32, u32)>);
367    let fitted = RwSignal::new(1.0_f64);
368    let measure = move || {
369        let Some(element) = web_sys::window()
370            .and_then(|window| window.document())
371            .and_then(|document| document.get_element_by_id("media-viewer-file"))
372        else {
373            return;
374        };
375        let natural = if let Some(image) = element.dyn_ref::<web_sys::HtmlImageElement>() {
376            (image.natural_width(), image.natural_height())
377        } else if let Some(video) = element.dyn_ref::<web_sys::HtmlVideoElement>() {
378            (video.video_width(), video.video_height())
379        } else {
380            return;
381        };
382        if natural.0 == 0 {
383            return;
384        }
385        size.set(Some(natural));
386        if let Some(element) = element.dyn_ref::<web_sys::HtmlElement>() {
387            fitted.set(f64::from(element.offset_width()) / f64::from(natural.0));
388        }
389    };
390    let resized = window_event_listener(ev::resize, move |_| measure());
391    on_cleanup(move || resized.remove());
392    let file_class = move || {
393        format!(
394            "max-h-full max-w-full select-none object-contain will-change-transform {} {}",
395            if is_video && zoom.get() == Zoom::default() {
396                "cursor-pointer"
397            } else {
398                "cursor-grab active:cursor-grabbing"
399            },
400            if zoom.get().scale * fitted.get() >= 3.0 {
401                "[image-rendering:pixelated]"
402            } else {
403                ""
404            }
405        )
406    };
407    let transform = move || zoom.get().transform();
408    let url = viewing.url.clone();
409    let file = if is_video {
410        view! {
411            <video
412                node_ref=video
413                id="media-viewer-file"
414                data-viewer-file
415                src=url
416                autoplay
417                playsinline
418                class=file_class
419                style:transform=transform
420                on:loadedmetadata=move |_| {
421                    if let Some(video) = video.get_untracked() {
422                        duration.set(video.duration());
423                    }
424                    measure();
425                }
426                on:durationchange=move |_| {
427                    if let Some(video) = video.get_untracked() {
428                        duration.set(video.duration());
429                    }
430                }
431                on:timeupdate=move |_| {
432                    if let Some(video) = video.get_untracked() {
433                        position.set(video.current_time());
434                    }
435                }
436                on:seeked=move |_| {
437                    if let Some(video) = video.get_untracked() {
438                        position.set(video.current_time());
439                    }
440                }
441                on:play=move |_| playing.set(true)
442                on:pause=move |_| playing.set(false)
443                on:ended=move |_| playing.set(false)
444                on:ratechange=move |_| {
445                    if let Some(video) = video.get_untracked() {
446                        speed.set(video.playback_rate());
447                    }
448                }
449                on:volumechange=move |_| {
450                    if let Some(video) = video.get_untracked() {
451                        muted.set(video.muted());
452                    }
453                }
454            ></video>
455        }
456        .into_any()
457    } else {
458        view! {
459            <img
460                id="media-viewer-file"
461                data-viewer-file
462                src=url
463                alt=viewing.alt.clone()
464                draggable="false"
465                class=file_class
466                style:transform=transform
467                on:load=move |_| measure()
468            />
469        }
470        .into_any()
471    };
472
473    let controls = is_video.then(|| {
474        let known = move || duration.get().is_finite() && duration.get() > 0.0;
475        view! {
476            <div
477                id="media-viewer-controls"
478                class="flex shrink-0 flex-wrap items-center gap-1 px-2 pt-1 pb-[max(0.25rem,env(safe-area-inset-bottom))]"
479            >
480                <button
481                    type="button"
482                    id="media-viewer-play"
483                    class=BAR_BUTTON
484                    aria-label=move || if playing.get() { "Pause" } else { "Play" }
485                    title=move || if playing.get() { "Pause (Space)" } else { "Play (Space)" }
486                    on:click=move |_| toggle_play()
487                >
488                    {move || {
489                        if playing.get() {
490                            view! { <Glyph icon=icondata::LuPause /> }
491                        } else {
492                            view! { <Glyph icon=icondata::LuPlay /> }
493                        }
494                    }}
495                </button>
496                <button
497                    type="button"
498                    id="media-viewer-back"
499                    class=BAR_BUTTON
500                    aria-label="Back a frame"
501                    title="Back a frame (,)"
502                    on:click=move |_| step(-FRAME)
503                >
504                    <Glyph icon=icondata::LuSkipBack />
505                </button>
506                <button
507                    type="button"
508                    id="media-viewer-forward"
509                    class=BAR_BUTTON
510                    aria-label="Forward a frame"
511                    title="Forward a frame (.)"
512                    on:click=move |_| step(FRAME)
513                >
514                    <Glyph icon=icondata::LuSkipForward />
515                </button>
516                <span class="shrink-0 px-1 font-mono text-xs tabular-nums" id="media-viewer-time">
517                    {move || clock(position.get())}
518                </span>
519                <input
520                    type="range"
521                    id="media-viewer-seek"
522                    aria-label="Position"
523                    class="h-11 min-w-32 flex-1 cursor-pointer accent-accent"
524                    min="0"
525                    step="any"
526                    disabled=move || !known()
527                    prop:max=move || if known() { duration.get() } else { 0.0 }
528                    prop:value=move || position.get()
529                    on:input=move |event| {
530                        if let Ok(to) = event_target_value(&event).parse::<f64>() {
531                            seek(to);
532                        }
533                    }
534                />
535                <span class="shrink-0 px-1 font-mono text-xs tabular-nums" id="media-viewer-duration">
536                    {move || clock(duration.get())}
537                </span>
538                <select
539                    id="media-viewer-speed"
540                    aria-label="Speed"
541                    title="Speed"
542                    class="min-h-11 shrink-0 rounded-md bg-transparent px-1 text-sm hover:bg-white/15 [&>option]:text-black"
543                    prop:value=move || speed.get().to_string()
544                    on:change=move |event| {
545                        if let (Ok(to), Some(video)) = (
546                            event_target_value(&event).parse::<f64>(),
547                            video.get_untracked(),
548                        ) {
549                            video.set_playback_rate(to);
550                        }
551                    }
552                >
553                    {SPEEDS
554                        .iter()
555                        .map(|speed| {
556                            view! { <option value=speed.to_string()>{format!("{speed}×")}</option> }
557                        })
558                        .collect_view()}
559                </select>
560                <button
561                    type="button"
562                    id="media-viewer-mute"
563                    class=BAR_BUTTON
564                    aria-label=move || if muted.get() { "Unmute" } else { "Mute" }
565                    title=move || if muted.get() { "Unmute" } else { "Mute" }
566                    on:click=move |_| {
567                        if let Some(video) = video.get_untracked() {
568                            video.set_muted(!video.muted());
569                        }
570                    }
571                >
572                    {move || {
573                        if muted.get() {
574                            view! { <Glyph icon=icondata::LuVolumeX /> }
575                        } else {
576                            view! { <Glyph icon=icondata::LuVolume2 /> }
577                        }
578                    }}
579                </button>
580            </div>
581        }
582    });
583
584    view! {
585        <div
586            node_ref=root
587            id="media-viewer"
588            role="dialog"
589            aria-modal="true"
590            aria-label=label
591            tabindex="-1"
592            class="fixed inset-0 z-[60] flex flex-col bg-black/95 text-white outline-none"
593            on:keydown=on_keydown
594        >
595            <div class="flex shrink-0 items-center gap-1 px-2 pt-[max(0.25rem,env(safe-area-inset-top))] pb-1">
596                <span class="flex min-w-0 flex-1 items-baseline gap-2 px-2 text-sm">
597                    <span id="media-viewer-title" class="min-w-0 truncate">{title}</span>
598                    {at.filter(|_| stepping).map(|at| view! {
599                        <span id="media-viewer-position" class="shrink-0 font-mono text-xs text-white/70 tabular-nums">
600                            {format!("{} / {count}", at + 1)}
601                        </span>
602                    })}
603                    <span id="media-viewer-size" class="shrink-0 font-mono text-xs text-white/70 tabular-nums">
604                        {move || size.get().map(|(width, height)| format!("{width}×{height}"))}
605                    </span>
606                </span>
607                <button
608                    type="button"
609                    id="media-viewer-zoom-out"
610                    class=BAR_BUTTON
611                    aria-label="Zoom out"
612                    title="Zoom out (-)"
613                    on:click=move |_| zoom_by(0.8)
614                >
615                    <Glyph icon=icondata::LuZoomOut />
616                </button>
617                <button
618                    type="button"
619                    id="media-viewer-fit"
620                    class=format!("{BAR_BUTTON} font-mono tabular-nums")
621                    aria-label="Fit to the window"
622                    title="Fit to the window (0)"
623                    on:click=move |_| zoom.set(Zoom::default())
624                >
625                    {move || zoom.get().percent()}
626                </button>
627                <button
628                    type="button"
629                    id="media-viewer-zoom-in"
630                    class=BAR_BUTTON
631                    aria-label="Zoom in"
632                    title="Zoom in (+)"
