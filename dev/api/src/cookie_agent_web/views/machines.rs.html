35/// How long a finger holds a runtime before it lifts to be dragged.
36const LONG_PRESS: Duration = Duration::from_millis(500);

38/// A drop target: a machine's card, named by its `data-machine-drop`.

41/// A runtime as its row shows it.
42#[derive(Clone, Debug, PartialEq)]
43struct RuntimeCard {
44    id: RuntimeId,
45    /// The machine it is on; `None` for one the daemon placed on none.
46    machine: Option<MachineName>,
47    pinned: bool,
48    connected: bool,
49    listing: Option<RuntimeListing>,
50}

52/// The open form: a machine's rename or merge, or a runtime's "Move to…".
53#[derive(Clone, Debug, PartialEq)]
54enum Form {
55    Machine(MachineAction, MachineEntry),
56    Move(RuntimeCard),
57}

59/// A drag under way, shared by the rows and the window's listeners.
60#[derive(Clone, Copy)]
61struct Dragging {
62    gesture: StoredValue<Drag>,
63    /// The row pressed, until the press ends.
64    pressed: StoredValue<Option<RuntimeCard>>,
65    /// The row lifted or dragged, which follows the pointer.
66    dragged: RwSignal<Option<RuntimeId>>,
67    offset: RwSignal<(f64, f64)>,
68    /// The machine under the pointer.
69    over: RwSignal<Option<String>>,
70    timer: StoredValue<Option<TimeoutHandle>>,
71}
72
73impl Dragging {
74    fn new() -> Self {
75        Self {
76            gesture: StoredValue::new(Drag::default()),
77            pressed: StoredValue::new(None),
78            dragged: RwSignal::new(None),
79            offset: RwSignal::new((0.0, 0.0)),
80            over: RwSignal::new(None),
81            timer: StoredValue::new(None),
82        }
83    }
84
85    fn stop_timer(self) {
86        if let Some(timer) = self.timer.get_value() {
87            timer.clear();
88        }
89        self.timer.set_value(None);
90    }
91
92    fn reset(self) {
93        browser::set_pressing(false);
94        self.stop_timer();
95        self.gesture.update_value(Drag::cancel);
96        self.pressed.set_value(None);
97        self.dragged.set(None);
98        self.offset.set((0.0, 0.0));
99        self.over.set(None);
100    }
101
102    /// Follows the pointer and lets go of it, on the window, for as long
103    /// as the view shows.
104    fn follow(self, workspace: Workspace, form: RwSignal<Option<Form>>) {
105        let moved = window_event_listener(ev::pointermove, move |event| {
106            let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
107            match self
108                .gesture
109                .try_update_value(|gesture| gesture.moved(x, y))
110                .unwrap_or(DragStep::Nothing)
111            {
112                DragStep::Nothing => {}
113                DragStep::Cancelled => {
114                    browser::set_pressing(false);
115                    self.stop_timer();
116                    self.pressed.set_value(None);
117                }
118                DragStep::Started => {
119                    self.stop_timer();
120                    form.set(None);
121                    self.dragged.set(
122                        self.pressed
123                            .with_value(|pressed| pressed.as_ref().map(|card| card.id.clone())),
124                    );
125                    self.over
126                        .set(browser::attribute_at(x, y, DROP, "data-machine-drop"));
127                }
128                DragStep::Moved { dx, dy } => {
129                    self.offset.set((dx, dy));
130                    self.over
131                        .set(browser::attribute_at(x, y, DROP, "data-machine-drop"));
132                    browser::scroll_near_edge("#page [data-page-body]", y);
133                }
134            }
135        });
136        let released = window_event_listener(ev::pointerup, move |_| {
137            let end = self
138                .gesture
139                .try_update_value(Drag::release)
140                .unwrap_or(DragEnd::Nothing);
141            let card = self.pressed.get_value();
142            let over = self.over.get_untracked();
143            self.reset();
144            let Some(card) = card else {
145                return;
146            };
147            match end {
148                DragEnd::Nothing => {}
149                DragEnd::Held => form.set(Some(Form::Move(card))),
150                DragEnd::Dropped => {
151                    if let Some(request) =
152                        drop_request(&card.id, card.machine.as_ref(), over.as_deref())
153                    {
154                        workspace.change_machines(request);
155                    }
156                }
157            }
158        });
159        let cancelled = window_event_listener(ev::pointercancel, move |_| self.reset());
160        let unselectable = StoredValue::new_local(browser::hold_selection(move || {
161            self.pressed
162                .try_with_value(Option::is_some)
163                .unwrap_or(false)
164        }));
165        let hold = StoredValue::new_local(browser::hold_touch_scroll(move || {
166            self.gesture
167                .try_with_value(Drag::holds_scroll)
168                .unwrap_or(false)
169        }));
170        on_cleanup(move || {
171            moved.remove();
172            released.remove();
173            cancelled.remove();
174            if let Some(hold) = hold.try_update_value(Option::take).flatten() {
175                hold.remove();
176            }
177            if let Some(hold) = unselectable.try_update_value(Option::take).flatten() {
178                hold.remove();
179            }
180            browser::set_pressing(false);
181        });
182    }
183}

185/// The daemon's machines and its runtimes: the Machines & runtimes view.
186pub fn machines_page(workspace: Workspace) -> impl IntoView {
187    poll_while_open(move || workspace.poll_runtimes());
188    let form = RwSignal::new(None::<Form>);
189    let adding = RwSignal::new(false);
190    let dragging = Dragging::new();
191    dragging.follow(workspace, form);
192    let machines = Memo::new(move |_| workspace.machines.get());
193    let registry = Memo::new(move |_| workspace.runtimes.get());
194    view! {
195        <div class="flex flex-wrap items-center gap-2">
196            <h2 class="min-w-0 flex-1 text-sm font-bold">"Machines"</h2>
197            <button
198                type="button"
199                id="machines-add"
200                class=PRIMARY
201                aria-expanded=move || adding.get().to_string()
202                on:click=move |_| {
203                    if !adding.get_untracked() {
204                        workspace.poll_config();
205                    }
206                    adding.update(|adding| *adding = !*adding);
207                }
208            >
209                <Glyph icon=icondata::FaPlusSolid />
210                "Add a runtime"
211            </button>
212        </div>
213        <p class=NOTE>
214            "Drag a runtime onto another machine to move it there (on a touch screen, hold it first), or use its Move to… button."
215        </p>
216        <Show when=move || adding.get()>
217            {add_runtime(workspace, adding)}
218        </Show>
219        {move || {
220            // The view reads both lists every second: only a change
221            // renders the cards again, so an open form keeps its text.
222            let machines = match machines.get() {
223                None => return loading("Reading machines…").into_any(),
224                Some(Err(error)) => return view! { <p class=ERROR>{error}</p> }.into_any(),
225                Some(Ok(listing)) => listing,
226            };
227            let registry = registry.get();
228            machine_cards(workspace, &machines, registry.as_ref(), form, dragging).into_any()
229        }}
230    }
231}

233/// Each machine's card, then the runtimes on none.
234fn machine_cards(
235    workspace: Workspace,
236    machines: &MachinesListResult,
237    registry: Option<&RuntimesListResult>,
238    form: RwSignal<Option<Form>>,
239    dragging: Dragging,
240) -> impl IntoView {
241    let listing = |id: &RuntimeId| {
242        registry.and_then(|registry| {
243            registry
244                .runtimes
245                .iter()
246                .find(|runtime| runtime.runtime_id == *id)
247                .cloned()
248        })
249    };
250    let mut groups: Vec<(MachineName, Vec<RuntimeCard>)> = Vec::new();
251    for entry in machine_entries(machines) {
252        match entry {
253            MachineEntry::Machine(name) => groups.push((name, Vec::new())),
254            MachineEntry::Runtime {
255                machine,
256                runtime,
257                assigned_by,
258                pinned,
259            } => {
260                let listed = listing(&runtime);
261                let card = RuntimeCard {
262                    connected: listed
263                        .as_ref()
264                        .map_or(assigned_by.is_some(), |listed| listed.connected),
265                    listing: listed,
266                    id: runtime,
267                    machine: Some(machine),
268                    pinned,
269                };
270                if let Some((_, cards)) = groups.last_mut() {
271                    cards.push(card);
272                }
273            }
274        }
275    }
276    let placed = |id: &RuntimeId| {
277        groups
278            .iter()
279            .any(|(_, cards)| cards.iter().any(|card| card.id == *id))
280    };
281    let unplaced = registry
282        .map(|registry| {
283            registry
284                .runtimes
285                .iter()
286                .filter(|runtime| !placed(&runtime.runtime_id))
287                .map(|runtime| RuntimeCard {
288                    id: runtime.runtime_id.clone(),
289                    machine: None,
290                    pinned: false,
291                    connected: runtime.connected,
292                    listing: Some(runtime.clone()),
293                })
294                .collect::<Vec<_>>()
295        })
296        .unwrap_or_default();
297    let names = groups
298        .iter()
299        .map(|(name, _)| name.clone())
300        .collect::<Vec<_>>();
301    let cards = groups
302        .into_iter()
303        .map(|(name, runtimes)| {
304            let info = machines
305                .machines
306                .iter()
307                .find(|machine| machine.name == name)
308                .cloned();
309            machine_card(
310                workspace,
311                name,
312                info,
313                runtimes,
314                names.clone(),
315                form,
316                dragging,
317            )
318        })
319        .collect_view();
320    view! {
321        {cards}
322        {(!unplaced.is_empty()).then(|| {
323            let rows = unplaced
324                .into_iter()
325                .map(|card| runtime_row(workspace, card, names.clone(), form, dragging))
326                .collect_view();
327            view! {
328                <section class="rounded-xl border border-toasted dark:border-ember" data-unplaced>
329                    <div class="flex items-center gap-3 border-b border-toasted px-3 py-2 dark:border-ember">
330                        <span class="inline-flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-toasted dark:bg-ember">
331                            <Glyph icon=icondata::FaCircleQuestionRegular class="text-lg" />
332                        </span>
333                        <div class="min-w-0 flex-1">
334                            <p class="text-sm font-bold">"On no machine yet"</p>
335                            <p class=NOTE>"Runtimes that have not connected: drag one onto a machine to pin it there."</p>
336                        </div>
337                    </div>
338                    <ul class="flex flex-col">{rows}</ul>
339                </section>
340            }
341        })}
342    }
343}

345/// A machine's card: its operating system's icon, its name, what it
346/// holds, its rename, merge, and forget, and its runtimes. While a runtime
347/// is dragged, the card takes it.
348fn machine_card(
349    workspace: Workspace,
350    name: MachineName,
351    info: Option<cookie_agent_protocol::MachineInfo>,
352    runtimes: Vec<RuntimeCard>,
353    names: Vec<MachineName>,
354    form: RwSignal<Option<Form>>,
355    dragging: Dragging,
356) -> impl IntoView {
357    let platform = info.as_ref().and_then(|info| info.platform.clone());
358    let memories = info.as_ref().map_or(0, |info| info.memories);
359    let detail = [
360        platform.clone(),
361        Some(match runtimes.len() {
362            1 => "1 runtime".to_owned(),
363            count => format!("{count} runtimes"),
364        }),
365        Some(if memories == 1 {
366            "1 memory".to_owned()
367        } else {
368            format!("{memories} memories")
369        }),
370    ]
371    .into_iter()
372    .flatten()
373    .collect::<Vec<_>>()
374    .join(" · ");
375    let entry = MachineEntry::Machine(name.clone());
376    let act = move |entry: MachineEntry, action: MachineAction| match machine_action_request(
377        &entry, action,
378    ) {
379        Ok(Some(request)) => workspace.change_machines(request),
380        Ok(None) => form.set(Some(Form::Machine(action, entry))),
381        Err(error) => workspace.status.set(error),
382    };
383    let actions = entry
384        .actions()
385        .into_iter()
386        .map(|action| {
387            let entry = entry.clone();
388            let (icon, label) = match action {
389                MachineAction::Rename => (icondata::FaPenSolid, "Rename the machine"),
390                MachineAction::Merge => (icondata::FaCodeMergeSolid, "Merge into another machine"),
391                _ => (icondata::FaTrashCanRegular, "Forget the machine"),
392            };
393            view! {
394                <button
395                    type="button"
396                    class=ICON_BUTTON
397                    aria-label=label
398                    title=label
399                    data-machine-action=action.label()
400                    on:click=move |_| act(entry.clone(), action)
401                >
402                    <Glyph icon=icon />
403                </button>
404            }
405        })
406        .collect_view();
407    let key = name.to_string();
408    let over = {
409        let key = key.clone();
410        move || dragging.over.get().as_deref() == Some(key.as_str())
411    };
412    let over_dark = over.clone();
413    let over_ring = over.clone();
414    // Another machine's runtime is dragged: this card can take it.
415    let target = {
416        let key = key.clone();
417        let runtimes = runtimes
418            .iter()
419            .map(|card| card.id.clone())
420            .collect::<Vec<_>>();
421        move || {
422            dragging
423                .dragged
424                .with(|dragged| dragged.as_ref().is_some_and(|id| !runtimes.contains(id)))
425                && dragging.over.get().as_deref() != Some(key.as_str())
426        }
427    };
428    let target_dark = target.clone();
429    let empty = runtimes.is_empty();
430    let rows = runtimes
431        .into_iter()
432        .map(|card| runtime_row(workspace, card, names.clone(), form, dragging))
433        .collect_view();
434    let form_entry = entry.clone();
435    view! {
436        <section
437            class="machine-card rounded-xl border border-toasted bg-parchment transition-colors dark:border-ember dark:bg-char"
438            class:ring-2=over_ring
439            class:ring-caramel=over.clone()
440            class:dark:ring-maple=over_dark
441            class:border-dashed=target.clone()
442            class:border-caramel=target
443            class:dark:border-maple=target_dark
444            data-machine-drop=key.clone()
445        >
446            <div class="machine-row flex items-center gap-3 border-b border-toasted py-1 pr-1 pl-3 dark:border-ember" data-machine-entry=key.clone()>
447                <span class="inline-flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-toasted dark:bg-ember" title=platform.clone().unwrap_or_else(|| "operating system not reported".into())>
448                    <Glyph icon=platform_icon(platform.as_deref()) class="text-lg" />
449                </span>
450                <div class="min-w-0 flex-1 py-1">
451                    <p class="truncate font-mono text-sm font-bold">{key.clone()}</p>
452                    <p class=NOTE>{detail}</p>
453                </div>
454                {actions}
455            </div>
456            {move || match form.get() {
457                Some(Form::Machine(action, entry)) if entry == form_entry => machine_form(workspace, action, entry, form).into_any(),
458                _ => ().into_any(),
459            }}
460            <ul class="flex flex-col">{rows}</ul>
461            {empty.then(|| view! { <p class=format!("{NOTE} px-3 py-3")>"No runtimes. Drop one here to pin it to this machine."</p> })}
462        </section>
463    }
464}

466/// A machine's rename or merge: the name it asks for.
467fn machine_form(
468    workspace: Workspace,
469    action: MachineAction,
470    entry: MachineEntry,
471    form: RwSignal<Option<Form>>,
472) -> impl IntoView {
473    let title = machine_form_title(action, &entry);
474    let submit = StoredValue::new_local(move || {
475        match machine_name_request(action, &entry, &field_value("machine-name")) {
476            Ok(request) => {
477                form.set(None);
478                workspace.change_machines(request);
479            }
480            Err(error) => workspace.status.set(error),
481        }
482    });
483    let submit = move || submit.with_value(|submit| submit());
484    view! {
485        <div class="flex flex-col gap-2 border-b border-toasted p-3 dark:border-ember">
486            <label class="flex flex-col gap-1 text-sm">
487                <span>{title}</span>
488                <input
489                    type="text"
490                    id="machine-name"
491                    class=FIELD
492                    on:keydown=move |event: ev::KeyboardEvent| {
493                        if event.key() == "Enter" {
494                            event.prevent_default();
495                            submit();
496                        }
497                    }
498                />
499            </label>
500            <div class="flex gap-2">
501                <button type="button" id="machine-apply" class=PRIMARY on:click=move |_| submit()>
502                    "Apply"
503                </button>
504                <button type="button" class=BUTTON on:click=move |_| form.set(None)>
505                    "Cancel"
506                </button>
507            </div>
508        </div>
509    }
510}

512/// A runtime's row: a grip, how it came to the daemon, its ID, whether it
513/// is connected and pinned, what it offers, and why it last failed; then
514/// "Move to…" and, when pinned, Unpin. Pressing the row drags it.
515fn runtime_row(
516    workspace: Workspace,
517    card: RuntimeCard,
518    names: Vec<MachineName>,
519    form: RwSignal<Option<Form>>,
520    dragging: Dragging,
521) -> impl IntoView {
522    let id = card.id.clone();
523    let lifetime = card.listing.as_ref().map(|listing| listing.lifetime);
524    let message = card
525        .listing
526        .as_ref()
527        .and_then(|listing| listing.message.clone())
528        .filter(|_| !card.connected);
529    let capabilities = card
530        .listing
531        .as_ref()
532        .map(|listing| {
533            listing
534                .capabilities
535                .iter()
536                .map(|capability| capability.name.to_string())
537                .collect::<Vec<_>>()
538        })
539        .unwrap_or_default();
540    let (status_icon, status_class, status) = if card.connected {
541        (
542            icondata::FaCircleSolid,
543            "text-green-700 dark:text-green-400",
544            "connected",
545        )
546    } else if message.is_some() {
547        (
548            icondata::FaCircleExclamationSolid,
549            "text-red-700 dark:text-red-400",
550            "not connected",
551        )
552    } else {
553        (icondata::FaCircleRegular, "opacity-60", "not connected")
554    };
555    let detail = [
556        Some(status.to_owned()),
557        lifetime.map(|lifetime| lifetime_label(lifetime).to_owned()),
558        card.listing
559            .as_ref()
560            .and_then(|listing| listing.platform.clone()),
561    ]
562    .into_iter()
563    .flatten()
564    .collect::<Vec<_>>()
565    .join(" · ");
566    let pressed = card.clone();
567    let on_pointerdown = move |event: ev::PointerEvent| {
568        // The row's buttons are clicks, not drags.
569        let on_button = event
570            .target()
571            .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
572            .and_then(|target| target.closest("button").ok().flatten())
573            .is_some();
574        let touch = event.pointer_type() == "touch";
575        if on_button || (!touch && event.button() != 0) {
576            return;
577        }
578        let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
579        dragging.stop_timer();
580        dragging
581            .gesture
582            .update_value(|gesture| gesture.press(x, y, touch));
583        dragging.pressed.set_value(Some(pressed.clone()));
584        if touch {
585            browser::set_pressing(true);
586            let id = pressed.id.clone();
587            let timer = set_timeout_with_handle(
588                move || {
589                    if dragging
590                        .gesture
591                        .try_update_value(Drag::elapse)
592                        .unwrap_or(false)
593                    {
594                        dragging.dragged.set(Some(id.clone()));
595                    }
596                },
597                LONG_PRESS,
598            )
599            .ok();
600            dragging.timer.set_value(timer);
601        } else {
602            // No text selection while the mouse drags.
603            event.prevent_default();
604        }
605    };
606    let lifted = {
607        let id = id.clone();
608        move || {
609            dragging
610                .dragged
611                .with(|dragged| dragged.as_ref() == Some(&id))
612        }
613    };
614    let moving = card.clone();
615    let unpinned = card.id.clone();
616    let picker_card = card.clone();
617    view! {
618        <li
619            class="runtime-entry no-callout relative flex cursor-grab touch-manipulation items-center gap-2 border-b border-toasted py-1 pr-1 pl-2 select-none last:border-b-0 dark:border-ember"
620            class:z-30=lifted.clone()
621            class:rounded-lg=lifted.clone()
622            class:shadow-xl=lifted.clone()
623            class:pointer-events-none=lifted.clone()
624            class:bg-cream=lifted.clone()
625            class:dark:bg-night=lifted.clone()
626            class:ring-2=lifted.clone()
627            class:ring-caramel=lifted.clone()
628            class:dark:ring-maple=lifted.clone()
629            style:transform={
630                let lifted = lifted.clone();
631                move || {
632                    let (dx, dy) = dragging.offset.get();
633                    if lifted() {
634                        format!("translate({dx}px, {dy}px) scale(1.02)")
635                    } else {
636                        String::new()
637                    }
638                }
639            }
640            data-runtime=id.to_string()
641            data-machine-entry=id.to_string()
642            data-dragging=move || lifted().then_some("true")
643            on:pointerdown=on_pointerdown
644            on:contextmenu=move |event: ev::MouseEvent| {
645                // A long press lifts the row; it is not the browser's menu.
646                if dragging.pressed.with_value(Option::is_some) {
647                    event.prevent_default();
648                }
649            }
650        >
651            <Glyph icon=icondata::FaGripVerticalSolid class="opacity-40" />
652            <span class="inline-flex h-8 w-8 shrink-0 items-center justify-center rounded-md bg-toasted dark:bg-ember" title=lifetime.map_or("not listed", lifetime_label)>
653                <Glyph icon=lifetime_icon(lifetime) />
654            </span>
655            <div class="flex min-w-0 flex-1 flex-col gap-0.5 py-1">
656                <span class="flex min-w-0 items-center gap-2">
657                    <span class="min-w-0 truncate font-mono text-sm font-bold">{id.to_string()}</span>
658                    {card.pinned.then(|| view! {
659                        <span class=format!("{STATE} inline-flex items-center gap-1") title="Pinned to this machine, whatever it reports">
660                            <Glyph icon=icondata::FaThumbtackSolid />
661                            "pinned"
662                        </span>
663                    })}
664                </span>
665                <span class=format!("flex min-w-0 items-center gap-1.5 {NOTE}")>
666                    <Glyph icon=status_icon class=format!("text-[0.6rem] {status_class}") />
667                    <span class="min-w-0 truncate">{detail}</span>
668                </span>
669                {(!capabilities.is_empty()).then(|| view! {
670                    <span class="flex flex-wrap gap-1">
671                        {capabilities.into_iter().map(|capability| view! {
672                            <span class="rounded bg-toasted px-1.5 font-mono text-[0.7rem] dark:bg-ember">{capability}</span>
673                        }).collect_view()}
674                    </span>
675                })}
676                {message.map(|message| view! { <span class="text-xs break-words text-red-800 dark:text-red-300">{message}</span> })}
677            </div>
678            <button
679                type="button"
680                class=format!("{BUTTON} px-2")
681                aria-label="Move to…"
682                title="Move to another machine"
683                data-machine-action=MachineAction::Move.label()
684                on:click=move |_| form.set(Some(Form::Move(moving.clone())))
685            >
686                <Glyph icon=icondata::FaArrowRightArrowLeftSolid />
687                <span class="hidden sm:inline">"Move to…"</span>
688            </button>
689            {card.pinned.then(|| view! {
690                <button
691                    type="button"
692                    class=ICON_BUTTON
693                    aria-label="Unpin"
694                    title="Unpin: follow the machine it reports again"
695                    data-machine-action=MachineAction::Unpin.label()
696                    on:click=move |_| workspace.change_machines(
697                        cookie_agent_client_state::runtimes::MachineRequest::Unassign { runtime: unpinned.clone() },
698                    )
699                >
700                    <Glyph icon=icondata::FaThumbtackSlashSolid />
701                </button>
702            })}
703        </li>
704        {move || match form.get() {
705            Some(Form::Move(open)) if open.id == picker_card.id => move_picker(workspace, open, names.clone(), form).into_any(),
706            _ => ().into_any(),
707        }}
708    }
709}

711/// "Move to…": the other machines, or a new machine's name.
712fn move_picker(
713    workspace: Workspace,
714    card: RuntimeCard,
715    names: Vec<MachineName>,
716    form: RwSignal<Option<Form>>,
717) -> impl IntoView {
718    let targets = names
719        .iter()
720        .filter(|name| Some(*name) != card.machine.as_ref())
721        .map(|name| {
722            let request = cookie_agent_client_state::runtimes::MachineRequest::Assign {
723                runtime: card.id.clone(),
724                machine: name.clone(),
725            };
726            view! {
727                <button
728                    type="button"
729                    class=BUTTON
730                    data-move-target=name.to_string()
731                    on:click=move |_| {
732                        form.set(None);
733                        workspace.change_machines(request.clone());
734                    }
735                >
736                    <Glyph icon=icondata::FaServerSolid />
737                    <span class="font-mono">{name.to_string()}</span>
738                </button>
739            }
740        })
741        .collect_view();
742    let runtime = card.id.clone();
743    let submit = StoredValue::new_local(move || {
744        match move_request(&runtime, &field_value("machine-name")) {
745            Ok(request) => {
746                form.set(None);
747                workspace.change_machines(request);
748            }
749            Err(error) => workspace.status.set(error),
750        }
751    });
752    let submit = move || submit.with_value(|submit| submit());
753    view! {
754        <li class=format!("{CONFIRM} m-2") data-move-picker=card.id.to_string()>
755            <p class="text-sm font-bold">{format!("Move {} to", card.id)}</p>
756            <div class="flex flex-wrap gap-2">{targets}</div>
757            <label class="flex flex-col gap-1 text-sm">
758                <span>"Or a new machine"</span>
759                <input
760                    type="text"
761                    id="machine-name"
762                    class=FIELD
763                    placeholder="machine name"
764                    on:keydown=move |event: ev::KeyboardEvent| {
765                        if event.key() == "Enter" {
766                            event.prevent_default();
767                            submit();
768                        }
769                    }
770                />
771            </label>
772            <div class="flex gap-2">
773                <button type="button" id="machine-apply" class=PRIMARY on:click=move |_| submit()>
774                    "Move"
775                </button>
776                <button type="button" class=BUTTON on:click=move |_| form.set(None)>
777                    "Cancel"
778                </button>
779            </div>
780        </li>
781    }
782}
