52/// A condition of the filter, as its builder edits it.
53#[derive(Clone, Copy)]
54struct Row {
55    key: usize,
56    header: RwSignal<bool>,
57    name: RwSignal<String>,
58    values: RwSignal<Vec<String>>,
59    /// The value being typed, added on Enter, a comma, or leaving.
60    typing: RwSignal<String>,
61}
62
63impl Row {
64    fn new(key: usize, condition: Condition) -> Self {
65        Self {
66            key,
67            header: RwSignal::new(condition.header),
68            name: RwSignal::new(condition.name),
69            values: RwSignal::new(condition.values),
70            typing: RwSignal::new(String::new()),
71        }
72    }
73
74    /// The condition, with what is being typed added.
75    fn condition(self) -> Condition {
76        let mut values = self.values.get_untracked();
77        let typing = self.typing.get_untracked();
78        values.extend(
79            typing
80                .split(',')
81                .map(str::trim)
82                .filter(|value| !value.is_empty())
83                .map(str::to_owned),
84        );
85        Condition {
86            header: self.header.get_untracked(),
87            name: self.name.get_untracked(),
88            values,
89        }
90    }
91
92    /// Adds what is being typed as values.
93    fn commit(self) {
94        let typed = self.typing.get_untracked();
95        let new: Vec<String> = typed
96            .split(',')
97            .map(str::trim)
98            .filter(|value| !value.is_empty())
99            .map(str::to_owned)
100            .collect();
101        if !new.is_empty() {
102            self.values.update(|values| {
103                for value in new {
104                    if !values.contains(&value) {
105                        values.push(value);
106                    }
107                }
108            });
109        }
110        self.typing.set(String::new());
111    }
112}

771/// One condition: its source, its name, its values as chips, and a field
772/// that adds more.
773fn condition_row(row: Row, rows: RwSignal<Vec<Row>>) -> impl IntoView {
774    let remove_value = move |value: String| {
775        row.values
776            .update(|values| values.retain(|kept| *kept != value));
777    };
778    view! {
779        <div class="flex flex-wrap items-center gap-2 rounded-lg bg-sunken p-2" data-webhook-condition>
780            <select
781                class=format!("{SELECT} w-auto sm:w-32")
782                aria-label="Source"
783                data-condition-source
784                on:change=move |event| row.header.set(event_target_value(&event) == "header")
785            >
786                <option value="header" selected=move || row.header.get()>"Header"</option>
787                <option value="json" selected=move || !row.header.get()>"JSON field"</option>
788            </select>
789            <input
790                type="text"
791                class=format!("{FIELD} w-auto min-w-0 flex-1 sm:max-w-[13rem]")
792                aria-label="Name"
793                data-condition-name
794                placeholder=move || if row.header.get() { "X-GitHub-Event" } else { "/action" }
795                prop:value=move || row.name.get()
796                on:input=move |event| row.name.set(event_target_value(&event))
797            />
798            <span class="text-[0.8125rem] text-muted">"is"</span>
799            <For each=move || row.values.get() key=|value| value.clone() let:value>
800                {
801                    let removed = value.clone();
802                    view! {
803                        <span class="inline-flex h-7 items-center gap-1 rounded-md bg-accent-soft pl-2 font-mono text-xs text-accent" data-condition-value=value.clone()>
804                            {value.clone()}
805                            <button
806                                type="button"
807                                class="hit-area inline-flex h-7 w-6 items-center justify-center rounded-md hover:bg-press"
808                                aria-label=format!("Remove {value}")
809                                on:click=move |_| remove_value(removed.clone())
810                            >
811                                <Glyph icon=icondata::LuX class="text-xs" />
812                            </button>
813                        </span>
814                    }
815                }
816            </For>
817            <input
818                type="text"
819                class=format!("{FIELD} w-auto min-w-[7rem] flex-1")
820                aria-label="Add a value"
821                data-condition-typing
822                placeholder="a value, Enter"
823                prop:value=move || row.typing.get()
824                on:input=move |event| {
825                    let text = event_target_value(&event);
826                    row.typing.set(text.clone());
827                    if text.contains(',') {
828                        row.commit();
829                    }
830                }
831                on:keydown=move |event: leptos::ev::KeyboardEvent| {
832                    if event.key() == "Enter" {
833                        event.prevent_default();
834                        row.commit();
835                    }
836                }
837                on:change=move |_| row.commit()
838            />
839            <button
840                type="button"
841                class=ICON_BUTTON
842                title="Remove this condition"
843                aria-label="Remove this condition"
844                on:click=move |_| rows.update(|rows| rows.retain(|kept| kept.key != row.key))
845            >
846                <Glyph icon=icondata::LuX />
847            </button>
848        </div>
849    }
