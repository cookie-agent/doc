1//! The first-run sheet (`docs/site/specs/default-models-and-onboarding.md`
2//! section 9): a four-step stepper over the daemon's onboarding API. The
3//! provider step is Settings' connect form; the model steps are searchable
4//! lists with variant chips; the kit step previews what installing the
5//! starter kit does to each file. Full screen on a phone.

41#[component]
42fn Stepper(workspace: Workspace) -> impl IntoView {
43    let onboarding = workspace.onboarding;
44    let step = RwSignal::new(
45        onboarding
46            .status
47            .with_untracked(|status| status.as_ref().map(first_step))
48            .unwrap_or(OnboardingStep::Provider),
49    );
50    let default_choice = RwSignal::new(None::<(ModelKey, Option<VariantId>)>);
51    let small_choice = RwSignal::new(None::<(ModelKey, Option<VariantId>)>);
52    let kit = RwSignal::new(None::<Result<OnboardingPreviewResult, String>>);
53    let kit_source = RwSignal::new(None::<OnboardingKitSource>);
54    let install = RwSignal::new(true);
55    let load_kit = move |source: Option<OnboardingKitSource>| {
56        kit.set(None);
57        kit_source.set(source.clone());
58        spawn_local(async move {
59            let preview = workspace.preview_kit(source).await;
60            install.set(preview.as_ref().is_ok_and(|preview| {
61                preview
62                    .files
63                    .iter()
64                    .any(|file| file.action != OnboardingFileAction::Skip)
65            }));
66            kit.set(Some(preview));
67        });
68    };
69    // The kit is read when its step first shows.
70    Effect::new(move |_| {
71        if step.get() == OnboardingStep::Kit && kit.with_untracked(Option::is_none) {
72            load_kit(kit_source.get_untracked());
73        }
74    });
75    let status = move || onboarding.status.get();
76    let refusal = move || {
77        status().and_then(|status| {
78            finish_refusal(
79                &status,
80                default_choice.with(Option::is_some),
81                small_choice.with(Option::is_some),
82            )
83        })
84    };
85    let finish = move || {
86        let models = workspace.models();
87        let input = |choice: Choice| {
88            choice.and_then(|(key, variant)| {
89                models
90                    .iter()
91                    .find(|model| model.key == key)
92                    .map(|model| model_input(model, variant.as_ref()))
93            })
94        };
95        let kit_ready = kit.with_untracked(|kit| matches!(kit, Some(Ok(_))));
96        let install_kit = install.get_untracked() && kit_ready;
97        workspace.run_onboarding(OnboardingRunParams {
98            install_kit,
99            source: if install_kit {
100                kit_source.get_untracked()
101            } else {
102                None
103            },
104            default_model: input(default_choice.get_untracked()),
105            small_model: input(small_choice.get_untracked()),
106        });
107    };
108    view! {
109        <div
110            id="onboarding"
111            role="dialog"
112            aria-modal="true"
113            aria-labelledby="onboarding-title"
114            class="fixed inset-0 z-50 flex items-stretch justify-center bg-black/40 sm:items-center sm:p-4"
115        >
116            <div class="flex h-full w-full min-w-0 flex-col bg-cream text-espresso sm:h-auto sm:max-h-[90vh] sm:max-w-2xl sm:rounded-xl sm:border sm:border-toasted dark:bg-night dark:text-latte sm:dark:border-ember">
117                <header class="flex items-center gap-2 border-b border-toasted px-4 py-2 dark:border-ember">
118                    <h2 id="onboarding-title" class="min-w-0 flex-1 truncate text-base font-bold">"Set up Cookie"</h2>
119                    <Show when=move || onboarding.result.with(Option::is_none)>
120                        <button type="button" id="onboarding-not-now" class=BUTTON on:click=move |_| workspace.dismiss_onboarding()>
121                            "Not now"
122                        </button>
123                    </Show>
124                </header>
125                {move || match onboarding.result.get() {
126                    Some(result) => summary(workspace, &result).into_any(),
127                    None => view! {
128                        <nav aria-label="Onboarding steps" class="grid grid-cols-2 gap-1 border-b border-toasted px-4 py-2 sm:flex sm:flex-wrap dark:border-ember">
129                            {OnboardingStep::ALL
130                                .into_iter()
131                                .enumerate()
132                                .map(|(index, each)| {
133                                    let current = move || step.get() == each;
134                                    let done = move || {
135                                        status().is_some_and(|status| each.done(&status))
136                                            || (each == OnboardingStep::DefaultModel && default_choice.with(Option::is_some))
137                                            || (each == OnboardingStep::SmallModel && small_choice.with(Option::is_some))
138                                    };
139                                    view! {
140                                        <button
141                                            type="button"
142                                            class="inline-flex min-h-11 min-w-0 items-center gap-1.5 rounded-md px-2 text-left text-sm hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke"
143                                            class:font-bold=current
144                                            class:bg-toasted=current
145                                            class:dark:bg-ember=current
146                                            aria-current=move || current().then_some("step")
147                                            data-step=step_id(each)
148                                            data-done=move || done().to_string()
149                                            on:click=move |_| step.set(each)
150                                        >
151                                            {move || if done() {
152                                                view! { <Glyph icon=icondata::FaCircleCheckSolid class="text-green-700 dark:text-green-400" /> }.into_any()
153                                            } else {
154                                                view! { <span class="inline-flex size-5 items-center justify-center rounded-full border border-current text-xs">{index + 1}</span> }.into_any()
155                                            }}
156                                            {each.label()}
157                                        </button>
158                                    }
159                                })
160                                .collect_view()}
161                        </nav>
162                        <div class="flex min-h-0 flex-1 flex-col gap-2 overflow-y-auto px-4 py-3" data-onboarding-body>
163                            {move || match step.get() {
164                                OnboardingStep::Provider => provider_step(workspace).into_any(),
165                                OnboardingStep::DefaultModel => model_step(workspace, false, default_choice).into_any(),
166                                OnboardingStep::SmallModel => model_step(workspace, true, small_choice).into_any(),
167                                OnboardingStep::Kit => kit_step(kit, install, load_kit).into_any(),
168                            }}
169                        </div>
170                        <footer class="flex flex-col gap-2 border-t border-toasted px-4 py-2 dark:border-ember">
171                            {move || onboarding.error.get().map(|message| view! { <p class=ERROR id="onboarding-error">{message}</p> })}
172                            <div class="flex flex-wrap items-center gap-2">
173                                <button
174                                    type="button"
175                                    id="onboarding-back"
176                                    class=BUTTON
177                                    disabled=move || step.get().previous().is_none()
178                                    on:click=move |_| {
179                                        if let Some(previous) = step.get_untracked().previous() {
180                                            step.set(previous);
181                                        }
182                                    }
183                                >
184                                    <Glyph icon=icondata::FaArrowLeftSolid />
185                                    "Back"
186                                </button>
187                                <span class=format!("min-w-0 flex-1 {NOTE}") id="onboarding-refusal">
188                                    {move || (step.get() == OnboardingStep::Kit).then(refusal).flatten()}
189                                </span>
190                                {move || match step.get().next() {
191                                    Some(next) => view! {
192                                        <button type="button" id="onboarding-next" class=PRIMARY on:click=move |_| step.set(next)>
193                                            "Next"
194                                            <Glyph icon=icondata::FaChevronRightSolid />
195                                        </button>
196                                    }
197                                    .into_any(),
198                                    None => view! {
199                                        <button
200                                            type="button"
201                                            id="onboarding-finish"
202                                            class=PRIMARY
203                                            disabled=move || refusal().is_some() || onboarding.running.get()
204                                            title=move || refusal().unwrap_or_default()
205                                            on:click=move |_| finish()
206                                        >
207                                            {move || if onboarding.running.get() { "Finishing…" } else { "Finish" }}
208                                        </button>
209                                    }
210                                    .into_any(),
211                                }}
212                            </div>
213                        </footer>
214                    }
215                    .into_any(),
216                }}
217            </div>
218        </div>
219    }
220}

222const fn step_id(step: OnboardingStep) -> &'static str {
223    match step {
224        OnboardingStep::Provider => "provider",
225        OnboardingStep::DefaultModel => "default-model",
226        OnboardingStep::SmallModel => "small-model",
227        OnboardingStep::Kit => "kit",
228    }

231/// The connected providers with their models, then Settings' connect form.
232fn provider_step(workspace: Workspace) -> impl IntoView {
233    let connected = move || {
234        let models = workspace.models();
235        workspace
236            .onboarding
237            .status
238            .get()
239            .map(|status| status.providers_connected)
240            .unwrap_or_default()
241            .into_iter()
242            .map(|provider| {
243                let count = models
244                    .iter()
245                    .filter(|model| model.key.provider_id() == provider)
246                    .count();
247                format!(
248                    "{provider} · {count} available model{}",
249                    if count == 1 { "" } else { "s" }
250                )
251            })
252            .collect::<Vec<_>>()
253    };
254    view! {
255        <p class="text-sm">"Connect at least one model provider. You can connect several."</p>
256        {move || {
257            let connected = connected();
258            if connected.is_empty() {
259                view! { <p class=NOTE id="onboarding-connected">"No provider is connected yet."</p> }.into_any()
260            } else {
261                view! {
262                    <ul class="flex flex-col gap-1" id="onboarding-connected">
263                        {connected
264                            .into_iter()
265                            .map(|line| view! {
266                                <li class="flex items-center gap-2 text-sm">
267                                    <Glyph icon=icondata::FaPlugSolid class="text-green-700 dark:text-green-400" />
268                                    {line}
269                                </li>
270                            })
271                            .collect_view()}
272                    </ul>
273                }
274                .into_any()
275            }
276        }}
277        {super::manage::connect_sheet(workspace)}
278    }
279}

281/// A searchable list of the connected providers' models with variant
282/// chips. The small model's list suggests nothing.
283fn model_step(workspace: Workspace, small: bool, choice: RwSignal<Choice>) -> impl IntoView {
284    let search = RwSignal::new(String::new());
285    let field = if small {
286        "small_model"
287    } else {
288        "default_model"
289    };
290    let current = move || {
291        workspace
292            .onboarding
293            .status
294            .get()
295            .and_then(|status: OnboardingStatus| {
296                if small {
297                    status.small_model
298                } else {
299                    status.default_model
300                }
301            })
302    };
303    let pick = move |model: &AvailableModelDescriptor| {
304        let variant = if small {
305            small_model_variant(model)
306        } else {
307            default_model_variant(model)
308        };
309        choice.set(Some((model.key.clone(), variant)));
310    };
311    view! {
312        <p class="text-sm">
313            {if small {
314                "The small model runs quick work: approvals, titles, and the explorer. It may be the default model."
315            } else {
316                "The default model runs your agents."
317            }}
318        </p>
319        {move || current().map(|current| view! {
320            <p class=NOTE data-current-model>{format!("{field} is {}; choosing another replaces it.", model_label(&current))}</p>
321        })}
322        <input
323            type="search"
324            class=FIELD
325            data-model-search
326            placeholder="Search models"
327            prop:value=move || search.get()
328            on:input=move |event| search.set(event_target_value(&event))
329        />
330        {move || {
331            let models = workspace.models();
332            let groups = model_groups(&models, &search.get());
333            if groups.is_empty() {
334                return view! {
335                    <p class=NOTE>
336                        {if models.is_empty() { "No model is available yet: connect a provider first." } else { "No model matches." }}
337                    </p>
338                }
339                .into_any();
340            }
341            let chosen = choice.get();
342            groups
343                .into_iter()
344                .map(|group| {
345                    let rows = group
346                        .models
347                        .into_iter()
348                        .map(|model| {
349                            let selected = chosen.as_ref().is_some_and(|(key, _)| *key == model.key);
350                            let picked = model.clone();
351                            let chips = selected.then(|| {
352                                let chosen_variant = chosen.as_ref().and_then(|(_, variant)| variant.clone());
353                                variants(model)
354                                    .into_iter()
355                                    .map(|variant| {
356                                        let on = chosen_variant.as_ref() == Some(&variant);
357                                        let key = model.key.clone();
358                                        let value = variant.clone();
359                                        view! {
360                                            <button
361                                                type="button"
362                                                class=BUTTON
363                                                class:bg-toasted=on
364                                                class:dark:bg-ember=on
365                                                aria-pressed=on.to_string()
366                                                data-variant=variant.to_string()
367                                                on:click=move |_| choice.set(Some((key.clone(), Some(value.clone()))))
368                                            >
369                                                {variant.to_string()}
370                                            </button>
371                                        }
372                                    })
373                                    .collect_view()
374                            });
375                            view! {
376                                <div class="flex flex-col gap-1">
377                                    <button
378                                        type="button"
379                                        class=ROW
380                                        class:bg-toasted=selected
381                                        class:dark:bg-ember=selected
382                                        aria-pressed=selected.to_string()
383                                        data-model=model.key.to_string()
384                                        on:click=move |_| pick(&picked)
385                                    >
386                                        <span class="flex w-full items-center gap-2">
387                                            <span class="min-w-0 flex-1 truncate text-sm font-bold">{model.display_name.clone()}</span>
388                                            {selected.then(|| view! { <Glyph icon=icondata::FaCheckSolid /> })}
389                                        </span>
390                                        <span class=format!("font-mono break-all {NOTE}")>{model.key.to_string()}</span>
391                                        <span class=NOTE>{format!("{} tokens of context", crate::format::compact_count(model.capabilities.context_tokens))}</span>
392                                    </button>
393                                    {chips.map(|chips| view! { <div class="flex flex-wrap gap-2 px-3 pb-1">{chips}</div> })}
394                                </div>
395                            }
396                        })
397                        .collect_view();
398                    view! {
399                        <section class="flex flex-col gap-1" data-provider-group=group.provider.to_string()>
400                            <h3 class="pt-2 text-xs font-bold uppercase tracking-wide text-ink dark:text-sand">{group.provider.to_string()}</h3>
401                            {rows}
402                        </section>
403                    }
404                })
405                .collect_view()
406                .into_any()
407        }}
408    }
409}

411/// What installing the starter kit does to each file, install or skip, and
412/// a downloaded kit instead of the release's.
413fn kit_step(
414    kit: RwSignal<Option<Result<OnboardingPreviewResult, String>>>,
415    install: RwSignal<bool>,
416    load_kit: impl Fn(Option<OnboardingKitSource>) + Copy + 'static,
417) -> impl IntoView {
418    let opened = RwSignal::new(HashSet::<String>::new());
419    let picked = move |event: leptos::ev::Event| {
420        let input: web_sys::HtmlInputElement = event_target(&event);
421        let Some(file) = crate::files::files_of(input.files()).into_iter().next() else {
422            return;
423        };
424        input.set_value("");
425        spawn_local(async move {
426            match crate::files::read_file(&file).await {
427                Ok(bytes) => load_kit(Some(archive_source(&bytes))),
428                Err(error) => kit.set(Some(Err(error))),
429            }
430        });
431    };
432    view! {
433        <p class="text-sm">"The starter kit: a commented config.toml, a primary agent with explore and worker subagents, and a skill for changing Cookie itself. Optional; your own files are never overwritten."</p>
434        {move || match kit.get() {
435            None => loading("Reading the starter kit…").into_any(),
436            Some(Err(message)) => view! {
437                <p class=ERROR id="onboarding-kit-error">{format!("The starter kit is not available: {message}")}</p>
438            }
439            .into_any(),
440            Some(Ok(preview)) => view! {
441                <p class=NOTE>{format!(
442                    "Kit {} from {}",
443                    preview.cookie_version.as_deref().map_or_else(|| format!("version {}", preview.kit_version), |version| format!("for cookie {version}")),
444                    preview.origin,
445                )}</p>
446                <ul class="flex flex-col gap-1">
447                    {preview
448                        .files
449                        .into_iter()
450                        .map(|file| {
451                            let path = file.path.clone();
452                            let toggled = path.clone();
453                            let shown = Memo::new(move |_| opened.with(|opened| opened.contains(&path)));
454                            let action = match file.action {
455                                OnboardingFileAction::Write => "write",
456                                OnboardingFileAction::Skip => "skip",
457                                OnboardingFileAction::WriteNew => "write_new",
458                                OnboardingFileAction::Unknown => "unknown",
459                            };
460                            let diff = file.diff.clone();
461                            view! {
462                                <li class="flex flex-col gap-1 rounded-md border border-toasted p-2 dark:border-ember" data-kit-file=file.path.clone() data-kit-action=action>
463                                    <span class="flex flex-wrap items-center gap-2">
464                                        <span class="min-w-0 flex-1 break-all font-mono text-sm">{file.path.clone()}</span>
465                                        <span class="shrink-0 rounded-full bg-toasted px-2 py-0.5 text-xs dark:bg-ember">{kit_file_status(&file)}</span>
466                                    </span>
467                                    {diff.map(|diff| view! {
468                                        <button type="button" class=BUTTON data-kit-diff on:click=move |_| opened.update(|opened| {
469                                            if !opened.remove(&toggled) {
470                                                opened.insert(toggled.clone());
471                                            }
472                                        })>
473                                            {move || if shown.get() { "Hide the diff" } else { "Show the diff" }}
474                                        </button>
475                                        <Show when=move || shown.get()>
476                                            <pre class="max-h-64 overflow-auto rounded-md bg-parchment p-2 font-mono text-xs dark:bg-char">{diff.clone()}</pre>
477                                        </Show>
478                                    })}
479                                </li>
480                            }
481                        })
482                        .collect_view()}
483                </ul>
484                <div class="flex flex-wrap gap-2" role="group" aria-label="Install the starter kit">
485                    <button type="button" id="onboarding-install" class=BUTTON class:bg-toasted=move || install.get() class:dark:bg-ember=move || install.get() aria-pressed=move || install.get().to_string() on:click=move |_| install.set(true)>
486                        "Install the kit"
487                    </button>
488                    <button type="button" id="onboarding-skip" class=BUTTON class:bg-toasted=move || !install.get() class:dark:bg-ember=move || !install.get() aria-pressed=move || (!install.get()).to_string() on:click=move |_| install.set(false)>
489                        "Skip it"
490                    </button>
491                </div>
492            }
493            .into_any(),
494        }}
495        <input type="file" id=KIT_INPUT class="hidden" accept=".gz,.tgz,application/gzip" on:change=picked />
496        <button type="button" id="onboarding-kit-file" class=BUTTON on:click=move |_| {
497            if let Some(input) = web_sys::window()
498                .and_then(|window| window.document())
499                .and_then(|document| document.get_element_by_id(KIT_INPUT))
500                .and_then(|element| wasm_bindgen::JsCast::dyn_into::<web_sys::HtmlElement>(element).ok())
501            {
502                input.click();
503            }
504        }>
505            <Glyph icon=icondata::FaFolderOpenRegular />
506            "Use a downloaded kit…"
507        </button>
508    }
509}

511/// The run's summary and the first session.
512fn summary(
513    workspace: Workspace,
514    result: &cookie_agent_protocol::OnboardingRunResult,
515) -> impl IntoView {
516    let lines = run_summary(result);
517    let ready = result.status.ready;
518    view! {
519        <div class="flex min-h-0 flex-1 flex-col gap-2 overflow-y-auto px-4 py-3" id="onboarding-summary">
520            <p class="text-sm font-bold">{if ready { "Cookie is set up." } else { "Saved." }}</p>
521            <ul class="flex flex-col gap-1">
522                {lines.into_iter().map(|line| view! { <li class="text-sm">{line}</li> }).collect_view()}
523            </ul>
524        </div>
525        <footer class="flex flex-wrap gap-2 border-t border-toasted px-4 py-2 dark:border-ember">
526            <button type="button" id="onboarding-first-session" class=PRIMARY on:click=move |_| workspace.start_first_session()>
527                "Start your first session"
528            </button>
529            <button type="button" id="onboarding-done" class=BUTTON on:click=move |_| workspace.close_onboarding()>
530                "Done"
531            </button>
532        </footer>
533    }
