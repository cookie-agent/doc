1//! The first-run sheet (`docs/site/specs/default-models-and-onboarding.md`
2//! section 9): a four-step stepper over the daemon's onboarding API. The
3//! provider step is Settings' connect form; the three model steps are
4//! searchable lists with variant chips that build each field's fallback list
5//! (`docs/site/specs/model-fallback-lists.md` section 6), the small and
6//! utility models optional. There is no starter kit: the daemon keeps the official agents
7//! itself. Full screen on a phone.
8

20/// A model step's fallback list, in order; empty keeps what is configured.
21type Choice = Vec<ModelChoice>;
22
23/// Which model a model step chooses.
24#[derive(Clone, Copy, Debug, Eq, PartialEq)]
25enum Slot {
26    Default,
27    Small,
28    Utility,
29}
30
31impl Slot {
32    const fn field(self) -> OnboardingModelField {
33        match self {
34            Self::Default => OnboardingModelField::DefaultModel,
35            Self::Small => OnboardingModelField::SmallModel,
36            Self::Utility => OnboardingModelField::UtilityModel,
37        }
38    }
39
40    /// The fields an entry of this one may name: the steps before it.
41    const fn earlier(self) -> &'static [OnboardingModelField] {
42        match self {
43            Self::Default => &[],
44            Self::Small => &[OnboardingModelField::DefaultModel],
45            Self::Utility => &[
46                OnboardingModelField::DefaultModel,
47                OnboardingModelField::SmallModel,
48            ],
49        }
50    }
51}

53/// The sheet, while it is open.
54#[component]
55pub fn OnboardingSheet(workspace: Workspace) -> impl IntoView {
56    move || {
57        workspace
58            .onboarding
59            .open
60            .get()
61            .then(|| view! { <Stepper workspace=workspace /> })
62    }
63}

65#[component]
66fn Stepper(workspace: Workspace) -> impl IntoView {
67    let onboarding = workspace.onboarding;
68    let step = RwSignal::new(
69        onboarding
70            .status
71            .with_untracked(|status| status.as_ref().map(first_step))
72            .unwrap_or(OnboardingStep::Provider),
73    );
74    let default_choice = RwSignal::new(Choice::new());
75    let small_choice = RwSignal::new(Choice::new());
76    let utility_choice = RwSignal::new(Choice::new());
77    let status = move || onboarding.status.get();
78    let refusal = move || {
79        status().and_then(|status| {
80            finish_refusal(
81                &status,
82                default_choice.with(|choice| !choice.is_empty()),
83                false,
84            )
85        })
86    };
87    let finish = move || {
88        let models = workspace.models();
89        let list = |choice: Choice| {
90            choice_entries(&choice, |key| models.iter().find(|model| model.key == *key))
91        };
92        // The daemon fetches the official agents first when it has none.
93        workspace.run_onboarding(OnboardingRunParams {
94            install_kit: false,
95            source: None,
96            default_model: None,
97            small_model: None,
98            utility_model: None,
99            default_model_list: list(default_choice.get_untracked()),
100            small_model_list: list(small_choice.get_untracked()),
101            utility_model_list: list(utility_choice.get_untracked()),
102            ..Default::default()
103        });
104    };
105    view! {
106        <div
107            id="onboarding"
108            role="dialog"
109            aria-modal="true"
110            aria-labelledby="onboarding-title"
111            class="fixed inset-0 z-50 flex items-stretch justify-center bg-black/40 sm:items-center sm:p-4"
112        >
113            <div class="flex h-full w-full min-w-0 flex-col bg-cream text-espresso sm:h-auto sm:max-h-[90vh] sm:max-w-2xl sm:rounded-xl sm:border sm:border-toasted dark:bg-night dark:text-latte sm:dark:border-ember">
114                <header class="flex items-center gap-2 border-b border-toasted px-4 py-2 dark:border-ember">
115                    <h2 id="onboarding-title" class="min-w-0 flex-1 truncate text-base font-bold">"Set up Cookie"</h2>
116                    <Show when=move || onboarding.result.with(Option::is_none)>
117                        <button type="button" id="onboarding-not-now" class=BUTTON on:click=move |_| workspace.dismiss_onboarding()>
118                            "Not now"
119                        </button>
120                    </Show>
121                </header>
122                {move || match onboarding.result.get() {
123                    Some(result) => summary(workspace, &result).into_any(),
124                    None => view! {
125                        <nav aria-label="Onboarding steps" class="grid grid-cols-2 gap-1 border-b border-toasted px-4 py-2 sm:flex sm:flex-wrap dark:border-ember">
126                            {OnboardingStep::ALL
127                                .into_iter()
128                                .enumerate()
129                                .map(|(index, each)| {
130                                    let current = move || step.get() == each;
131                                    let done = move || {
132                                        status().is_some_and(|status| each.done(&status))
133                                            || (each == OnboardingStep::DefaultModel && default_choice.with(|choice| !choice.is_empty()))
134                                            || (each == OnboardingStep::SmallModel && small_choice.with(|choice| !choice.is_empty()))
135                                            || (each == OnboardingStep::UtilityModel && utility_choice.with(|choice| !choice.is_empty()))
136                                    };
137                                    view! {
138                                        <button
139                                            type="button"
140                                            class="inline-flex min-h-11 min-w-0 items-center gap-1.5 rounded-md px-2 text-left text-sm hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke"
141                                            class:font-bold=current
142                                            class:bg-toasted=current
143                                            class:dark:bg-ember=current
144                                            aria-current=move || current().then_some("step")
145                                            data-step=step_id(each)
146                                            data-done=move || done().to_string()
147                                            on:click=move |_| step.set(each)
148                                        >
149                                            {move || if done() {
150                                                view! { <Glyph icon=icondata::FaCircleCheckSolid class="text-green-700 dark:text-green-400" /> }.into_any()
151                                            } else {
152                                                view! { <span class="inline-flex size-5 items-center justify-center rounded-full border border-current text-xs">{index + 1}</span> }.into_any()
153                                            }}
154                                            {each.label()}
155                                        </button>
156                                    }
157                                })
158                                .collect_view()}
159                        </nav>
160                        <div class="flex min-h-0 flex-1 flex-col gap-2 overflow-y-auto px-4 py-3" data-onboarding-body>
161                            {move || match step.get() {
162                                OnboardingStep::Provider => provider_step(workspace).into_any(),
163                                OnboardingStep::DefaultModel => model_step(workspace, Slot::Default, default_choice).into_any(),
164                                OnboardingStep::SmallModel => model_step(workspace, Slot::Small, small_choice).into_any(),
165                                OnboardingStep::UtilityModel => model_step(workspace, Slot::Utility, utility_choice).into_any(),
166                            }}
167                        </div>
168                        <footer class="flex flex-col gap-2 border-t border-toasted px-4 py-2 dark:border-ember">
169                            {move || onboarding.error.get().map(|message| view! { <p class=ERROR id="onboarding-error">{message}</p> })}
170                            <div class="flex flex-wrap items-center gap-2">
171                                <button
172                                    type="button"
173                                    id="onboarding-back"
174                                    class=BUTTON
175                                    disabled=move || step.get().previous().is_none()
176                                    on:click=move |_| {
177                                        if let Some(previous) = step.get_untracked().previous() {
178                                            step.set(previous);
179                                        }
180                                    }
181                                >
182                                    <Glyph icon=icondata::FaArrowLeftSolid />
183                                    "Back"
184                                </button>
185                                <span class=format!("min-w-0 flex-1 {NOTE}") id="onboarding-refusal">
186                                    {move || step.get().next().is_none().then(refusal).flatten()}
187                                </span>
188                                {move || match step.get().next() {
189                                    Some(next) => view! {
190                                        <button type="button" id="onboarding-next" class=PRIMARY on:click=move |_| step.set(next)>
191                                            "Next"
192                                            <Glyph icon=icondata::FaChevronRightSolid />
193                                        </button>
194                                    }
195                                    .into_any(),
196                                    None => view! {
197                                        <button
198                                            type="button"
199                                            id="onboarding-finish"
200                                            class=PRIMARY
201                                            disabled=move || refusal().is_some() || onboarding.running.get()
202                                            title=move || refusal().unwrap_or_default()
203                                            on:click=move |_| finish()
204                                        >
205                                            {move || if onboarding.running.get() { "Finishing…" } else { "Finish" }}
206                                        </button>
207                                    }
208                                    .into_any(),
209                                }}
210                            </div>
211                        </footer>
212                    }
213                    .into_any(),
214                }}
215            </div>
216        </div>
217    }
218}

220const fn step_id(step: OnboardingStep) -> &'static str {
221    match step {
222        OnboardingStep::Provider => "provider",
223        OnboardingStep::DefaultModel => "default-model",
224        OnboardingStep::SmallModel => "small-model",
225        OnboardingStep::UtilityModel => "utility-model",

229/// The connected providers with their models, then Settings' connect form.
230fn provider_step(workspace: Workspace) -> impl IntoView {
231    let connected = move || {
232        let models = workspace.models();
233        workspace
234            .onboarding
235            .status
236            .get()
237            .map(|status| status.providers_connected)
238            .unwrap_or_default()
239            .into_iter()
240            .map(|provider| {
241                let count = models
242                    .iter()
243                    .filter(|model| model.key.provider_id() == provider)
244                    .count();
245                format!(
246                    "{provider} · {count} available model{}",
247                    if count == 1 { "" } else { "s" }
248                )
249            })
250            .collect::<Vec<_>>()
251    };
252    view! {
253        <p class="text-sm">"Connect at least one model provider. You can connect several."</p>
254        {move || {
255            let connected = connected();
256            if connected.is_empty() {
257                view! { <p class=NOTE id="onboarding-connected">"No provider is connected yet."</p> }.into_any()
258            } else {
259                view! {
260                    <ul class="flex flex-col gap-1" id="onboarding-connected">
261                        {connected
262                            .into_iter()
263                            .map(|line| view! {
264                                <li class="flex items-center gap-2 text-sm">
265                                    <Glyph icon=icondata::FaPlugSolid class="text-green-700 dark:text-green-400" />
266                                    {line}
267                                </li>
268                            })
269                            .collect_view()}
270                    </ul>
271                }
272                .into_any()
273            }
274        }}
275        {super::manage::connect_sheet(workspace)}
276    }
277}

279/// A searchable list of the connected providers' models with variant
280/// chips: choosing a model adds it to the end of the field's fallback list,
281/// choosing it again takes it out; the small and utility models may also
282/// take the models of a field before them.
283fn model_step(workspace: Workspace, slot: Slot, choice: RwSignal<Choice>) -> impl IntoView {
284    let search = RwSignal::new(String::new());
285    let field = field_name(slot.field());
286    let current = move || {
287        workspace
288            .onboarding
289            .status
290            .get()
291            .and_then(|status| configured_label(&status, slot.field()))
292    };
293    let position = move |chosen: &Choice, key: &cookie_agent_protocol::ModelKey| {
294        chosen
295            .iter()
296            .position(|entry| matches!(entry, ModelChoice::Model { model, .. } if model == key))
297    };
298    let pick = move |model: &AvailableModelDescriptor| {
299        choice.update(|chosen| {
300            if let Some(index) = position(chosen, &model.key) {
301                chosen.remove(index);
302            } else if chosen.len() < cookie_agent_protocol::MAX_ONBOARDING_MODEL_ENTRIES {
303                let variant = match slot {
304                    Slot::Default => default_model_variant(model),
305                    Slot::Small => small_model_variant(model),
306                    Slot::Utility => utility_model_variant(model),
307                };
308                chosen.push(ModelChoice::Model {
309                    model: model.key.clone(),
310                    variant,
311                });
312            }
313        });
314    };
315    let same_as = slot
316        .earlier()
317        .iter()
318        .map(|earlier| {
319            let earlier = *earlier;
320            let entry = ModelChoice::Field(earlier);
321            let on = {
322                let entry = entry.clone();
323                move || choice.with(|chosen| chosen.contains(&entry))
324            };
325            let toggle = move |_| {
326                choice.update(|chosen| {
327                    if let Some(index) = chosen.iter().position(|each| *each == entry) {
328                        chosen.remove(index);
329                    } else {
330                        chosen.push(entry.clone());
331                    }
332                });
333            };
334            view! {
335                <button
336                    type="button"
337                    class=BUTTON
338                    class:bg-toasted=on.clone()
339                    class:dark:bg-ember=on.clone()
340                    aria-pressed=move || on().to_string()
341                    data-same-as=field_name(earlier)
342                    on:click=toggle
343                >
344                    {match earlier {
345                        OnboardingModelField::DefaultModel => "Same as the default model",
346                        _ => "Same as the small model",
347                    }}
348                </button>
349            }
350        })
351        .collect_view();
352    view! {
353        <p class="text-sm">
354            {match slot {
355                Slot::Default => "The default model runs your agents. Choose more for fallbacks: the first available one runs, and the next take over when it fails.",
356                Slot::Small => "Optional: the models of subagents that carry out work, such as the worker and the explorer. Without them they use the default model.",
357                Slot::Utility => "Optional: fast, cheap models for approvals and session titles. Without them they use the session's own model.",
358            }}
359        </p>
360        {move || current().map(|current| view! {
361            <p class=NOTE data-current-model>{format!("{field} is {current}; choosing models replaces it.")}</p>
362        })}
363        {(!slot.earlier().is_empty()).then(|| view! { <div class="flex flex-wrap gap-2">{same_as}</div> })}
364        {move || {
365            let chosen = choice.get();
366            (!chosen.is_empty()).then(|| view! {
367                <p class="text-sm" data-chosen-list>
368                    {format!(
369                        "In fallback order: {}",
370                        chosen.iter().map(choice_label).collect::<Vec<_>>().join(", then ")
371                    )}
372                </p>
373            })
374        }}
375        <input
376            type="search"
377            class=FIELD
378            data-model-search
379            placeholder="Search models"
380            prop:value=move || search.get()
381            on:input=move |event| search.set(event_target_value(&event))
382        />
383        {move || {
384            let models = workspace.models();
385            let groups = model_groups(&models, &search.get());
386            if groups.is_empty() {
387                return view! {
388                    <p class=NOTE>
389                        {if models.is_empty() { "No model is available yet: connect a provider first." } else { "No model matches." }}
390                    </p>
391                }
392                .into_any();
393            }
394            let chosen = choice.get();
395            groups
396                .into_iter()
397                .map(|group| {
398                    let rows = group
399                        .models
400                        .into_iter()
401                        .map(|model| {
402                            let index = position(&chosen, &model.key);
403                            let selected = index.is_some();
404                            let picked = model.clone();
405                            let chips = index.map(|index| {
406                                let chosen_variant = match &chosen[index] {
407                                    ModelChoice::Model { variant, .. } => variant.clone(),
408                                    ModelChoice::Field(_) => None,
409                                };
410                                variants(model)
411                                    .into_iter()
412                                    .map(|variant| {
413                                        let on = chosen_variant.as_ref() == Some(&variant);
414                                        let key = model.key.clone();
415                                        let value = variant.clone();
416                                        view! {
417                                            <button
418                                                type="button"
419                                                class=BUTTON
420                                                class:bg-toasted=on
421                                                class:dark:bg-ember=on
422                                                aria-pressed=on.to_string()
423                                                data-variant=variant.to_string()
424                                                on:click=move |_| choice.update(|chosen| {
425                                                    if let Some(index) = position(chosen, &key) {
426                                                        chosen[index] = ModelChoice::Model {
427                                                            model: key.clone(),
428                                                            variant: Some(value.clone()),
429                                                        };
430                                                    }
431                                                })
432                                            >
433                                                {variant.to_string()}
434                                            </button>
435                                        }
436                                    })
437                                    .collect_view()
438                            });
439                            view! {
440                                <div class="flex flex-col gap-1">
441                                    <button
442                                        type="button"
443                                        class=ROW
444                                        class:bg-toasted=selected
445                                        class:dark:bg-ember=selected
446                                        aria-pressed=selected.to_string()
447                                        data-model=model.key.to_string()
448                                        data-position=index.map(|index| (index + 1).to_string())
449                                        on:click=move |_| pick(&picked)
450                                    >
451                                        <span class="flex w-full items-center gap-2">
452                                            <span class="min-w-0 flex-1 truncate text-sm font-bold">{model.display_name.clone()}</span>
453                                            {index.map(|index| view! {
454                                                <span class="inline-flex size-5 items-center justify-center rounded-full border border-current text-xs" title="Place in the fallback list">{index + 1}</span>
455                                            })}
456                                        </span>
457                                        {(model.display_name.as_str() != model.key.to_string()).then(|| view! {
458                                            <span class=format!("font-mono break-all {NOTE}") data-doc-id>{model.key.to_string()}</span>
459                                        })}
460                                        <span class=NOTE>{format!("{} tokens of context", cookie_agent_client_state::format::compact_count(model.capabilities.context_tokens))}</span>
461                                    </button>
462                                    {chips.map(|chips| view! { <div class="flex flex-wrap gap-2 px-3 pb-1">{chips}</div> })}
463                                </div>
464                            }
465                        })
466                        .collect_view();
467                    view! {
468                        <section class="flex flex-col gap-1" data-provider-group=group.provider.to_string()>
469                            <h3 class="pt-2 text-xs font-bold uppercase tracking-wide text-ink dark:text-sand">{group.provider.to_string()}</h3>
470                            {rows}
471                        </section>
472                    }
473                })
474                .collect_view()
475                .into_any()
476        }}
477    }
478}

480/// The run's summary and the first session.
481fn summary(
482    workspace: Workspace,
483    result: &cookie_agent_protocol::OnboardingRunResult,
484) -> impl IntoView {
485    let lines = run_summary(result);
486    let ready = result.status.ready;
487    view! {
488        <div class="flex min-h-0 flex-1 flex-col gap-2 overflow-y-auto px-4 py-3" id="onboarding-summary">
489            <p class="text-sm font-bold">{if ready { "Cookie is set up." } else { "Saved." }}</p>
490            <ul class="flex flex-col gap-1">
491                {lines.into_iter().map(|line| view! { <li class="text-sm">{line}</li> }).collect_view()}
492            </ul>
493        </div>
494        <footer class="flex flex-wrap gap-2 border-t border-toasted px-4 py-2 dark:border-ember">
495            <button type="button" id="onboarding-first-session" class=PRIMARY on:click=move |_| workspace.start_first_session()>
496                "Start your first session"
497            </button>
498            <button type="button" id="onboarding-done" class=BUTTON on:click=move |_| workspace.close_onboarding()>
499                "Done"
500            </button>
501        </footer>
502    }
