50/// Reads `poll` every second while the sheet is open, as the TUI polls its
51/// open `/runtime` and `/mcp` panels.
52pub(super) fn poll_while_open(poll: impl Fn() + 'static) {
53    let ticker = set_interval_with_handle(poll, PANEL_POLL).ok();
54    on_cleanup(move || {
55        if let Some(ticker) = ticker {
56            ticker.clear();
57        }
58    });
59}

61/// The value of the text field `id`, read once.
62pub(super) fn field_value(id: &str) -> String {
63    web_sys::window()
64        .and_then(|window| window.document())
65        .and_then(|document| document.get_element_by_id(id))
66        .and_then(|element| element.dyn_into::<web_sys::HtmlInputElement>().ok())
67        .map(|input| input.value())

273/// A server's tools by its own names, collapsed under its tool count: those
274/// it offers, then those its filters left out, dimmed and struck through.
275/// With `manage`, each is a button that leaves it out or takes it back.
276/// Nothing for a server that has listed none.
277fn server_tools(
278    server: &McpServerInfo,
279    manage: Option<Workspace>,
280) -> Option<impl IntoView + use<>> {
281    let (offered, left_out) = server_tool_names(server);
282    if offered.is_empty() && left_out.is_empty() {
283        return None;
284    }
285    let chip = |name: &str, filtered: bool| {
286        let label = name.to_owned();
287        let chip = view! {
288            <span
289                class="rounded bg-toasted px-1.5 font-mono text-[0.7rem] dark:bg-ember"
290                class:line-through=filtered
291                class:opacity-50=filtered
292            >
293                {label}
294            </span>
295        };
296        let tool = name.to_owned();
297        let toggled = server.clone();
298        view! {
299            <li
300                title=if filtered {
301                    "left out by the server's include_tools or exclude_tools"
302                } else {
303                    "offered to agents"
304                }
305                data-mcp-tool=name.to_owned()
306                data-mcp-tool-filtered=filtered.then_some("true")
307            >
308                {match manage {
309                    Some(workspace) => view! {
310                        <button
311                            type="button"
312                            class="rounded hover:outline hover:outline-1 hover:outline-caramel dark:hover:outline-maple"
313                            aria-label=format!("{} {name}", if filtered { "Take back" } else { "Leave out" })
314                            data-mcp-tool-toggle
315                            on:click=move |_| workspace.toggle_mcp_tool(&toggled, &tool, !filtered)
316                        >
317                            {chip}
318                        </button>
319                    }
320                    .into_any(),
321                    None => chip.into_any(),
322                }}
323            </li>
324        }
325    };
326    let chips = offered
327        .iter()
328        .map(|name| chip(name, false))
329        .chain(left_out.iter().map(|name| chip(name, true)))
330        .collect_view();
331    Some(view! {
332        <details class="pl-4" data-mcp-tools>
333            <summary class=format!("cursor-pointer select-none {NOTE}")>
334                {if manage.is_some() { "Tools (select one to leave it out or take it back)" } else { "Tools" }}
335            </summary>
336            <ul class="mt-1 flex flex-wrap gap-1">{chips}</ul>
337        </details>
338    })
339}

341/// The MCP servers a runtime row's instance uses: each one's name, tool
342/// count, and state, `runtime-wide` for one its runtime shares across
343/// workspaces, and under it why it failed or what it waits for. A server
344/// that waits for a workspace says it starts with one. With `workspaces`,
345/// the home of their runtime, a server one workspace hosts says which.
346pub(crate) fn runtime_mcp_servers(
347    servers: Vec<McpServerInfo>,
348    workspaces: Option<Option<String>>,
349    manage: Option<(Workspace, RwSignal<Option<McpEdit>>)>,
350) -> impl IntoView {
351    let items = servers
352        .into_iter()
353        .map(|server| {
354            let tone = server_tone(&server);
355            let (dot, dot_class) = match tone {
356                StateTone::Ready => (icondata::FaCircleSolid, "text-green-700 dark:text-green-400"),
357                StateTone::Pending => (icondata::FaCircleHalfStrokeSolid, "text-amber-700 dark:text-amber-400"),
358                StateTone::Failed => (icondata::FaCircleExclamationSolid, "text-red-700 dark:text-red-400"),
359                StateTone::Off => (icondata::FaCircleRegular, "opacity-60"),
360            };
361            let failed = tone == StateTone::Failed;
362            let waiting = waits_for_workspace(&server);
363            let at = workspaces.as_ref().and_then(|home| {
364                server.workspace.as_deref().map(|workspace| {
365                    format!(
366                        "at {}",
367                        cookie_agent_client_state::paths::display_path(workspace, home.as_deref())
368                    )
369                })
370            });
371            view! {
372                <li
373                    class="flex min-w-0 flex-col gap-0.5"
374                    data-mcp-server=server.name.clone()
375                    data-mcp-state=mcp_state_label(server.state)
376                    data-mcp-scope=server.scope.map(|scope| scope.as_str())
377                    data-mcp-waiting=waiting.then_some("true")
378                >
379                    <div class="flex min-w-0 flex-wrap items-center gap-x-2 gap-y-0.5">
380                        <Glyph icon=dot class=format!("shrink-0 text-[0.6rem] {dot_class}") />
381                        <span class="min-w-16 flex-1 truncate font-mono text-sm">{server.name.clone()}</span>
382                        {(!waiting).then(|| view! { <span class=format!("shrink-0 {NOTE}")>{server_tools_label(&server)}</span> })}
383                        <span class=STATE class:text-red-700=failed class:dark:text-red-400=failed>{server_state_label(&server)}</span>
384                    </div>
385                    {scope_note(&server).map(|note| view! {
386                        <span class=format!("pl-4 {NOTE}") data-mcp-scope-note>
387                            <span class="rounded bg-toasted px-1.5 text-[0.7rem] dark:bg-ember">{note}</span>
388                            " one process every workspace of the runtime shares"
389                        </span>
390                    })}
391                    {at.map(|at| view! { <span class=format!("pl-4 {NOTE} truncate")>{at}</span> })}
392                    {server_tools(&server, manage.map(|(workspace, _)| workspace))}
393                    {manage.filter(|_| edit_refusal(&server).is_none()).map(|(workspace, form)| {
394                        let (edited, toggled, removed) = (server.clone(), server.clone(), server.clone());
395                        let enable = !server.definition.enabled;
396                        view! {
397                            <div class="flex flex-wrap gap-1 pl-3" data-mcp-actions>
398                                <button type="button" class=ROW_ACTION data-mcp-edit on:click=move |_| form.set(Some(McpEdit::edit(&edited)))>
399                                    "Edit"
400                                </button>
401                                <button type="button" class=ROW_ACTION data-mcp-toggle on:click=move |_| workspace.toggle_mcp(&toggled, enable)>
402                                    {if enable { "Enable" } else { "Disable" }}
403                                </button>
404                                <button type="button" class=ROW_ACTION data-mcp-remove on:click=move |_| workspace.remove_mcp(&removed)>
405                                    "Remove"
406                                </button>
407                            </div>
408                        }
409                    })}
410                    {server_message(&server).map(|message| view! {
411                        <span
412                            class=if failed {
413                                "pl-4 text-xs whitespace-pre-wrap break-words text-red-800 dark:text-red-300"
414                            } else {
415                                "pl-4 text-xs whitespace-pre-wrap break-words opacity-80"
416                            }
417                            data-mcp-message
418                        >
419                            {message}
420                        </span>
421                    })}
422                </li>
423            }
424        })
425        .collect_view();
426    view! {
427        <div class="runtime-mcp flex flex-col gap-1 rounded-md bg-parchment px-2 py-1.5 dark:bg-char" data-runtime-mcp>
428            <span class="text-xs font-bold opacity-70">"MCP servers"</span>
429            <ul class="flex flex-col gap-1">{items}</ul>
430        </div>
431    }
432}

470pub fn mcp_sheet(workspace: Workspace) -> impl IntoView {
471    poll_while_open(move || workspace.poll_mcp());
472    // The form: its fields, the server it edits, and where it is kept.
473    let form = RwSignal::new(None::<McpEdit>);
474    let persist = RwSignal::new(PersistChoice::Runtime);
475    let error = RwSignal::new(None::<String>);
476    let list = move || {
477        match workspace.mcp_servers.get() {
478        None => loading("Reading MCP servers…").into_any(),
479        Some(Err(message)) => view! { <p class=ERROR>{message}</p> }.into_any(),
480        Some(Ok(servers)) if servers.is_empty() => {
481            empty_state(icondata::FaServerSolid, "No MCP servers are configured.").into_any()
482        }
483        Some(Ok(servers)) => servers
484            .into_iter()
485            .map(|server| {
486                let refusal = Workspace::mcp_refusal(&server);
487                let needs_auth = server.state == McpServerState::NeedsAuth;
488                let name = server.name.clone();
489                let edited = server.clone();
490                let removed = server.clone();
491                let toggled = (server.clone(), !server.definition.enabled);
492                let reconnected = server.clone();
493                let editable = refusal.is_none();
494                view! {
495                    <div class="mcp-server flex flex-col gap-1 border-b border-toasted py-2 dark:border-ember" data-server=name.clone()>
496                        <div class="flex items-center gap-2">
497                            <span class="min-w-0 flex-1 truncate font-mono text-sm font-bold">{name.clone()}</span>
498                            <span class=format!("mcp-state {STATE}")>{mcp_state_label(server.state)}</span>
499                        </div>
500                        <span class=NOTE>
501                            {format!(
502                                "{} · {} · {}",
503                                runtime_label(&server),
504                                source_label(server.source),
505                                server_tools_label(&server)
506                            )}
507                        </span>
508                        <span class=format!("{NOTE} font-mono whitespace-pre-wrap break-all")>{server_definition_display(&server)}</span>
509                        {server.message.clone().map(|message| if needs_auth {
510                            // Why it waits for a sign-in, not a failure.
511                            view! { <span class=format!("{NOTE} whitespace-pre-wrap break-words") data-mcp-message>{message}</span> }.into_any()
512                        } else {
513                            view! { <span class=ERROR data-mcp-message>{message}</span> }.into_any()
514                        })}
515                        {needs_auth.then(|| sign_in_offer(workspace, &server))}
516                        {refusal.clone().map(|refusal| view! { <span class=NOTE>{refusal}</span> })}
517                        <div class="flex flex-wrap gap-2">
518                            <Show when=move || editable>
519                                <button type="button" class=BUTTON data-mcp-edit on:click={
520                                    let edited = edited.clone();
521                                    move |_| {
522                                        error.set(None);
523                                        form.set(Some(McpEdit::edit(&edited)));
524                                    }
525                                }>
526                                    "Edit"
527                                </button>
528                                <button type="button" class=BUTTON data-mcp-toggle on:click={
529                                    let toggled = toggled.clone();
530                                    move |_| workspace.toggle_mcp(&toggled.0, toggled.1)
531                                }>
532                                    {if toggled.1 { "Enable" } else { "Disable" }}

1143/// A provider's connect form, or its details when it cannot be connected.
1144/// Credentials stay in their password fields: they are read once when the
1145/// form is sent and never kept in the page's state.
1146fn provider_view(
1147    workspace: Workspace,
1148    provider: ProviderDescriptor,
1149    open: RwSignal<Option<ProviderDescriptor>>,
1150    confirm_disconnect: RwSignal<bool>,
1151) -> impl IntoView {
1152    let state = row_state(&provider, &[], None);
1153    let summary = model_counts_summary(&provider);
1154    let back = move |_| open.set(None);
1155    if state == ProviderRowState::Unsupported {
1156        let reason = provider
1157            .support
1158            .reason
1159            .as_ref()
1160            .map_or_else(|| "unsupported".to_owned(), ToString::to_string);
1161        return view! {
1162            <p class="text-sm font-bold">{format!("{} ({})", provider.display_name, provider.id)}</p>
1163            <p class="text-sm">{reason}</p>
1164            <p class=NOTE>{summary}</p>
1165            <button type="button" class=BUTTON on:click=back>"Back"</button>
1166        }
1167        .into_any();
1168    }
1169    if let Some(method) = cookie_agent_client_state::providers::sign_in_method(&provider).cloned() {
1170        return sign_in_view(workspace, provider, method, open, confirm_disconnect).into_any();
1171    }
1172    let Some(initial) = initial_auth_method(&provider).cloned() else {
1173        return view! {
1174            <p class="text-sm">"This provider has no store-backed authentication form."</p>
1175            <button type="button" class=BUTTON on:click=back>"Back"</button>
1176        }
1177        .into_any();
1178    };
1179    let method = RwSignal::new(initial.id.clone());
1180    let error = RwSignal::new(None::<String>);
1181    let submitted = RwSignal::new(false);
1182    let reconnect = provider.durable_connection.is_some();
1183    let provider_id = provider.id.clone();
1184    // A connect that finished without an error closes the form.
1185    Effect::new(move |_| {
1186        let operation = workspace
1187            .provider_operations
1188            .with(|operations| operations.get(&provider_id).cloned());
1189        if !submitted.get_untracked() {
1190            return;
1191        }
1192        match operation {
1193            None => {
1194                submitted.set(false);
1195                open.set(None);
1196            }
1197            Some(ProviderOperation::Error { message, .. }) => {
1198                submitted.set(false);
1199                error.set(Some(message));
1200            }
1201            Some(ProviderOperation::InProgress(_)) => {}
1202        }
1203    });
1204    let methods = provider.auth_methods.clone();
1205    let credential_methods = provider.auth_methods.clone();
1206    let setup_fields = provider.setup_fields.clone();
1207    let form_provider = provider.clone();
1208    let submit = move || {
1209        let provider = form_provider.clone();
1210        let Some(auth) = provider
1211            .auth_methods
1212            .iter()
1213            .find(|candidate| candidate.id == method.get_untracked())
1214            .cloned()
1215        else {
1216            return;
1217        };
1218        let typed_setup = provider
1219            .setup_fields
1220            .iter()
1221            .enumerate()
1222            .map(|(index, field)| (field, field_value(&format!("setup-{index}"))))
1223            .collect::<Vec<_>>();
1224        let setup = match setup_values(
1225            &typed_setup
1226                .iter()
1227                .map(|(field, value)| (*field, value.as_str()))
1228                .collect::<Vec<_>>(),
1229        ) {
1230            Ok(values) => values,
1231            Err(message) => {
1232                error.set(Some(format!("Invalid public setup: {message}")));
1233                return;
1234            }
1235        };
1236        let typed = auth
1237            .credentials
1238            .iter()
1239            .enumerate()
1240            .map(|(index, field)| (field, field_value(&format!("credential-{index}"))))
1241            .collect::<Vec<_>>();
1242        let credentials = credential_values(
1243            &typed
1244                .iter()
1245                .map(|(field, value)| (*field, value.as_str()))
1246                .collect::<Vec<_>>(),
1247        );
1248        drop(typed);
1249        let credentials = match credentials {
1250            Ok(values) => values,
1251            Err(message) => {
1252                error.set(Some(format!("Invalid credentials: {message}")));
1253                return;
1254            }
1255        };
1256        error.set(None);
1257        match workspace.connect_provider(provider, auth.id.clone(), setup, credentials, reconnect) {
1258            Ok(()) => submitted.set(true),
1259            Err(message) => error.set(Some(message)),
1260        }
1261    };
1262    let disconnect_provider = provider.clone();
1263    view! {
1264        <p class="text-sm font-bold">{format!("{} ({})", provider.display_name, provider.id)}</p>
1265        <p class=NOTE>{summary}</p>
1266        <p class=NOTE>{cookie_agent_client_state::providers::DURABLE_PROVIDER_COPY}</p>
1267        <Show when={
1268            let count = methods.len();
1269            move || count > 1
1270        }>
1271            <label class="flex flex-col gap-1 text-sm">
1272                <span>"Authentication"</span>
1273                <select
1274                    id="provider-auth-method"
1275                    class=FIELD
1276                    on:change=move |event| {
1277                        if let Ok(id) = cookie_agent_protocol::AuthMethodId::new(event_target_value(&event)) {
1278                            method.set(id);
1279                        }
1280                    }
1281                >
1282                    {methods
1283                        .iter()
1284                        .map(|candidate| {
1285                            let id = candidate.id.clone();
1286                            view! {
1287                                <option value=candidate.id.to_string() selected=move || method.get() == id>
1288                                    {candidate.display_name.to_string()}
1289                                </option>
1290                            }
1291                        })
1292                        .collect_view()}
1293                </select>
1294            </label>
1295        </Show>
1296        {move || {
1297            let current = method.get();
1298            credential_methods
1299                .iter()
1300                .find(|candidate| candidate.id == current)
1301                .map(|auth| {
1302                    auth.credentials
1303                        .iter()
1304                        .enumerate()
1305                        .map(|(index, field)| view! {
1306                            <label class="flex flex-col gap-1 text-sm">
1307                                <span>{format!("{}{}", field.display_name, if field.required { "" } else { " (optional)" })}</span>
1308                                <input
1309                                    type="password"
1310                                    id=format!("credential-{index}")
1311                                    class=FIELD
1312                                    autocomplete="off"
1313                                    spellcheck="false"
1314                                />
1315                                <span class=NOTE>{field.help.to_string()}</span>
1316                            </label>
1317                        })
1318                        .collect_view()
1319                })
1320        }}
1321        {setup_fields
1322            .iter()
1323            .enumerate()
1324            .map(|(index, field)| view! {
1325                <label class="flex flex-col gap-1 text-sm">
1326                    <span>{format!("{}{}", field.display_name, if field.required { "" } else { " (optional)" })}</span>
1327                    <input
1328                        type="text"
1329                        id=format!("setup-{index}")
1330                        class=FIELD
1331                        value=initial_setup_text(&disconnect_provider, field)
1332                    />
1333                    <span class=NOTE>{field.help.to_string()}</span>
1334                </label>
1335            })
1336            .collect_view()}
1337        {move || error.get().map(|message| view! { <p class=ERROR id="provider-error">{message}</p> })}
1338        <div class="flex flex-wrap gap-2">
1339            <button type="button" id="provider-connect" class=PRIMARY on:click=move |_| submit()>
1340                <Glyph icon=icondata::FaPlugSolid />
1341                {if reconnect { "Reconnect" } else { "Connect" }}
1342            </button>
1343            <button type="button" class=BUTTON on:click=back>"Cancel"</button>
1344            <Show when=move || reconnect>
1345                <button type="button" id="provider-disconnect" class=BUTTON on:click=move |_| confirm_disconnect.set(true)>
1346                    "Disconnect"
1347                </button>
1348            </Show>
1349        </div>
1350        <Show when=move || confirm_disconnect.get()>
1351            <div class=CONFIRM>
1352                <p class="text-sm">"Remove the stored connection and its credentials?"</p>
1353                <div class="flex gap-2">
1354                    <button type="button" id="provider-disconnect-confirm" class=PRIMARY on:click={
1355                        let provider = provider.clone();
1356                        move |_| {
1357                            confirm_disconnect.set(false);
1358                            open.set(None);
1359                            workspace.disconnect_provider(provider.clone());
1360                        }
1361                    }>
1362                        "Disconnect"
1363                    </button>
1364                    <button type="button" class=BUTTON on:click=move |_| confirm_disconnect.set(false)>"Keep it"</button>
1365                </div>
1366            </div>
1367        </Show>
1368    }
1369    .into_any()
1370}
