35/// Reads `poll` every second while the sheet is open, as the TUI polls its
36/// open `/runtime` and `/mcp` panels.
37fn poll_while_open(poll: impl Fn() + 'static) {
38    let ticker = set_interval_with_handle(poll, PANEL_POLL).ok();
39    on_cleanup(move || {
40        if let Some(ticker) = ticker {
41            ticker.clear();
42        }
43    });
44}

46/// The value of the text field `id`, read once.
47fn field_value(id: &str) -> String {
48    web_sys::window()
49        .and_then(|window| window.document())
50        .and_then(|document| document.get_element_by_id(id))
51        .and_then(|element| element.dyn_into::<web_sys::HtmlInputElement>().ok())
52        .map(|input| input.value())
53        .unwrap_or_default()
54}

93fn attachments_view(workspace: Workspace) -> impl IntoView {
94    let search = RwSignal::new(String::new());
95    let chosen = RwSignal::new(None::<RuntimeChoice>);
96    let attaching = RwSignal::new(false);

100            .into_iter()
101            .enumerate()

104                let message = row
105                    .instance
106                    .as_ref()
107                    .and_then(|instance| instance.message.clone());
108                let failed = status == Some(RuntimeInstanceStatus::Failed);
109                let detach_name = row.attachment.name.clone();
110                let retry_row = row.clone();
111                view! {
112                    <div class="runtime-row flex flex-col gap-1 border-b border-toasted py-2 dark:border-ember" data-attachment=row.attachment.name.to_string()>
113                        <div class="flex items-center gap-2">
114                            <span class="min-w-0 flex-1 truncate font-mono text-sm font-bold">
115                                {format!("{} · {}", row.attachment.name, row.attachment.kind)}
116                            </span>
117                            <span class="text-xs" class:text-red-700=failed class:dark:text-red-400=failed>
118                                {status_label(status)}
119                            </span>
120                        </div>
121                        <span class=NOTE>
122                            {format!(
123                                "{}/{}{}",
124                                row.attachment.runtime_id,
125                                row.attachment.capability,
126                                row.attachment
127                                    .workspace
128                                    .as_ref()
129                                    .map(|workspace| format!(" at {workspace}"))
130                                    .unwrap_or_default()
131                            )}

133                        {row.description.clone().map(|description| view! { <span class=NOTE>{description}</span> })}

143                    </div>

146            .collect_view()
147    };

150            .runtimes
151            .with(|listing| listing.as_ref().map(|listing| listing.runtimes.clone()))

165    };
166    view! {
167        <Show when=move || workspace.new_session.get()>
168            <p class=NOTE>"The runtimes the new session will attach; each starts now so its status shows."</p>
169        </Show>
170        {rows}
171        <Show
172            when=move || attaching.get()
173            fallback=move || view! {
174                <button type="button" id="runtime-attach" class=BUTTON on:click=move |_| attaching.set(true)>
175                    "Attach a runtime"
176                </button>
177            }
178        >
179            {move || match chosen.get() {
180                None => view! {
181                    <input
182                        type="search"
183                        id="runtime-search"
184                        class=FIELD
185                        placeholder="Search runtimes"
186                        prop:value=move || search.get()
187                        on:input=move |event| search.set(event_target_value(&event))
188                    />
189                    {move || {
190                        choices()
191                            .into_iter()
192                            .map(|choice| {
193                                let picked = choice.clone();
194                                view! {
195                                    <button
196                                        type="button"
197                                        class=ROW
198                                        data-choice=choice.label()

213                                                .unwrap_or_else(|| if choice.connected { String::new() } else { "not connected".into() })}
214                                        </span>
215                                        {choice.message.clone().map(|message| view! { <span class=ERROR>{message}</span> })}
216                                    </button>
217                                }
218                            })
219                            .collect_view()
220                    }}
221                    <button type="button" class=BUTTON on:click=move |_| attaching.set(false)>

258fn machines_view(workspace: Workspace) -> impl IntoView {
259    let form = RwSignal::new(None::<(MachineAction, MachineEntry)>);
260    let act = move |entry: MachineEntry, action: MachineAction| match machine_action_request(
261        &entry, action,
262    ) {
263        Ok(Some(request)) => workspace.change_machines(request),
264        Ok(None) => form.set(Some((action, entry))),
265        Err(error) => workspace.status.set(error),
266    };
267    move || {
268        let listing = workspace.machines.get();
269        let machines = match &listing {
270            None => return view! { <p class="text-sm">"Reading machines…"</p> }.into_any(),
271            Some(Err(error)) => return view! { <p class=ERROR>{error.clone()}</p> }.into_any(),
272            Some(Ok(listing)) => listing.clone(),
273        };
274        let rows = machine_entries(&machines)
275            .into_iter()
276            .map(|entry| {
277                let (label, detail, indent) = match &entry {
278                    MachineEntry::Machine(name) => {
279                        let machine = machines.machines.iter().find(|machine| machine.name == *name);
280                        let platform = machine
281                            .and_then(|machine| machine.platform.clone())
282                            .map(|platform| format!(" ({platform})"))
283                            .unwrap_or_default();
284                        let memories = machine.map_or(0, |machine| machine.memories);
285                        (
286                            format!("{name}{platform}"),
287                            if memories == 1 { "1 memory".to_owned() } else { format!("{memories} memories") },
288                            false,
289                        )
290                    }
291                    MachineEntry::Runtime { runtime, assigned_by, pinned, .. } => (
292                        runtime.to_string(),
293                        match (assigned_by, pinned) {
294                            (Some(_), true) => "connected · pinned".to_owned(),
295                            (Some(_), false) => "connected".to_owned(),
296                            (None, _) => "pinned · not connected".to_owned(),
297                        },
298                        true,
299                    ),
300                };
301                let actions = entry
302                    .actions()
303                    .into_iter()
304                    .map(|action| {
305                        let entry = entry.clone();
306                        view! {
307                            <button
308                                type="button"
309                                class=BUTTON
310                                data-machine-action=action.label()
311                                on:click=move |_| act(entry.clone(), action)
312                            >
313                                {action.label()}
314                            </button>
315                        }
316                    })
317                    .collect_view();
318                view! {
319                    <div class="machine-row flex flex-col gap-1 border-b border-toasted py-2 dark:border-ember" class:pl-6=indent data-machine-entry=label.clone()>
320                        <span class="font-mono text-sm" class:font-bold=!indent>{label.clone()}</span>
321                        <span class=NOTE>{detail}</span>
322                        <div class="flex flex-wrap gap-2">{actions}</div>
323                    </div>
324                }
325            })
326            .collect_view();
327        view! {
328            {rows}
329            {move || {
330                form.get()
331                    .map(|(action, entry)| {
332                        let title = machine_form_title(action, &entry);
333                        let submit = StoredValue::new_local(move || {
334                            match machine_name_request(action, &entry, &field_value("machine-name")) {
335                                Ok(request) => {
336                                    form.set(None);
337                                    workspace.change_machines(request);
338                                }
339                                Err(error) => workspace.status.set(error),
340                            }
341                        });
342                        let submit = move || submit.with_value(|submit| submit());
343                        view! {
344                            <div class="mt-2 flex flex-col gap-2 border-t border-toasted pt-2 dark:border-ember">
345                                <label class="flex flex-col gap-1 text-sm">
346                                    <span>{title}</span>
347                                    <input
348                                        type="text"
349                                        id="machine-name"
350                                        class=FIELD
351                                        on:keydown=move |event: ev::KeyboardEvent| {
352                                            if event.key() == "Enter" {
353                                                event.prevent_default();
354                                                submit();
355                                            }
356                                        }
357                                    />
358                                </label>
359                                <div class="flex gap-2">
360                                    <button type="button" id="machine-apply" class=BUTTON on:click=move |_| submit()>
361                                        "Apply"
362                                    </button>
363                                    <button type="button" class=BUTTON on:click=move |_| form.set(None)>
364                                        "Cancel"
365                                    </button>
366                                </div>
367                            </div>
368                        }
369                    })
370            }}
371        }
372        .into_any()
373    }

378pub fn mcp_sheet(workspace: Workspace) -> impl IntoView {
379    poll_while_open(move || workspace.poll_mcp());
380    // The form: its fields, the server it edits, and where it is kept.
381    let form = RwSignal::new(None::<(McpFields, Option<cookie_agent_protocol::McpServerInfo>)>);
382    let persist = RwSignal::new(PersistChoice::Runtime);
383    let error = RwSignal::new(None::<String>);
384    let list = move || {
385        match workspace.mcp_servers.get() {
386        None => view! { <p class="text-sm">"Reading MCP servers…"</p> }.into_any(),
387        Some(Err(message)) => view! { <p class=ERROR>{message}</p> }.into_any(),
388        Some(Ok(servers)) if servers.is_empty() => {
389            view! { <p class="text-sm">"No MCP servers are configured."</p> }.into_any()
390        }
391        Some(Ok(servers)) => servers
392            .into_iter()
393            .map(|server| {
394                let refusal = Workspace::mcp_refusal(&server);
395                let needs_auth = server.state == McpServerState::NeedsAuth;
396                let name = server.name.clone();
397                let edited = server.clone();
398                let removed = server.name.clone();
399                let toggled = (server.name.clone(), !server.definition.enabled);
400                let reconnected = server.clone();
401                let editable = refusal.is_none();
402                view! {
403                    <div class="mcp-server flex flex-col gap-1 border-b border-toasted py-2 dark:border-ember" data-server=name.clone()>
404                        <div class="flex items-center gap-2">
405                            <span class="min-w-0 flex-1 truncate font-mono text-sm font-bold">{name.clone()}</span>
406                            <span class="mcp-state text-xs">{mcp_state_label(server.state)}</span>
407                        </div>
408                        <span class=NOTE>
409                            {format!(
410                                "{} · {} · {} tools",
411                                runtime_label(&server),
412                                source_label(server.source),
413                                server.tool_count
414                            )}
415                        </span>
416                        <span class=format!("{NOTE} font-mono whitespace-pre-wrap break-all")>{definition_display(&server.definition)}</span>
417                        {server.message.clone().map(|message| view! { <span class=ERROR>{message}</span> })}
418                        {needs_auth.then(|| view! {
419                            <span class="mcp-auth text-sm">
420                                {format!(
421                                    "This server needs authorization, which the page cannot do: run `cookie mcp auth {name}` on a machine with a browser, or authorize it from the TUI."
422                                )}
423                            </span>
424                        })}
425                        {refusal.clone().map(|refusal| view! { <span class=NOTE>{refusal}</span> })}
426                        <div class="flex flex-wrap gap-2">
427                            <Show when=move || editable>
428                                <button type="button" class=BUTTON data-mcp-edit on:click={
429                                    let edited = edited.clone();
430                                    move |_| {
431                                        error.set(None);
432                                        form.set(Some((McpFields::edit(&edited), Some(edited.clone()))));
433                                    }
434                                }>
435                                    "Edit"
436                                </button>
437                                <button type="button" class=BUTTON data-mcp-toggle on:click={
438                                    let toggled = toggled.clone();
439                                    move |_| workspace.toggle_mcp(toggled.0.clone(), toggled.1)
440                                }>
441                                    {if toggled.1 { "Enable" } else { "Disable" }}
442                                </button>
443                                <button type="button" class=BUTTON data-mcp-remove on:click={
444                                    let removed = removed.clone();
445                                    move |_| workspace.remove_mcp(removed.clone())
446                                }>
447                                    "Remove"
448                                </button>
449                            </Show>
450                            <button type="button" class=BUTTON data-mcp-reconnect on:click=move |_| workspace.reconnect_mcp(&reconnected)>
451                                "Reconnect"
452                            </button>
453                        </div>
454                    </div>
455                }
456            })
457            .collect_view()
458            .into_any(),
459    }
460    };
461    view! {
462        {move || match form.get() {
463            None => view! {
464                {list}
465                <button
466                    type="button"
467                    id="mcp-new"
468                    class=BUTTON
469                    on:click=move |_| {
470                        error.set(None);
471                        form.set(Some((McpFields::add(), None)));
472                    }
473                >
474                    "New server"
475                </button>
476            }
477                .into_any(),
478            Some((fields, editing)) => mcp_form(workspace, form, persist, error, fields, editing).into_any(),
479        }}
480    }
481}

483fn mcp_form(
484    workspace: Workspace,
485    form: RwSignal<Option<(McpFields, Option<cookie_agent_protocol::McpServerInfo>)>>,
486    persist: RwSignal<PersistChoice>,
487    error: RwSignal<Option<String>>,
488    fields: McpFields,
489    editing: Option<cookie_agent_protocol::McpServerInfo>,
490) -> impl IntoView {
491    let transport = RwSignal::new(fields.transport);
492    let enabled = RwSignal::new(fields.enabled);
493    let extras = RwSignal::new(fields.extras.clone());
494    let editing_name = editing.as_ref().map(|server| server.name.clone());
495    let submit = move || {
496        let mut fields = McpFields {
497            name: field_value("mcp-name"),
498            transport: fields.transport,
499            endpoint: field_value("mcp-endpoint"),
500            extras: field_value("mcp-extras"),
501            environment: field_value("mcp-env"),
502            cwd: field_value("mcp-cwd"),
503            enabled: enabled.get_untracked(),
504        };
505        fields.transport = transport.get_untracked();
506        if fields.transport == McpTransport::Http {
507            fields.environment = "{}".into();
508        }
509        match workspace.submit_mcp(fields, editing.clone(), persist.get_untracked()) {
510            Ok(()) => form.set(None),
511            Err(message) => error.set(Some(message)),
512        }
513    };
514    view! {
515        <p class="text-sm font-bold">{if editing_name.is_some() { "Edit MCP server" } else { "New MCP server" }}</p>
516        <label class="flex flex-col gap-1 text-sm">
517            <span>"Name"</span>
518            <input type="text" id="mcp-name" class=FIELD value=fields.name.clone() readonly=editing_name.is_some() />
519        </label>
520        <label class="flex flex-col gap-1 text-sm">
521            <span>"Transport"</span>
522            <select
523                id="mcp-transport"
524                class=FIELD
525                on:change=move |event| {
526                    let next = if event_target_value(&event) == "http" { McpTransport::Http } else { McpTransport::Stdio };
527                    if next != transport.get_untracked() {
528                        transport.set(next);
529                        extras.set(next.extras().1.to_owned());
530                    }
531                }
532            >
533                <option value="stdio" selected=fields.transport == McpTransport::Stdio>"stdio"</option>
534                <option value="http" selected=fields.transport == McpTransport::Http>"http"</option>
535            </select>
536        </label>
537        <label class="flex flex-col gap-1 text-sm">
538            <span>{move || if transport.get() == McpTransport::Stdio { "Command" } else { "URL" }}</span>
539            <input type="text" id="mcp-endpoint" class=FIELD value=fields.endpoint.clone() />
540        </label>
541        <label class="flex flex-col gap-1 text-sm">
542            <span>{move || transport.get().extras().0}</span>
543            <input type="text" id="mcp-extras" class=FIELD prop:value=move || extras.get() on:input=move |event| extras.set(event_target_value(&event)) />
544        </label>
545        <Show when=move || transport.get() == McpTransport::Stdio>
546            <label class="flex flex-col gap-1 text-sm">
547                <span>"Environment JSON object"</span>
548                <input type="text" id="mcp-env" class=FIELD value=fields.environment.clone() />
549            </label>
550            <label class="flex flex-col gap-1 text-sm">
551                <span>"Working directory"</span>
552                <input type="text" id="mcp-cwd" class=FIELD value=fields.cwd.clone() />
553            </label>
554        </Show>
555        <label class="flex min-h-11 items-center gap-2 text-sm">
556            <input type="checkbox" id="mcp-enabled" class="h-5 w-5" prop:checked=move || enabled.get() on:change=move |_| enabled.update(|on| *on = !*on) />
557            <span>"Enabled"</span>
558        </label>
559        <label class="flex flex-col gap-1 text-sm">
560            <span>"Keep it"</span>
561            <select
562                id="mcp-persist"
563                class=FIELD
564                on:change=move |event| {
565                    persist.set(if event_target_value(&event) == "user" { PersistChoice::User } else { PersistChoice::Runtime });
566                }
567            >
568                <option value="runtime" selected=move || persist.get() == PersistChoice::Runtime>{PersistChoice::Runtime.label()}</option>
569                <option value="user" selected=move || persist.get() == PersistChoice::User>{PersistChoice::User.label()}</option>
570            </select>
571        </label>
572        {move || error.get().map(|message| view! { <p class=ERROR>{message}</p> })}
573        <div class="flex gap-2">
574            <button type="button" id="mcp-save" class=BUTTON on:click=move |_| submit()>
575                "Save"
576            </button>
577            <button type="button" class=BUTTON on:click=move |_| form.set(None)>
578                "Cancel"
579            </button>
580        </div>
581    }
582}

591        {move || match open.get() {
592            Some(provider) => provider_view(workspace, provider, open, confirm_disconnect).into_any(),
593            None => view! {
594                <input
595                    type="search"
596                    id="provider-search"
597                    class=FIELD
598                    placeholder="Search providers"
599                    prop:value=move || search.get()
600                    on:input=move |event| search.set(event_target_value(&event))
601                />
602                {move || workspace
603                    .with_runtime(|runtime| runtime.snapshot().map(catalog_status_line))
604                    .map(|line| view! { <p class=NOTE>{line}</p> })}
605                {move || {
606                    let query = search.get().trim().to_lowercase();
607                    let operations = workspace.provider_operations.get();
608                    workspace
609                        .providers()
610                        .into_iter()
611                        .filter(|provider| {
612                            query.is_empty()
613                                || provider.id.as_str().to_lowercase().contains(&query)
614                                || provider.display_name.to_string().to_lowercase().contains(&query)
615                        })
616                        .map(|provider| {
617                            let operation = operations.get(&provider.id).cloned();
618                            let state = row_state(&provider, &[], operation.as_ref());
619                            let detail = match (&operation, state) {
620                                (Some(ProviderOperation::InProgress(action)), _) => {
621                                    format!("{} in progress…", action_name(*action))
622                                }
623                                (Some(ProviderOperation::Error { message, .. }), _) => format!("error · {message}"),
624                                (_, ProviderRowState::Unsupported) => "unsupported".into(),
625                                (_, ProviderRowState::ConnectedReconnect) => "connected".into(),
626                                (_, ProviderRowState::Removed) => "removed from the current catalog".into(),
627                                _ => "disconnected".into(),
628                            };
629                            let note = unusable_models_note(&provider);
630                            let picked = provider.clone();
631                            view! {
632                                <button
633                                    type="button"
634                                    class=ROW
635                                    data-provider=provider.id.to_string()
636                                    on:click=move |_| {
637                                        confirm_disconnect.set(false);
638                                        open.set(Some(picked.clone()));
639                                    }
640                                >
641                                    <span class="text-sm font-bold">{format!("{} ({})", provider.display_name, provider.id)}</span>
642                                    <span class=NOTE>{detail}</span>
643                                    {note.map(|note| view! { <span class=NOTE>{note}</span> })}
644                                </button>
645                            }
646                        })
647                        .collect_view()
648                }}
649            }
650                .into_any(),
651        }}
652    }

655/// A provider's connect form, or its details when it cannot be connected.
656/// Credentials stay in their password fields: they are read once when the
657/// form is sent and never kept in the page's state.
658fn provider_view(
659    workspace: Workspace,
660    provider: ProviderDescriptor,
661    open: RwSignal<Option<ProviderDescriptor>>,
662    confirm_disconnect: RwSignal<bool>,
663) -> impl IntoView {
664    let state = row_state(&provider, &[], None);
665    let summary = model_counts_summary(&provider);
666    let back = move |_| open.set(None);
667    if state == ProviderRowState::Unsupported {
668        let reason = provider
669            .support
670            .reason
671            .as_ref()
672            .map_or_else(|| "unsupported".to_owned(), ToString::to_string);
673        return view! {
674            <p class="text-sm font-bold">{format!("{} ({})", provider.display_name, provider.id)}</p>
675            <p class="text-sm">{reason}</p>
676            <p class=NOTE>{summary}</p>
677            <button type="button" class=BUTTON on:click=back>"Back"</button>
678        }
679        .into_any();
680    }
681    let Some(initial) = initial_auth_method(&provider).cloned() else {
682        return view! {
683            <p class="text-sm">"This provider has no store-backed authentication form."</p>
684            <button type="button" class=BUTTON on:click=back>"Back"</button>
685        }
686        .into_any();
687    };
688    let method = RwSignal::new(initial.id.clone());
689    let error = RwSignal::new(None::<String>);
690    let submitted = RwSignal::new(false);
691    let reconnect = provider.durable_connection.is_some();
692    let provider_id = provider.id.clone();
693    // A connect that finished without an error closes the form.
694    Effect::new(move |_| {
695        let operation = workspace
696            .provider_operations
697            .with(|operations| operations.get(&provider_id).cloned());
698        if !submitted.get_untracked() {
699            return;
700        }
701        match operation {
702            None => {
703                submitted.set(false);
704                open.set(None);
705            }
706            Some(ProviderOperation::Error { message, .. }) => {
707                submitted.set(false);
708                error.set(Some(message));
709            }
710            Some(ProviderOperation::InProgress(_)) => {}
711        }
712    });
713    let methods = provider.auth_methods.clone();
714    let credential_methods = provider.auth_methods.clone();
715    let setup_fields = provider.setup_fields.clone();
716    let form_provider = provider.clone();
717    let submit = move || {
718        let provider = form_provider.clone();
719        let Some(auth) = provider
720            .auth_methods
721            .iter()
722            .find(|candidate| candidate.id == method.get_untracked())
723            .cloned()
724        else {
725            return;
726        };
727        let typed_setup = provider
728            .setup_fields
729            .iter()
730            .enumerate()
731            .map(|(index, field)| (field, field_value(&format!("setup-{index}"))))
732            .collect::<Vec<_>>();
733        let setup = match setup_values(
734            &typed_setup
735                .iter()
736                .map(|(field, value)| (*field, value.as_str()))
737                .collect::<Vec<_>>(),
738        ) {
739            Ok(values) => values,
740            Err(message) => {
741                error.set(Some(format!("Invalid public setup: {message}")));
742                return;
743            }
744        };
745        let typed = auth
746            .credentials
747            .iter()
748            .enumerate()
749            .map(|(index, field)| (field, field_value(&format!("credential-{index}"))))
750            .collect::<Vec<_>>();
751        let credentials = credential_values(
752            &typed
753                .iter()
754                .map(|(field, value)| (*field, value.as_str()))
755                .collect::<Vec<_>>(),
756        );
757        drop(typed);
758        let credentials = match credentials {
759            Ok(values) => values,
760            Err(message) => {
761                error.set(Some(format!("Invalid credentials: {message}")));
762                return;
763            }
764        };
765        error.set(None);
766        match workspace.connect_provider(provider, auth.id.clone(), setup, credentials, reconnect) {
767            Ok(()) => submitted.set(true),
768            Err(message) => error.set(Some(message)),
769        }
770    };
771    let disconnect_provider = provider.clone();
772    view! {
773        <p class="text-sm font-bold">{format!("{} ({})", provider.display_name, provider.id)}</p>
774        <p class=NOTE>{summary}</p>
775        <p class=NOTE>{cookie_agent_client_state::providers::DURABLE_PROVIDER_COPY}</p>
776        <Show when={
777            let count = methods.len();
778            move || count > 1
779        }>
780            <label class="flex flex-col gap-1 text-sm">
781                <span>"Authentication"</span>
782                <select
783                    id="provider-auth-method"
784                    class=FIELD
785                    on:change=move |event| {
786                        if let Ok(id) = cookie_agent_protocol::AuthMethodId::new(event_target_value(&event)) {
787                            method.set(id);
788                        }
789                    }
790                >
791                    {methods
792                        .iter()
793                        .map(|candidate| {
794                            let id = candidate.id.clone();
795                            view! {
796                                <option value=candidate.id.to_string() selected=move || method.get() == id>
797                                    {candidate.display_name.to_string()}
798                                </option>
799                            }
800                        })
801                        .collect_view()}
802                </select>
803            </label>
804        </Show>
805        {move || {
806            let current = method.get();
807            credential_methods
808                .iter()
809                .find(|candidate| candidate.id == current)
810                .map(|auth| {
811                    auth.credentials
812                        .iter()
813                        .enumerate()
814                        .map(|(index, field)| view! {
815                            <label class="flex flex-col gap-1 text-sm">
816                                <span>{format!("{}{}", field.display_name, if field.required { "" } else { " (optional)" })}</span>
817                                <input
818                                    type="password"
819                                    id=format!("credential-{index}")
820                                    class=FIELD
821                                    autocomplete="off"
822                                    spellcheck="false"
823                                />
824                                <span class=NOTE>{field.help.to_string()}</span>
825                            </label>
826                        })
827                        .collect_view()
828                })
829        }}
830        {setup_fields
831            .iter()
832            .enumerate()
833            .map(|(index, field)| view! {
834                <label class="flex flex-col gap-1 text-sm">
835                    <span>{format!("{}{}", field.display_name, if field.required { "" } else { " (optional)" })}</span>
836                    <input
837                        type="text"
838                        id=format!("setup-{index}")
839                        class=FIELD
840                        value=initial_setup_text(&disconnect_provider, field)
841                    />
842                    <span class=NOTE>{field.help.to_string()}</span>
843                </label>
844            })
845            .collect_view()}
846        {move || error.get().map(|message| view! { <p class=ERROR id="provider-error">{message}</p> })}
847        <div class="flex flex-wrap gap-2">
848            <button type="button" id="provider-connect" class=BUTTON on:click=move |_| submit()>
849                {if reconnect { "Reconnect" } else { "Connect" }}
850            </button>
851            <button type="button" class=BUTTON on:click=back>"Cancel"</button>
852            <Show when=move || reconnect>
853                <button type="button" id="provider-disconnect" class=BUTTON on:click=move |_| confirm_disconnect.set(true)>
854                    "Disconnect"
855                </button>
856            </Show>
857        </div>
858        <Show when=move || confirm_disconnect.get()>
859            <div class="flex flex-col gap-2 rounded border border-caramel p-2 dark:border-maple">
860                <p class="text-sm">"Remove the stored connection and its credentials?"</p>
861                <div class="flex gap-2">
862                    <button type="button" id="provider-disconnect-confirm" class=BUTTON on:click={
863                        let provider = provider.clone();
864                        move |_| {
865                            confirm_disconnect.set(false);
866                            open.set(None);
867                            workspace.disconnect_provider(provider.clone());
868                        }
869                    }>
870                        "Disconnect"
871                    </button>
872                    <button type="button" class=BUTTON on:click=move |_| confirm_disconnect.set(false)>"Keep it"</button>
873                </div>
874            </div>
875        </Show>
876    }
877    .into_any()
878}
