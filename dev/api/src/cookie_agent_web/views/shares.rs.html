18/// When a link stops working, in this browser's time zone.
19fn expiry_text(share: &Share) -> String {
20    share.expires_at.map_or_else(
21        || "Never expires".to_owned(),
22        |at| {
23            format!(
24                "Expires {}",
25                at.to_zoned(jiff::tz::TimeZone::system())
26                    .strftime("%b %-d, %Y, %H:%M")
27            )
28        },
29    )
30}
31
32/// "3 views", and when the last was.
33fn views_text(share: &Share) -> String {
34    let views = format!(
35        "{} view{}",
36        share.views,
37        if share.views == 1 { "" } else { "s" }
38    );
39    match share.last_viewed {
40        Some(at) => format!(
41            "{views}, last {}",
42            at.to_zoned(jiff::tz::TimeZone::system())
43                .strftime("%b %-d, %H:%M")
44        ),
45        None => views,
46    }
47}

49/// The expiry choices, as a select whose change calls `chosen`: `selected`
50/// chosen, or without one a "Change expiry…" prompt.
51fn lifetime_select(
52    id: Option<&'static str>,
53    selected: Option<ShareLifetime>,
54    chosen: impl Fn(ShareLifetime) + 'static,
55) -> impl IntoView {
56    view! {
57        <select
58            id=id
59            class=SELECT
60            aria-label="When the link expires"
61            data-share-lifetime
62            on:change=move |event| {
63                if let Some(lifetime) = ShareLifetime::from_code(&event_target_value(&event)) {
64                    chosen(lifetime);
65                }
66            }
67        >
68            {selected
69                .is_none()
70                .then(|| view! { <option value="" selected disabled>"Change expiry…"</option> })}
71            {ShareLifetime::ALL
72                .into_iter()
73                .map(|lifetime| {
74                    view! {
75                        <option value=lifetime.code() selected=selected == Some(lifetime)>
76                            {lifetime.label()}
77                        </option>
78                    }
79                })
80                .collect_view()}
81        </select>
82    }
83}

264/// The Sharing view: every live link, newest first, with *Copy*, *Change
265/// expiry*, and *Revoke*, and *Revoke all*.
266pub fn sharing_page(workspace: Workspace) -> impl IntoView {
267    Effect::new(move |_| workspace.refresh_shares());
268    let list = move || match workspace.share_list.get() {
269        None => loading("Listing share links…").into_any(),
270        Some(Err(error)) => view! { <p class=ERROR>{error}</p> }.into_any(),
271        Some(Ok(listed)) => {
272            let off = (!listed.enabled).then(|| {
273                view! {
274                    <p class=ERROR id="shares-off">
275                        "Share links are off on this daemon ([share] enabled = false): no link works until they are on again."
276                    </p>
277                }
278            });
279            if listed.shares.is_empty() {
280                return view! {
281                    {off}
282                    {empty_state(
283                        icondata::LuLink,
284                        "No share links. Share an asset from its card, or from the Assets view, to give it a public link.",
285                    )}
286                }
287                .into_any();
288            }
289            let count = listed.shares.len();
290            view! {
291                {off}
292                <div class="flex flex-wrap items-center gap-2">
293                    <p class=NOTE id="shares-count">
294                        {format!("{count} live link{}", if count == 1 { "" } else { "s" })}
295                    </p>
296                    <span class="flex-1"></span>
297                    <Show
298                        when=move || workspace.share_revoking_all.get()
299                        fallback=move || {
300                            view! {
301                                <button
302                                    type="button"
303                                    id="shares-revoke-all"
304                                    class=format!("{BUTTON} text-bad")
305                                    on:click=move |_| workspace.share_revoking_all.set(true)
306                                >
307                                    "Revoke all"
308                                </button>
309                            }
310                        }
311                    >
312                        <span class="flex flex-wrap items-center gap-2" id="shares-revoke-all-confirm">
313                            <span class="text-sm">"Revoke every link? They stop working at once."</span>
314                            <button
315                                type="button"
316                                id="shares-revoke-all-yes"
317                                class=format!("{BUTTON} border-bad text-bad")
318                                on:click=move |_| workspace.revoke_all_shares()
319                            >
320                                "Revoke all"
321                            </button>
322                            <button
323                                type="button"
324                                class=BUTTON
325                                on:click=move |_| workspace.share_revoking_all.set(false)
326                            >
327                                "Keep"
328                            </button>
329                        </span>
330                    </Show>
331                </div>
332                <div class="flex flex-col">
333                    {listed
334                        .shares
335                        .into_iter()
336                        .map(|share| share_row(workspace, share))
337                        .collect_view()}
338                </div>
339            }
340            .into_any()
341        }
342    };
343    view! {
344        <p class="text-sm text-muted">
345            "Public links to assets: anyone with one can open its asset in a browser until it expires or you revoke it."
346        </p>
347        {list}
348    }
349}

351/// Who made a link: the user, or an agent in a session.
352fn made_by(share: &Share) -> String {
353    let at = share
354        .created
355        .at
356        .to_zoned(jiff::tz::TimeZone::system())
357        .strftime("%b %-d, %H:%M")
358        .to_string();
359    match &share.created.agent {
360        Some(agent) => format!("Made by {agent} on {at}"),
361        None => format!("Made by you on {at}"),
362    }
363}

365fn share_row(workspace: Workspace, share: Share) -> impl IntoView {
366    let address = workspace.share_address(&share);
367    let copied = share.clone();
368    let revoked = share.id.clone();
369    let changed = share.id.clone();
370    let (store, asset, title) = (share.store.clone(), share.asset, share.title.clone());
371    view! {
372        <div class="flex flex-col gap-1 border-b border-line py-2" data-share-row=share.id.to_string()>
373            <div class="flex min-w-0 items-center gap-2">
374                <Glyph icon=icondata::LuLink class="text-accent" />
375                <button
376                    type="button"
377                    class="min-w-0 flex-1 truncate text-left text-sm font-bold hover:underline"
378                    data-share-title
379                    title="Open the share dialog"
380                    on:click=move |_| workspace.open_share_dialog(store.clone(), asset, title.clone())
381                >
382                    {share.title.clone()}
383                </button>
384                <span class=NOTE>
385                    {format!("{} · {}", share.media_type, cookie_agent_protocol::human_bytes(share.byte_length))}
386                </span>
387            </div>
388            <p class="truncate font-mono text-xs text-muted select-all" data-share-url>{address}</p>
389            <p class=NOTE data-share-meta>
390                {format!("{} · {} · {}", expiry_text(&share), views_text(&share), made_by(&share))}
391            </p>
392            <div class="flex flex-wrap items-center gap-1">
393                <button
394                    type="button"
395                    class=ROW_BUTTON
396                    data-share-action="copy"
397                    on:click=move |_| workspace.copy_share(&copied)
398                >
399                    <Glyph icon=icondata::LuCopy />
400                    "Copy"
401                </button>
402                {lifetime_select(
403                    None,
404                    None,
405                    move |lifetime| workspace.set_share_lifetime(changed.clone(), lifetime),
406                )}
407                <span class="flex-1"></span>
408                <button
409                    type="button"
410                    class=format!("{ROW_BUTTON} text-bad")
411                    data-share-action="revoke"
412                    on:click=move |_| workspace.revoke_share(revoked.clone())
413                >
414                    <Glyph icon=icondata::LuTrash2 />
415                    "Revoke"
416                </button>
417            </div>
418        </div>
419    }
