18/// When a link stops working, in this browser's time zone.
19fn expiry_text(share: &Share) -> String {
20    expiry_text_at(share.expires_at)
21}
22
23fn expiry_text_at(expires_at: Option<jiff::Timestamp>) -> String {
24    expires_at.map_or_else(
25        || "Never expires".to_owned(),
26        |at| {
27            format!(
28                "Expires {}",
29                at.to_zoned(jiff::tz::TimeZone::system())
30                    .strftime("%b %-d, %Y, %H:%M")
31            )
32        },
33    )
34}

36/// "3 views", and when the last was.
37fn views_text(share: &Share) -> String {
38    let views = format!(
39        "{} view{}",
40        share.views,
41        if share.views == 1 { "" } else { "s" }
42    );
43    match share.last_viewed {
44        Some(at) => format!(
45            "{views}, last {}",
46            at.to_zoned(jiff::tz::TimeZone::system())
47                .strftime("%b %-d, %H:%M")
48        ),
49        None => views,
50    }
51}

53/// The expiry choices, as a select whose change calls `chosen`: `selected`
54/// chosen, or without one a "Change expiry…" prompt.
55fn lifetime_select(
56    id: Option<&'static str>,
57    selected: Option<ShareLifetime>,
58    chosen: impl Fn(ShareLifetime) + 'static,
59) -> impl IntoView {
60    view! {
61        <select
62            id=id
63            class=SELECT
64            aria-label="When the link expires"
65            data-share-lifetime
66            on:change=move |event| {
67                if let Some(lifetime) = ShareLifetime::from_code(&event_target_value(&event)) {
68                    chosen(lifetime);
69                }
70            }
71        >
72            {selected
73                .is_none()
74                .then(|| view! { <option value="" selected disabled>"Change expiry…"</option> })}
75            {ShareLifetime::ALL
76                .into_iter()
77                .map(|lifetime| {
78                    view! {
79                        <option value=lifetime.code() selected=selected == Some(lifetime)>
80                            {lifetime.label()}
81                        </option>
82                    }
83                })
84                .collect_view()}
85        </select>
86    }
87}

268/// When a guest's link stops working, and what it allows, for the guest
269/// page's top bar.
270fn guest_bar_text(guest: &cookie_agent_protocol::GuestHello) -> String {
271    let who = if guest.can_send {
272        format!("Shared with you as {}", guest.label)
273    } else {
274        format!("Shared with you as {} · watch only", guest.label)
275    };
276    match guest.expires_at {
277        Some(at) => format!(
278            "{who} · link expires {}",
279            at.to_zoned(jiff::tz::TimeZone::system()).strftime("%b %-d")
280        ),
281        None => who,
282    }
283}

301/// The title bar's badge of a session with a live link: how many guests are
302/// connected, and a click opens the share dialog.
303#[component]
304pub(crate) fn LinkBadge(workspace: Workspace) -> impl IntoView {
305    move || {
306        if workspace.is_guest() || workspace.new_session.get() {
307            return None;
308        }
309        let session = workspace.root_of(workspace.selected.get()?);
310        let links = workspace.session_links(session);
311        if links.is_empty() {
312            return None;
313        }
314        let connected: u32 = links.iter().map(|link| link.connected).sum();
315        let title = match connected {
316            0 => "This conversation has a share link".to_owned(),
317            1 => "A guest is connected through a share link".to_owned(),
318            count => format!("{count} guests are connected through share links"),
319        };
320        Some(view! {
321            <button
322                type="button"
323                id="session-link-badge"
324                class="inline-flex min-h-9 shrink-0 items-center gap-1 rounded-full bg-accent-soft px-2 text-xs text-accent hover:bg-hover"
325                title=title.clone()
326                aria-label=title
327                on:click=move |_| workspace.open_session_share_dialog_of(session)
328            >
329                <Glyph icon=icondata::LuLink />
330                {(connected > 0).then(|| view! { <span data-guests-connected>{connected}</span> })}
331            </button>
332        })
333    }
334}

801/// The Sharing view: every live link, newest first, with *Copy*, *Change
802/// expiry*, and *Revoke*, and *Revoke all*.
803pub fn sharing_page(workspace: Workspace) -> impl IntoView {
804    Effect::new(move |_| workspace.refresh_shares());
805    let list = move || match workspace.share_list.get() {
806        None => loading("Listing share links…").into_any(),
807        Some(Err(error)) => view! { <p class=ERROR>{error}</p> }.into_any(),
808        Some(Ok(listed)) => {
809            let off = (!listed.enabled).then(|| {
810                view! {
811                    <p class=ERROR id="shares-off">
812                        "Share links are off on this daemon ([share] enabled = false): no link works until they are on again."
813                    </p>
814                }
815            });
816            if listed.shares.is_empty() && listed.sessions.is_empty() {
817                return view! {
818                    {off}
819                    {empty_state(
820                        icondata::LuLink,
821                        "No share links. Share an asset from its card, or a conversation from its ⋮ menu, to give it a public link.",
822                    )}
823                }
824                .into_any();
825            }
826            let count = listed.shares.len() + listed.sessions.len();
827            view! {
828                {off}
829                <div class="flex flex-wrap items-center gap-2">
830                    <p class=NOTE id="shares-count">
831                        {format!("{count} live link{}", if count == 1 { "" } else { "s" })}
832                    </p>
833                    <span class="flex-1"></span>
834                    <Show
835                        when=move || workspace.share_revoking_all.get()
836                        fallback=move || {
837                            view! {
838                                <button
839                                    type="button"
840                                    id="shares-revoke-all"
841                                    class=format!("{BUTTON} text-bad")
842                                    on:click=move |_| workspace.share_revoking_all.set(true)
843                                >
844                                    "Revoke all"
845                                </button>
846                            }
847                        }
848                    >
849                        <span class="flex flex-wrap items-center gap-2" id="shares-revoke-all-confirm">
850                            <span class="text-sm">"Revoke every link? They stop working at once."</span>
851                            <button
852                                type="button"
853                                id="shares-revoke-all-yes"
854                                class=format!("{BUTTON} border-bad text-bad")
855                                on:click=move |_| workspace.revoke_all_shares()
856                            >
857                                "Revoke all"
858                            </button>
859                            <button
860                                type="button"
861                                class=BUTTON
862                                on:click=move |_| workspace.share_revoking_all.set(false)
863                            >
864                                "Keep"
865                            </button>
866                        </span>
867                    </Show>
868                </div>
869                <div class="flex flex-col">

878                        .map(|share| share_row(workspace, share))
879                        .collect_view()}

881            }

883        }

886        {super::view_header(
887            "Sharing",
888            "Public links to assets and conversations: anyone with one can open it in a browser until it expires or you revoke it.",
889            (),
890        )}
891        {list}
892    }

895/// Who made a link: the user, or an agent in a session.
896fn made_by(share: &Share) -> String {
897    let at = share
898        .created
899        .at
900        .to_zoned(jiff::tz::TimeZone::system())
901        .strftime("%b %-d, %H:%M")
902        .to_string();
903    match &share.created.agent {
904        Some(agent) => format!("Made by {agent} on {at}"),
905        None => format!("Made by you on {at}"),
906    }
907}
