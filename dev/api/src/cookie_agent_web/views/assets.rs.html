23/// A choice of the filter or the kinds: a chip, filled while chosen.
24fn chip(
25    label: String,
26    chosen: Signal<bool>,
27    choose: impl Fn() + Send + Sync + 'static,
28    data: &'static str,
29) -> impl IntoView {
30    let value = label.clone();
31    view! {
32        <button
33            type="button"
34            role="radio"
35            class=move || format!("{CHIP} {}", if chosen.get() { CHIP_ON } else { CHIP_OFF })
36            aria-checked=move || chosen.get().to_string()
37            data-asset-filter=data
38            data-value=value
39            on:click=move |_| choose()
40        >
41            <Show when=move || chosen.get()>
42                <Glyph icon=icondata::LuCheck class="text-xs" />
43            </Show>
44            <span class="truncate">{label}</span>
45        </button>
46    }
47}

49/// The filter: whose assets (every project's, no project's, or one
50/// project's), and of which level.
51fn filter_panel(workspace: Workspace) -> impl IntoView {
52    let heading = "text-xs font-bold text-muted";
53    let project_chip =
54        move |label: String, project: Option<AssetProjectFilter>, data: &'static str| {
55            let wanted = project.clone();
56            chip(
57                label,
58                Signal::derive(move || {
59                    workspace
60                        .asset_filter
61                        .with(|filter| filter.project == wanted)
62                }),
63                move || {
64                    let mut filter = workspace.asset_filter.get_untracked();
65                    filter.project = project.clone();
66                    workspace.choose_asset_filter(filter);
67                },
68                data,
69            )
70        };
71    let projects = move || {
72        workspace
73            .projects
74            .get()
75            .into_iter()
76            .map(|project| {
77                project_chip(
78                    project.name.to_string(),
79                    Some(AssetProjectFilter::Project {
80                        project: project.id.clone(),
81                    }),
82                    "project",
83                )
84            })
85            .collect_view()
86    };
87    view! {
88        <div
89            id="assets-filter"
90            role="dialog"
91            aria-label="Filter assets"
92            class="flex flex-col gap-3 rounded-xl p-3 bg-raised shadow-[0_0_0_1px_var(--color-line),0_1px_2px_rgb(0_0_0/0.03)]"
93        >
94            <div class="flex flex-col gap-1.5">
95                <span class=heading>"Project"</span>
96                <div class="flex flex-wrap gap-1.5" role="radiogroup" aria-label="Project">
97                    {project_chip("All projects".into(), None, "project")}
98                    {project_chip("No project".into(), Some(AssetProjectFilter::NoProject), "project")}
99                    {projects}
100                </div>
101            </div>
102            <div class="flex flex-col gap-1.5">
103                <span class=heading>"Level"</span>
104                <div class="flex flex-wrap gap-1.5" role="radiogroup" aria-label="Level">
105                    {AssetFilter::LEVELS
106                        .into_iter()
107                        .map(|scope| {
108                            chip(
109                                AssetFilter::level_label(scope).to_owned(),
110                                Signal::derive(move || workspace.asset_filter.with(|filter| filter.scope == scope)),
111                                move || {
112                                    let mut filter = workspace.asset_filter.get_untracked();
113                                    filter.scope = scope;
114                                    workspace.choose_asset_filter(filter);
115                                },
116                                "level",
117                            )
118                        })
119                        .collect_view()}
120                </div>
121            </div>
122        </div>
123    }
124}

126/// What the filter lists, in words: "All projects · All levels".
127fn filter_summary(workspace: Workspace) -> String {
128    let filter = workspace.asset_filter.get();
129    let project = match &filter.project {
130        None => "All projects".to_owned(),
131        Some(AssetProjectFilter::NoProject) => "No project".to_owned(),
132        Some(AssetProjectFilter::Project { project }) => workspace
133            .projects
134            .with(|projects| {
135                projects
136                    .iter()
137                    .find(|known| known.id == *project)
138                    .map(|known| known.name.to_string())
139            })
140            .unwrap_or_else(|| project.to_string()),
141    };
142    format!("{project} · {}", AssetFilter::level_label(filter.scope))
143}

145pub fn assets_page(workspace: Workspace) -> impl IntoView {
146    let kind_button = move |kind: Option<AssetKind>| {
147        let label = kind.map_or("All", AssetKind::label);
148        let chosen = move || workspace.asset_kind.get() == kind;
149        view! {
150            <button
151                type="button"
152                role="radio"
153                class=move || format!("{CHIP} {}", if chosen() { CHIP_ON } else { CHIP_OFF })
154                aria-checked=move || chosen().to_string()
155                data-asset-kind=label
156                on:click=move |_| workspace.asset_kind.set(kind)
157            >
158                {label}
159            </button>
160        }
161    };
162    let project_name = move |project: &cookie_agent_protocol::ProjectId| {
163        workspace.projects.with_untracked(|projects| {
164            projects
165                .iter()
166                .find(|known| known.id == *project)
167                .map(|known| known.name.to_string())
168        })
169    };
170    let list = move || match workspace.asset_list.get() {
171        None => loading("Reading the assets…").into_any(),
172        Some(Err(message)) => view! { <p class=ERROR>{message}</p> }.into_any(),
173        Some(Ok(found)) if found.assets.is_empty() => empty_state(
174            icondata::LuImages,
175            "No assets here yet. Agents save them with write_asset.",
176        )
177        .into_any(),
178        Some(Ok(found)) => {
179            let query = workspace.asset_query.get();
180            let shown = assets_matching(&found.assets, &query, workspace.asset_kind.get())
181                .into_iter()
182                .cloned()
183                .collect::<Vec<_>>();
184            if shown.is_empty() {
185                return empty_state(icondata::LuSearch, "No assets match.").into_any();
186            }
187            let total = view! {
188                <p class=NOTE data-assets-total>
189                    {format!(
190                        "{} asset{} · {}",
191                        shown.len(),
192                        if shown.len() == 1 { "" } else { "s" },
193                        human_bytes(total_bytes(&shown))
194                    )}
195                </p>
196            };
197            let rows = workspace.asset_rows.get();
198            let more = shown.len().saturating_sub(rows);
199            let more = (more > 0).then(|| {
200                view! {
201                    <button
202                        type="button"
203                        class=BUTTON
204                        id="assets-more"
205                        on:click=move |_| {
206                            workspace
207                                .asset_rows
208                                .update(|rows| *rows += crate::workspace::ASSET_ROWS);
209                        }
210                    >
211                        {format!("Show more ({more} more)")}
212                    </button>
213                }
214            });
215            view! {
216                {total}
217                <div class="flex flex-col">
218                    {shown
219                        .into_iter()
220                        .take(rows)
221                        .map(|asset| {
222                            let place = asset_place(&asset, &found.sessions, project_name);
223                            asset_row(workspace, asset, place)
224                        })
225                        .collect_view()}
226                </div>
227                {more}
228            }
229            .into_any()
230        }
231    };
232    // The figure of the asset opened, filled once it is drawn.
233    Effect::new(move |_| {
234        if workspace.asset_open.get().is_some() {
235            request_animation_frame(move || {
236                if let Some(root) = document().get_element_by_id("assets-page") {
237                    crate::assets::show_within(&root, workspace);
238                }
239            });
240        }
241    });
242    let filter_open = move || workspace.asset_filter_open.get();
243    let narrowed = move || workspace.asset_filter.with(AssetFilter::changes) > 0;
244    view! {
245        <div class="flex flex-col gap-2" id="assets-page" data-assets-page>
246            <p class=NOTE>
247                "Files agents saved: a session's own, and those kept for a project, which every session of the project can use."
248            </p>
249            <div class="flex min-w-0 items-center gap-2">
250                <button
251                    type="button"
252                    class=format!("{BUTTON} relative")
253                    id="assets-filter-toggle"
254                    aria-label="Filter assets"
255                    title="Filter assets"
256                    aria-haspopup="dialog"
257                    aria-expanded=move || filter_open().to_string()
258                    aria-controls="assets-filter"
259                    on:click=move |_| workspace.asset_filter_open.update(|open| *open = !*open)
260                >
261                    <Glyph icon=icondata::LuListFilter />
262                    // A dot while the filter narrows the list.
263                    <Show when=narrowed>
264                        <span class="absolute top-1.5 right-1.5 h-2 w-2 rounded-full bg-accent ring-2 ring-surface" data-filter-dot></span>
265                    </Show>
266                </button>
267                <span class=format!("min-w-0 flex-1 truncate {NOTE}") data-assets-summary>
268                    {move || filter_summary(workspace)}
269                </span>
270            </div>
271            <Show when=filter_open>
272                {filter_panel(workspace)}
273            </Show>
274            <div class="flex flex-wrap gap-1.5" role="radiogroup" aria-label="Kind">
275                {kind_button(None)}
276                {AssetKind::ALL.into_iter().map(|kind| kind_button(Some(kind))).collect_view()}
277            </div>
278            <label class="relative flex items-center">
279                <Glyph icon=icondata::LuSearch class="pointer-events-none absolute left-3 text-xs text-muted" />
280                <input
281                    type="text"
282                    role="searchbox"
283                    id="assets-search"
284                    class="min-h-11 w-full min-w-0 rounded-lg bg-surface pr-3 pl-9 text-sm text-fg hover:shadow-[inset_0_0_0_1px_var(--color-faint)] focus:shadow-[inset_0_0_0_1px_var(--color-accent),0_0_0_3px_var(--color-accent-soft)] focus:outline-none shadow-[inset_0_0_0_1px_var(--color-line-2)]"
285                    aria-label="Search assets by title or file name"
286                    placeholder="Search by title or file name"
287                    prop:value=move || workspace.asset_query.get_untracked()
288                    on:input=move |event| {
289                        workspace.asset_query.set(event_target_value(&event));
290                        workspace.asset_rows.set(crate::workspace::ASSET_ROWS);
291                    }
292                />
293            </label>
294            {list}
295        </div>
296    }
297}

299/// Who last saved `asset`, and when: the session's title when the page
300/// lists it.
301fn saved_by(
302    workspace: Workspace,
303    asset: &Asset,
304) -> (String, Option<cookie_agent_protocol::SessionId>) {
305    let session = asset.updated.session;
306    let title = workspace.sessions.with_untracked(|sessions| {
307        sessions
308            .iter()
309            .find(|meta| meta.session_id == session)
310            .map(|meta| {
311                meta.title.as_ref().map_or_else(
312                    || meta.short_id.clone().unwrap_or_else(|| session.to_string()),
313                    |title| title.as_str().to_owned(),
314                )
315            })
316    });
317    let age = cookie_agent_protocol::memory_age(asset.updated.at, jiff::Timestamp::now());
318    match title {
319        Some(title) => (format!("saved by {title}, {age} ago"), Some(session)),
320        None => (format!("saved by another session, {age} ago"), None),
321    }
322}

324/// One asset: its kind, title, and size; who saved it and when; open,
325/// rename, download, and delete.
326fn asset_row(workspace: Workspace, asset: Asset, place: AssetPlace) -> impl IntoView {
327    let kind = AssetKind::of(&asset.media_type);
328    let id = asset.id;
329    let store = asset.store.clone();
330    let (saved, session) = saved_by(workspace, &asset);
331    let open = move || workspace.asset_open.get() == Some(id);
332    let renaming = move || {
333        workspace
334            .asset_renaming
335            .with(|renaming| renaming.as_ref().is_some_and(|(asset, _)| *asset == id))
336    };
337    let deleting = move || workspace.asset_deleting.get() == Some(id);
338    let block = asset.block();
339    let title = asset.title.clone();
340    let rename_store = store.clone();
341    let delete_store = store.clone();
342    let share_store = store.clone();
343    let share_row_title = title.clone();
344    let shared = move || {
345        workspace.share_list.with(|list| {
346            list.as_ref()
347                .and_then(|list| list.as_ref().ok())
348                .is_some_and(|list| list.shares.iter().any(|share| share.asset == id))
349        })
350    };
351    let download_asset = asset.clone();
352    let saved_link = match session {
353        Some(session) => view! {
354            <button
355                type="button"
356                class=format!("{NOTE} underline decoration-dotted underline-offset-2 hover:text-fg")
357                data-asset-session=session.to_string()
358                on:click=move |_| {
359                    workspace.open_session(session);
360                    workspace.open_view(crate::layout::View::Chat, None);
361                }
362            >
363                {saved.clone()}
364            </button>
365        }
366        .into_any(),
367        None => view! { <span class=NOTE>{saved.clone()}</span> }.into_any(),
368    };
369    // Where it lives: its project, and a session's own its session.
370    let level = if place.session.is_some() {
371        "session"
372    } else {
373        "project"
374    };
375    let tree = place.session.map(|(session, title)| {
376        // Untitled, as the session list names it.
377        let title = title.unwrap_or_else(|| {
378            workspace.sessions.with_untracked(|sessions| {
379                sessions
380                    .iter()
381                    .find(|meta| meta.session_id == session)
382                    .map_or_else(|| "Untitled session".to_owned(), super::session_label)
383            })
384        });
385        view! {
386            <span class=NOTE>"·"</span>
387            <button
388                type="button"
389                class=format!("{NOTE} min-w-0 truncate underline decoration-dotted underline-offset-2 hover:text-fg")
390                data-asset-tree=session.to_string()
391                title="Open the session"
392                on:click=move |_| {
393                    workspace.open_session(session);
394                    workspace.open_view(crate::layout::View::Chat, None);
395                }
396            >
397                {title}
398            </button>
399        }
400    });
401    // A right-click offers the asset's actions where the pointer is.
402    let context = RwSignal::new(None);
403    let context_items = {
404        let title = title.clone();
405        let download_asset = download_asset.clone();
406        let share_store = share_store.clone();
407        move || {
408            use super::row_menu::MenuItem;
409            let (title, download_asset) = (title.clone(), download_asset.clone());
410            let (share_store, share_title) = (share_store.clone(), title.clone());
411            vec![
412                MenuItem {
413                    id: "open",
414                    label: "Open",
415                    icon: icondata::LuEye,
416                    act: Callback::new(move |()| workspace.asset_open.set(Some(id))),
417                    danger: false,
418                },
419                MenuItem {
420                    id: "rename",
421                    label: "Rename",
422                    icon: icondata::LuPencil,
423                    act: Callback::new(move |()| {
424                        workspace.asset_renaming.set(Some((id, title.clone())))
425                    }),
426                    danger: false,
427                },
428                MenuItem {
429                    id: "download",
430                    label: "Download",
431                    icon: icondata::LuDownload,
432                    act: Callback::new(move |()| {
433                        crate::assets::download(workspace, download_asset.clone())
434                    }),
435                    danger: false,
436                },
437                MenuItem {
438                    id: "share",
439                    label: "Share link…",
440                    icon: icondata::LuShare2,
441                    act: Callback::new(move |()| {
442                        workspace.open_share_dialog(share_store.clone(), id, share_title.clone())
443                    }),
444                    danger: false,
445                },
446                MenuItem {
447                    id: "delete",
448                    label: "Delete…",
449                    icon: icondata::LuTrash2,
450                    act: Callback::new(move |()| workspace.asset_deleting.set(Some(id))),
451                    danger: true,
452                },
453            ]
454        }
455    };
456    view! {
457        {super::row_menu::context_menu(context, (), title.clone(), context_items)}
458        <div
459            class="flex flex-col gap-1 border-b border-line py-2"
460            data-asset-row=asset.handle.to_string()
461            on:contextmenu=super::row_menu::on_context(context)
462        >
463            <div class="flex min-w-0 flex-wrap items-center gap-2">
464                <Glyph icon=crate::icons::asset_icon(kind) class="text-accent" />
465                <span class="min-w-0 flex-1 truncate text-sm font-bold" data-asset-row-title>{title.clone()}</span>
466                <span class=NOTE>{format!("{} · {}", asset.media_type, human_bytes(asset.byte_length))}</span>
467            </div>
468            <div class="flex min-w-0 items-center gap-1" data-asset-place=level>
469                <Glyph
470                    icon=if level == "session" { icondata::LuMessagesSquare } else { icondata::LuFolder }
471                    class="text-xs text-muted"
472                />
473                <span class=format!("{NOTE} shrink-0") data-asset-project>{place.project}</span>
474                {tree}
475            </div>
476            <div class="flex min-w-0 flex-wrap items-center gap-1">
477                {saved_link}
478                <span class="flex-1"></span>
479                <button
480                    type="button"
481                    class=ROW_BUTTON
482                    data-asset-action="open"
483                    aria-expanded=move || open().to_string()
484                    on:click=move |_| workspace.asset_open.set(if open() { None } else { Some(id) })
485                >
486                    <Glyph icon=icondata::LuEye />
487                    {move || if open() { "Close" } else { "Open" }}
488                </button>
489                <button
490                    type="button"
491                    class=ROW_BUTTON
492                    data-asset-action="rename"
493                    on:click={
494                        let title = title.clone();
495                        move |_| workspace.asset_renaming.set(Some((id, title.clone())))
496                    }
497                >
498                    <Glyph icon=icondata::LuPencil />
499                    "Rename"
500                </button>
501                <button
502                    type="button"
503                    class=ROW_BUTTON
504                    data-asset-action="download"
505                    on:click=move |_| crate::assets::download(workspace, download_asset.clone())
506                >
507                    <Glyph icon=icondata::LuDownload />
508                    "Download"
509                </button>
510                <Show when=move || workspace.has_shares.get()>
511                    <button
512                        type="button"
513                        class=move || {
514                            if shared() {
515                                format!("{ROW_BUTTON} text-accent")
516                            } else {
517                                ROW_BUTTON.to_owned()
518                            }
519                        }
520                        data-asset-action="share"
521                        data-shared=move || shared().to_string()

523                            let (store, title) = (store.clone(), share_row_title.clone());

525                        }
526                    >
527                        <Glyph icon=icondata::LuShare2 />
528                        {move || if shared() { "Shared" } else { "Share" }}
529                    </button>
530                </Show>
531                <button
532                    type="button"
533                    class=ROW_BUTTON
534                    data-asset-action="delete"
535                    on:click=move |_| workspace.asset_deleting.set(Some(id))
536                >
537                    <Glyph icon=icondata::LuTrash2 />
538                    "Delete"
539                </button>
540            </div>
541            <Show when=renaming>
542                {
543                    let rename_store = rename_store.clone();

546                            class="flex flex-wrap items-center gap-2"

548                            on:submit={
549                                let rename_store = rename_store.clone();
550                                move |event: leptos::ev::SubmitEvent| {
551                                    event.prevent_default();
552                                    let title = workspace
553                                        .asset_renaming
554                                        .get_untracked()
555                                        .map(|(_, title)| title)
556                                        .unwrap_or_default();
557                                    workspace.rename_asset(rename_store.clone(), id, title);
558                                }
559                            }
560                        >
561                            <input
562                                type="text"
563                                id="asset-rename-title"
564                                class="min-h-11 min-w-0 flex-1 rounded-lg bg-surface px-3 text-sm shadow-[inset_0_0_0_1px_var(--color-line-2)]"
565                                maxlength="200"
566                                aria-label="The asset's title"
567                                prop:value=move || {
568                                    workspace
569                                        .asset_renaming
570                                        .with(|renaming| renaming.as_ref().map(|(_, title)| title.clone()))
571                                        .unwrap_or_default()
572                                }
