57/// The filter: whose assets (every project's, no project's, or one
58/// project's), and of which level.
59fn filter_panel(workspace: Workspace) -> impl IntoView {
60    let heading = "text-xs font-bold text-muted";
61    let project_chip =
62        move |label: String, project: Option<AssetProjectFilter>, data: &'static str| {
63            let wanted = project.clone();
64            chip(
65                label,
66                Signal::derive(move || {
67                    workspace
68                        .asset_filter
69                        .with(|filter| filter.project == wanted)
70                }),
71                move || {
72                    let mut filter = workspace.asset_filter.get_untracked();
73                    filter.project = project.clone();
74                    workspace.choose_asset_filter(filter);
75                },
76                data,
77            )
78        };
79    let projects = move || {
80        workspace
81            .projects
82            .get()
83            .into_iter()
84            .map(|project| {
85                project_chip(
86                    project.name.to_string(),
87                    Some(AssetProjectFilter::Project {
88                        project: project.id.clone(),
89                    }),
90                    "project",
91                )
92            })
93            .collect_view()
94    };
95    view! {
96        <div
97            id="assets-filter"
98            role="dialog"
99            aria-label="Filter assets"
100            class="flex flex-col gap-3 rounded-xl p-3 bg-raised shadow-[0_0_0_1px_var(--color-line),0_1px_2px_rgb(0_0_0/0.03)]"

102            <div class="flex flex-col gap-1.5">

104                <div class="flex flex-wrap gap-1.5" role="radiogroup" aria-label="Project">

109            </div>
110            <div class="flex flex-col gap-1.5">
111                <span class=heading>"Level"</span>
112                <div class="flex flex-wrap gap-1.5" role="radiogroup" aria-label="Level">

115                        .map(|scope| {
116                            chip(
117                                AssetFilter::level_label(scope).to_owned(),
118                                Signal::derive(move || workspace.asset_filter.with(|filter| filter.scope == scope)),
119                                move || {
120                                    let mut filter = workspace.asset_filter.get_untracked();

123                                },
124                                "level",

134/// What the filter lists, in words: "All projects · All levels".
135fn filter_summary(workspace: Workspace) -> String {
136    let filter = workspace.asset_filter.get();
137    let project = match &filter.project {
138        None => "All projects".to_owned(),
139        Some(AssetProjectFilter::NoProject) => "No project".to_owned(),
140        Some(AssetProjectFilter::Project { project }) => workspace
141            .projects
142            .with(|projects| {
143                projects
144                    .iter()
145                    .find(|known| known.id == *project)
146                    .map(|known| known.name.to_string())
147            })
148            .unwrap_or_else(|| project.to_string()),
149    };
150    format!("{project} · {}", AssetFilter::level_label(filter.scope))
151}

153/// The assets the tab lists now, as its kind and search narrow them, in
154/// its order: every one, not only the rows shown.
155pub(super) fn shown_assets(workspace: Workspace) -> Vec<Asset> {
156    let query = workspace.asset_query.get();
157    let kind = workspace.asset_kind.get();
158    workspace.asset_list.with(|list| match list {
159        Some(Ok(found)) => assets_matching(&found.assets, &query, kind)
160            .into_iter()
161            .cloned()
162            .collect(),
163        _ => Vec::new(),
164    })
165}

167/// Whether the media viewer shows a file of `media_type`.
168fn in_viewer(media_type: &str) -> bool {
169    media_type.starts_with("image/") || media_type.starts_with("video/")
170}
171
172/// Whether an asset of `media_type` opens on its own page.
173pub(super) fn on_page(media_type: &str) -> bool {
174    !in_viewer(media_type)
175}
176
177/// Opens `asset`, one of `shown`: a picture or a video in the media viewer,
178/// to step through every picture and video of `shown`; anything else on
179/// its page.
180fn open(workspace: Workspace, asset: &Asset, shown: &[Asset]) {
181    if !in_viewer(&asset.media_type) {
182        workspace.asset_page.set(Some(asset.clone()));
183        return;
184    }
185    let group = shown
186        .iter()
187        .filter(|shown| in_viewer(&shown.media_type))
188        .map(crate::assets::viewing_of)
189        .collect::<Vec<_>>();
190    let start = group.iter().position(|viewing| {
191        viewing
192            .asset
193            .as_ref()
194            .is_some_and(|(_, id)| *id == asset.id)
195    });
196    let first = start.and_then(|start| group.get(start).cloned());
197    workspace.media_group.set(group);
198    workspace.media_viewer.set(first);
199}

385/// What a thumbnail's observer calls back.
386type Seen = Closure<dyn Fn(js_sys::Array, web_sys::IntersectionObserver)>;
387
388thread_local! {
389    /// The one observer of every card's thumbnail, for the page's life:
390    /// it reads a thumbnail once it comes within a screen of the view.
391    static THUMBNAILS: std::cell::RefCell<Option<(web_sys::IntersectionObserver, Seen)>> = const { std::cell::RefCell::new(None) };
392    /// The workspace the observer fills thumbnails for.
393    static THUMBNAILS_FOR: std::cell::Cell<Option<Workspace>> = const { std::cell::Cell::new(None) };
394}
395
396/// Reads the thumbnail `thumb` of a card once it comes into view, or
397/// near it, and forgets it once it leaves the page.
398fn watch_thumbnail(thumb: &web_sys::Element, workspace: Workspace) {
399    THUMBNAILS_FOR.with(|kept| kept.set(Some(workspace)));
400    THUMBNAILS.with(|observer| {
401        let mut observer = observer.borrow_mut();
402        if observer.is_none() {
403            let seen = Seen::new(
404                |entries: js_sys::Array, observer: web_sys::IntersectionObserver| {
405                    let Some(workspace) = THUMBNAILS_FOR.with(std::cell::Cell::get) else {
406                        return;
407                    };
408                    for entry in entries.iter() {
409                        let Ok(entry) = entry.dyn_into::<web_sys::IntersectionObserverEntry>()
410                        else {
411                            continue;
412                        };
413                        let target = entry.target();
414                        if entry.is_intersecting() {
415                            observer.unobserve(&target);
416                            crate::assets::show_within(&target, workspace);
417                        } else if !target.is_connected() {
418                            observer.unobserve(&target);
419                        }
420                    }
421                },
422            );
423            let options = web_sys::IntersectionObserverInit::new();
424            options.set_root_margin(READ_AHEAD);
425            if let Ok(made) = web_sys::IntersectionObserver::new_with_options(
426                seen.as_ref().unchecked_ref(),
427                &options,
428            ) {
429                *observer = Some((made, seen));
430            }
431        }
432        if let Some((observer, _)) = observer.as_ref() {
433            observer.observe(thumb);
434        }
435    });
436}

438/// A card's thumbnail: `html`, read by [`watch_thumbnail`] once mounted.
439fn thumbnail(workspace: Workspace, html: String, class: &'static str) -> impl IntoView {
440    let node = NodeRef::<html::Span>::new();
441    Effect::new(move |_| {
442        if let Some(thumb) = node.get() {
443            watch_thumbnail(&thumb, workspace);
444        }
445    });
446    view! { <span node_ref=node class=class data-thumb inner_html=html></span> }
447}

449/// Where `asset` lives: its project, and a session's own asset's session,
450/// named from the tab's listing when it has one.
451pub(super) fn place_of(workspace: Workspace, asset: &Asset) -> AssetPlace {
452    let project_name = |project: &cookie_agent_protocol::ProjectId| {
453        workspace.projects.with_untracked(|projects| {
454            projects
455                .iter()
456                .find(|known| known.id == *project)
457                .map(|known| known.name.to_string())
458        })
459    };
460    let mut place = workspace.asset_list.with_untracked(|list| match list {
461        Some(Ok(found)) => asset_place(asset, &found.sessions, project_name),
462        _ => asset_place(asset, &[], project_name),
463    });
464    // Untitled, as the session list names it.
465    if let Some((session, title @ None)) = &mut place.session {
466        *title = Some(session_title(workspace, *session));
467    }
468    place
469}

471/// What the session list calls `session`.
472fn session_title(workspace: Workspace, session: SessionId) -> String {
473    workspace.sessions.with_untracked(|sessions| {
474        sessions
475            .iter()
476            .find(|meta| meta.session_id == session)
477            .map_or_else(|| "Untitled session".to_owned(), super::session_label)
478    })
479}

481/// Who last saved `asset`, and when: the session's title when the page
482/// lists it.
483pub(super) fn saved_by(workspace: Workspace, asset: &Asset) -> (String, Option<SessionId>) {
484    let session = asset.updated.session;
485    let title = workspace.sessions.with_untracked(|sessions| {
486        sessions
487            .iter()
488            .find(|meta| meta.session_id == session)
489            .map(super::session_label)
490    });
491    let age = cookie_agent_protocol::memory_age(asset.updated.at, jiff::Timestamp::now());
492    match title {
493        Some(title) => (format!("saved by {title}, {age} ago"), Some(session)),
494        None => (format!("saved by another session, {age} ago"), None),
495    }
496}

546        danger: false,
547    });

577    items.push(MenuItem {
578        id: "delete",

581        act: Callback::new(move |()| workspace.asset_deleting.set(Some(deleted.clone()))),
582        danger: true,
583    });
584    items
585}

587/// One asset, as a card of the grid or a row of the list: its thumbnail,
588/// title, and where it lives; a click opens it and ⋯ offers the rest.
589fn item(
590    workspace: Workspace,
591    asset: Asset,
592    place: AssetPlace,
593    shown: std::sync::Arc<Vec<Asset>>,
594    as_list: bool,
595) -> impl IntoView {
596    let kind = AssetKind::of(&asset.media_type);
597    let id = asset.id;
598    let shared = move || {
599        workspace.share_list.with(|list| {
600            list.as_ref()
601                .and_then(|list| list.as_ref().ok())
602                .is_some_and(|list| list.shares.iter().any(|share| share.asset == id))
603        })
604    };
605    let opened = {
606        let (asset, shown) = (asset.clone(), std::sync::Arc::clone(&shown));
607        Callback::new(move |()| open(workspace, &asset, &shown))
608    };
609    let items = {
610        let asset = asset.clone();
611        move || menu_items(workspace, &asset, Some(opened))
612    };
613    let menu = super::row_menu::row_menu(
614        workspace.asset_menu,
615        id,
616        asset.handle.to_string(),
617        asset.title.clone(),
618        if as_list {
619            "inline-flex size-9 shrink-0 items-center justify-center rounded-md text-muted hover:bg-hover hover:text-fg"
620        } else {
621            CORNER
622        },
623        items.clone(),
624    );
625    let context =
626        super::row_menu::context_menu(workspace.asset_context, id, asset.title.clone(), items);
627    let on_contextmenu = move |event: ev::MouseEvent| {
628        if !crate::browser::from_touch(&event) && super::row_menu::wants_context_menu(&event) {
629            event.prevent_default();
630            workspace.asset_context.set(Some((
631                id,
632                f64::from(event.client_x()),
633                f64::from(event.client_y()),
634            )));
635        }
636    };
637    let on_keydown = move |event: ev::KeyboardEvent| {
638        let on_card = event
639            .target()
640            .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
641            .is_some_and(|target| target.has_attribute("data-asset-row"));
642        if on_card && (event.key() == "Enter" || event.key() == " ") {
643            event.prevent_default();
644            opened.run(());
645        }
646    };
647    let level = if place.session.is_some() {
648        "session"
649    } else {
650        "project"
651    };
652    let tree = place.session.map(|(session, title)| {
653        let title = title.unwrap_or_else(|| session_title(workspace, session));
654        view! {
655            <span>"·"</span>
656            <button
657                type="button"
658                class="min-w-0 truncate underline decoration-dotted underline-offset-2 hover:text-fg"
659                data-asset-tree=session.to_string()
660                title="Open the session"
661                on:click=move |event| {
662                    event.stop_propagation();
663                    open_session(workspace, session);
664                }
665            >
666                {title}
667            </button>
668        }
669    });
670    let where_ = view! {
671        <span class=format!("flex min-w-0 items-center gap-1 {NOTE}") data-asset-place=level>
672            <Glyph
673                icon=if level == "session" { icondata::LuMessagesSquare } else { icondata::LuFolder }
674                class="shrink-0 text-xs"
675            />
676            <span class="shrink-0" data-asset-project>{place.project}</span>
677            {tree}
678        </span>
679    };
680    let title_line = view! {
681        <span class="flex min-w-0 items-center gap-1.5">
682            <Glyph icon=crate::icons::asset_icon(kind) class="shrink-0 text-[0.8125rem] text-accent" />
683            <span class="min-w-0 truncate text-[0.8125rem] font-semibold" data-asset-row-title>{asset.title.clone()}</span>
684            <Show when=shared>
685                <span class="shrink-0 text-xs text-accent" title="Shared" data-asset-shared>
686                    <Glyph icon=icondata::LuShare2 />
687                </span>
688            </Show>
689        </span>
690    };
691    let label =
692        crate::render::attachments::type_label_of(asset.filename.as_deref(), &asset.media_type);
693    let age = cookie_agent_protocol::memory_age(asset.updated.at, jiff::Timestamp::now());
694    let thumbnail_html = crate::assets::tab_thumbnail_html(&asset);
695    let handle = asset.handle.to_string();
696    let copy = handle.clone();
697    let row_attrs = (handle.clone(), asset.title.clone());
698    if as_list {
699        let (saved, _) = saved_by(workspace, &asset);
700        view! {
701            {context}
702            <div
703                class="flex min-w-0 cursor-pointer items-center gap-3 rounded-lg border-b border-line px-1.5 py-2 outline-none hover:bg-hover focus-visible:ring-2 focus-visible:ring-accent"
704                role="button"
705                tabindex="0"
706                aria-label=format!("Open {}", row_attrs.1)
707                data-asset-row=row_attrs.0
708                on:click=move |_| opened.run(())
709                on:keydown=on_keydown
710                on:contextmenu=on_contextmenu
711            >
712                {thumbnail(
713                    workspace,
714                    thumbnail_html,
715                    "relative aspect-[4/3] w-16 shrink-0 overflow-hidden rounded-md bg-sunken shadow-[0_0_0_1px_var(--color-line)] [&_[data-asset-icon]_span]:hidden [&_[data-asset-icon]_svg]:text-xl",
716                )}
717                <span class="flex min-w-0 flex-1 flex-col gap-0.5">
718                    {title_line}
719                    {where_}
720                </span>
721                <span class=format!("hidden shrink-0 text-right sm:block {NOTE}")>
722                    {format!("{} · {}", asset.media_type, human_bytes(asset.byte_length))}
723                    <br />
724                    {saved}
725                </span>
726                <span class="contents" on:click=|event| event.stop_propagation()>{menu}</span>
727            </div>
728        }
729        .into_any()
730    } else {
731        view! {
732            {context}
733            <div
734                class=CARD
735                role="button"
736                tabindex="0"
737                aria-label=format!("Open {}", row_attrs.1)
738                title=format!("{} · {}", row_attrs.1, row_attrs.0)
739                data-asset-row=row_attrs.0
740                on:click=move |_| opened.run(())
741                on:keydown=on_keydown
742                on:contextmenu=on_contextmenu
743            >
744                <span class=CARD_BOX>
745                    {thumbnail(workspace, thumbnail_html, "relative block aspect-[4/3] overflow-hidden bg-sunken")}
746                    <span class="flex min-w-0 flex-col gap-0.5 px-2.5 pt-2 pb-2.5">
747                        {title_line}
748                        <span class=format!("truncate {NOTE}")>
749                            {format!("{label} · {} · {age} ago", human_bytes(asset.byte_length))}
750                        </span>
751                        {where_}
752                    </span>
753                </span>
754                // Shown while its menu is open, which it holds.
755                <span
756                    class="absolute top-1.5 right-1.5 z-10 flex gap-1 opacity-0 transition-opacity group-hover:opacity-100 group-focus-within:opacity-100 pointer-coarse:opacity-100"
757                    class:opacity-100=move || workspace.asset_menu.get() == Some(id)
758                    on:click=|event| event.stop_propagation()
759                >
760                    <button
761                        type="button"
762                        class=format!("{CORNER} pointer-coarse:hidden")
763                        aria-label=format!("Copy the handle {handle}")
764                        title=format!("Copy {handle}")
765                        data-asset-action="copy"
766                        on:click=move |_| crate::assets::copy_handle(copy.clone(), workspace)
767                    >
768                        <Glyph icon=icondata::LuCopy />
769                    </button>
770                    {menu}
771                </span>
772            </div>
773        }
774        .into_any()
775    }
776}

778/// The rename and delete dialogs of an asset, from the tab, its page, or
779/// the media viewer's Details: over everything, the viewer included.
780#[component]
781pub(crate) fn AssetDialogs(workspace: Workspace) -> impl IntoView {
782    let field = NodeRef::<html::Input>::new();
783    Effect::new(move |_| {
784        if let Some(field) = field.get() {
785            let _ = field.focus();
786            field.select();
787        }
788    });
789    // The focus goes back to the viewer, for its keys, when it is open.
790    let refocus = move || {
791        request_animation_frame(|| {
792            if let Some(viewer) = document()
793                .get_element_by_id("media-viewer")
794                .and_then(|viewer| viewer.dyn_into::<web_sys::HtmlElement>().ok())
795            {
796                let _ = viewer.focus();
797            }
798        });
799    };
800    let close = move || {
801        workspace.asset_renaming.set(None);
802        workspace.asset_deleting.set(None);
803        refocus();
804    };
805    let open = move || {
806        workspace.asset_renaming.with(Option::is_some)
807            || workspace.asset_deleting.with(Option::is_some)
808    };
809    let save = move || {
810        if let Some((asset, title)) = workspace.asset_renaming.get_untracked() {
811            workspace.rename_asset(asset.store.clone(), asset.id, title);
812            refocus();
813        }
814    };
815    let body = move || {
816        if workspace.asset_renaming.with(Option::is_some) {
817            view! {
818                <form
819                    class="flex flex-col gap-3"
820                    data-asset-rename
821                    on:submit=move |event: ev::SubmitEvent| {
822                        event.prevent_default();
823                        save();
824                    }
825                >
826                    <h2 class="text-[0.9375rem] font-semibold" id="asset-dialog-title">"Rename"</h2>
827                    <input
828                        type="text"
829                        node_ref=field
830                        id="asset-rename-title"
831                        class="min-h-11 w-full min-w-0 rounded-lg bg-surface px-3 text-sm shadow-[inset_0_0_0_1px_var(--color-line-2)] focus:shadow-[inset_0_0_0_1px_var(--color-accent),0_0_0_3px_var(--color-accent-soft)] focus:outline-none lg:min-h-9"
832                        maxlength="200"
833                        aria-label="The asset's title"
834                        prop:value=move || {
835                            workspace
836                                .asset_renaming
837                                .with_untracked(|renaming| renaming.as_ref().map(|(_, title)| title.clone()))
838                                .unwrap_or_default()
839                        }
840                        on:input=move |event| {
841                            let title = event_target_value(&event);
842                            workspace.asset_renaming.update(|renaming| {
843                                if let Some((_, current)) = renaming {
844                                    *current = title;
845                                }
846                            });
847                        }
848                    />
849                    <div class="flex justify-end gap-2">
850                        <button type="button" class=BUTTON on:click=move |_| close()>"Cancel"</button>
851                        <button type="submit" class=PRIMARY id="asset-rename-save">"Save"</button>
852                    </div>
853                </form>
854            }
855            .into_any()
856        } else {
857            let asset = workspace.asset_deleting.get();
858            let title = asset
859                .as_ref()
860                .map(|asset| asset.title.clone())
861                .unwrap_or_default();
862            view! {
863                <div class="flex flex-col gap-3" data-asset-delete-confirm>
864                    <h2 class="text-[0.9375rem] font-semibold" id="asset-dialog-title">"Delete the asset?"</h2>
865                    <p class="text-sm text-fg-2">
866                        {format!("Delete {title}? Every place it was shown will say it was deleted.")}
867                    </p>
868                    <div class="flex justify-end gap-2">
869                        <button type="button" class=BUTTON on:click=move |_| close()>"Keep"</button>
870                        <button
871                            type="button"
872                            class=format!("{BUTTON} text-bad")
873                            id="asset-delete-confirm"
874                            on:click=move |_| {
875                                if let Some(asset) = workspace.asset_deleting.get_untracked() {
876                                    workspace.delete_asset(asset.store.clone(), asset.id);
877                                }
878                                refocus();
879                            }
880                        >
881                            "Delete"
882                        </button>
883                    </div>
884                </div>
885            }
886            .into_any()
887        }
888    };
889    view! {
890        <Show when=open>
891            <div class="fixed inset-0 z-[70] bg-scrim dark:bg-black/50" on:click=move |_| close()></div>
892            <div
893                id="asset-dialog"
894                role="dialog"
895                aria-modal="true"
896                aria-labelledby="asset-dialog-title"
897                class="fixed inset-x-0 bottom-0 z-[70] flex flex-col rounded-t-xl p-4 pb-[max(1rem,env(safe-area-inset-bottom))] text-sm text-fg sm:inset-x-auto sm:top-1/2 sm:bottom-auto sm:left-1/2 sm:w-[26rem] sm:max-w-[calc(100vw-2rem)] sm:-translate-x-1/2 sm:-translate-y-1/2 sm:rounded-xl bg-raised ring-1 ring-line shadow-menu"
898                on:keydown=move |event| {
899                    if event.key() == "Escape" {
900                        event.stop_propagation();
901                        close();
902                    }
903                }
904            >
905                {body}
906            </div>
907        </Show>
908    }
