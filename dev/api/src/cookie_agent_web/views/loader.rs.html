1//! The open session's transcript until its newest messages arrive: a skeleton
2//! of rows under a moving line, and a pill. A quick load shows nothing; once
3//! shown, the skeleton stays long enough to read, then fades as the
4//! transcript takes its place.
5
6use std::time::Duration;
7
8use leptos::prelude::*;
9
10use crate::workspace::Workspace;
11
12/// How long a load runs before the skeleton shows.
13const SHOW_AFTER: Duration = Duration::from_millis(150);
14/// How long the skeleton shows at least, once it does.
15const SHOWN_AT_LEAST: Duration = Duration::from_millis(400);
16/// How long it takes to fade (`.loader-out`).
17const FADE: Duration = Duration::from_millis(320);
18
19/// Where the skeleton is.
20#[derive(Clone, Copy, Debug, Eq, PartialEq)]
21enum Stage {
22    Hidden,
23    Shown,
24    Fading,
25}
26
27fn now() -> f64 {
28    js_sys::Date::now()
29}
30
31#[component]
32pub fn TranscriptLoader(workspace: Workspace) -> impl IntoView {
33    let loading = Memo::new(move |_| {
34        !workspace.new_session.get()
35            && workspace.selected.get().is_some_and(|session_id| {
36                workspace
37                    .opening
38                    .with(|opening| opening.contains(&session_id))
39            })
40    });
41    let stage = RwSignal::new(Stage::Hidden);
42    let shown_at = StoredValue::new(0.0_f64);
43    // Each change of `loading` outdates the timers set for the last one.
44    let turn = StoredValue::new(0_u64);
45    Effect::new(move |_| {
46        let now_loading = loading.get();
47        let this = turn.try_update_value(|turn| {
48            *turn += 1;
49            *turn
50        });
51        let current = move || turn.try_get_value() == this;
52        if now_loading {
53            if stage.get_untracked() == Stage::Fading {
54                stage.set(Stage::Shown);
55                return;
56            }
57            set_timeout(
58                move || {
59                    if current() && stage.get_untracked() == Stage::Hidden {
60                        shown_at.set_value(now());
61                        stage.set(Stage::Shown);
62                    }
63                },
64                SHOW_AFTER,
65            );
66        } else if stage.get_untracked() == Stage::Shown {
67            let shown_for = now() - shown_at.get_value();
68            #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
69            let wait = Duration::from_millis(
70                (SHOWN_AT_LEAST.as_secs_f64() * 1000.0 - shown_for).max(0.0) as u64,
71            );
72            set_timeout(
73                move || {
74                    if !current() {
75                        return;
76                    }
77                    stage.set(Stage::Fading);
78                    set_timeout(
79                        move || {
80                            if current() {
81                                stage.set(Stage::Hidden);
82                            }
83                        },
84                        FADE,
85                    );
86                },
87                wait,
88            );
89        }
90    });
91    view! {
92        <Show when=move || stage.get() != Stage::Hidden>
93            <div
94                class="pointer-events-none absolute inset-0 overflow-hidden bg-surface"
95                class:loader-out=move || stage.get() == Stage::Fading
96                role="status"
97                aria-label="Loading the session"
98                data-transcript-loader
99            >
100                <div class="loader-line loader-line-unknown absolute top-0 left-0 h-0.5 rounded-full bg-accent"></div>
101                <div class="mx-auto flex max-w-[45rem] flex-col gap-3 px-3 py-2 sm:px-6" aria-hidden="true">
102                    <Skeleton />
103                </div>
104                <div class="absolute inset-x-0 bottom-4 flex justify-center">
105                    <div class="inline-flex items-center gap-2 rounded-full bg-raised py-1.5 pr-3 pl-2.5 text-xs text-muted shadow-menu">
106                        <span class="size-3.5 animate-spin rounded-full border-2 border-line-2 border-t-accent"></span>
107                        "Loading"
108                    </div>
109                </div>
110            </div>
111        </Show>
112    }
113}

115/// Rows shaped like a transcript's: the pinned rows, a message, a reply
116/// with its work and two calls, and its answer.
117#[component]
118fn Skeleton() -> impl IntoView {
119    let line = |width: &'static str| {
120        view! { <div class="skeleton h-2.5 rounded-md" style=format!("width: {width}")></div> }
121    };
122    let call = |height: &'static str, title: &'static str| {
123        view! {
124            <div class="flex flex-col gap-3 rounded-xl p-3.5 ring-1 ring-line" style=format!("height: {height}")>
125                <div class="skeleton h-3" style=format!("width: {title}")></div>
126                <div class="skeleton flex-1"></div>
127            </div>
128        }
129    };
130    view! {
131        <div class="rise-in flex gap-2">
132            <div class="skeleton h-6 w-40 max-w-[40%] rounded-full"></div>
133            <div class="skeleton h-6 w-32 max-w-[35%] rounded-full"></div>
134        </div>
135        <div class="skeleton rise-in ml-auto h-11 w-[60%] rounded-2xl"></div>
136        <div class="rise-in flex flex-col gap-2.5">
137            <div class="flex items-center gap-2">
138                <div class="skeleton size-[18px] rounded-[5px]"></div>
139                <div class="skeleton h-2.5 w-28"></div>
140            </div>
141            {line("92%")}
142            {line("74%")}
143        </div>
144        <div class="rise-in">{line("170px")}</div>
145        <div class="rise-in">{call("11rem", "160px")}</div>
146        <div class="rise-in">{call("7.25rem", "240px")}</div>
147        <div class="rise-in flex flex-col gap-2.5">{line("88%")} {line("66%")} {line("52%")}</div>
148    }
