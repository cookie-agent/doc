1//! A model menu, for a cookie agent's models and an ACP agent's alike
2//! (`docs/site/specs/web-polish.md`, P4, changed 2026-11-16): a search
3//! field, the models in a list of their own that scrolls (grouped by
4//! provider when there are several), and a foot pinned under it (the effort
5//! control, and an ACP agent's other options).
6
7use cookie_agent_client_state::model_menu::{match_range, matches, provider_groups};
8use leptos::{ev, html, prelude::*};
9
10use super::{NOTE, ROW};
11use crate::{icons::Glyph, workspace::Workspace};
12
13/// One model of the menu.
14#[derive(Clone)]
15pub(super) struct MenuModel {
16    /// Its ID: the `provider/model` key, or the ACP option's value.
17    pub id: String,
18    pub name: String,
19    /// The provider it is grouped under; empty for an ACP agent's.
20    pub provider: String,
21    pub description: Option<String>,
22    pub chosen: bool,
23    /// Chooses it.
24    pub pick: Callback<()>,
25}
26
27/// A model's name with the part the search matched marked.
28fn marked_name(name: &str, query: &str) -> AnyView {
29    match match_range(name, query) {
30        Some((from, to)) => view! {
31            {name[..from].to_owned()}
32            <mark class="rounded-[0.2rem] bg-accent-soft px-px text-inherit">{name[from..to].to_owned()}</mark>
33            {name[to..].to_owned()}
34        }
35        .into_any(),
36        None => name.to_owned().into_any(),
37    }
38}

40/// The menu of `models`, `acp` saying whose they are, with `foot` pinned
41/// under the list. Typing filters the list; ↑ and ↓ move through it and
42/// Enter chooses.
43pub(super) fn model_menu(
44    workspace: Workspace,
45    models: Vec<MenuModel>,
46    acp: bool,
47    foot: Option<AnyView>,
48) -> impl IntoView {
49    let query = RwSignal::new(String::new());
50    // The row ↑ and ↓ reached, in the filtered list; none until they move.
51    let active = RwSignal::new(None::<usize>);
52    let field = NodeRef::<html::Input>::new();
53    let count = models.len();
54    let models = StoredValue::new(models);
55    let shown = move || {
56        let query = query.get();
57        models.with_value(|models| {
58            models
59                .iter()
60                .filter(|model| {
61                    matches(
62                        &query,
63                        &[
64                            &model.name,
65                            &model.provider,
66                            &model.id,
67                            model.description.as_deref().unwrap_or_default(),
68                        ],
69                    )
70                })
71                .cloned()
72                .collect::<Vec<_>>()
73        })
74    };
75    // On a desktop the field takes the keys at once; a phone's keyboard
76    // would cover the list, so there it waits for a tap. The list opens
77    // at the current model.
78    let narrow = workspace.narrow.get_untracked();
79    Effect::new(move |_| {
80        if let Some(field) = field.get()
81            && !narrow
82        {
83            let _ = field.focus();
84        }
85        crate::browser::reveal("[data-model-menu] [aria-current=true]");
86    });
87    let keydown = move |event: ev::KeyboardEvent| {
88        let rows = untrack(shown);
89        match event.key().as_str() {
90            "ArrowDown" | "ArrowUp" if !rows.is_empty() => {
91                event.prevent_default();
92                let last = rows.len() - 1;
93                let next = match (active.get_untracked(), event.key().as_str()) {
94                    (None, "ArrowDown") => 0,
95                    (None, _) => last,
96                    (Some(at), "ArrowDown") => (at + 1).min(last),
97                    (Some(at), _) => at.saturating_sub(1),
98                };
99                active.set(Some(next));
100                crate::browser::reveal("[data-model-menu] [data-active=true]");
101            }
102            "Enter" => {
103                event.prevent_default();
104                if let Some(model) = rows.get(active.get_untracked().unwrap_or(0)) {
105                    model.pick.run(());
106                }
107            }
108            _ => {}
109        }
110    };
111    let list = move || {
112        let query = query.get();
113        let rows = shown();
114        if rows.is_empty() {
115            return view! {
116                <p class=format!("px-3 py-4 text-center {NOTE}") data-model-none>
117                    {format!("No model matches “{}”.", query.trim())}
118                </p>
119            }
120            .into_any();
121        }
122        let active = active.get();
123        let mut index = 0usize;
124        provider_groups(rows, |model| model.provider.clone())
125            .into_iter()
126            .map(|(provider, members)| {
127                let rows = members
128                    .into_iter()
129                    .map(|model| {
130                        let at = index;
131                        index += 1;
132                        let highlighted = active == Some(at);
133                        let MenuModel {
134                            id,
135                            name,
136                            description,
137                            chosen,
138                            pick,
139                            ..
140                        } = model;
141                        view! {
142                            <button
143                                type="button"
144                                class=ROW
145                                class:bg-sel=chosen
146                                class:bg-hover=highlighted && !chosen
147                                aria-current=chosen.to_string()
148                                data-active=highlighted.to_string()
149                                title=id.clone()
150                                data-model=(!acp).then(|| id.clone())
151                                data-acp-model=acp.then(|| id.clone())
152                                on:click=move |_| pick.run(())
153                            >
154                                <span class="flex w-full items-center gap-2">
155                                    <span class="min-w-0 flex-1 truncate text-sm font-semibold">{marked_name(&name, &query)}</span>
156                                    {chosen.then(|| view! {
157                                        <Glyph icon=icondata::LuCheck class="text-accent" />
158                                        <span class="sr-only">"(current)"</span>
159                                    })}
160                                </span>
161                                {description.map(|description| view! { <span class=NOTE>{description}</span> })}
162                            </button>
163                        }
164                    })
165                    .collect_view();
166                view! {
167                    {provider.map(|provider| view! {
168                        <span class="px-3 pt-2.5 pb-0.5 text-[0.6875rem] font-semibold tracking-wide text-muted uppercase" data-model-group>{provider}</span>
169                    })}
170                    {rows}
171                }
172            })
173            .collect_view()
174            .into_any()
175    };
176    view! {
177        <div class="flex min-h-0 flex-col" data-model-menu>
178            <label class="mx-1 mb-1 flex min-h-11 items-center gap-2 rounded-lg bg-surface px-2.5 shadow-[inset_0_0_0_1px_var(--color-line-2)] focus-within:shadow-[inset_0_0_0_1px_var(--color-accent),0_0_0_3px_var(--color-accent-soft)] lg:min-h-9">
179                <Glyph icon=icondata::LuSearch class="shrink-0 text-muted" />
180                <input
181                    type="search"
182                    node_ref=field
183                    class="min-h-11 min-w-0 flex-1 bg-transparent text-sm placeholder:text-faint focus:outline-none lg:min-h-9"
184                    placeholder=format!("Search {count} models")
185                    aria-label="Search the models"
186                    data-model-search
187                    prop:value=move || query.get()
188                    on:input=move |event| {
189                        query.set(event_target_value(&event));
190                        active.set(None);
191                    }
192                    on:keydown=keydown
193                />
194            </label>
195            <div class="flex max-h-[min(24rem,45vh)] flex-col gap-0.5 overflow-y-auto overscroll-contain px-0.5" data-model-list>
196                {list}
197            </div>
198            {foot.map(|foot| view! {
199                <div class="mt-1 border-t border-line px-1 pt-2 pb-1" data-model-foot>{foot}</div>
200            })}
201        </div>
202    }
203}
