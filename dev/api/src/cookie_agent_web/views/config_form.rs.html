33/// The form, or why `config.toml` cannot be shown as one.
34pub(super) fn config_form(workspace: Workspace) -> impl IntoView {
35    let text = Memo::new(move |_| {
36        workspace.config.with(|config| match config {
37            Some(Ok(config)) => config
38                .documents
39                .iter()
40                .find(|document| document.path == "config.toml")
41                .map(|document| document.text.clone())
42                .unwrap_or_default(),
43            _ => String::new(),
44        })
45    });
46    // Discarding draws every row afresh, closing the open pickers.
47    let discarded = RwSignal::new(0_u32);
48    let form = Memo::new(move |_| form_of(&text.get()));
49    // The rest of `[speech]` waits for a backend.
50    let speech_on = Signal::derive(move || {
51        form.with(|form| {
52            form.as_ref().is_ok_and(|form| {
53                workspace
54                    .config_form_drafts
55                    .with(|drafts| form.speech_on(drafts))
56            })
57        })
58    });
59    let body = move || {
60        discarded.track();
61        match form.get() {
62            Err(error) => view! {
63                <p class=ERROR id="config-form-error">{error}</p>
64                <p class=NOTE>"Fix it with Edit the file; the settings return once it reads."</p>
65            }
66            .into_any(),
67            Ok(form) => view! {
68                <div class="flex flex-col gap-2.5" id="config-form">
69                    {form.sections.iter().cloned().map(|section| section_card(workspace, section, speech_on)).collect_view()}
70                </div>
71                <h2 class=CAPTION>"In their own panels"</h2>
72                <div class="grid gap-2 sm:grid-cols-2" id="config-managed">
73                    {form.managed.iter().cloned().map(|managed| managed_row(workspace, managed)).collect_view()}
74                </div>
75                {(!form.other.is_empty()).then(|| view! {
76                    <h2 class=CAPTION>"Other settings"</h2>
77                    <div class="overflow-hidden rounded-xl shadow-[0_0_0_1px_var(--color-line)]" id="config-other">
78                        {form.other.iter().map(|entry| other_box(workspace, entry)).collect_view()}
79                    </div>
80                })}
81            }
82            .into_any(),
83        }
84    };
85    let bar = move || {
86        let Ok(form) = form.get() else {
87            return None;
88        };
89        let drafts = workspace.config_form_drafts.get();
90        let changes = form.changes(&drafts);
91        let lines = match &changes {
92            Ok(changes) => changes.iter().map(change_line).collect::<Vec<_>>(),
93            Err(errors) => errors
94                .iter()
95                .map(|(path, error)| format!("{path}: {error}"))
96                .collect(),
97        };
98        let (count, blocked) = match &changes {
99            Ok(changes) => (changes.len(), false),
100            Err(errors) => (errors.len(), true),
101        };
102        (count > 0).then(|| {
103            let summary = match (blocked, count) {
104                (true, 1) => "1 value to fix".to_owned(),
105                (true, count) => format!("{count} values to fix"),
106                (false, 1) => "1 change".to_owned(),
107                (false, count) => format!("{count} changes"),
108            };
109            view! {
110                <div class=format!("{EDITOR_ACTIONS} mt-auto flex-col") id="config-form-bar">
111                    <div class="mx-auto flex w-full max-w-3xl flex-col gap-1.5">
112                        <div class="flex items-center gap-2">
113                            <span class="mr-auto text-[0.8125rem] font-medium" id="config-form-count">{summary}</span>
114                            <button
115                                type="button"
116                                id="config-form-apply"
117                                class=PRIMARY
118                                disabled=blocked
119                                on:click=move |_| workspace.apply_config_form(&form)
120                            >
121                                "Apply"
122                            </button>
123                            <button
124                                type="button"
125                                id="config-form-discard"
126                                class=BUTTON
127                                on:click=move |_| {
128                                    workspace.config_form_drafts.set(Default::default());
129                                    discarded.update(|count| *count += 1);
130                                }
131                            >
132                                "Discard"
133                            </button>
134                        </div>
135                        <ul class="max-h-24 overflow-y-auto font-mono text-[0.7188rem] leading-relaxed text-muted wrap-anywhere">
136                            {lines.into_iter().map(|line| view! { <li>{line}</li> }).collect_view()}
137                        </ul>
138                    </div>
139                </div>
140            }
141        })
142    };
143    view! {
144        {body}
145        {bar}
146    }
147}

149/// A `set` as a line for the bar.
150fn change_line(change: &cookie_agent_protocol::ConfigChange) -> String {
151    let cookie_agent_protocol::ConfigChange::Set { key, value } = change else {
152        return String::new();
153    };
154    let shown = shown_value(key, value);
155    let key = key.join(".");
156    if value.is_null() {
157        format!("{key} removed")
158    } else {
159        format!("{key} = {shown}")
160    }
161}

163fn section_card(
164    workspace: Workspace,
165    section: FormSection,
166    speech_on: Signal<bool>,
167) -> impl IntoView {
168    let waits_for_backends = section.section.id == "speech";
169    let authored = section.authored();
170    let set = section
171        .fields
172        .iter()
173        .filter(|field| field.value.is_some())
174        .count()
175        + section.other.len();
176    let other = section
177        .other
178        .iter()
179        .map(|entry| other_box(workspace, entry))
180        .collect_view();
181    view! {
182        <details class=CARD open=authored data-config-section=section.section.id>
183            <summary class=SUMMARY>
184                <Glyph icon=icondata::LuChevronRight class="text-xs text-faint transition-transform group-open/section:rotate-90" />
185                {section.section.title}
186                {(set > 0).then(|| view! { <span class="text-[0.6875rem] font-medium text-muted">{format!("{set} set in the file")}</span> })}
187            </summary>
188            {section.fields.iter().cloned().map(|field| {
189                let off = if waits_for_backends && field.field.kind != FieldKind::Backends {
190                    Signal::derive(move || !speech_on.get())
191                } else {
192                    Signal::stored(false)
193                };
194                field_row(workspace, field, off)
195            }).collect_view()}
196            {other}
197        </details>
198    }
199}

201/// One setting: what it is, its control, and the way back to its default.
202/// While `off`, it cannot be changed: it waits for another setting.
203fn field_row(workspace: Workspace, form_field: FormField, off: Signal<bool>) -> impl IntoView {
204    let field = form_field.field;
205    let path = field.path();
206    let draft = {
207        let path = path.clone();
208        Signal::derive(move || {
209            workspace
210                .config_form_drafts
211                .with(|drafts| drafts.get(&path).cloned())
212        })
213    };
214    let authored = form_field.value.is_some();
215    let pending = {
216        let form_field = form_field.clone();
217        move || {
218            draft
219                .get()
220                .is_some_and(|draft| matches!(form_field.change(&draft), Ok(Some(_))))
221        }
222    };
223    let problem = {
224        let form_field = form_field.clone();
225        move || {
226            draft
227                .get()
228                .and_then(|draft| form_field.change(&draft).err())
229        }
230    };
231    let picker_open = RwSignal::new(false);
232    let picker = RwSignal::new(Vec::<ModelChoice>::new());
233    let resettable = field.kind != FieldKind::Secret;
234    let reset = {
235        let path = path.clone();
236        move |_| {
237            if field.kind == FieldKind::Models {
238                picker_open.set(false);
239            }
240            workspace.config_form_drafts.update(|drafts| {
241                if authored {
242                    drafts.insert(path.clone(), String::new());
243                } else {
244                    drafts.remove(&path);
245                }
246            });
247        }
248    };
249    let show_reset = move || {
250        resettable
251            && draft
252                .get()
253                .map_or(authored, |draft| !draft.trim().is_empty())
254    };
255    let control = control(workspace, &form_field, draft, picker, picker_open);
256    let row_path = path.clone();
257    let shown_path = path.clone();
258    let highlight = pending.clone();
259    let picker_panel = (field.kind == FieldKind::Models).then(|| {
260        let slot = match field.key[0] {
261            "small_model" => Slot::Small,
262            "utility_model" => Slot::Utility,
263            _ => Slot::Default,
264        };
265        let form_field = form_field.clone();
266        let path = path.clone();
267        // What the picker holds becomes the draft; a list that says what
268        // the file says is none.
269        Effect::watch(
270            move || picker.get(),
271            move |chosen, _, _| {
272                if !picker_open.get_untracked() {
273                    return;
274                }
275                let default_of = default_variants(workspace);
276                let text = if chosen.is_empty() {
277                    String::new()
278                } else {
279                    models_to_value(chosen, default_of).to_string()
280                };
281                let unchanged = form_field
282                    .value
283                    .as_ref()
284                    .and_then(models_from_value)
285                    .map_or_else(String::new, |current| {
286                        models_to_value(&current, default_variants(workspace)).to_string()
287                    });
288                workspace.config_form_drafts.update(|drafts| {
289                    if text == unchanged {
290                        drafts.remove(&path);
291                    } else {
292                        drafts.insert(path.clone(), text);
293                    }
294                });
295            },
296            false,
297        );
298        move || {
299            picker_open.get().then(|| view! {
300                <div class="flex flex-col gap-2 border-t border-line bg-surface-2 px-3 py-3" data-model-picker>
301                    {model_step(workspace, slot, picker, false)}
302                    <div class="flex gap-2">
303                        <button type="button" class=PRIMARY data-picker-done on:click=move |_| picker_open.set(false)>"Done"</button>
304                        <button type="button" class=BUTTON data-picker-clear on:click=move |_| picker.set(Vec::new())>"Clear"</button>
305                    </div>
306                </div>
307            })
308        }
309    });
310    let backends_panel = (field.kind == FieldKind::Backends)
311        .then(|| backends_editor(workspace, form_field.clone(), draft));
312    view! {
313        <fieldset class="m-0 min-w-0 border-0 p-0" class:opacity-50=move || off.get() disabled=move || off.get() data-config-field=row_path>
314            <div class=ROW_GRID class:bg-sel=highlight>
315                <div class="min-w-0">
316                    <div class="flex flex-wrap items-center gap-2 text-[0.8125rem] font-medium">
317                        {field.label}
318                        {move || pending().then(|| view! { <span class="rounded-full bg-accent-soft px-2 text-[0.6875rem] font-medium text-accent">"not applied"</span> })}
319                    </div>
320                    <p class="text-xs text-muted">{field.help}</p>
321                    <p class="font-mono text-[0.6875rem] text-faint">{shown_path}</p>
322                </div>
323                <div class="col-start-1 row-start-2 min-w-0 lg:col-start-2 lg:row-start-1">
324                    {control}
325                    {move || problem().map(|problem| view! { <p class="mt-1 text-xs text-bad" data-field-error>{problem}</p> })}
326                </div>
327                <div class="col-start-2 row-span-2 row-start-1 flex justify-center lg:col-start-3 lg:row-span-1">
328                    {move || show_reset().then(|| {
329                        let reset = reset.clone();
330                        view! {
331                            <button type="button" class=ROW_ACTION title="Back to the default: removes the key" aria-label="Back to the default" data-field-reset on:click=reset>
332                                <Glyph icon=icondata::LuRotateCcw />
333                            </button>
334                        }
335                    })}
336                </div>
337            </div>
338            {picker_panel}
339            {backends_panel}
340        </fieldset>
341    }
342}

344/// Looks up the default variant of a model the daemon offers.
345fn default_variants(workspace: Workspace) -> impl Fn(&ModelKey) -> Option<VariantId> {
346    let models = workspace.models();
347    move |key| {
348        models
349            .iter()
350            .find(|model| &model.key == key)
351            .and_then(|model| model.default_variant.clone())
352    }
353}

355/// Stores `text` as the draft of `field`, or none when it is what the file
356/// holds.
357fn set_draft(workspace: Workspace, field: &FormField, text: String) {
358    let current = field.current_text();
359    let path = field.field.path();
360    workspace.config_form_drafts.update(|drafts| {
361        if text == current {
362            drafts.remove(&path);
363        } else {
364            drafts.insert(path, text);
365        }
366    });
367}

369fn control(
370    workspace: Workspace,
371    form_field: &FormField,
372    draft: Signal<Option<String>>,
373    picker: RwSignal<Vec<ModelChoice>>,
374    picker_open: RwSignal<bool>,
375) -> AnyView {
376    let field: &'static ConfigField = form_field.field;
377    let current = form_field.current_text();
378    let for_change = form_field.clone();
379    match field.kind {
380        FieldKind::Secret => {
381            let label = if form_field.value.is_some() {
382                "•••••••• (set)"
383            } else {
384                "not set"
385            };
386            view! { <p class="font-mono text-[0.8125rem] text-muted lg:text-right">{label}</p> }
387                .into_any()
388        }
389        FieldKind::Bool => {
390            let shown = form_field.clone();
391            view! {
392                <label class="-ml-3 flex min-h-11 w-fit min-w-11 cursor-pointer items-center justify-center lg:ml-auto lg:-mr-3">
393                    <input
394                        type="checkbox"
395                        class="hit-area h-5 w-5 accent-accent"
396                        aria-label=field.label
397                        prop:checked=move || shown.shown_text(draft.get().as_deref()) == "true"
398                        on:change=move |event| set_draft(workspace, &for_change, event_target_checked(&event).to_string())
399                    />
400                </label>
401            }
402            .into_any()
403        }
404        FieldKind::Choice(options) => {
405            let value = move || draft.get().unwrap_or_else(|| current.clone());
406            let value = Signal::derive(value);
407            let default = field.default_text().unwrap_or_default();
408            view! {
409                <select
410                    class=FIELD
411                    aria-label=field.label
412                    on:change=move |event| set_draft(workspace, &for_change, event_target_value(&event))
413                >
414                    <option value="" prop:selected=move || value.get().is_empty()>{format!("default ({default})")}</option>
415                    {options.iter().map(|&option| view! {
416                        <option value=option prop:selected=move || value.get() == option>{option}</option>
417                    }).collect_view()}
418                </select>
419            }
420            .into_any()
421        }
422        FieldKind::Choices(options) => {
423            let value = Signal::derive(move || draft.get().unwrap_or_else(|| current.clone()));
424            let chosen = move || {
425                value.with(|text| {
426                    text.split(',')
427                        .map(str::trim)
428                        .filter(|name| !name.is_empty())
429                        .map(str::to_owned)
430                        .collect::<Vec<_>>()
431                })
432            };
433            view! {
434                <div class="flex flex-wrap gap-x-2.5 gap-y-1 font-mono text-xs">
435                    {options.iter().map(|&option| {
436                        let for_change = for_change.clone();
437                        view! {
438                            <label class="flex min-h-11 cursor-pointer items-center gap-1.5 lg:min-h-7">
439                                <input
440                                    type="checkbox"
441                                    class="hit-area h-4 w-4 accent-accent"
442                                    prop:checked=move || chosen().iter().any(|name| name == option)
443                                    on:change=move |event| {
444                                        let on = event_target_checked(&event);
445                                        let mut names = chosen();
446                                        names.retain(|name| name != option);
447                                        if on {
448                                            names.push(option.to_owned());
449                                        }
450                                        // In the list's order, so ticking and
451                                        // unticking comes back to the file's.
452                                        let ordered = options.iter().copied().filter(|name| names.iter().any(|chosen| chosen == name)).collect::<Vec<_>>();
453                                        set_draft(workspace, &for_change, ordered.join(","));
454                                    }
455                                />
456                                {option}
457                            </label>
458                        }
459                    }).collect_view()}
460                </div>
461            }
462            .into_any()
463        }
464        FieldKind::Models => {
465            let shown = form_field.clone();
466            let note = match field.default {
467                cookie_agent_client_state::config_form::FieldDefault::Note(note) => note,
468                cookie_agent_client_state::config_form::FieldDefault::Value(value) => value,
469            };
470            let chosen_now = move || {
471                let text = draft
472                    .get_untracked()
473                    .unwrap_or_else(|| shown.current_text());
474                serde_json::from_str::<Value>(&text)
475                    .ok()
476                    .as_ref()
477                    .and_then(models_from_value)
478                    .unwrap_or_default()
479            };
480            let label_of = {
481                let form_field = form_field.clone();
482                move || {
483                    let text = draft.get().unwrap_or_else(|| form_field.current_text());
484                    serde_json::from_str::<Value>(&text)
485                        .ok()
486                        .as_ref()
487                        .and_then(models_from_value)
488                        .filter(|choices| !choices.is_empty())
489                        .map(|choices| {
490                            choices
491                                .iter()
492                                .map(|choice| display_choice(workspace, choice))
493                                .collect::<Vec<_>>()
494                                .join(", then ")
495                        })
496                }
497            };
498            // The IDs, for the tooltip.
499            let ids_of = {
500                let form_field = form_field.clone();
501                move || {
502                    let text = draft.get().unwrap_or_else(|| form_field.current_text());
503                    serde_json::from_str::<Value>(&text)
504                        .ok()
505                        .as_ref()
506                        .and_then(models_from_value)
507                        .map(|choices| {
508                            choices
509                                .iter()
510                                .map(choice_label)
511                                .collect::<Vec<_>>()
512                                .join(", then ")
513                        })
514                        .unwrap_or_default()
515                }
516            };
517            let has_value = {
518                let label_of = label_of.clone();
519                move || label_of().is_some()
520            };
521            // One picker: the models by their display names
522            // (docs/site/specs/web-polish.md, P6).
523            view! {
524                <div class="flex flex-col items-start gap-1 lg:items-end">
525                    <button
526                        type="button"
527                        class=format!("{BUTTON} w-full max-w-[18rem] justify-between lg:w-[16rem]")
528                        title=ids_of
529                        data-model-change
530                        on:click=move |_| {
531                            if picker_open.get_untracked() {
532                                picker_open.set(false);
533                                return;
534                            }
535                            // A model's variant shows on its chips, the
536                            // default's included.
537                            let models = workspace.models();
538                            let chosen = chosen_now().into_iter().map(|choice| match choice {
539                                ModelChoice::Model { model, variant: None } => {
540                                    let variant = models.iter().find(|each| each.key == model).and_then(|each| each.default_variant.clone());
541                                    ModelChoice::Model { model, variant }
542                                }
543                                other => other,
544                            }).collect();
545                            picker.set(chosen);
546                            picker_open.set(true);
547                        }
548                    >
549                        <span class="flex min-w-0 items-center gap-1.5">
550                            <Glyph icon=icondata::LuSparkles class="shrink-0 text-muted" />
551                            <span class="min-w-0 truncate" class:text-faint=move || !has_value() data-model-label>
552                                {move || label_of().unwrap_or_else(|| note.to_owned())}
553                            </span>
554                        </span>
555                        <Glyph icon=icondata::LuChevronDown class="shrink-0 text-xs text-muted" />
556                    </button>
557                </div>
558            }
559            .into_any()
560        }
561        FieldKind::Backends => {
562            let count = move || backend_rows(&draft.get().unwrap_or_else(|| current.clone())).len();
563            view! {
564                <p class="text-[0.8125rem] text-muted lg:text-right" data-backend-count>
565                    {move || match count() {
566                        0 => "off".to_owned(),
567                        1 => "1 backend".to_owned(),
568                        count => format!("{count} backends"),
569                    }}
570                </p>
571            }
572            .into_any()
573        }
574        FieldKind::Keyed(keys) => {
575            let value = Signal::derive(move || draft.get().unwrap_or_else(|| current.clone()));
576            let default = field.default_text().unwrap_or_default();
577            // The key and number shown: the draft's, else the default's.
578            let parts = move || {
579                let text = value.get();
580                let text = if text.trim().is_empty() {
581                    default.to_owned()
582                } else {
583                    text
584                };
585                let (key, number) = text.split_once('=').unwrap_or((text.as_str(), ""));
586                (key.trim().to_owned(), number.trim().to_owned())
587            };
588            let for_key = for_change.clone();
589            view! {
590                <div class="flex gap-1.5">
591                    <select
592                        class=format!("{FIELD} w-auto shrink-0")
593                        aria-label=format!("{} as", field.label)
594                        on:change=move |event| {
595                            let (_, number) = parts();
596                            set_draft(workspace, &for_key, format!("{}={number}", event_target_value(&event)));
597                        }
598                    >
599                        {keys.iter().map(|&key| {
600                            view! { <option value=key prop:selected=move || parts().0 == key>{key}</option> }
601                        }).collect_view()}
602                    </select>
603                    <input
604                        type="text"
605                        class=format!("{FIELD} min-w-0")
606                        aria-label=field.label
607                        autocomplete="off"
608                        inputmode="numeric"
609                        prop:value=move || parts().1
610                        on:input=move |event| {
611                            let (key, _) = parts();
612                            set_draft(workspace, &for_change, format!("{key}={}", event_target_value(&event).trim()));
613                        }
614                    />
615                </div>
616            }
617            .into_any()
618        }
619        FieldKind::Integer | FieldKind::Number | FieldKind::Text => {
620            let placeholder = match field.default {
621                cookie_agent_client_state::config_form::FieldDefault::Value(value)
622                | cookie_agent_client_state::config_form::FieldDefault::Note(value) => value,
623            };
624            let numeric = matches!(field.kind, FieldKind::Integer | FieldKind::Number);
625            view! {
626                <input
627                    type="text"
628                    class=FIELD
629                    aria-label=field.label
630                    autocomplete="off"
631                    spellcheck="false"
632                    inputmode=if numeric { "numeric" } else { "text" }
633                    placeholder=placeholder
634                    prop:value=move || draft.get().unwrap_or_else(|| current.clone())
635                    on:input=move |event| set_draft(workspace, &for_change, event_target_value(&event))
636                />
637            }
638            .into_any()
639        }
640    }
641}

643/// The backends a [`FieldKind::Backends`] text lists; blank is none.
644fn backend_rows(text: &str) -> Vec<Map<String, Value>> {
645    serde_json::from_str::<Vec<Value>>(text)
646        .unwrap_or_default()
647        .into_iter()
648        .filter_map(|row| match row {
649            Value::Object(row) => Some(row),
650            _ => None,
651        })
652        .collect()
653}

655/// `[speech] backends`, one editable row each, in the order they are tried.
656/// A row's `auth` and `headers` are kept as they are; the file changes them.
657fn backends_editor(
658    workspace: Workspace,
659    form_field: FormField,
660    draft: Signal<Option<String>>,
661) -> impl IntoView {
662    let current = form_field.current_text();
663    let rows = Memo::new(move |_| backend_rows(&draft.get().unwrap_or_else(|| current.clone())));
664    let write = {
665        let form_field = form_field.clone();
666        move |rows: Vec<Map<String, Value>>| {
667            let text = if rows.is_empty() {
668                String::new()
669            } else {
670                Value::Array(rows.into_iter().map(Value::Object).collect()).to_string()
671            };
672            set_draft(workspace, &form_field, text);
673        }
674    };
675    let edit = {
676        let write = write.clone();
677        move |index: usize, change: &dyn Fn(&mut Map<String, Value>)| {
678            let mut all = rows.get_untracked();
679            if let Some(row) = all.get_mut(index) {
680                change(row);
681                write(all);
682            }
683        }
684    };
685    // The providers a backend can use: those with a connection or
686    // configuration.
687    let providers = move || {
688        workspace
689            .providers()
690            .into_iter()
691            .filter(|provider| {
692                provider.presence == cookie_agent_protocol::ProviderPresence::Current
693                    && provider.configuration
694                        != cookie_agent_protocol::ProviderConfigurationState::Unconfigured
695            })
696            .map(|provider| (provider.id.to_string(), provider.display_name.to_string()))
697            .collect::<Vec<_>>()
698    };
699    let add = {
700        let write = write.clone();
701        move |_| {
702            let mut all = rows.get_untracked();
703            let mut row = Map::new();
704            match providers().into_iter().next() {
705                Some((id, _)) => row.insert("provider".into(), Value::String(id)),
706                None => row.insert("endpoint".into(), Value::String(String::new())),
707            };
708            row.insert("model".into(), Value::String(String::new()));
709            all.push(row);
710            write(all);
711        }
712    };
713    let text_of = move |index: usize, key: &'static str| {
714        move || {
715            rows.with(|rows| {
716                rows.get(index)
717                    .and_then(|row| row.get(key))
718                    .map(|value| match value {
719                        Value::String(text) => text.clone(),
720                        other => other.to_string(),
721                    })
722                    .unwrap_or_default()
723            })
724        }
725    };
726    let row_view = move |index: usize| {
727        let has = move |key: &'static str| {
728            rows.with(|rows| rows.get(index).is_some_and(|row| row.contains_key(key)))
729        };
730        let source = {
731            let edit = edit.clone();
732            move |event: leptos::ev::Event| {
733                let chosen = event_target_value(&event);
734                edit(index, &|row| {
735                    if chosen.is_empty() {
736                        row.remove("provider");
737                        row.entry("endpoint")
738                            .or_insert(Value::String(String::new()));
739                    } else {
740                        row.remove("endpoint");
741                        row.remove("auth");
742                        row.insert("provider".into(), Value::String(chosen.clone()));
743                    }
744                });
745            }
746        };
747        // A text key: blank removes an optional one and keeps a required
748        // one, so the row says what it lacks.
749        let typed = {
750            let edit = edit.clone();
751            move |key: &'static str, required: bool| {
752                let edit = edit.clone();
753                move |event: leptos::ev::Event| {
754                    let text = event_target_value(&event);
755                    let text = text.trim();
756                    edit(index, &|row| {
757                        if text.is_empty() && !required {
758                            row.remove(key);
759                        } else if key == "timeout_secs" {
760                            let value = text
761                                .parse::<i64>()
762                                .map_or_else(|_| Value::String(text.to_owned()), Value::from);
763                            row.insert(key.into(), value);
764                        } else {
765                            row.insert(key.into(), Value::String(text.to_owned()));
766                        }
767                    });
768                }
769            }
770        };
771        let remove = {
772            let write = write.clone();
773            move |_| {
774                let mut all = rows.get_untracked();
775                if index < all.len() {
776                    all.remove(index);
777                    write(all);
778                }
779            }
780        };
781        let raise = {
782            let write = write.clone();
783            move |_| {
784                let mut all = rows.get_untracked();
785                if index > 0 && index < all.len() {
786                    all.swap(index - 1, index);
787                    write(all);
788                }
789            }
790        };
791        let on_endpoint = typed("endpoint", true);
792        let provider = text_of(index, "provider");
793        let chosen = text_of(index, "provider");
794        let options = move || {
795            let mut options = providers();
796            let named = chosen();
797            if !named.is_empty() && !options.iter().any(|(id, _)| *id == named) {
798                options.push((named.clone(), named));
799            }
800            options
801        };
802        view! {
803            <li class="flex flex-col gap-1.5 border-t border-line pt-2.5 first:border-t-0 first:pt-0" data-backend=index>
804                <div class="flex items-center gap-2">
805                    <span class="text-[0.6875rem] font-semibold text-muted">{move || if index == 0 { "First".to_owned() } else { format!("Then {}", index + 1) }}</span>
806                    <span class="mr-auto"></span>
807                    {(index > 0).then(|| view! {
808                        <button type="button" class=ROW_ACTION title="Try it earlier" aria-label="Try it earlier" data-backend-raise on:click=raise>
809                            <Glyph icon=icondata::LuArrowUp />
810                        </button>
811                    })}
812                    <button type="button" class=ROW_ACTION title="Remove the backend" aria-label="Remove the backend" data-backend-remove on:click=remove>
813                        <Glyph icon=icondata::LuTrash2 />
814                    </button>
815                </div>
816                <div class="grid gap-1.5 sm:grid-cols-2">
817                    <select class=FIELD aria-label="Service" on:change=source data-backend-source>
818                        {move || options().into_iter().map(|(id, name)| {
819                            let selected = provider() == id;
820                            view! { <option value=id.clone() selected=selected>{format!("{name} ({id})")}</option> }
821                        }).collect_view()}
822                        <option value="" selected=move || !has("provider")>"OpenAI-compatible endpoint"</option>
823                    </select>
824                    <input type="text" class=FIELD aria-label="Model" placeholder="model, such as gpt-4o-mini-transcribe" autocomplete="off" spellcheck="false"
825                        prop:value=text_of(index, "model") on:input=typed("model", true) data-backend-model />
826                    {move || (!has("provider")).then(|| view! {
827                        <input type="text" class=format!("{FIELD} sm:col-span-2") aria-label="Endpoint" placeholder="https://host/v1, or http:// to a loopback address" autocomplete="off" spellcheck="false"
828                            prop:value=text_of(index, "endpoint") on:input=on_endpoint.clone() data-backend-endpoint />
829                    })}
830                    <input type="text" class=FIELD aria-label="Name" placeholder="name: the provider, else the host" autocomplete="off" spellcheck="false"
831                        prop:value=text_of(index, "name") on:input=typed("name", false) />
832                    <input type="text" class=FIELD aria-label="Time limit in seconds" placeholder="time limit, seconds: none" autocomplete="off" inputmode="numeric"
833                        prop:value=text_of(index, "timeout_secs") on:input=typed("timeout_secs", false) />
834                </div>
835                {move || (has("auth") || has("headers")).then(|| view! {
836                    <p class="text-xs text-muted">
837                        "Its auth and headers are kept as they are; "
838                        <button type="button" class="font-medium text-accent hover:underline" on:click=move |_| workspace.edit_config("config.toml")>"change them in the file"</button>
839                        "."
840                    </p>
841                })}
842            </li>
843        }
844    };
845    view! {
846        <div class="flex flex-col gap-2.5 border-t border-line bg-surface-2 px-3 py-3" data-backends>
847            <ol class="flex flex-col gap-2.5">
848                <For each=move || 0..rows.with(Vec::len) key=|index| *index children=row_view.clone() />
849            </ol>
850            {move || rows.with(Vec::is_empty).then(|| view! {
851                <p class=NOTE>"Add a backend to turn voice input on; the settings below wait for one."</p>
852            })}
853            <div>
854                <button type="button" class=BUTTON on:click=add data-backend-add>
855                    <Glyph icon=icondata::LuPlus />
856                    "Add backend"
857                </button>
858            </div>
859        </div>
860    }
861}

917/// A model choice by its display names, as the composer's chip names a
918/// model and its level: `Claude Opus 5.5 · High`.
919fn display_choice(workspace: Workspace, choice: &ModelChoice) -> String {
920    match choice {
921        ModelChoice::Model { model, variant } => {
922            let name = workspace.with_runtime(|runtime| runtime.model_names().short_label(model));
923            let level = variant.as_ref().map(|variant| {
924                workspace
925                    .models()
926                    .into_iter()
927                    .find(|each| &each.key == model)
928                    .and_then(|each| each.variants.into_iter().find(|each| &each.id == variant))
929                    .map_or_else(|| variant.to_string(), |each| each.display_name)
930            });
931            cookie_agent_client_state::effort::chip_label(&name, level.as_deref())
932        }
933        other => choice_label(other),
934    }
