45/// The sidebar button: it shows and hides the sidebar; on a phone's second
46/// level (an item open in a view's list) it goes back up to the list. A
47/// dot on it says a session is unread while the sidebar is hidden.
48#[component]
49pub fn SidebarButton(workspace: Workspace) -> impl IntoView {
50    let shown = move || {
51        workspace.sidebar_open.get() || (workspace.desktop.get() && workspace.sidebar_pinned.get())
52    };
53    let back_up = move || {
54        workspace.view.get() != View::Chat && workspace.narrow.get() && !workspace.at_list()
55    };
56    let title = move || {
57        if back_up() {
58            format!("Back to {}", workspace.view.get().label())
59        } else if workspace.sidebar_open.get() {
60            "Hide the sidebar".to_owned()
61        } else {
62            "Show or hide the sidebar".to_owned()
63        }
64    };
65    view! {
66        <button
67            type="button"
68            class=format!("{ICON_BUTTON} relative")
69            aria-label=move || if back_up() { "Back" } else { "Sidebar" }
70            title=title
71            data-sidebar-toggle
72            aria-expanded=move || shown().to_string()
73            on:click=move |_| workspace.toggle_sidebar()
74        >
75            <Glyph
76                icon=Signal::derive(move || {
77                    if back_up() { icondata::LuArrowLeft } else { icondata::LuPanelLeft }
78                })
79                class="text-lg"
80            />
81            // A session is unread and the sidebar, which would show it, is
82            // hidden.
83            <Show when=move || !shown() && workspace.any_unread()>
84                <span
85                    class="absolute top-2 right-2 h-2 w-2 rounded-full bg-accent ring-2 ring-surface"
86                    data-unread-any
87                    aria-hidden="true"
88                ></span>
89            </Show>
90        </button>
91    }
92}

94/// A view other than Chat, as a full page: its title bar, then its
95/// content. Agents' title bar has the sidebar button, and so has a phone's
96/// second level, to go back up to the list; a detail page has a back
97/// arrow, which on a phone only a third level needs.
98#[component]
99pub fn PageView(workspace: Workspace, connection: Connection) -> impl IntoView {
100    let back = move || {
101        let depth = workspace
102            .open_layers()
103            .iter()
104            .filter(|layer| matches!(layer, Layer::Section | Layer::Editor))
105            .count();
106        depth > usize::from(workspace.narrow.get())
107    };
108    let title = move || match workspace.view.get() {
109        View::Settings if workspace.narrow.get() => workspace
110            .settings_section
111            .get()
112            .map_or("Settings", SettingsSection::label),
113        view => view.label(),
114    };
115    view! {
116        <section
117            id="page"
118            class="flex min-w-0 flex-1 flex-col"
119            data-view=move || workspace.view.get().id()
120        >
121            // An asset open on its page has a head of its own.
122            <div
123                class="flex h-12 shrink-0 items-center gap-1 border-b border-line px-2"
124                class:hidden=move || workspace.view.get() == View::Assets && workspace.asset_page.with(Option::is_some)
125            >
126                <SidebarButton workspace=workspace />
127                <Show when=back>
128                    <button
129                        type="button"
130                        class=ICON_BUTTON
131                        aria-label="Back"
132                        title="Back"
133                        data-page-back
134                        on:click=move |_| workspace.go_back()
135                    >
136                        <Glyph icon=icondata::LuArrowLeft class="text-lg" />
137                    </button>
138                </Show>
139                <h1 class="min-w-0 flex-1 truncate px-1 text-[0.9375rem] font-semibold tracking-tight" id="page-title">{title}</h1>
140            </div>
141            {move || match workspace.view.get() {
142                View::Settings => settings_page(workspace, connection).into_any(),
143                View::Agents => agents_page(workspace).into_any(),
144                // The list stays, hidden, under an asset's page, so Back
145                // finds it scrolled where it was.
146                View::Assets => view! {
147                    <div
148                        class="flex min-h-0 flex-1 flex-col overflow-y-auto px-3 py-4 sm:px-8 sm:py-6"
149                        class:hidden=move || workspace.asset_page.with(Option::is_some)
150                        data-page-body
151                    >
152                        <div class="mx-auto flex w-full max-w-6xl flex-1 flex-col gap-3">
153                            {super::assets::assets_page(workspace)}
154                        </div>
155                    </div>
156                    <super::asset_page::AssetPage workspace />
157                }
158                .into_any(),
159                view => view! {
160                    <div class="flex min-h-0 flex-1 flex-col overflow-y-auto px-3 py-4 sm:px-8 sm:py-6" data-page-body>
161                        <div class="mx-auto flex w-full max-w-3xl flex-1 flex-col gap-3">
162                            {match view {
163                                View::Machines => machines_page(workspace).into_any(),
164                                View::Memory => memory_sheet(workspace).into_any(),
165                                View::Cron => cron_page(workspace).into_any(),
166                                View::Skills => super::skills::skills_page(workspace).into_any(),
167                                View::Sharing => super::shares::sharing_page(workspace).into_any(),
168                                View::Mcp => mcp_sheet(workspace).into_any(),
169                                View::Chat | View::Settings | View::Agents | View::Assets => ().into_any(),
170                            }}
171                        </div>
172                    </div>
173                }
174                .into_any(),
175            }}
176        </section>
177    }
178}

193/// A Settings section's row: its icon and name; a phone, whose list is the
194/// page, says what it holds below.
195const SECTION_ROW: &str = "group flex w-full shrink-0 flex-col items-start gap-0.5 rounded-lg px-2.5 py-1.5 text-left transition-colors hover:bg-hover active:bg-press aria-[current=page]:bg-sel compact:min-h-12 compact:px-3 compact:py-2";

197/// Settings: the section list beside the open section; on a phone the
198/// list, then one section at a time.
199fn settings_page(workspace: Workspace, connection: Connection) -> impl IntoView {
200    let entries = SettingsSection::ALL
201        .into_iter()
202        .map(|section| {
203            let current = move || workspace.shown_section() == Some(section);
204            view! {
205                <button
206                    type="button"
207                    class=SECTION_ROW
208                    aria-current=move || current().then_some("page")
209                    data-section=section.id()
210                    on:click=move |_| workspace.open_section(Some(section))
211                >
212                    <span class="flex w-full items-center gap-2.5">
213                        <Glyph icon=section_icon(section) class="text-base text-muted group-aria-[current=page]:text-accent" />
214                        <span class="min-w-0 flex-1 text-[0.8125rem] text-fg-2 group-aria-[current=page]:font-semibold group-aria-[current=page]:text-fg">{section.label()}</span>
215                        <Glyph icon=icondata::LuChevronRight class="text-xs opacity-60 sm:hidden" />
216                    </span>
217                    <span class="hidden pl-[1.625rem] text-xs text-muted compact:block">{section.note()}</span>
218                </button>
219            }
220        })
221        .collect_view();
222    view! {
223        <div class="flex min-h-0 flex-1">
224            <nav
225                aria-label="Settings sections"
226                class="flex w-full shrink-0 flex-col gap-px overflow-y-auto p-2 sm:w-56 sm:border-r sm:border-line"
227                class:hidden=move || workspace.shown_section().is_some() && workspace.narrow.get()
228            >
229                <Show when=move || workspace.onboarding.status.with(Option::is_some)>
230                    <button
231                        type="button"
232                        id="settings-onboard"
233                        class=SECTION_ROW
234                        on:click=move |_| workspace.open_onboarding()
235                    >
236                        <span class="flex w-full items-center gap-2.5">
237                            <Glyph icon=icondata::LuWandSparkles class="text-base text-accent" />
238                            <span class="min-w-0 flex-1 text-[0.8125rem] font-medium text-fg-2">"Set up Cookie"</span>
239                        </span>
240                        <span class="hidden pl-[1.625rem] text-xs text-muted compact:block">"model providers or an ACP agent, and tools"</span>
241                    </button>
242                </Show>
243                {entries}
244            </nav>
245            {move || {
246                workspace.shown_section().map(|section| view! {
247                    <div class="flex min-h-0 min-w-0 flex-1 flex-col overflow-y-auto px-3 py-4 sm:px-8 sm:py-6" data-page-body data-section-body=section.id()>
248                        <div class="mx-auto flex w-full max-w-3xl flex-1 flex-col gap-3">
249                            {match section {
250                                SettingsSection::Configuration => config_sheet(workspace).into_any(),
251                                SettingsSection::Providers => connect_sheet(workspace).into_any(),
252                                SettingsSection::Plugins => plugins_sheet(workspace).into_any(),
253                                SettingsSection::Notifications => super::notifications::notifications_section(workspace, connection).into_any(),
254                                SettingsSection::Display => display_section(workspace).into_any(),
255                                SettingsSection::Browser => browser_section(connection).into_any(),
256                                SettingsSection::About => crate::app::about_section(connection).into_any(),
257                            }}
258                        </div>
259                    </div>
260                })
261            }}
262        </div>
263    }
264}
