27/// A rail button: an icon in a 44 px square, its name in a tooltip beside
28/// it (`data-tip`).
29const RAIL_BUTTON: &str = "relative inline-flex h-11 w-11 shrink-0 items-center justify-center rounded-lg text-ink hover:bg-toasted hover:text-espresso active:bg-crust dark:text-sand dark:hover:bg-ember dark:hover:text-latte dark:active:bg-smoke";

31/// The icon of a view on the rail.
32pub fn view_icon(view: View) -> crate::icons::Icon {
33    match view {
34        View::Chat => icondata::FaCommentsSolid,
35        View::Agents => icondata::FaRobotSolid,
36        View::Settings => icondata::FaGearSolid,
37        View::Machines => icondata::FaServerSolid,
38        View::Memory => icondata::FaBrainSolid,
39    }
40}

42/// The rail: the views, then the connection and the theme at its foot.
43/// `primary` is the one always in the page (the drawer has a second).
44#[component]
45pub fn Rail(
46    workspace: Workspace,
47    connection: Connection,
48    theme: RwSignal<ThemePreference>,
49    primary: bool,
50) -> impl IntoView {
51    let buttons = View::ALL
52        .into_iter()
53        .map(|view| {
54            let current = move || workspace.view.get() == view;
55            view! {
56                <button
57                    type="button"
58                    class=RAIL_BUTTON
59                    class:bg-toasted=current
60                    class:text-espresso=current
61                    class:dark:bg-ember=current
62                    class:dark:text-latte=current
63                    aria-label=view.label()
64                    data-tip=view.label()
65                    data-view=view.id()
66                    aria-current=move || current().then_some("page")
67                    on:click=move |_| workspace.open_view(view, None)
68                >
69                    // The active view's mark: a bar at the rail's edge.
70                    <span
71                        class="absolute top-2 bottom-2 -left-1.5 w-1 rounded-r bg-caramel dark:bg-maple"
72                        class:hidden=move || !current()
73                        aria-hidden="true"
74                    ></span>
75                    <Glyph icon=view_icon(view) class="text-lg" />
76                </button>
77            }
78        })
79        .collect_view();
80    view! {
81        <nav
82            aria-label="Views"
83            class="rail flex w-14 shrink-0 flex-col items-center gap-1 border-r border-toasted bg-parchment py-2 dark:border-ember dark:bg-char"
84            data-rail=if primary { "primary" } else { "drawer" }
85        >
86            <img src="icons/icon.svg" alt="" class="mb-1 h-7 w-7 short:hidden" />
87            {buttons}
88            <span class="flex-1"></span>
89            <ConnectionStatus connection=connection compact=true primary=primary />
90            <ThemeToggle theme=theme class=RAIL_BUTTON in_rail=true />
91        </nav>
92    }
93}

95/// The sessions over the page: a phone's drawer (the rail beside the
96/// sessions, 85 % wide), or the sessions sliding over a tablet's
97/// transcript. A desktop pins them instead.
98#[component]
99pub fn Drawer(
100    workspace: Workspace,
101    connection: Connection,
102    theme: RwSignal<ThemePreference>,
103) -> impl IntoView {
104    view! {
105        <Show when=move || workspace.sidebar_open.get()>
106            <div
107                class="fixed inset-0 z-30 bg-espresso/30 lg:hidden dark:bg-night/60"
108                data-drawer-backdrop
109                on:click=move |_| workspace.sidebar_open.set(false)
110            ></div>
111            <div
112                id="drawer"
113                role="dialog"
114                aria-label="Sessions"
115                class="sheet-frame fixed top-0 left-0 z-40 flex w-[85%] max-w-sm shadow-xl sm:bottom-0 sm:left-14 sm:w-80 lg:hidden"
116            >
117                <div class="flex sm:hidden">
118                    <Rail workspace=workspace connection=connection theme=theme primary=false />
119                </div>
120                <div class="flex min-w-0 flex-1 flex-col overflow-y-auto border-r border-toasted bg-parchment dark:border-ember dark:bg-char">
121                    <SessionList workspace=workspace />
122                </div>
123            </div>
124        </Show>
125    }
126}

128/// The sidebar button: it shows and hides the sessions.
129#[component]
130pub fn SidebarButton(workspace: Workspace, #[prop(optional)] class: &'static str) -> impl IntoView {
131    let shown = move || {
132        workspace.sidebar_open.get()
133            || (workspace.view.get() == View::Chat && workspace.sidebar_pinned.get())
134    };
135    view! {
136        <button
137            type="button"
138            class=format!("{ICON_BUTTON} {class}")
139            aria-label="Sessions"
140            title="Show or hide the sessions"
141            data-sidebar-toggle
142            aria-expanded=move || shown().to_string()
143            on:click=move |_| workspace.toggle_sidebar()
144        >
145            <Glyph icon=icondata::FaTableColumnsSolid class="text-lg" />
146        </button>
147    }
148}

150/// A view other than Chat, as a full page: its title bar, then its
151/// content. A phone's title bar has the sidebar button, and a back arrow
152/// on a detail page.
153#[component]
154pub fn PageView(workspace: Workspace, connection: Connection) -> impl IntoView {
155    let detail = move || {
156        workspace
157            .open_layers()
158            .iter()
159            .any(|layer| matches!(layer, Layer::Section | Layer::Editor))
160    };
161    let title = move || match workspace.view.get() {
162        View::Settings if workspace.narrow.get() => workspace
163            .settings_section
164            .get()
165            .map_or("Settings", SettingsSection::label),
166        view => view.label(),
167    };
168    view! {
169        <section
170            id="page"
171            class="flex min-w-0 flex-1 flex-col"
172            data-view=move || workspace.view.get().id()
173        >
174            <div class="flex h-12 shrink-0 items-center gap-1 border-b border-toasted px-2 dark:border-ember">
175                <SidebarButton workspace=workspace class="sm:hidden" />
176                <Show when=detail>
177                    <button
178                        type="button"
179                        class=ICON_BUTTON
180                        aria-label="Back"
181                        title="Back"
182                        data-page-back
183                        on:click=move |_| workspace.go_back()
184                    >
185                        <Glyph icon=icondata::FaArrowLeftSolid class="text-lg" />
186                    </button>
187                </Show>
188                <h1 class="min-w-0 flex-1 truncate px-1 text-lg font-bold" id="page-title">{title}</h1>
189            </div>
190            {move || match workspace.view.get() {
191                View::Settings => settings_page(workspace, connection).into_any(),
192                view => view! {
193                    <div class="flex min-h-0 flex-1 flex-col overflow-y-auto p-3" data-page-body>
194                        <div class="mx-auto flex w-full max-w-3xl flex-1 flex-col gap-2">
195                            {match view {
196                                View::Agents => config_sheet(workspace, ConfigScope::Agents).into_any(),
197                                View::Machines => machines_page(workspace).into_any(),
198                                View::Memory => memory_sheet(workspace).into_any(),
199                                View::Chat | View::Settings => ().into_any(),
200                            }}
201                        </div>
202                    </div>
203                }
204                .into_any(),
205            }}
206        </section>
207    }
208}

210/// Settings: the section list beside the open section; on a phone the
211/// list, then one section at a time.
212fn settings_page(workspace: Workspace, connection: Connection) -> impl IntoView {
213    let entries = SettingsSection::ALL
214        .into_iter()
215        .map(|section| {
216            let current = move || workspace.shown_section() == Some(section);
217            view! {
218                <button
219                    type="button"
220                    class=format!("{ROW} min-h-12")
221                    class:bg-toasted=current
222                    class:dark:bg-ember=current
223                    aria-current=move || current().then_some("page")
224                    data-section=section.id()
225                    on:click=move |_| workspace.open_section(Some(section))
226                >
227                    <span class="flex w-full items-center gap-2">
228                        <span class="min-w-0 flex-1 text-sm font-bold">{section.label()}</span>
229                        <Glyph icon=icondata::FaChevronRightSolid class="text-xs opacity-60 sm:hidden" />
230                    </span>
231                    <span class=NOTE>{section.note()}</span>
232                </button>
233            }
234        })
235        .collect_view();
236    view! {
237        <div class="flex min-h-0 flex-1">
238            <nav
239                aria-label="Settings sections"
240                class="flex w-full shrink-0 flex-col gap-1 overflow-y-auto p-2 sm:w-60 sm:border-r sm:border-toasted sm:dark:border-ember"
241                class:hidden=move || workspace.shown_section().is_some() && workspace.narrow.get()
242            >
243                {entries}
244            </nav>
245            {move || {
246                workspace.shown_section().map(|section| view! {
247                    <div class="flex min-h-0 min-w-0 flex-1 flex-col overflow-y-auto p-3" data-page-body data-section-body=section.id()>
248                        <div class="mx-auto flex w-full max-w-3xl flex-1 flex-col gap-2">
249                            {match section {
250                                SettingsSection::Configuration => config_sheet(workspace, ConfigScope::Configuration).into_any(),
251                                SettingsSection::Providers => connect_sheet(workspace).into_any(),
252                                SettingsSection::Mcp => mcp_sheet(workspace).into_any(),
253                                SettingsSection::Plugins => plugins_sheet(workspace).into_any(),
254                                SettingsSection::Browser => browser_section(workspace, connection).into_any(),
255                            }}
256                        </div>
257                    </div>
258                })
259            }}
260        </div>
261    }
262}

264/// This browser: installing the page as an app, and what it keeps for the
265/// daemon: the diagnostics the transcript shows, and the token.
266fn browser_section(workspace: Workspace, connection: Connection) -> impl IntoView {
267    let confirming = RwSignal::new(false);
268    view! {
269        {super::install::install_section()}
270        <h2 class="mt-4 text-sm font-bold">"Diagnostics in the transcript"</h2>
271        <p class=NOTE>"The same choice as Diagnostics in the session's ⋮ menu; it stays in this browser."</p>
272        {events_sheet(workspace)}
273        <h2 class="mt-4 text-sm font-bold">"This daemon"</h2>
274        <p class=NOTE>"The browser keeps the daemon's token to connect again without a link."</p>
275        <ForgetDaemon connection=connection confirming=confirming primary=false on_forget=|| {} />
276    }
