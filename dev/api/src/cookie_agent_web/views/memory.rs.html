26/// The icon of a layer's heading.
27fn layer_icon(layer: MemoryLayer) -> crate::icons::Icon {
28    match layer {
29        MemoryLayer::User => icondata::FaUserSolid,
30        MemoryLayer::Agent => icondata::FaRobotSolid,
31        MemoryLayer::Machine => icondata::FaServerSolid,
32        MemoryLayer::Project => icondata::FaFolderSolid,
33        MemoryLayer::Unknown => icondata::FaBoxSolid,

37pub fn memory_sheet(workspace: Workspace) -> impl IntoView {
38    let confirm_delete = RwSignal::new(None::<(String, String)>);
39    let confirm_project = RwSignal::new(None::<String>);
40    let new_url = RwSignal::new("project://".to_owned());
41    // The folders opened or folded on this page; the rest are as this
42    // browser keeps them, folded when it keeps nothing.
43    let toggled = RwSignal::new(HashMap::<String, bool>::new());
44    let is_open = move |project: &str| {
45        toggled
46            .with(|toggled| toggled.get(project).copied())
47            .unwrap_or_else(|| open_from_storage(browser::load(&folder_key(project)).as_deref()))
48    };
49    move || {
50        if let Some(edit) = workspace.memory_edit.get() {
51            return memory_editor(workspace, edit).into_any();
52        }
53        if let Some(pending) = workspace.memory_pending.get() {
54            return pending_choices(workspace, pending).into_any();
55        }
56        let listing = match workspace.memory.get() {
57            None => return loading("Reading memories…").into_any(),
58            Some(Err(error)) => return view! { <p class=ERROR>{error}</p> }.into_any(),
59            Some(Ok(listing)) => listing,
60        };
61        let lines = memory_lines(&listing, is_open)
62            .into_iter()
63            .map(|line| match line {
64                MemoryLine::Project {
65                    name,
66                    memories,
67                    open,
68                } => {
69                    let strays = project_strays(&listing, &name);
70                    project_folder(
71                        workspace,
72                        name,
73                        memories,
74                        open,
75                        strays,
76                        toggled,
77                        confirm_project,
78                        new_url,
79                    )
80                    .into_any()
81                }
82                MemoryLine::Row(row) => {
83                    memory_row(workspace, row, confirm_delete, new_url).into_any()
84                }
85            })
86            .collect_view();
87        view! {
88            {lines}
89            {move || confirm_delete.get().map(|(url, revision)| {
90                let deleted = (url.clone(), revision);
91                view! {
92                    <div class=CONFIRM>
93                        <p class="text-sm">{format!("Delete {url}?")}</p>
94                        <div class="flex gap-2">
95                            <button type="button" id="memory-delete-confirm" class=PRIMARY on:click=move |_| {
96                                confirm_delete.set(None);
97                                workspace.delete_memory(deleted.0.clone(), deleted.1.clone());
98                            }>
99                                "Delete"
100                            </button>
101                            <button type="button" class=BUTTON on:click=move |_| {
102                                confirm_delete.set(None);
103                                workspace.status.set("Delete cancelled.".into());
104                            }>
105                                "Keep it"
106                            </button>
107                        </div>
108                    </div>
109                }
110            })}
111            <div class="mt-3 flex flex-col gap-2 border-t border-toasted pt-3 dark:border-ember">
112                <label class="flex flex-col gap-1 text-sm">
113                    <span class="font-bold">"New memory"</span>
114                    <input
115                        type="text"
116                        id="memory-new-url"
117                        class=FIELD
118                        prop:value=move || new_url.get()
119                        on:input=move |event| new_url.set(event_target_value(&event))
120                    />
121                </label>
122                <button type="button" id="memory-new" class=PRIMARY on:click=move |_| workspace.new_memory(&new_url.get_untracked())>
123                    <Glyph icon=icondata::FaPlusSolid />
124                    "Create"
125                </button>
126            </div>
127        }
128        .into_any()
129    }
130}

132/// A row of the shared list: a layer's heading or a machine's group (a
133/// click fills in its prefix for a new memory), a memory, or what is not
134/// a memory.
135fn memory_row(
136    workspace: Workspace,
137    row: MemoryRow,
138    confirm_delete: RwSignal<Option<(String, String)>>,
139    new_url: RwSignal<String>,
140) -> impl IntoView {
141    let prefix = row.new_prefix();
142    match row {
143        MemoryRow::Layer(layer) => view! {
144            <button
145                type="button"
146                class=format!("{ROW} mt-2")
147                title="Create a memory in this layer"
148                data-memory-layer=layer.as_str()
149                on:click=move |_| new_url.set(prefix.clone())
150            >
151                <span class="flex items-center gap-2 text-xs font-bold tracking-wide text-ink uppercase dark:text-sand">
152                    <Glyph icon=layer_icon(layer) class="opacity-70" />
153                    {layer_title(layer)}
154                </span>
155            </button>
156        }
157        .into_any(),
158        MemoryRow::Group { name, .. } => view! {
159            <button
160                type="button"
161                class=format!("{ROW} pl-6")
162                title="Create a memory in this group"
163                on:click=move |_| new_url.set(prefix.clone())
164            >
165                <span class="flex items-center gap-2 font-mono text-sm">
166                    <Glyph icon=icondata::FaServerSolid class="opacity-70" />
167                    {name}
168                </span>
169            </button>
170        }
171        .into_any(),
172        MemoryRow::Memory(memory) => {
173            let url = memory.url.clone();
174            let deleted = (memory.url.clone(), memory.revision.clone());
175            let description = memory.description.clone().unwrap_or_default();
176            let indent = if memory.layer == MemoryLayer::Project {
177                "pl-10"
178            } else {
179                "pl-6"
180            };
181            view! {
182                <div class=format!("memory-row flex items-center gap-2 {indent}") data-memory=memory.url.clone() data-revision=memory.revision.clone()>
183                    <button type="button" class=format!("{ROW} min-w-0 flex-1") on:click=move |_| workspace.edit_memory(url.clone())>
184                        <span class="flex w-full items-center gap-2 font-mono text-sm">
185                            <Glyph icon=icondata::FaFileLinesRegular class="opacity-60" />
186                            <span class="min-w-0 truncate">{memory_name(&memory).to_owned()}</span>
187                        </span>
188                        <span class=format!("pl-6 {NOTE}")>{description}</span>
189                        {memory.invalid.clone().map(|reason| view! { <span class=ERROR>{reason}</span> })}
190                    </button>
191                    <button
192                        type="button"
193                        class=ICON_BUTTON
194                        aria-label="Delete"
195                        title="Delete"
196                        data-memory-delete
197                        on:click=move |_| confirm_delete.set(Some(deleted.clone()))
198                    >
199                        <Glyph icon=icondata::FaTrashCanRegular />
200                    </button>
201                </div>
202            }
203            .into_any()
204        }
205        MemoryRow::IgnoredHeading => view! { <p class="mt-2 text-sm font-bold">"Not memories"</p> }.into_any(),
206        MemoryRow::Ignored(entry) => view! {
207            <p class=format!("{NOTE} pl-6")>{format!("{} — {}", entry.path, entry.reason)}</p>
208        }
209        .into_any(),
210    }
211}

213/// A project's folder: a click opens or folds it, and the browser keeps
214/// the choice; its + starts a memory in it, and its trash deletes its
215/// memories behind a confirmation.
216#[expect(clippy::too_many_arguments, reason = "the folder's shared state")]
217fn project_folder(
218    workspace: Workspace,
219    name: String,
220    memories: usize,
221    open: bool,
222    strays: usize,
223    toggled: RwSignal<HashMap<String, bool>>,
224    confirm_project: RwSignal<Option<String>>,
225    new_url: RwSignal<String>,
226) -> impl IntoView {
227    let toggle = {
228        let name = name.clone();
229        move |_| {
230            let opened = !open;
231            if opened {
232                browser::store(&folder_key(&name), "open");
233            } else {
234                browser::forget(&folder_key(&name));
235            }
236            toggled.update(|toggled| {
237                toggled.insert(name.clone(), opened);
238            });
239        }
240    };
241    let prefix = format!("project://{name}/");
242    let confirming = {
243        let name = name.clone();
244        move || confirm_project.get().as_deref() == Some(name.as_str())
245    };
246    let deleted = name.clone();
247    let asked = name.clone();
248    let count = if memories == 1 {
249        "1 memory".to_owned()
250    } else {
251        format!("{memories} memories")
252    };
253    let kept = match strays {
254        0 => "The folder goes with them.".to_owned(),
255        1 => "Its 1 file that is not a memory stays, and so does the folder.".to_owned(),
256        strays => format!("Its {strays} files that are not memories stay, and so does the folder."),
257    };
258    let memory_question =
259        format!("Delete project {name} and its {count}? This cannot be undone. {kept}");
260    // A daemon that keeps projects deletes the project itself, with its
261    // sessions: the question counts both.
262    let project = {
263        let name = name.clone();
264        move || {
265            workspace
266                .projects
267                .with(|projects| {
268                    projects
269                        .iter()
270                        .find(|project| project.name.as_str() == name)
271                        .cloned()
272                })
273                .filter(|_| workspace.has_projects.get())
274        }
275    };
276    let question = {
277        let project = project.clone();
278        move || {
279            project().map_or_else(
280                || memory_question.clone(),
281                |project| cookie_agent_client_state::projects::delete_question(&project),
282            )
283        }
284    };
285    view! {
286        <div class="flex items-center gap-1 pl-3" data-project=name.clone()>
287            <button
288                type="button"
289                class=format!("{ROW} min-w-0 flex-1")
290                aria-expanded=open.to_string()
291                title=if open { "Fold the project" } else { "Open the project" }
292                data-project-toggle
293                on:click=toggle
294            >
295                <span class="flex w-full items-center gap-2 font-mono text-sm">
296                    <Glyph
297                        icon=if open { icondata::FaChevronDownSolid } else { icondata::FaChevronRightSolid }
298                        class="w-3 text-xs opacity-60"
299                    />
300                    <Glyph
301                        icon=if open { icondata::FaFolderOpenSolid } else { icondata::FaFolderSolid }
302                        class="text-caramel dark:text-maple"
303                    />
304                    <span class="min-w-0 flex-1 truncate font-bold">{name.clone()}</span>
305                    <span class=STATE>{memories.to_string()}</span>
306                </span>
307            </button>
308            <button
309                type="button"
310                class=ICON_BUTTON
311                aria-label="New memory in this project"
312                title="New memory in this project"
313                data-project-new
314                on:click=move |_| new_url.set(prefix.clone())
315            >
316                <Glyph icon=icondata::FaPlusSolid />
317            </button>
318            <button
319                type="button"
320                class=ICON_BUTTON
321                aria-label="Delete the project"
322                title=move || if workspace.has_projects.get() { "Delete the project with its sessions and memories" } else { "Delete the project's memories" }
323                data-project-delete
324                on:click=move |_| confirm_project.set(Some(asked.clone()))
325            >
326                <Glyph icon=icondata::FaTrashCanRegular />
327            </button>
328        </div>
329        <Show when=confirming>
330            <div class=format!("{CONFIRM} ml-6") role="alertdialog" data-project-confirm>
331                <p class="text-sm">{question.clone()}</p>
332                {move || {
333                    workspace
334                        .project_error
335                        .get()
336                        .map(|(_, message)| view! { <p class=ERROR data-project-error>{message}</p> })
337                }}
338                <div class="flex gap-2">
339                    <button
340                        type="button"
341                        id="project-delete-confirm"
342                        class=PRIMARY
343                        on:click={
344                            let deleted = deleted.clone();
345                            let project = project.clone();
346                            move |_| {
347                                match project() {
348                                    Some(project) => workspace.remove_project(project.id),
349                                    None => {
350                                        confirm_project.set(None);
351                                        workspace.delete_project(deleted.clone());
352                                    }
353                                }
354                            }
355                        }
356                    >
357                        "Delete"
358                    </button>
359                    <button type="button" class=BUTTON on:click=move |_| confirm_project.set(None)>
360                        "Keep it"
361                    </button>
362                </div>
363            </div>
364        </Show>
365    }
366}

368/// The memory open in the editor: an `agent://` file as its parts, the
369/// daemon's fixed, else the whole file as text.
370fn memory_editor(workspace: Workspace, edit: MemoryEdit) -> impl IntoView {
371    let info = workspace.memory.with_untracked(|listing| {
372        listing
373            .as_ref()
374            .and_then(|listing| listing.as_ref().ok())
375            .and_then(|listing| {
376                listing
377                    .memories
378                    .iter()
379                    .find(|memory| memory.url == edit.url)
380                    .cloned()
381            })
382    });
383    let parts = info
384        .as_ref()
385        .and_then(|info| agent_memory_parts(info, &edit.text));
386    let original = info
387        .as_ref()
388        .and_then(|info: &MemoryInfo| agent_memory_parts(info, &edit.original));
389    let url = edit.url.clone();
390    let body = match &parts {
391        Some(parts) => view! {
392            <div class="flex flex-col gap-1" data-agent-memory>
393                <span class="text-xs font-bold text-ink dark:text-sand">"Description"</span>
394                <p
395                    id="memory-description"
396                    class="flex items-start gap-2 rounded-md border border-dashed border-crust px-3 py-2 text-sm dark:border-smoke"
397                    title="The daemon keeps it: the agent's own description"
398                >
399                    <Glyph icon=icondata::FaLockSolid class="mt-1 text-xs opacity-60" />
400                    <span>{parts.description.clone()}</span>
401                </p>
402                <span class="mt-1 text-xs font-bold text-ink dark:text-sand">"Keywords"</span>
403                <div class="flex flex-wrap items-center gap-1.5">
404                    {parts.fixed.iter().map(|keyword| view! {
405                        <span class=format!("{STATE} inline-flex items-center gap-1 font-mono") title="The daemon keeps it" data-fixed-keyword>
406                            <Glyph icon=icondata::FaLockSolid class="text-[0.6rem] opacity-60" />
407                            {keyword.clone()}
408                        </span>
409                    }).collect_view()}
410                </div>
411                <input
412                    type="text"
413                    id="memory-keywords"
414                    class=FIELD
415                    placeholder="more keywords, separated by commas"
416                    value=parts.extra.join(", ")
417                    aria-label="More keywords"
418                />
419                <p class=NOTE>"The daemon owns the description and the first keywords (the agent and its mode); the rest are yours."</p>
420                <span class="mt-1 text-xs font-bold text-ink dark:text-sand">"Memory"</span>
421            </div>
422            <textarea id="memory-text" class=EDITOR spellcheck="false">{parts.body.clone()}</textarea>
423        }
424        .into_any(),
425        None => view! {
426            <textarea id="memory-text" class=EDITOR spellcheck="false">{edit.text.clone()}</textarea>
427        }
428        .into_any(),
429    };
430    let save = StoredValue::new(parts.clone());
431    view! {
432        <div class=EDITOR_PAGE>
433            <p class="font-mono text-sm font-bold" id="memory-url">{url}</p>
434            {body}
435            <div class=EDITOR_ACTIONS>
436                <button
437                    type="button"
438                    id="memory-save"
439                    class=PRIMARY
440                    on:click=move |_| {
441                        let text = match save.get_value() {
442                            Some(parts) => {
443                                let typed = AgentMemoryParts {
444                                    extra: AgentMemoryParts::typed_keywords(&field_value("memory-keywords")),
445                                    body: area_value("memory-text"),
446                                    ..parts
447                                };
448                                // Unchanged parts are the text as read, so
449                                // nothing is written.
450                                if original.as_ref() == Some(&typed) {
451                                    workspace.memory_edit.with_untracked(|edit| {
452                                        edit.as_ref().map(|edit| edit.original.clone()).unwrap_or_default()
453                                    })
454                                } else {
455                                    typed.text()
456                                }
457                            }
458                            None => area_value("memory-text"),
459                        };
460                        workspace.save_memory(text);
461                    }
462                >
463                    "Save"
464                </button>
465                <button type="button" class=BUTTON on:click=move |_| workspace.memory_edit.set(None)>
466                    "Cancel"
467                </button>
468            </div>
469        </div>
470    }
471}

473/// A write that was not saved: what the view offers to do with it.
474fn pending_choices(workspace: Workspace, pending: MemoryPending) -> impl IntoView {
475    let choice = |id: &'static str, label: &'static str, choice: MemoryChoice| {
476        view! {
477            <button type="button" id=id class=BUTTON on:click=move |_| workspace.resolve_memory(choice)>
478                {label}
479            </button>
480        }
481    };
482    match pending {
483        MemoryPending::Invalid { edit, message } => view! {
484            <p class=ERROR id="memory-pending">{format!("{} is not a valid memory: {message}", edit.url)}</p>
485            <div class="flex gap-2">
486                {choice("memory-edit-again", "Edit again", MemoryChoice::EditAgain)}
487                {choice("memory-drop", "Drop the edit", MemoryChoice::Drop)}
488            </div>
489        }
490        .into_any(),
491        MemoryPending::Changed { edit } => view! {
492            <p class="text-sm" id="memory-pending">{format!("{} changed since it was opened. Reload it and lose this edit, or overwrite it with this edit?", edit.url)}</p>
493            <div class="flex flex-wrap gap-2">
494                {choice("memory-reload", "Reload current", MemoryChoice::Reload)}
495                {choice("memory-overwrite", "Overwrite", MemoryChoice::Overwrite)}
496                {choice("memory-drop", "Drop the edit", MemoryChoice::Drop)}
497            </div>
498        }
499        .into_any(),
500    }
