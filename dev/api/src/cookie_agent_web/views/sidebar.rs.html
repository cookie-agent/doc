2//! New session, the views under *Workspace*, the sessions by project, and
3//! Settings, the connection, and the theme at its foot.

5//! A desktop pins it beside every view, or hides it (the sidebar button,

35/// The views the sidebar lists under *Workspace*: every view but Chat,

40    View::Memory,

43    View::Machines,

49/// The sidebar entry of `view`.
50fn nav_entry(workspace: Workspace, view: View) -> impl IntoView {
51    let current = move || workspace.view.get() == view;
52    // Scheduled prompts need a daemon that keeps cron jobs.
53    let offered = move || {
54        (view != View::Cron || workspace.has_crons.get())
55            && (view != View::Webhooks || workspace.has_webhooks.get())
56            && (view != View::Sharing || workspace.has_shares.get())
57    };
58    let label = match view {
59        // Short enough for the sidebar's two columns.
60        View::Mcp => "MCP",
61        _ => view.label(),
62    };
63    view! {
64        <button
65            type="button"
66            class=move || if workspace.narrow.get() { NAV_TILE } else { NAV_ROW }
67            data-view=view.id()
68            aria-current=move || current().then_some("page")
69            class:hidden=move || !offered()
70            on:click=move |_| workspace.open_from_sidebar(view)
71        >
72            <Glyph
73                icon=view_icon(view)
74                class="text-base text-muted transition-colors group-hover:text-fg-2 group-aria-[current=page]:text-accent"
75            />
76            <span class="max-w-full truncate px-1">{label}</span>
77        </button>
78    }
79}

81/// The sidebar: what [`WorkspaceView`](super::WorkspaceView) shows beside
82/// every view, or as the page below desktop widths.
83#[component]
84pub fn Sidebar(
85    workspace: Workspace,
86    connection: Connection,
87    theme: RwSignal<ThemePreference>,
88) -> impl IntoView {
89    // Below desktop widths the sidebar is the page, over the view.
90    let paged = move || workspace.sidebar_open.get();
91    let nav = WORKSPACE_VIEWS
92        .into_iter()
93        .map(|view| nav_entry(workspace, view))
94        .collect_view();
95    view! {
96        <aside
97            id="sidebar"
98            class="z-10 w-64 shrink-0 flex-col border-r border-line bg-surface-2"
99            class:hidden=move || !paged()
100            class:flex=paged
101            class:absolute=paged
102            class:inset-0=paged
103            class:w-full=paged
104            // Settings takes the sidebar's place on a desktop
105            // (docs/site/specs/web-polish.md, P6).
106            class:lg:flex=move || workspace.sidebar_pinned.get() && workspace.view.get() != View::Settings
107            aria-label="Sidebar"
108        >
109            <div class="flex h-12 shrink-0 items-center gap-2 px-3">
110                <img src="icons/icon.svg" alt="" class="h-6 w-6 shrink-0" />
111                <span class="min-w-0 flex-1 truncate text-sm font-semibold tracking-tight">"Cookie Agent"</span>
112                <Show when=paged>
113                    <super::shell::SidebarButton workspace=workspace />
114                </Show>
115            </div>
116            <nav aria-label="Views" class="flex shrink-0 flex-col gap-px px-2" data-sidebar-views>
117                <button
118                    type="button"
119                    class=NAV_ROW
120                    data-new-session
121                    aria-current=move || {
122                        (workspace.view.get() == View::Chat && workspace.new_session.get())
123                            .then_some("page")
124                    }
125                    on:click=move |_| workspace.start_new_session()
126                >
127                    <Glyph icon=icondata::LuSquarePen class="text-base text-muted group-hover:text-fg-2 group-aria-[current=page]:text-accent" />
128                    <span class="truncate">"New session"</span>
129                </button>
130                // Two columns on a desktop, so the sessions keep the room
131                // (docs/site/specs/web-polish.md, P1); a phone's tiles.
132                <div class=move || if workspace.narrow.get() { "mt-2 grid grid-cols-4 gap-1" } else { "mt-1.5 grid grid-cols-2 gap-x-1 gap-y-px" }>{nav}</div>
133            </nav>
134            <SessionList workspace=workspace />
135            <div class="flex shrink-0 items-center gap-1 border-t border-line px-2 py-1.5">
136                <button
137                    type="button"
138                    class=format!("{NAV_ROW} min-w-0 flex-1")
139                    data-view=View::Settings.id()
140                    aria-current=move || (workspace.view.get() == View::Settings).then_some("page")
141                    on:click=move |_| workspace.open_from_sidebar(View::Settings)
142                >
143                    <Glyph icon=view_icon(View::Settings) class="text-base text-muted group-hover:text-fg-2 group-aria-[current=page]:text-accent" />
144                    <span class="truncate">"Settings"</span>
145                </button>
146                <ConnectionStatus connection=connection compact=true primary=true />
147                <ThemeToggle theme=theme class=FOOT_BUTTON />
148            </div>
149        </aside>
150    }
151}
