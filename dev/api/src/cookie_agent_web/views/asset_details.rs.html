1//! The media viewer's Details (`docs/site/specs/asset-opening.md` section
2//! 4): beside the file on a wide window and a sheet from the bottom on a
3//! narrow one, an asset's title, type and size, handle, where it lives, and
4//! who saved it, with *Rename*, *Open the session*, and *Delete…*.

11const ACTION: &str = "inline-flex min-h-9 items-center gap-1.5 rounded-lg border border-white/15 px-2.5 text-xs hover:bg-white/10";
12const LINK: &str = "text-left underline decoration-dotted underline-offset-2 hover:text-white";

14/// The Details of the asset `id` of `store`, read when they open and again
15/// after a rename.
16#[component]
17pub(super) fn AssetDetails(workspace: Workspace, store: AssetStore, id: AssetId) -> impl IntoView {
18    let found = RwSignal::new(None::<Result<Option<Asset>, String>>);
19    Effect::new(move |_| {
20        // Read again once a rename closes its dialog.
21        if workspace.asset_renaming.with(Option::is_some) {
22            return;
23        }
24        let store = store.clone();
25        spawn_local(async move {
26            let read = workspace.get_asset(store, id).await;
27            found.try_set(Some(read));
28        });
29    });
30    let body = move || match found.get() {
31        None => view! { <p class="text-white/60">"Loading…"</p> }.into_any(),
32        Some(Err(error)) => view! { <p class="text-white/60">{error}</p> }.into_any(),
33        Some(Ok(None)) => {
34            view! { <p class="text-white/60">"This asset was deleted."</p> }.into_any()
35        }
36        Some(Ok(Some(asset))) => details(workspace, asset).into_any(),
37    };
38    view! {
39        <aside
40            id="media-viewer-info"
41            aria-label="Details"
42            class="absolute inset-x-0 bottom-0 z-20 flex max-h-[45%] flex-col gap-3 overflow-y-auto rounded-t-2xl border-t border-white/10 bg-[#141416] p-4 text-[0.8125rem] lg:static lg:max-h-none lg:w-72 lg:shrink-0 lg:rounded-none lg:border-t-0 lg:border-l lg:bg-white/[0.03]"
43            on:click=|event| event.stop_propagation()
44        >
45            {body}
46        </aside>
47    }
48}

50fn details(workspace: Workspace, asset: Asset) -> impl IntoView {
51    let place = super::assets::place_of(workspace, &asset);
52    let (saved, saved_session) = super::assets::saved_by(workspace, &asset);
53    // Its row is "Saved".
54    let saved = saved.trim_start_matches("saved ").to_owned();
55    let lives_in = match place.session {
56        Some((session, title)) => {
57            let title = title.unwrap_or_else(|| "its session".to_owned());
58            view! {
59                {format!("{} · ", place.project)}
60                <button type="button" class=LINK data-details-session on:click=move |_| open_session(workspace, session)>
61                    {title}
62                </button>
63            }
64            .into_any()
65        }
66        None if matches!(asset.store, AssetStore::Shared) => place.project.into_any(),
67        None => format!("{} · project", place.project).into_any(),
68    };
69    let saved = match saved_session {
70        Some(session) => view! {
71            <button type="button" class=LINK on:click=move |_| open_session(workspace, session)>{saved}</button>
72        }
73        .into_any(),
74        None => saved.into_any(),
75    };
76    let session = saved_session.or(match asset.store {
77        AssetStore::Session { session } => Some(session),
78        _ => None,
79    });
80    let (rename, delete) = (asset.clone(), asset.clone());
81    view! {
82        <h3 class="flex min-w-0 items-center gap-2 text-sm font-bold" id="media-viewer-info-title">
83            <Glyph icon=crate::icons::asset_icon(cookie_agent_client_state::assets::AssetKind::of(&asset.media_type)) />
84            <span class="min-w-0 wrap-anywhere">{asset.title.clone()}</span>
85        </h3>
86        <dl class="grid grid-cols-[5rem_1fr] gap-x-3 gap-y-1.5">
87            <dt class="text-white/55">"Type"</dt>
88            <dd class="min-w-0 wrap-anywhere">{format!("{} · {}", asset.media_type, human_bytes(asset.byte_length))}</dd>
89            <dt class="text-white/55">"Handle"</dt>
90            <dd class="font-mono text-xs select-all">{asset.handle.to_string()}</dd>
91            <dt class="text-white/55">"Lives in"</dt>
92            <dd class="min-w-0">{lives_in}</dd>
93            <dt class="text-white/55">"Saved"</dt>
94            <dd class="min-w-0">{saved}</dd>
95        </dl>
96        <div class="flex flex-wrap gap-1.5">
97            <button
98                type="button"
99                class=ACTION
100                data-details-action="rename"
101                on:click=move |_| workspace.asset_renaming.set(Some((rename.clone(), rename.title.clone())))
102            >
103                <Glyph icon=icondata::LuPencil />
104                "Rename"
105            </button>
106            {session.map(|session| view! {
107                <button type="button" class=ACTION data-details-action="session" on:click=move |_| open_session(workspace, session)>
108                    <Glyph icon=icondata::LuMessagesSquare />
109                    "Open the session"
110                </button>
111            })}
112            <button
113                type="button"
114                class=format!("{ACTION} border-bad/40 text-bad")
115                data-details-action="delete"
116                on:click=move |_| workspace.asset_deleting.set(Some(delete.clone()))
117            >
118                <Glyph icon=icondata::LuTrash2 />
119                "Delete…"
120            </button>
121        </div>
122    }
123}

125/// Closes the viewer and the asset page, and opens `session` in the chat.
126fn open_session(workspace: Workspace, session: cookie_agent_protocol::SessionId) {
127    workspace.media_viewer.set(None);
128    workspace.asset_page.set(None);
129    workspace.open_session(session);
130    workspace.open_view(View::Chat, None);
131}
