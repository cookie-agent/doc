23/// How long a finger holds a session row before it lifts to be dragged,
24/// or a header before it shows its buttons.

27/// A drop target: a project section's header, named by its

44/// The project a section key names: `None` for "No project".
45fn project_of_key(key: &str) -> Option<Option<ProjectId>> {
46    if key == "none" {
47        return Some(None);
48    }
49    key.strip_prefix("project:")
50        .and_then(|id| ProjectId::new(id).ok())
51        .map(Some)
52}

54/// A session row's drag, shared by the rows, the headers, and the window's
55/// listeners.
56#[derive(Clone, Copy)]
57pub(super) struct SessionDrag {
58    gesture: StoredValue<Drag>,
59    /// The row pressed, until the press ends.
60    pressed: StoredValue<Option<SessionId>>,
61    /// The row lifted or dragged, which follows the pointer.
62    pub(super) dragged: RwSignal<Option<SessionId>>,
63    pub(super) offset: RwSignal<(f64, f64)>,
64    /// The section whose header is under the pointer.
65    over: RwSignal<Option<String>>,
66    timer: StoredValue<Option<TimeoutHandle>>,
67    /// A drag or a long press just ended: the click that follows does not
68    /// open the row.
69    swallow_click: StoredValue<bool>,
70    /// The list, which scrolls when a drag nears its edge.
71    pub(super) list: NodeRef<html::Nav>,
72}
73
74impl SessionDrag {
75    pub(super) fn new() -> Self {
76        Self {
77            gesture: StoredValue::new(Drag::default()),
78            pressed: StoredValue::new(None),
79            dragged: RwSignal::new(None),
80            offset: RwSignal::new((0.0, 0.0)),
81            over: RwSignal::new(None),
82            timer: StoredValue::new(None),
83            swallow_click: StoredValue::new(false),
84            list: NodeRef::new(),
85        }
86    }
87
88    fn stop_timer(self) {
89        if let Some(timer) = self.timer.get_value() {
90            timer.clear();
91        }
92        self.timer.set_value(None);
93    }
94
95    fn reset(self) {
96        browser::set_pressing(false);
97        self.stop_timer();
98        self.gesture.update_value(Drag::cancel);
99        self.pressed.set_value(None);
100        self.dragged.set(None);
101        self.offset.set((0.0, 0.0));
102        self.over.set(None);
103    }
104
105    /// Whether the click that just came must be ignored; it is ignored
106    /// once.
107    pub(super) fn take_click(self) -> bool {
108        let swallow = self.swallow_click.get_value();
109        self.swallow_click.set_value(false);
110        swallow
111    }
112
113    /// A press on `session_id`'s row: a mouse drags once it moves (only
114    /// where there are headers to drop on), a finger after a long press,
115    /// which let go in place reveals the row's buttons.
116    pub(super) fn press(
117        self,
118        workspace: Workspace,
119        session_id: SessionId,
120        event: &ev::PointerEvent,
121    ) {
122        let touch = event.pointer_type() == "touch";
123        if !touch && (event.button() != 0 || !workspace.has_projects.get_untracked()) {
124            return;
125        }
126        let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
127        self.stop_timer();
128        self.swallow_click.set_value(false);
129        self.gesture
130            .update_value(|gesture| gesture.press(x, y, touch));
131        self.pressed.set_value(Some(session_id));
132        if touch {
133            browser::set_pressing(true);
134            let timer = set_timeout_with_handle(
135                move || {
136                    if self.gesture.try_update_value(Drag::elapse).unwrap_or(false) {
137                        self.dragged.set(Some(session_id));
138                    }
139                },
140                LONG_PRESS,
141            )
142            .ok();
143            self.timer.set_value(timer);
144        }
145    }
146
147    /// Follows the pointer and lets go of it, on the window, for as long
148    /// as the list shows. A long press let go in place reveals the row's
149    /// buttons in `actions`; a drop on a header moves the session.
150    pub(super) fn follow(self, workspace: Workspace, actions: RowActions) {
151        let moved = window_event_listener(ev::pointermove, move |event| {
152            let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
153            match self
154                .gesture
155                .try_update_value(|gesture| gesture.moved(x, y))
156                .unwrap_or(DragStep::Nothing)
157            {
158                DragStep::Nothing => {}
159                DragStep::Cancelled => {
160                    browser::set_pressing(false);
161                    self.stop_timer();
162                    self.pressed.set_value(None);
163                }
164                DragStep::Started => {
165                    self.stop_timer();
166                    browser::set_pressing(true);
167                    actions.revealed.set(None);
168                    self.dragged.set(self.pressed.get_value());
169                    self.over
170                        .set(browser::attribute_at(x, y, DROP, "data-project-drop"));
171                }
172                DragStep::Moved { dx, dy } => {
173                    self.offset.set((dx, dy));
174                    self.over
175                        .set(browser::attribute_at(x, y, DROP, "data-project-drop"));
176                    if let Some(list) = self.list.get_untracked() {
177                        browser::scroll_element_near_edge(&list, y);
178                    }
179                }
180            }
181        });
182        let released = window_event_listener(ev::pointerup, move |_| {
183            let end = self
184                .gesture
185                .try_update_value(Drag::release)
186                .unwrap_or(DragEnd::Nothing);
187            let pressed = self.pressed.get_value();
188            let over = self.over.get_untracked();
189            self.reset();
190            let Some(session_id) = pressed else {
191                return;
192            };
193            // The click a release may bring comes before any timer: past it,
194            // a click (the keyboard's, say) opens the row again.
195            if end != DragEnd::Nothing {
196                self.swallow_click.set_value(true);
197                set_timeout(
198                    move || {
199                        let _ = self.swallow_click.try_set_value(false);
200                    },
201                    std::time::Duration::ZERO,
202                );
203            }
204            match end {
205                DragEnd::Nothing => {}
206                DragEnd::Held => {
207                    actions.confirming.set(None);
208                    actions.revealed.set(Some(session_id));
209                }
210                DragEnd::Dropped => {
211                    if let Some(project) = over.as_deref().and_then(project_of_key) {
212                        workspace.move_session(session_id, project);
213                    }
214                }
215            }
216        });
217        let cancelled = window_event_listener(ev::pointercancel, move |_| self.reset());
218        let unselectable = StoredValue::new_local(browser::hold_selection(move || {
219            self.pressed
220                .try_with_value(Option::is_some)
221                .unwrap_or(false)
222        }));
223        let hold = StoredValue::new_local(browser::hold_touch_scroll(move || {
224            self.gesture
225                .try_with_value(Drag::holds_scroll)
226                .unwrap_or(false)
227        }));
228        on_cleanup(move || {
229            moved.remove();
230            released.remove();
231            cancelled.remove();
232            if let Some(hold) = hold.try_update_value(Option::take).flatten() {
233                hold.remove();
234            }
235            if let Some(hold) = unselectable.try_update_value(Option::take).flatten() {
236                hold.remove();
237            }
238            browser::set_pressing(false);
239        });
240    }
241
242    /// Whether `session_id`'s row is lifted or dragged.
243    pub(super) fn lifts(self, session_id: SessionId) -> bool {
244        self.dragged.get() == Some(session_id)
245    }
246}

279/// The session list as project sections, then "No project".
280pub(super) fn project_list(
281    workspace: Workspace,
282    now: RwSignal<jiff::Timestamp>,
283    actions: RowActions,
284    drag: SessionDrag,
285    headers: HeaderActions,
286) -> impl IntoView {
287    let sections = Memo::new(move |_| {
288        workspace.projects.with(|projects| {
289            workspace
290                .sessions
291                .with(|sessions| project_sections(projects, sessions))
292        })
293    });
294    move || {
295        sections
296            .get()
297            .into_iter()
298            .map(|section| section_view(workspace, section, now, actions, drag, headers))
299            .collect_view()
300    }
301}

303/// One section: its header, then its sessions while it is open.
304fn section_view(
305    workspace: Workspace,
306    section: ProjectSection,
307    now: RwSignal<jiff::Timestamp>,
308    actions: RowActions,
309    drag: SessionDrag,
310    headers: HeaderActions,
311) -> impl IntoView {
312    let key = section.key();
313    let open = {
314        let key = key.clone();
315        Memo::new(move |_| workspace.section_open(&key))
316    };
317    let sessions = section.sessions.clone();
318    let empty = sessions.is_empty();
319    view! {
320        <section class="flex flex-col gap-1" data-project-section=key.clone() data-open=move || open.get().to_string()>
321            {project_header(workspace, section.project.clone(), key.clone(), open, drag, headers, sessions.iter().map(|session| session.session_id).collect())}
322            <Show when=move || open.get()>
323                {
324                    let sessions = sessions.clone();
325                    move || {
326                        if empty {
327                            return view! { <p class=format!("px-3 pb-1 {NOTE}") data-section-empty>"No sessions yet."</p> }.into_any();
328                        }
329                        sessions
330                            .clone()
331                            .into_iter()
332                            .map(|session| session_row(workspace, session, now, actions, drag))
333                            .collect_view()
334                            .into_any()
335                    }
336                }
337            </Show>
338        </section>
339    }
340}

342/// A section's header: a click folds or opens it. It shows the project's
343/// emoji (a folder without one) and its name, then its default
344/// attachments as badges. `+` starts a session in it; a project's pencil
345/// and delete show on hover, or after a long press on a touch screen.
346/// Sessions dropped on it move into it.
347fn project_header(
348    workspace: Workspace,
349    project: Option<ProjectInfo>,
350    key: String,
351    open: Memo<bool>,
352    drag: SessionDrag,
353    headers: HeaderActions,
354    members: Vec<SessionId>,
355) -> impl IntoView {
356    let project_id = project.as_ref().map(|project| project.id.clone());
357    let label = project.as_ref().map_or_else(
358        || "No project".to_owned(),
359        |project| project.name.to_string(),
360    );
361    let emoji = project.as_ref().and_then(|project| project.emoji.clone());
362    let editable = project.is_some();
363    let defaults = project
364        .as_ref()
365        .and_then(|project| project.defaults.clone());
366    let badge_line = NodeRef::<html::Span>::new();
367    crate::badges::keep_fitted(badge_line);
368    let badges = {
369        let defaults = defaults.clone();
370        move || {
371            let homes = workspace.homes();
372            let shown = workspace
373                .runtimes
374                .with(|listing| project_badges(defaults.as_ref(), listing.as_ref(), &homes));
375            crate::badges::fit_soon(badge_line);
376            crate::badges::badge_line(shown)
377        }
378    };
379    let has_badges = {
380        let defaults = defaults.clone();
381        move || {
382            let homes = workspace.homes();
383            workspace.runtimes.with(|listing| {
384                !project_badges(defaults.as_ref(), listing.as_ref(), &homes).is_empty()
385            })
386        }
387    };
388    let tooltip = {
389        let defaults = defaults.clone();
390        let label = label.clone();
391        move || {
392            let homes = workspace.homes();
393            let places = defaults.as_ref().map(|defaults| {
394                workspace
395                    .runtimes
396                    .with(|listing| project_badges_tooltip(defaults, listing.as_ref(), &homes))
397            });
398            match places {
399                Some(places)
400                    if defaults
401                        .as_ref()
402                        .is_some_and(|defaults| !defaults.runtimes.is_empty()) =>
403                {
404                    format!("{label}\nNew sessions attach:\n{places}")
405                }
406                _ => label.clone(),
407            }
408        }
409    };
410    let HeaderActions {
411        hovered,
412        revealed,
413        confirming,
414        press,
415        ..
416    } = headers;
417    let shown = {
418        let key = key.clone();
419        move || {
420            editable
421                && (hovered.get().as_deref() == Some(key.as_str())
422                    || revealed.get().as_deref() == Some(key.as_str()))
423        }
424    };
425    // A dragged session from another section can land here.
426    let target = {
427        let key = key.clone();
428        move || {
429            drag.dragged
430                .get()
431                .is_some_and(|dragged| !members.contains(&dragged))
432                && drag.over.get().as_deref() == Some(key.as_str())
433        }
434    };
435    let on_pointerdown = {
436        let key = key.clone();
437        move |event: ev::PointerEvent| {
438            if event.pointer_type() != "touch" || !editable {
439                return;
440            }
441            let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
442            press.update_value(|press| press.start(x, y));
443            headers.stop_timer();
444            let key = key.clone();
445            let timer = set_timeout_with_handle(
446                move || {
447                    if press.try_update_value(Press::elapse).unwrap_or(false) {
448                        revealed.set(Some(key.clone()));
449                    }
450                },
451                LONG_PRESS,
452            )
453            .ok();
454            headers.timer.set_value(timer);
455        }
456    };
457    let on_pointermove = move |event: ev::PointerEvent| {
458        let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
459        if press
460            .try_update_value(|press| press.moved(x, y))
461            .unwrap_or(false)
462        {
463            headers.stop_timer();
464        }
465    };
466    let release = move |_| {
467        press.update_value(Press::cancel);
468        headers.stop_timer();
469    };
470    let toggle_key = key.clone();
471    let menu_key = key.clone();
472    let enter_key = key.clone();
473    let leave_key = key.clone();
474    let error_key = key.clone();
475    let new_in = project_id.clone();
476    let edit_id = project_id.clone();
477    let delete_id = project_id.clone();
478    let confirm_id = project_id.clone();
479    let confirm_project = project.clone();
480    let shown_buttons = shown.clone();
481    let ring = target.clone();
482    view! {
483        <div
484            class=move || format!(
485                "project-header group relative rounded-md {}",
486                if ring() { "bg-toasted ring-2 ring-caramel dark:bg-ember dark:ring-maple" } else { "" },
487            )
488            data-project-drop=key.clone()
489            data-project-header=project_id.as_ref().map_or_else(|| "none".to_owned(), ToString::to_string)
490            data-drop-target=move || target().then_some("true")
491            on:pointerenter=move |event: ev::PointerEvent| {
492                if event.pointer_type() != "touch" {
493                    hovered.set(Some(enter_key.clone()));
494                }
495            }
496            on:pointerleave=move |_| {
497                if hovered.get_untracked().as_deref() == Some(leave_key.as_str()) {
498                    hovered.set(None);
499                }
500            }
501        >
502            <button
503                type="button"
504                class="no-callout flex w-full flex-col items-stretch rounded-md text-left select-none hover:bg-toasted active:bg-crust group-hover:bg-toasted dark:hover:bg-ember dark:active:bg-smoke dark:group-hover:bg-ember"
505                aria-expanded=move || open.get().to_string()
506                title=tooltip
507                data-section-toggle
508                on:pointerdown=on_pointerdown
509                on:pointermove=on_pointermove
510                on:pointerup=release
511                on:pointercancel=release
512                on:contextmenu=move |event: ev::MouseEvent| {
513                    // A long press shows the header's buttons, not the
514                    // browser's menu.
515                    if revealed.get_untracked().as_deref() == Some(menu_key.as_str()) {
516                        event.prevent_default();
517                    }
518                }
519                on:click=move |_| {
520                    if press.try_update_value(Press::click).unwrap_or(true) {
521                        workspace.toggle_section(&toggle_key);
522                    }
523                }
524            >
525                // The name's line keeps clear of `+`; the pencil and trash
526                // lie over it when they show, without moving it. The
527                // badges' line below has the width to itself.
528                <span class="flex min-h-11 w-full items-center gap-2 pr-12 pl-2">
529                    <span class="inline-flex shrink-0 opacity-70 transition-transform" class:rotate-90=move || open.get()>
530                        <Glyph icon=icondata::FaChevronRightSolid class="text-[0.625rem]" />
531                    </span>
532                    {match emoji.clone() {
533                        Some(emoji) => view! { <span class="w-5 shrink-0 text-center text-base leading-none" data-project-emoji>{emoji}</span> }.into_any(),
534                        None if editable => view! { <Glyph icon=icondata::FaFolderSolid class="w-5 justify-center text-caramel dark:text-maple" /> }.into_any(),
535                        None => view! { <Glyph icon=icondata::FaFolderOpenRegular class="w-5 justify-center opacity-70" /> }.into_any(),
536                    }}
537                    <span class=move || format!("min-w-0 flex-1 truncate text-sm font-bold {}", if editable { "" } else { "text-ink dark:text-sand" }) data-project-name>
538                        {label.clone()}
539                    </span>
540                </span>
541                <Show when=has_badges.clone()>
542                    <span class="flex min-w-0 items-center gap-1 overflow-hidden pb-1.5 pr-2 pl-9 text-xs" node_ref=badge_line data-project-badges>
543                        {badges.clone()}
544                    </span>
545                </Show>
546            </button>
547            // Level with the name's line: the pencil and delete (on hover,
548            // or after a long press), then `+`, always there.
549            <div
550                class=move || format!(
551                    "absolute top-0 right-0 flex h-11 items-center gap-0.5 rounded-tr-md pr-0.5 {}",
552                    if shown_buttons() { HEADER_COVER } else { "" },
553                )
554                data-header-actions
555            >
556                <Show when=shown.clone()>
557                    <button
558                        type="button"
559                        class=HEADER_ACTION
560                        aria-label="Edit the project"
561                        title="Edit the project"
562                        data-header-edit
563                        on:click={
564                            let edit_id = edit_id.clone();
565                            move |_| {
566                                revealed.set(None);
567                                workspace.open_project_editor(edit_id.clone(), false);
568                            }
569                        }
570                    >
571                        <Glyph icon=icondata::FaPenSolid />
572                    </button>
573                    <button
574                        type="button"
575                        class=HEADER_ACTION
576                        aria-label="Delete the project"
577                        title="Delete the project with its sessions and memories"
578                        data-header-delete
579                        on:click={
580                            let delete_id = delete_id.clone();
581                            move |_| {
582                                revealed.set(None);
583                                confirming.set(delete_id.clone());
584                            }
585                        }
586                    >
587                        <Glyph icon=icondata::FaTrashCanRegular />
588                    </button>
589                </Show>
590                <button
591                    type="button"
592                    class=HEADER_ACTION
593                    aria-label=if editable { "New session in this project" } else { "New session in no project" }
594                    title=if editable { "New session in this project" } else { "New session in no project" }
595                    data-header-new
596                    on:click=move |_| workspace.new_session_in(new_in.clone())
597                >
598                    <Glyph icon=icondata::FaPlusSolid />
599                </button>
600            </div>
601        </div>
602        <Show when=move || confirm_id.is_some() && confirming.get() == confirm_id>
603            {
604                let question = confirm_project.as_ref().map(delete_question).unwrap_or_default();
605                let project = confirm_project.as_ref().map(|project| project.id.clone());
606                view! {
607                    <div class="mx-2 mb-1 flex flex-col gap-2 rounded-lg border border-caramel bg-cream p-2 text-sm dark:border-maple dark:bg-night" role="alertdialog" data-project-confirm-box>
608                        <p>{question}</p>
609                        <div class="flex flex-wrap gap-2">
610                            <button
611                                type="button"
612                                class=PRIMARY

644/// "Move to project…": the projects, then "No project"; the session's own

669                class:dark:bg-ember=chosen
670                data-move-to=id.as_ref().map_or_else(|| "none".to_owned(), ToString::to_string)

672                    workspace.move_session(session_id, id.clone());
673                    workspace.discard_panel();
674                }
675            >
