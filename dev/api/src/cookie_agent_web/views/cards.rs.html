19/// A card's frame, with its selected look.
20fn card_class(selected: bool, offline: bool) -> String {
21    let state = if selected {
22        "border-caramel bg-toasted dark:border-maple dark:bg-ember"
23    } else if offline {
24        "border-toasted opacity-60 dark:border-ember"
25    } else {
26        "border-crust bg-cream hover:border-caramel hover:bg-parchment dark:border-smoke dark:bg-night dark:hover:border-maple dark:hover:bg-char"
27    };
28    let cursor = if offline && !selected {
29        "cursor-not-allowed"
30    } else {
31        "cursor-pointer"
32    };
33    format!("runtime-card flex min-w-0 flex-col gap-2 rounded-lg border-2 p-3 {state} {cursor}")
34}

36fn state_class(state: CardState) -> &'static str {
37    match state {
38        CardState::Ready => "text-green-700 dark:text-green-400",
39        CardState::StartsOnUse => "text-caramel dark:text-maple",
40        CardState::Offline => "text-ink dark:text-sand",
41    }
42}

44/// The new session's runtime cards, above the composer whose first message
45/// creates the session with them.
46#[component]
47pub fn NewSessionCards(workspace: Workspace) -> impl IntoView {
48    let groups = Memo::new(move |_| workspace.card_groups());
49    let cards = Memo::new(move |_| workspace.cards());
50    let blocker = move || {
51        workspace
52            .selection
53            .with(|selection| cards.with(|cards| selection.blocker(cards)))
54    };
55    let note = move || {
56        workspace
57            .selection
58            .with(|selection| cards.with(|cards| selection.prefix_note(cards)))
59    };
60    // Changing the cards answers why the last message did not create the
61    // session.
62    Effect::new(move |previous: Option<()>| {
63        workspace.selection.track();
64        if previous.is_some() {
65            workspace.draft_error.set(None);
66        }
67    });
68    view! {
69        <section
70            id="new-session"
71            aria-label="New session"
72            aria-busy=move || workspace.creating.get().to_string()
73            class="mx-auto flex w-full max-w-3xl flex-col gap-3 py-3"
74            class:opacity-60=move || workspace.creating.get()
75            class:pointer-events-none=move || workspace.creating.get()
76        >
77            // The title bar names the view: no heading here.
78            <super::projects::NewSessionProject workspace=workspace />
79            <p class=NOTE>
80                "Pick where the new session works: click a card to select it, and give a coding runtime its workspace. With only global, the session has no workspace: it hands work that needs files or commands to subagents on the runtimes it names. The first message creates the session, starting the runtimes that are not running."
81            </p>
82            {move || {
83                if workspace.runtimes.with(Option::is_none) {
84                    return super::loading("Reading the runtimes…").into_any();
85                }
86                groups
87                    .get()
88                    .into_iter()
89                    .map(|group| group_view(workspace, group, workspace.new_cards()))
90                    .collect_view()
91                    .into_any()
92            }}
93            <div class="flex flex-col gap-1">
94                {move || note().map(|note| view! {
95                    <p id="prefix-note" class="flex items-start gap-1.5 text-xs">
96                        <Glyph icon=icondata::FaCircleInfoSolid class="mt-0.5 text-caramel dark:text-maple" />
97                        <span>{note}</span>
98                    </p>
99                })}
100                {move || blocker().map(|blocker| view! {
101                    <p id="start-blocker" class=NOTE>{blocker}</p>
102                })}
103                <p class="flex items-start gap-2 text-sm" id="draft-places">
104                    <Glyph icon=icondata::FaFolderOpenSolid class="mt-0.5 text-caramel dark:text-maple" />
105                    <span class="min-w-0 break-words font-mono" data-draft-places>
106                        {move || workspace.places_label().unwrap_or_default()}
107                    </span>
108                </p>
109            </div>
110        </section>
111    }
112}

114/// Above the composer of a new session: that the first message is creating
115/// it, or why it did not.
116#[component]
117pub fn DraftNotice(workspace: Workspace) -> impl IntoView {
118    view! {
119        <Show when=move || workspace.new_session.get()>
120            {move || {
121                if workspace.creating.get() {
122                    Some(view! {
123                        <p id="draft-creating" role="status" class="mx-3 mt-2 flex items-center gap-2 text-sm text-caramel dark:text-maple">
124                            <Glyph icon=icondata::FaCircleNotchSolid class="animate-spin" />
125                            "Creating the session…"
126                        </p>
127                    }.into_any())
128                } else {
129                    workspace.draft_error.get().map(|error| view! {
130                        <p id="draft-error" role="alert" class=format!("{} mx-3 mt-2", super::ERROR)>{error}</p>
131                    }.into_any())
132                }
133            }}
134        </Show>
135    }
136}

138/// A machine's cards under a header that folds them, selected in `set`.
139pub(super) fn group_view(workspace: Workspace, group: Group, set: CardSet) -> impl IntoView {
140    let name = group.name();
141    let toggle_name = name.clone();
142    let header = group.clone();
143    let shown = {
144        let group = group.clone();
145        Memo::new(move |_| workspace.group_shown(&group, set))
146    };
147    let summary = {
148        let group = group.clone();
149        move || set.selection.with(|selection| group.summary(selection))
150    };
151    let cards = group.cards.clone();
152    view! {
153        // Two cards a row only where the cards' own container is wide (the
154        // new session's page), one in a narrow one (the project editor).
155        <div class="@container flex flex-col gap-2" data-group=name data-open=move || shown.get().to_string()>
156            <button
157                type="button"
158                class="flex min-h-11 w-full items-center gap-2 rounded-md px-2 text-left hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke"
159                aria-expanded=move || shown.get().to_string()
160                data-group-toggle=toggle_name
161                on:click=move |_| workspace.toggle_group(&header, set)
162            >
163                <span
164                    class="inline-flex transition-transform"
165                    class:rotate-90=move || shown.get()
166                >
167                    <Glyph icon=icondata::FaChevronRightSolid class="text-xs" />
168                </span>
169                <Glyph icon=if group.daemon { icondata::FaHouseSolid } else { icondata::FaComputerSolid } class="text-caramel dark:text-maple" />
170                <span class="min-w-0 flex-1 truncate text-sm" data-group-summary>{summary}</span>
171            </button>
172            <Show when=move || shown.get()>
173                <div class="grid grid-cols-1 items-start gap-2 @lg:grid-cols-2">
174                    {cards.clone().into_iter().map(|card| card_view(workspace, card, set)).collect_view()}
175                </div>
176            </Show>
177        </div>
178    }
179}

181/// One card: a click anywhere selects it, except in its fields and
182/// buttons.
183fn card_view(workspace: Workspace, card: Card, set: CardSet) -> impl IntoView {
184    let key = card.key();
185    let selected = {
186        let card = card.clone();
187        Memo::new(move |_| set.selection.with(|selection| selection.selected(&card)))
188    };
189    let offline = card.state == CardState::Offline;
190    let on_click = {
191        let card = card.clone();
192        move |event: ev::MouseEvent| {
193            let inside_control = event
194                .target()
195                .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
196                .and_then(|target| {
197                    target
198                        .closest("input, button, a, [data-card-fields]")
199                        .ok()
200                        .flatten()
201                })
202                .is_some();
203            if !inside_control {
204                workspace.toggle_card(&card, set);
205            }
206        }
207    };
208    let toggle = card.clone();
209    let fields = card.clone();
210    let label = card.label();
211    view! {
212        <div
213            class=move || card_class(selected.get(), offline)
214            data-card=key.clone()
215            data-state=card.state.label()
216            data-selected=move || selected.get().to_string()
217            on:click=on_click
218        >
219            <div class="flex items-start gap-2">
220                <button
221                    type="button"
222                    class="flex min-h-11 min-w-0 flex-1 items-start gap-2 rounded-md text-left disabled:cursor-not-allowed"
223                    aria-pressed=move || selected.get().to_string()
224                    aria-label=format!("{label}: select")
225                    disabled=move || offline && !selected.get()
226                    data-card-toggle
227                    on:click=move |_| workspace.toggle_card(&toggle, set)
228                >
229                    <Glyph
230                        icon=Signal::derive(move || if selected.get() { icondata::FaCircleCheckSolid } else { icondata::FaCircleRegular })
231                        class="mt-0.5 text-lg text-caramel dark:text-maple"
232                    />
233                    <span class="flex min-w-0 flex-1 flex-col">
234                        <span class="flex min-w-0 items-center gap-1.5">
235                            // Its machine's colour, as on the session rows.
236                            {crate::badges::runtime_dot(
237                                &card.machine.as_ref().map_or_else(|| card.runtime_id.to_string(), ToString::to_string),
238                                card.runtime_id.as_str(),
239                            )}
240                            <span class="truncate font-mono text-sm font-bold">{card.label()}</span>
241                        </span>
242                        {(!card.description.is_empty()).then(|| view! {
243                            <span class=format!("line-clamp-2 {NOTE}")>{card.description.clone()}</span>
244                        })}
245                    </span>
246                </button>
247                <span class=format!("mt-1 shrink-0 text-xs font-bold {}", state_class(card.state)) data-card-state>
248                    {card.state.label()}
249                </span>
250            </div>
251            {card.message.clone().filter(|_| offline).map(|message| view! {
252                <p class=format!("break-words {NOTE}")>{message}</p>
253            })}
254            {move || {
255                (selected.get() && fields.takes_workspace)
256                    .then(|| workspace_fields(workspace, fields.clone(), set))
257            }}
258            {
259                let key = key.clone();
260                move || {
261                    // A refusal of a card without a workspace field.
262                    (!card.takes_workspace)
263                        .then(|| set.errors.with(|errors| errors.get(&(key.clone(), 0)).cloned()))
264                        .flatten()
265                        .map(|error| view! { <p class=super::ERROR data-card-error>{error}</p> })
266                }
267            }
268        </div>
269    }
270}

272/// A selected card's workspace fields, their suggestions, and "+ another
273/// workspace".
274fn workspace_fields(workspace: Workspace, card: Card, set: CardSet) -> impl IntoView {
275    let list = format!("workspaces-{}", card.runtime_id);
276    let suggestions = workspace.workspace_suggestions(&card.runtime_id);
277    let rows = {
278        let card = card.clone();
279        let list = list.clone();
280        move || {
281            let workspaces = set.selection.with(|selection| selection.workspaces(&card));
282            let removable = workspaces.len() > 1;
283            workspaces
284                .into_iter()
285                .enumerate()
286                .map(|(index, text)| {
287                    let edited = card.clone();
288                    let removed = card.clone();
289                    let error = set
290                        .errors
291                        .with(|errors| errors.get(&(card.key(), index)).cloned());
292                    view! {
293                        <div class="flex flex-col gap-1">
294                            <div class="flex items-center gap-1">
295                                <input
296                                    type="text"
297                                    class=FIELD
298                                    list=list.clone()
299                                    placeholder="~/project"
300                                    autocomplete="off"
301                                    spellcheck="false"
302                                    aria-label=format!("Workspace {} of {}", index + 1, card.label())
303                                    data-workspace-field=index.to_string()
304                                    prop:value=text
305                                    on:input=move |event| {
306                                        let text = event_target_value(&event);
307                                        set.selection.update(|selection| {
308                                            selection.set_workspace(&edited, index, text);
309                                        });
310                                    }
311                                />
312                                {removable.then(|| view! {
313                                    <button
314                                        type="button"
315                                        class=ICON_BUTTON
316                                        aria-label="Remove this workspace"
317                                        title="Remove this workspace"
318                                        data-remove-workspace=index.to_string()
319                                        on:click=move |_| set.selection.update(|selection| {
320                                            selection.remove_workspace(&removed, index);
321                                        })
322                                    >
323                                        <Glyph icon=icondata::FaXmarkSolid />
324                                    </button>
325                                })}
326                            </div>
327                            {error.map(|error| view! { <p class=super::ERROR data-card-error>{error}</p> })}
328                        </div>
329                    }
330                })
331                .collect_view()
332        }
333    };
334    let added = card.clone();
335    view! {
336        <div class="flex flex-col gap-2" data-card-fields>
337            {rows}
338            <datalist id=list>
339                {suggestions.into_iter().map(|suggestion| view! { <option value=suggestion></option> }).collect_view()}
340            </datalist>
341            <button
342                type="button"
343                class=format!("{BUTTON} self-start")
344                data-add-workspace
345                on:click=move |_| set.selection.update(|selection| selection.add_workspace(&added))
346            >
347                <Glyph icon=icondata::FaPlusSolid />
348                "another workspace"
349            </button>
350        </div>
351    }
