19/// A card's frame, with its selected look.
20fn card_class(selected: bool, offline: bool) -> String {
21    let state = if selected {
22        "border-accent bg-sunken"
23    } else if offline {
24        "border-line opacity-60"
25    } else {
26        "border-line-2 bg-surface hover:border-accent hover:bg-hover"
27    };
28    let cursor = if offline && !selected {
29        "cursor-not-allowed"
30    } else {
31        "cursor-pointer"
32    };
33    format!("runtime-card flex min-w-0 flex-col gap-2 rounded-lg border-2 p-3 {state} {cursor}")
34}

36fn state_class(state: CardState) -> &'static str {
37    match state {
38        CardState::Ready => "text-ok",
39        CardState::StartsOnUse => "text-accent",
40        CardState::Offline => "text-muted",
41    }
42}

44/// The new session's runtime cards, above the composer whose first message
45/// creates the session with them.
46#[component]
47pub fn NewSessionCards(workspace: Workspace) -> impl IntoView {
48    let groups = Memo::new(move |_| workspace.card_groups());
49    let cards = Memo::new(move |_| workspace.cards());
50    let blocker = move || {
51        workspace
52            .selection
53            .with(|selection| cards.with(|cards| selection.blocker(cards)))
54    };
55    // Changing the cards answers why the last message did not create the
56    // session.
57    Effect::new(move |previous: Option<()>| {
58        workspace.selection.track();
59        if previous.is_some() {
60            workspace.draft_error.set(None);
61        }
62    });
63    // The chips' words (docs/site/specs/web-polish.md, P2).
64    let runs_on = move || {
65        cards.with(|cards| {
66            workspace.selection.with(|selection| {
67                let selected: Vec<&Card> = cards
68                    .iter()
69                    .filter(|card| selection.selected(card))
70                    .collect();
71                crate::cards::runs_on(&selected)
72            })
73        })
74    };
75    // The folder chip, only when a coding runtime has a folder: "global
76    // only" is said by Runs on.
77    let folder = move || {
78        workspace
79            .places_label()
80            .filter(|places| places.contains('/'))
81            .map(|places| crate::places::folder_names(&places))
82            .unwrap_or_default()
83    };
84    let skills_off = move || super::tool_groups::cards_skills_off(workspace, workspace.new_cards());
85    let panel = move || workspace.draft_panel.get();
86    let toggle = move |which: DraftPanel| {
87        workspace.draft_panel.update(|open| {
88            *open = if *open == Some(which) {
89                None
90            } else {
91                Some(which)
92            }
93        });
94    };
95    view! {
96        <section
97            id="new-session"
98            aria-label="New session"
99            aria-busy=move || workspace.creating.get().to_string()
100            class="mx-auto flex w-full max-w-3xl flex-col gap-3 py-3"
101            class:opacity-60=move || workspace.creating.get()
102            class:pointer-events-none=move || workspace.creating.get()
103        >
104            // The composer comes first: a heading over it, and one row of
105            // chips for what the session gets, each opening its panel.
106            <div class="flex flex-col items-center pt-4 pb-1">
107                <h2 class="text-center text-[1.375rem] font-semibold tracking-tight text-fg">"What should cookie work on?"</h2>
108            </div>
109            <div class="flex flex-wrap items-center justify-center gap-1.5" data-draft-chips>
110                <super::projects::NewSessionProject workspace=workspace />
111                <button
112                    type="button"
113                    class=move || chip_class(panel() == Some(DraftPanel::Where))
114                    aria-expanded=move || (panel() == Some(DraftPanel::Where)).to_string()
115                    aria-controls="draft-where"
116                    title="Where the session runs: click for the runtimes"
117                    data-new-session-chip="where"
118                    on:click=move |_| toggle(DraftPanel::Where)
119                >
120                    <Glyph icon=icondata::LuHouse class="shrink-0 text-muted" />
121                    <span class="min-w-0 truncate" data-runs-on>{runs_on}</span>
122                    <Glyph icon=icondata::LuChevronDown class="shrink-0 text-[0.625rem] opacity-50" />
123                </button>
124                {move || {
125                    let folder = folder();
126                    (!folder.is_empty()).then(|| view! {
127                        <button
128                            type="button"
129                            class=chip_class(false)
130                            title=move || workspace.places_label().unwrap_or_default()
131                            data-new-session-chip="folder"
132                            on:click=move |_| workspace.draft_panel.set(Some(DraftPanel::Where))
133                        >
134                            <Glyph icon=icondata::LuFolder class="shrink-0 text-muted" />
135                            <span class="min-w-0 truncate font-mono text-xs">{folder}</span>
136                        </button>
137                    })
138                }}
139                <button
140                    type="button"
141                    class=move || chip_class(panel() == Some(DraftPanel::Tools))
142                    aria-expanded=move || (panel() == Some(DraftPanel::Tools)).to_string()
143                    aria-controls="draft-tools"
144                    title="The tools the runtimes bring, and the skills"
145                    data-new-session-chip="tools"
146                    on:click=move |_| toggle(DraftPanel::Tools)
147                >
148                    <Glyph icon=icondata::LuWandSparkles class="shrink-0 text-muted" />
149                    <span>"Tools & skills"</span>
150                    {move || {
151                        let off = skills_off();
152                        (off > 0).then(|| view! { <span class="text-xs font-semibold text-warn">{format!("{off} off")}</span> })
153                    }}
154                    <Glyph icon=icondata::LuChevronDown class="shrink-0 text-[0.625rem] opacity-50" />
155                </button>
156            </div>
157            {move || blocker().map(|blocker| view! {
158                <p id="start-blocker" class=format!("text-center {NOTE}")>{blocker}</p>
159            })}
160            <Show when=move || panel() == Some(DraftPanel::Where)>
161                <div id="draft-where" class=PANEL>
162                    <p class=NOTE>
163                        "Pick the runtimes this session uses; a coding runtime needs a folder. With only global, the session hands work on files to subagents."
164                    </p>
165                    {move || {
166                        if workspace.runtimes.with(Option::is_none) {
167                            return super::loading("Reading the runtimes…").into_any();
168                        }
169                        groups
170                            .get()
171                            .into_iter()
172                            .map(|group| group_view(workspace, group, workspace.new_cards()))
173                            .collect_view()
174                            .into_any()
175                    }}
176                    <p class="flex items-start gap-2 text-sm" id="draft-places">
177                        <Glyph icon=icondata::LuFolderOpen class="mt-0.5 text-accent" />
178                        <span class="min-w-0 break-words font-mono" data-draft-places>
179                            {move || workspace.places_label().unwrap_or_default()}
180                        </span>
181                    </p>
182                </div>
183            </Show>
184            <Show when=move || panel() == Some(DraftPanel::Tools)>
185                <div id="draft-tools" class=PANEL>
186                    {super::tool_groups::draft_figures(workspace, workspace.new_cards())}
187                    {super::tool_groups::cards_skills(workspace, workspace.new_cards(), "Skills", "new-session-skills")}
188                </div>
189            </Show>
190        </section>
191    }
192}

230/// A machine's cards under a header that folds them, selected in `set`.
231pub(super) fn group_view(workspace: Workspace, group: Group, set: CardSet) -> impl IntoView {
232    let name = group.name();
233    let toggle_name = name.clone();
234    let header = group.clone();
235    let shown = {
236        let group = group.clone();
237        Memo::new(move |_| workspace.group_shown(&group, set))
238    };
239    let summary = {
240        let group = group.clone();
241        move || set.selection.with(|selection| group.summary(selection))
242    };
243    let cards = group.cards.clone();
244    view! {
245        // Two cards a row only where the cards' own container is wide (the
246        // new session's page), one in a narrow one (the project editor).
247        <div class="@container flex flex-col gap-2" data-group=name data-open=move || shown.get().to_string()>
248            <button
249                type="button"
250                class="flex min-h-11 w-full items-center gap-2 rounded-md px-2 text-left hover:bg-hover active:bg-press"
251                aria-expanded=move || shown.get().to_string()
252                data-group-toggle=toggle_name
253                on:click=move |_| workspace.toggle_group(&header, set)
254            >
255                <span
256                    class="inline-flex transition-transform"
257                    class:rotate-90=move || shown.get()
258                >
259                    <Glyph icon=icondata::LuChevronRight class="text-xs" />
260                </span>
261                <Glyph icon=if group.daemon { icondata::LuHouse } else { icondata::LuMonitor } class="text-accent" />
262                <span class="min-w-0 flex-1 truncate text-sm" data-group-summary>{summary}</span>
263            </button>
264            <Show when=move || shown.get()>
265                <div class="grid grid-cols-1 items-start gap-2 @lg:grid-cols-2">
266                    {cards.clone().into_iter().map(|card| card_view(workspace, card, set)).collect_view()}
267                </div>
268            </Show>
269        </div>
270    }
271}

273/// One card: a click anywhere selects it, except in its fields and
274/// buttons.
275fn card_view(workspace: Workspace, card: Card, set: CardSet) -> impl IntoView {
276    let key = card.key();
277    let selected = {
278        let card = card.clone();
279        Memo::new(move |_| set.selection.with(|selection| selection.selected(&card)))
280    };
281    let offline = card.state == CardState::Offline;
282    let on_click = {
283        let card = card.clone();
284        move |event: ev::MouseEvent| {
285            let inside_control = event
286                .target()
287                .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
288                .and_then(|target| {
289                    target
290                        .closest("input, button, a, [data-card-fields], [data-card-tools]")
291                        .ok()
292                        .flatten()
293                })
294                .is_some();
295            if !inside_control {
296                workspace.toggle_card(&card, set);
297            }
298        }
299    };
300    let toggle = card.clone();
301    let fields = card.clone();
302    let label = card.label();
303    view! {
304        <div
305            class=move || card_class(selected.get(), offline)
306            data-card=key.clone()
307            data-state=card.state.label()
308            data-selected=move || selected.get().to_string()
309            on:click=on_click
310        >
311            <div class="flex items-start gap-2">
312                <button
313                    type="button"
314                    class="flex min-h-11 min-w-0 flex-1 items-start gap-2 rounded-md text-left disabled:cursor-not-allowed"
315                    aria-pressed=move || selected.get().to_string()
316                    aria-label=format!("{label}: select")
317                    disabled=move || offline && !selected.get()
318                    data-card-toggle
319                    on:click=move |_| workspace.toggle_card(&toggle, set)
320                >
321                    <Glyph
322                        icon=Signal::derive(move || if selected.get() { icondata::LuCircleCheck } else { icondata::LuCircle })
323                        class="mt-0.5 text-lg text-accent"
324                    />
325                    <span class="flex min-w-0 flex-1 flex-col">
326                        <span class="flex min-w-0 items-center gap-1.5">
327                            // Its machine's colour, as on the session rows.
328                            {crate::badges::runtime_dot(
329                                &card.machine.as_ref().map_or_else(|| card.runtime_id.to_string(), ToString::to_string),
330                                card.runtime_id.as_str(),
331                            )}
332                            <span class="truncate text-sm font-semibold">{card.label()}</span>
333                        </span>
334                        {(!card.description.is_empty()).then(|| view! {
335                            <span class=format!("line-clamp-2 {NOTE}")>{card.description.clone()}</span>
336                        })}
337                    </span>
338                </button>
339                <span class=format!("mt-1 shrink-0 text-xs font-bold {}", state_class(card.state)) data-card-state>
340                    {card.state.label()}
341                </span>
342            </div>
343            {card.message.clone().filter(|_| offline).map(|message| view! {
344                <p class=format!("break-words {NOTE}")>{message}</p>
345            })}
346            {move || {
347                (selected.get() && fields.takes_workspace)
348                    .then(|| workspace_fields(workspace, fields.clone(), set))
349            }}
350            {super::tool_groups::card_tools(workspace, card.clone(), set, selected)}
351            {
352                let key = key.clone();
353                move || {
354                    // A refusal of a card without a workspace field.
355                    (!card.takes_workspace)
356                        .then(|| set.errors.with(|errors| errors.get(&(key.clone(), 0)).cloned()))
357                        .flatten()
358                        .map(|error| view! { <p class=super::ERROR data-card-error>{error}</p> })
359                }
360            }
361        </div>
362    }
363}

365/// A selected card's workspace fields, their suggestions, and "+ another
366/// workspace".
367fn workspace_fields(workspace: Workspace, card: Card, set: CardSet) -> impl IntoView {
368    let list = format!("workspaces-{}", card.runtime_id);
369    let suggestions = workspace.workspace_suggestions(&card.runtime_id);
370    let rows = {
371        let card = card.clone();
372        let list = list.clone();
373        move || {
374            let workspaces = set.selection.with(|selection| selection.workspaces(&card));
375            let removable = workspaces.len() > 1;
376            workspaces
377                .into_iter()
378                .enumerate()
379                .map(|(index, text)| {
380                    let edited = card.clone();
381                    let removed = card.clone();
382                    // A runtime that started while the session was being
383                    // created has a card per capability now (`alt` became
384                    // `alt/coding`): the error its first card got shows here.
385                    let error = set.errors.with(|errors| {
386                        errors
387                            .get(&(card.key(), index))
388                            .or_else(|| errors.get(&(card.runtime_id.to_string(), index)))
389                            .cloned()
390                    });
391                    view! {
392                        <div class="flex flex-col gap-1">
393                            <div class="flex items-center gap-1">
394                                <input
395                                    type="text"
396                                    class=FIELD
397                                    list=list.clone()
398                                    placeholder="~/project"
399                                    autocomplete="off"
400                                    spellcheck="false"
401                                    aria-label=format!("Workspace {} of {}", index + 1, card.label())
402                                    data-workspace-field=index.to_string()
403                                    prop:value=text
404                                    on:input=move |event| {
405                                        let text = event_target_value(&event);
406                                        set.selection.update(|selection| {
407                                            selection.set_workspace(&edited, index, text);
408                                        });
409                                    }
410                                />
411                                {removable.then(|| view! {
412                                    <button
413                                        type="button"
414                                        class=ICON_BUTTON
415                                        aria-label="Remove this workspace"
416                                        title="Remove this workspace"
417                                        data-remove-workspace=index.to_string()
418                                        on:click=move |_| set.selection.update(|selection| {
419                                            selection.remove_workspace(&removed, index);
420                                        })
421                                    >
422                                        <Glyph icon=icondata::LuX />
423                                    </button>
424                                })}
425                            </div>
426                            {error.map(|error| view! { <p class=super::ERROR data-card-error>{error}</p> })}
427                        </div>
428                    }
429                })
430                .collect_view()
431        }
432    };
433    // One workspace a card: one machine offers each tool once, so a second
434    // directory there is reached by absolute path.
435    view! {
436        <div class="flex flex-col gap-2" data-card-fields>
437            {rows}
438            <datalist id=list>
439                {suggestions.into_iter().map(|suggestion| view! { <option value=suggestion></option> }).collect_view()}
440            </datalist>
441        </div>
442    }
