1//! What is live in the open session, as badges in its title bar
2//! (docs/site/specs/quiet-ink.md, D6): the goal, the calls running in the
3//! background, the approvals waiting, and the context's use. A badge comes
4//! when its state starts, follows it, and goes when it ends; a click opens
5//! the session panel at its tab. The context ring always shows once a
6//! request has measured it.

21/// One badge: a quiet pill that tints on hover and settles in as it comes.
22const BADGE: &str = "badge-in inline-flex h-11 lg:h-7 items-center gap-1.5 rounded-md px-2 text-xs font-medium tabular-nums text-fg-2 transition-colors hover:bg-hover hover:text-fg active:bg-press";

24/// How much of `window` `tokens` fill, from 0 to 1.
25fn filled(tokens: u64, window: Option<u64>) -> Option<f64> {
26    let window = window.filter(|window| *window > 0)?;
27    #[allow(clippy::cast_precision_loss)]
28    Some((tokens as f64 / window as f64).clamp(0.0, 1.0))
29}

31/// The context ring's title.
32pub(super) fn context_title(tokens: u64, window: Option<u64>) -> String {
33    match window {
34        Some(window) => format!(
35            "Context: {} of {} tokens",
36            format_count(tokens),
37            format_count(window)
38        ),
39        None => format!(
40            "Context: {} tokens in the last request",
41            format_count(tokens)
42        ),
43    }
44}

46/// A ring `fraction` of the way round, in the accent.
47fn ring(fraction: f64) -> impl IntoView {
48    // r = 6: a circumference of 37.7.
49    const ROUND: f64 = 37.7;
50    view! {
51        <svg class="h-3.5 w-3.5 -rotate-90" viewBox="0 0 16 16" aria-hidden="true">
52            <circle cx="8" cy="8" r="6" fill="none" stroke="var(--color-line-2)" stroke-width="2.25"></circle>
53            <circle
54                cx="8"
55                cy="8"
56                r="6"
57                fill="none"
58                stroke="var(--color-accent)"
59                stroke-width="2.25"
60                stroke-linecap="round"
61                stroke-dasharray=format!("{ROUND}")
62                stroke-dashoffset=format!("{:.2}", ROUND * (1.0 - fraction))
63            ></circle>
64        </svg>
65    }
66}

68/// The open session's live badges.
69#[component]
70pub fn LiveBadges(workspace: Workspace) -> impl IntoView {
71    let goal = Memo::new(move |_| workspace.goal().filter(|goal| goal_bar_shows(goal.status)));
72    let background = Memo::new(move |_| workspace.background_calls().len());
73    let approvals = Memo::new(move |_| workspace.approvals().len());
74    let context = move || {
75        workspace.viewed.with(|viewed| {
76            viewed
77                .context_tokens
78                .map(|tokens| (tokens, viewed.context_window))
79        })
80    };
81    let goal_badge = move || {
82        goal.get().map(|goal| {
83            let paused = goal.status == GoalStatus::Paused;
84            let done = goal.items.iter().filter(|item| item.finished).count();
85            let total = goal.items.len();
86            let objective = goal.objective.split_whitespace().collect::<Vec<_>>().join(" ");
87            let title = if paused {
88                format!("Goal (paused): {objective}")
89            } else if total > 0 {
90                format!("Goal: {objective} ({done} of {total} done)")
91            } else {
92                format!("Goal: {objective}")
93            };
94            view! {
95                <button
96                    type="button"
97                    class=BADGE
98                    id="goal-badge"
99                    data-goal-status=if paused { "paused" } else { "active" }
100                    aria-label=title.clone()
101                    title=title
102                    on:click=move |_| workspace.panel.set(Some(Panel::Goal))
103                >
104                    <Glyph
105                        icon=if paused { icondata::LuCirclePause } else { icondata::LuTarget }
106                        class=if paused { "text-sm text-muted" } else { "text-sm text-accent" }
107                    />
108                    {(total > 0).then(|| view! { <span data-goal-progress>{format!("{done}/{total}")}</span> })}
109                </button>
110            }
111        })
112    };
113    view! {
114        <div class="flex shrink-0 items-center gap-0.5" data-live-badges>
115            {goal_badge}
116            <Show when=move || { background.get() > 0 }>
117                <button
118                    type="button"
119                    class=BADGE
120                    id="background-badge"
121                    aria-label=move || format!("{} in the background", background.get())
122                    title=move || format!("{} running in the background: click for them", background.get())
123                    on:click=move |_| workspace.run_command(Command::Background)
124                >
125                    <Glyph icon=icondata::LuLoaderCircle class="animate-spin text-sm text-accent" />
126                    <span>{move || background.get()}</span>
127                </button>
128            </Show>
129            <Show when=move || { approvals.get() > 0 }>
130                <button
131                    type="button"
132                    class=format!("{BADGE} bg-warn-soft text-warn hover:bg-warn-soft hover:text-warn")
133                    id="approvals-badge"
134                    aria-label=move || format!("{} waiting for you", super::approvals_label(approvals.get()))
135                    title="Approvals waiting: click to answer"
136                    on:click=move |_| workspace.panel.set(Some(Panel::Approvals))
137                >
138                    <Glyph icon=icondata::LuShield class="text-sm" />
139                    <span>{move || approvals.get()}</span>
140                </button>
141            </Show>
142            {move || context().map(|(tokens, window)| {
143                let title = context_title(tokens, window);
144                view! {
145                    <button
146                        type="button"
147                        class=BADGE
148                        id="context"
149                        aria-label=title.clone()
150                        title=title
151                        on:click=move |_| workspace.run_command(Command::Usage)
152                    >
153                        {ring(filled(tokens, window).unwrap_or(0.0))}
154                        <span class="text-muted">
155                            {match filled(tokens, window) {
156                                Some(fraction) => format!("{:.0}%", fraction * 100.0),
157                                None => compact_count(tokens),
158                            }}
159                        </span>
160                    </button>
161                }
162            })}
163        </div>
164    }
165}

167#[cfg(test)]
168mod tests {
169    use super::filled;
170
171    #[test]
172    fn the_ring_fills_by_the_share_of_a_known_window() {
173        assert_eq!(filled(0, Some(400_000)), Some(0.0));
174        assert_eq!(filled(100_000, Some(400_000)), Some(0.25));
175        assert_eq!(filled(500_000, Some(400_000)), Some(1.0), "never past full");
176        assert_eq!(filled(10, None), None, "no window, no share");
177        assert_eq!(filled(10, Some(0)), None);
178    }
