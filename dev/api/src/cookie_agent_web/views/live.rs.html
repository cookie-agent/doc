1//! What is live in the open session, as badges in its title bar
2//! (docs/site/specs/quiet-ink.md, D6): the goal, the calls running in the
3//! background, the approvals waiting, and the context's use. A badge comes
4//! when its state starts, follows it, and goes when it ends; a click opens
5//! the session panel at its tab. The context ring always shows once a
6//! request has measured it.

22/// One badge: a quiet pill that tints on hover and settles in as it comes.
23const BADGE: &str = "badge-in inline-flex h-11 lg:h-7 items-center gap-1.5 rounded-md px-2 text-xs font-medium tabular-nums text-fg-2 transition-colors hover:bg-hover hover:text-fg active:bg-press";

25/// How much of `window` `tokens` fill, from 0 to 1.
26fn filled(tokens: u64, window: Option<u64>) -> Option<f64> {
27    let window = window.filter(|window| *window > 0)?;
28    #[allow(clippy::cast_precision_loss)]
29    Some((tokens as f64 / window as f64).clamp(0.0, 1.0))
30}

32/// The context ring's title.
33pub(super) fn context_title(tokens: u64, window: Option<u64>) -> String {
34    match window {
35        Some(window) => format!(
36            "Context: {} of {} tokens",
37            format_count(tokens),
38            format_count(window)
39        ),
40        None => format!(
41            "Context: {} tokens in the last request",
42            format_count(tokens)
43        ),
44    }
45}

47/// A ring `fraction` of the way round, in the accent.
48fn ring(fraction: f64) -> impl IntoView {
49    // r = 6: a circumference of 37.7.
50    const ROUND: f64 = 37.7;
51    view! {
52        <svg class="h-3.5 w-3.5 -rotate-90" viewBox="0 0 16 16" aria-hidden="true">
53            <circle cx="8" cy="8" r="6" fill="none" stroke="var(--color-line-2)" stroke-width="2.25"></circle>
54            <circle
55                cx="8"
56                cy="8"
57                r="6"
58                fill="none"
59                stroke="var(--color-accent)"
60                stroke-width="2.25"
61                stroke-linecap="round"
62                stroke-dasharray=format!("{ROUND}")
63                stroke-dashoffset=format!("{:.2}", ROUND * (1.0 - fraction))
64            ></circle>
65        </svg>
66    }
67}

69/// The cache badge's icon and its colour (`docs/site/specs/cache-aware-idle.md`
70/// section 4.5): a flame while warm, graded by heat; a snowflake once
71/// expired; a spinner while compacting.
72fn cache_icon(badge: &CacheBadge) -> (icondata::Icon, &'static str, bool) {
73    match badge {
74        CacheBadge::Compacting => (icondata::LuLoaderCircle, "text-sm text-accent", true),
75        CacheBadge::InUse { .. } | CacheBadge::Forever { .. } => {
76            (icondata::LuFlame, "text-sm text-hot", false)
77        }
78        CacheBadge::Counting { heat, .. } => match heat {
79            CacheHeat::Hot => (icondata::LuFlame, "text-sm text-hot", false),
80            CacheHeat::Warm => (icondata::LuFlame, "text-sm text-hot opacity-60", false),
81            CacheHeat::Cooling => (icondata::LuFlame, "text-sm text-muted", false),
82            CacheHeat::Expired => (icondata::LuSnowflake, "text-sm text-faint", false),
83        },
84    }
85}

87/// The open session's live badges.
88#[component]
89pub fn LiveBadges(workspace: Workspace) -> impl IntoView {
90    // The cache badge counts down while it shows.
91    let now = RwSignal::new(jiff::Timestamp::now());
92    let ticker = set_interval_with_handle(
93        move || now.set(jiff::Timestamp::now()),
94        std::time::Duration::from_secs(1),
95    )
96    .ok();
97    on_cleanup(move || {
98        if let Some(ticker) = ticker {
99            ticker.clear();
100        }
101    });
102    let cache_badge = move || {
103        let viewed = workspace.viewed.get();
104        let clock = viewed.cache_clock?;
105        let session = workspace.selected.get()?;
106        let plan = workspace.sessions.with(|sessions| {
107            sessions
108                .iter()
109                .find(|meta| meta.session_id == session)
110                .and_then(|meta| meta.cache_compaction.clone())
111        });
112        let now = now.get();
113        let badge = clock.badge(now, viewed.running, plan.as_ref());
114        let zone = crate::cron::zone_at(now);
115        let title = detail_lines(&clock, &badge, plan.as_ref(), now, &zone).join("\n");
116        let (icon, class, spins) = cache_icon(&badge);
117        let text = badge_text(&badge);
118        let planned = matches!(badge, CacheBadge::Counting { planned: true, .. });
119        let cooling = matches!(
120            badge,
121            CacheBadge::Counting {
122                heat: CacheHeat::Cooling,
123                ..
124            }
125        );
126        Some(view! {
127            <button
128                type="button"
129                class=format!("{BADGE} relative")
130                id="cache-badge"
131                aria-label=title.clone()
132                title=title
133                on:click=move |_| workspace.run_command(Command::Usage)
134            >
135                <Glyph icon=icon class=if spins { format!("{class} animate-spin") } else { class.to_owned() } />
136                {(!text.is_empty()).then(|| view! {
137                    <span class=if cooling { "text-warn" } else { "text-muted" }>{text}</span>
138                })}
139                {planned.then(|| view! {
140                    <span class="absolute right-1 top-1 h-1.5 w-1.5 rounded-full bg-accent" data-cache-planned></span>
141                })}
142            </button>
143        })
144    };
145    let goal = Memo::new(move |_| workspace.goal().filter(|goal| goal_bar_shows(goal.status)));
146    let background = Memo::new(move |_| workspace.background_calls().len());
147    let approvals = Memo::new(move |_| workspace.approvals().len());
148    let context = move || {
149        workspace.viewed.with(|viewed| {
150            viewed
151                .context_tokens
152                .map(|tokens| (tokens, viewed.context_window))
153        })
154    };
155    let goal_badge = move || {
156        goal.get().map(|goal| {
157            let paused = goal.status == GoalStatus::Paused;
158            let done = goal.items.iter().filter(|item| item.finished).count();
159            let total = goal.items.len();
160            let objective = goal.objective.split_whitespace().collect::<Vec<_>>().join(" ");
161            let title = if paused {
162                format!("Goal (paused): {objective}")
163            } else if total > 0 {
164                format!("Goal: {objective} ({done} of {total} done)")
165            } else {
166                format!("Goal: {objective}")
167            };
168            view! {
169                <button
170                    type="button"
171                    class=BADGE
172                    id="goal-badge"
173                    data-goal-status=if paused { "paused" } else { "active" }
174                    aria-label=title.clone()
175                    title=title
176                    on:click=move |_| workspace.panel.set(Some(Panel::Goal))
177                >
178                    <Glyph
179                        icon=if paused { icondata::LuCirclePause } else { icondata::LuTarget }
180                        class=if paused { "text-sm text-muted" } else { "text-sm text-accent" }
181                    />
182                    {(total > 0).then(|| view! { <span data-goal-progress>{format!("{done}/{total}")}</span> })}
183                </button>
184            }
185        })
186    };
187    view! {
188        <div class="flex shrink-0 items-center gap-0.5" data-live-badges>
189            {goal_badge}
190            {cache_badge}
191            <Show when=move || { background.get() > 0 }>
192                <button
193                    type="button"
194                    class=BADGE
195                    id="background-badge"
196                    aria-label=move || format!("{} in the background", background.get())
197                    title=move || format!("{} running in the background: click for them", background.get())
198                    on:click=move |_| workspace.run_command(Command::Background)
199                >
200                    <Glyph icon=icondata::LuLoaderCircle class="animate-spin text-sm text-accent" />
201                    <span>{move || background.get()}</span>
202                </button>
203            </Show>
204            <Show when=move || { approvals.get() > 0 }>
205                <button
206                    type="button"
207                    class=format!("{BADGE} bg-warn-soft text-warn hover:bg-warn-soft hover:text-warn")
208                    id="approvals-badge"
209                    aria-label=move || format!("{} waiting for you", super::approvals_label(approvals.get()))
210                    title="Approvals waiting: click to answer"
211                    on:click=move |_| workspace.panel.set(Some(Panel::Approvals))
212                >
213                    <Glyph icon=icondata::LuShield class="text-sm" />
214                    <span>{move || approvals.get()}</span>
215                </button>
216            </Show>
217            {move || context().map(|(tokens, window)| {
218                let title = context_title(tokens, window);
219                view! {
220                    <button
221                        type="button"
222                        class=BADGE
223                        id="context"
224                        aria-label=title.clone()
225                        title=title
226                        on:click=move |_| workspace.run_command(Command::Usage)
227                    >
228                        {ring(filled(tokens, window).unwrap_or(0.0))}
229                        <span class="text-muted">
230                            {match filled(tokens, window) {
231                                Some(fraction) => format!("{:.0}%", fraction * 100.0),
232                                None => compact_count(tokens),
233                            }}
234                        </span>
235                    </button>
236                }
237            })}
238        </div>
239    }
240}

242#[cfg(test)]
243mod tests {
244    use super::filled;
245
246    #[test]
247    fn the_ring_fills_by_the_share_of_a_known_window() {
248        assert_eq!(filled(0, Some(400_000)), Some(0.0));
249        assert_eq!(filled(100_000, Some(400_000)), Some(0.25));
250        assert_eq!(filled(500_000, Some(400_000)), Some(1.0), "never past full");
251        assert_eq!(filled(10, None), None, "no window, no share");
252        assert_eq!(filled(10, Some(0)), None);
253    }
