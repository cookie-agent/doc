26/// The viewed session's tree, folded where the user folded it, shown as
27/// `/agent-panel` says.
28#[component]
29pub fn TreePanel(workspace: Workspace) -> impl IntoView {
30    let rows = move || {
31        let collapsed = workspace.collapsed.get();
32        workspace
33            .tree
34            .with(|tree| tree.as_ref().map(|tree| tree_rows(tree, &collapsed)))
35            .unwrap_or_default()
36    };
37    view! {
38        <Show when=move || workspace.tree_panel_visible()>
39            <nav id="agent-tree" class="flex max-h-72 flex-col gap-1 overflow-y-auto border-b border-toasted p-2 dark:border-ember" aria-label="Subagents">
40                <p class="px-2 text-xs font-bold">"Agents"</p>
41                {move || {
42                    rows()
43                        .into_iter()
44                        .map(|row| {
45                            let session_id = row.session_id;
46                            let current = move || workspace.selected.get() == Some(session_id);
47                            let indent = format!("{}rem", row.depth);
48                            let fold = row.has_children.then(|| {
49                                view! {
50                                    <button
51                                        type="button"
52                                        class="min-h-11 min-w-11 rounded hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke"
53                                        aria-label="Fold"
54                                        aria-expanded=(!row.collapsed).to_string()
55                                        on:click=move |_| workspace.toggle_folded(session_id)
56                                    >
57                                        {if row.collapsed { "▸" } else { "▾" }}
58                                    </button>
59                                }
60                            });
61                            view! {
62                                <div class="flex items-center" style:padding-left=indent>
63                                    {fold}
64                                    <button
65                                        type="button"
66                                        class=ROW
67                                        class:bg-toasted=current
68                                        class:dark:bg-ember=current
69                                        data-tree-session=session_id.to_string()
70                                        on:click=move |_| workspace.open_session(session_id)
71                                    >
72                                        <span class="w-full truncate text-sm">
73                                            {if row.running { "● " } else { "" }}
74                                            {row.label.clone()}
75                                        </span>
76                                    </button>
77                                </div>
78                            }
79                        })
80                        .collect_view()
81                }}
82            </nav>
83        </Show>
84    }
85}

87/// The root session's goal, with the controls its status allows.
88#[component]
89pub fn GoalBar(workspace: Workspace) -> impl IntoView {
90    let goal = Memo::new(move |_| workspace.goal());
91    view! {
92        {move || {
93            goal.get()
94                .map(|goal| {
95                    let read_only = !workspace.writable();
96                    let mut summary = goal.objective.split_whitespace().collect::<Vec<_>>().join(" ");
97                    if matches!(
98                        goal.status,
99                        cookie_agent_protocol::GoalStatus::Completed
100                            | cookie_agent_protocol::GoalStatus::Cancelled
101                    ) {
102                        summary.push_str(&format!(" · {}", status_name(goal.status)));
103                    }
104                    let controls = goal_actions(goal.status, read_only)
105                        .iter()
106                        .map(|action| {
107                            let action = *action;
108                            view! {
109                                <button
110                                    type="button"
111                                    class=BUTTON
112                                    data-goal-action=action_label(action).to_lowercase()
113                                    on:click=move |_| {
114                                        workspace.goal_command(GoalCommand::Lifecycle(action))
115                                    }
116                                >
117                                    {action_label(action)}
118                                </button>
119                            }
120                        })
121                        .collect_view();
122                    view! {
123                        <div id="goal-bar" class=BAR>
124                            <button
125                                type="button"
126                                class=format!("{ROW} min-w-0 flex-1 justify-center")
127                                on:click=move |_| workspace.panel.set(Some(crate::workspace::Panel::Goal))
128                            >
129                                <span class="w-full truncate">{format!("🎯 {summary}")}</span>
130                            </button>
131                            {controls}
132                        </div>
133                    }
134                })
135        }}
136    }

235/// The attach and camera buttons and their hidden file inputs, offered
236/// only for the media the selected model accepts.
237#[component]
238pub fn AttachButtons(workspace: Workspace) -> impl IntoView {
239    let media = Memo::new(move |_| {
240        workspace
241            .accepted_media()
242            .map(|accepted| accepted.media)
243            .unwrap_or_default()
244    });
245    let accept = move || accept_attribute(&media.get());
246    let camera_accept = move || {
247        let media = media.get();
248        accept_attribute(
249            &media
250                .into_iter()
251                .filter(|entry| entry.kind.as_str() == "image")
252                .collect::<Vec<_>>(),
253        )
254    };
255    let on_change = move |event: ev::Event| {
256        let Some(input) = event.target().and_then(|target| {
257            wasm_bindgen::JsCast::dyn_into::<web_sys::HtmlInputElement>(target).ok()
258        }) else {
259            return;
260        };
261        let files = files_of(input.files());
262        input.set_value("");
263        workspace.attach_files(files);
264    };
265    view! {
266        <Show when=move || !media.with(Vec::is_empty)>
267            <input
268                type="file"
269                id=FILE_INPUT
270                class="hidden"
271                multiple
272                accept=accept
273                on:change=on_change
274            />
275            <button
276                type="button"
277                id="attach"
278                class=format!("{BUTTON} disabled:opacity-60")
279                aria-label="Attach a file"
280                title="Attach a file"
281                disabled=move || !workspace.writable()
282                on:click=move |_| pick_files(false)
283            >
284                "📎"
285            </button>
286            <Show when=move || accepts_images(&media.get())>
287                <input
288                    type="file"
289                    id=CAMERA_INPUT
290                    class="hidden"
291                    accept=camera_accept
292                    capture="environment"
293                    on:change=on_change
294                />
295                <button
296                    type="button"
297                    class=format!("{BUTTON} disabled:opacity-60 md:hidden")
298                    aria-label="Take a photo"
299                    disabled=move || !workspace.writable()
300                    on:click=move |_| pick_files(true)
301                >
302                    "📷"
303                </button>
304            </Show>
305        </Show>
306    }

309pub fn goal_sheet(workspace: Workspace) -> impl IntoView {
310    move || {
311        let goal = workspace.goal();
312        let current = match &goal {
313            Some(goal) => {
314                let items = goal
315                    .items
316                    .iter()
317                    .map(|item| {
318                        let mark = if item.finished { "☑" } else { "☐" };
319                        view! { <li>{format!("{mark} {}", item.description)}</li> }
320                    })
321                    .collect_view();
322                view! {
323                    <p class="text-sm font-bold">{format!("Goal ({})", status_name(goal.status))}</p>
324                    <p id="goal-objective" class="text-sm whitespace-pre-wrap">{goal.objective.clone()}</p>
325                    <ul class="list-none text-xs">{items}</ul>
326                }
327                    .into_any()
328            }
329            None => view! { <p class="text-sm">"No goal is set."</p> }.into_any(),
330        };
331        let controls = goal.map(|goal| {
332            goal_actions(goal.status, !workspace.writable())
333                .iter()
334                .map(|action| {
335                    let action = *action;
336                    view! {
337                        <button
338                            type="button"
339                            class=BUTTON
340                            on:click=move |_| workspace.goal_command(GoalCommand::Lifecycle(action))
341                        >
342                            {action_label(action)}
343                        </button>
344                    }
345                })
346                .collect_view()
347        });
348        view! {
349            {current}
350            <div class="flex flex-wrap gap-2">{controls}</div>
351            <Show when=move || workspace.viewing_root() && workspace.writable()>
352                <label class="mt-3 flex flex-col gap-1 text-sm">
353                    <span>"New objective"</span>
354                    <textarea
355                        id="goal-input"
356                        rows="3"
357                        class=FIELD
358                        prop:value=move || workspace.goal_draft.get()
359                        on:input=move |event| workspace.goal_draft.set(event_target_value(&event))
360                    ></textarea>
361                </label>
362                <button
363                    type="button"
364                    id="goal-set"
365                    class=BUTTON
366                    on:click=move |_| {
367                        let objective = workspace.goal_draft.get_untracked();
368                        if objective.trim().is_empty() {
369                            workspace.status.set("goal objective must not be empty".into());
370                            return;
371                        }
372                        workspace.goal_draft.set(String::new());
373                        workspace.panel.set(None);
374                        workspace.goal_command(GoalCommand::Objective(objective.trim().to_owned()));
375                    }
376                >
377                    "Set the goal"
378                </button>
379            </Show>
380            <Show when=move || !workspace.viewing_root()>
381                <p class="text-xs">"Goals belong to root sessions."</p>
382            </Show>
383        }
384    }
385}

387/// How long ago a background call was detached, as the TUI says it.
388fn elapsed_label(seconds: i64) -> String {
389    match seconds {
390        ..60 => format!("{}s", seconds.max(0)),
391        60..3600 => format!("{}m{:02}s", seconds / 60, seconds % 60),
392        _ => format!("{}h{:02}m", seconds / 3600, seconds % 3600 / 60),
393    }
394}

396pub fn background_sheet(workspace: Workspace) -> impl IntoView {
397    // The elapsed times tick while the sheet is open.
398    let now = RwSignal::new(jiff::Timestamp::now());
399    let ticker = set_interval_with_handle(
400        move || now.set(jiff::Timestamp::now()),
401        Duration::from_secs(1),
402    )
403    .ok();
404    on_cleanup(move || {
405        if let Some(ticker) = ticker {
406            ticker.clear();
407        }
408    });
409    move || {
410        let calls = workspace.background_calls();
411        if calls.is_empty() {
412            return view! { <p class="text-sm">"No calls are running in the background."</p> }
413                .into_any();
414        }
415        calls
416            .into_iter()
417            .map(|call| {
418                let job = call.job.clone();
419                let elapsed = now.get().duration_since(call.detached_at).as_secs();
420                view! {
421                    <div class="background-call flex items-center gap-2 border-b border-toasted py-2 dark:border-ember">
422                        <div class="min-w-0 flex-1">
423                            <p class="truncate font-mono text-sm">{format!("{} · {}", call.job, call.tool)}</p>
424                            <p class="truncate text-xs">
425                                {call.display.clone().unwrap_or_default()}
426                                {format!(" · {}", elapsed_label(elapsed))}
427                            </p>
428                        </div>
429                        <button
430                            type="button"
431                            class=BUTTON
432                            data-stop=call.job.clone()
433                            on:click=move |_| workspace.stop_background(job.clone())
434                        >
435                            "Stop"
436                        </button>
437                    </div>
438                }
439            })
440            .collect_view()
441            .into_any()
442    }
443}

445pub fn skills_sheet(workspace: Workspace) -> impl IntoView {
446    let list = move || match workspace.skills.get() {
447        None => view! { <p class="text-sm">"Reading skills…"</p> }.into_any(),
448        Some(Err(error)) => view! { <p class="text-sm">{error}</p> }.into_any(),
449        Some(Ok(skills)) => {
450            let query = workspace.skill_query.get();
451            let shown = crate::skills::matching_skills(&skills, &query)
452                .into_iter()
453                .cloned()
454                .collect::<Vec<_>>();
455            if shown.is_empty() {
456                return view! { <p class="text-sm">"No matching skill."</p> }.into_any();
457            }
458            shown
459                .into_iter()
460                .map(|skill| {
461                    let picked = skill.clone();
462                    let name = skill.name.clone();
463                    let current = move || {
464                        workspace
465                            .skill_pick
466                            .with(|pick| pick.as_ref().is_some_and(|pick| pick.name == name))
467                    };
468                    view! {
469                        <button
470                            type="button"
471                            class=ROW
472                            class:bg-toasted=current.clone()
473                            class:dark:bg-ember=current
474                            data-skill=skill.name.clone()
475                            on:click=move |_| workspace.skill_pick.set(Some(picked.clone()))
476                        >
477                            <span class="font-mono text-sm font-bold">{skill.name.clone()}</span>
478                            <span class="text-xs text-ink dark:text-sand">{skill.description.clone()}</span>
479                        </button>
480                    }
481                })
482                .collect_view()
483                .into_any()
484        }
485    };
486    view! {
487        <input
488            type="search"
489            id="skill-search"
490            class=FIELD
491            placeholder="Search skills"
492            prop:value=move || workspace.skill_query.get()
493            on:input=move |event| workspace.skill_query.set(event_target_value(&event))
494        />
495        {list}
496        {move || {
497            workspace
498                .skill_pick
499                .get()
500                .map(|skill| {
501                    let hint = skill.argument_hint.clone().unwrap_or_else(|| "arguments".into());
502                    view! {
503                        <div class="mt-3 flex flex-col gap-2 border-t border-toasted pt-3 dark:border-ember">
504                            <p class="font-mono text-sm">{format!("/{}", skill.name)}</p>
505                            <input
506                                type="text"
507                                id="skill-args"
508                                class=FIELD
509                                placeholder=hint
510                                prop:value=move || workspace.skill_args.get()
511                                on:input=move |event| workspace.skill_args.set(event_target_value(&event))
512                                on:keydown=move |event: ev::KeyboardEvent| {
513                                    if event.key() == "Enter" {
514                                        event.prevent_default();
515                                        workspace.run_skill();
516                                    }
517                                }
518                            />
519                            <button type="button" id="skill-run" class=BUTTON on:click=move |_| workspace.run_skill()>
520                                "Run the skill"
521                            </button>
522                        </div>
523                    }
524                })
525        }}
526    }
527}

529pub fn permissions_sheet(workspace: Workspace) -> impl IntoView {
530    let new_action = RwSignal::new(None::<PermissionAction>);
531    let new_pattern = RwSignal::new(String::new());
532    let new_effect = RwSignal::new(PermissionEffect::Ask);
533    let table = move || match workspace.permissions.get() {
534        None => view! { <p class="text-sm">"Reading permissions…"</p> }.into_any(),
535        Some(Err(error)) => view! { <p class="text-sm">{error}</p> }.into_any(),
536        Some(Ok(permissions)) => {
537            let editable = workspace.writable();
538            let rows = permission_rows(&permissions)
539                .into_iter()
540                .map(|row| {
541                    let segments = EFFECTS
542                        .iter()
543                        .map(|effect| {
544                            let effect = *effect;
545                            let action = row.action.clone();
546                            let resource = row.resource.clone();
547                            let active = effect == row.effect;
548                            view! {
549                                <button
550                                    type="button"
551                                    class="min-h-11 min-w-11 px-2 text-xs hover:bg-toasted active:bg-crust disabled:opacity-60 dark:hover:bg-ember dark:active:bg-smoke"
552                                    class:font-bold=active
553                                    class:underline=active
554                                    aria-pressed=active.to_string()
555                                    data-effect=effect_label(effect)
556                                    disabled=!editable
557                                    on:click=move |_| {
558                                        if !active {
559                                            workspace.set_permission(action.clone(), &resource, effect);
560                                        }
561                                    }
562                                >
563                                    {effect_label(effect)}
564                                </button>
565                            }
566                        })
567                        .collect_view();
568                    let remove = (row.source == PermissionRuleSource::SessionOverlay && editable).then(|| {
569                        let action = row.action.clone();
570                        let resource = row.resource.clone();
571                        view! {
572                            <button
573                                type="button"
574                                class="min-h-11 min-w-11 rounded text-xs hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke"
575                                aria-label="Remove the override"
576                                data-remove=format!("{} {}", row.action, row.resource)
577                                on:click=move |_| workspace.clear_permission(action.clone(), &resource)
578                            >
579                                "×"
580                            </button>
581                        }
582                    });
583                    view! {
584                        <tr class="permission-row border-b border-toasted dark:border-ember" data-rule=format!("{} {}", row.action, row.resource)>
585                            <td class="py-1 pr-2 font-mono text-xs">{row.action.to_string()}</td>
586                            <td class="py-1 pr-2 font-mono text-xs break-all">{row.resource.clone()}</td>
587                            <td class="py-1"><div class="flex">{segments}</div></td>
588                            <td class="py-1 text-xs">{source_label(row.source)}</td>
589                            <td>{remove}</td>
590                        </tr>
591                    }
592                })
593                .collect_view();
594            view! {
595                <table class="w-full text-left">
596                    <thead>
597                        <tr class="text-xs">
598                            <th>"Action"</th>
599                            <th>"Pattern"</th>
600                            <th>"Effect"</th>
601                            <th>"From"</th>
602                            <th></th>
603                        </tr>
604                    </thead>
605                    <tbody>{rows}</tbody>
606                </table>
607            }
608            .into_any()
609        }
610    };
611    let actions = move || {
612        workspace.permissions.with(|permissions| {
613            permission_actions(
614                permissions
615                    .as_ref()
616                    .and_then(|permissions| permissions.as_ref().ok())
617                    .map(Vec::as_slice),
618            )
619        })
620    };
621    view! {
622        <Show when=move || workspace.permissions_draft()>
623            <p class="text-xs">"Rules for the new session, previewed against its agent; the first message creates the session with them."</p>
624        </Show>
625        {table}
626        <Show when=move || workspace.writable()>
627            <div id="permission-form" class="mt-3 flex flex-col gap-2 border-t border-toasted pt-3 dark:border-ember">
628                <p class="text-sm font-bold">"New rule"</p>
629                <select
630                    id="permission-action"
631                    class=FIELD
632                    on:change=move |event| {
633                        new_action.set(PermissionAction::new(event_target_value(&event)).ok());
634                    }
635                >
636                    {move || {
637                        let actions = actions();
638                        if new_action.get_untracked().is_none() {
639                            new_action.set(actions.first().cloned());
640                        }
641                        actions
642                            .into_iter()
643                            .map(|action| {
644                                let selected = new_action.get().as_ref() == Some(&action);
645                                view! {
646                                    <option value=action.to_string() selected=selected>
647                                        {action.to_string()}
648                                    </option>
649                                }
650                            })
651                            .collect_view()
652                    }}
653                </select>
654                <input
655                    type="text"
656                    id="permission-pattern"
657                    class=FIELD
658                    placeholder="pattern, such as git *"
659                    prop:value=move || new_pattern.get()
660                    on:input=move |event| new_pattern.set(event_target_value(&event))
661                />
662                <div class="flex">
663                    {EFFECTS
664                        .iter()
665                        .map(|effect| {
666                            let effect = *effect;
667                            view! {
668                                <button
669                                    type="button"
670                                    class=BUTTON
671                                    class:font-bold=move || new_effect.get() == effect
672                                    aria-pressed=move || (new_effect.get() == effect).to_string()
673                                    data-new-effect=effect_label(effect)
674                                    on:click=move |_| new_effect.set(effect)
675                                >
676                                    {effect_label(effect)}
677                                </button>
678                            }
679                        })
680                        .collect_view()}
681                </div>
682                <button
683                    type="button"
684                    id="permission-add"
685                    class=BUTTON
686                    on:click=move |_| {
687                        let Some(action) = new_action.get_untracked() else {
688                            return;
689                        };
690                        let pattern = new_pattern.get_untracked();
691                        if pattern.trim().is_empty() {
692                            workspace.status.set("a rule needs a pattern".into());
693                            return;
694                        }
695                        new_pattern.set(String::new());
696                        workspace.set_permission(action, &pattern, new_effect.get_untracked());
697                    }
698                >
699                    "Add the rule"
700                </button>
701            </div>
702        </Show>
703    }
704}

706pub fn message_sheet(workspace: Workspace) -> impl IntoView {
707    move || {
708        let Some(menu) = workspace.user_menu.get() else {
709            return ().into_any();
710        };
711        let writable = workspace.writable();
712        let actions = if menu.confirming_revert {
713            view! {
714                <p class="text-sm font-bold">"Revert to before this message?"</p>
715                <p class="text-xs">
716                    "This message and everything after it will be removed from the conversation. Its text goes back into the composer so you can edit and resend. To keep both versions, fork instead."
717                </p>
718                <div class="flex flex-wrap gap-2">
719                    <button type="button" id="revert-confirm" class=BUTTON on:click=move |_| workspace.revert()>
720                        "Revert"
721                    </button>
722                    <button type="button" class=BUTTON on:click=move |_| workspace.confirm_revert(false)>
723                        "Back"
724                    </button>
725                </div>
726            }
727                .into_any()
728        } else {
729            view! {
730                <button type="button" id="message-copy" class=ROW on:click=move |_| workspace.copy_message()>
731                    <span class="text-sm font-bold">"Copy"</span>
732                    <span class="text-xs">"copy the message text"</span>
733                </button>
734                <Show when=move || writable>
735                    <button type="button" id="message-revert" class=ROW on:click=move |_| workspace.confirm_revert(true)>
736                        <span class="text-sm font-bold">"Revert"</span>
737                        <span class="text-xs">"remove this message and every later turn; edit and resend it"</span>
738                    </button>
739                    <button type="button" id="message-fork" class=ROW on:click=move |_| workspace.fork()>
740                        <span class="text-sm font-bold">"Fork"</span>
741                        <span class="text-xs">"a new session with the conversation up to this message"</span>
742                    </button>
743                </Show>
744            }
745                .into_any()
746        };
747        view! {
748            <textarea id="message-text" readonly rows="4" class=FIELD>{menu.text.clone()}</textarea>
749            {actions}
750        }
751        .into_any()
752    }
