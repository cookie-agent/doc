194/// The attach and camera buttons and their hidden file inputs, offered
195/// only for the media the selected model accepts.
196#[component]
197pub fn AttachButtons(workspace: Workspace) -> impl IntoView {
198    let media = Memo::new(move |_| {
199        workspace
200            .accepted_media()
201            .map(|accepted| accepted.media)
202            .unwrap_or_default()
203    });
204    let accept = move || accept_attribute(&media.get());
205    let camera_accept = move || {
206        let media = media.get();
207        accept_attribute(
208            &media
209                .into_iter()
210                .filter(|entry| entry.kind.as_str() == "image")
211                .collect::<Vec<_>>(),
212        )
213    };
214    let on_change = move |event: ev::Event| {
215        let Some(input) = event.target().and_then(|target| {
216            wasm_bindgen::JsCast::dyn_into::<web_sys::HtmlInputElement>(target).ok()
217        }) else {
218            return;
219        };
220        let files = files_of(input.files());
221        input.set_value("");
222        workspace.attach_files(files);
223    };
224    view! {
225        <Show when=move || !media.with(Vec::is_empty)>
226            <input
227                type="file"
228                id=FILE_INPUT
229                class="hidden"
230                multiple
231                accept=accept
232                on:change=on_change
233            />
234            <button
235                type="button"
236                id="attach"
237                class=ATTACH_BUTTON
238                aria-label="Attach a file"
239                title="Attach a file"
240                disabled=move || !workspace.writable()
241                on:click=move |_| pick_files(false)
242            >
243                <Glyph icon=icondata::FaPaperclipSolid class="text-base" />
244            </button>
245            <Show when=move || accepts_images(&media.get())>
246                <input
247                    type="file"
248                    id=CAMERA_INPUT
249                    class="hidden"
250                    accept=camera_accept
251                    capture="environment"
252                    on:change=on_change
253                />
254                <button
255                    type="button"
256                    class=format!("{ATTACH_BUTTON} md:hidden")
257                    aria-label="Take a photo"
258                    title="Take a photo"
259                    disabled=move || !workspace.writable()
260                    on:click=move |_| pick_files(true)
261                >
262                    <Glyph icon=icondata::FaCameraSolid class="text-base" />
263                </button>
264            </Show>
265        </Show>
266    }

440/// How long ago a background call was detached, as the TUI says it.
441fn elapsed_label(seconds: i64) -> String {
442    match seconds {
443        ..60 => format!("{}s", seconds.max(0)),
444        60..3600 => format!("{}m{:02}s", seconds / 60, seconds % 60),
445        _ => format!("{}h{:02}m", seconds / 3600, seconds % 3600 / 60),
446    }
447}

449pub fn background_sheet(workspace: Workspace) -> impl IntoView {
450    // The elapsed times tick while the sheet is open.
451    let now = RwSignal::new(jiff::Timestamp::now());
452    let ticker = set_interval_with_handle(
453        move || now.set(jiff::Timestamp::now()),
454        Duration::from_secs(1),
455    )
456    .ok();
457    on_cleanup(move || {
458        if let Some(ticker) = ticker {
459            ticker.clear();
460        }
461    });
462    move || {
463        let calls = workspace.background_calls();
464        if calls.is_empty() {
465            return empty_state(
466                icondata::FaHourglassHalfSolid,
467                "No calls are running in the background.",
468            )
469            .into_any();
470        }
471        calls
472            .into_iter()
473            .map(|call| {
474                let job = call.job.clone();
475                let elapsed = now.get().duration_since(call.detached_at).as_secs();
476                view! {
477                    <div class="background-call flex items-center gap-2 border-b border-toasted py-2 dark:border-ember">
478                        <div class="min-w-0 flex-1">
479                            <p class="truncate font-mono text-sm">{format!("{} · {}", call.job, call.tool)}</p>
480                            <p class=format!("truncate {NOTE}")>
481                                {call.display.clone().unwrap_or_default()}
482                                {format!(" · {}", elapsed_label(elapsed))}
483                            </p>
484                        </div>
485                        // An ACP agent's own command: the agent runs it, and
486                        // only the agent can stop it.
487                        {if call.agent_task {
488                            view! {
489                                <span class=NOTE data-agent-task=call.job.clone()>"run by the agent"</span>
490                            }
491                            .into_any()
492                        } else {
493                            view! {
494                                <button
495                                    type="button"
496                                    class=BUTTON
497                                    data-stop=call.job.clone()
498                                    on:click=move |_| workspace.stop_background(job.clone())
499                                >
500                                    <Glyph icon=icondata::FaCircleStopSolid />
501                                    "Stop"
502                                </button>
503                            }
504                            .into_any()
505                        }}
506                    </div>
507                }
508            })
509            .collect_view()
510            .into_any()
511    }
512}

612/// The new-rule form's fields, which an edited rule fills.
613#[derive(Clone, Copy)]
614struct RuleForm {
615    action: RwSignal<String>,
616    runtime: RwSignal<String>,
617    pattern: RwSignal<String>,
618    effect: RwSignal<PermissionEffect>,
619    /// The session rule being edited, which the form's rule replaces.
620    editing: RwSignal<Option<RuleRef>>,
621}
622
623impl RuleForm {
624    fn new() -> Self {
625        Self {
626            action: RwSignal::new(String::new()),
627            runtime: RwSignal::new(String::new()),
628            pattern: RwSignal::new(String::new()),
629            effect: RwSignal::new(PermissionEffect::Ask),
630            editing: RwSignal::new(None),
631        }
632    }
633
634    fn edit(self, row: &PermissionRow) {
635        self.action.set(row.action.to_string());
636        self.runtime.set(
637            row.runtime
638                .as_ref()
639                .map(ToString::to_string)
640                .unwrap_or_default(),
641        );
642        self.pattern.set(row.resource.clone());
643        self.effect.set(row.effect);
644        self.editing.set(RuleRef::of(row));
645    }
646
647    fn reset(self) {
648        self.action.set(String::new());
649        self.runtime.set(String::new());
650        self.pattern.set(String::new());
651        self.effect.set(PermissionEffect::Ask);
652        self.editing.set(None);
653    }
654
655    /// Why the typed key is refused; none while the action is empty.
656    fn key_error(self) -> Option<String> {
657        let action = self.action.get();
658        if action.trim().is_empty() {
659            return None;
660        }
661        parse_rule_key(&action, &self.runtime.get()).err()
662    }
663
664    fn submit(self, workspace: Workspace) {
665        let (action, runtime) =
666            match parse_rule_key(&self.action.get_untracked(), &self.runtime.get_untracked()) {
667                Ok(key) => key,
668                Err(error) => {
669                    workspace.status.set(error);
670                    return;
671                }
672            };
673        let pattern = self.pattern.get_untracked();
674        if pattern.trim().is_empty() {
675            workspace.status.set("a rule needs a pattern".into());
676            return;
677        }
678        let replacing = self.editing.get_untracked();
679        let effect = self.effect.get_untracked();
680        self.reset();
681        workspace.set_permission(action, runtime, &pattern, effect, replacing);
682    }
683}

685/// A rule's key as the table shows it: the action, then `@runtime` when the
686/// rule names one.
687fn rule_key_view(row: &PermissionRow) -> impl IntoView + use<> {
688    let runtime = row.runtime.as_ref().map(|runtime| {
689        view! { <span class="text-ink dark:text-sand" data-rule-runtime>{format!("@{runtime}")}</span> }
690    });
691    view! {
692        <span class="font-bold">{row.action.to_string()}</span>
693        {runtime}
694    }
695}

697pub fn permissions_sheet(workspace: Workspace) -> impl IntoView {
698    let form = RuleForm::new();
699    let table = move || match workspace.permissions.get() {
700        None => loading("Reading permissions…").into_any(),
701        Some(Err(error)) => view! { <p class=super::ERROR>{error}</p> }.into_any(),
702        Some(Ok(permissions)) => {
703            let editable = workspace.writable();
704            let rows = permission_rows(&permissions)
705                .into_iter()
706                .map(|row| {
707                    let name = format!("{} {}", row.key(), row.resource);
708                    let segments = EFFECTS
709                        .iter()
710                        .map(|effect| {
711                            let effect = *effect;
712                            let action = row.action.clone();
713                            let runtime = row.runtime.clone();
714                            let resource = row.resource.clone();
715                            let active = effect == row.effect;
716                            view! {
717                                <button
718                                    type="button"
719                                    class="min-h-11 min-w-11 px-2 text-xs hover:bg-toasted active:bg-crust disabled:cursor-not-allowed disabled:opacity-50 dark:hover:bg-ember dark:active:bg-smoke"
720                                    class:font-bold=active
721                                    class:bg-toasted=active
722                                    class:dark:bg-ember=active
723                                    aria-pressed=active.to_string()
724                                    data-effect=effect_label(effect)
725                                    disabled=!editable
726                                    on:click=move |_| {
727                                        if !active {
728                                            workspace.set_permission(
729                                                action.clone(),
730                                                runtime.clone(),
731                                                &resource,
732                                                effect,
733                                                None,
734                                            );
735                                        }
736                                    }
737                                >
738                                    {effect_label(effect)}
739                                </button>
740                            }
741                        })
742                        .collect_view();
743                    let session_rule = (row.source == PermissionRuleSource::SessionOverlay && editable)
744                        .then(|| RuleRef::of(&row))
745                        .flatten();
746                    let controls = session_rule.map(|rule| {
747                        let edited = row.clone();
748                        view! {
749                            <button
750                                type="button"
751                                class=ICON_BUTTON
752                                aria-label="Edit the override"
753                                title="Edit the override"
754                                data-edit=name.clone()
755                                on:click=move |_| form.edit(&edited)
756                            >
757                                <Glyph icon=icondata::FaPenSolid />
758                            </button>
759                            <button
760                                type="button"
761                                class=ICON_BUTTON
762                                aria-label="Remove the override"
763                                title="Remove the override"
764                                data-remove=name.clone()
765                                on:click=move |_| workspace.clear_permission(rule.clone())
766                            >
767                                <Glyph icon=icondata::FaXmarkSolid />
768                            </button>
769                        }
770                    });
771                    // Rows wrap rather than scroll: the rule's name takes
772                    // the room left, and its controls go below it when a
773                    // narrow screen or a long name needs.
774                    view! {
775                        <div
776                            class="permission-row flex min-w-0 flex-wrap items-center gap-x-3 gap-y-1 border-b border-toasted py-2 dark:border-ember"
777                            data-rule=name.clone()
778                        >
779                            <span class="min-w-0 flex-1 basis-32 font-mono text-xs wrap-anywhere">
780                                {rule_key_view(&row)}
781                                " "
782                                <span>{row.resource.clone()}</span>
783                            </span>
784                            <span class="flex min-w-0 flex-wrap items-center gap-2">
785                                <span class="inline-flex shrink-0 overflow-hidden rounded-md border border-crust dark:border-smoke" role="group" aria-label="Effect">
786                                    {segments}
787                                </span>
788                                <span class=format!("min-w-14 wrap-anywhere {NOTE}")>{source_label(row.source)}</span>
789                                {controls}
790                            </span>
791                        </div>
792                    }
793                })
794                .collect_view();
795            view! { <div class="flex min-w-0 flex-col">{rows}</div> }.into_any()
796        }
797    };
798    let actions = move || {
799        workspace.permissions.with(|permissions| {
800            permission_actions(
801                permissions
802                    .as_ref()
803                    .and_then(|permissions| permissions.as_ref().ok())
804                    .map(Vec::as_slice),
805            )
806        })
807    };
808    let runtimes = move || {
809        workspace
810            .runtimes
811            .with(|listing| listing.as_ref().map(rule_runtimes).unwrap_or_default())
812    };
813    let takes = move || takes_runtime(&form.action.get());
814    view! {
815        <Show when=move || workspace.permissions_draft()>
816            <p class=NOTE>"Rules for the new session, previewed against its agent; the first message creates the session with them."</p>
817        </Show>
818        {table}
819        <Show when=move || workspace.writable()>
820            <div id="permission-form" class="mt-3 flex flex-col gap-2 border-t border-toasted pt-3 dark:border-ember">
821                <p class="text-sm font-bold">
822                    {move || if form.editing.get().is_some() { "Edit the rule" } else { "New rule" }}
823                </p>
824                <div class="flex min-w-0 items-center gap-2">
825                    <input
826                        type="text"
827                        id="permission-action"
828                        class=format!("{FIELD} w-full min-w-0 flex-1")
829                        list="permission-action-options"
830                        placeholder="action, such as bash"
831                        aria-label="Action"
832                        autocomplete="off"
833                        autocapitalize="off"
834                        spellcheck="false"
835                        aria-invalid=move || form.key_error().is_some().to_string()
836                        prop:value=move || form.action.get()
837                        on:input=move |event| {
838                            let action = event_target_value(&event);
839                            if !takes_runtime(&action) {
840                                form.runtime.set(String::new());
841                            }
842                            form.action.set(action);
843                        }
844                    />
845                    <span class="shrink-0 font-mono text-sm" aria-hidden="true">"@"</span>
846                    <input
847                        type="text"
848                        id="permission-runtime"
849                        class=format!("{FIELD} w-full min-w-0 flex-1 disabled:cursor-not-allowed disabled:opacity-50")
850                        list="permission-runtime-options"
851                        placeholder=move || if takes() { "any runtime" } else { "no runtime" }
852                        aria-label="Runtime (empty for every runtime)"
853                        title=move || if takes() { "The runtime the rule is limited to; empty for every runtime" } else { NO_RUNTIME }
854                        autocomplete="off"
855                        autocapitalize="off"
856                        spellcheck="false"
857                        disabled=move || !takes()
858                        prop:value=move || form.runtime.get()
859                        on:input=move |event| form.runtime.set(event_target_value(&event))
860                    />
861                </div>
862                <datalist id="permission-action-options">
863                    {move || actions().into_iter().map(|action| view! { <option value=action.to_string()></option> }).collect_view()}
864                </datalist>
865                <datalist id="permission-runtime-options">
866                    {move || runtimes().into_iter().map(|runtime| view! { <option value=runtime.to_string()></option> }).collect_view()}
867                </datalist>
868                {move || form.key_error().map(|error| view! { <p class=NOTE data-key-error>{error}</p> })}
869                <input
870                    type="text"
871                    id="permission-pattern"
872                    class=FIELD
873                    placeholder="pattern, such as git *"
874                    aria-label="Pattern"
875                    prop:value=move || form.pattern.get()
876                    on:input=move |event| form.pattern.set(event_target_value(&event))
877                    on:keydown=move |event: ev::KeyboardEvent| {
878                        if event.key() == "Enter" {
879                            event.prevent_default();
880                            form.submit(workspace);
881                        }
882                    }
883                />
884                <div class="inline-flex self-start overflow-hidden rounded-md border border-crust dark:border-smoke" role="group" aria-label="Effect">
885                    {EFFECTS
886                        .iter()
887                        .map(|effect| {
888                            let effect = *effect;
889                            view! {
890                                <button
891                                    type="button"
892                                    class="min-h-11 min-w-11 px-3 text-sm hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke"
893                                    class:font-bold=move || form.effect.get() == effect
894                                    class:bg-toasted=move || form.effect.get() == effect
895                                    class:dark:bg-ember=move || form.effect.get() == effect
896                                    aria-pressed=move || (form.effect.get() == effect).to_string()
897                                    data-new-effect=effect_label(effect)
898                                    on:click=move |_| form.effect.set(effect)
899                                >
900                                    {effect_label(effect)}
901                                </button>
902                            }
903                        })
904                        .collect_view()}
905                </div>
906                <div class="flex flex-wrap gap-2">
907                    <button
908                        type="button"
909                        id="permission-add"
910                        class=PRIMARY
911                        disabled=move || form.key_error().is_some()
912                        on:click=move |_| form.submit(workspace)
913                    >
914                        {move || if form.editing.get().is_some() {
915                            view! { <Glyph icon=icondata::FaCheckSolid />"Save the rule" }.into_any()
916                        } else {
917                            view! { <Glyph icon=icondata::FaPlusSolid />"Add the rule" }.into_any()
918                        }}
919                    </button>
920                    <Show when=move || form.editing.get().is_some()>
921                        <button type="button" id="permission-cancel" class=BUTTON on:click=move |_| form.reset()>
922                            "Cancel"
923                        </button>
924                    </Show>
925                </div>
926            </div>
927        </Show>
928    }
929}
