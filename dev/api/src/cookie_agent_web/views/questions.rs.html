1//! The question an agent asks the user (ACP elicitation), as a card above
2//! the composer: the oldest of the viewed session's waiting questions, who
3//! asks, and its message, then a form to fill in, a page to open, or, for
4//! a kind this client does not know, nothing to fill in. Submit (or Done,
5//! for a page) accepts, Decline says no, and Cancel dismisses it without
6//! choosing. The card leaves once the daemon records the answer.

24/// `url` split around its host, for showing the host highlighted: the part
25/// before it, the host, and the rest. Without a host the whole URL comes
26/// first.
27pub(crate) fn host_parts(url: &str) -> (String, String, String) {
28    let Some(host) = url_host(url) else {
29        return (url.to_owned(), String::new(), String::new());
30    };
31    let start = url.find("://").map_or(0, |index| index + 3);
32    match url[start..].find(host) {
33        Some(offset) => {
34            let at = start + offset;
35            (
36                url[..at].to_owned(),
37                host.to_owned(),
38                url[at + host.len()..].to_owned(),
39            )
40        }
41        None => (url.to_owned(), String::new(), String::new()),
42    }
43}

45/// "1 of 3" for the first of several waiting questions; nothing for one.
46pub(crate) fn queue_label(count: usize) -> Option<String> {
47    (count > 1).then(|| format!("1 of {count}"))
48}

50/// The `type` of the input a text field's format asks for.
51pub(crate) fn text_input_type(format: Option<&str>) -> &'static str {
52    match format {
53        Some("email") => "email",
54        Some("uri") => "url",
55        Some("date") => "date",
56        _ => "text",
57    }
58}

60/// The viewed session's oldest waiting question, if any.
61#[component]
62pub(super) fn QuestionCard(workspace: Workspace) -> impl IntoView {
63    let first = Memo::new(move |_| workspace.questions().into_iter().next());
64    let count = Memo::new(move |_| workspace.questions().len());
65    move || {
66        first
67            .get()
68            .map(|question| question_card(workspace, question, count))
69    }
70}

72fn question_card(
73    workspace: Workspace,
74    question: QuestionState,
75    count: Memo<usize>,
76) -> impl IntoView {
77    let error = RwSignal::new(None::<String>);
78    let sending = RwSignal::new(false);
79    let problem = RwSignal::new(None::<FieldProblem>);
80    let id = question.request.elicitation_id.to_string();
81    let heading = question.heading();
82    let region_label = heading.clone();
83    let message = question.request.message.clone();
84    let stored = StoredValue::new(question.clone());
85    let send = move |action: ElicitationAction, content: Option<ElicitationContent>| {
86        sending.set(true);
87        error.set(None);
88        stored.with_value(|question| {
89            workspace.answer_question(question, action, content, move |failure| {
90                sending.set(false);
91                error.set(failure);
92            });
93        });
94    };
95    let draft = RwSignal::new(match question.kind() {
96        QuestionKind::Form(form) => Some(FormDraft::new(form)),
97        _ => None,
98    });
99    let (body, accept_label) = match question.kind() {
100        QuestionKind::Form(form) => (form_view(&id, form, draft, problem).into_any(), Some("Submit")),
101        QuestionKind::Url(url) => (url_view(url).into_any(), Some("Done")),
102        QuestionKind::Unsupported => (
103            view! {
104                <p class=NOTE>"This client cannot show this kind of question: decline or cancel it."</p>
105            }
106            .into_any(),
107            None,
108        ),
109    };
110    let accept = move |_| {
111        let form = stored.with_value(|question| question.request.form.clone());
112        match form {
113            Some(form) => match draft.with_untracked(|draft| {
114                draft
115                    .as_ref()
116                    .map(|draft| draft.content(&form))
117                    .unwrap_or_else(|| Ok(ElicitationContent::new()))
118            }) {
119                Ok(content) => {
120                    problem.set(None);
121                    send(ElicitationAction::Accept, Some(content));
122                }
123                Err(found) => problem.set(Some(found)),
124            },
125            None => send(ElicitationAction::Accept, None),
126        }
127    };
128    view! {
129        <section
130            id="question"
131            data-question=id
132            role="region"
133            aria-label=region_label
134            class="flex flex-col gap-2 rounded-lg border-2 border-caramel bg-parchment p-3 dark:border-maple dark:bg-char"
135        >
136            <p class="flex items-center gap-2 text-sm font-bold">
137                <Glyph icon=icondata::FaCircleQuestionSolid class="text-caramel dark:text-maple" />
138                <span class="min-w-0 flex-1 break-words" data-question-heading>{heading}</span>
139                {move || queue_label(count.get()).map(|label| view! { <span class=NOTE>{label}</span> })}
140            </p>
141            <p class="text-sm whitespace-pre-wrap break-words" data-question-message>{message}</p>
142            {body}
143            {move || error.get().map(|error| view! { <p class=ERROR role="alert" data-question-error>{error}</p> })}
144            <div class="flex flex-wrap gap-2">
145                {accept_label.map(|label| view! {
146                    <button type="button" class=PRIMARY data-answer="accept" disabled=move || sending.get() on:click=accept>
147                        <Glyph icon=icondata::FaCheckSolid />
148                        {label}
149                    </button>
150                })}
151                <button
152                    type="button"
153                    class=BUTTON
154                    data-answer="decline"
155                    disabled=move || sending.get()
156                    on:click=move |_| send(ElicitationAction::Decline, None)
157                >
158                    <Glyph icon=icondata::FaXmarkSolid />
159                    "Decline"
160                </button>
161                <button
162                    type="button"
163                    class=BUTTON
164                    data-answer="cancel"
165                    title="Dismiss the question without choosing"
166                    disabled=move || sending.get()
167                    on:click=move |_| send(ElicitationAction::Cancel, None)
168                >
169                    "Cancel"
170                </button>
171            </div>
172        </section>
173    }
174}

176/// A URL question: the whole URL with its host picked out, what looks off
177/// about it, and the link, which opens only on the user's click.
178fn url_view(url: &str) -> impl IntoView {
179    let (before, host, after) = host_parts(url);
180    let warnings = url_warnings(url);
181    let host_label = url_host(url).unwrap_or("the page").to_owned();
182    view! {
183        <p class="rounded-md bg-cream px-3 py-2 font-mono text-sm break-all dark:bg-night" data-question-url>
184            <span class="opacity-70">{before}</span>
185            <strong class="text-espresso dark:text-latte" data-question-host>{host}</strong>
186            <span class="opacity-70">{after}</span>
187        </p>
188        {(!warnings.is_empty()).then(|| view! {
189            <ul class="flex flex-col gap-1 text-sm text-red-800 dark:text-red-300" data-question-warnings>
190                {warnings.into_iter().map(|warning| view! {
191                    <li class="flex items-start gap-2">
192                        <Glyph icon=icondata::FaTriangleExclamationSolid />
193                        <span>{format!("Careful: {warning}.")}</span>
194                    </li>
195                }).collect_view()}
196            </ul>
197        })}
198        <p class=NOTE>"Open the page, finish there, then choose Done."</p>
199        <a
200            class=format!("{BUTTON} self-start")
201            href=url.to_owned()
202            target="_blank"
203            rel="noopener noreferrer"
204            data-question-open
205        >
206            <Glyph icon=icondata::FaArrowUpRightFromSquareSolid />
207            {format!("Open {host_label}")}
208        </a>
209    }
210}

212/// A form: one control per field, in the form's order.
213fn form_view(
214    id: &str,
215    form: &ElicitationForm,
216    draft: RwSignal<Option<FormDraft>>,
217    problem: RwSignal<Option<FieldProblem>>,
218) -> impl IntoView {
219    let set = move |index: usize, value: DraftValue| {
220        draft.update(|draft| {
221            if let Some(slot) = draft.as_mut().and_then(|draft| draft.values.get_mut(index)) {
222                *slot = value;
223            }
224        });
225        problem.update(|problem| {
226            if problem
227                .as_ref()
228                .is_some_and(|problem| problem.index == index)
229            {
230                *problem = None;
231            }
232        });
233    };
234    let current = move |index: usize| {
235        draft.with_untracked(|draft| {
236            draft
237                .as_ref()
238                .and_then(|draft| draft.values.get(index).cloned())
239        })
240    };
241    view! {
242        {form.title.clone().map(|title| view! { <p class="text-sm font-bold">{title}</p> })}
243        {form.description.clone().map(|description| view! { <p class=NOTE>{description}</p> })}
244        <div class="flex flex-col gap-3">
245            {form.fields.iter().enumerate().map(|(index, field)| {
246                let control = field_control(id, index, field, current(index), set);
247                field_view(index, field, control, problem)
248            }).collect_view()}
249        </div>
250    }
251}

253fn field_view(
254    index: usize,
255    field: &ElicitationField,
256    control: AnyView,
257    problem: RwSignal<Option<FieldProblem>>,
258) -> impl IntoView {
259    let hint = field_hint(field);
260    view! {
261        <div class="flex flex-col gap-1" data-field=field.name.clone()>
262            <span class="text-sm font-bold">
263                {field.label().to_owned()}
264                {field.required.then(|| view! { <span class="text-caramel dark:text-maple" aria-hidden="true">" *"</span> })}
265            </span>
266            {field.description.clone().map(|description| view! { <span class=NOTE>{description}</span> })}
267            {(!hint.is_empty()).then(|| view! { <span class=NOTE data-field-hint>{hint}</span> })}
268            {control}
269            {move || problem.get().filter(|problem| problem.index == index).map(|problem| view! {
270                <p class=ERROR role="alert" data-field-problem>{problem.message}</p>
271            })}
272        </div>
273    }
274}

276/// The input of field `index`, starting at `value`; `set` records each
277/// change in the draft.
278fn field_control(
279    id: &str,
280    index: usize,
281    field: &ElicitationField,
282    value: Option<DraftValue>,
283    set: impl Fn(usize, DraftValue) + Copy + Send + Sync + 'static,
284) -> AnyView {
285    let label = field.label().to_owned();
286    let group = format!("question-{id}-{index}");
287    match (field.input.known(), value) {
288        (Some(ElicitationInput::Text { max_length, format, .. }), Some(DraftValue::Text(text))) => {
289            let kind = text_input_type(format.as_deref());
290            let placeholder = (format.as_deref() == Some("date-time")).then_some("2026-10-24T10:00:00Z");
291            view! {
292                <input
293                    type=kind
294                    class=FIELD
295                    aria-label=label
296                    maxlength=max_length.map(|max| max.to_string())
297                    placeholder=placeholder
298                    prop:value=text
299                    on:input=move |event| set(index, DraftValue::Text(event_target_value(&event)))
300                />
301            }
302            .into_any()
303        }
304        (
305            Some(ElicitationInput::Number { .. } | ElicitationInput::Integer { .. }),
306            Some(DraftValue::Text(text)),
307        ) => {
308            let decimal = matches!(field.input.known(), Some(ElicitationInput::Number { .. }));
309            view! {
310                <input
311                    type="text"
312                    inputmode=if decimal { "decimal" } else { "numeric" }
313                    class=format!("{FIELD} font-mono")
314                    aria-label=label
315                    prop:value=text
316                    on:input=move |event| set(index, DraftValue::Text(event_target_value(&event)))
317                />
318            }
319            .into_any()
320        }
321        (Some(ElicitationInput::Choice { choices, .. }), Some(DraftValue::Choice(chosen))) => {
322            let none = (!field.required).then(|| {
323                let group = group.clone();
324                view! {
325                    <label class=OPTION>
326                        <input
327                            type="radio"
328                            class="mt-0.5 h-5 w-5 shrink-0"
329                            name=group
330                            prop:checked=chosen.is_none()
331                            on:change=move |_| set(index, DraftValue::Choice(None))
332                        />
333                        <span class="text-sm opacity-70">"No choice"</span>
334                    </label>
335                }
336            });
337            view! {
338                <fieldset class="flex flex-col" aria-label=label>
339                    {choices.iter().enumerate().map(|(choice_index, choice)| {
340                        let group = group.clone();
341                        view! {
342                            <label class=OPTION>
343                                <input
344                                    type="radio"
345                                    class="mt-0.5 h-5 w-5 shrink-0"
346                                    name=group
347                                    data-choice=choice.value.clone()
348                                    prop:checked=chosen == Some(choice_index)
349                                    on:change=move |_| set(index, DraftValue::Choice(Some(choice_index)))
350                                />
351                                <span class="flex min-w-0 flex-col">
352                                    <span class="text-sm break-words">{choice.title.clone()}</span>
353                                    {choice.description.clone().map(|description| view! { <span class=NOTE>{description}</span> })}
354                                </span>
355                            </label>
356                        }
357                    }).collect_view()}
358                    {none}
359                </fieldset>
360            }
361            .into_any()
362        }
363        (
364            Some(ElicitationInput::MultiChoice { choices, .. }),
365            Some(DraftValue::MultiChoice(chosen)),
366        ) => {
367            let picked = RwSignal::new(chosen);
368            view! {
369                <fieldset class="flex flex-col" aria-label=label>
370                    {choices.iter().enumerate().map(|(choice_index, choice)| view! {
371                        <label class=OPTION>
372                            <input
373                                type="checkbox"
374                                class="mt-0.5 h-5 w-5 shrink-0"
375                                data-choice=choice.value.clone()
376                                prop:checked=picked.with_untracked(|picked| picked.get(choice_index).copied().unwrap_or(false))
377                                on:change=move |event| {
378                                    let on = event_target_checked(&event);
379                                    picked.update(|picked| {
380                                        if let Some(slot) = picked.get_mut(choice_index) {
381                                            *slot = on;
382                                        }
383                                    });
384                                    set(index, DraftValue::MultiChoice(picked.get_untracked()));
385                                }
386                            />
387                            <span class="flex min-w-0 flex-col">
388                                <span class="text-sm break-words">{choice.title.clone()}</span>
389                                {choice.description.clone().map(|description| view! { <span class=NOTE>{description}</span> })}
390                            </span>
391                        </label>
392                    }).collect_view()}
393                </fieldset>
394            }
395            .into_any()
396        }
397        (Some(ElicitationInput::Boolean { .. }), Some(DraftValue::Boolean(on))) => view! {
398            <label class=OPTION>
399                <input
400                    type="checkbox"
401                    class="mt-0.5 h-5 w-5 shrink-0"
402                    aria-label=label
403                    prop:checked=on
404                    on:change=move |event| set(index, DraftValue::Boolean(event_target_checked(&event)))
405                />
406                <span class="text-sm">"Yes"</span>
407            </label>
408        }
409        .into_any(),
410        _ => view! { <p class=NOTE data-field-unsupported>"This client cannot fill it in."</p> }.into_any(),
411    }
412}

418    #[test]
419    fn the_host_is_picked_out_of_the_url() {
420        assert_eq!(
421            host_parts("https://user@auth.example.com:8443/connect?id=1"),
422            (
423                "https://user@".to_owned(),
424                "auth.example.com".to_owned(),
425                ":8443/connect?id=1".to_owned()
426            )
427        );
428        assert_eq!(
429            host_parts("mailto:someone"),
430            ("mailto:someone".to_owned(), String::new(), String::new())
431        );
432    }

434    #[test]
435    fn several_questions_say_which_is_shown() {
436        assert_eq!(queue_label(1), None);
437        assert_eq!(queue_label(3).as_deref(), Some("1 of 3"));
438    }
