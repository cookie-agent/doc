1//! The side panel (`docs/site/specs/assets.md` section 8.1): an HTML page
2//! or a Markdown document a shelf's card opened, beside the transcript on a
3//! desktop's width and over it, full screen, on narrower windows. Its head
4//! has the asset's title and handle, *Preview* and *Source*, *Copy handle*,
5//! *Full screen*, *Download*, and *Close* (Escape too). Beside the
6//! transcript, a divider on its edge sets how much of the row it takes.
7
8use leptos::{ev, html, prelude::*};
9
10use crate::{
11    assets::OpenAsset,
12    browser,
13    icons::Glyph,
14    layout::{ASSET_PANEL_SHARE_KEY, asset_panel_share, parse_asset_panel_share},
15    workspace::Workspace,
16};
17
18const BUTTON: &str = "inline-flex min-h-9 min-w-9 shrink-0 items-center justify-center gap-1 rounded-md px-2 text-xs hover:bg-hover active:bg-press";
19const TAB: &str = "inline-flex min-h-8 items-center rounded-md border px-2 text-xs";
20const TAB_ON: &str = "border-fg bg-accent text-on-accent";
21const TAB_OFF: &str = "border-line-2 hover:bg-hover";
22/// How far an arrow key moves the divider, in CSS pixels.
23const DIVIDER_STEP: f64 = 32.0;
24
25/// The panel, open while `workspace.asset_panel` holds an asset. A session
26/// opened in its place closes it.
27#[component]
28pub(crate) fn AssetPanel(workspace: Workspace) -> impl IntoView {
29    Effect::new(
30        move |previous: Option<Option<cookie_agent_protocol::SessionId>>| {
31            let selected = workspace.selected.get();
32            if previous.is_some_and(|previous| previous != selected) {
33                workspace.asset_panel.set(None);
34            }
35            selected
36        },
37    );
38    move || {
39        workspace
40            .asset_panel
41            .get()
42            .map(|open| view! { <Panel workspace open /> })
43    }
44}

46/// The divider between the transcript and the panel: a drag or the arrow
47/// keys set `share`, the part of the row the panel takes, which the browser
48/// remembers, and a double click goes back to the default width. Its parent
49/// `root` is the panel, whose parent is the row.
50#[component]
51fn Divider(
52    root: NodeRef<html::Aside>,
53    share: RwSignal<Option<f64>>,
54    dragging: RwSignal<bool>,
55) -> impl IntoView {
56    let handle = NodeRef::<html::Div>::new();
57    // How far right of the panel's edge the press was, so the edge does not
58    // jump to the pointer.
59    let grab = StoredValue::new(0.0_f64);
60    // Puts the panel's left edge where `edge` says, given where it is.
61    let place = move |edge: &dyn Fn(f64) -> f64| {
62        let Some(panel) = root.get_untracked() else {
63            return;
64        };
65        let Some(row) = panel.parent_element() else {
66            return;
67        };
68        let (row, panel) = (
69            row.get_bounding_client_rect(),
70            panel.get_bounding_client_rect(),
71        );
72        let left = edge(panel.left());
73        if let Some(next) = asset_panel_share(row.width(), row.right() - left) {
74            share.set(Some(next));
75        }
76    };
77    let keep = move || {
78        if let Some(share) = share.get_untracked() {
79            browser::store(ASSET_PANEL_SHARE_KEY, &share.to_string());
80        }
81    };
82    let on_pointerdown = move |event: ev::PointerEvent| {
83        if event.button() != 0 {
84            return;
85        }
86        // No text selection under the drag.
87        event.prevent_default();
88        if let Some(panel) = root.get_untracked() {
89            grab.set_value(f64::from(event.client_x()) - panel.get_bounding_client_rect().left());
90        }
91        // Moves over the transcript or over a page's frame come here too.
92        if let Some(handle) = handle.get_untracked() {
93            let _ = handle.set_pointer_capture(event.pointer_id());
94        }
95        dragging.set(true);
96    };
97    let on_pointermove = move |event: ev::PointerEvent| {
98        if dragging.get_untracked() {
99            let left = f64::from(event.client_x()) - grab.get_value();
100            place(&|_| left);
101        }
102    };
103    let on_pointerup = move |_: ev::PointerEvent| {
104        if dragging.get_untracked() {
105            dragging.set(false);
106            keep();
107        }
108    };
109    let on_keydown = move |event: ev::KeyboardEvent| {
110        let by = match event.key().as_str() {
111            "ArrowLeft" => -DIVIDER_STEP,
112            "ArrowRight" => DIVIDER_STEP,
113            _ => return,
114        };
115        event.prevent_default();
116        place(&|left| left + by);
117        keep();
118    };
119    view! {
120        <div
121            node_ref=handle
122            id="asset-panel-divider"
123            role="separator"
124            aria-orientation="vertical"
125            aria-label="Resize the panel"
126            aria-valuenow=move || share.get().map(|share| (share * 100.0).round().to_string())
127            tabindex="0"
128            title="Drag to resize, double-click to reset"
129            class="group absolute inset-y-0 -left-px z-10 hidden w-2 cursor-col-resize touch-none outline-none lg:block"
130            on:pointerdown=on_pointerdown
131            on:pointermove=on_pointermove
132            on:pointerup=on_pointerup
133            on:pointercancel=on_pointerup
134            on:dblclick=move |_| {
135                share.set(None);
136                browser::forget(ASSET_PANEL_SHARE_KEY);
137            }
138            on:keydown=on_keydown
139        >
140            <div
141                class="h-full w-0.5 group-hover:bg-accent group-focus-visible:bg-accent"
142                class:bg-accent=move || dragging.get()
143            ></div>
144        </div>
145    }
146}

148#[component]
149fn Panel(workspace: Workspace, open: OpenAsset) -> impl IntoView {
150    let source = RwSignal::new(false);
151    let full = RwSignal::new(false);
152    let body = NodeRef::<html::Div>::new();
153    let root = NodeRef::<html::Aside>::new();
154    let share = RwSignal::new(
155        browser::load(ASSET_PANEL_SHARE_KEY).and_then(|stored| parse_asset_panel_share(&stored)),
156    );
157    let dragging = RwSignal::new(false);
158    let close = move || workspace.asset_panel.set(None);
159    {
160        let open = open.clone();
161        Effect::new(move |_| {
162            let source = source.get();
163            let Some(element) = body.get() else {
164                return;
165            };
166            element.set_inner_html(&open.placeholder_html(source));
167            crate::assets::show_within(&element, workspace);
168        });
169    }
170    // The focus goes to the panel, for Escape.
171    Effect::new(move |_| {
172        if let Some(root) = root.get() {
173            let _ = root.focus();
174        }
175    });
176    let has_source = open.has_source();
177    let title = open.title.clone();
178    let label = title.clone();
179    let handle = open.handle.clone();
180    let copy = handle.clone();
181    let (origin, file_title) = (open.source.clone(), open.title.clone());
182    let copyable = !handle.is_empty();
183    let shareable = match &open.source {
184        crate::assets::Source::Asset { store, asset } => Some((store.clone(), *asset)),
185        crate::assets::Source::Artifact { .. } => None,
186    };
187    let share_title = open.title.clone();
188    let tab = move |is_source: bool| {
189        move || {
190            format!(
191                "{TAB} {}",
192                if source.get() == is_source {
193                    TAB_ON
194                } else {
195                    TAB_OFF
196                }
197            )
198        }
199    };
200    let class = move || {
201        if full.get() {
202            "fixed inset-0 z-50 flex flex-col bg-surface-2 outline-none"
203        } else {
204            "fixed inset-0 z-40 flex flex-col bg-surface-2 outline-none lg:relative lg:z-auto lg:w-[clamp(320px,var(--asset-panel-width,min(48rem,48%)),calc(100%-320px))] lg:shrink-0 lg:border-l lg:border-line"
205        }
206    };
207    view! {
208        <aside
209            node_ref=root
210            id="asset-panel"
211            role="dialog"
212            aria-label=label
213            tabindex="-1"
214            class=class
215            style=move || {
216                share
217                    .get()
218                    .map(|share| format!("--asset-panel-width:{:.2}%", share * 100.0))
219                    .unwrap_or_default()
220            }
221            data-full=move || full.get().to_string()
222            on:keydown=move |event: leptos::ev::KeyboardEvent| {
223                if event.key() == "Escape" {
224                    event.prevent_default();
225                    if full.get_untracked() {
226                        full.set(false);
227                    } else {
228                        close();
229                    }
230                }
231            }
232        >
233            <Show when=move || !full.get()>
234                <Divider root share dragging />
235            </Show>
236            <div class="flex min-w-0 shrink-0 flex-wrap items-center gap-1 border-b border-line px-2 py-1.5">
237                <span class="flex min-w-0 flex-1 flex-col px-1">
238                    <span class="truncate text-sm font-bold" id="asset-panel-title">{title}</span>
239                    <span class="truncate font-mono text-[0.6875rem] text-muted select-all" id="asset-panel-handle">
240                        {handle}
241                    </span>
242                </span>
243                <Show when=move || has_source>
244                    <span class="flex shrink-0 gap-1" role="tablist">
245                        <button
246                            type="button"
247                            role="tab"
248                            id="asset-panel-preview"
249                            class=tab(false)
250                            aria-selected=move || (!source.get()).to_string()
251                            on:click=move |_| source.set(false)
252                        >
253                            "Preview"
254                        </button>
255                        <button
256                            type="button"
257                            role="tab"
258                            id="asset-panel-source"
259                            class=tab(true)
260                            aria-selected=move || source.get().to_string()
261                            on:click=move |_| source.set(true)
262                        >
263                            "Source"
264                        </button>
265                    </span>
266                </Show>
267                // A stored file has no handle to copy.
268                <Show when=move || copyable>
269                    <button
270                        type="button"
271                        id="asset-panel-copy"
272                        class=BUTTON
273                        title="Copy the handle"
274                        aria-label="Copy the handle"
275                        on:click={
276                            let copy = copy.clone();
277                            move |_| crate::assets::copy_handle(copy.clone(), workspace)
278                        }
279                    >
280                        <Glyph icon=icondata::LuCopy />
281                    </button>
282                </Show>
283                {shareable.map(|(store, asset)| {
284                    view! {
285                        <Show when=move || workspace.has_shares.get()>
286                            <button
287                                type="button"
288                                id="asset-panel-share"
289                                class=BUTTON
290                                title="Share link"
291                                aria-label="Share link"
292                                on:click={
293                                    let (store, title) = (store.clone(), share_title.clone());
294                                    move |_| workspace.open_share_dialog(store.clone(), asset, title.clone())
295                                }
296                            >
297                                <Glyph icon=icondata::LuShare2 />
298                            </button>
299                        </Show>
300                    }
301                })}
302                <button
303                    type="button"
304                    id="asset-panel-full"
305                    class=BUTTON
306                    title=move || if full.get() { "Beside the chat" } else { "Full screen" }
307                    aria-label=move || if full.get() { "Beside the chat" } else { "Full screen" }
308                    aria-pressed=move || full.get().to_string()
309                    on:click=move |_| full.update(|full| *full = !*full)
310                >
311                    <Glyph icon=Signal::derive(move || {
312                        if full.get() {
313                            icondata::LuMinimize2
314                        } else {
315                            icondata::LuMaximize2
316                        }
317                    }) />
318                </button>
319                <button
320                    type="button"
321                    id="asset-panel-download"
322                    class=BUTTON
323                    title="Download"
324                    aria-label="Download"
325                    on:click=move |_| crate::assets::download_source(workspace, origin.clone(), file_title.clone())
326                >
327                    <Glyph icon=icondata::LuDownload />
328                </button>
329                <button
330                    type="button"
331                    id="asset-panel-close"
332                    class=BUTTON
333                    title="Close (Esc)"
334                    aria-label="Close"
335                    on:click=move |_| close()
336                >
337                    <Glyph icon=icondata::LuX />
338                </button>
339            </div>
340            // A page's frame would otherwise take the pointer from the drag.
341            <div
342                node_ref=body
343                class="flex min-h-0 flex-1 flex-col overflow-auto"
344                class:pointer-events-none=move || dragging.get()
345                id="asset-panel-body"
346            ></div>
347        </aside>
348    }
