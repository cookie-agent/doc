1//! The Skills view: the skills the daemon offers every session (the user's
2//! and the official ones), or with a session open that session's, its
3//! workspace's included; grouped by where they come from, searched by name
4//! and description, and each with what it is for, how to invoke it, and
5//! where it lives. Its button refreshes the official skills now.

17/// A group's icon.
18fn source_icon(source: SkillSource) -> crate::icons::Icon {
19    match source {
20        SkillSource::User => icondata::LuUser,
21        SkillSource::Project => icondata::LuFolder,
22        SkillSource::Official => icondata::LuBadgeCheck,
23    }
24}

26pub fn skills_page(workspace: Workspace) -> impl IntoView {
27    let session_open = move || workspace.selected.get().is_some();
28    // Another session opened: its skills, while the view lists a session's;
29    // none open, the daemon's again.
30    Effect::new(
31        move |previous: Option<Option<cookie_agent_protocol::SessionId>>| {
32            let selected = workspace.selected.get();
33            if previous.is_some_and(|previous| previous != selected)
34                && workspace.skill_list_session.get_untracked()
35            {
36                workspace.list_session_skills(selected.is_some());
37            }
38            selected
39        },
40    );
41    let scope_button = move |session: bool, label: &'static str, id: &'static str| {
42        let chosen = move || workspace.skill_list_session.get() == session;
43        view! {
44            <button
45                type="button"
46                role="radio"
47                class=move || format!(
48                    "inline-flex min-h-11 items-center rounded-full border px-3 text-sm lg:min-h-9 {}",
49                    if chosen() {
50                        "border-fg bg-accent text-on-accent"
51                    } else {
52                        "border-line-2 hover:bg-hover active:bg-press"
53                    },
54                )
55                aria-checked=move || chosen().to_string()
56                id=id
57                on:click=move |_| {
58                    if !chosen() {
59                        workspace.list_session_skills(session);
60                    }
61                }
62            >
63                {label}
64            </button>
65        }
66    };
67    // The daemon's own skills live on its machine: under its home.
68    let home = move || {
69        workspace.runtime_homes.with(|homes| {
70            homes
71                .get_str(RuntimeId::GLOBAL)
72                .or_else(|| homes.get_str(RuntimeId::LOCAL))
73                .map(str::to_owned)
74        })
75    };
76    let list = move || {
77        let query = workspace.skill_list_query.get();
78        match workspace.skill_list.get() {
79            None => loading("Reading the skills…").into_any(),
80            Some(Err(message)) => view! { <p class=ERROR>{message}</p> }.into_any(),
81            Some(Ok(skills)) if skills.is_empty() => {
82                empty_state(icondata::LuBookOpen, "No skills yet.").into_any()
83            }
84            Some(Ok(skills)) => {
85                let groups = skill_groups(&skills, &query);
86                if groups.is_empty() {
87                    return empty_state(icondata::LuSearch, "No skills match.").into_any();
88                }
89                let home = home();
90                groups
91                    .into_iter()
92                    .map(|(source, group)| {
93                        let rows = group
94                            .into_iter()
95                            .map(|skill| skill_row(skill, home.as_deref()))
96                            .collect_view();
97                        view! {
98                            <section class="flex flex-col" data-skill-group=format!("{source:?}").to_lowercase()>
99                                <h2 class="flex items-center gap-2 pt-3 pb-1 text-sm font-bold">
100                                    <Glyph icon=source_icon(source) class="text-accent" />
101                                    {source_label(source)}
102                                </h2>
103                                {rows}
104                            </section>
105                        }
106                    })
107                    .collect_view()
108                    .into_any()
109            }
110        }
111    };
112    view! {
113        <div class="flex flex-col gap-2" data-skills-page>
114            <p class=NOTE>
115                "Skills are instructions an agent loads when a task calls for them; you can run one with /skills in a session."
116            </p>
117            <div class="flex flex-wrap items-center gap-2">
118                <Show when=session_open>
119                    <div class="flex flex-wrap gap-1.5" role="radiogroup" aria-label="Which skills">
120                        {scope_button(false, "Every session's", "skills-daemon")}
121                        {scope_button(true, "This session's", "skills-session")}
122                    </div>
123                </Show>
124                <span class="flex-1"></span>
125                <button
126                    type="button"
127                    class=BUTTON
128                    id="skills-update"
129                    disabled=move || workspace.skills_updating.get()
130                    on:click=move |_| workspace.update_official_skills()
131                >
132                    <span class="inline-flex" class:animate-spin=move || workspace.skills_updating.get()>
133                        <Glyph icon=icondata::LuRefreshCw />
134                    </span>
135                    {move || if workspace.skills_updating.get() { "Updating…" } else { "Update official skills" }}
136                </button>
137            </div>
138            {move || workspace.skills_update_note.get().map(|(text, failed)| {
139                view! { <p class=if failed { ERROR } else { NOTE } data-skills-update-note>{text}</p> }
140            })}
141            <label class="relative flex items-center">
142                <Glyph icon=icondata::LuSearch class="pointer-events-none absolute left-3 text-xs text-muted" />
143                <input
144                    type="text"
145                    role="searchbox"
146                    id="skills-search"
147                    class="min-h-11 w-full min-w-0 rounded-lg bg-surface pr-3 pl-9 text-sm text-fg hover:shadow-[inset_0_0_0_1px_var(--color-faint)] focus:shadow-[inset_0_0_0_1px_var(--color-accent),0_0_0_3px_var(--color-accent-soft)] focus:outline-none shadow-[inset_0_0_0_1px_var(--color-line-2)]"
148                    aria-label="Search skills by name or description"
149                    placeholder="Search by name or description"
150                    prop:value=move || workspace.skill_list_query.get_untracked()
151                    on:input=move |event| workspace.skill_list_query.set(event_target_value(&event))
152                />
153            </label>
154            {list}
155        </div>
156    }
157}

159/// One skill: its name, how it is invoked, and its marks; what it does and
160/// when; where it lives.
161fn skill_row(skill: &SkillDescriptor, home: Option<&str>) -> impl IntoView + use<> {
162    let invocation = match &skill.argument_hint {
163        Some(hint) => format!("/{} {hint}", skill.name),
164        None => format!("/{}", skill.name),
165    };
166    // What sets it apart, beside its name.
167    let mut marks = Vec::new();
168    if let Some(version) = &skill.version {
169        marks.push(("version", format!("v{version}")));
170    }
171    if !skill.precedence_winner {
172        marks.push(("shadowed", "overridden".to_owned()));
173    }
174    if !skill.user_invocable {
175        marks.push(("agent-only", "agent only".to_owned()));
176    }
177    if !skill.visible {
178        marks.push(("hidden", "hidden from the model".to_owned()));
179    }
180    match skill.permission_effect {
181        PermissionEffect::Deny => marks.push(("denied", "denied".to_owned())),
182        PermissionEffect::Ask => marks.push(("asks", "asks first".to_owned())),
183        PermissionEffect::Allow => {}
184    }
185    let marks = marks
186        .into_iter()
187        .map(|(id, text)| view! { <span class=STATE data-skill-mark=id>{text}</span> })
188        .collect_view();
189    let when = skill.when_to_use.clone();
190    let location = display_path(&skill.location, home);
191    view! {
192        <div
193            class="flex flex-col gap-1 border-b border-line py-2"
194            class:opacity-60=!skill.precedence_winner
195            data-skill=skill.name.clone()
196        >
197            <div class="flex flex-wrap items-center gap-2">
198                <span class="min-w-0 truncate font-mono text-sm font-bold" title=invocation.clone()>{invocation.clone()}</span>
199                {marks}
200            </div>
201            <p class="text-sm whitespace-pre-wrap break-words">{skill.description.clone()}</p>
202            {when.map(|when| view! { <p class=format!("{NOTE} whitespace-pre-wrap break-words")>{format!("When: {when}")}</p> })}
203            <span class=format!("{NOTE} font-mono break-all") title=skill.location.clone()>{location}</span>
204        </div>
205    }
