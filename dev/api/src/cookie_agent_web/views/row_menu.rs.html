1//! A row's ⋮ menu: the row's rarer actions as a short list, where more of
2//! them can go. A wide window opens it under the ⋮ (over it near the
3//! bottom of the screen); a phone shows it as a sheet from the bottom, as
4//! the session's sheets are. Escape, a click outside, or an item closes
5//! it, and nothing in it presses the row under it.
6
7use icondata::Icon;
8use leptos::{ev, html, prelude::*};
9use wasm_bindgen::JsCast as _;
10
11use crate::{icons::Glyph, layout::menu_place};

13/// An item's height in a wide window (44 px), for placing the menu.
14const ITEM_HEIGHT: f64 = 44.0;
15/// The menu's padding and border, top and bottom together.
16const MENU_PADDING: f64 = 10.0;
17
18/// One action of a row's menu.
19#[derive(Clone)]
20pub(super) struct MenuItem {
21    /// Its `data-menu-item`.
22    pub id: &'static str,
23    pub label: &'static str,
24    pub icon: Icon,
25    /// What it does; the menu has closed by then.
26    pub act: Callback<()>,
27}
28
29/// A row's ⋮, styled `class`, and the menu of `items` it opens; no ⋮ while
30/// there are none. `open` names the one open menu of the rows sharing it,
31/// `key` this one's; `name` is the ⋮'s `data-row-menu`, and `title` heads
32/// the phone's sheet and names the menu.
33pub(super) fn row_menu<K>(
34    open: RwSignal<Option<K>>,
35    key: K,
36    name: String,
37    title: String,
38    class: &'static str,
39    items: impl Fn() -> Vec<MenuItem> + Send + Sync + Clone + 'static,
40) -> impl IntoView
41where
42    K: Clone + PartialEq + Send + Sync + 'static,
43{
44    let trigger = NodeRef::<html::Button>::new();
45    let is_open = {
46        let key = key.clone();
47        move || open.with(|open| open.as_ref() == Some(&key))
48    };
49    // Where the menu stands in a wide window: CSS variables, which a
50    // phone's sheet ignores.
51    let place = RwSignal::new(String::new());
52    let items = StoredValue::new(items);
53    let listed = move || items.with_value(|items| items());
54    let title = StoredValue::new(title);
55    let toggle = {
56        let key = key.clone();
57        let is_open = is_open.clone();
58        move |_| {
59            if is_open() {
60                open.set(None);
61                return;
62            }
63            let Some(button) = trigger.get_untracked() else {
64                return;
65            };
66            let frame = button.get_bounding_client_rect();
67            let window = window();
68            let size = |value: Result<wasm_bindgen::JsValue, _>| {
69                value.ok().and_then(|value| value.as_f64()).unwrap_or(0.0)
70            };
71            let viewport = (size(window.inner_width()), size(window.inner_height()));
72            #[allow(clippy::cast_precision_loss)]
73            let height = listed().len() as f64 * ITEM_HEIGHT + MENU_PADDING;
74            let at = menu_place(
75                (frame.top(), frame.bottom(), frame.right()),
76                viewport,
77                height,
78            );
79            let px = |value: Option<f64>| {
80                value.map_or_else(|| "auto".to_owned(), |value| format!("{value}px"))
81            };
82            place.set(format!(
83                "--menu-top: {}; --menu-bottom: {}; --menu-right: {}px",
84                px(at.top),
85                px(at.bottom),
86                at.right,
87            ));
88            open.set(Some(key.clone()));
89        }
90    };
91    let close = move |refocus: bool| {
92        open.set(None);
93        if refocus && let Some(button) = trigger.get_untracked() {
94            let _ = button.focus();
95        }
96    };
97    let has_items = move || !listed().is_empty();
98    view! {
99        <Show when=has_items>
100            <button
101                type="button"
102                class=class
103                aria-label="More actions"
104                title="More actions"
105                aria-haspopup="menu"
106                aria-expanded={
107                    let is_open = is_open.clone();
108                    move || is_open().to_string()
109                }
110                data-row-menu=name.clone()
111                node_ref=trigger
112                on:click=toggle.clone()
113            >
114                <Glyph icon=icondata::FaEllipsisVerticalSolid />
115            </button>
116            <Show when=is_open.clone()>
117                {move || menu(listed(), title.get_value(), place, close)}
118            </Show>
119        </Show>
120    }
121}

123/// The open menu over a backdrop that closes it.
124fn menu(
125    items: Vec<MenuItem>,
126    title: String,
127    place: RwSignal<String>,
128    close: impl Fn(bool) + Copy + Send + Sync + 'static,
129) -> impl IntoView {
130    let list = NodeRef::<html::Div>::new();
131    let heading = title.clone();
132    // Escape closes it wherever the focus is; the keyboard starts on the
133    // first item.
134    let escape = window_event_listener(ev::keydown, move |event| {
135        if event.key() == "Escape" {
136            event.prevent_default();
137            close(true);
138        }
139    });
140    let escape = StoredValue::new_local(Some(escape));
141    on_cleanup(move || {
142        if let Some(escape) = escape.try_update_value(Option::take).flatten() {
143            escape.remove();
144        }
145    });
146    Effect::new(move |_| {
147        if let Some(first) = list
148            .get()
149            .and_then(|list| menu_items(&list).into_iter().next())
150        {
151            let _ = first.focus();
152        }
153    });
154    // Up and down move between the items; Tab leaves the menu for the ⋮.
155    let on_keydown = move |event: ev::KeyboardEvent| {
156        let Some(list) = list.get_untracked() else {
157            return;
158        };
159        let items = menu_items(&list);
160        let focused = document().active_element();
161        let at = items
162            .iter()
163            .position(|item| focused.as_ref() == Some(item.unchecked_ref()));
164        let next = match event.key().as_str() {
165            "ArrowDown" => at.map_or(0, |at| (at + 1) % items.len()),
166            "ArrowUp" => at.map_or(items.len() - 1, |at| (at + items.len() - 1) % items.len()),
167            "Home" => 0,
168            "End" => items.len() - 1,
169            "Tab" => {
170                event.prevent_default();
171                close(true);
172                return;
173            }
174            _ => return,
175        };
176        event.prevent_default();
177        if let Some(item) = items.get(next) {
178            let _ = item.focus();
179        }
180    };
181    view! {
182        <div
183            class="fixed inset-0 z-50 compact:bg-espresso/30 compact:dark:bg-night/60"
184            data-row-menu-backdrop
185            on:click=move |_| close(false)
186            // A wheel would scroll the list from under the menu.
187            on:wheel=move |_| close(false)
188        ></div>
189        <div
190            role="menu"
191            aria-label=title
192            class="fixed top-(--menu-top) right-(--menu-right) bottom-(--menu-bottom) z-50 flex min-w-48 flex-col rounded-lg border border-toasted bg-cream p-1 text-espresso shadow-lg dark:border-ember dark:bg-night dark:text-latte compact:top-auto compact:right-0 compact:bottom-0 compact:left-0 compact:min-w-0 compact:rounded-none compact:rounded-t-2xl compact:border-x-0 compact:border-b-0 compact:p-2 compact:pb-[max(0.5rem,env(safe-area-inset-bottom))]"
193            style=move || place.get()
194            data-row-menu-list
195            node_ref=list
196            on:keydown=on_keydown
197        >
198            <span class="mx-auto mb-1 hidden h-1 w-10 rounded-full bg-crust compact:block dark:bg-smoke" aria-hidden="true"></span>
199            <p class="hidden truncate px-3 pt-1 pb-2 text-sm font-bold compact:block" aria-hidden="true">{heading}</p>
200            {items
201                .into_iter()
202                .map(|item| {
203                    let act = item.act;
204                    view! {
205                        <button
206                            type="button"
207                            role="menuitem"
208                            class="flex min-h-11 w-full items-center gap-3 rounded-md px-3 text-left text-sm whitespace-nowrap hover:bg-toasted focus-visible:bg-toasted active:bg-crust dark:hover:bg-ember dark:focus-visible:bg-ember dark:active:bg-smoke compact:min-h-12"
209                            data-menu-item=item.id
210                            on:click=move |_| {
211                                close(false);
212                                act.run(());
213                            }
214                        >
215                            <Glyph icon=item.icon class="w-5 justify-center text-caramel dark:text-maple" />
216                            <span>{item.label}</span>
217                        </button>
218                    }
219                })
220                .collect_view()}
221        </div>
222    }
223}

225/// The menu's items, in order.
226fn menu_items(list: &web_sys::HtmlDivElement) -> Vec<web_sys::HtmlElement> {
227    let Ok(found) = list.query_selector_all("[role=menuitem]") else {
228        return Vec::new();
229    };
230    (0..found.length())
231        .filter_map(|index| found.item(index))
232        .filter_map(|node| node.dyn_into::<web_sys::HtmlElement>().ok())
233        .collect()
234}
