11/// The control for `levels`, showing `current`; choosing a level calls
12/// `on_choose` with its ID. A drag moves the pill and chooses where it is
13/// let go; a click or a key chooses at once. `large` is the phone's
14/// finger-sized pill.
15#[component]
16pub fn EffortControl(
17    levels: Levels,
18    current: Option<String>,
19    on_choose: Callback<String>,
20    #[prop(optional)] large: bool,
21) -> impl IntoView {
22    let form = levels.form();
23    if form == Form::Hidden {
24        return ().into_any();
25    }
26    let initial = current.as_deref().and_then(|id| levels.position(id));
27    let default = levels.default.as_deref().and_then(|id| levels.position(id));
28    let levels = StoredValue::new(levels);
29    let count = levels.with_value(|levels| levels.levels.len());
30    // The level shown: it follows a drag before the drag chooses it.
31    let shown = RwSignal::new(initial);
32    let chosen = StoredValue::new(initial);
33    let choose = move |index: usize| {
34        shown.set(Some(index));
35        if chosen.get_value() == Some(index) {
36            return;
37        }
38        chosen.set_value(Some(index));
39        if let Some(id) =
40            levels.with_value(|levels| levels.levels.get(index).map(|level| level.id.clone()))
41        {
42            on_choose.run(id);
43        }
44    };
45    let label = move |index: usize| levels.with_value(|levels| levels.levels[index].label.clone());
46    let header = move || {
47        let name = levels.with_value(|levels| levels.name);
48        let value = shown.get().map(label);
49        let default_label = default.map(label);
50        let reset = default.filter(|default| shown.get() != Some(*default));
51        view! {
52            <div class="flex items-baseline gap-2 px-1">
53                <span class="text-xs font-semibold text-muted">{name}</span>
54                {value.map(|value| view! { <span class="effort-value text-[0.8125rem] font-semibold" data-effort-value>{value}</span> })}
55                {default_label.map(|label| view! { <span class="text-[0.6875rem] text-faint">{format!("default {label}")}</span> })}
56                {reset.map(|default| view! {
57                    <button
58                        type="button"
59                        class="ml-auto rounded px-1 text-xs text-accent hover:bg-hover"
60                        data-effort-reset
61                        on:click=move |_| choose(default)
62                    >
63                        "Reset"
64                    </button>
65                })}
66            </div>
67        }
68    };
69    let description = move || {
70        shown
71            .get()
72            .and_then(|index| levels.with_value(|levels| levels.levels[index].description.clone()))
73    };
74    let scale = levels.with_value(|levels| levels.scale);
75    // How warm the pill is: from the accent at the middle level to the hot
76    // colour at the top, on a scale only.
77    let heat = move || match (scale, shown.get()) {
78        (true, Some(index)) if count > 1 => {
79            let position = index as f64 / (count - 1) as f64;
80            ((position - 0.5) * 2.0).max(0.0)
81        }
82        _ => 0.0,
83    };
84    let body = match form {
85        Form::Switch => {
86            let on = move || shown.get() == Some(1);
87            view! {
88                <button
89                    type="button"
90                    role="switch"
91                    class="flex min-h-9 items-center gap-2.5 rounded-lg px-1 text-left text-[0.8125rem] font-semibold"
92                    aria-checked=move || on().to_string()
93                    data-effort-switch
94                    on:click=move |_| choose(usize::from(!on()))
95                >
96                    <span class="effort-switch" class:effort-switch-on=on aria-hidden="true"></span>
97                    {move || shown.get().map(label)}
98                </button>
99            }
100            .into_any()
101        }
102        Form::List => view! {
103            <div class="flex flex-col" role="listbox" aria-label=move || levels.with_value(|levels| levels.name)>
104                {(0..count).map(|index| view! {
105                    <button
106                        type="button"
107                        role="option"
108                        class="flex min-h-9 items-center gap-2 rounded-lg px-2 text-left text-[0.8125rem] hover:bg-hover"
109                        aria-selected=move || (shown.get() == Some(index)).to_string()
110                        data-effort-level=move || levels.with_value(|levels| levels.levels[index].id.clone())
111                        on:click=move |_| choose(index)
112                    >
113                        <span class="min-w-0 flex-1 truncate">{label(index)}</span>
114                        <Show when=move || shown.get() == Some(index)>
115                            <Glyph icon=icondata::LuCheck class="text-accent" />
116                        </Show>
117                    </button>
118                }).collect_view()}
119            </div>
120        }
121        .into_any(),
122        Form::Pill { short } => pill(levels, count, short, large, shown, choose).into_any(),
123        Form::Hidden => ().into_any(),
124    };
125    view! {
126        <div
127            class="effort flex flex-col gap-1.5"
128            class:effort-large=large
129            style=move || format!("--effort-heat: {:.3}", heat())
130            data-effort=move || levels.with_value(|levels| levels.name)
131        >
132            {header}
133            {body}
134            {move || description().map(|text| view! { <p class="px-1 text-xs text-muted" data-effort-description>{text}</p> })}
135            {scale.then(|| view! {
136                <div class="flex justify-between px-1 text-[0.6875rem] text-faint" aria-hidden="true">
137                    <span>"Faster · cheaper"</span>
138                    <span>"Deeper · slower"</span>
139                </div>
140            })}
141        </div>
142    }
143    .into_any()
144}

146/// The sliding pill: one segment per level, a raised pill under the shown
147/// one. It is one slider to assistive technology.
148fn pill(
149    levels: StoredValue<Levels>,
150    count: usize,
151    short: bool,
152    large: bool,
153    shown: RwSignal<Option<usize>>,
154    choose: impl Fn(usize) + Copy + Send + Sync + 'static,
155) -> impl IntoView {
156    let rail = NodeRef::<leptos::html::Div>::new();
157    let dragging = RwSignal::new(false);
158    // The segment under a pointer at `x`, in the rail's own coordinates.
159    let at = move |x: f64| -> Option<usize> {
160        let rail = rail.get_untracked()?;
161        let frame = rail.get_bounding_client_rect();
162        if frame.width() <= 0.0 {
163            return None;
164        }
165        let fraction = ((x - frame.left()) / frame.width()).clamp(0.0, 0.999_9);
166        Some((fraction * count as f64) as usize)
167    };
168    let tick = move |index: usize| {
169        if shown.get_untracked() != Some(index) {
170            shown.set(Some(index));
171            // A short buzz at each new segment, where a phone offers one.
172            let _ = window().navigator().vibrate_with_duration(4);
173        }
174    };
175    let on_pointerdown = move |event: ev::PointerEvent| {
176        if event.button() != 0 {
177            return;
178        }
179        if let Some(rail) = rail.get_untracked() {
180            let _ = rail.set_pointer_capture(event.pointer_id());
181            let _ = rail.focus();
182        }
183        dragging.set(true);
184        if let Some(index) = at(f64::from(event.client_x())) {
185            tick(index);
186        }
187    };
188    let on_pointermove = move |event: ev::PointerEvent| {
189        if dragging.get_untracked()
190            && let Some(index) = at(f64::from(event.client_x()))
191        {
192            tick(index);
193        }
194    };
195    let release = move |_: ev::PointerEvent| {
196        if dragging.get_untracked() {
197            dragging.set(false);
198            if let Some(index) = shown.get_untracked() {
199                choose(index);
200            }
201        }
202    };
203    let on_keydown = move |event: ev::KeyboardEvent| {
204        let current = shown.get_untracked();
205        let next = match event.key().as_str() {
206            "ArrowLeft" | "ArrowDown" => current.map_or(0, |index| index.saturating_sub(1)),
207            "ArrowRight" | "ArrowUp" => current.map_or(0, |index| (index + 1).min(count - 1)),
208            "Home" => 0,
209            "End" => count - 1,
210            _ => return,
211        };
212        event.prevent_default();
213        choose(next);
214    };
215    let name = levels.with_value(|levels| levels.name);
216    view! {
217        <div
218            node_ref=rail
219            class="effort-rail"
220            class:effort-dragging=move || dragging.get()
221            class:effort-rail-large=large
222            role="slider"
223            tabindex="0"
224            aria-label=name
225            aria-valuemin="0"
226            aria-valuemax=(count - 1).to_string()
227            aria-valuenow=move || shown.get().map(|index| index.to_string())
228            aria-valuetext=move || shown.get().map(|index| levels.with_value(|levels| levels.levels[index].label.clone()))
229            style=move || format!(
230                "--effort-count: {count}; --effort-at: {}",
231                shown.get().unwrap_or_default(),
232            )
233            data-effort-pill
234            on:pointerdown=on_pointerdown
235            on:pointermove=on_pointermove
236            on:pointerup=release
237            on:pointercancel=release
238            on:keydown=on_keydown
239        >
240            <Show when=move || shown.get().is_some()>
241                <span class="effort-knob" aria-hidden="true"></span>
242            </Show>
243            {(0..count).map(|index| {
244                let (whole, id) = levels.with_value(|levels| {
245                    let level = &levels.levels[index];
246                    (level.label.clone(), level.id.clone())
247                });
248                let text = if short { short_label(&whole) } else { whole.clone() };
249                view! {
250                    <span
251                        class="effort-segment"
252                        class:effort-segment-on=move || shown.get() == Some(index)
253                        title=whole
254                        data-effort-level=id
255                    >
256                        {text}
257                    </span>
258                }
259            }).collect_view()}
260        </div>
261    }
262}
