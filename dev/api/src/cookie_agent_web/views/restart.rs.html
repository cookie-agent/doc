1//! Restarting the daemon: the banner while its configuration waits for a
2//! restart or while a restart waits, and the confirmation that offers a
3//! restart when idle or now. The calls are `crate::workspace::restart`.

5use cookie_agent_client_state::restart::{
6    RESTART_WHEN_IDLE, restart_explanation, restart_now_label, running_turns,

8use cookie_agent_protocol::RestartWhen;
9use leptos::prelude::*;
10
11use super::{BUTTON, PRIMARY};

14/// "Restart needed to apply: plugins · Restart daemon", or, once a restart
15/// is scheduled, "Restarting when idle…".
16#[component]
17pub(crate) fn RestartBanner(workspace: Workspace) -> impl IntoView {
18    let restarting = move || workspace.restarting.get();
19    let needed = move || workspace.restart_banner();
20    view! {
21        {move || {
22            if let Some(text) = restarting() {
23                return Some(
24                    view! {
25                        <div
26                            id="restart-banner"
27                            role="status"
28                            data-restarting
29                            class="flex shrink-0 items-center gap-2 border-b border-caramel bg-toasted px-3 py-2 text-sm dark:border-maple dark:bg-ember"
30                        >
31                            <Glyph icon=icondata::FaArrowsRotateSolid class="animate-spin text-caramel dark:text-maple" />
32                            <span class="min-w-0 flex-1">{text}</span>
33                        </div>
34                    }
35                    .into_any(),
36                );
37            }
38            needed().map(|text| {
39                view! {
40                    <div
41                        id="restart-banner"
42                        role="region"
43                        aria-label="Restart needed"
44                        class="flex shrink-0 flex-wrap items-center gap-2 border-b border-toasted bg-parchment py-1 pr-1 pl-3 text-sm dark:border-ember dark:bg-char"
45                    >
46                        <Glyph icon=icondata::FaTriangleExclamationSolid class="text-caramel dark:text-maple" />
47                        <span class="min-w-0 flex-1 py-2" id="restart-needed">{text}</span>
48                        <button
49                            type="button"
50                            id="restart-banner-button"
51                            class=BUTTON
52                            on:click=move |_| workspace.ask_restart()
53                        >
54                            "Restart daemon"
55                        </button>
56                    </div>
57                }
58                .into_any()
59            })
60        }}
61    }
62}

64/// The confirmation: when idle (the default) or now, naming the turns a
65/// restart now interrupts. A sheet at the bottom of a phone, a centered
66/// dialog elsewhere.
67#[component]
68pub(crate) fn RestartDialog(workspace: Workspace) -> impl IntoView {
69    let running = move || workspace.sessions.with(|sessions| running_turns(sessions));
70    let close = move || workspace.restart_confirm.set(false);
71    view! {
72        <Show when=move || workspace.restart_confirm.get()>
73            <div class="fixed inset-0 z-40 bg-espresso/30 dark:bg-black/50" on:click=move |_| close()></div>
74            <div
75                id="restart-dialog"
76                role="alertdialog"
77                aria-modal="true"
78                aria-labelledby="restart-title"
79                class="fixed inset-x-0 bottom-0 z-50 flex max-h-[85vh] flex-col gap-3 overflow-y-auto rounded-t-xl border border-toasted bg-cream p-4 pb-[max(1rem,env(safe-area-inset-bottom))] text-sm shadow-lg sm:inset-x-auto sm:top-1/2 sm:bottom-auto sm:left-1/2 sm:w-[28rem] sm:max-w-[calc(100vw-2rem)] sm:-translate-x-1/2 sm:-translate-y-1/2 sm:rounded-xl dark:border-ember dark:bg-night"
80                on:keydown=move |event| {
81                    if event.key() == "Escape" {
82                        close();
83                    }
84                }
85            >
86                <p id="restart-title" class="font-bold">"Restart the daemon?"</p>
87                <p class="text-ink dark:text-sand" id="restart-explanation">{move || restart_explanation(running())}</p>
88                <div class="flex flex-col gap-2 sm:flex-row-reverse sm:flex-wrap">
89                    <button
90                        type="button"
91                        id="restart-idle"
92                        class=PRIMARY
93                        autofocus
94                        on:click=move |_| workspace.restart(RestartWhen::Idle)
95                    >
96                        {RESTART_WHEN_IDLE}
97                    </button>
98                    <button
99                        type="button"
100                        id="restart-now"
101                        class=BUTTON
102                        on:click=move |_| workspace.restart(RestartWhen::Now)
103                    >
104                        {move || restart_now_label(running())}
105                    </button>
106                    <button type="button" id="restart-cancel" class=BUTTON on:click=move |_| close()>
107                        "Cancel"
108                    </button>
109                </div>
110            </div>
111        </Show>
112    }
