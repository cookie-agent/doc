1//! The GitHub star card (`docs/site/specs/github-star.md` section 2): over
2//! the transcript's bottom right corner, the width of the transcript on a
3//! phone. [`crate::workspace::StarPhase`] says what it shows.
4
5use cookie_agent_protocol::{STAR_REPOSITORY_URL, StarAnswer};
6use leptos::prelude::*;
7
8use super::{BUTTON, ICON_BUTTON, PRIMARY};
9use crate::{
10    icons::Glyph,
11    workspace::{StarPhase, Workspace},
12};
13
14/// **Star on GitHub ↗**: the repository's page in a new tab.
15fn page_link(workspace: Workspace) -> impl IntoView {
16    view! {
17        <a
18            data-star-open
19            class=format!("{PRIMARY} flex-1")
20            href=STAR_REPOSITORY_URL
21            target="_blank"
22            rel="noopener noreferrer"
23            on:click=move |_| workspace.answer_star(StarAnswer::Opened)
24        >
25            <Glyph icon=icondata::LuStar />
26            "Star on GitHub"
27            <Glyph icon=icondata::LuExternalLink />
28        </a>
29    }
30}
31
32#[component]
33pub(crate) fn StarCard(workspace: Workspace) -> impl IntoView {
34    let phase = move || workspace.star.get();
35    let body = move || match phase() {
36        StarPhase::Hidden => ().into_any(),
37        StarPhase::Ask { login } => {
38            let star = match login {
39                Some(login) => view! {
40                    <button
41                        type="button"
42                        data-star-star
43                        class=format!("{PRIMARY} flex-1")
44                        on:click=move |_| workspace.answer_star(StarAnswer::Star)
45                    >
46                        <Glyph icon=icondata::LuStar />
47                        {format!("Star as @{login}")}
48                    </button>
49                }
50                .into_any(),
51                None => page_link(workspace).into_any(),
52            };
53            view! {
54                <button
55                    type="button"
56                    data-star-later
57                    class=format!("{ICON_BUTTON} absolute top-1 right-1")
58                    aria-label="Later"
59                    title="Later"
60                    on:click=move |_| workspace.answer_star(StarAnswer::Later)
61                >
62                    <Glyph icon=icondata::LuX />
63                </button>
64                <p id="star-title" class="flex items-center gap-2 pr-10 font-semibold">
65                    <Glyph icon=icondata::LuStar class="text-accent" />
66                    "Enjoying Cookie Agent?"
67                </p>
68                <p class="text-muted">"A star on GitHub helps others find it."</p>
69                <div class="flex flex-wrap gap-1.5">
70                    {star}
71                    <button
72                        type="button"
73                        data-star-later
74                        class=BUTTON
75                        on:click=move |_| workspace.answer_star(StarAnswer::Later)
76                    >
77                        "Later"
78                    </button>
79                </div>
80                <div class="-mt-1 -mr-1.5 -mb-2 flex justify-end">
81                    <button
82                        type="button"
83                        data-star-dismiss
84                        class="inline-flex min-h-9 items-center rounded-md px-2 text-xs text-muted hover:bg-hover hover:text-fg active:bg-press"
85                        on:click=move |_| workspace.answer_star(StarAnswer::Dismiss)
86                    >
87                        "Don't remind me"
88                    </button>
89                </div>
90            }
91            .into_any()
92        }
93        StarPhase::Starring { login } => view! {
94            <p id="star-title" class="flex items-center gap-2 font-semibold">
95                <Glyph icon=icondata::LuStar class="text-accent" />
96                "Enjoying Cookie Agent?"
97            </p>
98            <p class="text-muted">"A star on GitHub helps others find it."</p>
99            <button type="button" class=PRIMARY disabled>
100                <Glyph icon=icondata::LuLoaderCircle class="animate-spin" />
101                {format!("Starring as @{login}…")}
102            </button>
103        }
104        .into_any(),
105        StarPhase::Thanked { opened } => view! {
106            <p id="star-title" class="flex items-center gap-2 font-semibold" data-star-thanked>
107                <Glyph icon=icondata::LuStar class="text-ok [&_svg]:fill-current" />
108                {if opened { "Thanks!" } else { "Thanks for the star!" }}
109            </p>
110            {opened.then(|| view! { <p class="text-muted">"GitHub opened in a new tab."</p> })}
111        }
112        .into_any(),
113        StarPhase::Failed { login } => view! {
114            <p id="star-title" class="flex items-center gap-2 font-semibold" data-star-failed>
115                <Glyph icon=icondata::LuCircleAlert class="text-accent" />
116                {format!("Couldn't star as @{login}")}
117            </p>
118            <p class="text-muted">"Star it on GitHub instead?"</p>
119            <div class="flex flex-wrap gap-1.5">
120                {page_link(workspace)}
121                <button type="button" class=BUTTON on:click=move |_| workspace.close_star()>
122                    "Close"
123                </button>
124            </div>
125        }
126        .into_any(),
127    };
128    view! {
129        <Show when=move || phase() != StarPhase::Hidden>
130            <div
131                id="star-card"
132                role="dialog"
133                aria-labelledby="star-title"
134                class="absolute inset-x-2 bottom-3 z-20 flex flex-col gap-2.5 rounded-xl bg-raised p-3.5 text-sm ring-1 ring-line shadow-menu sm:inset-x-auto sm:right-4 sm:w-80"
135            >
136                {body}
137            </div>
138        </Show>
139    }
140}
