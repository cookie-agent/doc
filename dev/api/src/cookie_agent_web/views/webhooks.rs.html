61pub fn webhooks_page(workspace: Workspace) -> impl IntoView {
62    let confirm = RwSignal::new(None::<(WebhookId, Confirm)>);
63    let menu = RwSignal::new(None::<WebhookId>);
64    let context = RwSignal::new(None::<(WebhookId, f64, f64)>);
65    // The form renders once per opening: its fields keep what is typed.
66    let form = Memo::new(move |_| workspace.webhook_form.get());
67    move || {
68        if let Some(form) = form.get() {
69            return super::webhook_form::webhook_form(workspace, form).into_any();
70        }
71        let listing = match workspace.webhooks.get() {
72            None => return loading("Reading webhooks…").into_any(),
73            Some(Err(error)) => return view! { <p class=ERROR>{error}</p> }.into_any(),
74            Some(Ok(listing)) => listing,
75        };
76        let receiving = listing.receiving;
77        if let Some(webhook) = workspace.webhook_selected.get().and_then(|id| {
78            listing
79                .webhooks
80                .iter()
81                .find(|webhook| webhook.id == id)
82                .cloned()
83        }) {
84            return webhook_page(workspace, webhook, receiving, confirm, menu).into_any();
85        }
86        let rows = if listing.webhooks.is_empty() {
87            empty_state(
88                icondata::LuWebhook,
89                "No webhooks. Add one here, or from a session's ⋮ menu.",
90            )
91            .into_any()
92        } else {
93            view! {
94                <div class="flex flex-col gap-2" data-webhook-list>
95                    {listing
96                        .webhooks
97                        .into_iter()
98                        .map(|webhook| webhook_row(workspace, webhook, confirm, menu, context))
99                        .collect_view()}
100                </div>
101            }
102            .into_any()
103        };
104        view! {
105            {super::view_header(
106                "Webhooks",
107                "URLs that turn each request they receive into a message in a session, or start one.",
108                view! {
109                    <button type="button" id="webhook-new" class=PRIMARY on:click=move |_| workspace.new_webhook()>
110                        <Glyph icon=icondata::LuPlus />
111                        "New webhook"
112                    </button>
113                },
114            )}
115            {notices(workspace, receiving)}
116            {rows}
117        }
118        .into_any()
119    }
120}

226/// A row's health line: the dot, the last event and when, and the last
227/// refusal with its reason (H1, W2, W3).
228fn health_line(health: Health, now: Timestamp) -> impl IntoView {
229    let Health {
230        tone,
231        last_event,
232        refusal,
233    } = health;
234    let tone_class = if tone == Tone::Bad {
235        "text-bad"
236    } else {
237        "text-muted"
238    };
239    view! {
240        <span class="flex min-w-0 flex-wrap items-center gap-x-3 gap-y-0.5 text-xs" data-webhook-health data-tone=format!("{tone:?}").to_lowercase()>
241            <span class=format!("inline-flex min-w-0 items-center gap-1.5 {tone_class}")>
242                <span class=format!("h-2 w-2 shrink-0 rounded-full {}", dot(tone)) aria-hidden="true"></span>
243                {match &last_event {
244                    Some((at, outcome)) => view! {
245                        <span data-webhook-last title=exact_time(*at)>
246                            {format!("Last event {} · {outcome}", relative_time(now, *at))}
247                        </span>
248                    }
249                    .into_any(),
250                    None => view! { <span data-webhook-last>"No events yet"</span> }.into_any(),
251                }}
252            </span>
253            {refusal.map(|refusal| view! {
254                <span class="min-w-0 text-warn" data-webhook-refusal title=exact_time(refusal.at)>
255                    {format!(
256                        "{} {} {}: {}",
257                        refusal.count,
258                        refusal.what,
259                        relative_time(now, refusal.at),
260                        refusal.reason
261                    )}
262                </span>
263            })}
264        </span>
265    }

290/// A webhook's actions, for its ⋯ and a right-click: Edit, a test event,
291/// New URL, New secret (with a check), and Delete; the last three ask on
292/// its page first.
293fn menu_items(
294    workspace: Workspace,
295    webhook: &Webhook,
296    confirm: RwSignal<Option<(WebhookId, Confirm)>>,
297) -> impl Fn() -> Vec<MenuItem> + Send + Sync + Clone + 'static + use<> {
298    let webhook = webhook.clone();
299    move || {
300        let ask = |what: Confirm| {
301            let id = webhook.id.clone();
302            Callback::new(move |()| {
303                // The confirmation is on the webhook's page.
304                if workspace.webhook_selected.get_untracked() != Some(id.clone()) {
305                    workspace.show_webhook(id.clone());
306                }
307                confirm.set(Some((id.clone(), what)));
308            })
309        };
310        let (edited, tested) = (webhook.clone(), webhook.id.clone());
311        let mut items = vec![
312            MenuItem {
313                id: "edit",
314                label: "Edit",
315                icon: icondata::LuPencil,
316                act: Callback::new(move |()| workspace.edit_webhook(edited.clone())),
317                danger: false,
318            },
319            MenuItem {
320                id: "test",
321                label: "Send a test event",
322                icon: icondata::LuSend,
323                act: Callback::new(move |()| workspace.test_webhook(tested.clone())),
324                danger: false,
325            },
326            MenuItem {
327                id: "rotate-url",
328                label: "New URL…",
329                icon: icondata::LuRefreshCw,
330                act: ask(Confirm::RotateUrl),
331                danger: false,
332            },
333        ];
334        if webhook.verify.is_some() {
335            items.push(MenuItem {
336                id: "rotate-secret",
337                label: "New secret…",
338                icon: icondata::LuKeyRound,
339                act: ask(Confirm::RotateSecret),
340                danger: false,
341            });
342        }
343        items.push(MenuItem {
344            id: "delete",
345            label: "Delete…",
346            icon: icondata::LuTrash2,
347            act: ask(Confirm::Delete),
348            danger: true,
349        });
350        items
351    }
352}

354/// A webhook's row (H1): a press opens its page.
355fn webhook_row(
356    workspace: Workspace,
357    webhook: Webhook,
358    confirm: RwSignal<Option<(WebhookId, Confirm)>>,
359    menu: RwSignal<Option<WebhookId>>,
360    context: RwSignal<Option<(WebhookId, f64, f64)>>,
361) -> impl IntoView {
362    let now = Timestamp::now();
363    let id = webhook.id.clone();
364    let title = webhook_title(&webhook);
365    let enabled = webhook.enabled;
366    let tags = row_tags(&webhook);
367    let items = menu_items(workspace, &webhook, confirm);
368    let open = {
369        let id = id.clone();
370        move |_| workspace.show_webhook(id.clone())
371    };
372    let opened_context = {
373        let id = id.clone();
374        move |event: leptos::ev::MouseEvent| {
375            if !crate::browser::from_touch(&event) && wants_context_menu(&event) {
376                event.prevent_default();
377                context.set(Some((
378                    id.clone(),
379                    f64::from(event.client_x()),
380                    f64::from(event.client_y()),
381                )));
382            }
383        }
384    };
385    view! {
386        {context_menu(context, id.clone(), title.clone(), items.clone())}
387        <div
388            class="webhook relative flex items-start gap-3 rounded-xl bg-raised py-2 pr-1 pl-3 shadow-[0_0_0_1px_var(--color-line),0_1px_2px_rgb(0_0_0/0.03)] transition-colors hover:bg-hover"
389            class:opacity-70=!enabled
390            on:contextmenu=opened_context
391            data-webhook=id.to_string()
392            data-title=title.clone()
393            data-enabled=enabled.to_string()
394        >
395            <button
396                type="button"
397                class="absolute inset-0 rounded-xl"
398                aria-label=format!("Open {title}")
399                data-webhook-open
400                on:click=open
401            ></button>
402            <span class="mt-1.5 inline-flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-sunken text-fg-2">
403                <Glyph icon=webhook_icon(webhook.verify.as_ref()) />
404            </span>
405            <span class="flex min-w-0 flex-1 flex-col gap-0.5 py-1">
406                <span class="flex min-w-0 items-center gap-2">
407                    <span class="min-w-0 truncate text-sm font-semibold" data-webhook-title>{title.clone()}</span>
408                    {tags.into_iter().map(|tag| view! { <span class=TAG data-webhook-tag>{tag}</span> }).collect_view()}
409                </span>
410                {target_line(workspace, &webhook)}
411                {health_line(health(&webhook), now)}
412            </span>
413            {switch(workspace, &webhook)}
414            {row_menu(menu, id.clone(), id.to_string(), title.clone(), MORE, items)}
415        </div>
416    }
417}

419/// An open webhook (H2): its head, its actions, a ready card when it is
420/// new (H5), and its three tabs.
421fn webhook_page(
422    workspace: Workspace,
423    webhook: Webhook,
424    receiving: bool,
425    confirm: RwSignal<Option<(WebhookId, Confirm)>>,
426    menu: RwSignal<Option<WebhookId>>,
427) -> impl IntoView {
428    let title = webhook_title(&webhook);
429    let url = workspace.webhook_url(&webhook);
430    let subtitle = {
431        let words = workspace
432            .projects
433            .with_untracked(|projects| target_words(&webhook, projects, name_of(workspace)));
434        let target = match words.session {
435            Some((_, name)) => format!("into {name}"),
436            None => words.rest,
437        };
438        format!("{} · {target}", check_label(webhook.verify.as_ref()))
439    };
440    let items = menu_items(workspace, &webhook, confirm);
441    let created = {
442        let id = webhook.id.clone();
443        move || {
444            workspace.webhook_revealed.with(|shown| {
445                shown
446                    .as_ref()
447                    .is_some_and(|shown| shown.webhook == id && shown.created)
448            })
449        }
450    };
451    let ready = {
452        let (webhook, url) = (webhook.clone(), url.clone());
453        move || created().then(|| ready_card(workspace, &webhook, url.clone()))
454    };
455    let tested = webhook.id.clone();
456    let copied_url = url.clone();
457    let events = webhook.history.len();
458    let tab = workspace.webhook_tab;
459    let tab_button = move |which: WebhookTab, label: &'static str, count: Option<usize>| {
460        view! {
461            <button
462                type="button"
463                role="tab"
464                class="inline-flex min-h-11 shrink-0 items-center gap-1.5 border-b-2 px-2 text-[0.8125rem] whitespace-nowrap transition-colors aria-selected:border-accent aria-selected:font-semibold aria-selected:text-fg border-transparent text-muted hover:text-fg lg:min-h-9"
465                aria-selected=move || (tab.get() == which).to_string()
466                data-webhook-tab=label
467                on:click=move |_| tab.set(which)
468            >
469                {label}
470                {count.filter(|count| *count > 0).map(|count| view! {
471                    <span class="rounded-md bg-sunken px-1 text-[0.6875rem] font-semibold text-muted">{count}</span>
472                })}
473            </button>
474        }
475    };
476    let body = {
477        let webhook = webhook.clone();
478        let url = url.clone();
479        move || match tab.get() {
480            WebhookTab::Setup => setup_tab(workspace, webhook.clone(), url.clone()).into_any(),
481            WebhookTab::Events => events_tab(workspace, &webhook).into_any(),
482            WebhookTab::Rules => rules_tab(workspace, &webhook, &url).into_any(),
483        }
484    };
485    view! {
486        <div class="flex flex-col gap-3" id="webhook-page" data-webhook-page=webhook.id.to_string()>
487            <div class="flex items-center gap-3">
488                <span class="inline-flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-sunken text-fg-2">
489                    <Glyph icon=webhook_icon(webhook.verify.as_ref()) class="text-lg" />
490                </span>
491                <div class="min-w-0 flex-1">
492                    <h2 class="truncate text-xl font-semibold tracking-tight" id="webhook-page-title">{title.clone()}</h2>
493                    <p class="truncate text-[0.8125rem] text-muted" data-webhook-subtitle>{subtitle}</p>
494                </div>
495                {switch(workspace, &webhook)}
496                {row_menu(menu, webhook.id.clone(), webhook.id.to_string(), title.clone(), ICON_BUTTON, items)}
497            </div>
498            // Two big actions, side by side on a phone (H6).
499            <div class="flex gap-2" data-webhook-actions>
500                <button
501                    type="button"
502                    class=format!("{BUTTON} flex-1 sm:flex-none")
503                    data-webhook-action="test"
504                    on:click=move |_| workspace.test_webhook(tested.clone())
505                >
506                    <Glyph icon=icondata::LuSend />
507                    "Send a test event"
508                </button>
509                <button
510                    type="button"
511                    class=format!("{BUTTON} flex-1 sm:flex-none")
512                    data-webhook-action="copy-url"
513                    on:click=move |_| copy(workspace, "URL", copied_url.clone())
514                >
515                    <Glyph icon=icondata::LuCopy />
516                    "Copy URL"
517                </button>
518            </div>
519            {notices(workspace, receiving)}
520            {confirmation(workspace, &webhook, confirm)}
521            {ready}
522            <div class="flex gap-1 overflow-x-auto border-b border-line" role="tablist" aria-label="Webhook">
523                {tab_button(WebhookTab::Setup, "Setup", None)}
524                {tab_button(WebhookTab::Events, "Events", Some(events))}
525                {tab_button(WebhookTab::Rules, "Rules & limits", None)}
526            </div>
527            <div class="flex flex-col gap-2" role="tabpanel">{body}</div>
528        </div>
529    }
530}

570/// The question a menu's New URL, New secret, or Delete asks first.
571fn confirmation(
572    workspace: Workspace,
573    webhook: &Webhook,
574    confirm: RwSignal<Option<(WebhookId, Confirm)>>,
575) -> impl IntoView + use<> {
576    let id = webhook.id.clone();
577    let title = webhook_title(webhook);
578    move || {
579        let what = confirm
580            .get()
581            .filter(|(asked, _)| *asked == id)
582            .map(|(_, what)| what)?;
583        let (id, title) = (id.clone(), title.clone());
584        let question = match what {
585            Confirm::Delete => format!(
586                "Delete {title}? Its URL stops working at once; the sessions it started stay."
587            ),
588            Confirm::RotateUrl => format!(
589                "Give {title} a new URL? The old one stops working at once, so update every sender."
590            ),
591            Confirm::RotateSecret => format!(
592                "Give {title} a new secret? Requests signed with the old one are refused from now on."
593            ),
594        };
595        let verb = match what {
596            Confirm::Delete => "Delete",
597            Confirm::RotateUrl => "New URL",
598            Confirm::RotateSecret => "New secret",
599        };

758/// Events (H3): the counters, then each event newest first.

812/// An event's row (H3): a dot, the outcome in words, a test tag, the time,
813/// and its session; a press opens its request.
814fn event_row(
815    workspace: Workspace,
816    webhook: &Webhook,
817    event: &WebhookEvent,
818    zone: &jiff::tz::TimeZone,
819    now: Timestamp,
820) -> impl IntoView + use<> {
821    let kind = outcome_kind(event);
822    let tone = match kind {
823        OutcomeKind::Failed => Tone::Bad,
824        OutcomeKind::Pending | OutcomeKind::Waiting => Tone::Warn,
825        OutcomeKind::Unknown | OutcomeKind::Ended => Tone::Quiet,
826        OutcomeKind::Delivered | OutcomeKind::Completed => Tone::Good,
827    };
828    let outcome = {
829        let outcome = event_outcome(event);
830        let outcome = outcome
831            .strip_prefix("test · ")
832            .unwrap_or(&outcome)
833            .to_owned();
834        let mut chars = outcome.chars();
835        chars
836            .next()
837            .map(|first| first.to_uppercase().chain(chars).collect::<String>())
838            .unwrap_or_default()
839    };
840    let mut when = vec![
841        event_time(event, zone),
842        relative_time(now, event.received_at),
843    ];
844    if let Some(key) = &event.thread_key {
845        when.push(format!("thread {key}"));
846    }
847    let id = event.id.clone();
848    let hook = webhook.id.clone();
849    let path = webhook.path.clone();
850    let open = {
851        let (hook, id) = (hook.clone(), id.clone());
852        Memo::new(move |_| {
853            workspace.webhook_event.with(|open| {
854                open.as_ref()
855                    .is_some_and(|open| open.webhook == hook && open.event == id)
856            })
857        })
858    };
859    let toggle = {
860        let (hook, id) = (hook.clone(), id.clone());
861        move |_| workspace.toggle_webhook_event(hook.clone(), id.clone())
862    };
863    let session = event.session_id.map(|session| {
864        let name = name_of(workspace)(session).unwrap_or_else(|| "its session".to_owned());
865        view! {
866            <button
867                type="button"
868                class="inline-flex min-h-11 max-w-[40%] shrink-0 items-center gap-1.5 rounded-lg px-2 text-[0.8125rem] text-accent hover:bg-hover lg:min-h-9"
869                data-webhook-event-session=session.to_string()
870                title="Open the session"
871                on:click=move |_| {
872                    workspace.open_view(View::Chat, None);
873                    workspace.open_session(session);
874                }
875            >
876                <Glyph icon=icondata::LuExternalLink class="shrink-0" />
877                <span class="truncate">{name}</span>
878            </button>
879        }
880    });
881    view! {
882        <div class="flex flex-col" data-webhook-event=id.clone()>
883            <div class="flex items-center gap-1 pr-1">
884                <button
885                    type="button"
886                    class="flex min-h-14 min-w-0 flex-1 items-center gap-3 rounded-lg px-3 text-left hover:bg-hover"
887                    aria-expanded=move || open.get().to_string()
888                    title=format!("{} · show the request", exact_time(event.received_at))
889                    on:click=toggle
890                >
891                    <span class="inline-flex shrink-0 text-muted transition-transform" class:rotate-90=move || open.get()>
892                        <Glyph icon=icondata::LuChevronRight />
893                    </span>
894                    <span class=format!("h-2 w-2 shrink-0 rounded-full {}", dot(tone)) aria-hidden="true"></span>
895                    <span class="flex min-w-0 flex-1 flex-col">
896                        <span class="flex min-w-0 items-center gap-2 text-sm font-medium" class:text-bad=tone == Tone::Bad>
897                            <span class="min-w-0 break-words" data-webhook-event-outcome>{outcome}</span>
898                            {event.test.then(|| view! { <span class="inline-flex h-5 shrink-0 items-center rounded-md bg-accent-soft px-1.5 text-[0.6875rem] font-semibold text-accent" data-webhook-test>"test"</span> })}
899                        </span>
900                        <span class="truncate text-xs text-muted">{when.join(" · ")}</span>
901                    </span>
902                </button>
903                {session}
904            </div>
905            {move || open.get().then(|| event_request(workspace, path.clone()))}
906        </div>
907    }
908}

910/// The open event's request: the method and path, the headers as a table,
911/// and the body, pretty-printed when it is JSON.
912fn event_request(workspace: Workspace, path: String) -> impl IntoView {
913    move || {
914        let request = workspace
915            .webhook_event
916            .with(|open| open.as_ref().and_then(|open| open.request.clone()));
917        match request {
918            None => loading("Reading the request…").into_any(),
919            Some(Err(error)) => view! { <p class=ERROR>{error}</p> }.into_any(),
920            Some(Ok(request)) => {
921                let headers = request
922                    .headers
923                    .iter()
924                    .map(|header| {
925                        view! {
926                            <tr>
927                                <td class="w-40 py-0.5 pr-3 align-top text-muted">{header.name.clone()}</td>
928                                <td class="py-0.5 font-mono break-all">{header.value.clone()}</td>
929                            </tr>
930                        }
931                    })
932                    .collect_view();
933                let body = match (request.body_text, request.body_base64) {
934                    (Some(text), _) => serde_json::from_str::<serde_json::Value>(&text)
935                        .ok()
936                        .and_then(|value| serde_json::to_string_pretty(&value).ok())
937                        .unwrap_or(text),
938                    (None, Some(_)) => format!("({} bytes of binary content)", request.event.bytes),
939                    (None, None) => "(no body)".to_owned(),
940                };
941                let copied = body.clone();
942                view! {
943                    <div class="flex flex-col gap-2 border-t border-line py-3 pr-3 pl-12" data-webhook-request>
944                        <div class="flex items-center gap-2">
945                            <span class="rounded-md bg-accent-soft px-1.5 py-0.5 font-mono text-[0.6875rem] font-semibold text-accent">{request.method}</span>
946                            <span class="min-w-0 flex-1 truncate font-mono text-xs text-muted">{path.clone()}</span>
947                            <button
948                                type="button"
949                                class=ICON_BUTTON
950                                title="Copy the body"
951                                aria-label="Copy the body"
952                                on:click=move |_| copy(workspace, "Body", copied.clone())
953                            >
954                                <Glyph icon=icondata::LuCopy />
955                            </button>
956                        </div>
957                        <table class="w-full text-xs" data-webhook-headers><tbody>{headers}</tbody></table>
958                        <pre class="max-h-96 overflow-auto rounded-lg bg-sunken px-3 py-2 font-mono text-xs leading-relaxed whitespace-pre-wrap break-words" data-webhook-body>{body}</pre>
959                    </div>
960                }
961                .into_any()
962            }
963        }
964    }
965}
