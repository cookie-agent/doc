53pub fn webhooks_page(workspace: Workspace) -> impl IntoView {
54    let confirm = RwSignal::new(None::<(WebhookId, Confirm)>);
55    // The form renders once per opening: its fields keep what is typed.
56    let form = Memo::new(move |_| workspace.webhook_form.get());
57    move || {
58        if let Some(form) = form.get() {
59            return webhook_form(workspace, form).into_any();
60        }
61        let listing = match workspace.webhooks.get() {
62            None => return loading("Reading webhooks…").into_any(),
63            Some(Err(error)) => return view! { <p class=ERROR>{error}</p> }.into_any(),
64            Some(Ok(listing)) => listing,
65        };
66        let rows = if listing.webhooks.is_empty() {
67            empty_state(
68                icondata::LuWebhook,
69                "No webhooks. Add one here, or from a session's ⋮ menu.",
70            )
71            .into_any()
72        } else {
73            listing
74                .webhooks
75                .into_iter()
76                .map(|webhook| webhook_row(workspace, webhook, confirm))
77                .collect_view()
78                .into_any()
79        };
80        view! {
81            {super::view_header(
82                "Webhooks",
83                "URLs that turn each request they receive into a message in a session, or start one.",
84                view! {
85                    <button type="button" id="webhook-new" class=PRIMARY on:click=move |_| workspace.new_webhook()>
86                        <Glyph icon=icondata::LuPlus />
87                        "New webhook"
88                    </button>
89                },
90            )}
91            {(!listing.receiving).then(|| view! {
92                <p class=ERROR id="webhooks-receiving-off">
93                    "Receiving is off ([webhooks] enabled = false): every URL answers 404, and the webhooks stay."
94                </p>
95            })}
96            {move || workspace.webhook_error.get().map(|error| view! { <p class=ERROR id="webhook-error">{error}</p> })}
97            {move || workspace.webhook_revealed.get().map(|revealed| revealed_note(workspace, revealed))}
98            {rows}
99        }
100        .into_any()
101    }
102}

104/// A secret or URL the daemon returned once: shown with a copy button
105/// until dismissed.
106fn revealed_note(
107    workspace: Workspace,
108    revealed: crate::workspace::WebhookRevealed,
109) -> impl IntoView {
110    let url = revealed.new_url.then(|| {
111        workspace
112            .webhook_list()
113            .into_iter()
114            .find(|webhook| webhook.id == revealed.webhook)
115            .map(|webhook| workspace.webhook_url(&webhook))
116    });
117    let heading = if revealed.secret.is_some() {
118        format!(
119            "{}: copy its secret now; it is shown here until you close this.",
120            revealed.title
121        )
122    } else {
123        format!("{}: its URL.", revealed.title)
124    };
125    view! {
126        <div class="flex flex-col gap-2 rounded-xl bg-raised p-3 shadow-[0_0_0_1px_var(--color-accent),0_0_0_4px_var(--color-accent-soft)]" role="status" id="webhook-revealed">
127            <p class="text-sm font-bold">{heading}</p>
128            {url.flatten().map(|url| copy_line(workspace, "URL", url, "webhook-revealed-url"))}
129            {revealed.secret.map(|secret| copy_line(workspace, "Secret", secret, "webhook-revealed-secret"))}
130            <div>
131                <button type="button" class=BUTTON id="webhook-revealed-close" on:click=move |_| workspace.webhook_revealed.set(None)>
132                    "Done"
133                </button>
134            </div>
135        </div>
136    }
137}

139/// A value to copy: its label, the value, and a copy button.
140fn copy_line(
141    workspace: Workspace,
142    label: &'static str,
143    value: String,
144    id: &'static str,
145) -> impl IntoView {
146    let copied = value.clone();
147    view! {

149            <span class="text-xs font-bold text-muted">{label}</span>
150            <div class="flex items-start gap-2">
151                <code class=CODE id=id>{value}</code>
152                <button
153                    type="button"
154                    class=BUTTON
155                    title=format!("Copy the {}", label.to_lowercase())
156                    aria-label=format!("Copy the {}", label.to_lowercase())
157                    data-copy=id
158                    on:click=move |_| {
159                        let text = copied.clone();
160                        leptos::task::spawn_local(async move {
161                            let done = crate::browser::copy_text(&text).await;
162                            workspace.status.set(if done {
163                                format!("{label} copied.")
164                            } else {
165                                "This browser does not let the page copy; select the text instead.".to_owned()
166                            });
167                        });
168                    }
169                >
170                    <Glyph icon=icondata::LuCopy />
171                </button>
172            </div>
173        </div>
174    }

177/// A webhook's row: a click shows or hides its details.
178fn webhook_row(
179    workspace: Workspace,
180    webhook: Webhook,
181    confirm: RwSignal<Option<(WebhookId, Confirm)>>,
182) -> impl IntoView {
183    let id = webhook.id.clone();
184    let selected = {
185        let id = id.clone();
186        Memo::new(move |_| {
187            workspace
188                .webhook_selected
189                .with(|selected| selected.as_ref() == Some(&id))
190        })
191    };
192    let title = webhook_title(&webhook);
193    let target = workspace.projects.with_untracked(|projects| {
194        target_label(&webhook.target, projects, |session| {
195            workspace
196                .sessions
197                .with_untracked(|sessions| session_name(sessions, session))
198        })
199    });
200    let state = state_label(&webhook);
201    let check = check_label(webhook.verify.as_ref());
202    let requests = counts_summary(&webhook.counts);
203    let failing = last_outcome(&webhook) == Some(OutcomeKind::Failed);
204    let enabled = webhook.enabled;
205    let toggle = {
206        let id = id.clone();
207        move |_| {
208            workspace.webhook_selected.update(|selected| {
209                *selected = if selected.as_ref() == Some(&id) {
210                    None
211                } else {
212                    Some(id.clone())
213                };
214            });
215        }
216    };
217    let details = {
218        let webhook = webhook.clone();
219        move || {
220            selected
221                .get()
222                .then(|| webhook_details(workspace, webhook.clone(), confirm))
223        }
224    };
225    // A right-click offers the webhook's actions without opening it.
226    let context = RwSignal::new(None);
227    let context_items = {
228        let webhook = webhook.clone();
229        move || {
230            use super::row_menu::MenuItem;
231            let (edited, toggled, tested, asked) = (
232                webhook.clone(),
233                webhook.id.clone(),
234                webhook.id.clone(),
235                webhook.id.clone(),
236            );
237            let enabled = webhook.enabled;
238            vec![
239                MenuItem {
240                    id: "edit",
241                    label: "Edit",
242                    icon: icondata::LuPencil,
243                    act: Callback::new(move |()| workspace.edit_webhook(edited.clone())),
244                    danger: false,
245                },
246                MenuItem {
247                    id: if enabled { "disable" } else { "enable" },
248                    label: if enabled { "Disable" } else { "Enable" },
249                    icon: if enabled {
250                        icondata::LuPause
251                    } else {
252                        icondata::LuPlay
253                    },
254                    act: Callback::new(move |()| {
255                        workspace.set_webhook_enabled(toggled.clone(), !enabled)
256                    }),
257                    danger: false,
258                },
259                MenuItem {
260                    id: "test",
261                    label: "Send a test event",
262                    icon: icondata::LuSend,
263                    act: Callback::new(move |()| workspace.test_webhook(tested.clone())),
264                    danger: false,
265                },
266                MenuItem {
267                    id: "delete",
268                    label: "Delete…",
269                    icon: icondata::LuTrash2,
270                    act: Callback::new(move |()| {
271                        // The confirmation is in the webhook's details.
272                        workspace.webhook_selected.set(Some(asked.clone()));
273                        confirm.set(Some((asked.clone(), Confirm::Delete)));
274                    }),
275                    danger: true,
276                },
277            ]
278        }
279    };
280    view! {
281        {super::row_menu::context_menu(context, (), title.clone(), context_items)}
282        <div
283            on:contextmenu=super::row_menu::on_context(context)
284            class=move || format!(
285                "webhook flex flex-col gap-1 rounded-xl p-1 transition-shadow {}",
286                if selected.get() {
287                    "bg-raised shadow-[0_0_0_1px_var(--color-accent),0_0_0_4px_var(--color-accent-soft)]"
288                } else {
289                    "bg-raised shadow-[0_0_0_1px_var(--color-line),0_1px_2px_rgb(0_0_0/0.03)]"
290                }
291            )
292            data-webhook=id.to_string()
293            data-title=title.clone()
294            data-enabled=enabled.to_string()
295        >
296            <div class="relative">
297            <button
298                type="button"
299                role="switch"
300                class="absolute top-1 right-1 z-[1] inline-flex h-11 w-11 items-center justify-center rounded-full lg:top-2.5 lg:right-2.5 lg:h-7"
301                aria-checked=enabled.to_string()
302                aria-label=if enabled { "Turn the webhook off" } else { "Turn the webhook on" }
303                title=if enabled { "On: click to turn it off" } else { "Off: click to turn it on" }
304                data-webhook-switch
305                on:click={
306                    let id = id.clone();
307                    move |_| workspace.set_webhook_enabled(id.clone(), !enabled)
308                }
309            >
310                <span class="effort-switch" class:effort-switch-on=enabled aria-hidden="true"></span>
311            </button>
312            <button
313                type="button"
314                class=format!("{ROW} pr-16")
315                aria-expanded=move || selected.get().to_string()
316                on:click=toggle
317            >
318                <span class="flex w-full items-center gap-2">
319                    <Glyph
320                        icon=icondata::LuWebhook
321                        class=if enabled { "text-accent" } else { "opacity-50" }
322                    />
323                    <span class="min-w-0 flex-1 truncate text-sm font-bold" data-webhook-title>{title.clone()}</span>
324                    <span class=STATE data-webhook-state>{state}</span>
325                </span>
326                <span class="pl-6 text-[0.8125rem] text-fg-2" data-webhook-target>{target}</span>
327                <span class=format!("flex w-full flex-wrap gap-x-3 pl-6 {NOTE}")>
328                    <span data-webhook-check>{check}</span>
329                    <span data-webhook-requests class:text-bad=failing>{requests}</span>
330                </span>
331            </button>
332            </div>
333            {details}
334        </div>
335    }
336}

338/// The selected webhook: its URL, details, actions, and events, newest
339/// first. The target and the session that created it link to their
340/// sessions.
341fn webhook_details(
342    workspace: Workspace,
343    webhook: Webhook,
344    confirm: RwSignal<Option<(WebhookId, Confirm)>>,
345) -> impl IntoView {
346    let zone = zone_at(jiff::Timestamp::now());
347    let url = workspace.webhook_url(&webhook);
348    let private = workspace.webhook_urls_private();
349    let title = |session| {
350        workspace
351            .sessions
352            .with_untracked(|sessions| session_name(sessions, session))
353    };
354    let lines = workspace
355        .projects
356        .with_untracked(|projects| detail_lines(&webhook, &url, &zone, projects, title))
357        .into_iter()
358        // The URL has its own line, with a copy button.
359        .filter(|detail| detail.label != "url")
360        .map(|detail| {
361            let value = match detail.session {
362                Some(session) => view! {
363                    <button
364                        type="button"
365                        class=LINK
366                        data-webhook-session=session.to_string()
367                        title="Open this session"
368                        on:click=move |_| {
369                            workspace.open_view(crate::layout::View::Chat, None);
370                            workspace.open_session(session);
371                        }
372                    >
373                        {detail.value}
374                    </button>
375                }
376                .into_any(),
377                None => detail.value.into_any(),
378            };
379            view! {
380                <dt class="text-xs font-bold text-muted">{detail.label}</dt>
381                <dd class="min-w-0 text-sm break-words" data-webhook-detail=detail.label>{value}</dd>
382            }
383        })
384        .collect_view();
385    let prompt = webhook.prompt.clone();
386    let thread_prompt = match &webhook.target {
387        cookie_agent_protocol::WebhookTarget::New(template) => template.thread_prompt.clone(),
388        _ => None,
389    };
390    let history = if webhook.history.is_empty() {
391        view! { <p class=NOTE>"No events yet. Send a test event to try it."</p> }.into_any()
392    } else {
393        events_newest_first(&webhook)
394            .map(|event| event_line(workspace, webhook.id.clone(), event, &zone))
395            .collect_view()
396            .into_any()
397    };
398    let title = webhook_title(&webhook);
399    let has_secret = webhook.verify.is_some();
400    let edited = webhook.clone();
401    let (toggled, enabled) = (webhook.id.clone(), webhook.enabled);
402    let tested = webhook.id.clone();
403    let id = webhook.id.clone();
404    let ask = move |what: Confirm| {
405        let id = id.clone();
406        move |_| confirm.set(Some((id.clone(), what)))
407    };
408    let revealed = (webhook.id.clone(), title.clone());
409    let confirming = {
410        let id = webhook.id.clone();
411        move || {
412            confirm
413                .get()
414                .filter(|(asked, _)| *asked == id)
415                .map(|(_, what)| what)
416        }
417    };
418    let confirmed = (webhook.id.clone(), title.clone());
419    view! {
420        <div class="flex flex-col gap-3 px-3 pb-2" data-webhook-details>
421            {copy_line(workspace, "URL", url, "webhook-url")}
422            {private.then(|| view! {
423                <p class=NOTE data-webhook-private>
424                    "This URL works only where cookie's page can be reached. Set [public] url to the address a tunnel or Funnel gives the public port, so other services can reach it."
425                </p>
426            })}
427            <dl class="grid grid-cols-[auto_1fr] items-baseline gap-x-3 gap-y-1">{lines}</dl>
428            {thread_prompt.map(|text| view! {
429                <div class="flex flex-col gap-1">
430                    <span class="text-xs font-bold text-muted">"Thread prompt"</span>
431                    <p class="rounded-md bg-sunken px-3 py-2 text-sm whitespace-pre-wrap break-words" data-webhook-thread-prompt>{text}</p>
432                </div>
433            })}
434            {prompt.map(|text| view! {
435                <div class="flex flex-col gap-1">
436                    <span class="text-xs font-bold text-muted">"Prompt"</span>
437                    <p class="rounded-md bg-sunken px-3 py-2 text-sm whitespace-pre-wrap break-words" data-webhook-prompt>{text}</p>
438                </div>
439            })}
440            <div class="flex flex-wrap gap-2" data-webhook-actions>
441                <button type="button" class=BUTTON data-webhook-action="edit" on:click=move |_| workspace.edit_webhook(edited.clone())>
442                    <Glyph icon=icondata::LuPencil />
443                    "Edit"
444                </button>
445                <button
446                    type="button"
447                    class=BUTTON
448                    data-webhook-action=if enabled { "disable" } else { "enable" }
449                    on:click=move |_| workspace.set_webhook_enabled(toggled.clone(), !enabled)
450                >
451                    <Glyph icon=if enabled { icondata::LuPause } else { icondata::LuPlay } />
452                    {if enabled { "Disable" } else { "Enable" }}
453                </button>
454                <button type="button" class=BUTTON data-webhook-action="test" on:click=move |_| workspace.test_webhook(tested.clone())>
455                    <Glyph icon=icondata::LuSend />
456                    "Send a test event"
457                </button>
458                <button type="button" class=BUTTON data-webhook-action="rotate-url" on:click=ask(Confirm::RotateUrl)>
459                    <Glyph icon=icondata::LuRefreshCw />
460                    "New URL"
461                </button>
462                {has_secret.then(|| {
463                    let revealed = revealed.clone();
464                    view! {
465                        <button type="button" class=BUTTON data-webhook-action="rotate-secret" on:click=ask(Confirm::RotateSecret)>
466                            <Glyph icon=icondata::LuKeyRound />
467                            "New secret"
468                        </button>
469                        <button
470                            type="button"
471                            class=BUTTON
472                            data-webhook-action="reveal"
473                            on:click=move |_| workspace.reveal_webhook_secret(revealed.0.clone(), revealed.1.clone())
474                        >
475                            <Glyph icon=icondata::LuEye />
476                            "Show secret"
477                        </button>
478                    }
479                })}
480                <button type="button" class=BUTTON data-webhook-action="delete" on:click=ask(Confirm::Delete)>
481                    <Glyph icon=icondata::LuTrash2 />
482                    "Delete"
483                </button>
484            </div>
485            {move || confirming().map(|what| {
486                let (id, title) = confirmed.clone();
487                let question = match what {
488                    Confirm::Delete => format!("Delete {title}? Its URL stops working at once; the sessions it started stay."),
489                    Confirm::RotateUrl => format!("Give {title} a new URL? The old one stops working at once, so update every sender."),
490                    Confirm::RotateSecret => format!("Give {title} a new secret? Requests signed with the old one are refused from now on."),
491                };
492                let verb = match what {
493                    Confirm::Delete => "Delete",
494                    Confirm::RotateUrl => "New URL",
495                    Confirm::RotateSecret => "New secret",
496                };
497                view! {
498                    <div class=CONFIRM role="alertdialog" data-webhook-confirm>
499                        <p class="text-sm">{question}</p>
500                        <div class="flex gap-2">
501                            <button type="button" id="webhook-confirm" class=PRIMARY on:click=move |_| {
502                                confirm.set(None);
503                                match what {
504                                    Confirm::Delete => workspace.delete_webhook(id.clone(), title.clone()),
505                                    Confirm::RotateUrl => workspace.rotate_webhook(id.clone(), WebhookRotate::Url),
506                                    Confirm::RotateSecret => workspace.rotate_webhook(id.clone(), WebhookRotate::Secret),
507                                }
508                            }>
509                                {verb}
510                            </button>
511                            <button type="button" class=BUTTON on:click=move |_| confirm.set(None)>
512                                "Keep it"
513                            </button>
514                        </div>
515                    </div>
516                }
517            })}
518            <div class="flex flex-col gap-1" data-webhook-history>
519                <span class="text-xs font-bold text-muted">"Events"</span>
520                {history}
521            </div>
522        </div>
523    }
524}

526/// One event: when, what it did, its thread, and the session it went to;
527/// a press opens its request.
528fn event_line(
529    workspace: Workspace,
530    webhook: WebhookId,
531    event: &WebhookEvent,
532    zone: &jiff::tz::TimeZone,
533) -> impl IntoView {
534    let when = event_time(event, zone);
535    let outcome = event_outcome(event);
536    let failed = matches!(
537        cookie_agent_client_state::webhook::outcome_kind(event),
538        OutcomeKind::Failed
539    );
540    let thread = event.thread_key.clone();
541    let session = event.session_id;
542    let id = event.id.clone();
543    let open = {
544        let (webhook, id) = (webhook.clone(), id.clone());
545        Memo::new(move |_| {
546            workspace.webhook_event.with(|open| {
547                open.as_ref()
548                    .is_some_and(|open| open.webhook == webhook && open.event == id)
549            })
550        })
551    };
552    let toggle = {
553        let (webhook, id) = (webhook.clone(), id.clone());
554        move |_| workspace.toggle_webhook_event(webhook.clone(), id.clone())
555    };
556    view! {
557        <div class="flex flex-col" data-webhook-event=id.clone()>
558            <div class="flex min-h-11 flex-wrap items-center gap-x-3 text-sm">
559                <button
560                    type="button"
561                    class="inline-flex min-h-11 min-w-0 flex-1 items-center gap-3 rounded-lg px-1 text-left hover:bg-hover"
562                    aria-expanded=move || open.get().to_string()
563                    title="Show the request"
564                    on:click=toggle
565                >
566                    <span class="inline-flex transition-transform" class:rotate-90=move || open.get()>
567                        <Glyph icon=icondata::LuChevronRight />
568                    </span>
569                    <span class="font-mono text-xs">{when}</span>
570                    <span class="min-w-0 flex-1 break-words" class:text-bad=failed data-webhook-event-outcome>{outcome}</span>
571                    {thread.map(|key| view! { <span class=NOTE>{format!("thread {key}")}</span> })}
572                </button>
573                {session.map(|session| view! {
574                    <button
575                        type="button"
576                        class=BUTTON
577                        data-webhook-event-session=session.to_string()
578                        on:click=move |_| {
579                            workspace.open_view(crate::layout::View::Chat, None);
580                            workspace.open_session(session);
581                        }
582                    >
583                        <Glyph icon=icondata::LuExternalLink />
584                        "Open session"
585                    </button>
586                })}
587            </div>
588            {move || open.get().then(|| event_request(workspace))}
589        </div>
590    }
591}

593/// The open event's request: its method, headers, and body.
594fn event_request(workspace: Workspace) -> impl IntoView {
595    move || {
596        let request = workspace
597            .webhook_event
598            .with(|open| open.as_ref().and_then(|open| open.request.clone()));
599        match request {
600            None => loading("Reading the request…").into_any(),
601            Some(Err(error)) => view! { <p class=ERROR>{error}</p> }.into_any(),
602            Some(Ok(request)) => {
603                let headers = request
604                    .headers
605                    .iter()
606                    .map(|header| format!("{}: {}", header.name, header.value))
607                    .collect::<Vec<_>>()
608                    .join("\n");
609                let body = match (request.body_text, request.body_base64) {
610                    (Some(text), _) => text,
611                    (None, Some(_)) => format!("({} bytes of binary content)", request.event.bytes),
612                    (None, None) => "(no body)".to_owned(),
613                };
614                view! {
615                    <div class="flex flex-col gap-2 py-2 pl-7" data-webhook-request>
616                        <code class=CODE>{format!("{}\n{headers}", request.method)}</code>
617                        <code class=format!("{CODE} max-h-80 overflow-y-auto") data-webhook-body>{body}</code>
618                    </div>
619                }
620                .into_any()
621            }
622        }
623    }
624}

626/// The webhook form: a new webhook, or an edit of one.
627#[allow(clippy::too_many_lines, reason = "one form, field by field")]
628fn webhook_form(workspace: Workspace, form: WebhookForm) -> impl IntoView {
629    let WebhookForm {
630        editing,
631        draft,
632        session,
633        template,
634    } = form;
635    let name = RwSignal::new(draft.name.clone());
636    let prompt = RwSignal::new(draft.prompt.clone());
637    let thread_key = RwSignal::new(draft.thread_key.clone());
638    let thread_prompt = RwSignal::new(draft.thread_prompt.clone());
639    let delivery = RwSignal::new(draft.delivery);
640    let verify = RwSignal::new(draft.verify.clone());
641    let secret = RwSignal::new(String::new());
642    let filter = RwSignal::new(draft.filter.clone());
643    let headers = RwSignal::new(draft.headers.clone());
644    let max_per_hour = RwSignal::new(draft.max_per_hour.clone());
645    let max_concurrent = RwSignal::new(draft.max_concurrent.clone());
646    let max_failures = RwSignal::new(draft.max_consecutive_failures.clone());
647    let new = RwSignal::new(draft.target.is_new());
648    let agent = RwSignal::new(if draft.agent.is_empty() {
649        template.agent.to_string()
650    } else {
651        draft.agent.clone()
652    });
653    let model = RwSignal::new(draft.model.clone());
654    let mode = RwSignal::new(draft.permission_mode);
655    let creating = editing.is_none();
656    let threaded = move || new.get() && !thread_key.with(|key| key.trim().is_empty());
657    let target_note = editing.as_ref().map(|webhook| {
658        workspace.projects.with_untracked(|projects| {
659            target_label(&webhook.target, projects, |id| {
660                workspace
661                    .sessions
662                    .with_untracked(|sessions| session_name(sessions, id))
663            })
664        })
665    });
666    let this_session = session.map(|(id, subagent)| {
667        let kind = if subagent {
668            "this subagent"
669        } else {
670            "this session"
671        };
672        workspace
673            .sessions
674            .with_untracked(|sessions| session_name(sessions, id))
675            .map_or_else(|| kind.to_owned(), |name| format!("{kind}: {name}"))
676    });
677    let agents = {
678        let mut agents: Vec<String> = untrack(|| workspace.selectable_agents())
679            .into_iter()
680            .map(|agent| agent.id.to_string())
681            .collect();
682        if !agents.contains(&agent.get_untracked()) {
683            agents.insert(0, agent.get_untracked());
684        }
685        agents
686    };
687    let pins = {
688        let mut pins = model_pins(&untrack(|| workspace.models()));
689        let current = model.get_untracked();
690        if !current.is_empty() && !pins.contains(&current) {
691            pins.insert(0, current);
692        }
693        pins
694    };
695    let build = {
696        let draft = draft.clone();
697        let template = template.clone();
698        move || {
699            let mut draft = WebhookDraft {
700                name: name.get_untracked(),
701                prompt: prompt.get_untracked(),
702                thread_key: thread_key.get_untracked(),
703                thread_prompt: thread_prompt.get_untracked(),
704                delivery: delivery.get_untracked(),
705                verify: verify.get_untracked(),
706                secret: secret.get_untracked(),
707                filter: filter.get_untracked(),
708                headers: headers.get_untracked(),
709                max_per_hour: max_per_hour.get_untracked(),
710                max_concurrent: max_concurrent.get_untracked(),
711                max_consecutive_failures: max_failures.get_untracked(),
712                permission_mode: mode.get_untracked(),
713                model: model.get_untracked(),
714                agent: agent.get_untracked(),
715                ..draft.clone()
716            };
717            if creating {
718                draft.target = match (new.get_untracked(), session) {
719                    (false, Some((session, subagent))) => {
720                        WebhookDraftTarget::Session { session, subagent }
721                    }
722                    _ => WebhookDraftTarget::New {
723                        template: template.clone(),
724                    },
725                };
726            }
727            draft
728        }
729    };
730    let heading = match &editing {
731        Some(webhook) => format!("Edit {}", webhook_title(webhook)),
732        None => "Add a webhook".to_owned(),
733    };
734    let text_field = move |id: &'static str,
735                           label: &'static str,
736                           value: RwSignal<String>,
737                           placeholder: &'static str| {
738        view! {
739            <label class=LABEL>
740                <span class="font-bold">{label}</span>
741                <input
742                    type="text"
743                    id=id
744                    class=FIELD
745                    placeholder=placeholder
746                    prop:value=move || value.get()
747                    on:input=move |event| value.set(event_target_value(&event))
748                />
749            </label>
750        }
751    };
752    view! {
753        <div class=EDITOR_PAGE id="webhook-form">
754            <h2 class="text-sm font-bold">{heading}</h2>
755            {text_field("webhook-name", "Name", name, "optional; lists show its ID without one")}
756            {match target_note {
757                Some(target) => view! {
758                    <p class="text-sm" id="webhook-target">{format!("Target: {target}")}</p>
759                }
760                .into_any(),
761                None => view! {
762                    <div class="flex flex-col gap-1" role="radiogroup" aria-label="Target">
763                        <span class="text-sm font-bold">"Target"</span>
764                        <div class="flex flex-wrap gap-2">
765                            {this_session.clone().map(|label| view! {
766                                <button
767                                    type="button"
768                                    id="webhook-target-session"
769                                    role="radio"
770                                    class=move || format!("{BUTTON} {}", if new.get() { "" } else { "bg-sunken" })
771                                    aria-checked=move || (!new.get()).to_string()
772                                    on:click=move |_| new.set(false)
773                                >
774                                    {label}
775                                </button>
776                            })}
777                            <button
778                                type="button"
779                                id="webhook-target-new"
780                                role="radio"
781                                class=move || format!("{BUTTON} {}", if new.get() { "bg-sunken" } else { "" })
782                                aria-checked=move || new.get().to_string()
783                                on:click=move |_| new.set(true)
784                            >
785                                "New sessions"
786                            </button>
787                        </div>
788                    </div>
789                }
790                .into_any(),
791            }}
792            <Show when=move || new.get()>
793                <div class="grid gap-2 sm:grid-cols-2">
794                    <label class=LABEL>
795                        <span class="font-bold">"Agent"</span>
796                        <select id="webhook-agent" class=SELECT on:change=move |event| agent.set(event_target_value(&event))>
797                            {agents.clone().into_iter().map(|id| {
798                                let (chosen, value) = (id.clone(), id.clone());
799                                view! { <option value=value selected=move || agent.get() == chosen>{id}</option> }
800                            }).collect_view()}
801                        </select>
802                    </label>
803                    <label class=LABEL>
804                        <span class="font-bold">"Model"</span>
805                        <select id="webhook-model" class=SELECT on:change=move |event| model.set(event_target_value(&event))>
806                            <option value="" selected=move || model.with(String::is_empty)>"the agent's own models"</option>
807                            {pins.clone().into_iter().map(|pin| {
808                                let (chosen, value) = (pin.clone(), pin.clone());
809                                view! { <option value=value selected=move || model.get() == chosen>{pin}</option> }
810                            }).collect_view()}
811                        </select>
812                    </label>
813                    <label class=LABEL>
814                        <span class="font-bold">"Permission mode"</span>
815                        <select
816                            id="webhook-mode"
817                            class=SELECT
818                            on:change=move |event| {
819                                let value = event_target_value(&event);
820                                mode.set(MODES.iter().map(|choice| choice.mode).find(|choice| permission_mode_label(*choice) == value));
821                            }
822                        >
823                            <option value="" selected=move || mode.get().is_none()>
824                                {format!("{} (the default)", permission_mode_label(PermissionMode::AutoApproveN))}
825                            </option>
826                            {MODES.iter().map(|choice| {
827                                let value = choice.mode;
828                                view! {
829                                    <option value=permission_mode_label(value) selected=move || mode.get() == Some(value)>
830                                        {format!("{} — {}", permission_mode_label(value), choice.description)}
831                                    </option>
832                                }
833                            }).collect_view()}
834                        </select>
835                    </label>
836                    {text_field("webhook-max-concurrent", "Sessions at once", max_concurrent, "1; 0 or less for no limit")}
837                </div>
838                {text_field("webhook-thread-key", "Thread key", thread_key, "json:/issue/number  or  header:X-Thread")}
839                <p class=NOTE>"Events with the same key go into one session, which the first of them starts. Without a key, each event starts its own session."</p>
840                <label class=LABEL>
841                    <span class="font-bold">"Thread prompt"</span>
842                    <textarea
843                        id="webhook-thread-prompt"
844                        class=AREA
845                        placeholder="optional; sent as your message when a session starts, and again after each compaction"
846                        prop:value=move || thread_prompt.get()
847                        on:input=move |event| thread_prompt.set(event_target_value(&event))
848                    ></textarea>
849                </label>
850            </Show>
851            <label class=LABEL>
852                <span class="font-bold">"Prompt"</span>
853                <textarea
854                    id="webhook-prompt"
855                    class=AREA
856                    placeholder="optional; sent with every event, inside its webhook_event block"
857                    prop:value=move || prompt.get()
858                    on:input=move |event| prompt.set(event_target_value(&event))
859                ></textarea>
860            </label>
861            <Show when=move || !new.get() || threaded()>
862                <label class=LABEL>
863                    <span class="font-bold">"Delivery"</span>
864                    <select
865                        id="webhook-delivery"
866                        class=SELECT
867                        on:change=move |event| delivery.set(if event_target_value(&event) == "steer" {
868                            ProducerDeliveryMode::Steer
869                        } else {
870                            ProducerDeliveryMode::Queue
871                        })
872                    >
873                        {[ProducerDeliveryMode::Queue, ProducerDeliveryMode::Steer].map(|choice| view! {
874                            <option value=delivery_label(choice) selected=move || delivery.get() == choice>
875                                {match choice {
876                                    ProducerDeliveryMode::Queue => "queue: after the current turn",
877                                    ProducerDeliveryMode::Steer => "steer: into the running turn",
878                                }}
879                            </option>
880                        })}
881                    </select>
882                </label>
883            </Show>
884            <div class="grid gap-2 sm:grid-cols-2">
885                {text_field("webhook-verify", "Check", verify, "github, standard, bearer, or hmac_sha256 header=…")}
886                {text_field("webhook-secret", "Secret", secret, "blank: the daemon makes one, or keeps it")}
887            </div>
888            <p class=NOTE>"Without a check, the URL is the only key. hmac_sha256 also takes encoding=hex|base64 and prefix=…"</p>
889            <label class=LABEL>
890                <span class="font-bold">"Filter"</span>
891                <textarea
892                    id="webhook-filter"
893                    class=format!("{AREA} min-h-[4rem] font-mono")
894                    placeholder="one condition per line, such as header:X-GitHub-Event=issues,pull_request or json:/action=opened"
895                    prop:value=move || filter.get()
896                    on:input=move |event| filter.set(event_target_value(&event))
897                ></textarea>
898            </label>
899            <div class="grid gap-2 sm:grid-cols-3">
900                {text_field("webhook-headers", "Headers shown", headers, "X-GitHub-Event, …")}
901                {text_field("webhook-max-per-hour", "Events per hour", max_per_hour, "the daemon's default")}
902                {text_field("webhook-max-failures", "Failures before off", max_failures, "the daemon's default")}
903            </div>
904            {move || workspace.webhook_error.get().map(|error| view! { <p class=ERROR id="webhook-error">{error}</p> })}
905            <div class=EDITOR_ACTIONS>
906                <button type="button" id="webhook-save" class=PRIMARY on:click=move |_| workspace.save_webhook(build())>
907                    {if creating { "Add" } else { "Save" }}
908                </button>
909                <button type="button" id="webhook-cancel" class=BUTTON on:click=move |_| workspace.close_webhook_form()>
910                    "Cancel"
911                </button>
912            </div>
913        </div>
914    }
915}

917/// A webhook session's mark in the session list: a click opens the
918/// webhook whose event started the session, while a press that moves
919/// drags the row like the rest of it.
920pub fn webhook_mark(workspace: Workspace, session: SessionWebhook) -> impl IntoView {
921    let id = session.webhook_id.clone();
922    let title = {
923        let session = session.clone();
924        move || {
925            workspace.webhooks.with(|webhooks| {
926                let list = webhooks
927                    .as_ref()
928                    .and_then(|webhooks| webhooks.as_ref().ok())
929                    .map(|webhooks| webhooks.webhooks.as_slice())
930                    .unwrap_or_default();
931                mark_title(&session, list)
932            })
933        }
934    };
935    view! {
936        <span
937            class="hit-area relative inline-flex shrink-0 text-accent opacity-80 hover:opacity-100"
938            role="link"
939            tabindex="0"
940            aria-label=title.clone()
941            title=title
942            data-session-webhook=session.webhook_id.to_string()
943            on:click=move |event: leptos::ev::MouseEvent| {
944                event.stop_propagation();
945                workspace.open_webhook(id.clone());
946            }
947        >
948            <Glyph icon=icondata::LuWebhook class="text-xs" />
949        </span>
950    }
