1//! The Agents view: the agent documents in a sidebar like Chat's sessions
2//! (the shared agents, then each preset's), and the open document's
3//! editor beside it. A desktop pins the sidebar, which the sidebar button
4//! hides and shows; a narrower window shows it as the page in place of the
5//! editor. A phone shows the agents as the page until one opens. Under the
6//! documents, the ACP agents every runtime offers (`[acp.agents]`), whose
7//! form opens over the page. Above the documents, the official agents, the
8//! ACP agents, and the templates, each with where its document comes from,
9//! why it cannot run, and Customize, Turn off, Reset, and Replace
10//! (`docs/site/specs/acp-onboarding.md` section 7).

29/// The icon of an agent by the mode its document gives it.
30fn mode_icon(mode: Option<&str>) -> crate::icons::Icon {
31    match mode {
32        Some("primary") => icondata::LuBot,
33        Some("subagent") => icondata::LuUserCog,
34        Some("all") => icondata::LuUsers,
35        _ => icondata::LuFileText,
36    }
37}

39/// Whether `path` is an agent document, which this view edits.
40fn is_agent(path: &str) -> bool {
41    path.starts_with("agents/")
42}

44/// The Agents view: the sidebar, then the open agent or what to do. Below
45/// desktop widths it shows one at a time: the agents fill the page while
46/// the sidebar is open, and on a phone until one opens; the title bar's
47/// sidebar button returns to them.
48pub fn agents_page(workspace: Workspace) -> impl IntoView {
49    // A phone shows the list until an agent opens, then the agent alone;
50    // a wider window both, side by side.
51    let phone = move || workspace.narrow.get();
52    let editing = move || {
53        workspace
54            .config_edit
55            .with(|edit| edit.as_ref().is_some_and(|edit| is_agent(&edit.path)))
56            || workspace.agent_open.with(Option::is_some)
57    };
58    let listing = move || phone() && !editing();
59    view! {
60        <div class="flex min-h-0 flex-1">
61            <aside
62                id="agents-sidebar"
63                class="flex shrink-0 flex-col border-r border-line"
64                class:hidden=move || phone() && editing()
65                class:w-full=phone
66                class:w-72=move || !phone()
67            >
68                {move || listing().then(|| {
69                    workspace.config_error.get().map(|error| view! { <p class=format!("{ERROR} mx-3 mt-3") id="config-error">{error}</p> })
70                })}
71                <AgentList workspace=workspace />
72            </aside>
73            <div
74                class="flex min-h-0 min-w-0 flex-1 flex-col overflow-y-auto p-3"
75                class:hidden=listing
76                data-page-body
77            >
78                <div class="mx-auto flex w-full max-w-3xl flex-1 flex-col gap-2">
79                    {agent_detail(workspace)}
80                </div>
81            </div>
82        </div>
83    }
84}

86/// The agent documents by preset, with New agent and Reload above them,
87/// then the ACP agents every runtime offers.
88#[component]
89pub fn AgentList(workspace: Workspace) -> impl IntoView {
90    let creating = RwSignal::new(false);
91    // `global`'s ACP agent declarations, read once the daemon shares them.
92    let acp = super::machines::Manage::new();
93    Effect::new(move |_| {
94        if workspace.shares_acp_agents.get() {
95            acp.read_agents(
96                workspace,
97                [
98                    cookie_agent_protocol::RuntimeId::new(cookie_agent_protocol::RuntimeId::GLOBAL)
99                        .expect("`global` is a runtime ID"),
100                ],
101            );
102        }
103    });
104    let create = move || {
105        workspace.agent_tab.set(crate::workspace::AgentTab::File);
106        workspace.new_agent(&field_value("config-new-path"));
107    };
108    // The panel's rows follow the configuration.
109    Effect::new(move |_| {
110        workspace.config.track();
111        workspace.poll_agent_documents();
112    });
113    view! {
114        <div class="flex flex-col gap-2 p-3">
115            <div class="flex gap-2">
116                <button
117                    type="button"
118                    class=format!("{PRIMARY} flex-1")
119                    data-new-agent
120                    aria-expanded=move || creating.get().to_string()
121                    on:click=move |_| creating.update(|creating| *creating = !*creating)
122                >
123                    <Glyph icon=icondata::LuPlus />
124                    "New agent"
125                </button>
126                <button
127                    type="button"
128                    id="config-reload"
129                    class=ICON_BUTTON
130                    aria-label="Reload from disk"
131                    title="Reload the documents from disk, after editing them by hand"
132                    on:click=move |_| workspace.reload_config()
133                >
134                    <Glyph icon=icondata::LuRefreshCw />
135                </button>
136            </div>
137            <Show when=move || creating.get()>
138                <label class="flex flex-col gap-1 text-xs font-bold text-muted">
139                    <span>"Document"</span>
140                    <input
141                        type="text"
142                        id="config-new-path"
143                        class=FIELD
144                        value="agents/"
145                        on:keydown=move |event: leptos::ev::KeyboardEvent| {
146                            if event.key() == "Enter" {
147                                event.prevent_default();
148                                create();
149                            }
150                        }
151                    />
152                </label>
153                <p class=NOTE>"agents/<id>.md, or agents/<preset>/<id>.md for a preset's agent"</p>
154                <div class="flex gap-2">
155                    <button type="button" id="config-new" class=PRIMARY on:click=move |_| create()>
156                        "Create"
157                    </button>
158                    <button type="button" class=BUTTON on:click=move |_| creating.set(false)>
159                        "Cancel"
160                    </button>
161                </div>
162            </Show>
163        </div>
164        <nav class="flex min-h-0 flex-1 flex-col gap-1 overflow-y-auto px-2 pb-3" aria-label="Agents">
165            {agent_panel(workspace)}
166            {move || {
167                let config = match workspace.config.get() {
168                    None => return loading("Reading the agents…").into_any(),
169                    Some(Err(error)) => return view! { <p class=ERROR>{error}</p> }.into_any(),
170                    Some(Ok(config)) => config,
171                };
172                let groups = agent_groups(&config);
173                if groups.is_empty() {
174                    return empty_state(icondata::LuBot, "No agent documents yet. Create one with New agent.")
175                        .into_any();
176                }
177                groups
178                    .into_iter()
179                    .map(|group| {
180                        let (icon, heading) = match &group.preset {
181                            None => (icondata::LuUsers, "Shared agents".to_owned()),
182                            Some(preset) => (icondata::LuLayers, format!("Preset {preset}")),
183                        };
184                        let empty = group.agents.is_empty();
185                        let rows = group
186                            .agents
187                            .into_iter()
188                            .map(|agent| agent_row(workspace, agent))
189                            .collect_view();
190                        view! {
191                            <h3
192                                class="mt-2 flex items-center gap-2 px-3 text-xs font-bold tracking-wide text-muted uppercase"
193                                data-preset=group.preset.clone().unwrap_or_default()
194                            >
195                                <Glyph icon=icon class="opacity-70" />
196                                <span class="min-w-0 truncate">{heading}</span>
197                            </h3>
198                            {empty.then(|| view! { <p class=format!("{NOTE} px-3")>"No agents in this preset."</p> })}
199                            {rows}
200                        }
201                    })
202                    .collect_view()
203                    .into_any()
204            }}
205            <Show when=move || workspace.shares_acp_agents.get()>
206                <div class="mt-3">{super::machines::shared_agents(workspace, acp)}</div>
207            </Show>
208        </nav>
209        {super::machines::acp_dialog(workspace, acp, "agents-acp-form")}
210    }
211}

213/// The official agents, the ACP agents, and the templates, by group, each
214/// with its actions.
215fn agent_panel(workspace: Workspace) -> impl IntoView {
216    use cookie_agent_client_state::agent_documents::{
217        AgentDocumentGroup, actions, group, origin_label, unavailable_line,
218    };
219    // An action that asks first, waiting for its second click.
220    let pending = RwSignal::new(None::<(String, AgentDocumentAction)>);
221    // The one open ⋯ menu among the rows.
222    let menu = RwSignal::new(None::<String>);
223    move || {
224        let documents = match workspace.agent_documents.get() {
225            None => return None,
226            Some(Err(error)) => return Some(view! { <p class=ERROR>{error}</p> }.into_any()),
227            Some(Ok(documents)) => documents,
228        };
229        let agents = workspace.with_runtime(|runtime| {
230            runtime
231                .snapshot()
232                .map(|snapshot| snapshot.agents.clone())
233                .unwrap_or_default()
234        });
235        let sections = [
236            AgentDocumentGroup::Official,
237            AgentDocumentGroup::Acp,
238            AgentDocumentGroup::Templates,
239        ]
240        .into_iter()
241        .filter_map(|each| {
242            let rows = documents
243                .iter()
244                .filter(|info| info.preset.is_none() && group(info) == each)
245                .cloned()
246                .collect::<Vec<_>>();
247            (!rows.is_empty()).then(|| {
248                let rows = rows
249                    .into_iter()
250                    .map(|info| {
251                        let reason = unavailable_line(&info, &agents);
252                        let description = agents
253                            .iter()
254                            .find(|agent| agent.id.as_str() == info.id.as_str())
255                            .map(|agent| agent.description.clone())
256                            .filter(|description| !description.is_empty());
257                        // An action that asks first asks on the row.
258                        let asking = {
259                            let id = info.id.clone();
260                            move || pending.get().filter(|(pending, _)| *pending == id).map(|(_, action)| action)
261                        };
262                        let confirm = {
263                            let (id, preset, path) = (info.id.clone(), info.preset.clone(), info.path.clone());
264                            move || {
265                                asking().map(|action| {
266                                    let (id, preset, path) = (id.clone(), preset.clone(), path.clone());
267                                    view! {
268                                        <span class="flex flex-wrap items-center gap-2 text-xs" data-agent-confirm>
269                                            <span class="text-muted">{format!("{}: this overwrites your document.", action.label())}</span>
270                                            <button
271                                                type="button"
272                                                class=format!("{PANEL_BUTTON} text-bad")
273                                                data-agent-action=format!("{:?}", action).to_lowercase()
274                                                on:click=move |_| {
275                                                    pending.set(None);
276                                                    workspace.agent_document_action(action, id.clone(), preset.clone(), path.clone());
277                                                }
278                                            >
279                                                {action.label()}
280                                            </button>
281                                            <button type="button" class=PANEL_BUTTON on:click=move |_| pending.set(None)>"Cancel"</button>
282                                        </span>
283                                    }
284                                })
285                            }
286                        };
287                        // A right-click offers the same actions; one that
288                        // asks first asks on the row, as its button does.
289                        let context = RwSignal::new(None);
290                        let context_items = {
291                            let info = info.clone();
292                            move || {
293                                actions(&info)
294                                    .into_iter()
295                                    .map(|action| {
296                                        let (id, preset, path) =
297                                            (info.id.clone(), info.preset.clone(), info.path.clone());
298                                        super::row_menu::MenuItem {
299                                            id: "agent-action",
300                                            label: action.label(),
301                                            icon: icondata::LuBot,
302                                            act: Callback::new(move |()| {
303                                                if action.confirms() {
304                                                    pending.set(Some((id.clone(), action)));
305                                                } else {
306                                                    workspace.agent_document_action(
307                                                        action,
308                                                        id.clone(),
309                                                        preset.clone(),
310                                                        path.clone(),
311                                                    );
312                                                }
313                                            }),
314                                            danger: action.confirms(),
315                                        }
316                                    })
317                                    .collect::<Vec<_>>()
318                            }
319                        };
320                        let menu_items = context_items.clone();
321                        // A click opens it on Resolved: the user's document
322                        // in the editor, else the agent read-only
323                        // (docs/site/specs/agent-shortcuts.md section 3.1).
324                        let open = {
325                            let info = info.clone();
326                            move |_| match &info.path {
327                                Some(path) => workspace.show_agent_document(path),
328                                None => workspace.open_agent_read_only(crate::workspace::AgentOpen {
329                                    id: info.id.clone(),
330                                    preset: info.preset.clone(),
331                                    path: None,
332                                    origin: origin_label(&info),
333                                    info: Some(info.clone()),
334                                }),
335                            }
336                        };
337                        let current = {
338                            let (id, path) = (info.id.clone(), info.path.clone());
339                            move || match &path {
340                                Some(path) => workspace
341                                    .config_edit
342                                    .with(|edit| edit.as_ref().is_some_and(|edit| edit.path == *path)),
343                                None => workspace.agent_open.with(|open| {
344                                    open.as_ref().is_some_and(|open| open.path.is_none() && open.id == id)
345                                }),
346                            }
347                        };
348                        let current_name = current.clone();
349                        view! {
350                            {super::row_menu::context_menu(context, (), info.id.clone(), context_items)}
351                            // Its name, what it does, and its actions in ⋯
352                            // (docs/site/specs/web-polish.md, P5).
353                            <div
354                                class="group relative flex flex-col gap-0.5 rounded-lg px-3 py-1.5 transition-colors hover:bg-hover"
355                                class:bg-sel=current
356                                data-agent-document=info.id.clone()
357                                on:contextmenu=super::row_menu::on_context(context)
358                            >
359                                // The whole row opens it; its name reads as the
360                                // link, and ⋯ stays above.
361                                <button
362                                    type="button"
363                                    class="absolute inset-0 rounded-lg"
364                                    aria-label=format!("Show what {} runs", info.id)
365                                    aria-current=move || current_name().then_some("true")
366                                    data-agent-open=info.id.clone()
367                                    on:click=open
368                                ></button>
369                                <span class="pointer-events-none relative flex w-full items-center gap-2">
370                                    <span class="min-w-0 truncate text-sm font-semibold decoration-line-2 underline-offset-[3px] group-hover:underline">
371                                        {info.id.clone()}
372                                    </span>
373                                    <span class=STATE>{origin_label(&info)}</span>
374                                    <span class="flex-1"></span>
375                                    <span class="pointer-events-auto">
376                                        {super::row_menu::row_menu(menu, info.id.clone(), info.id.clone(), info.id.clone(), ROW_MENU_BUTTON, menu_items)}
377                                    </span>
378                                </span>
379                                {description.map(|description| view! { <span class=format!("{NOTE} pointer-events-none relative line-clamp-2") data-agent-description>{description}</span> })}
380                                {reason.map(|reason| view! { <span class=format!("{NOTE} pointer-events-none relative") data-agent-unavailable>{reason}</span> })}
381                                <span class="relative">{confirm}</span>
382                            </div>
383                        }
384                    })
385                    .collect_view();
386                view! {
387                    <h3 class="mt-2 flex items-center gap-2 px-3 text-xs font-bold tracking-wide text-muted uppercase">
388                        {each.label()}
389                    </h3>
390                    {rows}
391                }
392            })
393        })
394        .collect_view();
395        Some(
396            view! { <section id="agent-panel" class="flex flex-col gap-1">{sections}</section> }
397                .into_any(),
398        )
399    }
400}

408/// An agent's row: its mode's icon, its ID, whether it is off or has an
409/// edit the daemon refused, and its description; a click opens it.
410fn agent_row(workspace: Workspace, agent: AgentEntry) -> impl IntoView {
411    let path = agent.path.clone();
412    let current = {
413        let path = path.clone();
414        move || {
415            workspace
416                .config_edit
417                .with(|edit| edit.as_ref().is_some_and(|edit| edit.path == path))
418        }
419    };
420    let draft = {
421        let path = path.clone();
422        move || {
423            workspace
424                .config_drafts
425                .with(|drafts| drafts.contains_key(&path))
426        }
427    };
428    let opened = path.clone();
429    let title = path.clone();
430    let summary = agent.summary;
431    let note = summary
432        .description
433        .clone()
434        .or_else(|| summary.mode.clone())
435        .unwrap_or_default();
436    let current_class = current.clone();
437    view! {
438        <div data-config-path=path.clone()>
439            <button
440                type="button"
441                class=format!("group {ROW}")
442                class:bg-sel=current_class
443                aria-current=move || current().then_some("true")
444                title=title
445                on:click=move |_| workspace.show_agent_document(&opened)
446            >
447                <span class="flex w-full items-center gap-2">
448                    <Glyph icon=mode_icon(summary.mode.as_deref()) class="text-sm opacity-80" />
449                    <span class="min-w-0 flex-1 truncate text-sm font-semibold decoration-line-2 underline-offset-[3px] group-hover:underline">{agent.id}</span>
450                    {(!summary.enabled).then(|| view! { <span class=STATE>"off"</span> })}
451                    <Show when=draft.clone()>
452                        <span class="text-xs text-accent">"not applied"</span>
453                    </Show>
454                </span>
455                <span class=format!("w-full truncate pl-6 {NOTE}")>{note}</span>
456            </button>
457        </div>
458    }
459}

461/// The open agent's editor, the agent shown read-only, or, with neither
462/// open, what the view offers.
463fn agent_detail(workspace: Workspace) -> impl IntoView {
464    let confirm_delete = RwSignal::new(false);
465    move || {
466        let Some(edit) = workspace
467            .config_edit
468            .get()
469            .filter(|edit| is_agent(&edit.path))
470        else {
471            confirm_delete.set(false);
472            if let Some(open) = workspace.agent_open.get() {
473                return read_only_agent(workspace, open).into_any();
474            }
475            return view! {
476                {move || workspace.config_error.get().map(|error| view! { <p class=ERROR id="config-error">{error}</p> })}
477                <div class="flex flex-col items-center gap-3 px-4 py-10 text-center text-sm text-muted" data-agents-empty>
478                    <Glyph icon=icondata::LuBot class="text-3xl opacity-70" />
479                    <p>"Pick an agent to see what it runs, or create one with New agent."</p>
480                    <button
481                        type="button"
482                        class=BUTTON
483                        class:lg:hidden=move || workspace.sidebar_pinned.get()
484                        data-show-agents
485                        on:click=move |_| workspace.toggle_sidebar()
486                    >
487                        <Glyph icon=icondata::LuPanelLeft />
488                        "Show the agents"
489                    </button>
490                </div>
491            }
492            .into_any();
493        };
494        let path = edit.path.clone();
495        let preset = path
496            .strip_prefix("agents/")
497            .and_then(|rest| rest.split_once('/'))
498            .map(|(preset, _)| preset.to_owned());
499        let deletable = edit.original.is_some();
500        let deleted = path.clone();
501        // The form reads the file's text as it opens, and writes each
502        // change back into it, so Save saves either.
503        let tab = workspace.agent_tab;
504        let fields = RwSignal::new(AgentSummary::default());
505        let open_form = move || {
506            fields.set(crate::agents::agent_summary(&area_value("config-text")));
507            tab.set(AgentTab::Form);
508        };
509        if tab.get_untracked() == AgentTab::Form {
510            fields.set(crate::agents::agent_summary(&edit.text));
511        }
512        // A new document has nothing applied to resolve.
513        if !deletable && tab.get_untracked() == AgentTab::Resolved {
514            tab.set(AgentTab::File);
515        }
516        let resolved_params = AgentsResolvedParams {
517            path: Some(path.clone()),
518            ..AgentsResolvedParams::default()
519        };
520        // Resolved shows the applied document: say so when the editor
521        // holds changes.
522        let unapplied = {
523            let original = edit.original.clone();
524            let text = edit.text.clone();
525            move || original.as_deref() != Some(text.as_str())
526        };
527        view! {
528            <div class=EDITOR_PAGE>
529                <div class="flex items-center gap-2">
530                    <Glyph icon=mode_icon(agent_summary(&edit.text).mode.as_deref()) class="text-lg opacity-80" />
531                    <h2 class="min-w-0 truncate text-xl font-semibold tracking-tight" id="agent-name">{agent_id(&path).to_owned()}</h2>
532                    {preset.map(|preset| view! {
533                        <span class=STATE>
534                            {format!("preset {preset}")}
535                        </span>
536                    })}
537                    {(!deletable).then(|| view! { <span class=STATE>"new"</span> })}
538                </div>
539                <div class="flex flex-wrap items-center gap-2">
540                    <p class=format!("min-w-0 flex-1 font-mono {NOTE}") id="config-path">{path.clone()}</p>
541                    // The same text three ways (docs/site/specs/web-polish.md,
542                    // P5; docs/site/specs/agent-shortcuts.md section 3.1).
543                    <div class="inline-flex rounded-lg bg-sunken p-0.5 text-xs" role="tablist" aria-label="Show as">
544                        <button type="button" role="tab" class=move || seg_class(tab.get() == AgentTab::Form) aria-selected=move || (tab.get() == AgentTab::Form).to_string() data-agent-form-tab on:click=move |_| open_form()>"Form"</button>
545                        <button type="button" role="tab" class=move || seg_class(tab.get() == AgentTab::File) aria-selected=move || (tab.get() == AgentTab::File).to_string() data-agent-file-tab on:click=move |_| tab.set(AgentTab::File)>"File"</button>
546                        <Show when=move || deletable>
547                            <button type="button" role="tab" class=move || seg_class(tab.get() == AgentTab::Resolved) aria-selected=move || (tab.get() == AgentTab::Resolved).to_string() data-agent-resolved-tab title="The document after every inherit, read-only" on:click=move |_| tab.set(AgentTab::Resolved)>
548                                <Glyph icon=icondata::LuLayers class="mr-1" />
549                                "Resolved"
550                            </button>
551                        </Show>
552                    </div>
553                </div>
554                {move || workspace.config_error.get().map(|error| view! { <p class=ERROR id="config-error">{error}</p> })}
555                <Show when=move || tab.get() == AgentTab::Form>
556                    {agent_form(fields, tab)}
557                </Show>
558                <Show when=move || tab.get() == AgentTab::Resolved>
559                    {
560                        let unapplied = unapplied.clone();
561                        let params = resolved_params.clone();
562                        view! {
563                            <Show when=unapplied.clone()>
564                                <p class=NOTE data-resolved-unapplied>"Resolved shows the applied document. Apply to see your changes."</p>

725/// The colour of the `index`th document of a chain of `length`: the accent
726/// for the first, faint for the root, a hue for one between.
727fn source_class(index: usize, length: usize) -> &'static str {
728    if length <= 1 {
729        ""
730    } else if index + 1 == length {
731        "bg-line-2"
732    } else if index == 0 {
733        "bg-accent"
734    } else {
735        "bg-hue-3 dark:bg-hue-3-dark"
736    }
737}

739/// What a key's tag says it replaced: the inherited value when it was one
740/// line, else how many lines (rules, for a permission action).
741fn replaced_note(key: &str, replaced: &str) -> String {
742    let lines = replaced.lines().collect::<Vec<_>>();
743    let value = |line: &str| {
744        let value = line
745            .split_once(':')
746            .map_or(line, |(_, value)| value)
747            .trim()
748            .trim_start_matches("- ")
749            .to_owned();
750        if value.chars().count() > 36 {
751            format!("{}…", value.chars().take(34).collect::<String>().trim_end())
752        } else {
753            value
754        }
755    };
756    match lines.as_slice() {
757        [line] => format!("was {}", value(line)),
758        [_, line] if !line.trim_start().starts_with('#') => format!("was {}", value(line.trim())),
759        _ => format!(
760            "replaces {} inherited {}",
761            lines.len().saturating_sub(1),
762            if key.starts_with("permissions.") {
763                "rules"
764            } else {
765                "lines"
766            },
767        ),
768    }
769}

771/// An agent document after every `inherit` (`agents.resolved`): its chain,
772/// then the text with each key's source marked
773/// (`docs/site/specs/agent-shortcuts.md` section 3.2).
774fn resolved_view(workspace: Workspace, params: AgentsResolvedParams) -> impl IntoView {
775    let resolved =
776        RwSignal::new(None::<(AgentsResolvedParams, Result<AgentsResolvedResult, String>)>);
777    let marks = RwSignal::new(crate::browser::load(MARKS_KEY).as_deref() != Some("off"));
778    let unfolded = RwSignal::new(false);
779    // Read again as the configuration changes, after an Apply among others.
780    {
781        let params = params.clone();
782        Effect::new(move |_| {
783            workspace.config.track();
784            workspace.read_resolved_agent(params.clone(), resolved);
785        });
786    }
787    move || {
788        let Some((asked, result)) = resolved.get() else {
789            return loading("Resolving the document…").into_any();
790        };
791        if asked != params {
792            return loading("Resolving the document…").into_any();
793        }
794        let document = match result {
795            Ok(document) => document,
796            Err(error) => {
797                return view! { <p class=ERROR data-resolved-error>{error}</p> }.into_any();
798            }
799        };
800        resolved_document(workspace, document, marks, unfolded).into_any()
801    }
802}

804fn resolved_document(
805    workspace: Workspace,
806    document: AgentsResolvedResult,
807    marks: RwSignal<bool>,
808    unfolded: RwSignal<bool>,
809) -> impl IntoView {
810    let chain = document.chain.clone();
811    let length = chain.len();
812    let index_of = {
813        let chain = chain.clone();
814        move |path: &str| {
815            chain
816                .iter()
817                .position(|each| each == path)
818                .unwrap_or(length.saturating_sub(1))
819        }
820    };
821    let pills = chain
822        .iter()
823        .enumerate()
824        .map(|(index, path)| {
825            let opened = path.clone();
826            view! {
827                {(index > 0).then(|| view! { <Glyph icon=icondata::LuArrowRight class="text-faint" /> })}
828                <button
829                    type="button"
830                    class="hit-area inline-flex items-center gap-1.5 rounded-full bg-raised px-2 py-0.5 font-mono text-xs shadow-[0_0_0_1px_var(--color-line-2)] hover:bg-sunken"
831                    title=format!("Open {path}")
832                    data-chain-document=path.clone()
833                    on:click=move |_| workspace.open_agent_path(&opened)
834                >
835                    <span class=format!("h-1.5 w-1.5 rounded-full {}", source_class(index, length))></span>
836                    {path.clone()}
837                </button>
838            }
839        })
840        .collect_view();
841    let ghost = (!document.own_document && !chain.first().is_some_and(|path| path.starts_with("official_agents/") && length == 1)).then(|| {
842        view! {
843            <span class="rounded-full px-2 py-0.5 text-xs text-muted italic shadow-[inset_0_0_0_1px_var(--color-line)]">"no document of yours"</span>
844            <Glyph icon=icondata::LuArrowRight class="text-faint" />
845        }
846    });
847    let chain_row = view! {
848        <div class="flex flex-wrap items-center gap-1.5 text-xs" data-resolved-chain>
849            <span class="mr-0.5 text-muted">{if length > 1 { "Inherits" } else { "From" }}</span>
850            {ghost}
851            {pills}
852            {(length == 1).then(|| view! { <span class=NOTE>"It inherits nothing."</span> })}
853        </div>
854    };
855    let several = length > 1;
856    let narrow = workspace.narrow.get_untracked();
857    // Each line's key, by line number.
858    let lines = document.text.lines().map(str::to_owned).collect::<Vec<_>>();
859    let key_at = |line: u32| {
860        document
861            .keys
862            .iter()
863            .find(|key| key.first_line <= line && line <= key.last_line)
864    };
865    let prompt = document.prompt.clone();
866    let values = document.values.clone();
867    let mut rows = Vec::new();
868    for (index, text) in lines.iter().enumerate() {
869        let number = u32::try_from(index + 1).unwrap_or(u32::MAX);
870        let in_prompt = prompt
871            .as_ref()
872            .is_some_and(|prompt| prompt.first_line <= number && number <= prompt.last_line);
873        if let Some(prompt) = prompt.as_ref().filter(|prompt| prompt.first_line == number) {
874            let bar = source_class(index_of(&prompt.source), length);
875            rows.push(
876                view! {
877                    <div class="flex items-center gap-2 py-0.5 pr-3 font-sans text-[0.6875rem] text-muted" data-resolved-prompt>
878                        <span class=move || format!("w-[3px] self-stretch {}", if marks.get() { bar } else { "" })></span>
879                        <span class="ml-2">{format!("The prompt of {}", prompt.source)}</span>
880                    </div>
881                }
882                .into_any(),
883            );
884        }
885        let folded = prompt.as_ref().filter(|_| in_prompt).map(|prompt| {
886            (
887                number - prompt.first_line >= PROMPT_FOLD,
888                prompt.last_line - prompt.first_line + 1,
889            )
890        });
891        let (hidden, total) = folded.unwrap_or((false, 0));
892        let bar = if in_prompt {
893            prompt
894                .as_ref()
895                .map(|prompt| source_class(index_of(&prompt.source), length))
896        } else {
897            key_at(number).map(|key| source_class(index_of(&key.source), length))
898        };
899        let tag = key_at(number)
900            .filter(|key| key.first_line == number && several)
901            .map(|key| {
902                let dot = source_class(index_of(&key.source), length);
903                let note = key.replaced.as_deref().map(|replaced| replaced_note(&key.key, replaced));
904                view! {
905                    <span class=move || format!(
906                        "{} items-center gap-1.5 font-sans text-[0.6875rem] whitespace-nowrap text-muted {}",
907                        if marks.get() { "inline-flex" } else { "hidden" },
908                        if narrow { "col-start-2 pl-3" } else { "ml-4 justify-self-end" },
909                    )>
910                        <span class=format!("h-1.5 w-1.5 rounded-full {dot}")></span>
911                        {key.source.clone()}
912                        {note.map(|note| view! { <span class="text-faint">{format!("· {note}")}</span> })}
913                    </span>
914                }
915            });
916        let value = values
917            .iter()
918            .filter(|value| text.contains(&value.name))
919            .map(|value| {
920                view! { <span class="ml-2 rounded bg-sunken px-1 font-sans text-[0.6875rem] text-muted">{format!("→ {}", value.value)}</span> }
921            })
922            .collect_view();
923        let comment = text.trim_start().starts_with('#');
924        let text = text.clone();
925        rows.push(
926            view! {
927                <div
928                    class=if narrow { "grid grid-cols-[3px_1fr] items-baseline pr-3" } else { "grid grid-cols-[3px_1fr_auto] items-baseline pr-3" }
929                    class:hidden=move || hidden && !unfolded.get()
930                >
931                    <span class=move || format!("w-[3px] self-stretch rounded-r-sm {}", if marks.get() { bar.unwrap_or("") } else { "" })></span>
932                    <span class="min-w-0 pl-3 break-words whitespace-pre-wrap" class:text-muted=comment>{text}{value}</span>
933                    {tag}
934                </div>
935            }
936            .into_any(),
937        );
938        if let Some(prompt) = prompt.as_ref().filter(|prompt| {
939            in_prompt && number == prompt.first_line + PROMPT_FOLD - 1 && total > PROMPT_FOLD
940        }) {
941            let hidden_lines = prompt.last_line - prompt.first_line + 1 - PROMPT_FOLD;
942            rows.push(
943                view! {
944                    <div class="py-0.5 pl-[15px]" class:hidden=move || unfolded.get()>
945                        <button type="button" class="font-sans text-xs text-accent hover:underline" data-resolved-unfold on:click=move |_| unfolded.set(true)>
946                            {format!("Show {hidden_lines} more lines")}
947                        </button>
948                    </div>
949                }
950                .into_any(),
951            );
952        }
953    }
954    view! {
955        <div class="flex flex-col gap-2" data-resolved>
956            {chain_row}
957            <div class="flex flex-wrap items-center justify-between gap-2">
958                {(!document.comments).then(|| view! { <span class=NOTE data-resolved-no-comments>"Shown without comments."</span> })}
959                <span class="flex-1"></span>
960                {several.then(|| view! {
961                    <label class="inline-flex cursor-pointer items-center gap-1.5 text-xs text-muted select-none">
962                        <input
963                            type="checkbox"
964                            class="accent-accent"
965                            prop:checked=move || marks.get()
966                            data-resolved-marks
967                            on:change=move |event| {
968                                let on = event_target_checked(&event);
969                                marks.set(on);
970                                crate::browser::store(MARKS_KEY, if on { "on" } else { "off" });
971                            }
972                        />
973                        "Mark where each key comes from"
974                    </label>
975                })}
976            </div>
977            <div class="overflow-x-auto rounded-lg bg-surface-2 py-2 font-mono text-[0.8125rem] leading-relaxed shadow-[inset_0_0_0_1px_var(--color-line)]" data-resolved-text>
978                {rows}
979            </div>
980        </div>
981    }
982}

993/// The agent editor's form: its description and mode, written into the
994/// file's front matter as they change (`agents::set_front_matter`); the
995/// rest of the document is the file's.
996fn agent_form(fields: RwSignal<AgentSummary>, tab: RwSignal<AgentTab>) -> impl IntoView {
997    let write = move |key: &'static str, value: String| {
998        let text = area_value("config-text");
999        match crate::agents::set_front_matter(&text, key, &value) {
1000            Some(text) => {
1001                if let Some(area) = document()
1002                    .get_element_by_id("config-text")
1003                    .and_then(|area| area.dyn_into::<web_sys::HtmlTextAreaElement>().ok())
1004                {
1005                    area.set_value(&text);
1006                }
1007            }
1008            // A key the file writes as a block: the file edits it.
1009            None => tab.set(AgentTab::File),
1010        }
1011    };
1012    let mode =
1013        move || fields.with(|fields| fields.mode.clone().unwrap_or_else(|| "primary".to_owned()));
1014    let mode_button = move |value: &'static str, label: &'static str| {
1015        view! {

1018                class=move || seg_class(mode() == value)
1019                aria-pressed=move || (mode() == value).to_string()
1020                data-agent-mode=value
1021                on:click=move |_| {
1022                    fields.update(|fields| fields.mode = Some(value.to_owned()));
1023                    write("mode", value.to_owned());
1024                }
1025            >
1026                {label}
1027            </button>
1028        }
1029    };
1030    view! {
1031        <div class="grid grid-cols-1 items-center gap-x-5 gap-y-3 rounded-xl bg-raised p-4 shadow-[0_0_0_1px_var(--color-line)] sm:grid-cols-[9rem_1fr]" data-agent-form>

1034                type="text"
1035                id="agent-form-description"
1036                class=FIELD
1037                prop:value=move || fields.with(|fields| fields.description.clone().unwrap_or_default())
1038                on:input=move |event| {
1039                    let value = event_target_value(&event);
1040                    fields.update(|fields| fields.description = Some(value.clone()));
1041                    write("description", value);
1042                }
1043            />
1044            <span class="text-[0.8125rem] text-muted">"Mode"</span>
1045            <div class="inline-flex w-fit rounded-lg bg-sunken p-0.5 text-xs">
1046                {mode_button("primary", "Primary")}
1047                {mode_button("subagent", "Subagent")}
1048                {mode_button("all", "Both")}
1049            </div>
1050            <span class="text-[0.8125rem] text-muted sm:self-start">"The rest"</span>
1051            <p class=NOTE>
1052                "Models, permissions, tool groups, and the instructions are in the file; "
1053                <button type="button" class="text-accent underline-offset-2 hover:underline" on:click=move |_| tab.set(AgentTab::File)>"edit them there"</button>
