1//! The Agents view: the agent documents in a sidebar like Chat's sessions
2//! (the shared agents, then each preset's), and the open document's
3//! editor beside it. A desktop pins the sidebar, which the sidebar button
4//! hides and shows; a narrower window shows it as the page in place of the
5//! editor. A phone shows the agents as the page until one opens. Under the
6//! documents, the ACP agents every runtime offers (`[acp.agents]`), whose
7//! form opens over the page. Above the documents, the official agents, the
8//! ACP agents, and the templates, each with where its document comes from,
9//! why it cannot run, and Customize, Turn off, Reset, and Replace
10//! (`docs/site/specs/acp-onboarding.md` section 7).

28/// The icon of an agent by the mode its document gives it.
29fn mode_icon(mode: Option<&str>) -> crate::icons::Icon {
30    match mode {
31        Some("primary") => icondata::LuBot,
32        Some("subagent") => icondata::LuUserCog,
33        Some("all") => icondata::LuUsers,
34        _ => icondata::LuFileText,
35    }
36}

38/// Whether `path` is an agent document, which this view edits.
39fn is_agent(path: &str) -> bool {
40    path.starts_with("agents/")
41}

43/// The Agents view: the sidebar, then the open agent or what to do. Below
44/// desktop widths it shows one at a time: the agents fill the page while
45/// the sidebar is open, and on a phone until one opens; the title bar's
46/// sidebar button returns to them.
47pub fn agents_page(workspace: Workspace) -> impl IntoView {
48    // A phone shows the list until an agent opens, then the agent alone;
49    // a wider window both, side by side.
50    let phone = move || workspace.narrow.get();
51    let editing = move || {
52        workspace
53            .config_edit
54            .with(|edit| edit.as_ref().is_some_and(|edit| is_agent(&edit.path)))
55    };
56    let listing = move || phone() && !editing();
57    view! {
58        <div class="flex min-h-0 flex-1">
59            <aside
60                id="agents-sidebar"
61                class="flex shrink-0 flex-col border-r border-line"
62                class:hidden=move || phone() && editing()
63                class:w-full=phone
64                class:w-72=move || !phone()
65            >
66                {move || listing().then(|| {
67                    workspace.config_error.get().map(|error| view! { <p class=format!("{ERROR} mx-3 mt-3") id="config-error">{error}</p> })
68                })}
69                <AgentList workspace=workspace />
70            </aside>
71            <div
72                class="flex min-h-0 min-w-0 flex-1 flex-col overflow-y-auto p-3"
73                class:hidden=listing
74                data-page-body
75            >
76                <div class="mx-auto flex w-full max-w-3xl flex-1 flex-col gap-2">
77                    {agent_detail(workspace)}
78                </div>
79            </div>
80        </div>
81    }
82}

84/// The agent documents by preset, with New agent and Reload above them,
85/// then the ACP agents every runtime offers.
86#[component]
87pub fn AgentList(workspace: Workspace) -> impl IntoView {
88    let creating = RwSignal::new(false);
89    // `global`'s ACP agent declarations, read once the daemon shares them.
90    let acp = super::machines::Manage::new();
91    Effect::new(move |_| {
92        if workspace.shares_acp_agents.get() {
93            acp.read_agents(
94                workspace,
95                [
96                    cookie_agent_protocol::RuntimeId::new(cookie_agent_protocol::RuntimeId::GLOBAL)
97                        .expect("`global` is a runtime ID"),
98                ],
99            );
100        }
101    });
102    let create = move || workspace.new_agent(&field_value("config-new-path"));
103    // The panel's rows follow the configuration.
104    Effect::new(move |_| {
105        workspace.config.track();
106        workspace.poll_agent_documents();
107    });
108    view! {
109        <div class="flex flex-col gap-2 p-3">
110            <div class="flex gap-2">
111                <button
112                    type="button"
113                    class=format!("{PRIMARY} flex-1")
114                    data-new-agent
115                    aria-expanded=move || creating.get().to_string()
116                    on:click=move |_| creating.update(|creating| *creating = !*creating)
117                >
118                    <Glyph icon=icondata::LuPlus />
119                    "New agent"
120                </button>
121                <button
122                    type="button"
123                    id="config-reload"
124                    class=ICON_BUTTON
125                    aria-label="Reload from disk"
126                    title="Reload the documents from disk, after editing them by hand"
127                    on:click=move |_| workspace.reload_config()
128                >
129                    <Glyph icon=icondata::LuRefreshCw />
130                </button>
131            </div>
132            <Show when=move || creating.get()>
133                <label class="flex flex-col gap-1 text-xs font-bold text-muted">
134                    <span>"Document"</span>
135                    <input
136                        type="text"
137                        id="config-new-path"
138                        class=FIELD
139                        value="agents/"
140                        on:keydown=move |event: leptos::ev::KeyboardEvent| {
141                            if event.key() == "Enter" {
142                                event.prevent_default();
143                                create();
144                            }
145                        }
146                    />
147                </label>
148                <p class=NOTE>"agents/<id>.md, or agents/<preset>/<id>.md for a preset's agent"</p>
149                <div class="flex gap-2">
150                    <button type="button" id="config-new" class=PRIMARY on:click=move |_| create()>
151                        "Create"
152                    </button>
153                    <button type="button" class=BUTTON on:click=move |_| creating.set(false)>
154                        "Cancel"
155                    </button>
156                </div>
157            </Show>
158        </div>
159        <nav class="flex min-h-0 flex-1 flex-col gap-1 overflow-y-auto px-2 pb-3" aria-label="Agents">
160            {agent_panel(workspace)}
161            {move || {
162                let config = match workspace.config.get() {
163                    None => return loading("Reading the agents…").into_any(),
164                    Some(Err(error)) => return view! { <p class=ERROR>{error}</p> }.into_any(),
165                    Some(Ok(config)) => config,
166                };
167                let groups = agent_groups(&config);
168                if groups.is_empty() {
169                    return empty_state(icondata::LuBot, "No agent documents yet. Create one with New agent.")
170                        .into_any();
171                }
172                groups
173                    .into_iter()
174                    .map(|group| {
175                        let (icon, heading) = match &group.preset {
176                            None => (icondata::LuUsers, "Shared agents".to_owned()),
177                            Some(preset) => (icondata::LuLayers, format!("Preset {preset}")),
178                        };
179                        let empty = group.agents.is_empty();
180                        let rows = group
181                            .agents
182                            .into_iter()
183                            .map(|agent| agent_row(workspace, agent))
184                            .collect_view();
185                        view! {
186                            <h3
187                                class="mt-2 flex items-center gap-2 px-3 text-xs font-bold tracking-wide text-muted uppercase"
188                                data-preset=group.preset.clone().unwrap_or_default()
189                            >
190                                <Glyph icon=icon class="opacity-70" />
191                                <span class="min-w-0 truncate">{heading}</span>
192                            </h3>
193                            {empty.then(|| view! { <p class=format!("{NOTE} px-3")>"No agents in this preset."</p> })}
194                            {rows}
195                        }
196                    })
197                    .collect_view()
198                    .into_any()
199            }}
200            <Show when=move || workspace.shares_acp_agents.get()>
201                <div class="mt-3">{super::machines::shared_agents(workspace, acp)}</div>
202            </Show>
203        </nav>
204        {super::machines::acp_dialog(workspace, acp, "agents-acp-form")}
205    }
206}

208/// The official agents, the ACP agents, and the templates, by group, each
209/// with its actions.
210fn agent_panel(workspace: Workspace) -> impl IntoView {
211    use cookie_agent_client_state::agent_documents::{
212        AgentDocumentGroup, actions, group, origin_label, unavailable_line,
213    };
214    // An action that asks first, waiting for its second click.
215    let pending = RwSignal::new(None::<(String, AgentDocumentAction)>);
216    // The one open ⋯ menu among the rows.
217    let menu = RwSignal::new(None::<String>);
218    move || {
219        let documents = match workspace.agent_documents.get() {
220            None => return None,
221            Some(Err(error)) => return Some(view! { <p class=ERROR>{error}</p> }.into_any()),
222            Some(Ok(documents)) => documents,
223        };
224        let agents = workspace.with_runtime(|runtime| {
225            runtime
226                .snapshot()
227                .map(|snapshot| snapshot.agents.clone())
228                .unwrap_or_default()
229        });
230        let sections = [
231            AgentDocumentGroup::Official,
232            AgentDocumentGroup::Acp,
233            AgentDocumentGroup::Templates,
234        ]
235        .into_iter()
236        .filter_map(|each| {
237            let rows = documents
238                .iter()
239                .filter(|info| info.preset.is_none() && group(info) == each)
240                .cloned()
241                .collect::<Vec<_>>();
242            (!rows.is_empty()).then(|| {
243                let rows = rows
244                    .into_iter()
245                    .map(|info| {
246                        let reason = unavailable_line(&info, &agents);
247                        let description = agents
248                            .iter()
249                            .find(|agent| agent.id.as_str() == info.id.as_str())
250                            .map(|agent| agent.description.clone())
251                            .filter(|description| !description.is_empty());
252                        // An action that asks first asks on the row.
253                        let asking = {
254                            let id = info.id.clone();
255                            move || pending.get().filter(|(pending, _)| *pending == id).map(|(_, action)| action)
256                        };
257                        let confirm = {
258                            let (id, preset, path) = (info.id.clone(), info.preset.clone(), info.path.clone());
259                            move || {
260                                asking().map(|action| {
261                                    let (id, preset, path) = (id.clone(), preset.clone(), path.clone());
262                                    view! {
263                                        <span class="flex flex-wrap items-center gap-2 text-xs" data-agent-confirm>
264                                            <span class="text-muted">{format!("{}: this overwrites your document.", action.label())}</span>
265                                            <button
266                                                type="button"
267                                                class=format!("{PANEL_BUTTON} text-bad")
268                                                data-agent-action=format!("{:?}", action).to_lowercase()
269                                                on:click=move |_| {
270                                                    pending.set(None);
271                                                    workspace.agent_document_action(action, id.clone(), preset.clone(), path.clone());
272                                                }
273                                            >
274                                                {action.label()}
275                                            </button>
276                                            <button type="button" class=PANEL_BUTTON on:click=move |_| pending.set(None)>"Cancel"</button>
277                                        </span>
278                                    }
279                                })
280                            }
281                        };
282                        // A right-click offers the same actions; one that
283                        // asks first asks on the row, as its button does.
284                        let context = RwSignal::new(None);
285                        let context_items = {
286                            let info = info.clone();
287                            move || {
288                                actions(&info)
289                                    .into_iter()
290                                    .map(|action| {
291                                        let (id, preset, path) =
292                                            (info.id.clone(), info.preset.clone(), info.path.clone());
293                                        super::row_menu::MenuItem {
294                                            id: "agent-action",
295                                            label: action.label(),
296                                            icon: icondata::LuBot,
297                                            act: Callback::new(move |()| {
298                                                if action.confirms() {
299                                                    pending.set(Some((id.clone(), action)));
300                                                } else {
301                                                    workspace.agent_document_action(
302                                                        action,
303                                                        id.clone(),
304                                                        preset.clone(),
305                                                        path.clone(),
306                                                    );
307                                                }
308                                            }),
309                                            danger: action.confirms(),
310                                        }
311                                    })
312                                    .collect::<Vec<_>>()
313                            }
314                        };
315                        let menu_items = context_items.clone();
316                        view! {
317                            {super::row_menu::context_menu(context, (), info.id.clone(), context_items)}
318                            // Its name, what it does, and its actions in ⋯
319                            // (docs/site/specs/web-polish.md, P5).
320                            <div
321                                class="group flex flex-col gap-0.5 rounded-lg px-3 py-1.5 transition-colors hover:bg-hover"
322                                data-agent-document=info.id.clone()
323                                on:contextmenu=super::row_menu::on_context(context)
324                            >
325                                <span class="flex w-full items-center gap-2">
326                                    <span class="min-w-0 truncate text-sm font-semibold">{info.id.clone()}</span>
327                                    <span class=STATE>{origin_label(&info)}</span>
328                                    <span class="flex-1"></span>
329                                    {super::row_menu::row_menu(menu, info.id.clone(), info.id.clone(), info.id.clone(), ROW_MENU_BUTTON, menu_items)}
330                                </span>
331                                {description.map(|description| view! { <span class=format!("{NOTE} line-clamp-2") data-agent-description>{description}</span> })}
332                                {reason.map(|reason| view! { <span class=NOTE data-agent-unavailable>{reason}</span> })}
333                                {confirm}
334                            </div>
335                        }
336                    })
337                    .collect_view();
338                view! {
339                    <h3 class="mt-2 flex items-center gap-2 px-3 text-xs font-bold tracking-wide text-muted uppercase">
340                        {each.label()}
341                    </h3>
342                    {rows}
343                }
344            })
345        })
346        .collect_view();
347        Some(
348            view! { <section id="agent-panel" class="flex flex-col gap-1">{sections}</section> }
349                .into_any(),
350        )
351    }
352}

360/// An agent's row: its mode's icon, its ID, whether it is off or has an
361/// edit the daemon refused, and its description; a click opens it.
362fn agent_row(workspace: Workspace, agent: AgentEntry) -> impl IntoView {
363    let path = agent.path.clone();
364    let current = {
365        let path = path.clone();
366        move || {
367            workspace
368                .config_edit
369                .with(|edit| edit.as_ref().is_some_and(|edit| edit.path == path))
370        }
371    };
372    let draft = {
373        let path = path.clone();
374        move || {
375            workspace
376                .config_drafts
377                .with(|drafts| drafts.contains_key(&path))
378        }
379    };
380    let opened = path.clone();
381    let title = path.clone();
382    let summary = agent.summary;
383    let note = summary
384        .description
385        .clone()
386        .or_else(|| summary.mode.clone())
387        .unwrap_or_default();
388    let current_class = current.clone();
389    view! {
390        <div data-config-path=path.clone()>
391            <button
392                type="button"
393                class=ROW
394                class:bg-sel=current_class
395                aria-current=move || current().then_some("true")
396                title=title
397                on:click=move |_| {
398                    workspace.edit_config(&opened);
399                    workspace.sidebar_open.set(false);
400                }
401            >
402                <span class="flex w-full items-center gap-2">
403                    <Glyph icon=mode_icon(summary.mode.as_deref()) class="text-sm opacity-80" />
404                    <span class="min-w-0 flex-1 truncate text-sm font-semibold">{agent.id}</span>
405                    {(!summary.enabled).then(|| view! { <span class=STATE>"off"</span> })}
406                    <Show when=draft.clone()>
407                        <span class="text-xs text-accent">"not applied"</span>
408                    </Show>
409                </span>
410                <span class=format!("w-full truncate pl-6 {NOTE}")>{note}</span>
411            </button>
412        </div>
413    }
414}

416/// The open agent's editor, or, with none open, what the view offers.
417fn agent_detail(workspace: Workspace) -> impl IntoView {
418    let confirm_delete = RwSignal::new(false);
419    move || {
420        let Some(edit) = workspace
421            .config_edit
422            .get()
423            .filter(|edit| is_agent(&edit.path))
424        else {
425            confirm_delete.set(false);
426            return view! {
427                {move || workspace.config_error.get().map(|error| view! { <p class=ERROR id="config-error">{error}</p> })}
428                <div class="flex flex-col items-center gap-3 px-4 py-10 text-center text-sm text-muted" data-agents-empty>
429                    <Glyph icon=icondata::LuBot class="text-3xl opacity-70" />
430                    <p>"Pick an agent to edit its document, or create one with New agent."</p>
431                    <button
432                        type="button"
433                        class=BUTTON
434                        class:lg:hidden=move || workspace.sidebar_pinned.get()
435                        data-show-agents
436                        on:click=move |_| workspace.toggle_sidebar()
437                    >
438                        <Glyph icon=icondata::LuPanelLeft />
439                        "Show the agents"
440                    </button>
441                </div>
442            }
443            .into_any();
444        };
445        let path = edit.path.clone();
446        let preset = path
447            .strip_prefix("agents/")
448            .and_then(|rest| rest.split_once('/'))
449            .map(|(preset, _)| preset.to_owned());
450        let deletable = edit.original.is_some();
451        let deleted = path.clone();
452        // The form reads the file's text as it opens, and writes each
453        // change back into it, so Save saves either.
454        let form = RwSignal::new(false);
455        let fields = RwSignal::new(AgentSummary::default());
456        let open_form = move || {
457            fields.set(crate::agents::agent_summary(&area_value("config-text")));
458            form.set(true);
459        };
460        view! {
461            <div class=EDITOR_PAGE>
462                <div class="flex items-center gap-2">
463                    <Glyph icon=mode_icon(agent_summary(&edit.text).mode.as_deref()) class="text-lg opacity-80" />
464                    <h2 class="min-w-0 truncate text-xl font-semibold tracking-tight" id="agent-name">{agent_id(&path).to_owned()}</h2>
465                    {preset.map(|preset| view! {
466                        <span class=STATE>
467                            {format!("preset {preset}")}
468                        </span>
469                    })}
470                    {(!deletable).then(|| view! { <span class=STATE>"new"</span> })}
471                </div>
472                <div class="flex flex-wrap items-center gap-2">
473                    <p class=format!("min-w-0 flex-1 font-mono {NOTE}") id="config-path">{path.clone()}</p>
474                    // The same text two ways (docs/site/specs/web-polish.md, P5).
475                    <div class="inline-flex rounded-lg bg-sunken p-0.5 text-xs" role="tablist" aria-label="Edit as">
476                        <button type="button" role="tab" class=move || seg_class(form.get()) aria-selected=move || form.get().to_string() data-agent-form-tab on:click=move |_| open_form()>"Form"</button>
477                        <button type="button" role="tab" class=move || seg_class(!form.get()) aria-selected=move || (!form.get()).to_string() data-agent-file-tab on:click=move |_| form.set(false)>"File"</button>
478                    </div>
479                </div>
480                {move || workspace.config_error.get().map(|error| view! { <p class=ERROR id="config-error">{error}</p> })}
481                <Show when=move || form.get()>
482                    {agent_form(fields, form)}
483                </Show>
484                <textarea id="config-text" class=EDITOR class:hidden=move || form.get() spellcheck="false">{edit.text.clone()}</textarea>
485                <Show when=move || confirm_delete.get()>
486                    <div class=CONFIRM role="alertdialog">
487                        <p class="text-sm">{format!("Delete {}? This cannot be undone.", deleted.clone())}</p>
488                        <div class="flex gap-2">
489                            <button
490                                type="button"
491                                id="config-delete-confirm"
492                                class=PRIMARY
493                                on:click={
494                                    let deleted = deleted.clone();
495                                    move |_| {
496                                        confirm_delete.set(false);
497                                        workspace.delete_config(deleted.clone());
498                                    }
499                                }
500                            >
501                                "Delete"
502                            </button>
503                            <button type="button" class=BUTTON on:click=move |_| confirm_delete.set(false)>
504                                "Keep it"
505                            </button>
506                        </div>
507                    </div>
508                </Show>
509                <div class=EDITOR_ACTIONS>
510                    <button
511                        type="button"
512                        id="config-apply"
513                        class=PRIMARY
514                        on:click=move |_| workspace.apply_config(area_value("config-text"))
515                    >
516                        "Apply"
517                    </button>
518                    <button type="button" class=BUTTON on:click=move |_| workspace.config_edit.set(None)>
519                        "Close"
520                    </button>
521                    <span class="flex-1"></span>
522                    <Show when=move || deletable>
523                        <button
524                            type="button"
525                            class=BUTTON
526                            data-config-delete
527                            on:click=move |_| confirm_delete.set(true)
528                        >
529                            <Glyph icon=icondata::LuTrash2 />
530                            "Delete"
531                        </button>
532                    </Show>
533                </div>
534            </div>
535        }
536        .into_any()
537    }

540/// A segment of the editor's Form | File switch.
541fn seg_class(on: bool) -> &'static str {
542    if on {
543        "inline-flex min-h-11 items-center rounded-md bg-raised px-3 font-medium text-fg shadow-[0_1px_2px_rgb(0_0_0/0.08)] lg:min-h-0 lg:px-2.5 lg:py-1"

549/// The agent editor's form: its description and mode, written into the
550/// file's front matter as they change (`agents::set_front_matter`); the
551/// rest of the document is the file's.
552fn agent_form(fields: RwSignal<AgentSummary>, form: RwSignal<bool>) -> impl IntoView {
553    let write = move |key: &'static str, value: String| {
554        let text = area_value("config-text");
555        match crate::agents::set_front_matter(&text, key, &value) {
556            Some(text) => {
557                if let Some(area) = document()
558                    .get_element_by_id("config-text")
559                    .and_then(|area| area.dyn_into::<web_sys::HtmlTextAreaElement>().ok())
560                {
561                    area.set_value(&text);
562                }
563            }
564            // A key the file writes as a block: the file edits it.
565            None => form.set(false),
566        }
567    };
568    let mode =
569        move || fields.with(|fields| fields.mode.clone().unwrap_or_else(|| "primary".to_owned()));
570    let mode_button = move |value: &'static str, label: &'static str| {
571        view! {
572            <button
573                type="button"
574                class=move || seg_class(mode() == value)
575                aria-pressed=move || (mode() == value).to_string()
576                data-agent-mode=value
577                on:click=move |_| {
578                    fields.update(|fields| fields.mode = Some(value.to_owned()));
579                    write("mode", value.to_owned());
580                }
581            >
582                {label}
583            </button>
584        }
585    };
586    view! {
