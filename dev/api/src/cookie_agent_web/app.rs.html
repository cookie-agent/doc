23/// How long a check of the connection after the tab became visible may take
24/// before the page redials.
25const VISIBLE_CHECK_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
26
27/// Where the page's connection stands.
28#[derive(Clone, Debug, PartialEq)]
29pub(crate) enum Phase {
30    /// Waiting for a URL and a token.
31    Idle,
32    Connecting,
33    Connected {
34        address: String,
35        engine_version: String,
36        protocol_version: String,
37    },
38    /// The connection dropped; the page redials with the shared backoff.
39    Reconnecting {
40        reason: String,
41    },
42    /// Connecting failed; `hint` says what to change.
43    Failed {
44        message: String,
45        hint: Option<String>,
46    },
47}

49/// The page's connection and its inputs, shared by the components.
50#[derive(Clone, Copy)]
51pub(crate) struct Connection {
52    url: RwSignal<String>,
53    token: RwSignal<String>,
54    phase: RwSignal<Phase>,
55    identity: StoredValue<ClientIdentity>,
56    client: StoredValue<Option<Client>>,
57    /// Bumped by every connect and forget, so a superseded attempt or
58    /// watcher stops.
59    generation: StoredValue<u64>,
60    /// What the page shows of the daemon; it outlives a redial.
61    workspace: Workspace,
62    /// The workspace has had a connection, so a redial keeps it on screen.
63    attached: RwSignal<bool>,
64}
65
66impl Connection {
67    fn new() -> Self {
68        let filled = browser::page_href()
69            .as_deref()
70            .and_then(daemon_url_for_page)
71            .unwrap_or_default();
72        let Launch {
73            url,
74            token,
75            store,
76            fragment,
77        } = launch(
78            &filled,
79            browser::load(DAEMON_URL_KEY).as_deref(),
80            &browser::page_fragment().unwrap_or_default(),
81            |url| browser::load(&token_key(url)),
82        );
83        // A token in the fragment is kept for the daemon and taken out of
84        // the address bar, so it never lands in history or a bookmark.
85        if let Some(token) = store {
86            browser::store(&token_key(&url), &token);
87        }
88        if let Some(fragment) = fragment {
89            browser::replace_fragment(&fragment);
90        }
91        Self {
92            url: RwSignal::new(url),
93            token: RwSignal::new(token),
94            phase: RwSignal::new(Phase::Idle),
95            identity: StoredValue::new(connection::tab_identity()),
96            client: StoredValue::new(None),
97            generation: StoredValue::new(0),
98            workspace: Workspace::new(),
99            attached: RwSignal::new(false),
100        }
101    }
102
103    fn next_generation(self) -> u64 {
104        self.generation.update_value(|generation| *generation += 1);
105        self.generation.get_value()
106    }
107
108    fn is_current(self, generation: u64) -> bool {
109        self.generation.get_value() == generation
110    }
111
112    /// Connects with the URL and token the screen holds.
113    fn connect(self) {
114        let url = self.url.get_untracked().trim().to_owned();
115        let token = token_from_input(&self.token.get_untracked());
116        if token != self.token.get_untracked() {
117            self.token.set(token.clone());
118        }
119        if !connects_by_itself(Some(&url), Some(&token)) {
120            self.phase.set(Phase::Failed {
121                message: "Enter the daemon URL and its token.".into(),
122                hint: None,
123            });
124            return;
125        }
126        self.drop_client();
127        let generation = self.next_generation();
128        self.phase.set(Phase::Connecting);
129        spawn_local(async move {
130            let opened = connection::open(&url, &token, self.identity.get_value()).await;
131            if !self.is_current(generation) {
132                return;
133            }
134            match opened {
135                Ok((client, hello)) => {
136                    browser::store(DAEMON_URL_KEY, &url);
137                    browser::store(&token_key(&url), &token);
138                    self.adopt(generation, &url, client, &hello);
139                }
140                Err(error) => self.phase.set(failure(&error)),
141            }
142        });
143    }
144
145    /// Makes `client` the page's connection and watches it for a drop.
146    fn adopt(self, generation: u64, url: &str, client: Client, hello: &ServerHello) {
147        let deliveries = client.subscribe_deliveries();
148        self.client.set_value(Some(client.clone()));
149        self.workspace.adopt(client);
150        self.attached.set(true);
151        self.phase.set(Phase::Connected {
152            address: url.to_owned(),
153            engine_version: hello.engine_version.clone(),
154            protocol_version: hello.protocol_version.to_string(),
155        });
156        let Some(mut deliveries) = deliveries else {
157            return;
158        };
159        spawn_local(async move {
160            let reason = loop {
161                match deliveries.recv().await {
162                    Some(ClientDelivery::Disconnected { error }) => break error.to_string(),
163                    Some(delivery) if self.is_current(generation) => {
164                        self.workspace.handle(delivery);
165                    }
166                    Some(_) => {}
167                    None => break "the connection closed".to_owned(),
168                }
169            };
170            if self.is_current(generation) {
171                self.reconnect(generation, reason);
172            }
173        });
174    }
175
176    /// Redials after a drop with the shared backoff and the tab's identity,
177    /// until it connects or the user connects or forgets anew.
178    fn reconnect(self, generation: u64, reason: String) {
179        self.drop_client();
180        self.workspace.disconnected();
181        self.phase.set(Phase::Reconnecting { reason });
182        let url = self.url.get_untracked();
183        let token = self.token.get_untracked();
184        spawn_local(async move {
185            let mut delay = RECONNECT_FIRST_DELAY;
186            loop {
187                gloo_timers::future::sleep(delay).await;
188                if !self.is_current(generation) {
189                    return;
190                }
191                match connection::open(&url, &token, self.identity.get_value()).await {
192                    Ok((client, hello)) if self.is_current(generation) => {
193                        self.adopt(generation, &url, client, &hello);
194                        return;
195                    }
196                    Ok((client, _)) => {
197                        client.shutdown();
198                        return;
199                    }
200                    Err(error) => {
201                        if !self.is_current(generation) {
202                            return;
203                        }
204                        self.phase.set(Phase::Reconnecting {
205                            reason: crate::errors::describe(&error),
206                        });
207                        delay = next_reconnect_delay(delay);
208                    }
209                }
210            }
211        });
212    }
213
214    /// Checks the connection once the tab is visible again, as a phone
215    /// suspends background tabs, and redials when it does not answer.
216    fn check(self) {
217        let Some(client) = self.client.get_value() else {
218            return;
219        };
220        let generation = self.generation.get_value();
221        spawn_local(async move {
222            let answer = std::pin::pin!(client.runtime_snapshot());
223            let timeout = gloo_timers::future::sleep(VISIBLE_CHECK_TIMEOUT);
224            let answered = matches!(
225                futures_util::future::select(answer, timeout).await,
226                futures_util::future::Either::Left((Ok(_), _))
227            );
228            if !answered && self.is_current(generation) {
229                let generation = self.next_generation();
230                self.reconnect(generation, "the daemon stopped answering".into());
231            }
232        });
233    }
234
235    /// Deletes the stored token and disconnects.
236    pub(crate) fn forget(self) {
237        self.next_generation();
238        self.drop_client();
239        browser::forget(&token_key(self.url.get_untracked().trim()));
240        self.token.set(String::new());
241        self.attached.set(false);
242        self.phase.set(Phase::Idle);
243    }
244
245    fn drop_client(self) {
246        if let Some(client) = self.client.get_value() {
247            client.shutdown();
248        }
249        self.client.set_value(None);
250    }
251}
252
253/// What the connect screen shows for a failed connection.
254fn failure(error: &ClientError) -> Phase {
255    match error {
256        ClientError::ConnectionRefusedBeforeHandshake { .. } => Phase::Failed {
257            message: crate::errors::describe(error),
258            hint: Some(refused_before_handshake_hint(
259                &browser::page_origin().unwrap_or_default(),
260            )),
261        },
262        _ if error.error_code() == Some("unauthenticated") => Phase::Failed {
263            message: "The daemon refused the token.".into(),
264            hint: Some(
265                "Run `cookie web` on the daemon's machine for a link with its token, \
266                 or paste the token from its ready line."
267                    .into(),
268            ),
269        },
270        _ => Phase::Failed {
271            message: crate::errors::describe(error),
272            hint: None,
273        },
274    }
275}

277/// The page.
278#[component]
279pub fn App() -> impl IntoView {
280    let theme = RwSignal::new(ThemePreference::parse(browser::load(THEME_KEY).as_deref()));
281    let system_dark = RwSignal::new(browser::system_prefers_dark());
282    browser::on_system_scheme_change(move || system_dark.set(browser::system_prefers_dark()));
283    Effect::new(move || {
284        let preference = theme.get();
285        browser::set_dark(preference.is_dark(system_dark.get()));
286        browser::store(THEME_KEY, preference.as_str());
287    });
288
289    Install::provide();
290    let connection = Connection::new();
291    browser::on_visible(move || connection.check());
292    browser::follow_visual_viewport();
293    if connects_by_itself(
294        Some(&connection.url.get_untracked()),
295        Some(&connection.token.get_untracked()),
296    ) {
297        connection.connect();
298    }
299
300    view! {
301        <div class="app-frame flex flex-col overflow-hidden bg-cream text-espresso dark:bg-night dark:text-latte">
302            {move || {
303                let phase = connection.phase.get();
304                let attached = connection.attached.get();
305                match phase {
306                    Phase::Connected { .. } => {
307                        view! {
308                            <InstallBanner />
309                            <WorkspaceView workspace=connection.workspace connection=connection theme=theme />
310                        }
311                            .into_any()
312                    }
313                    Phase::Reconnecting { reason } if attached => {
314                        view! {
315                            <div id="reconnecting" role="status" class="flex items-center gap-2 border-b border-caramel bg-toasted px-3 py-2 text-sm dark:border-maple dark:bg-ember">
316                                <Glyph icon=icondata::FaArrowsRotateSolid class="animate-spin text-caramel dark:text-maple" />
317                                <span class="min-w-0 flex-1">{format!("Connection lost ({reason}); reconnecting…")}</span>
318                            </div>
319                            <InstallBanner />
320                            <WorkspaceView workspace=connection.workspace connection=connection theme=theme />
321                        }
322                            .into_any()
323                    }
324                    phase => {
325                        view! {
326                            <Header connection=connection theme=theme />
327                            <main class="flex flex-1 items-start justify-center overflow-y-auto px-4 py-10">
328                                <ConnectScreen connection=connection phase=phase />
329                            </main>
330                        }
331                            .into_any()
332                    }
333                }
334            }}
335        </div>
336    }
337}

389/// The connection's state as a small indicator; a click shows the daemon's
390/// details, the way to forget it, and the page's credits. `compact` shows
391/// only the icon (the rail's); `primary` gives it the ids the page's
392/// scripts find it by, for the one instance that is always there.
393#[component]
394pub(crate) fn ConnectionStatus(
395    connection: Connection,
396    compact: bool,
397    primary: bool,
398) -> impl IntoView {
399    let open = RwSignal::new(false);
400    let confirming = RwSignal::new(false);
401    let close = move || {
402        open.set(false);
403        confirming.set(false);
404    };
405    let indicator = move || {
406        let (icon, label, class) = match connection.phase.get() {
407            Phase::Connected { .. } => (icondata::FaCircleSolid, "connected", "text-[0.5rem]"),
408            Phase::Connecting => (
409                icondata::FaCircleNotchSolid,
410                "connecting",
411                "animate-spin text-sm",
412            ),
413            Phase::Reconnecting { .. } => (
414                icondata::FaArrowsRotateSolid,
415                "reconnecting",
416                "animate-spin text-sm",
417            ),
418            Phase::Idle | Phase::Failed { .. } => {
419                (icondata::FaPlugCircleXmarkSolid, "offline", "text-sm")
420            }
421        };
422        view! {
423            <Glyph icon=icon class=class />
424            <span
425                class=if compact { "sr-only" } else { "text-xs" }
426                id=(primary && label == "connected").then_some("connected")
427            >
428                {label}
429            </span>
430        }
431    };
432    let tone = move || match connection.phase.get() {
433        Phase::Connected { .. } => "text-green-700 dark:text-green-400",
434        Phase::Connecting | Phase::Reconnecting { .. } => "text-caramel dark:text-maple",
435        Phase::Idle | Phase::Failed { .. } => "text-ink dark:text-sand",
436    };
437    let details = move || match connection.phase.get() {
438        Phase::Connected {
439            address,
440            engine_version,
441            protocol_version,
442        } => vec![
443            ("Daemon", address),
444            ("Version", format!("cookie {engine_version}")),
445            ("Client protocol", protocol_version),
446        ],
447        Phase::Connecting => vec![("State", "connecting…".to_owned())],
448        Phase::Reconnecting { reason } => vec![("Connection lost", reason)],
449        Phase::Idle => vec![("State", "not connected".to_owned())],
450        Phase::Failed { message, .. } => vec![("Connecting failed", message)],
451    };
452    let summary = move || {
453        details()
454            .into_iter()
455            .map(|(name, value)| format!("{name}: {value}"))
456            .collect::<Vec<_>>()
457            .join("\n")
458    };
459    view! {
460        <div class="relative">
461            <button
462                type="button"
463                id=primary.then_some("connection")
464                aria-label=move || {
465                    let state = match connection.phase.get() {
466                        Phase::Connected { .. } => "connected",
467                        Phase::Connecting => "connecting",
468                        Phase::Reconnecting { .. } => "reconnecting",
469                        Phase::Idle | Phase::Failed { .. } => "offline",
470                    };
471                    format!("Connection: {state}")
472                }
473                class=move || format!("inline-flex min-h-11 min-w-11 items-center justify-center gap-1.5 rounded-full hover:bg-toasted active:bg-crust dark:hover:bg-ember dark:active:bg-smoke {} {}", if compact { "" } else { "px-3" }, tone())
474                aria-haspopup="dialog"
475                aria-expanded=move || open.get().to_string()
476                title=summary
477                on:click=move |_| {
478                    if open.get_untracked() { close() } else { open.set(true) }
479                }
480            >
481                {indicator}
482            </button>
483            <Show when=move || open.get()>
484                <div class="fixed inset-0 z-30" on:click=move |_| close()></div>
485                <div
486                    id="connection-details"
487                    role="dialog"
488                    aria-label="Connection"
489                    class=if compact {
490                        "fixed bottom-3 left-3 z-40 flex w-80 max-w-[calc(100vw-1.5rem)] flex-col gap-3 rounded-lg border border-toasted bg-cream p-3 text-sm shadow-lg sm:left-16 dark:border-ember dark:bg-night"
491                    } else {
492                        "absolute top-full right-0 z-40 mt-1 flex w-80 max-w-[calc(100vw-1.5rem)] flex-col gap-3 rounded-lg border border-toasted bg-cream p-3 text-sm shadow-lg dark:border-ember dark:bg-night"
493                    }
494                >
495                    <div class="flex items-center gap-2">
496                        <p class="flex-1 font-bold">"Connection"</p>
497                        <button
498                            type="button"
499                            id="connection-close"
500                            class=HEADER_BUTTON
501                            aria-label="Close"
502                            title="Close"
503                            on:click=move |_| close()
504                        >
505                            <Glyph icon=icondata::FaXmarkSolid />
506                        </button>
507                    </div>
508                    <dl class="grid grid-cols-[auto_1fr] gap-x-3 gap-y-1">
509                        {move || {
510                            details()
511                                .into_iter()
512                                .map(|(name, value)| {
513                                    view! {
514                                        <dt class="text-ink dark:text-sand">{name}</dt>
515                                        <dd class="min-w-0 font-mono text-xs break-all">{value}</dd>
516                                    }
517                                })
518                                .collect_view()
519                        }}
520                    </dl>
521                    <Show when=move || connection.attached.get()>
522                        <ForgetDaemon connection=connection confirming=confirming primary=primary on_forget=close />
523                    </Show>
524                    <Credits />
525                </div>
526            </Show>
527        </div>
528    }
529}
