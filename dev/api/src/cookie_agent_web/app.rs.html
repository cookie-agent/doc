29/// How long a check of the connection after the tab became visible may take
30/// before the page redials.
31const VISIBLE_CHECK_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
32
33/// How often the page checks whether the user went idle, which ends what
34/// it told the daemon it shows.
35const PRESENCE_CHECK: std::time::Duration = std::time::Duration::from_secs(30);

37/// Where the page's connection stands.
38#[derive(Clone, Debug, PartialEq)]
39pub(crate) enum Phase {
40    /// Waiting for a URL and a token.
41    Idle,
42    Connecting,
43    Connected {
44        address: String,
45        engine_version: String,
46        protocol_version: String,
47    },
48    /// The connection dropped; the page redials with the shared backoff.
49    Reconnecting {
50        reason: String,
51    },
52    /// Connecting failed; `hint` says what to change.
53    Failed {
54        message: String,
55        hint: Option<String>,
56    },
57}

59/// The page's connection and its inputs, shared by the components.
60#[derive(Clone, Copy)]
61pub(crate) struct Connection {
62    url: RwSignal<String>,
63    token: RwSignal<String>,
64    phase: RwSignal<Phase>,
65    identity: StoredValue<ClientIdentity>,
66    client: StoredValue<Option<Client>>,
67    /// Bumped by every connect and forget, so a superseded attempt or
68    /// watcher stops.
69    generation: StoredValue<u64>,
70    /// What the page shows of the daemon; it outlives a redial.
71    workspace: Workspace,
72    /// The workspace has had a connection, so a redial keeps it on screen.
73    attached: RwSignal<bool>,
74    /// The session a notification asked to open, until a connection opens
75    /// it.
76    pending_session: StoredValue<Option<String>>,
77    /// The asset or attached file a `#…&asset=` or `#…&attachment=` link
78    /// asked to show once its session opens.
79    pending_asset: StoredValue<Option<push::LinkTarget>>,
80    /// The daemon's status (client protocol 1.65.0): how its start goes
81    /// while the page waits for it, then what it is busy with.
82    daemon: RwSignal<Option<DaemonStatus>>,
83    /// This page is a conversation share's guest page.
84    pub(crate) is_guest: bool,
85    /// The link stopped working, or never did.
86    guest_gone: RwSignal<bool>,
87}

89impl Connection {
90    fn new() -> Self {
91        let guest_socket = browser::page_href()
92            .as_deref()
93            .and_then(crate::guest::socket_url);
94        let filled = browser::page_href()
95            .as_deref()
96            .and_then(daemon_url_for_page)
97            .unwrap_or_default();
98        let address_fragment = browser::page_fragment().unwrap_or_default();
99        let Launch {
100            mut url,
101            mut token,
102            store,
103            fragment,
104        } = launch(
105            &filled,
106            browser::load(DAEMON_URL_KEY).as_deref(),
107            &address_fragment,
108            |url| browser::load(&token_key(url)),
109        );
110        if let Some(socket) = &guest_socket {
111            url.clone_from(socket);
112            token.clear();
113        }
114        // A notification opens the page at `#session=<id>`: its session
115        // is on the daemon that pushed it (push notifications spec 9.3).
116        let pending_session = push::session_from_fragment(&address_fragment);
117        // A figure's link is the page of the daemon that shows it, unlike a
118        // notification's, which is on the daemon that pushed it.
119        let pending_asset = push::link_target(&address_fragment);
120        if pending_session.is_some() && pending_asset.is_none() && store.is_none() {
121            let daemon =
122                push::notification_daemon(&url, browser::load(PUSH_DAEMON_URL_KEY).as_deref());
123            if daemon != url {
124                token = browser::load(&token_key(&daemon)).unwrap_or_default();
125                url = daemon;
126            }
127        }
128        // A token in the fragment is kept for the daemon and taken out of
129        // the address bar, so it never lands in history or a bookmark.
130        if let Some(token) = store {
131            browser::store(&token_key(&url), &token);
132        }
133        if let Some(fragment) = fragment {
134            browser::replace_fragment(&fragment);
135        }
136        if pending_session.is_some() {
137            browser::replace_fragment(&push::fragment_without_session(
138                &browser::page_fragment().unwrap_or_default(),
139            ));
140        }
141        Self {
142            url: RwSignal::new(url),
143            token: RwSignal::new(token),
144            phase: RwSignal::new(Phase::Idle),
145            identity: StoredValue::new(connection::tab_identity()),
146            client: StoredValue::new(None),
147            generation: StoredValue::new(0),
148            workspace: Workspace::new(),
149            attached: RwSignal::new(false),
150            pending_session: StoredValue::new(pending_session),
151            pending_asset: StoredValue::new(pending_asset),
152            daemon: RwSignal::new(None),
153            is_guest: guest_socket.is_some(),
154            guest_gone: RwSignal::new(false),
155        }
156    }

158    /// The daemon URL the page connects to.
159    pub(crate) fn url(self) -> String {
160        self.url.get_untracked().trim().to_owned()
161    }
162
163    /// Opens `session` from a clicked notification, on the daemon this
164    /// browser's notifications come from: switching to it first when the
165    /// page is on another, or leaving it filled in on the connect screen
166    /// when the page keeps no token for it.
167    fn open_notified(self, session: String) {
168        let daemon =
169            push::notification_daemon(&self.url(), browser::load(PUSH_DAEMON_URL_KEY).as_deref());
170        if daemon == self.url() && self.client.get_value().is_some() {
171            self.workspace.open_notified_session(&session);
172            return;
173        }
174        self.pending_session.set_value(Some(session));
175        if daemon == self.url() {
176            return;
177        }
178        let token = browser::load(&token_key(&daemon)).unwrap_or_default();
179        self.url.set(daemon);
180        self.token.set(token.clone());
181        if connects_by_itself(Some(&self.url()), Some(&token)) {
182            self.connect();
183        } else {
184            self.next_generation();
185            self.drop_client();
186            self.attached.set(false);
187            self.phase.set(Phase::Idle);
188        }
189    }
190
191    fn next_generation(self) -> u64 {
192        self.generation.update_value(|generation| *generation += 1);
193        self.generation.get_value()
194    }
195
196    fn is_current(self, generation: u64) -> bool {
197        self.generation.get_value() == generation
198    }

200    /// Connects with the URL and token the screen holds.
201    fn connect(self) {
202        let url = self.url.get_untracked().trim().to_owned();
203        let token = token_from_input(&self.token.get_untracked());
204        if token != self.token.get_untracked() {
205            self.token.set(token.clone());
206        }
207        if !self.is_guest && !connects_by_itself(Some(&url), Some(&token)) {
208            self.phase.set(Phase::Failed {
209                message: "Enter the daemon URL and its token.".into(),
210                hint: None,
211            });
212            return;
213        }
214        self.drop_client();
215        let generation = self.next_generation();
216        self.phase.set(Phase::Connecting);
217        self.daemon.set(None);
218        spawn_local(async move {
219            let opened = self.open_ready(generation, &url, &token).await;
220            if !self.is_current(generation) {
221                return;
222            }
223            match opened {
224                Ok((client, hello)) => {
225                    if !self.is_guest {
226                        browser::store(DAEMON_URL_KEY, &url);
227                        browser::store(&token_key(&url), &token);
228                    }
229                    self.adopt(generation, &url, client, &hello);
230                }
231                Err(_) if self.is_guest => self.guest_gone.set(true),
232                Err(error) => self.phase.set(failure(&error)),
233            }
234        });
235    }

237    /// Connects and handshakes, then waits for a daemon that is still
238    /// starting to be ready, showing how its start goes meanwhile.
239    async fn open_ready(
240        self,
241        generation: u64,
242        url: &str,
243        token: &str,
244    ) -> Result<(Client, ServerHello), ClientError> {
245        let (client, hello) = connection::open(url, token, self.identity.get_value()).await?;
246        let waited = client
247            .wait_ready_with(
248                &hello,
249                |status| {
250                    if self.is_current(generation) {
251                        self.daemon.set(Some(status.clone()));
252                    }
253                },
254                gloo_timers::future::sleep,
255            )
256            .await;
257        if let Err(error) = waited {
258            client.shutdown();
259            return Err(error);
260        }
261        Ok((client, hello))
262    }

264    /// Makes `client` the page's connection and watches it for a drop.
265    fn adopt(self, generation: u64, url: &str, client: Client, hello: &ServerHello) {
266        // Only the daemon that serves this page serves plugin pages below
267        // it; another daemon's ticket would replace this one's.
268        let serves_page = crate::browser::page_origin()
269            .is_some_and(|origin| cookie_agent_client_state::plugins::serves_page(url, &origin));
270        self.workspace.plugin_pages_here.set(serves_page);
271        self.workspace.guest.set(hello.guest.clone());
272        if let Some(ticket) = hello.page_ticket.as_deref().filter(|_| serves_page) {
273            crate::browser::set_page_ticket(ticket);
274        }
275        self.workspace.share_url.set(hello.share_url.clone());
276        self.workspace
277            .share_page_base
278            .set(cookie_agent_client_state::shares::page_base(url).unwrap_or_default());
279        // The page waited for the daemon to be ready; what it is busy with
280        // comes in its notifications from here on.
281        self.daemon.set(
282            hello
283                .status
284                .clone()
285                .filter(|status| !daemon_status::is_starting(status)),
286        );
287        let deliveries = client.subscribe_deliveries();
288        self.client.set_value(Some(client.clone()));
289        self.workspace.adopt(client);
290        if hello.guest.is_none() {
291            self.workspace.resubscribe_push(url.to_owned());
292        }
293        if let Some(session) = self
294            .pending_session
295            .try_update_value(Option::take)
296            .flatten()
297        {
298            self.workspace.open_notified_session(&session);
299            match self.pending_asset.try_update_value(Option::take).flatten() {
300                Some(push::LinkTarget::Asset(handle)) => {
301                    crate::assets::reveal(handle, self.workspace);
302                }
303                Some(push::LinkTarget::Attachment(sha256)) => {
304                    crate::assets::reveal_attachment(sha256, self.workspace);
305                }
306                None => {}
307            }
308        }
309        self.attached.set(true);
310        self.phase.set(Phase::Connected {
311            address: url.to_owned(),
312            engine_version: hello.engine_version.clone(),
313            protocol_version: hello.protocol_version.to_string(),
314        });
315        let Some(mut deliveries) = deliveries else {
316            return;
317        };
318        spawn_local(async move {
319            let reason = loop {
320                let Some(first) = deliveries.recv().await else {
321                    break "the connection closed".to_owned();
322                };
323                // Everything already delivered applies as one batch: the
324                // page hears of the store once, not once a delivery.
325                let mut next = Some(first);
326                let mut ended = None;
327                self.workspace.batch(|| {
328                    while let Some(delivery) = next.take() {
329                        match delivery {
330                            ClientDelivery::Disconnected { error } => {
331                                ended = Some(error.to_string());
332                                return;
333                            }
334                            ClientDelivery::DaemonStatus(status) if self.is_current(generation) => {
335                                self.daemon.set(Some(status));
336                            }
337                            delivery if self.is_current(generation) => {
338                                self.workspace.handle(delivery);
339                            }
340                            _ => {}
341                        }
342                        next = deliveries.try_recv().ok();
343                    }
344                });
345                if let Some(reason) = ended {
346                    break reason;
347                }
348            };
349            if self.is_current(generation) {
350                self.reconnect(generation, reason);
351            }
352        });
353    }

355    /// Redials after a drop with the shared backoff and the tab's identity,
356    /// until it connects or the user connects or forgets anew.
357    fn reconnect(self, generation: u64, reason: String) {
358        self.drop_client();
359        self.workspace.disconnected();
360        self.phase.set(Phase::Reconnecting { reason });
361        let url = self.url.get_untracked();
362        let token = self.token.get_untracked();
363        // A daemon that went down to restart listens again in moments.
364        let for_restart = cookie_agent_client_state::subscriptions::restarting(
365            self.daemon.get_untracked().as_ref(),
366        );
367        spawn_local(async move {
368            let mut delays = RedialDelays::new(for_restart);
369            loop {
370                gloo_timers::future::sleep(delays.next_delay()).await;
371                if !self.is_current(generation) {
372                    return;
373                }
374                match self.open_ready(generation, &url, &token).await {
375                    Ok((client, hello)) if self.is_current(generation) => {
376                        self.adopt(generation, &url, client, &hello);
377                        return;
378                    }
379                    Ok((client, _)) => {
380                        client.shutdown();
381                        return;
382                    }
383                    Err(error) => {
384                        if !self.is_current(generation) {
385                            return;
386                        }
387                        // A browser gives no reason for a socket it could not
388                        // open, so the page asks: 404 says the link is revoked
389                        // or expired. A daemon that is down says nothing, and
390                        // is redialed.
391                        if self.is_guest && browser::page_is_gone().await {
392                            self.guest_gone.set(true);
393                            return;
394                        }
395                        self.phase.set(Phase::Reconnecting {
396                            reason: crate::errors::describe(&error),
397                        });
398                    }
399                }
400            }
401        });
402    }

404    /// Checks the connection once the tab is visible again, as a phone
405    /// suspends background tabs, and redials when it does not answer.
406    fn check(self) {
407        let Some(client) = self.client.get_value() else {
408            return;
409        };
410        let generation = self.generation.get_value();
411        spawn_local(async move {
412            let answer = std::pin::pin!(client.daemon_status());
413            let timeout = gloo_timers::future::sleep(VISIBLE_CHECK_TIMEOUT);
414            let answered = matches!(
415                futures_util::future::select(answer, timeout).await,
416                futures_util::future::Either::Left((Ok(_), _))
417            );
418            if !answered && self.is_current(generation) {
419                let generation = self.next_generation();
420                self.reconnect(generation, "the daemon stopped answering".into());
421            }
422        });
423    }

425    /// Deletes the stored token and disconnects.
426    pub(crate) fn forget(self) {
427        self.next_generation();
428        // The connection stays open until the daemon dropped this
429        // browser's push subscription, so it notifies here no more.
430        let client = self.client.get_value();
431        self.client.set_value(None);
432        let url = self.url();
433        spawn_local(async move {
434            crate::workspace::release_push(url, client.clone()).await;
435            if let Some(client) = client {
436                client.shutdown();
437            }
438        });
439        browser::forget(&token_key(self.url.get_untracked().trim()));
440        self.token.set(String::new());
441        self.attached.set(false);
442        self.phase.set(Phase::Idle);
443    }
444
445    fn drop_client(self) {
446        if let Some(client) = self.client.get_value() {
447            client.shutdown();
448        }
449        self.client.set_value(None);
450    }
451}

453/// What the connect screen shows for a failed connection.
454fn failure(error: &ClientError) -> Phase {
455    match error {
456        ClientError::ConnectionRefusedBeforeHandshake { .. } => Phase::Failed {
457            message: crate::errors::describe(error),
458            hint: Some(refused_before_handshake_hint()),
459        },
460        _ if error.error_code() == Some("unauthenticated") => Phase::Failed {
461            message: "The daemon refused the token.".into(),
462            hint: Some(
463                "Run `cookie web` on the daemon's machine for a link with its token, \
464                 or paste the token from its ready line."
465                    .into(),
466            ),
467        },
468        _ => Phase::Failed {
469            message: crate::errors::describe(error),
470            hint: None,
471        },
472    }
473}

475/// The page.
476#[component]
477pub fn App() -> impl IntoView {
478    let theme = RwSignal::new(ThemePreference::parse(browser::load(THEME_KEY).as_deref()));
479    let system_dark = RwSignal::new(browser::system_prefers_dark());
480    browser::on_system_scheme_change(move || system_dark.set(browser::system_prefers_dark()));
481    Effect::new(move || {
482        let preference = theme.get();
483        browser::set_dark(preference.is_dark(system_dark.get()));
484        browser::store(THEME_KEY, preference.as_str());
485    });
486
487    Install::provide();
488    // A sign-in's callback leaves the address bar before anything else
489    // reads or rewrites the address.
490    let callback = oauth::take_callback();
491    let connection = Connection::new();
492    let landing = RwSignal::new(None::<Landing>);
493    let workspace = connection.workspace;
494    // Another tab's transcript choice applies here too.
495    browser::on_stored_change(crate::connect::TRANSCRIPT_KEY, move |stored| {
496        workspace.show_transcript(crate::connect::transcript_from_storage(stored.as_deref()));
497    });
498    // The tab that started a sign-in takes the callback another tab of the
499    // page landed with.
500    oauth::listen(move |state, url| {
501        if let Some(server) = oauth::started_here(&state) {
502            oauth::post(&oauth::ChannelMessage::Taken {
503                state: state.clone(),
504            });
505            workspace.finish_sign_in(server, url, Some(state));
506        }
507    });
508    if let Some(callback) = callback {
509        spawn_local(land(callback, workspace, landing));
510    }
511    browser::on_visible(move || connection.check());
512    // Push notifications: a click opens its session, and the daemon hears
513    // what the page shows, so it does not push that (spec section 8).
514    browser::on_open_session_message(move |session| connection.open_notified(session));
515    browser::on_attention_change(move || workspace.report_presence());
516    browser::on_user_input(move || workspace.note_input());
517    Effect::new(move |_| {
518        workspace.selected.track();
519        workspace.new_session.track();
520        workspace.report_presence();
521    });
522    spawn_local(async move {
523        loop {
524            gloo_timers::future::sleep(PRESENCE_CHECK).await;
525            workspace.report_presence();
526        }
527    });
528    browser::follow_visual_viewport();
529    crate::views::follow_status(workspace);
530    if connection.is_guest
531        || connects_by_itself(
532            Some(&connection.url.get_untracked()),
533            Some(&connection.token.get_untracked()),
534        )
535    {
536        connection.connect();
537    }
538
539    let connected = Memo::new(move |_| matches!(connection.phase.get(), Phase::Connected { .. }));
540    let reconnecting_reason = Memo::new(move |_| match connection.phase.get() {
541        Phase::Reconnecting { reason } if connection.attached.get() => Some(reason),
542        _ => None,
543    });
544    let shows_workspace =
545        Memo::new(move |_| connected.get() || reconnecting_reason.with(Option::is_some));
546    view! {
547        <div class="app-frame flex flex-col overflow-hidden bg-surface text-fg">
548            <OAuthLanding landing=landing />
549            {move || {
550                // Only a change of screen remounts what the page shows: the
551                // workspace stays mounted while the page reconnects, so the
552                // composer's focus, the scroll positions, and open panels
553                // outlive a dropped connection.
554                if connection.guest_gone.get() {
555                    return view! {
556                        <main id="guest-gone" class="flex flex-1 flex-col items-center justify-center gap-2 px-4 text-center">
557                            <h1 class="text-xl font-semibold">"This link isn't available"</h1>
558                            <p class="text-muted">"It may have expired or been revoked. Ask the person who shared it for a new one."</p>
559                        </main>
560                    }
561                        .into_any();
562                }
563                if connection.is_guest && !shows_workspace.get() {
564                    return view! {
565                        <main class="flex flex-1 items-center justify-center px-4 text-muted">"Opening the conversation…"</main>
566                    }
567                        .into_any();
568                }
569                if !shows_workspace.get() {
570                    return view! {
571                        <Header connection=connection theme=theme />
572                        <main class="flex flex-1 items-start justify-center overflow-y-auto px-4 py-10">
573                            {move || view! { <ConnectScreen connection=connection phase=connection.phase.get() /> }}
574                        </main>
575                    }
576                        .into_any();
577                }
578                view! {
579                    <Show when=move || reconnecting_reason.get().is_some()>
580                        <div id="reconnecting" role="status" class="flex items-center gap-2 border-b border-accent bg-sunken px-3 py-2 text-sm">
581                            <Glyph icon=icondata::LuRefreshCw class="animate-spin text-accent" />
582                            <span class="min-w-0 flex-1">
583                                {move || {
584                                    match connection.daemon.get().filter(daemon_status::is_starting) {
585                                        Some(status) => daemon_status::starting_text(&status),
586                                        None => format!(
587                                            "Connection lost ({}); reconnecting…",
588                                            reconnecting_reason.get().unwrap_or_default(),
589                                        ),
590                                    }
591                                }}
592                            </span>
593                        </div>
594                        <StartingProgress connection=connection />
595                    </Show>
596                    <Show when=move || connected.get()>
597                        <DaemonNotice connection=connection />
598                        <RestartBanner workspace=connection.workspace />
599                        <RestartDialog workspace=connection.workspace />
600                        <ShareDialogView workspace=connection.workspace />
601                        <SessionShareDialogView workspace=connection.workspace />
602                        <MediaViewer workspace=connection.workspace />
603                    </Show>
604                    <Show when=move || !connection.is_guest>
605                        <InstallBanner />
606                    </Show>
607                    <WorkspaceView workspace=connection.workspace connection=connection theme=theme />
608                }
609                    .into_any()
610            }}
611        </div>
612    }
613}

615/// What the daemon is busy with, once connected: a task that does not
616/// block it, such as a restart waiting for idle, or the daemon stopping.
617#[component]
618fn DaemonNotice(connection: Connection) -> impl IntoView {
619    let notice = Memo::new(move |_| {
620        connection
621            .daemon
622            .get()
623            .as_ref()
624            .and_then(daemon_status::notice_text)
625    });
626    view! {

722/// The connection's state as a small indicator; a click shows the daemon's

724/// only the icon (the rail's); `primary` gives it the ids the page's

726#[component]
727pub(crate) fn ConnectionStatus(
728    connection: Connection,
729    compact: bool,
730    primary: bool,
731) -> impl IntoView {
732    let open = RwSignal::new(false);
733    let confirming = RwSignal::new(false);
734    let close = move || {
735        open.set(false);
736        confirming.set(false);
737    };
738    let indicator = move || {
739        let (icon, label, class) = match connection.phase.get() {
740            Phase::Connected { .. } => (
741                icondata::LuCircle,
742                "connected",
743                "text-[0.5rem] [&_svg]:fill-current",
744            ),
745            Phase::Connecting => (
746                icondata::LuLoaderCircle,
747                "connecting",
748                "animate-spin text-sm",
749            ),

751                icondata::LuRefreshCw,
752                "reconnecting",
753                "animate-spin text-sm",
754            ),

756        };
757        view! {
758            <Glyph icon=icon class=class />
759            <span
760                class=if compact { "sr-only" } else { "text-xs" }
761                id=(primary && label == "connected").then_some("connected")
762            >
763                {label}
764            </span>
765        }
766    };

773        Phase::Connected {
774            address,
775            engine_version,
776            protocol_version,

778            ("Daemon", address),
779            ("Version", format!("cookie {engine_version}")),
780            ("Client protocol", protocol_version),
781        ],
782        Phase::Connecting => vec![("State", "connecting…".to_owned())],
783        Phase::Reconnecting { reason } => vec![("Connection lost", reason)],
784        Phase::Idle => vec![("State", "not connected".to_owned())],
785        Phase::Failed { message, .. } => vec![("Connecting failed", message)],
786    };
787    let summary = move || {
788        details()
789            .into_iter()
790            .map(|(name, value)| format!("{name}: {value}"))
791            .collect::<Vec<_>>()

793    };
794    view! {

801                        Phase::Connected { .. } => "connected",

803                        Phase::Reconnecting { .. } => "reconnecting",
804                        Phase::Idle | Phase::Failed { .. } => "offline",
805                    };
806                    format!("Connection: {state}")
807                }
808                class=move || format!("inline-flex min-h-11 min-w-11 items-center justify-center gap-1.5 rounded-full hover:bg-hover active:bg-press {} {}", if compact { "" } else { "px-3" }, tone())
809                aria-haspopup="dialog"
810                aria-expanded=move || open.get().to_string()
811                title=summary
812                on:click=move |_| {
813                    if open.get_untracked() { close() } else { open.set(true) }
814                }
815            >
816                {indicator}
817            </button>
818            <Show when=move || open.get()>
819                <div class="fixed inset-0 z-30" on:click=move |_| close()></div>
820                <div
821                    id="connection-details"
822                    role="dialog"
823                    aria-label="Connection"
824                    class=if compact {
825                        "fixed bottom-3 left-3 z-40 flex w-80 max-w-[calc(100vw-1.5rem)] flex-col gap-3 rounded-lg border border-line bg-surface p-3 text-sm shadow-lg sm:left-16"
826                    } else {
827                        "absolute top-full right-0 z-40 mt-1 flex w-80 max-w-[calc(100vw-1.5rem)] flex-col gap-3 rounded-lg border border-line bg-surface p-3 text-sm shadow-lg"
828                    }
829                >
830                    <div class="flex items-center gap-2">
831                        <p class="flex-1 font-bold">"Connection"</p>
832                        <button
833                            type="button"
834                            id="connection-close"
835                            class=HEADER_BUTTON
836                            aria-label="Close"
837                            title="Close"
838                            on:click=move |_| close()
839                        >
840                            <Glyph icon=icondata::LuX />
841                        </button>
842                    </div>
843                    <dl class="grid grid-cols-[auto_1fr] gap-x-3 gap-y-1">
844                        {move || {
845                            details()
846                                .into_iter()
847                                .map(|(name, value)| {
848                                    view! {
849                                        <dt class="text-muted">{name}</dt>
850                                        <dd class="min-w-0 font-mono text-xs break-all">{value}</dd>
851                                    }
852                                })
853                                .collect_view()
854                        }}
855                    </dl>
856                    // The daemon can restart from here, as from Settings ›
857                    // Configuration: the same confirmation, once the panel
858                    // is closed.
859                    <Show when=move || {
860                        matches!(connection.phase.get(), Phase::Connected { .. })
861                            && connection.workspace.can_restart.get()
862                    }>
863                        <button
864                            type="button"
865                            id=primary.then_some("connection-restart")
866                            data-connection-restart
867                            class="inline-flex min-h-11 items-center justify-center gap-2 self-start rounded-lg px-3 text-sm hover:bg-sunken active:bg-press bg-raised shadow-[0_0_0_1px_var(--color-line-2)]"
868                            on:click=move |_| {
869                                close();
870                                connection.workspace.ask_restart();
871                            }
872                        >
873                            <Glyph icon=icondata::LuPower />
874                            "Restart daemon"
875                        </button>
876                    </Show>
877                    <Show when=move || connection.attached.get()>
878                        <ForgetDaemon connection=connection confirming=confirming primary=primary on_forget=close />
879                    </Show>
880                </div>
881            </Show>
882        </div>
883    }
884}

1224    /// Offering the callback to the tab that started the sign-in.
1225    HandingOver,
1226    /// That tab took it; this one closes when the browser lets it.
1227    HandedOver { server: String },
1228    /// The sign-in waited too long.
1229    Expired { server: String },
1230    /// No sign-in this browser started, or one that already ended: the
1231    /// address, to paste where the sign-in started.
1232    Unknown { url: String },
1233}
1234
1235/// Finishes the sign-in `callback` answers: in the tab that started it when
1236/// that tab takes it, else here. A callback whose sign-in this browser does
1237/// not know, or that waited too long, is refused.
1238async fn land(callback: oauth::Callback, workspace: Workspace, landing: RwSignal<Option<Landing>>) {
1239    let lookup = callback
1240        .state
1241        .as_deref()
1242        .map_or(Lookup::Unknown, oauth::lookup);
1243    match lookup {
1244        Lookup::Unknown => landing.set(Some(Landing::Unknown { url: callback.url })),
1245        Lookup::Expired(server) => {
1246            if let Some(state) = &callback.state {
1247                oauth::forget(state);
1248            }
1249            workspace.refresh_sign_ins();
1250            landing.set(Some(Landing::Expired { server }));
1251        }
1252        Lookup::Found(server) => {
1253            let state = callback.state.unwrap_or_default();
1254            landing.set(Some(Landing::HandingOver));
1255            if oauth::hand_off(&state, &callback.url).await {
1256                landing.set(Some(Landing::HandedOver { server }));
1257                oauth::close_this_tab();
1258            } else {
1259                landing.set(None);
1260                workspace.finish_sign_in(server, callback.url, Some(state));
1261            }
1262        }
1263    }
1264}
