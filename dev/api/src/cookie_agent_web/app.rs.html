21/// How long a check of the connection after the tab became visible may take
22/// before the page redials.
23const VISIBLE_CHECK_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
24
25/// Where the page's connection stands.
26#[derive(Clone, Debug, PartialEq)]
27enum Phase {
28    /// Waiting for a URL and a token.
29    Idle,
30    Connecting,
31    Connected {
32        address: String,
33        engine_version: String,
34        protocol_version: String,
35    },
36    /// The connection dropped; the page redials with the shared backoff.
37    Reconnecting {
38        reason: String,
39    },
40    /// Connecting failed; `hint` says what to change.
41    Failed {
42        message: String,
43        hint: Option<String>,
44    },
45}
46
47/// The page's connection and its inputs, shared by the components.
48#[derive(Clone, Copy)]
49struct Connection {
50    url: RwSignal<String>,
51    token: RwSignal<String>,
52    phase: RwSignal<Phase>,
53    identity: StoredValue<ClientIdentity>,
54    client: StoredValue<Option<Client>>,
55    /// Bumped by every connect and forget, so a superseded attempt or
56    /// watcher stops.
57    generation: StoredValue<u64>,
58    /// What the page shows of the daemon; it outlives a redial.
59    workspace: Workspace,
60    /// The workspace has had a connection, so a redial keeps it on screen.
61    attached: RwSignal<bool>,
62}
63
64impl Connection {
65    fn new() -> Self {
66        let filled = browser::page_href()
67            .as_deref()
68            .and_then(daemon_url_for_page)
69            .unwrap_or_default();
70        let url = browser::load(DAEMON_URL_KEY).unwrap_or(filled);
71        // A token in the fragment is kept for the daemon and taken out of
72        // the address bar, so it never lands in history or a bookmark.
73        let fragment = browser::page_fragment().unwrap_or_default();
74        if let Some(token) = token_from_fragment(&fragment) {
75            browser::store(&token_key(&url), &token);
76            browser::replace_fragment(&fragment_without_token(&fragment));
77        }
78        let token = browser::load(&token_key(&url)).unwrap_or_default();
79        Self {
80            url: RwSignal::new(url),
81            token: RwSignal::new(token),
82            phase: RwSignal::new(Phase::Idle),
83            identity: StoredValue::new(connection::tab_identity()),
84            client: StoredValue::new(None),
85            generation: StoredValue::new(0),
86            workspace: Workspace::new(),
87            attached: RwSignal::new(false),
88        }
89    }
90
91    fn next_generation(self) -> u64 {
92        self.generation.update_value(|generation| *generation += 1);
93        self.generation.get_value()
94    }
95
96    fn is_current(self, generation: u64) -> bool {
97        self.generation.get_value() == generation
98    }
99
100    /// Connects with the URL and token the screen holds.
101    fn connect(self) {
102        let url = self.url.get_untracked().trim().to_owned();
103        let token = self.token.get_untracked().trim().to_owned();
104        if !connects_by_itself(Some(&url), Some(&token)) {
105            self.phase.set(Phase::Failed {
106                message: "Enter the daemon URL and its token.".into(),
107                hint: None,
108            });
109            return;
110        }
111        self.drop_client();
112        let generation = self.next_generation();
113        self.phase.set(Phase::Connecting);
114        spawn_local(async move {
115            let opened = connection::open(&url, &token, self.identity.get_value()).await;
116            if !self.is_current(generation) {
117                return;
118            }
119            match opened {
120                Ok((client, hello)) => {
121                    browser::store(DAEMON_URL_KEY, &url);
122                    browser::store(&token_key(&url), &token);
123                    self.adopt(generation, &url, client, &hello);
124                }
125                Err(error) => self.phase.set(failure(&error)),
126            }
127        });
128    }
129
130    /// Makes `client` the page's connection and watches it for a drop.
131    fn adopt(self, generation: u64, url: &str, client: Client, hello: &ServerHello) {
132        let deliveries = client.subscribe_deliveries();
133        self.client.set_value(Some(client.clone()));
134        self.workspace.adopt(client);
135        self.attached.set(true);
136        self.phase.set(Phase::Connected {
137            address: url.to_owned(),
138            engine_version: hello.engine_version.clone(),
139            protocol_version: hello.protocol_version.to_string(),
140        });
141        let Some(mut deliveries) = deliveries else {
142            return;
143        };
144        spawn_local(async move {
145            let reason = loop {
146                match deliveries.recv().await {
147                    Some(ClientDelivery::Disconnected { error }) => break error.to_string(),
148                    Some(delivery) if self.is_current(generation) => {
149                        self.workspace.handle(delivery);
150                    }
151                    Some(_) => {}
152                    None => break "the connection closed".to_owned(),
153                }
154            };
155            if self.is_current(generation) {
156                self.reconnect(generation, reason);
157            }
158        });
159    }
160
161    /// Redials after a drop with the shared backoff and the tab's identity,
162    /// until it connects or the user connects or forgets anew.
163    fn reconnect(self, generation: u64, reason: String) {
164        self.drop_client();
165        self.workspace.disconnected();
166        self.phase.set(Phase::Reconnecting { reason });
167        let url = self.url.get_untracked();
168        let token = self.token.get_untracked();
169        spawn_local(async move {
170            let mut delay = RECONNECT_FIRST_DELAY;
171            loop {
172                gloo_timers::future::sleep(delay).await;
173                if !self.is_current(generation) {
174                    return;
175                }
176                match connection::open(&url, &token, self.identity.get_value()).await {
177                    Ok((client, hello)) if self.is_current(generation) => {
178                        self.adopt(generation, &url, client, &hello);
179                        return;
180                    }
181                    Ok((client, _)) => {
182                        client.shutdown();
183                        return;
184                    }
185                    Err(error) => {
186                        if !self.is_current(generation) {
187                            return;
188                        }
189                        self.phase.set(Phase::Reconnecting {
190                            reason: error.to_string(),
191                        });
192                        delay = next_reconnect_delay(delay);
193                    }
194                }
195            }
196        });
197    }
198
199    /// Checks the connection once the tab is visible again, as a phone
200    /// suspends background tabs, and redials when it does not answer.
201    fn check(self) {
202        let Some(client) = self.client.get_value() else {
203            return;
204        };
205        let generation = self.generation.get_value();
206        spawn_local(async move {
207            let answer = std::pin::pin!(client.runtime_snapshot());
208            let timeout = gloo_timers::future::sleep(VISIBLE_CHECK_TIMEOUT);
209            let answered = matches!(
210                futures_util::future::select(answer, timeout).await,
211                futures_util::future::Either::Left((Ok(_), _))
212            );
213            if !answered && self.is_current(generation) {
214                let generation = self.next_generation();
215                self.reconnect(generation, "the daemon stopped answering".into());
216            }
217        });
218    }
219
220    /// Deletes the stored token and disconnects.
221    fn forget(self) {
222        self.next_generation();
223        self.drop_client();
224        browser::forget(&token_key(self.url.get_untracked().trim()));
225        self.token.set(String::new());
226        self.attached.set(false);
227        self.phase.set(Phase::Idle);
228    }
229
230    fn drop_client(self) {
231        if let Some(client) = self.client.get_value() {
232            client.shutdown();
233        }
234        self.client.set_value(None);
235    }
236}
237
238/// What the connect screen shows for a failed connection.
239fn failure(error: &ClientError) -> Phase {
240    match error {
241        ClientError::ConnectionRefusedBeforeHandshake { .. } => Phase::Failed {
242            message: error.to_string(),
243            hint: Some(refused_before_handshake_hint(
244                &browser::page_origin().unwrap_or_default(),
245            )),
246        },
247        _ if error.error_code() == Some("unauthenticated") => Phase::Failed {
248            message: "The daemon refused the token.".into(),
249            hint: Some(
250                "Run `cookie web` on the daemon's machine for a link with its token, \
251                 or paste the token from its ready line."
252                    .into(),
253            ),
254        },
255        _ => Phase::Failed {
256            message: error.to_string(),
257            hint: None,
258        },
259    }
260}

262/// The page.
263#[component]
264pub fn App() -> impl IntoView {
265    let theme = RwSignal::new(ThemePreference::parse(browser::load(THEME_KEY).as_deref()));
266    let system_dark = RwSignal::new(browser::system_prefers_dark());
267    browser::on_system_scheme_change(move || system_dark.set(browser::system_prefers_dark()));
268    Effect::new(move || {
269        let preference = theme.get();
270        browser::set_dark(preference.is_dark(system_dark.get()));
271        browser::store(THEME_KEY, preference.as_str());
272    });
273
274    let connection = Connection::new();
275    browser::on_visible(move || connection.check());
276    if connects_by_itself(
277        Some(&connection.url.get_untracked()),
278        Some(&connection.token.get_untracked()),
279    ) {
280        connection.connect();
281    }
282
283    view! {
284        <div class="flex h-dvh flex-col overflow-hidden bg-cream text-espresso dark:bg-night dark:text-latte">
285            <Header connection=connection theme=theme />
286            {move || {
287                let phase = connection.phase.get();
288                let attached = connection.attached.get();
289                match phase {
290                    Phase::Connected { .. } => {
291                        view! { <WorkspaceView workspace=connection.workspace /> }.into_any()
292                    }
293                    Phase::Reconnecting { reason } if attached => {
294                        view! {
295                            <div id="reconnecting" class="border-b border-caramel bg-toasted px-3 py-1 text-sm dark:border-maple dark:bg-ember">
296                                {format!("Connection lost ({reason}); reconnecting…")}
297                            </div>
298                            <WorkspaceView workspace=connection.workspace />
299                        }
300                            .into_any()
301                    }
302                    phase => {
303                        view! {
304                            <main class="flex flex-1 items-start justify-center overflow-y-auto px-4 py-10">
305                                <ConnectScreen connection=connection phase=phase />
306                            </main>
307                        }
308                            .into_any()
309                    }
310                }
311            }}
312        </div>
313    }
314}

316#[component]
317fn Header(connection: Connection, theme: RwSignal<ThemePreference>) -> impl IntoView {
318    let status = move || {
319        match connection.phase.get() {
320        Phase::Idle | Phase::Failed { .. } => view! { <span>"not connected"</span> }.into_any(),
321        Phase::Connecting => view! { <span>"connecting…"</span> }.into_any(),
322        Phase::Connected {
323            address,
324            engine_version,
325            protocol_version,
326        } => view! {
327            <span id="connected">
328                {format!("{address} · cookie {engine_version} · client protocol {protocol_version}")}
329            </span>
330        }
331            .into_any(),
332        Phase::Reconnecting { .. } => view! { <span>"reconnecting…"</span> }.into_any(),
333    }
334    };
335    view! {
336        <header class="flex items-center gap-3 border-b border-toasted bg-parchment px-4 py-1 dark:border-ember dark:bg-char">
337            <span class="font-mono font-bold">"cookie agent"</span>
338            <span class="min-w-0 flex-1 truncate font-mono text-xs text-ink dark:text-sand">{status}</span>
339            <Show when=move || connection.attached.get()>
340                <button
341                    type="button"
342                    id="forget"
343                    class="min-h-11 rounded border border-crust px-3 py-1 text-sm hover:bg-toasted active:bg-crust dark:border-smoke dark:hover:bg-ember dark:active:bg-smoke"
344                    on:click=move |_| connection.forget()
345                >
346                    "Forget this daemon"
347                </button>
348            </Show>
349            <button
350                type="button"
351                class="min-h-11 rounded border border-crust px-3 py-1 text-sm hover:bg-toasted active:bg-crust dark:border-smoke dark:hover:bg-ember dark:active:bg-smoke"
352                on:click=move |_| theme.update(|preference| *preference = preference.next())
353            >
354                {move || theme.get().label()}
355            </button>
356        </header>
357    }
358}

360#[component]
361fn ConnectScreen(connection: Connection, phase: Phase) -> impl IntoView {
362    let busy = matches!(phase, Phase::Connecting | Phase::Reconnecting { .. });
363    let notice = match phase {
364        Phase::Failed { message, hint } => Some((message, hint)),
365        Phase::Reconnecting { reason } => {
366            Some((format!("Connection lost ({reason}); reconnecting…"), None))
367        }
368        _ => None,
369    };
370    view! {
371        <form
372            class="flex w-full max-w-xl flex-col gap-4 rounded-lg border border-toasted bg-parchment p-6 dark:border-ember dark:bg-char"
373            on:submit=move |event| {
374                event.prevent_default();
375                connection.connect();
376            }
377        >
378            <h1 class="text-xl font-bold">"Connect to a cookie daemon"</h1>
379            <label class="flex flex-col gap-1 text-sm">
380                <span>"Daemon URL"</span>
381                <input
382                    type="url"
383                    class="rounded border border-crust bg-cream px-3 py-2 font-mono text-espresso hover:border-caramel focus:border-caramel focus:outline-none dark:border-smoke dark:bg-night dark:text-latte dark:hover:border-maple dark:focus:border-maple"
384                    placeholder="wss://host:7419/ws"
385                    prop:value=move || connection.url.get()
386                    on:input=move |event| {
387                        let url = event_target_value(&event);
388                        connection
389                            .token
390                            .set(browser::load(&token_key(url.trim())).unwrap_or_default());
391                        connection.url.set(url);
392                    }
393                />
394            </label>
395            <label class="flex flex-col gap-1 text-sm">
396                <span>"Token"</span>
397                <input
398                    type="password"
399                    autocomplete="off"
400                    class="rounded border border-crust bg-cream px-3 py-2 font-mono text-espresso hover:border-caramel focus:border-caramel focus:outline-none dark:border-smoke dark:bg-night dark:text-latte dark:hover:border-maple dark:focus:border-maple"
401                    placeholder="paste the token, or open the link `cookie web` prints"
402                    prop:value=move || connection.token.get()
403                    on:input=move |event| connection.token.set(event_target_value(&event))
404                />
405            </label>
406            {notice
407                .map(|(message, hint)| {
408                    view! {
409                        <div class="rounded border border-caramel p-3 text-sm dark:border-maple">
410                            <p>{message}</p>
411                            {hint.map(|hint| view! { <p class="mt-2 font-mono">{hint}</p> })}
412                        </div>
413                    }
414                })}
415            <div class="flex flex-wrap gap-3">
416                <button
417                    type="submit"
418                    disabled=busy
419                    class="rounded bg-espresso px-4 py-2 font-bold text-cream hover:bg-caramel active:bg-ink disabled:opacity-60 dark:bg-latte dark:text-night dark:hover:bg-maple dark:active:bg-sand"
420                >
421                    {if busy { "Connecting…" } else { "Connect" }}
422                </button>
423                <button
424                    type="button"
425                    class="rounded border border-crust px-4 py-2 hover:bg-toasted active:bg-crust dark:border-smoke dark:hover:bg-ember dark:active:bg-smoke"
426                    on:click=move |_| connection.forget()
427                >
428                    "Forget this daemon"
429                </button>
430            </div>
431        </form>
432    }
