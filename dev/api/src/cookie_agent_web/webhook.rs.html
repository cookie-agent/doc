1//! The Webhooks view's rules beyond what every client shares
2//! (`cookie_agent_client_state::webhook`): the template a new-session
3//! webhook starts from, and what a webhook session's mark says.
4
5use cookie_agent_client_state::webhook::webhook_title;
6use cookie_agent_protocol::{AgentId, ProjectId, SessionWebhook, Webhook, WebhookTemplate};
7
8/// The template a new-session webhook starts from: `agent` of `preset`, in
9/// `project`; the daemon attaches what `session.create` would.
10pub fn new_template(
11    agent: AgentId,
12    preset: Option<String>,
13    project: Option<ProjectId>,
14) -> WebhookTemplate {
15    WebhookTemplate {
16        preset,
17        project,
18        ..WebhookTemplate::for_agent(agent)
19    }
20}
21
22/// What a webhook session's mark says: the webhook that started it, by
23/// its title while the daemon still has it, and the thread it is.
24pub fn mark_title(session: &SessionWebhook, webhooks: &[Webhook]) -> String {
25    let thread = session
26        .thread_key
27        .as_deref()
28        .map(|key| format!(", thread {key}"))
29        .unwrap_or_default();
30    match webhooks
31        .iter()
32        .find(|webhook| webhook.id == session.webhook_id)
33    {
34        Some(webhook) => format!("Started by the webhook {}{thread}", webhook_title(webhook)),
35        None => format!(
36            "Started by the webhook {}{thread}, since deleted",
37            session.webhook_id
38        ),
39    }
40}

42#[cfg(test)]
43mod tests {
44    use cookie_agent_protocol::SessionWebhook;
45
46    use super::{mark_title, new_template};
47
48    #[test]
49    fn a_new_session_template_keeps_its_preset_and_project_and_attaches_global() {
50        let template = new_template(
51            "builder".parse().expect("agent"),
52            Some("fast".into()),
53            Some(
54                "0190a5b2-0000-7000-8000-0000000000c1"
55                    .parse()
56                    .expect("project"),
57            ),
58        );
59        assert_eq!(template.agent.as_str(), "builder");
60        assert_eq!(template.preset.as_deref(), Some("fast"));
61        assert!(template.project.is_some());
62        assert!(template.global);
63        assert_eq!(template.thread_key, None);
64        assert_eq!(template.model, None);
65    }
66
67    #[test]
68    fn the_mark_finds_its_webhook_or_knows_it_is_gone() {
69        let session: SessionWebhook = serde_json::from_value(serde_json::json!({
70            "webhook_id": "w7k2m9qa", "event_id": "e3j8x0pq", "thread_key": "1234"
71        }))
72        .expect("session webhook");
73        let gone = mark_title(&session, &[]);
74        assert!(gone.contains("w7k2m9qa"));
75        assert!(gone.contains("1234"));
76        let unthreaded = SessionWebhook {
77            thread_key: None,
78            ..session
79        };
80        assert!(!mark_title(&unthreaded, &[]).contains("thread"));
81    }
