1//! The new-session runtime cards, independent of the browser: one card per
2//! runtime capability that takes a workspace, plus `global`, grouped by the
3//! machine each runtime runs on; which cards a new session selects and at
4//! which workspaces; the defaults, the workspace suggestions, the note
5//! about tool-name prefixes, and what the browser remembers.
6//!
7//! The selection keeps the text the user typed: the session is created
8//! with it, and the daemon and the runtime resolve it (`~/…` included).

35/// Whether a card's runtime can start a session now.
36#[derive(Clone, Copy, Debug, Eq, PartialEq)]
37pub enum CardState {
38    /// Connected (or built in).
39    Ready,
40    /// Not running, but the daemon starts it when a session opens it.
41    StartsOnUse,
42    /// A runtime that dials in and is not connected: it cannot be opened.
43    Offline,
44}

46impl CardState {
47    pub const fn label(self) -> &'static str {
48        match self {
49            Self::Ready => "ready",
50            Self::StartsOnUse => "starts on use",
51            Self::Offline => "offline",
52        }
53    }

56/// One card: a runtime capability a session can attach.
57#[derive(Clone, Debug, Eq, PartialEq)]
58pub struct Card {
59    pub runtime_id: RuntimeId,
60    /// `None` for a runtime that has not said what it offers yet: attaching
61    /// takes its only capability.
62    pub capability: Option<CapabilityName>,
63    pub takes_workspace: bool,
64    pub state: CardState,
65    pub machine: Option<MachineName>,
66    pub description: String,
67    /// Why the runtime last failed to connect.
68    pub message: Option<String>,
69}
70
71impl Card {
72    /// What identifies the card in the selection and the page.
73    pub fn key(&self) -> String {
74        card_key(
75            self.runtime_id.as_str(),
76            self.capability.as_ref().map(CapabilityName::as_str),
77        )
78    }
79
80    pub fn is_global(&self) -> bool {
81        self.runtime_id.as_str() == RuntimeId::GLOBAL
82    }
83
84    /// `local/coding`, or the runtime alone before it said what it offers.
85    pub fn label(&self) -> String {
86        match &self.capability {
87            Some(capability) => format!("{}/{capability}", self.runtime_id),
88            None => self.runtime_id.to_string(),
89        }
90    }
91}

93fn card_key(runtime_id: &str, capability: Option<&str>) -> String {
94    match capability {
95        Some(capability) => format!("{runtime_id}/{capability}"),
96        None => runtime_id.to_owned(),
97    }
98}

100/// The cards `listing` offers: `global` first, then `local`, then every
101/// other runtime's capabilities that take a workspace by runtime ID, and a
102/// runtime that has not said what it offers as one card that asks for a
103/// workspace.
104pub fn cards(listing: &RuntimesListResult) -> Vec<Card> {
105    let mut cards = Vec::new();
106    for runtime in &listing.runtimes {
107        let state = if runtime.connected || runtime.runtime_id.as_str() == RuntimeId::GLOBAL {
108            CardState::Ready
109        } else if runtime.lifetime == RuntimeLifetime::DialIn {
110            CardState::Offline
111        } else {
112            CardState::StartsOnUse
113        };
114        let card =
115            |capability: Option<CapabilityName>, takes_workspace, description: String| Card {
116                runtime_id: runtime.runtime_id.clone(),
117                capability,
118                takes_workspace,
119                state,
120                machine: runtime.machine.clone(),
121                description,
122                message: (!runtime.connected)
123                    .then(|| runtime.message.clone())
124                    .flatten(),
125            };
126        if runtime.runtime_id.as_str() == RuntimeId::GLOBAL {
127            cards.push(card(
128                Some(CapabilityName::global()),
129                false,
130                "The daemon's own tools, MCP servers, and memory".into(),
131            ));
132        } else if runtime.capabilities.is_empty() {
133            // It has not said what it offers yet: say how it comes to run.
134            let how = match runtime.lifetime {
135                RuntimeLifetime::DialIn => "Dials in · not connected since the daemon started",
136                RuntimeLifetime::Builtin => "Built in",
137                RuntimeLifetime::Spawned => {
138                    "Template runtime · the daemon starts it when a session needs it"
139                }
140            };
141            cards.push(card(None, true, how.to_owned()));
142        } else {
143            for capability in runtime.capabilities.iter().filter(|c| c.takes_workspace) {
144                cards.push(card(
145                    Some(capability.name.clone()),
146                    true,
147                    capability.description.clone(),
148                ));
149            }
150        }
151    }
152    cards.sort_by_key(|card| {
153        (
154            !card.is_global(),
155            card.runtime_id.as_str() != RuntimeId::LOCAL,
156            card.runtime_id.clone(),
157            card.capability.clone(),
158        )
159    });
160    cards
161}

163/// The machine this daemon runs on: `local`'s, else `global`'s.
164pub fn daemon_machine(listing: &RuntimesListResult) -> Option<MachineName> {
165    let machine_of = |id: &str| {
166        listing
167            .runtimes
168            .iter()
169            .find(|runtime| runtime.runtime_id.as_str() == id)
170            .and_then(|runtime| runtime.machine.clone())
171    };
172    machine_of(RuntimeId::LOCAL).or_else(|| machine_of(RuntimeId::GLOBAL))
173}

175/// The cards of one machine.
176#[derive(Clone, Debug, Eq, PartialEq)]
177pub struct Group {
178    /// `None` for runtimes whose machine the daemon cannot tell.
179    pub machine: Option<MachineName>,
180    /// This daemon's machine, which also holds `global`.
181    pub daemon: bool,
182    pub cards: Vec<Card>,
183}
184
185impl Group {
186    /// The group's name and its key in the browser's storage.
187    pub fn name(&self) -> String {
188        self.machine
189            .as_ref()
190            .map_or_else(|| "other machines".to_owned(), ToString::to_string)
191    }
192
193    /// "yunhao-labpc · 2 runtimes · 1 ready · 1 selected".
194    pub fn summary(&self, selection: &Selection) -> String {
195        let mut runtimes = self
196            .cards
197            .iter()
198            .map(|card| card.runtime_id.as_str())
199            .collect::<Vec<_>>();
200        runtimes.dedup();
201        let ready = self
202            .cards
203            .iter()
204            .filter(|card| card.state == CardState::Ready)
205            .map(|card| card.runtime_id.as_str())
206            .collect::<std::collections::BTreeSet<_>>()
207            .len();
208        let selected = self
209            .cards
210            .iter()
211            .filter(|card| selection.selected(card))
212            .count();
213        format!(
214            "{} · {} · {ready} ready · {selected} selected",
215            self.name(),
216            plural(runtimes.len(), "runtime"),
217        )
218    }
219
220    /// Whether the group shows its cards: as the browser remembers it, else
221    /// open for this daemon's machine and for a machine with a selected card.
222    pub fn open(&self, stored: Option<&str>, selection: &Selection) -> bool {
223        match stored {
224            Some("open") => true,
225            Some("closed") => false,
226            _ => self.daemon || self.cards.iter().any(|card| selection.selected(card)),
227        }
228    }
229}

239/// `cards` grouped by machine: this daemon's machine first (with `global`),
240/// then the others by name, and runtimes of no known machine last.
241pub fn groups(cards: &[Card], daemon: Option<&MachineName>) -> Vec<Group> {
242    let mut by_machine = BTreeMap::<Option<MachineName>, Vec<Card>>::new();
243    for card in cards {
244        let machine = if card.is_global() {
245            daemon.cloned().or_else(|| card.machine.clone())
246        } else {
247            card.machine.clone()
248        };
249        by_machine.entry(machine).or_default().push(card.clone());
250    }
251    let mut groups = by_machine
252        .into_iter()
253        .map(|(machine, cards)| Group {
254            daemon: machine.is_some() && machine.as_ref() == daemon,
255            machine,
256            cards,
257        })
258        .collect::<Vec<_>>();
259    // A daemon whose machine is unknown keeps its built-in runtimes first.
260    let holds_global = |group: &Group| group.cards.iter().any(Card::is_global);
261    groups.sort_by_key(|group| {
262        (
263            !(group.daemon || holds_global(group)),
264            group.machine.is_none(),
265        )
266    });
267    groups
268}

274/// One selected card and the workspaces it opens, in field order.
275#[derive(Clone, Debug, Eq, PartialEq)]
276pub struct Pick {
277    pub runtime_id: String,
278    pub capability: Option<String>,
279    /// One entry per workspace field (empty while not filled in); none for
280    /// a card that takes no workspace.
281    pub workspaces: Vec<String>,
282}
283
284impl Pick {
285    fn key(&self) -> String {
286        card_key(&self.runtime_id, self.capability.as_deref())
287    }
288
289    /// Whether the pick is of `card`. A runtime that has not said what it
290    /// offers has one card without a capability, which becomes its
291    /// capability's card once it connected: either matches the other.
292    fn is_of(&self, card: &Card) -> bool {
293        self.runtime_id == card.runtime_id.as_str()
294            && match (&self.capability, &card.capability) {
295                (Some(picked), Some(offered)) => picked == offered.as_str(),
296                _ => true,
297            }
298    }
299}

333impl Selection {
334    /// The selection of a session attached as `attachments`, its
335    /// workspaces shown under their runtime's home, as the fields show them.
336    pub fn of_session(attachments: &[RuntimeAttachment], homes: &RuntimeHomes) -> Self {
337        let mut selection = Self {
338            global: false,
339            picks: Vec::new(),
340        };
341        for attachment in attachments {
342            if attachment.runtime_id.as_str() == RuntimeId::GLOBAL {
343                selection.global = true;
344                continue;
345            }
346            let Some(workspace) = &attachment.workspace else {
347                continue;
348            };
349            let shown = homes.display(&attachment.runtime_id, workspace);
350            let key = card_key(
351                attachment.runtime_id.as_str(),
352                Some(attachment.capability.as_str()),
353            );
354            match selection.picks.iter_mut().find(|pick| pick.key() == key) {
355                Some(pick) => pick.workspaces.push(shown),
356                None => selection.picks.push(Pick {
357                    runtime_id: attachment.runtime_id.to_string(),
358                    capability: Some(attachment.capability.to_string()),
359                    workspaces: vec![shown],
360                }),
361            }
362        }
363        selection
364    }

366    /// The selection of a project's default attachments, their workspaces
367    /// shown under their runtime's home. A runtime named without a
368    /// capability takes its card whatever the capability turns out to be.
369    pub fn of_project(defaults: &ProjectDefaults, homes: &RuntimeHomes) -> Self {
370        let mut selection = Self {
371            global: defaults.global,
372            picks: Vec::new(),
373        };
374        for runtime in &defaults.runtimes {
375            let shown = runtime
376                .workspace
377                .as_deref()
378                .map(|workspace| homes.display(&runtime.runtime_id, workspace));
379            let key = card_key(
380                runtime.runtime_id.as_str(),
381                runtime.capability.as_ref().map(CapabilityName::as_str),
382            );
383            match selection.picks.iter_mut().find(|pick| pick.key() == key) {
384                Some(pick) => pick.workspaces.extend(shown),
385                None => selection.picks.push(Pick {
386                    runtime_id: runtime.runtime_id.to_string(),
387                    capability: runtime.capability.as_ref().map(ToString::to_string),
388                    workspaces: shown.into_iter().collect(),
389                }),
390            }
391        }
392        selection
393    }

395    /// What a project's defaults keep of the selection: `global`, and each
396    /// pick at each of its workspaces (as typed), in pick order. Unlike
397    /// [`Self::requests`] it needs no card, so a runtime that is not
398    /// listed now keeps its place.
399    pub fn project_requests(&self) -> (bool, Vec<RuntimeAttachRequest>) {
400        let mut requests = Vec::new();
401        for pick in &self.picks {
402            let Ok(runtime_id) = RuntimeId::new(pick.runtime_id.clone()) else {
403                continue;
404            };
405            let capability = pick
406                .capability
407                .as_ref()
408                .and_then(|name| CapabilityName::new(name.clone()).ok());
409            let request = |workspace: Option<String>| RuntimeAttachRequest {
410                runtime_id: runtime_id.clone(),
411                capability: capability.clone(),
412                workspace,
413            };
414            if pick.workspaces.is_empty() {
415                requests.push(request(None));
416            }
417            for workspace in &pick.workspaces {
418                requests.push(request(Some(workspace.trim().to_owned())));
419            }
420        }
421        (self.global, requests)
422    }
423
424    fn pick(&self, card: &Card) -> Option<&Pick> {
425        self.picks.iter().find(|pick| pick.is_of(card))
426    }
427
428    fn pick_mut(&mut self, card: &Card) -> Option<&mut Pick> {
429        self.picks.iter_mut().find(|pick| pick.is_of(card))
430    }
431
432    pub fn selected(&self, card: &Card) -> bool {
433        if card.is_global() {
434            self.global
435        } else {
436            self.pick(card).is_some()
437        }
438    }

440    /// Selects or deselects `card`; an offline card cannot be selected.
441    /// Returns whether the selection changed.
442    pub fn toggle(&mut self, card: &Card) -> bool {
443        if card.is_global() {
444            self.global = !self.global;
445            return true;
446        }
447        if self.selected(card) {
448            self.picks.retain(|pick| !pick.is_of(card));
449            return true;
450        }
451        if card.state == CardState::Offline {
452            return false;
453        }
454        self.picks.push(Pick {
455            runtime_id: card.runtime_id.to_string(),
456            capability: card.capability.as_ref().map(ToString::to_string),
457            workspaces: if card.takes_workspace {
458                vec![String::new()]
459            } else {
460                Vec::new()
461            },
462        });
463        true
464    }

466    /// The workspace fields of `card`, in order.
467    pub fn workspaces(&self, card: &Card) -> Vec<String> {
468        self.pick(card)
469            .map(|pick| pick.workspaces.clone())
470            .unwrap_or_default()
471    }

473    pub fn set_workspace(&mut self, card: &Card, index: usize, text: String) {
474        if let Some(field) = self
475            .pick_mut(card)
476            .and_then(|pick| pick.workspaces.get_mut(index))
477        {
478            *field = text;
479        }
480    }

482    /// "+ another workspace": the same runtime at a second directory.
483    pub fn add_workspace(&mut self, card: &Card) {
484        if let Some(pick) = self.pick_mut(card) {
485            pick.workspaces.push(String::new());
486        }
487    }

489    /// Removes a workspace field; the last one stays.
490    pub fn remove_workspace(&mut self, card: &Card, index: usize) {
491        if let Some(pick) = self.pick_mut(card)
492            && pick.workspaces.len() > 1
493            && index < pick.workspaces.len()
494        {
495            pick.workspaces.remove(index);
496        }
497    }

499    /// Why the first message cannot create the session yet, else `None`: a
500    /// selected card with an empty workspace field, or one gone offline.
501    pub fn blocker(&self, cards: &[Card]) -> Option<String> {
502        for card in cards
503            .iter()
504            .filter(|card| !card.is_global() && self.selected(card))
505        {
506            if card.state == CardState::Offline {
507                return Some(format!("{} is offline", card.label()));
508            }
509            if card.takes_workspace
510                && self
511                    .workspaces(card)
512                    .iter()
513                    .any(|workspace| workspace.trim().is_empty())
514            {
515                return Some(format!("enter a workspace for {}", card.label()));
516            }
517        }
518        None
519    }

521    /// What the session opens, in card order: each selected card at each
522    /// of its workspaces (as typed), and whether it attaches `global`.
523    pub fn requests(&self, cards: &[Card]) -> (bool, Vec<RuntimeAttachRequest>) {
524        let (global, requests) = self.keyed_requests(cards);
525        (
526            global,
527            requests
528                .into_iter()
529                .map(|(_, _, request)| request)
530                .collect(),
531        )
532    }
533
534    /// [`Self::requests`], each with its card's key and its field's index,
535    /// where a refusal shows.
536    pub fn keyed_requests(&self, cards: &[Card]) -> (bool, Vec<KeyedRequest>) {
537        let mut requests = Vec::new();
538        for card in cards
539            .iter()
540            .filter(|card| !card.is_global() && self.selected(card))
541        {
542            let request = |workspace: Option<String>| RuntimeAttachRequest {
543                runtime_id: card.runtime_id.clone(),
544                capability: card.capability.clone(),
545                workspace,
546            };
547            if card.takes_workspace {
548                for (index, workspace) in self.workspaces(card).into_iter().enumerate() {
549                    requests.push((
550                        card.key(),
551                        index,
552                        request(Some(workspace.trim().to_owned())),
553                    ));
554                }
555            } else {
556                requests.push((card.key(), 0, request(None)));
557            }
558        }
559        (self.global, requests)
560    }
561}

563/// The card field a refusal of a new session's runtimes names, among the
564/// `requests` it was created with ([`Selection::keyed_requests`]): the
565/// first request whose runtime the message quotes as `` `id` ``, as the
566/// daemon names runtimes, preferring one whose workspace it names too.
567/// `None` when it names none of them: the error is the session's.
568pub fn failing_request(message: &str, requests: &[KeyedRequest]) -> Option<(String, usize)> {
569    let named = requests
570        .iter()
571        .filter(|(_, _, request)| message.contains(&format!("`{}`", request.runtime_id)))
572        .collect::<Vec<_>>();
573    named
574        .iter()
575        .find(|(_, _, request)| {
576            request
577                .workspace
578                .as_deref()
579                .is_some_and(|workspace| !workspace.is_empty() && message.contains(workspace))
580        })
581        .or_else(|| named.first())
582        .map(|(key, index, _)| (key.clone(), *index))
583}

585/// The workspaces a field for `runtime_id` suggests, shown under the
586/// runtime's `home`: the ones this browser opened there lately (`recent`,
587/// newest first), then the workspaces sessions attach there, most recently
588/// active first; each once.
589pub fn suggestions(
590    runtime_id: &str,
591    recent: &[String],
592    sessions: &[SessionMeta],
593    home: Option<&str>,
594) -> Vec<String> {
595    let mut sessions = sessions.iter().collect::<Vec<_>>();
596    sessions.sort_by_key(|session| std::cmp::Reverse(session.last_activity));
597    let attached = sessions.into_iter().flat_map(|session| {
598        session
599            .attachments
600            .iter()
601            .filter(|attachment| attachment.runtime_id.as_str() == runtime_id)
602            .filter_map(|attachment| attachment.workspace.clone())
603    });
604    let mut shown = Vec::new();
605    for workspace in recent.iter().cloned().chain(attached) {
606        let display = cookie_agent_client_state::paths::display_path(&workspace, home);
607        if !shown.contains(&display) {
608            shown.push(display);
609        }
610        if shown.len() == SUGGESTION_LIMIT {
611            break;
612        }
613    }
614    shown
615}

617/// `recent` with `workspace` first, once, kept to [`RECENT_LIMIT`].
618pub fn remember(recent: &[String], workspace: &str) -> Vec<String> {
619    std::iter::once(workspace.to_owned())
620        .chain(recent.iter().filter(|kept| *kept != workspace).cloned())
621        .take(RECENT_LIMIT)
622        .collect()
623}
