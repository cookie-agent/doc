119/// A session's name in the list: its title, else its short ID.
120fn session_label(session: &SessionMeta) -> String {
121    session
122        .title
123        .as_ref()
124        .map(ToString::to_string)
125        .or_else(|| session.short_id.clone())
126        .unwrap_or_else(|| session.session_id.to_string())
127}

129#[component]
130fn SessionList(workspace: Workspace) -> impl IntoView {
131    let directories = move || {
132        let mut directories = workspace.sessions.with(|sessions| {
133            sessions
134                .iter()
135                .map(|session| session.cwd_identity.to_string())
136                .collect::<Vec<_>>()
137        });
138        directories.sort();
139        directories.dedup();
140        directories
141    };
142    let shown = move || {
143        let filter = workspace.directory_filter.get();
144        let mut sessions = workspace.sessions.with(|sessions| {
145            sessions
146                .iter()
147                .filter(|session| {
148                    matches!(session.origin, cookie_agent_protocol::SessionOrigin::Root)
149                        && filter
150                            .as_deref()
151                            .is_none_or(|filter| session.cwd_identity.as_str() == filter)
152                })
153                .cloned()
154                .collect::<Vec<_>>()
155        });
156        sessions.sort_by_key(|session| std::cmp::Reverse(session.last_activity));
157        sessions
158    };
159    // The rows' "5m ago" moves on while the list shows.
160    let now = RwSignal::new(jiff::Timestamp::now());
161    let ticker = set_interval_with_handle(
162        move || now.set(jiff::Timestamp::now()),
163        Duration::from_secs(30),
164    )
165    .ok();
166    on_cleanup(move || {
167        if let Some(ticker) = ticker {
168            ticker.clear();
169        }
170    });
171    view! {
172        <div class="flex flex-col gap-2 p-3">
173            <button type="button" class=PRIMARY data-new-session on:click=move |_| workspace.start_new_session()>
174                <Glyph icon=icondata::FaPlusSolid />
175                "New session"
176            </button>
177            <label class="flex flex-col gap-1 text-xs font-bold text-ink dark:text-sand">
178                <span>"Directory"</span>
179                <select
180                    class="min-h-11 rounded-md border border-crust bg-cream px-2 text-sm font-normal text-espresso hover:border-caramel dark:border-smoke dark:bg-night dark:text-latte dark:hover:border-maple"
181                    on:change=move |event| {
182                        let value = event_target_value(&event);
183                        workspace.directory_filter.set((!value.is_empty()).then_some(value));
184                    }
185                >
186                    <option value="">"All directories"</option>
187                    {move || {
188                        directories()
189                            .into_iter()
190                            .map(|directory| {
191                                let selected = workspace.directory_filter.get().as_deref()
192                                    == Some(directory.as_str());
193                                view! {
194                                    <option value=directory.clone() selected=selected>
195                                        {directory.clone()}
196                                    </option>
197                                }
198                            })
199                            .collect_view()
200                    }}
201                </select>
202            </label>
203        </div>
204        <nav class="flex min-h-0 flex-1 flex-col gap-1 overflow-y-auto px-2 pb-3" aria-label="Sessions">
205            {move || {
206                let sessions = shown();
207                if sessions.is_empty() {
208                    return empty_state(icondata::FaFolderOpenSolid, "No sessions yet. Start one with New session.")
209                        .into_any();
210                }
211                sessions
212                    .into_iter()
213                    .map(|session| {
214                        let session_id = session.session_id;
215                        let current = move || workspace.selected.get() == Some(session_id);
216                        let mark = icons::session_status(session.status);
217                        let directory = session.cwd_identity.to_string();
218                        let last_activity = session.last_activity;
219                        let title = format!("{directory}\n{}", mark.label);
220                        view! {
221                            <button
222                                type="button"
223                                class=ROW
224                                class:bg-toasted=current
225                                class:dark:bg-ember=current
226                                aria-current=move || current().then_some("true")
227                                data-session=session_id.to_string()
228                                title=title
229                                on:click=move |_| workspace.open_session(session_id)
230                            >
231                                <span class="flex w-full items-center gap-2">
232                                    <Glyph icon=mark.icon class=mark.class("text-xs") />
233                                    <span class="sr-only">{mark.label}</span>
234                                    <span class="min-w-0 flex-1 truncate text-sm font-bold">
235                                        {session_label(&session)}
236                                    </span>
237                                </span>
238                                <span class=format!("flex w-full min-w-0 items-center gap-1 pl-5 {NOTE}")>
239                                    <Glyph icon=icondata::FaFolderSolid class="opacity-70" />
240                                    <span class="min-w-0 truncate">{basename(&directory).to_owned()}</span>
241                                    <span class="shrink-0">
242                                        {move || format!("· {}", relative_time(now.get(), last_activity))}
243                                    </span>
244                                </span>
245                            </button>
246                        }
247                    })
248                    .collect_view()
249                    .into_any()
250            }}
251        </nav>
252    }
253}

255#[component]
256fn Toolbar(workspace: Workspace) -> impl IntoView {
257    let title = move || {
258        if workspace.new_session.get() {
259            return "New session".to_owned();
260        }
261        workspace
262            .selected
263            .get()
264            .and_then(|session_id| {
265                workspace
266                    .sessions
267                    .with(|sessions| {
268                        sessions
269                            .iter()
270                            .find(|session| session.session_id == session_id)
271                            .map(session_label)
272                    })
273                    .or_else(|| {
274                        // A subagent's session is in the tree, not the list.
275                        workspace.tree.with(|tree| {
276                            tree.as_ref()
277                                .and_then(|tree| crate::tree::find(tree, session_id))
278                                .map(|session| {
279                                    format!(
280                                        "{} · {}",
281                                        session.creation_selection.agent,
282                                        session_label(session)
283                                    )
284                                })
285                        })
286                    })
287            })
288            .unwrap_or_else(|| "No session".into())
289    };
290    let agent = move || {
291        workspace
292            .draft
293            .get()
294            .map(|draft| draft.agent.to_string())
295            .unwrap_or_else(|| "agent".into())
296    };
297    let model = move || {
298        let draft = workspace.draft.get();
299        workspace.with_runtime(|runtime| {
300            draft
301                .map(|draft| runtime.model_names().selection_label(&draft.model))
302                .unwrap_or_else(|| "model".into())
303        })
304    };
305    let read_only = move || {
306        workspace.selected.get().is_some_and(|session_id| {
307            !workspace.new_session.get()
308                && workspace
309                    .read_only
310                    .with(|read_only| read_only.contains(&session_id))
311        })
312    };
313    view! {
314        <div class="flex flex-wrap items-center gap-x-2 gap-y-1 border-b border-toasted px-3 py-2 dark:border-ember">
315            <button
316                type="button"
317                class=format!("{ICON_BUTTON} md:hidden")
318                aria-label="Sessions"
319                title="Sessions"
320                on:click=move |_| workspace.panel.set(Some(Panel::Sessions))
321            >
322                <Glyph icon=icondata::FaBarsSolid class="text-lg" />
323            </button>
324            <div class="flex min-w-0 flex-1 basis-40 items-center gap-2">
325                <h1 class="min-w-0 truncate text-lg font-bold" id="session-title" title=title>
326                    {title}
327                </h1>
328                <Show when=read_only>
329                    <span class="inline-flex shrink-0 items-center gap-1 rounded-full bg-toasted px-2 py-1 text-xs dark:bg-ember">
330                        <Glyph icon=icondata::FaLockSolid />
331                        "read-only snapshot"
332                    </span>
333                </Show>
334            </div>
335            <div class="flex min-w-0 items-center gap-2">
336                <button
337                    type="button"
338                    class=CHIP
339                    title=move || format!("Agent: {} (click to change)", agent())
340                    on:click=move |_| workspace.panel.set(Some(Panel::Agents))
341                >
342                    <Glyph icon=icondata::FaRobotSolid class="text-caramel dark:text-maple" />
343                    <span class="sr-only">"Agent: "</span>
344                    <span class="min-w-0 truncate">{agent}</span>
345                    <Glyph icon=icondata::FaChevronDownSolid class="text-[0.625rem] opacity-60" />
346                </button>
347                <button
348                    type="button"
349                    class=CHIP
350                    title=move || format!("Model: {} (click to change)", model())
351                    on:click=move |_| workspace.panel.set(Some(Panel::Models))
352                >
353                    <Glyph icon=icondata::FaMicrochipSolid class="text-caramel dark:text-maple" />

372#[component]
373fn Transcript(workspace: Workspace) -> impl IntoView {
374    let cache = StoredValue::new(HashMap::<u64, Arc<str>>::new());
375    let items = Memo::new(move |_| {
376        let Some(session_id) = workspace
377            .selected
378            .get()
379            .filter(|_| !workspace.new_session.get())
380        else {
381            return Vec::new();
382        };
383        let level = workspace.level.get();
384        let expanded = workspace.expanded.get();
385        let names = workspace.with_runtime(|runtime| runtime.model_names().clone());
386        workspace.with_store(|store| {
387            let Some(state) = store.sessions.get(&session_id) else {
388                return Vec::new();
389            };
390            let view = TranscriptView {
391                session_id,
392                state,
393                names: &names,
394                minimum_level: level,
395                expanded: &expanded,
396            };
397            cache
398                .try_update_value(|cache| {
399                    let mut kept = HashMap::new();
400                    let rows = state
401                        .transcript
402                        .iter()
403                        .filter_map(|item| {
404                            let key = item_key(&view, item);
405                            let html = match cache.get(&key) {
406                                Some(html) => Arc::clone(html),
407                                None => Arc::from(item_html(&view, item)?),
408                            };
409                            kept.insert(key, Arc::clone(&html));
410                            Some((key, html))
411                        })
412                        .collect::<Vec<_>>();
413                    *cache = kept;
414                    rows
415                })
416                .unwrap_or_default()
417        })
418    });
419    let container = NodeRef::<html::Div>::new();
420    let following = StoredValue::new(true);
421    Effect::new(move |_| {
422        items.track();
423        let Some(element) = container.get() else {
424            return;
425        };
426        let element: web_sys::Element = element.into();
427        request_animation_frame(move || {
428            if following.get_value() {
429                element.set_scroll_top(element.scroll_height());
430            }
431            crate::highlight::highlight_within(&element);
432            crate::media::show_within(&element, workspace);
433        });
434    });
435    let on_click = move |event: ev::MouseEvent| {
436        let Some(target) = event
437            .target()
438            .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
439            .and_then(|target| target.closest("[data-action]").ok().flatten())
440        else {
441            return;
442        };
443        let action = target.get_attribute("data-action").unwrap_or_default();
444        let call = target.get_attribute("data-call");
445        match action.as_str() {
446            "toggle-tool" => {
447                if let Some(call) = call {
448                    workspace.expanded.update(|expanded| {
449                        if !expanded.remove(&call) {
450                            expanded.insert(call);
451                        }
452                    });
453                }
454            }
455            "full" => {
456                let block = target
457                    .get_attribute("data-block")
458                    .and_then(|block| block.parse().ok());
459                if let (Some(call), Some(block)) = (call, block)
460                    && let Ok(call) = call.parse::<cookie_agent_protocol::ToolCallId>()
461                {
462                    workspace.toggle_full_content(call, block);
463                }
464            }
465            "toggle-block" => {
466                if let Some(key) = target.get_attribute("data-block-key") {
467                    workspace.expanded.update(|expanded| {
468                        if !expanded.remove(&key) {
469                            expanded.insert(key);
470                        }
471                    });
472                }
473            }
474            "user-menu" => {
475                if let Some(seq) = target
476                    .get_attribute("data-seq")
477                    .and_then(|seq| seq.parse().ok())
478                {
479                    workspace.open_user_menu(seq);
480                }
481            }
482            "background" => {
483                if let Some(call) = call.and_then(|call| call.parse().ok()) {
484                    workspace.request_background(call);
485                }
486            }
487            "open-session" => {
488                if let Some(session_id) = target
489                    .get_attribute("data-session")
490                    .and_then(|session| session.parse().ok())
491                {
492                    workspace.open_session(session_id);
493                }
494            }
495            _ => {}
496        }
497    };
498    // Following stops only when the user scrolls away from the bottom
499    // (wheel, touch, or the scrollbar), never when an item's node is
500    // replaced or the box shrinks under a bar.
501    let near_bottom = move || {
502        container.get().is_some_and(|element| {
503            let element: web_sys::Element = element.into();
504            element.scroll_height() - element.scroll_top() - element.client_height() < 80
505        })
506    };
507    let user_scrolling = StoredValue::new(false);
508    let on_scroll = move |_| {
509        if near_bottom() {
510            following.set_value(true);
511        } else if user_scrolling.get_value() {
512            following.set_value(false);
513        }
514    };
515    view! {
516        <div
517            node_ref=container
518            id="transcript"
519            class="min-h-0 flex-1 overflow-y-auto px-3 py-2"
520            on:click=on_click
521            on:scroll=on_scroll
522            on:wheel=move |event: ev::WheelEvent| {
523                if event.delta_y() < 0.0 {
524                    following.set_value(false);
525                }
526            }
527            on:touchstart=move |_| user_scrolling.set_value(true)
528            on:touchend=move |_| user_scrolling.set_value(false)
529            on:pointerdown=move |_| user_scrolling.set_value(true)
530            on:pointerup=move |_| user_scrolling.set_value(false)
531        >
532            <Show when=move || workspace.new_session.get()>
533                <NewSessionCard workspace=workspace />
534            </Show>
535            <div class="mx-auto flex max-w-4xl flex-col gap-2">
536                <For
537                    each=move || items.get()
538                    key=|(key, _)| *key
539                    children=move |(_, html)| {
540                        view! { <div class="transcript-item min-w-0" inner_html=html.to_string()></div> }
541                    }
542                />
543            </div>
544        </div>
545    }
546}

578fn QueueStrip(workspace: Workspace) -> impl IntoView {

580    view! {

593                                        title="Take back into the composer"
594                                        on:click=move |_| workspace.recall(seq)
595                                    >

623    let textarea = NodeRef::<html::Textarea>::new();
624    let long_pressed = StoredValue::new(false);

626    let menu_open = RwSignal::new(false);
627    let completions = move || completions(&workspace.composer.get());
628    let insert_newline = move || {
629        let Some(textarea) = textarea.get() else {
630            return;
631        };
632        let start = textarea.selection_start().ok().flatten().unwrap_or(0);
633        let end = textarea.selection_end().ok().flatten().unwrap_or(start);
634        let _ = textarea.set_range_text_with_start_and_end_and_mode("\n", start, end, "end");
635        workspace.composer.set(textarea.value());
636    };
637    let on_keydown = move |event: ev::KeyboardEvent| {
638        let key = event.key();
639        let action = composer_action(KeyPress {
640            key: &key,
641            alt: event.alt_key(),
642            shift: event.shift_key(),
643            ctrl: event.ctrl_key(),
644            meta: event.meta_key(),
645            empty: workspace.composer.with_untracked(String::is_empty),
646        });
647        match action {
648            ComposerAction::Type => return,
649            ComposerAction::Steer => workspace.submit(InputDelivery::Steer),
650            ComposerAction::Queue => workspace.submit(InputDelivery::Queue),
651            ComposerAction::Newline if event.shift_key() => return,
652            ComposerAction::Newline => insert_newline(),
653            ComposerAction::Cancel => workspace.cancel(),
654            ComposerAction::Recall => workspace.recall_newest(),
655        }
656        event.prevent_default();
657    };
658    view! {
659        <div class="relative border-t border-toasted px-3 py-2 dark:border-ember">
660            <Show when=move || !completions().is_empty()>
661                <div class="absolute bottom-full left-3 z-20 mb-1 flex max-h-64 w-80 max-w-[calc(100%-1.5rem)] flex-col overflow-y-auto rounded-lg border border-crust bg-cream p-1 shadow-lg dark:border-smoke dark:bg-night">
662                    {move || {

664                            .into_iter()

678                                                    let _ = textarea.focus();
679                                                }
680                                            } else {
681                                                workspace.composer.set(String::new());
682                                                workspace.run_command(command, String::new());
683                                            }
684                                        }
685                                    >
686                                        <span class="font-mono text-sm font-bold">{format!("/{}", spec.name)}</span>

1206fn usage_sheet(workspace: Workspace) -> impl IntoView {
1207    move || match workspace.usage.get() {
1208        None => loading("Reading usage…").into_any(),
1209        Some(usage) => {
1210            let rows = |usage: &cookie_agent_protocol::UsageRollup| {
1211                let mut rows = vec![
1212                    ("input tokens", usage.input_tokens.to_string()),
1213                    ("output tokens", usage.output_tokens.to_string()),
1214                    ("reasoning tokens", usage.reasoning_tokens.to_string()),
1215                    ("cache reads", usage.cache_read_tokens.to_string()),
1216                    ("cache writes", usage.cache_write_tokens.to_string()),
1217                    ("requests", usage.request_count.to_string()),
1218                ];
1219                if let Some(cost) = usage.estimated_cost_usd {
1220                    rows.push(("estimated cost", format!("${cost:.4}")));
1221                }
1222                rows.into_iter()
1223                    .map(|(name, value)| view! {
1224                        <dt class="border-b border-toasted py-1 text-ink dark:border-ember dark:text-sand">{name}</dt>
1225                        <dd class="border-b border-toasted py-1 text-right font-mono dark:border-ember">{value}</dd>
1226                    })
1227                    .collect_view()
1228            };
1229            let session = match &usage.session {
1230                Ok(rollup) => {
1231                    view! { <dl class="grid grid-cols-2 text-sm">{rows(rollup)}</dl> }.into_any()
1232                }
1233                Err(error) => view! { <p class="text-sm">{error.clone()}</p> }.into_any(),
1234            };
1235            let tree = match &usage.tree {
1236                Ok((rollup, count)) => view! {
1237                    <h3 class="mt-4 text-sm font-bold">
1238                        {if *count == 1 { "Session tree (1 session)".to_owned() } else { format!("Session tree ({count} sessions)") }}
1239                    </h3>
1240                    <dl class="grid grid-cols-2 text-sm">{rows(rollup)}</dl>
1241                }
1242                    .into_any(),
1243                Err(error) => view! { <p class="text-sm">{error.clone()}</p> }.into_any(),
1244            };
1245            view! {
1246                <h3 class="text-sm font-bold">"This session"</h3>
1247                {session}
1248                {tree}
1249            }
1250            .into_any()
1251        }
1252    }
1253}
