49/// A session's name in the list: its title, else its short ID.
50fn session_label(session: &SessionMeta) -> String {
51    session
52        .title
53        .as_ref()
54        .map(ToString::to_string)
55        .or_else(|| session.short_id.clone())
56        .unwrap_or_else(|| session.session_id.to_string())
57}

59#[component]
60fn SessionList(workspace: Workspace) -> impl IntoView {
61    let directories = move || {
62        let mut directories = workspace.sessions.with(|sessions| {
63            sessions
64                .iter()
65                .map(|session| session.cwd_identity.to_string())
66                .collect::<Vec<_>>()
67        });
68        directories.sort();
69        directories.dedup();
70        directories
71    };
72    let shown = move || {
73        let filter = workspace.directory_filter.get();
74        let mut sessions = workspace.sessions.with(|sessions| {
75            sessions
76                .iter()
77                .filter(|session| {
78                    matches!(session.origin, cookie_agent_protocol::SessionOrigin::Root)
79                        && filter
80                            .as_deref()
81                            .is_none_or(|filter| session.cwd_identity.as_str() == filter)
82                })
83                .cloned()
84                .collect::<Vec<_>>()
85        });
86        sessions.sort_by_key(|session| std::cmp::Reverse(session.last_activity));
87        sessions
88    };
89    view! {
90        <div class="flex flex-col gap-2 p-3">
91            <button type="button" class=BUTTON on:click=move |_| workspace.start_new_session()>
92                "New session"
93            </button>
94            <label class="flex flex-col gap-1 text-xs">
95                <span>"Directory"</span>
96                <select
97                    class="min-h-11 rounded border border-crust bg-cream px-2 text-sm hover:border-caramel dark:border-smoke dark:bg-night dark:hover:border-maple"
98                    on:change=move |event| {
99                        let value = event_target_value(&event);
100                        workspace.directory_filter.set((!value.is_empty()).then_some(value));
101                    }
102                >
103                    <option value="">"All directories"</option>
104                    {move || {
105                        directories()
106                            .into_iter()
107                            .map(|directory| {
108                                let selected = workspace.directory_filter.get().as_deref()
109                                    == Some(directory.as_str());
110                                view! {
111                                    <option value=directory.clone() selected=selected>
112                                        {directory.clone()}
113                                    </option>
114                                }
115                            })
116                            .collect_view()
117                    }}
118                </select>
119            </label>
120        </div>
121        <nav class="flex min-h-0 flex-1 flex-col gap-1 overflow-y-auto px-2 pb-3" aria-label="Sessions">
122            {move || {
123                shown()
124                    .into_iter()
125                    .map(|session| {
126                        let session_id = session.session_id;
127                        let current = move || workspace.selected.get() == Some(session_id);
128                        view! {
129                            <button
130                                type="button"
131                                class=ROW
132                                class:bg-toasted=current
133                                data-session=session_id.to_string()
134                                on:click=move |_| workspace.open_session(session_id)
135                            >
136                                <span class="w-full truncate text-sm font-bold">
137                                    {session_label(&session)}
138                                </span>
139                                <span class="w-full truncate font-mono text-xs text-ink dark:text-sand">
140                                    {format!("{:?} · {}", session.status, session.cwd_identity)
141                                        .to_lowercase()}
142                                </span>
143                            </button>
144                        }
145                    })
146                    .collect_view()
147            }}
148        </nav>
149    }
150}

152#[component]
153fn Toolbar(workspace: Workspace) -> impl IntoView {
154    let title = move || {
155        if workspace.new_session.get() {
156            return "New session".to_owned();
157        }
158        workspace
159            .selected
160            .get()
161            .and_then(|session_id| {
162                workspace.sessions.with(|sessions| {
163                    sessions
164                        .iter()
165                        .find(|session| session.session_id == session_id)
166                        .map(session_label)
167                })
168            })
169            .unwrap_or_else(|| "No session".into())
170    };
171    let agent = move || {
172        workspace
173            .draft
174            .get()
175            .map(|draft| draft.agent.to_string())
176            .unwrap_or_else(|| "agent".into())
177    };
178    let model = move || {
179        let draft = workspace.draft.get();
180        workspace.with_runtime(|runtime| {
181            draft
182                .map(|draft| runtime.model_names().selection_label(&draft.model))
183                .unwrap_or_else(|| "model".into())
184        })
185    };
186    let read_only = move || {
187        workspace.selected.get().is_some_and(|session_id| {
188            !workspace.new_session.get()
189                && workspace
190                    .read_only
191                    .with(|read_only| read_only.contains(&session_id))
192        })
193    };
194    view! {
195        <div class="flex flex-wrap items-center gap-2 border-b border-toasted px-3 py-2 dark:border-ember">
196            <button
197                type="button"
198                class=format!("{BUTTON} md:hidden")
199                on:click=move |_| workspace.panel.set(Some(Panel::Sessions))
200            >
201                "Sessions"
202            </button>
203            <h1 class="min-w-0 flex-1 truncate font-bold" id="session-title">
204                {title}
205            </h1>
206            <Show when=read_only>
207                <span class="rounded bg-toasted px-2 py-1 text-xs dark:bg-ember">"read-only snapshot"</span>
208            </Show>
209            <button
210                type="button"
211                class=BUTTON
212                on:click=move |_| workspace.panel.set(Some(Panel::Agents))
213            >
214                {agent}
215            </button>
216            <button
217                type="button"
218                class=BUTTON
219                on:click=move |_| workspace.panel.set(Some(Panel::Models))
220            >
221                {model}
222            </button>
223            <button
224                type="button"
225                class=BUTTON
226                aria-label="Commands"
227                on:click=move |_| workspace.panel.set(Some(Panel::Commands))
228            >
229                "☰"
230            </button>
231        </div>
232    }

235#[component]
236fn Transcript(workspace: Workspace) -> impl IntoView {
237    let cache = StoredValue::new(HashMap::<u64, Arc<str>>::new());
238    let items = Memo::new(move |_| {
239        let Some(session_id) = workspace
240            .selected
241            .get()
242            .filter(|_| !workspace.new_session.get())
243        else {
244            return Vec::new();
245        };
246        let level = workspace.level.get();
247        let expanded = workspace.expanded.get();
248        let names = workspace.with_runtime(|runtime| runtime.model_names().clone());
249        workspace.with_store(|store| {
250            let Some(state) = store.sessions.get(&session_id) else {
251                return Vec::new();
252            };
253            let view = TranscriptView {
254                state,
255                names: &names,
256                minimum_level: level,
257                expanded: &expanded,
258            };
259            cache
260                .try_update_value(|cache| {
261                    let mut kept = HashMap::new();
262                    let rows = state
263                        .transcript
264                        .iter()
265                        .filter_map(|item| {
266                            let key = item_key(&view, item);
267                            let html = match cache.get(&key) {
268                                Some(html) => Arc::clone(html),
269                                None => Arc::from(item_html(&view, item)?),
270                            };
271                            kept.insert(key, Arc::clone(&html));
272                            Some((key, html))
273                        })
274                        .collect::<Vec<_>>();
275                    *cache = kept;
276                    rows
277                })
278                .unwrap_or_default()
279        })
280    });
281    let container = NodeRef::<html::Div>::new();
282    let following = StoredValue::new(true);
283    Effect::new(move |_| {
284        items.track();
285        let Some(element) = container.get() else {
286            return;
287        };
288        let element: web_sys::Element = element.into();
289        request_animation_frame(move || {
290            if following.get_value() {
291                element.set_scroll_top(element.scroll_height());
292            }
293            crate::highlight::highlight_within(&element);
294            crate::media::show_within(&element, workspace);
295        });
296    });
297    let on_click = move |event: ev::MouseEvent| {
298        let Some(target) = event
299            .target()
300            .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
301            .and_then(|target| target.closest("[data-action]").ok().flatten())
302        else {
303            return;
304        };
305        let action = target.get_attribute("data-action").unwrap_or_default();
306        let call = target.get_attribute("data-call");
307        match action.as_str() {
308            "toggle-tool" => {
309                if let Some(call) = call {
310                    workspace.expanded.update(|expanded| {
311                        if !expanded.remove(&call) {
312                            expanded.insert(call);
313                        }
314                    });
315                }
316            }
317            "full" => {
318                let block = target
319                    .get_attribute("data-block")
320                    .and_then(|block| block.parse().ok());
321                if let (Some(call), Some(block)) = (call, block)
322                    && let Ok(call) = call.parse::<cookie_agent_protocol::ToolCallId>()
323                {
324                    workspace.toggle_full_content(call, block);
325                }
326            }
327            "open-session" => {
328                if let Some(session_id) = target
329                    .get_attribute("data-session")
330                    .and_then(|session| session.parse().ok())
331                {
332                    workspace.open_session(session_id);
333                }
334            }
335            _ => {}
336        }
337    };
338    let on_scroll = move |_| {
339        if let Some(element) = container.get() {
340            let element: web_sys::Element = element.into();
341            following.set_value(
342                element.scroll_height() - element.scroll_top() - element.client_height() < 80,
343            );
344        }
345    };
346    view! {
347        <div
348            node_ref=container
349            id="transcript"
350            class="min-h-0 flex-1 overflow-y-auto px-3 py-2"
351            on:click=on_click
352            on:scroll=on_scroll
353        >
354            <Show when=move || workspace.new_session.get()>
355                <NewSessionCard workspace=workspace />
356            </Show>
357            <div class="mx-auto flex max-w-4xl flex-col gap-2">
358                <For
359                    each=move || items.get()
360                    key=|(key, _)| *key
361                    children=move |(_, html)| {
362                        view! { <div class="transcript-item min-w-0" inner_html=html.to_string()></div> }
363                    }
364                />
365            </div>
366        </div>
367    }
368}

431#[component]
432fn Composer(workspace: Workspace) -> impl IntoView {
433    let textarea = NodeRef::<html::Textarea>::new();
434    let long_pressed = StoredValue::new(false);
435    let press_timer = StoredValue::new(None::<TimeoutHandle>);
436    let menu_open = RwSignal::new(false);
437    let completions = move || completions(&workspace.composer.get());
438    let insert_newline = move || {
439        let Some(textarea) = textarea.get() else {
440            return;
441        };
442        let start = textarea.selection_start().ok().flatten().unwrap_or(0);
443        let end = textarea.selection_end().ok().flatten().unwrap_or(start);
444        let _ = textarea.set_range_text_with_start_and_end_and_mode("\n", start, end, "end");
445        workspace.composer.set(textarea.value());
446    };
447    let on_keydown = move |event: ev::KeyboardEvent| {
448        let key = event.key();
449        let action = composer_action(KeyPress {
450            key: &key,
451            alt: event.alt_key(),
452            shift: event.shift_key(),
453            ctrl: event.ctrl_key(),
454            meta: event.meta_key(),
455            empty: workspace.composer.with_untracked(String::is_empty),
456        });
457        match action {
458            ComposerAction::Type => return,
459            ComposerAction::Steer => workspace.submit(InputDelivery::Steer),
460            ComposerAction::Queue => workspace.submit(InputDelivery::Queue),
461            ComposerAction::Newline if event.shift_key() => return,
462            ComposerAction::Newline => insert_newline(),
463            ComposerAction::Cancel => workspace.cancel(),
464            ComposerAction::Recall => workspace.recall_newest(),
465        }
466        event.prevent_default();
467    };
468    view! {
469        <div class="relative border-t border-toasted px-3 py-2 dark:border-ember">
470            <Show when=move || !completions().is_empty()>
471                <div class="absolute bottom-full left-3 mb-1 flex max-h-64 flex-col overflow-y-auto rounded border border-crust bg-cream shadow dark:border-smoke dark:bg-night">
472                    {move || {
473                        completions()
474                            .into_iter()
475                            .map(|spec| {
476                                let command = spec.command;
477                                view! {
478                                    <button
479                                        type="button"
480                                        class=ROW
481                                        on:click=move |_| {
482                                            workspace.composer.set(String::new());
483                                            workspace.run_command(command);
484                                        }
485                                    >
486                                        <span class="font-mono text-sm">{format!("/{}", spec.name)}</span>
487                                        <span class="text-xs text-ink dark:text-sand">{spec.description}</span>
488                                    </button>
489                                }
490                            })
491                            .collect_view()
492                    }}
493                </div>
494            </Show>
495            <div class="flex items-end gap-2">
496                <textarea
497                    node_ref=textarea
498                    id="composer"
499                    rows="2"
500                    class="min-h-11 flex-1 resize-y rounded border border-crust bg-cream px-3 py-2 font-mono text-sm hover:border-caramel focus:border-caramel focus:outline-none disabled:opacity-60 dark:border-smoke dark:bg-night dark:hover:border-maple dark:focus:border-maple"
501                    placeholder="Message · / for commands"
502                    disabled=move || !workspace.writable()
503                    prop:value=move || workspace.composer.get()
504                    on:input=move |event| workspace.composer.set(event_target_value(&event))
505                    on:keydown=on_keydown
506                ></textarea>
507                <div class="relative flex">
508                    <button
509                        type="button"
510                        id="send"
511                        class="min-h-11 rounded-l bg-espresso px-4 font-bold text-cream hover:bg-caramel active:bg-ink disabled:opacity-60 dark:bg-latte dark:text-night dark:hover:bg-maple dark:active:bg-sand"
512                        disabled=move || !workspace.writable()
513                        on:pointerdown=move |_| {
514                            long_pressed.set_value(false);
515                            let handle = set_timeout_with_handle(
516                                move || {
517                                    long_pressed.set_value(true);
518                                    workspace.submit(InputDelivery::Queue);
519                                },
520                                LONG_PRESS,
521                            )
522                            .ok();
523                            press_timer.set_value(handle);
524                        }
525                        on:pointerup=move |_| {
526                            if let Some(handle) = press_timer.get_value() {
527                                handle.clear();
528                            }
529                        }
530                        on:pointerleave=move |_| {
531                            if let Some(handle) = press_timer.get_value() {
532                                handle.clear();
533                            }
534                        }
535                        on:click=move |_| {
536                            if !long_pressed.get_value() {
537                                workspace.submit(InputDelivery::Steer);
538                            }
539                            long_pressed.set_value(false);
540                        }
541                    >
542                        "Send"
543                    </button>
544                    <button
545                        type="button"
546                        aria-label="More ways to send"
547                        class="min-h-11 rounded-r border-l border-cream bg-espresso px-2 text-cream hover:bg-caramel active:bg-ink dark:border-night dark:bg-latte dark:text-night dark:hover:bg-maple dark:active:bg-sand"
548                        on:click=move |_| menu_open.update(|open| *open = !*open)
549                    >
550                        "▾"
551                    </button>
552                    <Show when=move || menu_open.get()>
553                        <div class="absolute right-0 bottom-full mb-1 flex w-48 flex-col rounded border border-crust bg-cream shadow dark:border-smoke dark:bg-night">
554                            <button
555                                type="button"
556                                id="queue-send"
557                                class=ROW
558                                on:click=move |_| {
559                                    menu_open.set(false);
560                                    workspace.submit(InputDelivery::Queue);
561                                }
562                            >
563                                "Queue behind the run"
564                            </button>
565                            <button
566                                type="button"
567                                id="cancel-run"
568                                class=ROW
569                                on:click=move |_| {
570                                    menu_open.set(false);
571                                    workspace.cancel();
572                                }
573                            >
574                                "Cancel the run"
575                            </button>
576                        </div>
577                    </Show>
578                </div>
579            </div>
580        </div>
581    }
582}

584#[component]
585fn StatusBar(workspace: Workspace) -> impl IntoView {
586    let running = move || {
587        workspace.selected.get().is_some_and(|session_id| {
588            workspace.with_store(|store| {
589                store.sessions.get(&session_id).is_some_and(|state| {
590                    state.active_run.is_some()
591                        || state
592                            .tools
593                            .values()
594                            .any(|tool| tool.status == ToolStatus::Running)
595                })
596            })
597        })
598    };
599    let cost = move || {
600        workspace
601            .selected
602            .get()
603            .and_then(|session_id| {
604                workspace
605                    .costs
606                    .with(|costs| costs.get(&session_id).copied())
607            })
608            .map(|cost| format!("${cost:.4}"))
609    };
610    let context = move || {
611        workspace.selected.get().and_then(|session_id| {
612            workspace.with_store(|store| {
613                store
614                    .sessions
615                    .get(&session_id)
616                    .and_then(|state| state.context_tokens)
617            })
618        })
619    };
620    let approvals = Memo::new(move |_| workspace.approvals().len());
621    view! {
622        <div id="status-bar" class="flex flex-wrap items-center gap-2 border-t border-toasted px-3 py-1 text-xs dark:border-ember">
623            <button
624                type="button"
625                id="permission-mode"
626                class=format!("{BUTTON} font-mono")
627                title="Permission mode: click for the next"
628                on:click=move |_| workspace.cycle_permission_mode()
629            >
630                {move || permission_mode_label(workspace.permission_mode())}
631            </button>
632            <Show when=move || { approvals.get() > 0 }>
633                <button
634                    type="button"
635                    id="approvals-badge"
636                    class="min-h-11 rounded bg-caramel px-3 font-bold text-cream hover:bg-espresso active:bg-ink dark:bg-maple dark:text-night dark:hover:bg-latte dark:active:bg-sand"
637                    on:click=move |_| workspace.panel.set(Some(Panel::Approvals))
638                >
639                    {move || format!("{} approval(s) waiting", approvals.get())}
640                </button>
641            </Show>
642            <Show when=running>
643                <span class="text-caramel dark:text-maple">"working…"</span>
644            </Show>
645            {move || cost().map(|cost| view! {
646                <button
647                    type="button"
648                    class=BUTTON
649                    on:click=move |_| workspace.run_command(crate::commands::Command::Usage)
650                >
651                    {cost}
652                </button>
653            })}
654            {move || context().map(|tokens| view! { <span class="font-mono">{format!("ctx {tokens}")}</span> })}

803fn approvals_sheet(workspace: Workspace) -> impl IntoView {
804    move || {
805        let approvals = workspace.approvals();
806        if approvals.is_empty() {
807            return view! { <p class="text-sm">"No approval is waiting."</p> }.into_any();
808        }
809        approvals
810            .into_iter()
811            .map(|approval| {
812                let (title, preview) = workspace.with_store(|store| {
813                    store
814                        .sessions
815                        .get(&approval.session_id)
816                        .and_then(|state| state.tools.get(&approval.tool_call_id))
817                        .map(|tool| {
818                            (
819                                tool.compact_title(),
820                                display_html(&tool.presentation.preview, BlockPlace::plain),
821                            )
822                        })
823                        .unwrap_or_else(|| ("a tool call".into(), String::new()))
824                });
825                let decide = move |decision: ApprovalUserDecision| {
826                    let approval = approval.clone();
827                    move |_| workspace.respond(approval.clone(), decision)
828                };
829                view! {
830                    <div class="approval flex flex-col gap-2 rounded border border-caramel p-3 dark:border-maple">
831                        <p class="font-mono text-sm">{title}</p>
832                        <div class="min-w-0" inner_html=preview></div>
833                        <div class="flex flex-wrap gap-2">
834                            <button type="button" class=BUTTON data-decision="approve_once" on:click=decide(ApprovalUserDecision::ApproveOnce)>
835                                "Approve once"
836                            </button>
837                            <button type="button" class=BUTTON data-decision="approve_tree" on:click=decide(ApprovalUserDecision::ApproveTree)>
838                                "Approve for this session tree"
839                            </button>
840                            <button type="button" class=BUTTON data-decision="reject" on:click=decide(ApprovalUserDecision::Reject)>
841                                "Reject"
842                            </button>
843                        </div>
844                    </div>
845                }
846            })
847            .collect_view()
848            .into_any()
849    }
850}

852fn usage_sheet(workspace: Workspace) -> impl IntoView {
853    move || match workspace.usage.get() {
854        None => view! { <p class="text-sm">"Reading usage…"</p> }.into_any(),
855        Some(usage) => {
856            let rows = |usage: &cookie_agent_protocol::UsageRollup| {
857                let mut rows = vec![
858                    ("input tokens", usage.input_tokens.to_string()),
859                    ("output tokens", usage.output_tokens.to_string()),
860                    ("reasoning tokens", usage.reasoning_tokens.to_string()),
861                    ("cache reads", usage.cache_read_tokens.to_string()),
862                    ("cache writes", usage.cache_write_tokens.to_string()),
863                    ("requests", usage.request_count.to_string()),
864                ];
865                if let Some(cost) = usage.estimated_cost_usd {
866                    rows.push(("estimated cost", format!("${cost:.4}")));
867                }
868                rows.into_iter()
869                    .map(|(name, value)| view! { <dt>{name}</dt><dd class="text-right font-mono">{value}</dd> })
870                    .collect_view()
871            };
872            let session = match &usage.session {
873                Ok(rollup) => {
874                    view! { <dl class="grid grid-cols-2 gap-x-3 text-sm">{rows(rollup)}</dl> }
875                        .into_any()
876                }
877                Err(error) => view! { <p class="text-sm">{error.clone()}</p> }.into_any(),
878            };
879            let tree = match &usage.tree {
880                Ok((rollup, count)) => view! {
881                    <p class="mt-3 text-sm font-bold">{format!("Session tree ({count} sessions)")}</p>
882                    <dl class="grid grid-cols-2 gap-x-3 text-sm">{rows(rollup)}</dl>
883                }
884                    .into_any(),
885                Err(error) => view! { <p class="text-sm">{error.clone()}</p> }.into_any(),
886            };
887            view! {
888                <p class="text-sm font-bold">"This session"</p>
889                {session}
890                {tree}
891            }
892            .into_any()
893        }
894    }
895}
