219#[component]
220fn SessionList(workspace: Workspace) -> impl IntoView {
221    // Each runtime at each workspace sessions work in, under its home.
222    let places = move || {
223        let homes = workspace.homes();
224        workspace.sessions.with(|sessions| {
225            let roots = sessions
226                .iter()
227                .filter(|session| {
228                    matches!(session.origin, cookie_agent_protocol::SessionOrigin::Root)
229                })
230                .cloned()
231                .collect::<Vec<_>>();
232            crate::places::filter_options(&roots, &homes)
233        })
234    };
235    let shown = move || {
236        let filter = workspace.directory_filter.get();
237        let mut sessions = workspace.sessions.with(|sessions| {
238            sessions
239                .iter()
240                .filter(|session| {
241                    matches!(session.origin, cookie_agent_protocol::SessionOrigin::Root)
242                        && filter
243                            .as_deref()
244                            .is_none_or(|filter| crate::places::matches(session, filter))
245                })
246                .cloned()
247                .collect::<Vec<_>>()
248        });
249        // Running sessions first, then the rest; newest first in each.
250        cookie_agent_client_state::rows::order_sessions(&mut sessions);
251        sessions
252    };
253    // The rows' "5m ago" moves on while the list shows.
254    let now = RwSignal::new(jiff::Timestamp::now());
255    let ticker = set_interval_with_handle(
256        move || now.set(jiff::Timestamp::now()),
257        Duration::from_secs(30),
258    )
259    .ok();
260    on_cleanup(move || {
261        if let Some(ticker) = ticker {
262            ticker.clear();
263        }
264    });
265    // Renaming and deleting: the row a long press revealed the actions
266    // of, the row whose delete waits for confirmation, and the row whose
267    // title is a field.
268    let actions = RowActions {
269        hovered: RwSignal::new(None),
270        revealed: RwSignal::new(None),
271        confirming: RwSignal::new(None),
272        renaming: RwSignal::new(None),
273        draft: RwSignal::new(String::new()),
274        fresh: StoredValue::new(false),
275    };
276    // Dragging a session onto a project's header, and the headers' own
277    // buttons.
278    let drag = projects::SessionDrag::new();
279    drag.follow(workspace, actions);
280    let headers = projects::HeaderActions::new();
281    let list = drag.list;
282    view! {
283        <div class="flex flex-col gap-2 p-3">
284            <div class="flex gap-2">
285                <button type="button" class=format!("{PRIMARY} flex-1") data-new-session on:click=move |_| workspace.start_new_session()>
286                    <Glyph icon=icondata::FaPlusSolid />
287                    "New session"
288                </button>
289                <Show when=move || workspace.has_projects.get()>
290                    <button
291                        type="button"
292                        class=BUTTON
293                        id="new-project"
294                        aria-label="New project…"
295                        title="New project…"
296                        on:click=move |_| workspace.open_project_editor(None, false)
297                    >
298                        <Glyph icon=icondata::FaFolderPlusSolid />
299                    </button>
300                </Show>
301            </div>
302            // A daemon without projects: the workspace filter instead of
303            // project sections.
304            <Show when=move || !workspace.has_projects.get()>
305            <label class="flex flex-col gap-1 text-xs font-bold text-ink dark:text-sand">
306                <span>"Workspace"</span>
307                <select
308                    id="place-filter"
309                    class="min-h-11 rounded-md border border-crust bg-cream px-2 text-sm font-normal text-espresso hover:border-caramel dark:border-smoke dark:bg-night dark:text-latte dark:hover:border-maple"
310                    on:change=move |event| {
311                        let value = event_target_value(&event);
312                        workspace.directory_filter.set((!value.is_empty()).then_some(value));
313                    }
314                >
315                    <option value="">"All workspaces"</option>
316                    {move || {
317                        places()
318                            .into_iter()
319                            .map(|(value, label)| {
320                                let selected = workspace.directory_filter.get().as_deref()
321                                    == Some(value.as_str());
322                                view! {
323                                    <option value=value selected=selected>
324                                        {label}
325                                    </option>
326                                }
327                            })
328                            .collect_view()
329                    }}
330                </select>
331            </label>
332            </Show>
333        </div>
334        <nav class="flex min-h-0 flex-1 flex-col gap-1 overflow-x-hidden overflow-y-auto px-2 pb-3" aria-label="Sessions" node_ref=list>
335            {move || {
336                if workspace.has_projects.get() {
337                    return projects::project_list(workspace, now, actions, drag, headers).into_any();
338                }
339                let sessions = shown();
340                if sessions.is_empty() {
341                    return empty_state(icondata::FaFolderOpenSolid, "No sessions yet. Start one with New session.")
342                        .into_any();
343                }
344                sessions
345                    .into_iter()
346                    .map(|session| session_row(workspace, session, now, actions, drag))
347                    .collect_view()
348                    .into_any()
349            }}
350        </nav>
351    }
352}

354/// The session list's row actions, shared by its rows: move, rename, and
355/// delete.
356#[derive(Clone, Copy)]
357struct RowActions {
358    /// The row a mouse (or pen) is over, which shows its buttons.
359    hovered: RwSignal<Option<SessionId>>,
360    /// The row a long press offered its actions under.
361    revealed: RwSignal<Option<SessionId>>,
362    confirming: RwSignal<Option<SessionId>>,
363    /// The row whose title is a field, and what the field holds; kept
364    /// here, so a list redrawn while the user types keeps it.
365    renaming: RwSignal<Option<SessionId>>,
366    draft: RwSignal<String>,
367    /// The field just opened: its text is selected once it shows.
368    fresh: StoredValue<bool>,
369}
370
371impl RowActions {
372    /// Turns `session`'s title into a field holding it.
373    fn begin_rename(self, workspace: Workspace, session_id: SessionId) {
374        self.revealed.set(None);
375        self.confirming.set(None);
376        self.draft.set(workspace.title_draft(session_id));
377        self.fresh.set_value(true);
378        self.renaming.set(Some(session_id));
379    }
380}

382/// The right padding a row keeps clear of its action buttons: room for
383/// them while they show (or the keyboard is in the row), else the row's
384/// own, which leaves its badges the room. A third button left of the
385/// pencil makes it wider still.
386fn row_padding(shown: bool, buttons: usize) -> &'static str {
387    match (shown, buttons) {
388        (true, 0 | 1) => "pr-12",
389        (false, 0 | 1) => "pr-3 group-has-focus-visible:pr-12",
390        (true, 2) => "pr-24",
391        (false, 2) => "pr-3 group-has-focus-visible:pr-24",
392        (true, _) => "pr-36",
393        (false, _) => "pr-3 group-has-focus-visible:pr-36",
394    }
395}

397/// A root session's row: a click opens it. On hover (or a long press on
398/// a touch screen) its buttons offer to rename it, in place, and to
399/// delete it, behind an inline confirmation. Its second line says where
400/// it runs, as badges. The open session's row has its subagents under it.
401fn session_row(
402    workspace: Workspace,
403    session: SessionMeta,
404    now: RwSignal<jiff::Timestamp>,
405    actions: RowActions,
406    drag: projects::SessionDrag,
407) -> impl IntoView {
408    let session_id = session.session_id;
409    let current = move || workspace.selected.get() == Some(session_id);
410    // The root of the open tree, whether the root itself or one of its
411    // subagents is open, and only when it has subagents: its row keeps the
412    // chevron that folds and unfolds the tree.
413    let tree_root =
414        move || workspace.tree_root.get() == Some(session_id) && workspace.tree_has_subagents();
415    let mark = icons::session_status(session.status);
416    let activity = cookie_agent_client_state::activity::session_activity(&session);
417    let status_mark = icons::row_status(&activity);
418    // What the row's tooltip and a screen reader say of it: what it waits
419    // on, else its status.
420    let status_text = match &activity {
421        cookie_agent_client_state::activity::SessionActivity::Waiting(kinds) => {
422            cookie_agent_client_state::activity::waiting_label(kinds)
423        }
424        _ => mark.label.to_owned(),
425    };
426    let last_activity = session.last_activity;
427    // Where it runs, once the runtimes are known; the tooltip lists every
428    // workspace with its path.
429    // Fitted to the line: machines whole, the last ones left out for a
430    // `+N` when even the workspaces gone to `…` leave too little room.
431    let badges = {
432        let session = session.clone();
433        move |line: NodeRef<html::Span>| {
434            let homes = workspace.homes();
435            let listed = workspace.lists_attachments.get();
436            let shown = workspace
437                .runtimes
438                .with(|listing| session_badges(&session, listing.as_ref(), &homes, listed));
439            crate::badges::fit_soon(line);
440            crate::badges::badge_line(shown)
441        }
442    };
443    let badge_line = NodeRef::<html::Span>::new();
444    crate::badges::keep_fitted(badge_line);
445    let title = {
446        let session = session.clone();
447        let status = status_text.clone();
448        move || {
449            let homes = workspace.homes();
450            let listed = workspace.lists_attachments.get();
451            let places = workspace
452                .runtimes
453                .with(|listing| badges_tooltip(&session, listing.as_ref(), &homes, listed));
454            format!("{places}\n{status}")
455        }
456    };
457    let label = session_label(&session);
458    let confirm_label = label.clone();
459    let RowActions {
460        hovered,
461        revealed,
462        confirming,
463        renaming,
464        ..
465    } = actions;
466    let is_renaming = move || renaming.get() == Some(session_id);
467    let movable = move || workspace.has_projects.get();
468    let buttons = move || 1 + usize::from(workspace.can_delete.get()) + usize::from(movable());
469    let shown = move || hovered.get() == Some(session_id) && !is_renaming();
470    // A mouse drags the row onto a project's header at once; a finger
471    // after a long press, which let go in place reveals the buttons.
472    let on_pointerdown = move |event: ev::PointerEvent| {
473        if !is_renaming() {
474            drag.press(workspace, session_id, &event);
475        }
476    };
477    let lifted = move || drag.lifts(session_id);
478    let field_badges = badges.clone();
479    view! {
480        <div
481            class=move || format!(
482                "session-row group relative {}",
483                if lifted() {
484                    "pointer-events-none z-30 rounded-md bg-cream shadow-xl ring-2 ring-caramel dark:bg-night dark:ring-maple"
485                } else {
486                    ""
487                },
488            )
489            style:transform=move || {
490                let (dx, dy) = drag.offset.get();
491                if lifted() { format!("translate({dx}px, {dy}px) scale(1.02)") } else { String::new() }
492            }
493            data-dragging=move || lifted().then_some("true")
494            data-session-row=session_id.to_string()
495            on:pointerenter=move |event: ev::PointerEvent| {
496                if event.pointer_type() != "touch" {
497                    hovered.set(Some(session_id));
498                }
499            }
500            on:pointerleave=move |_| {
501                if hovered.get_untracked() == Some(session_id) {
502                    hovered.set(None);
503                }
504            }
505        >
506            <Show
507                when=move || !is_renaming()
508                fallback=move || rename_row(workspace, session_id, status_mark, actions, field_badges.clone())
509            >
510            <button
511                type="button"
512                // The row looks hovered while the pointer is on its buttons
513                // too, which sit over it, or the keyboard is in it.
514                // One class closure for all of it: `class:` toggles beside
515                // it would be dropped each time it changes.
516                class=move || format!(
517                    "{ROW} no-callout select-none {} {} group-hover:bg-toasted group-has-focus-visible:bg-toasted dark:group-hover:bg-ember dark:group-has-focus-visible:bg-ember",
518                    row_padding(shown(), buttons()),
519                    if current() { "bg-toasted dark:bg-ember" } else { "" },
520                )
521                aria-current=move || current().then_some("true")
522                data-session=session_id.to_string()
523                title=title.clone()
524                on:pointerdown=on_pointerdown
525                on:contextmenu=move |event: ev::MouseEvent| {
526                    // A touch screen's long press lifts the row and offers
527                    // its actions, not the browser's menu.
528                    if lifted() || revealed.get_untracked() == Some(session_id) {
529                        event.prevent_default();
530                    }
531                }
532                // The open session's row switches its subagent tree between
533                // the running ones and all of them.
534                aria-expanded=move || tree_root().then(|| workspace.tree_expanded.get().to_string())
535                on:click=move |_| {
536                    if !drag.take_click() {
537                        if current() && tree_root() && !workspace.new_session.get_untracked() {
538                            workspace.toggle_tree();
539                        } else {
540                            workspace.open_session(session_id);

741/// A row whose title is a field: Enter renames, Escape (or leaving it)
742/// keeps the title, and an emptied field offers the automatic title.
743fn rename_row<B: IntoView + 'static>(
744    workspace: Workspace,
745    session_id: SessionId,
746    mark: Option<icons::StatusMark>,
747    actions: RowActions,
748    badges: impl Fn(NodeRef<html::Span>) -> B + Send + Sync + Clone + 'static,
749) -> impl IntoView {
750    let line = NodeRef::<html::Span>::new();
751    crate::badges::keep_fitted(line);
752    let RowActions {
753        renaming,
754        draft,
755        fresh,
756        ..
757    } = actions;
758    let field = NodeRef::<html::Input>::new();
759    // The field takes the focus as it shows; its text is selected the
760    // first time, so typing replaces the title.
761    Effect::new(move |_| {
762        if let Some(input) = field.get() {
763            request_animation_frame(move || {
764                let _ = input.focus();
765                if fresh.get_value() {
766                    fresh.set_value(false);
767                    input.select();
768                }
769            });
770        }
771    });
772    let cancel = move || {
773        if renaming.get_untracked() == Some(session_id) {
774            renaming.set(None);
775        }
776    };
777    let save = move || {
778        if workspace.submit_title(session_id, &draft.get_untracked()) {
779            renaming.set(None);
780        }
781    };
782    let empty = move || draft.with(|draft| draft.trim().is_empty());
783    view! {
784        <div
785            class="flex min-h-11 w-full flex-col items-start gap-1 rounded-md bg-toasted px-3 py-2 dark:bg-ember"
786            data-renaming=session_id.to_string()
787            on:focusout=move |event: ev::FocusEvent| {
788                // Leaving the row keeps the title; moving within it does not.
789                let inside = event
790                    .related_target()
791                    .and_then(|target| target.dyn_into::<web_sys::Node>().ok())
792                    .zip(event.current_target().and_then(|target| target.dyn_into::<web_sys::Node>().ok()))
793                    .is_some_and(|(target, row)| row.contains(Some(&target)));
794                if !inside {
795                    cancel();
796                }
797            }
798        >
799            <span class="flex w-full items-center gap-2">
800                {mark.map(|mark| view! { <Glyph icon=mark.icon class=mark.class("text-xs") /> })}
801                <input
802                    type="text"
803                    node_ref=field
804                    class=TITLE_FIELD
805                    aria-label="Session title"
806                    placeholder="The automatic title"
807                    autocomplete="off"
808                    data-rename-field=session_id.to_string()
809                    prop:value=move || draft.get()
810                    on:input=move |event| draft.set(event_target_value(&event))
811                    on:keydown=move |event: ev::KeyboardEvent| match event.key().as_str() {
812                        "Enter" => {
813                            event.prevent_default();
814                            save();
815                        }
816                        "Escape" => {
817                            event.prevent_default();
818                            cancel();
819                        }
820                        _ => {}
821                    }
822                />
823            </span>
824            <Show
825                when=empty
826                fallback=move || view! {
827                    <span class=format!("flex w-full min-w-0 items-center gap-1 {NOTE}")>
828                        <span class="flex min-w-0 flex-1 items-center gap-1 overflow-hidden" node_ref=line>
829                            {
830                                let badges = badges.clone();
831                                move || badges(line)
832                            }
833                        </span>
834                    </span>
835                }
836            >
837                <button
838                    type="button"
839                    class=format!("{AUTO_TITLE} -ml-2")
840                    data-rename-reset=session_id.to_string()
841                    // A tap must not take the focus from the field first,
842                    // which would close it.
843                    on:pointerdown=move |event: ev::PointerEvent| event.prevent_default()
844                    on:click=move |_| {
845                        renaming.set(None);
846                        workspace.rename_session(session_id, SessionRenameChange::Reset);
847                    }
848                >
849                    <Glyph icon=icondata::FaWandMagicSparklesSolid />
850                    "Use the automatic title"
851                </button>
852            </Show>
853        </div>
854    }
855}

1468/// A touch held on a user message's bubble: after [`LONG_PRESS`] without
1469/// moving it opens the message's menu, as the ⋯ does with a mouse.
1470#[derive(Clone, Copy)]
1471struct MessagePress {
1472    workspace: Workspace,
1473    press: StoredValue<Press>,
1474    timer: StoredValue<Option<TimeoutHandle>>,
1475    /// The message the press is on, until it ends.
1476    seq: StoredValue<Option<u64>>,
1477}
1478
1479impl MessagePress {
1480    fn down(self, event: &ev::PointerEvent) {
1481        self.cancel();
1482        self.seq.set_value(None);
1483        if event.pointer_type() != "touch" {
1484            return;
1485        }
1486        let Some(seq) = event
1487            .target()
1488            .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
1489            .and_then(|target| target.closest("[data-bubble]").ok().flatten())
1490            .and_then(|bubble| bubble.closest(".user-row").ok().flatten())
1491            .and_then(|row| row.get_attribute("data-seq"))
1492            .and_then(|seq| seq.parse().ok())
1493        else {
1494            return;
1495        };
1496        let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
1497        self.press.update_value(|press| press.start(x, y));
1498        self.seq.set_value(Some(seq));
1499        let timer = set_timeout_with_handle(
1500            move || {
1501                if self.press.try_update_value(Press::elapse).unwrap_or(false) {
1502                    self.workspace.open_user_menu(seq);
1503                }
1504            },
1505            LONG_PRESS,
1506        )
1507        .ok();
1508        self.timer.set_value(timer);
1509    }
1510
1511    fn moved(self, event: &ev::PointerEvent) {
1512        let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
1513        if self
1514            .press
1515            .try_update_value(|press| press.moved(x, y))
1516            .unwrap_or(false)
1517        {
1518            self.stop_timer();
1519        }
1520    }
1521
1522    /// The finger lifted or the browser took the gesture over.
1523    fn cancel(self) {
1524        self.press.update_value(Press::cancel);
1525        self.stop_timer();
1526    }
1527
1528    /// Whether a press is on a bubble now, held or just fired.
1529    fn held(self) -> bool {
1530        self.seq.get_value().is_some()
1531    }
1532
1533    fn stop_timer(self) {
1534        if let Some(timer) = self.timer.get_value() {
1535            timer.clear();
1536        }
1537        self.timer.set_value(None);
1538    }
1539}
