152/// A line of text with its code spans drawn as code instead of showing
153/// their backticks (`format::code_spans`).
154fn with_code(line: &str) -> impl IntoView + use<> {
155    use cookie_agent_client_state::format::{Span, code_spans};
156    code_spans(line)
157        .into_iter()
158        .map(|span| match span {
159            Span::Text(text) => text.to_owned().into_any(),
160            Span::Code(code) => {
161                view! { <code class=INLINE_CODE>{code.to_owned()}</code> }.into_any()
162            }
163        })
164        .collect_view()
165}

208/// The connected page.
209#[component]
210pub fn WorkspaceView(
211    workspace: Workspace,
212    connection: Connection,
213    theme: RwSignal<ThemePreference>,
214) -> impl IntoView {
215    // The view mounts again only when the page comes back from the connect
216    // screen (another daemon, say), so its listeners go with it.
217    let phone = crate::browser::on_media_change(crate::browser::PHONE_QUERY, move |narrow| {
218        workspace.narrow.set(narrow);
219    });
220    // A window that grows to a desktop's pins the sidebar instead.
221    let desktop = crate::browser::on_media_change(crate::browser::DESKTOP_QUERY, move |desktop| {
222        workspace.desktop.set(desktop);
223        if desktop {
224            workspace.sidebar_open.set(false);
225        }
226    });
227    provide_context(workspace);
228    let history = workspace.follow_history();
229    let listeners = StoredValue::new_local(vec![phone, desktop, history]);
230    on_cleanup(move || {
231        for listener in listeners
232            .try_update_value(std::mem::take)
233            .into_iter()
234            .flatten()
235            .flatten()
236        {
237            listener.remove();
238        }
239    });
240    let chat = move || workspace.view.get() == View::Chat;
241    // Below desktop widths the sidebar is the page, over the view, which
242    // keeps its place under it.
243    let paged = move || workspace.sidebar_open.get();
244    view! {
245        <div class="relative flex min-h-0 flex-1">
246            <Show when=move || !workspace.is_guest()>
247                <Sidebar workspace=workspace connection=connection theme=theme />
248            </Show>
249            <div
250                class="flex min-w-0 flex-1"
251                class:hidden=move || !chat()
252                class:invisible=paged
253                inert=move || paged().then_some("")
254            >
255                <ChatView workspace=workspace />
256                <asset_panel::AssetPanel workspace=workspace />
257            </div>
258            <Show when=move || !chat()>
259                <div class="flex min-w-0 flex-1" class:invisible=paged inert=move || paged().then_some("")>
260                    <PageView workspace=workspace connection=connection />
261                </div>
262            </Show>
263            <Sheet workspace=workspace />
264            <onboarding::OnboardingSheet workspace=workspace />
265        </div>
266    }
267}

350#[component]
351fn SessionList(workspace: Workspace) -> impl IntoView {
352    // Whether the search holds something, and the titles it looks in.
353    let searching = Memo::new(move |_| search_query(workspace).is_some());
354    let labels = session_rows::search_labels(workspace, searching);
355    // The root rows a list without projects shows, in order.
356    let shown = Memo::new(move |_| {
357        let filter = workspace.directory_filter.get();
358        let query = search_query(workspace);
359        labels.with(|labels| {
360            workspace.sessions.with(|sessions| {
361                let mut shown = sessions
362                    .iter()
363                    .filter(|session| {
364                        matches!(session.origin, cookie_agent_protocol::SessionOrigin::Root)
365                            && filter
366                                .as_deref()
367                                .is_none_or(|filter| crate::places::matches(session, filter))
368                            && workspace.filter_passes(session)
369                            && session_rows::search_matches(
370                                labels,
371                                session.session_id,
372                                query.as_deref(),
373                            )
374                    })
375                    .collect::<Vec<_>>();
376                // Running and pinned sessions first, then the rest; newest
377                // first in each.
378                cookie_agent_client_state::rows::order_sessions(&mut shown);
379                shown
380                    .into_iter()
381                    .map(|session| session.session_id)
382                    .collect::<Vec<_>>()
383            })
384        })
385    });
386    let nothing_shown = Memo::new(move |_| shown.with(Vec::is_empty));
387    // The rows' "5m ago" moves on while the list shows.
388    let now = RwSignal::new(jiff::Timestamp::now());
389    let ticker = set_interval_with_handle(
390        move || now.set(jiff::Timestamp::now()),
391        Duration::from_secs(30),
392    )
393    .ok();
394    on_cleanup(move || {
395        if let Some(ticker) = ticker {
396            ticker.clear();
397        }
398    });
399    // Renaming and deleting: the row a long press revealed the actions
400    // of, the row whose delete waits for confirmation, and the row whose
401    // title is a field.
402    let actions = RowActions {
403        hovered: RwSignal::new(None),
404        revealed: RwSignal::new(None),
405        confirming: RwSignal::new(None),
406        renaming: RwSignal::new(None),
407        draft: RwSignal::new(String::new()),
408        fresh: StoredValue::new(false),
409        menu: RwSignal::new(None),
410        context: RwSignal::new(None),
411    };
412    // Dragging a session onto a project's header, and the headers' own
413    // buttons.
414    let press = session_rows::RowPress::new();
415    let headers = projects::HeaderActions::new();
416    let list = NodeRef::<html::Nav>::new();
417    let rows = session_rows::SessionRows {
418        workspace,
419        slots: session_rows::SessionSlots::new(workspace),
420        now,
421        actions,
422        press,
423    };
424    // The project sections, worked out once for the list and Select all.
425    let sections = projects::listed_sections(workspace, searching, labels);
426    // The root rows the list shows, in page order, for Select all.
427    let listed = Memo::new(move |_| {
428        if workspace.has_projects.get() {
429            projects::listed_rows(workspace, sections, searching)
430        } else {
431            shown.get()
432        }
433    });
434    // A Shift+click ranges over these rows.
435    workspace.picking.shown.set_value(Some(listed));
436    on_cleanup(move || workspace.picking.shown.set_value(None));
437    // The filter left every section empty: the list says so above them,
438    // with the way to change it. A memo of its own, so the sections, which
439    // change as a session opens, never draw the list anew: that would lose
440    // the list's scroll, which the browser keeps on a row it shows.
441    let offer = Memo::new(move |_| {
442        !searching.get()
443            && projects::sections_hold_none(sections)
444            && (workspace.filter_changes() > 0 || workspace.has_older_sessions.get())
445    });
446    view! {
447        // `relative`: a desktop's filter opens over the list, under its
448        // button.
449        <div class="relative flex shrink-0 flex-col gap-1 px-2 pt-3">
450            <Show when=move || !workspace.picking.active.get() fallback=move || selection_bar(workspace, listed)>
451                <SessionTools workspace=workspace />
452            </Show>
453            <SessionFilterPanel workspace=workspace />
454        </div>
455        <nav class="flex min-h-0 flex-1 flex-col gap-px overflow-x-hidden overflow-y-auto px-2 pb-3" aria-label="Sessions" node_ref=list>
456            {move || {
457                if workspace.has_projects.get() {
458                    return view! {
459                        {move || offer.get().then(|| filter_offer(workspace, icondata::LuListFilter, "No sessions in this time range or filter."))}
460                        {projects::project_list(rows, sections, searching, headers)}
461                    }
462                    .into_any();
463                }
464                if nothing_shown.get() {
465                    if searching.get() {
466                        return empty_state(icondata::LuSearch, "No sessions match.").into_any();
467                    }
468                    if workspace.filter_changes() > 0 {
469                        return filter_offer(workspace, icondata::LuListFilter, "No sessions match the filter.").into_any();
470                    }
471                    // Older sessions than the window reaches: a longer date
472                    // in the filter lists them.
473                    if workspace.has_older_sessions.get() {
474                        return filter_offer(workspace, icondata::LuHistory, "No sessions in this time range; older ones are a longer date away.").into_any();
475                    }
476                    return empty_state(icondata::LuFolderOpen, "No sessions yet. Start one with New session.")
477                        .into_any();
478                }
479                session_rows::keyed_rows(rows, move || shown.get()).into_any()
480            }}
481        </nav>
482    }
483}

485/// The second row of the session list's header: Select, Search, and
486/// Filter, with what the filter lists beside them. Search turns the row
487/// into its search box, which grows out of the Search button as the other
488/// buttons fade, and shrinks back into it as they return: the box stays in
489/// the page, clipped to the button's place while closed. It takes the
490/// focus as it opens; Escape or its × closes it, shows every session again,
491/// and gives the focus back to Search.
492#[component]
493fn SessionTools(workspace: Workspace) -> impl IntoView {
494    let row = NodeRef::<html::Div>::new();
495    let search_button = NodeRef::<html::Button>::new();
496    let field = NodeRef::<html::Input>::new();
497    let searching = move || workspace.session_search.with(Option::is_some);
498    // Where Search sits in the row, measured as the box opens and closes:
499    // the box grows from and shrinks to it.
500    let origin = RwSignal::new(String::new());
501    let measure = move || {
502        if let (Some(row), Some(button)) = (row.get_untracked(), search_button.get_untracked()) {
503            let left = button.offset_left();
504            let right = (row.client_width() - left - button.offset_width()).max(0);
505            origin.set(format!(
506                "--search-left: {left}px; --search-right: {right}px"
507            ));
508        }
509    };
510    let open = move |_| {
511        measure();
512        workspace.filter_open.set(false);
513        workspace.session_search.set(Some(String::new()));
514        // Focused at once, in the tap: a phone opens its keyboard only for
515        // a focus the tap itself makes. The page shows the box open only
516        // after the tap, so it is opened here first, as it will be.
517        if let (Some(row), Some(input)) = (row.get_untracked(), field.get_untracked()) {
518            let _ = row.set_attribute("data-searching", "true");
519            if let Some(search) = row.query_selector(".session-search").ok().flatten() {
520                let _ = search.remove_attribute("inert");
521            }
522            let _ = input.focus();
523        }
524    };
525    let close = move || {
526        measure();
527        workspace.session_search.set(None);
528        // Given back at once too, the button out of its inert first.
529        if let Some(button) = search_button.get_untracked() {
530            let _ = button.remove_attribute("inert");
531            let _ = button.focus();

3553/// Brings the markdown blocks under `element` from `drawn` to `blocks`:
3554/// the blocks both start with stay, the rest go, and the new ones are added
3555/// after them.
3556fn draw_blocks(element: &web_sys::Element, drawn: &mut Vec<(Arc<str>, u32)>, blocks: &[Arc<str>]) {
3557    let kept = drawn
3558        .iter()
3559        .zip(blocks)
3560        .take_while(|((old, _), new)| old == *new)
3561        .count();
3562    let stale: u32 = drawn[kept..].iter().map(|(_, nodes)| nodes).sum();
3563    for _ in 0..stale {
3564        if let Some(last) = element.last_child() {
3565            let _ = element.remove_child(&last);
3566        }
3567    }
3568    drawn.truncate(kept);
3569    for block in &blocks[kept..] {
3570        let before = element.child_nodes().length();
3571        let _ = element.insert_adjacent_html("beforeend", block);
3572        let after = element.child_nodes().length();
3573        // A code block is highlighted alone, once it is added and near the view.
3574        if fills_in(block) {
3575            let nodes = element.child_nodes();
3576            for index in before..after {
3577                if let Some(node) = nodes
3578                    .item(index)
3579                    .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
3580                    && !crate::near_view::watch(&node)
3581                {
3582                    crate::highlight::highlight_within(&node);
3583                }
3584            }
3585        }
3586        drawn.push((Arc::clone(block), after - before));
3587    }
3588}

4309    });

4743/// What `build` builds, built again only when what `key` reads comes out
4744/// different, not on whatever else `build` reads: a sheet over the session
4745/// store would otherwise be built again on each of its revisions, every
4746/// event of a working session, its rows starting over un-hovered each time,
4747/// so that their hover blinks.
4748pub(super) fn built_on<K, V>(
4749    key: impl Fn() -> K + Send + Sync + 'static,
4750    build: impl Fn() -> V + Send + Sync + 'static,
4751) -> impl IntoView
4752where
4753    K: PartialEq + Send + Sync + 'static,
4754    V: IntoView + 'static,
4755{
4756    let key = Memo::new(move |_| key());
4757    move || {
4758        // Read, not tracked: `track` subscribes without running the memo,
4759        // which then never reads, nor follows, what it is made of.
4760        key.with(|_| ());
4761        untrack(&build)
4762    }
4763}

5763/// The session's ⋮ menu: every command, by name, icon, and what it does,
5764/// in rows a finger can hit; each opens its sheet. The session's ACP
5765/// agent's commands follow, each badged as the agent's; one of them stands
5766/// in for cookie's command of its name. A filter on top narrows both.
5767fn commands_sheet(workspace: Workspace) -> impl IntoView {
5768    let has_crons = workspace.has_crons.get_untracked();
5769    let has_webhooks = workspace.has_webhooks.get_untracked();
5770    let filter = RwSignal::new(String::new());
5771    let query = move || filter.get().trim().trim_start_matches('/').to_lowercase();
5772    let matches = |query: &str, texts: [&str; 2]| {
5773        query.is_empty() || texts.iter().any(|text| text.to_lowercase().contains(query))
5774    };
5775    // The agent's commands and which of cookie's they stand in for, as
5776    // values: read from the store, which changes with every event of a
5777    // working session, they would build every row again with each, and
5778    // the hover of the one under the pointer would blink.
5779    let replaced = Memo::new(move |_| {
5780        MENU.iter()
5781            .filter(|entry| workspace.agent_replaces(entry.id))
5782            .map(|entry| entry.id)
5783            .collect::<Vec<_>>()
5784    });
5785    let commands = Memo::new(move |_| workspace.agent_commands());
5786    // What is live behind a panel entry, at its end: the cost, the goal's
5787    // progress, the calls in the background.
5788    let live = move |command: Command| -> Option<String> {
5789        match command {
5790            Command::Usage => workspace.selected.get().and_then(|session_id| {
5791                workspace
5792                    .costs
5793                    .with(|costs| costs.get(&session_id).copied())
5794                    .map(cost_label)
5795            }),
5796            Command::Goal => workspace
5797                .goal()
5798                .filter(|goal| cookie_agent_client_state::rows::goal_bar_shows(goal.status))
5799                .filter(|goal| !goal.items.is_empty())
5800                .map(|goal| {
5801                    format!(
5802                        "{}/{}",
5803                        goal.items.iter().filter(|item| item.finished).count(),
5804                        goal.items.len()
5805                    )
5806                }),
5807            Command::Background => Some(workspace.background_calls().len())
5808                .filter(|count| *count > 0)
5809                .map(|count| count.to_string()),
5810            _ => None,
5811        }
5812    };
5813    let row = move |entry: &'static crate::commands::MenuEntry| {
5814        let command = entry.command;
5815        view! {
5816            <button
5817                type="button"
5818                class=MENU_ROW
5819                title=entry.description
5820                data-command=entry.id
5821                on:click=move |_| {
5822                    workspace.panel.set(None);
5823                    workspace.run_command(command);
5824                }
5825            >
5826                <Glyph icon=entry.icon class="w-4 shrink-0 justify-center text-muted" />
5827                <span class="min-w-0 flex-1 truncate">{entry.label}</span>
5828                {move || live(command).map(|value| view! { <span class="shrink-0 text-xs text-muted tabular-nums">{value}</span> })}
5829            </button>
5830        }
5831    };
5832    // The session's actions, then what opens in the session panel
5833    // (docs/site/specs/web-polish.md, P4).
5834    let own = move || {
5835        let query = query();
5836        let replaced = replaced.get();
5837        let offered: Vec<&'static crate::commands::MenuEntry> = MENU
5838            .iter()
5839            .filter(|entry| crate::commands::offered(entry.command, has_crons, has_webhooks))
5840            .filter(|entry| {
5841                !workspace.is_guest() || crate::commands::offered_to_guest(entry.command)
5842            })
5843            .filter(|entry| !replaced.contains(&entry.id))
5844            .filter(|entry| matches(&query, [entry.label, entry.description]))
5845            .collect();
5846        let in_panel = |entry: &&&crate::commands::MenuEntry| {
5847            SESSION_TABS
5848                .iter()
5849                .any(|(_, command, _, _)| *command == entry.command)
5850        };
5851        let actions = offered
5852            .iter()
5853            .filter(|entry| !in_panel(entry))
5854            .map(|entry| row(entry))
5855            .collect_view();
5856        let panel: Vec<_> = offered.iter().filter(|entry| in_panel(entry)).collect();
5857        view! {
5858            <div class="flex flex-col">{actions}</div>
5859            {(!panel.is_empty()).then(|| view! {
5860                <h3 class="mt-1 border-t border-line px-2.5 pt-2 pb-0.5 text-xs font-semibold text-muted">"Open in the session panel"</h3>
5861                <div class="flex flex-col">{panel.into_iter().map(|entry| row(entry)).collect_view()}</div>
5862            })}
5863        }
5864    };
5865    let agent = move || {
5866        let query = query();
5867        let rows = commands.with(|commands| {
5868            commands
5869                .iter()
5870                .filter(|command| {
5871                    // A guest's agent commands are the ones that stand in for
5872                    // compact and the goal.
5873                    (!workspace.is_guest() || matches!(command.name.as_str(), "compact" | "goal"))
5874                        && cookie_agent_client_state::acp::command_listed(
5875                            &command.name,
5876                            query.trim_start_matches('/'),
5877                        )
5878                        && matches(
5879                            &query,
5880                            [command.name.as_str(), command.description.as_str()],
5881                        )
5882                })
5883                .cloned()
5884                .collect::<Vec<_>>()
5885        });
5886        (!rows.is_empty()).then(|| {
5887            let rows = rows
5888                .into_iter()
5889                .map(|command| {
5890                    let name = command.name.clone();
5891                    let hint = command.input_hint.clone().unwrap_or_default();
5892                    let description = command.description.clone();
5893                    view! {
5894                        <button
5895                            type="button"
5896                            class=format!("{ROW} min-h-12")
5897                            data-agent-command=name.clone()
5898                            on:click=move |_| workspace.choose_agent_command(command.clone())
5899                        >
5900                            <span class="flex w-full min-w-0 flex-col">
5901                                <span class="flex flex-wrap items-baseline gap-x-2">
5902                                    <span class="agent-badge rounded px-1 text-[0.625rem] uppercase bg-raised shadow-[0_0_0_1px_var(--color-accent),0_0_0_4px_var(--color-accent-soft)]">"agent"</span>
5903                                    <span class="font-mono text-sm font-bold">{cookie_agent_client_state::acp::command_label(&name)}</span>
5904                                    <span class=format!("font-mono {NOTE}")>{hint}</span>
5905                                </span>
5906                                <span class=NOTE>{description}</span>
5907                            </span>
5908                        </button>
5909                    }
5910                })
5911                .collect_view();
5912            view! {
5913                <h3 class="mt-1 border-t border-line px-2.5 pt-2 pb-0.5 text-xs font-semibold text-muted">"The agent's commands"</h3>
5914                <div id="agent-commands" class="flex flex-col gap-2">{rows}</div>
5915            }
5916        })
5917    };
5918    view! {
5919        <input
5920            id="command-filter"
5921            type="search"
5922            class=depth::FIELD
5923            placeholder="Filter commands"
5924            aria-label="Filter commands"
5925            prop:value=move || filter.get()
5926            on:input=move |event| filter.set(event_target_value(&event))
5927        />
5928        {own}
5929        {agent}
5930    }
5931}
