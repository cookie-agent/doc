128#[component]
129pub fn WorkspaceView(
130    workspace: Workspace,
131    connection: Connection,
132    theme: RwSignal<ThemePreference>,
133) -> impl IntoView {
134    // The view remounts with every connection, so its listeners go with it.
135    let phone = crate::browser::on_media_change(crate::browser::PHONE_QUERY, move |narrow| {
136        workspace.narrow.set(narrow);
137    });
138    // A window that grows to a desktop's pins the sidebar instead.
139    let desktop = crate::browser::on_media_change(crate::browser::DESKTOP_QUERY, move |desktop| {
140        if desktop {
141            workspace.sidebar_open.set(false);
142        }
143    });
144    let history = workspace.follow_history();
145    let listeners = StoredValue::new_local(vec![phone, desktop, history]);
146    on_cleanup(move || {
147        for listener in listeners
148            .try_update_value(std::mem::take)
149            .into_iter()
150            .flatten()
151            .flatten()
152        {
153            listener.remove();
154        }
155    });
156    let chat = move || workspace.view.get() == View::Chat;
157    view! {
158        <div class="flex min-h-0 flex-1">
159            <div class="hidden sm:flex">
160                <Rail workspace=workspace connection=connection theme=theme primary=true />
161            </div>
162            <div class="flex min-w-0 flex-1" class:hidden=move || !chat()>
163                <aside
164                    class="hidden w-72 shrink-0 flex-col border-r border-toasted bg-parchment dark:border-ember dark:bg-char"
165                    class:lg:flex=move || workspace.sidebar_pinned.get()
166                    id="sidebar"
167                >
168                    <SessionList workspace=workspace />
169                </aside>
170                <ChatView workspace=workspace />
171            </div>
172            <Show when=move || !chat()>
173                <PageView workspace=workspace connection=connection />
174            </Show>
175            <Drawer workspace=workspace connection=connection theme=theme />
176            <Sheet workspace=workspace />
177        </div>
178    }

232#[component]
233fn SessionList(workspace: Workspace) -> impl IntoView {
234    // Each runtime at each workspace sessions work in, under its home.
235    let places = move || {
236        let homes = workspace.homes();
237        workspace.sessions.with(|sessions| {
238            let roots = sessions
239                .iter()
240                .filter(|session| {
241                    matches!(session.origin, cookie_agent_protocol::SessionOrigin::Root)
242                })
243                .cloned()
244                .collect::<Vec<_>>();
245            crate::places::filter_options(&roots, &homes)
246        })
247    };
248    let shown = move || {
249        let filter = workspace.directory_filter.get();
250        let mut sessions = workspace.sessions.with(|sessions| {
251            sessions
252                .iter()
253                .filter(|session| {
254                    matches!(session.origin, cookie_agent_protocol::SessionOrigin::Root)
255                        && filter
256                            .as_deref()
257                            .is_none_or(|filter| crate::places::matches(session, filter))
258                })
259                .cloned()
260                .collect::<Vec<_>>()
261        });
262        // Running sessions first, then the rest; newest first in each.
263        cookie_agent_client_state::rows::order_sessions(&mut sessions);
264        sessions
265    };
266    // The rows' "5m ago" moves on while the list shows.
267    let now = RwSignal::new(jiff::Timestamp::now());
268    let ticker = set_interval_with_handle(
269        move || now.set(jiff::Timestamp::now()),
270        Duration::from_secs(30),
271    )
272    .ok();
273    on_cleanup(move || {
274        if let Some(ticker) = ticker {
275            ticker.clear();
276        }
277    });
278    // Renaming and deleting: the row a long press revealed the actions
279    // of, the row whose delete waits for confirmation, and the row whose
280    // title is a field.
281    let actions = RowActions {
282        hovered: RwSignal::new(None),
283        revealed: RwSignal::new(None),
284        confirming: RwSignal::new(None),
285        renaming: RwSignal::new(None),
286        draft: RwSignal::new(String::new()),
287        fresh: StoredValue::new(false),
288    };
289    // Dragging a session onto a project's header, and the headers' own
290    // buttons.
291    let drag = projects::SessionDrag::new();
292    drag.follow(workspace, actions);
293    let headers = projects::HeaderActions::new();
294    let list = drag.list;
295    view! {
296        <div class="flex flex-col gap-2 p-3">
297            <div class="flex gap-2">
298                <button type="button" class=format!("{PRIMARY} flex-1") data-new-session on:click=move |_| workspace.start_new_session()>
299                    <Glyph icon=icondata::FaPlusSolid />
300                    "New session"
301                </button>
302                <Show when=move || workspace.has_projects.get()>
303                    <button
304                        type="button"
305                        class=BUTTON
306                        id="new-project"
307                        aria-label="New project…"
308                        title="New project…"
309                        on:click=move |_| workspace.open_project_editor(None, false)
310                    >
311                        <Glyph icon=icondata::FaFolderPlusSolid />
312                    </button>
313                </Show>
314            </div>
315            // A daemon without projects: the workspace filter instead of
316            // project sections.
317            <Show when=move || !workspace.has_projects.get()>
318            <label class="flex flex-col gap-1 text-xs font-bold text-ink dark:text-sand">
319                <span>"Workspace"</span>
320                <select
321                    id="place-filter"
322                    class="min-h-11 rounded-md border border-crust bg-cream px-2 text-sm font-normal text-espresso hover:border-caramel dark:border-smoke dark:bg-night dark:text-latte dark:hover:border-maple"
323                    on:change=move |event| {
324                        let value = event_target_value(&event);
325                        workspace.directory_filter.set((!value.is_empty()).then_some(value));
326                    }
327                >
328                    <option value="">"All workspaces"</option>
329                    {move || {
330                        places()
331                            .into_iter()
332                            .map(|(value, label)| {
333                                let selected = workspace.directory_filter.get().as_deref()
334                                    == Some(value.as_str());
335                                view! {
336                                    <option value=value selected=selected>
337                                        {label}
338                                    </option>
339                                }
340                            })
341                            .collect_view()
342                    }}
343                </select>
344            </label>
345            </Show>
346        </div>
347        <nav class="flex min-h-0 flex-1 flex-col gap-1 overflow-x-hidden overflow-y-auto px-2 pb-3" aria-label="Sessions" node_ref=list>
348            {move || {
349                if workspace.has_projects.get() {
350                    return projects::project_list(workspace, now, actions, drag, headers).into_any();
351                }
352                let sessions = shown();
353                if sessions.is_empty() {
354                    return empty_state(icondata::FaFolderOpenSolid, "No sessions yet. Start one with New session.")
355                        .into_any();
356                }
357                sessions
358                    .into_iter()
359                    .map(|session| session_row(workspace, session, now, actions, drag))
360                    .collect_view()
361                    .into_any()
362            }}
363        </nav>
364    }
365}

367/// The session list's row actions, shared by its rows: move, rename, and
368/// delete.
369#[derive(Clone, Copy)]
370struct RowActions {
371    /// The row a mouse (or pen) is over, which shows its buttons.
372    hovered: RwSignal<Option<SessionId>>,
373    /// The row a long press offered its actions under.
374    revealed: RwSignal<Option<SessionId>>,
375    confirming: RwSignal<Option<SessionId>>,
376    /// The row whose title is a field, and what the field holds; kept
377    /// here, so a list redrawn while the user types keeps it.
378    renaming: RwSignal<Option<SessionId>>,
379    draft: RwSignal<String>,
380    /// The field just opened: its text is selected once it shows.
381    fresh: StoredValue<bool>,
382}
383
384impl RowActions {
385    /// Turns `session`'s title into a field holding it.
386    fn begin_rename(self, workspace: Workspace, session_id: SessionId) {
387        self.revealed.set(None);
388        self.confirming.set(None);
389        self.draft.set(workspace.title_draft(session_id));
390        self.fresh.set_value(true);
391        self.renaming.set(Some(session_id));
392    }
393}

395/// A root session's row: a click opens it. On hover (or a long press on
396/// a touch screen) its buttons offer to rename it, in place, and to
397/// delete it, behind an inline confirmation. Its second line says where
398/// it runs, as badges. The open session's row has its subagents under it.
399fn session_row(
400    workspace: Workspace,
401    session: SessionMeta,
402    now: RwSignal<jiff::Timestamp>,
403    actions: RowActions,
404    drag: projects::SessionDrag,
405) -> impl IntoView {
406    let session_id = session.session_id;
407    let current = move || workspace.selected.get() == Some(session_id);
408    // The root of the open tree, whether the root itself or one of its
409    // subagents is open, and only when it has subagents: its row keeps the
410    // chevron that folds and unfolds the tree.
411    let tree_root =
412        move || workspace.tree_root.get() == Some(session_id) && workspace.tree_has_subagents();
413    let mark = icons::session_status(session.status);
414    let activity = cookie_agent_client_state::activity::session_activity(&session);
415    let status_mark = icons::row_status(&activity);
416    // What the row's tooltip and a screen reader say of it: what it waits
417    // on, else its status.
418    let status_text = match &activity {
419        cookie_agent_client_state::activity::SessionActivity::Waiting(kinds) => {
420            cookie_agent_client_state::activity::waiting_label(kinds)
421        }
422        _ => mark.label.to_owned(),
423    };
424    let last_activity = session.last_activity;
425    // A run session of a cron job: its clock opens the job.
426    let cron = session.cron.clone();
427    // Where it runs, once the runtimes are known; the tooltip lists every
428    // workspace with its path.
429    // Fitted to the line: machines whole, the last ones left out for a
430    // `+N` when even the workspaces gone to `…` leave too little room.
431    let badges = {
432        let session = session.clone();
433        move |line: NodeRef<html::Span>| {
434            let homes = workspace.homes();
435            let listed = workspace.lists_attachments.get();
436            let shown = workspace
437                .runtimes
438                .with(|listing| session_badges(&session, listing.as_ref(), &homes, listed));
439            crate::badges::fit_soon(line);
440            crate::badges::badge_line(shown)
441        }
442    };
443    let badge_line = NodeRef::<html::Span>::new();
444    crate::badges::keep_fitted(badge_line);
445    let title = {
446        let session = session.clone();
447        let status = status_text.clone();
448        move || {
449            let homes = workspace.homes();
450            let listed = workspace.lists_attachments.get();
451            let places = workspace
452                .runtimes
453                .with(|listing| badges_tooltip(&session, listing.as_ref(), &homes, listed));
454            format!("{places}\n{status}")
455        }
456    };
457    let label = session_label(&session);
458    let confirm_label = label.clone();
459    let RowActions {
460        hovered,
461        revealed,
462        confirming,
463        renaming,
464        ..
465    } = actions;
466    let is_renaming = move || renaming.get() == Some(session_id);
467    let movable = move || workspace.has_projects.get();
468    let shown = move || hovered.get() == Some(session_id) && !is_renaming();
469    // A mouse drags the row onto a project's header at once; a finger
470    // after a long press, which let go in place reveals the buttons.
471    let on_pointerdown = move |event: ev::PointerEvent| {
472        if !is_renaming() {
473            drag.press(workspace, session_id, &event);
474        }
475    };
476    let lifted = move || drag.lifts(session_id);
477    let field_badges = badges.clone();
478    view! {
479        <div
480            class=move || format!(
481                "session-row group relative {}",
482                if lifted() {
483                    "pointer-events-none z-30 rounded-md bg-cream shadow-xl ring-2 ring-caramel dark:bg-night dark:ring-maple"
484                } else {
485                    ""
486                },
487            )
488            style:transform=move || {
489                let (dx, dy) = drag.offset.get();
490                if lifted() { format!("translate({dx}px, {dy}px) scale(1.02)") } else { String::new() }
491            }
492            data-dragging=move || lifted().then_some("true")
493            data-session-row=session_id.to_string()
494            on:pointerenter=move |event: ev::PointerEvent| {
495                if event.pointer_type() != "touch" {
496                    hovered.set(Some(session_id));
497                }
498            }
499            on:pointerleave=move |_| {
500                if hovered.get_untracked() == Some(session_id) {
501                    hovered.set(None);
502                }
503            }
504        >
505            <Show
506                when=move || !is_renaming()
507                fallback=move || rename_row(workspace, session_id, status_mark, actions, field_badges.clone())
508            >
509            <button
510                type="button"
511                // The row looks hovered while the pointer is on its buttons
512                // too, which sit over it, or the keyboard is in it.
513                // One class closure for all of it: `class:` toggles beside
514                // it would be dropped each time it changes.
515                class=move || format!(
516                    "{ROW} no-callout select-none pr-3 {} group-hover:bg-toasted group-has-focus-visible:bg-toasted dark:group-hover:bg-ember dark:group-has-focus-visible:bg-ember",
517                    if current() { "bg-toasted dark:bg-ember" } else { "" },
518                )
519                aria-current=move || current().then_some("true")
520                data-session=session_id.to_string()
521                title=title.clone()
522                on:pointerdown=on_pointerdown
523                on:contextmenu=move |event: ev::MouseEvent| {
524                    // A touch screen's long press lifts the row and offers
525                    // its actions, not the browser's menu.
526                    if lifted() || revealed.get_untracked() == Some(session_id) {
527                        event.prevent_default();
528                    }
529                }
530                // The open session's row switches its subagent tree between
531                // the running ones and all of them.
532                aria-expanded=move || tree_root().then(|| workspace.tree_expanded.get().to_string())
533                on:click=move |_| {
534                    if !drag.take_click() {
535                        if current() && tree_root() && !workspace.new_session.get_untracked() {
536                            workspace.toggle_tree();
537                        } else {
538                            workspace.open_session(session_id);
539                        }
540                    }
541                }

750/// A row whose title is a field: Enter renames, Escape (or leaving it)
751/// keeps the title, and an emptied field offers the automatic title.
752fn rename_row<B: IntoView + 'static>(
753    workspace: Workspace,
754    session_id: SessionId,
755    mark: Option<icons::StatusMark>,
756    actions: RowActions,
757    badges: impl Fn(NodeRef<html::Span>) -> B + Send + Sync + Clone + 'static,
758) -> impl IntoView {
759    let line = NodeRef::<html::Span>::new();
760    crate::badges::keep_fitted(line);
761    let RowActions {
762        renaming,
763        draft,
764        fresh,
765        ..
766    } = actions;
767    let field = NodeRef::<html::Input>::new();
768    // The field takes the focus as it shows; its text is selected the
769    // first time, so typing replaces the title.
770    Effect::new(move |_| {
771        if let Some(input) = field.get() {
772            request_animation_frame(move || {
773                let _ = input.focus();
774                if fresh.try_get_value() == Some(true) {
775                    fresh.set_value(false);
776                    input.select();
777                }
778            });
779        }
780    });
781    let cancel = move || {
782        if renaming.get_untracked() == Some(session_id) {
783            renaming.set(None);
784        }
785    };
786    let save = move || {
787        if workspace.submit_title(session_id, &draft.get_untracked()) {
788            renaming.set(None);
789        }
790    };
791    let empty = move || draft.with(|draft| draft.trim().is_empty());
792    view! {
793        <div
794            class="flex min-h-11 w-full flex-col items-start gap-1 rounded-md bg-toasted px-3 py-2 dark:bg-ember"
795            data-renaming=session_id.to_string()
796            on:focusout=move |event: ev::FocusEvent| {
797                // Leaving the row keeps the title; moving within it does not.
798                let inside = event
799                    .related_target()
800                    .and_then(|target| target.dyn_into::<web_sys::Node>().ok())
801                    .zip(event.current_target().and_then(|target| target.dyn_into::<web_sys::Node>().ok()))
802                    .is_some_and(|(target, row)| row.contains(Some(&target)));
803                if !inside {
804                    cancel();
805                }
806            }
807        >
808            <span class="flex w-full items-center gap-2">
809                {mark.map(|mark| view! { <Glyph icon=mark.icon class=mark.class("text-xs") /> })}
810                <input
811                    type="text"
812                    node_ref=field
813                    class=TITLE_FIELD
814                    aria-label="Session title"
815                    placeholder="The automatic title"
816                    autocomplete="off"
817                    data-rename-field=session_id.to_string()
818                    prop:value=move || draft.get()
819                    on:input=move |event| draft.set(event_target_value(&event))
820                    on:keydown=move |event: ev::KeyboardEvent| match event.key().as_str() {
821                        "Enter" => {
822                            event.prevent_default();
823                            save();
824                        }
825                        "Escape" => {
826                            event.prevent_default();
827                            cancel();
828                        }
829                        _ => {}
830                    }
831                />
832            </span>
833            <Show
834                when=empty
835                fallback=move || view! {
836                    <span class=format!("flex w-full min-w-0 items-center gap-1 {NOTE}")>
837                        <span class="flex min-w-0 flex-1 items-center gap-1 overflow-hidden" node_ref=line>
838                            {
839                                let badges = badges.clone();
840                                move || badges(line)
841                            }
842                        </span>
843                    </span>
844                }
845            >
846                <button
847                    type="button"
848                    class=format!("{AUTO_TITLE} -ml-2")
849                    data-rename-reset=session_id.to_string()
850                    // A tap must not take the focus from the field first,
851                    // which would close it.
852                    on:pointerdown=move |event: ev::PointerEvent| event.prevent_default()
853                    on:click=move |_| {
854                        renaming.set(None);
855                        workspace.rename_session(session_id, SessionRenameChange::Reset);
856                    }
857                >
858                    <Glyph icon=icondata::FaWandMagicSparklesSolid />
859                    "Use the automatic title"
860                </button>
861            </Show>
862        </div>
863    }
864}

1494/// A touch held on a user message's bubble: after [`LONG_PRESS`] without
1495/// moving it opens the message's menu, as the ⋯ does with a mouse.
1496#[derive(Clone, Copy)]
1497struct MessagePress {
1498    workspace: Workspace,
1499    press: StoredValue<Press>,
1500    timer: StoredValue<Option<TimeoutHandle>>,
1501    /// The message the press is on, until it ends.
1502    seq: StoredValue<Option<u64>>,
1503}
1504
1505impl MessagePress {
1506    fn down(self, event: &ev::PointerEvent) {
1507        self.cancel();
1508        self.press.set_value(Press::default());
1509        self.seq.set_value(None);
1510        if event.pointer_type() != "touch" {
1511            return;
1512        }
1513        let Some(seq) = event
1514            .target()
1515            .and_then(|target| target.dyn_into::<web_sys::Element>().ok())
1516            .and_then(|target| target.closest("[data-bubble]").ok().flatten())
1517            .and_then(|bubble| bubble.closest(".user-row").ok().flatten())
1518            .and_then(|row| row.get_attribute("data-seq"))
1519            .and_then(|seq| seq.parse().ok())
1520        else {
1521            return;
1522        };
1523        let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
1524        self.press.update_value(|press| press.start(x, y));
1525        self.seq.set_value(Some(seq));
1526        let timer = set_timeout_with_handle(
1527            move || {
1528                if self.press.try_update_value(Press::elapse).unwrap_or(false) {
1529                    self.workspace.open_user_menu(seq);
1530                }
1531            },
1532            LONG_PRESS,
1533        )
1534        .ok();
1535        self.timer.set_value(timer);
1536    }
1537
1538    fn moved(self, event: &ev::PointerEvent) {
1539        let (x, y) = (f64::from(event.client_x()), f64::from(event.client_y()));
1540        if self
1541            .press
1542            .try_update_value(|press| press.moved(x, y))
1543            .unwrap_or(false)
1544        {
1545            self.stop_timer();
1546        }
1547    }
1548
1549    /// The finger lifted or the browser took the gesture over.
1550    fn cancel(self) {
1551        self.press.update_value(Press::cancel);
1552        self.stop_timer();
1553    }
1554
1555    /// Whether the last press opened the menu, until the next one starts.
1556    fn fired(self) -> bool {
1557        self.press.with_value(Press::fired)
1558    }
1559
1560    /// Whether a press is on a bubble now, held or just fired.
1561    fn held(self) -> bool {
1562        self.seq.get_value().is_some()
1563    }
1564
1565    fn stop_timer(self) {
1566        if let Some(timer) = self.timer.get_value() {
1567            timer.clear();
1568        }
1569        self.timer.set_value(None);
1570    }
1571}
