1//! The page's icons: Font Awesome Free 6, whose artwork is CC BY 4.0, from
2//! the `icondata` crate. Components show them through `leptos_icons`; the
3//! transcript, which renders to HTML text, inlines the same SVG. Each icon
4//! is a static the linker keeps only when the page names it.
5//!
6//! Icons inherit the text colour (`currentColor`) and are hidden from
7//! assistive technology: the button or text beside them carries the name.
8

16/// `icon` as an inline element, `class` sizing or colouring it.
17#[component]
18pub fn Glyph(
19    #[prop(into)] icon: Signal<Icon>,
20    #[prop(into, optional)] class: String,
21) -> impl IntoView {
22    view! {
23        <span class=format!("inline-flex shrink-0 items-center {class}") aria-hidden="true">
24            <leptos_icons::Icon icon=icon />
25        </span>
26    }
27}

29/// `icon` as SVG markup for HTML the page renders as text, `class` on the
30/// `<svg>`: a `<use>` of the icon's symbol in the page's sprite, which
31/// holds each icon's paths once however many transcript rows show it.
32pub fn svg(icon: Icon, class: &str) -> String {
33    format!(
34        "<svg class=\"inline-block shrink-0 {class}\" aria-hidden=\"true\" width=\"1em\" height=\"1em\" fill=\"currentColor\"><use href=\"#{}\"></use></svg>",
35        symbol(icon)
36    )
37}

39thread_local! {
40    /// The icons whose symbol is in the sprite.
41    static SYMBOLS: std::cell::RefCell<std::collections::HashSet<&'static str>> =
42        std::cell::RefCell::default();
43}
44
45/// The ID of `icon`'s symbol in the page's sprite, which gets the symbol
46/// the first time the icon is asked for.
47fn symbol(icon: Icon) -> String {
48    use std::hash::BuildHasher as _;
49    let id = format!(
50        "icon-{:x}",
51        foldhash::fast::FixedState::default().hash_one(icon.data)
52    );
53    if SYMBOLS.with(|symbols| symbols.borrow_mut().insert(icon.data)) {
54        add_symbol(&id, icon);
55    }
56    id
57}

59/// Adds `icon`'s symbol to the sprite, a hidden `<svg>` at the end of the
60/// page made with the first.
61#[cfg(target_arch = "wasm32")]
62fn add_symbol(id: &str, icon: Icon) {
63    const SPRITE: &str = "icon-sprite";
64    let Some(document) = web_sys::window().and_then(|window| window.document()) else {
65        return;
66    };
67    let sprite = match document.get_element_by_id(SPRITE) {
68        Some(sprite) => sprite,
69        None => {
70            let Some(body) = document.body() else {
71                return;
72            };
73            let _ = body.insert_adjacent_html(
74                "beforeend",
75                &format!(
76                    "<svg id=\"{SPRITE}\" aria-hidden=\"true\" style=\"position:absolute;width:0;height:0;overflow:hidden\"></svg>"
77                ),
78            );
79            let Some(sprite) = document.get_element_by_id(SPRITE) else {
80                return;
81            };
82            sprite
83        }
84    };
85    let _ = sprite.insert_adjacent_html(
86        "beforeend",
87        &format!(
88            "<symbol id=\"{id}\" viewBox=\"{}\">{}</symbol>",
89            icon.view_box.unwrap_or("0 0 512 512"),
90            icon.data
91        ),
92    );
93}

98/// The icon of a tool call's presentation hint, by the hint and never by
99/// the tool's name; a hint this build does not know is generic.
100pub fn hint_icon(hint: Option<&str>) -> Icon {
101    match hint.unwrap_or(ToolHint::GENERIC) {
102        ToolHint::SHELL => icondata::FaTerminalSolid,
103        ToolHint::FILE_READ => icondata::FaFileLinesSolid,
104        ToolHint::FILE_WRITE | ToolHint::FILE_EDIT => icondata::FaPenToSquareSolid,
105        ToolHint::FETCH => icondata::FaGlobeSolid,
106        ToolHint::SEARCH => icondata::FaMagnifyingGlassSolid,
107        ToolHint::DELEGATE => icondata::FaRobotSolid,
108        ToolHint::MESSAGE => icondata::FaEnvelopeSolid,
109        ToolHint::SKILL => icondata::FaWandMagicSparklesSolid,
110        ToolHint::GOAL => icondata::FaBullseyeSolid,
111        _ => icondata::FaWrenchSolid,
112    }
113}

115/// How a status shows: its icon, its name, the colour classes, and whether
116/// the icon turns while the thing runs.
117#[derive(Clone, Copy, Debug, PartialEq)]
118pub struct StatusMark {
119    pub icon: Icon,
120    pub label: &'static str,
121    pub tone: &'static str,
122    pub spins: bool,
123}
124
125impl StatusMark {
126    /// The icon's classes: its colour, turning while it runs, and `extra`.
127    pub fn class(self, extra: &str) -> String {
128        let spin = if self.spins { "animate-spin" } else { "" };
129        format!("{} {spin} {extra}", self.tone)
130    }
131}

133const RUNNING: &str = "text-caramel dark:text-maple";
134const DONE: &str = "text-green-700 dark:text-green-400";
135const FAILED: &str = "text-red-700 dark:text-red-400";
136const QUIET: &str = "text-ink dark:text-sand";
137
138/// A tool call's status.
139pub fn tool_status(status: &ToolStatus, backgrounded: bool) -> StatusMark {
140    let (icon, label, tone, spins) = match status {
141        ToolStatus::Running if backgrounded => (
142            icondata::FaHourglassHalfSolid,
143            "in the background",
144            QUIET,
145            false,
146        ),
147        ToolStatus::Running => (icondata::FaCircleNotchSolid, "running", RUNNING, true),
148        ToolStatus::Completed => (icondata::FaCircleCheckSolid, "done", DONE, false),
149        ToolStatus::Failed => (icondata::FaCircleXmarkSolid, "failed", FAILED, false),
150        ToolStatus::Cancelled => (icondata::FaBanSolid, "cancelled", QUIET, false),
151        ToolStatus::Interrupted => (icondata::FaCirclePauseSolid, "interrupted", QUIET, false),
152    };
153    StatusMark {
154        icon,
155        label,
156        tone,
157        spins,
158    }
159}

161/// A session's status in the session list.
162pub fn session_status(status: SessionStatus) -> StatusMark {
163    let (icon, label, tone, spins) = match status {
164        SessionStatus::Running => (icondata::FaCircleNotchSolid, "running", RUNNING, true),
165        SessionStatus::Idle => (icondata::FaCircleRegular, "idle", QUIET, false),
166        SessionStatus::Completed => (icondata::FaCircleCheckSolid, "completed", DONE, false),
167        SessionStatus::Failed => (icondata::FaCircleXmarkSolid, "failed", FAILED, false),
168        SessionStatus::Cancelled => (icondata::FaBanSolid, "cancelled", QUIET, false),
169        SessionStatus::Interrupted => (icondata::FaCirclePauseSolid, "interrupted", QUIET, false),
170        SessionStatus::Unknown => (icondata::FaCircleQuestionRegular, "unknown", QUIET, false),
171    };
172    StatusMark {
173        icon,
174        label,
175        tone,
176        spins,
177    }
178}

180/// The mark a session's row shows before its title, if any: a turning
181/// circle while it runs, a clock while it waits on a producer (an active
182/// goal, a background call, a subagent, or an external producer) to start
183/// its next run, and a mark when its last run failed or was interrupted. A
184/// finished, idle, or cancelled session has none, so its title starts at
185/// the row's edge. Running outranks waiting, which outranks a problem.
186pub fn row_status(activity: &SessionActivity) -> Option<StatusMark> {
187    match activity {
188        SessionActivity::Running => Some(session_status(SessionStatus::Running)),
189        SessionActivity::Waiting(_) => Some(StatusMark {
190            icon: icondata::FaClockRegular,
191            label: "waiting",
192            tone: RUNNING,
193            spins: false,
194        }),
195        SessionActivity::Problem(status @ (SessionStatus::Failed | SessionStatus::Interrupted)) => {
196            Some(session_status(*status))
197        }
198        SessionActivity::Problem(_) | SessionActivity::None => None,
199    }
200}
201
202/// [`row_status`] of a row that knows only its status, such as a
203/// subagent's in the tree: subagents report no producer they wait on.
204pub fn row_status_of(status: SessionStatus) -> Option<StatusMark> {
205    row_status(&match status {
206        SessionStatus::Running => SessionActivity::Running,
207        SessionStatus::Failed | SessionStatus::Cancelled | SessionStatus::Interrupted => {
208            SessionActivity::Problem(status)
209        }
210        SessionStatus::Idle | SessionStatus::Completed | SessionStatus::Unknown => {
211            SessionActivity::None
212        }
213    })
214}

226/// The permission mode's icon, a shield of its own for each mode (a
227/// phone's bottom bar shows only the icon), and what the mode does.
228pub fn permission_mode(mode: PermissionMode) -> (Icon, &'static str) {
229    match mode {
230        PermissionMode::AutoApprove => (
231            icondata::FaShieldHalvedSolid,
232            "the classifier approves safe calls and asks you about the rest",
233        ),
234        PermissionMode::AutoApproveN => (
235            icondata::FaShieldSolid,
236            "the classifier approves safe calls and rejects the rest without asking",
237        ),
238        PermissionMode::AutoApproveY => (
239            icondata::FaShieldHeartSolid,
240            "the classifier approves safe calls and approves the rest without asking",
241        ),
242        PermissionMode::Ask => (
243            icondata::FaUserShieldSolid,
244            "every call that needs approval asks you",
245        ),
246        PermissionMode::Yolo => (
247            icondata::FaShieldVirusSolid,
248            "every call that would ask is approved",
249        ),
250    }
251}

262    #[test]
263    fn every_permission_mode_has_its_own_icon() {
264        use cookie_agent_protocol::PermissionMode;
265        let mut icons = [
266            PermissionMode::AutoApprove,
267            PermissionMode::AutoApproveN,
268            PermissionMode::AutoApproveY,
269            PermissionMode::Ask,
270            PermissionMode::Yolo,
271        ]
272        .map(|mode| permission_mode(mode).0.data)
273        .to_vec();
274        icons.sort_unstable();
275        icons.dedup();
276        assert_eq!(icons.len(), 5);
277    }

279    #[test]
280    fn every_documented_hint_has_its_own_icon_and_others_are_generic() {
281        let generic = hint_icon(Some(ToolHint::GENERIC));
282        assert_eq!(hint_icon(None), generic);
283        assert_eq!(hint_icon(Some("teleport")), generic);
284        assert_eq!(hint_icon(Some("shell")), icondata::FaTerminalSolid);
285        assert_eq!(
286            hint_icon(Some(ToolHint::FILE_WRITE)),
287            hint_icon(Some(ToolHint::FILE_EDIT))
288        );
289        let mut distinct = ToolHint::DOCUMENTED
290            .iter()
291            .filter(|hint| **hint != ToolHint::FILE_EDIT)
292            .map(|hint| hint_icon(Some(hint)).data)
293            .collect::<Vec<_>>();
294        let documented = distinct.len();
295        distinct.sort_unstable();
296        distinct.dedup();
297        assert_eq!(distinct.len(), documented, "two hints share an icon");
298    }

300    #[test]
301    fn statuses_name_themselves_and_only_running_ones_spin() {
302        for status in [
303            ToolStatus::Running,
304            ToolStatus::Completed,
305            ToolStatus::Failed,
306            ToolStatus::Cancelled,
307            ToolStatus::Interrupted,
308        ] {
309            let mark = tool_status(&status, false);
310            assert_eq!(mark.spins, status == ToolStatus::Running, "{status:?}");
311            assert!(!mark.label.is_empty());
312        }
313        let backgrounded = tool_status(&ToolStatus::Running, true);
314        assert!(!backgrounded.spins);
315        assert_eq!(backgrounded.label, "in the background");
316        assert_eq!(tool_status(&ToolStatus::Failed, false).label, "failed");
317
318        assert!(session_status(SessionStatus::Running).spins);
319        assert_eq!(session_status(SessionStatus::Completed).label, "completed");
320        assert_eq!(
321            session_status(SessionStatus::Failed).icon,
322            tool_status(&ToolStatus::Failed, false).icon
323        );
324        assert_eq!(session_status(SessionStatus::Unknown).label, "unknown");
325        // Rows mark only what runs, what waits, and what went wrong.
326        assert!(row_status(&SessionActivity::Running).is_some_and(|mark| mark.spins));
327        let waiting = row_status(&SessionActivity::Waiting(vec![
328            cookie_agent_protocol::WaitingOn::Goal,
329        ]))
330        .expect("a waiting mark");
331        assert!(!waiting.spins);
332        assert_ne!(waiting.icon, session_status(SessionStatus::Running).icon);
333        assert!(row_status(&SessionActivity::Problem(SessionStatus::Failed)).is_some());
334        assert!(row_status(&SessionActivity::Problem(SessionStatus::Interrupted)).is_some());
335        assert!(row_status(&SessionActivity::Problem(SessionStatus::Cancelled)).is_none());
336        assert!(row_status(&SessionActivity::None).is_none());
337    }

339    #[test]
340    fn inline_svg_is_hidden_and_takes_the_text_colour() {
341        let markup = svg(icondata::FaCheckSolid, "text-xs");
342        assert!(markup.starts_with("<svg class=\"inline-block shrink-0 text-xs\""));
343        assert!(markup.contains("aria-hidden=\"true\""));
344        assert!(markup.contains("fill=\"currentColor\""));
345        // The paths are in the sprite, once, under one ID per icon.
346        assert!(!markup.contains(icondata::FaCheckSolid.data));
347        assert_eq!(markup, svg(icondata::FaCheckSolid, "text-xs"));
348        assert_ne!(
349            markup,
350            svg(icondata::FaBanSolid, "text-xs"),
351            "two icons share a symbol"
352        );
353    }
