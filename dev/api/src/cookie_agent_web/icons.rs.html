1//! The page's icons: Font Awesome Free 6, whose artwork is CC BY 4.0, from
2//! the `icondata` crate. Components show them through `leptos_icons`; the
3//! transcript, which renders to HTML text, inlines the same SVG. Each icon
4//! is a static the linker keeps only when the page names it.
5//!
6//! Icons inherit the text colour (`currentColor`) and are hidden from
7//! assistive technology: the button or text beside them carries the name.

21/// `icon` as an inline element, `class` sizing or colouring it.
22#[component]
23pub fn Glyph(
24    #[prop(into)] icon: Signal<Icon>,
25    #[prop(into, optional)] class: String,
26) -> impl IntoView {
27    view! {
28        <span class=format!("inline-flex shrink-0 items-center {class}") aria-hidden="true">
29            <leptos_icons::Icon icon=icon />
30        </span>
31    }
32}
33
34/// `icon` as SVG markup for HTML the page renders as text, `class` on the
35/// `<svg>`.
36pub fn svg(icon: Icon, class: &str) -> String {
37    format!(
38        "<svg class=\"inline-block shrink-0 {class}\" aria-hidden=\"true\" width=\"1em\" height=\"1em\" viewBox=\"{}\" fill=\"currentColor\">{}</svg>",
39        icon.view_box.unwrap_or("0 0 512 512"),
40        icon.data
41    )
42}

44/// The icon of a tool call's presentation hint, by the hint and never by
45/// the tool's name; a hint this build does not know is generic.
46pub fn hint_icon(hint: Option<&str>) -> Icon {
47    match hint.unwrap_or(ToolHint::GENERIC) {
48        ToolHint::SHELL => icondata::FaTerminalSolid,
49        ToolHint::FILE_READ => icondata::FaFileLinesSolid,
50        ToolHint::FILE_WRITE | ToolHint::FILE_EDIT => icondata::FaPenToSquareSolid,
51        ToolHint::FETCH => icondata::FaGlobeSolid,
52        ToolHint::SEARCH => icondata::FaMagnifyingGlassSolid,
53        ToolHint::DELEGATE => icondata::FaRobotSolid,
54        ToolHint::MESSAGE => icondata::FaEnvelopeSolid,
55        ToolHint::SKILL => icondata::FaWandMagicSparklesSolid,
56        ToolHint::GOAL => icondata::FaBullseyeSolid,
57        _ => icondata::FaWrenchSolid,
58    }
59}

61/// How a status shows: its icon, its name, the colour classes, and whether
62/// the icon turns while the thing runs.
63#[derive(Clone, Copy, Debug, PartialEq)]
64pub struct StatusMark {
65    pub icon: Icon,
66    pub label: &'static str,
67    pub tone: &'static str,
68    pub spins: bool,
69}
70
71impl StatusMark {
72    /// The icon's classes: its colour, turning while it runs, and `extra`.
73    pub fn class(self, extra: &str) -> String {
74        let spin = if self.spins { "animate-spin" } else { "" };
75        format!("{} {spin} {extra}", self.tone)
76    }
77}

79const RUNNING: &str = "text-caramel dark:text-maple";
80const DONE: &str = "text-green-700 dark:text-green-400";
81const FAILED: &str = "text-red-700 dark:text-red-400";
82const QUIET: &str = "text-ink dark:text-sand";
83
84/// A tool call's status.
85pub fn tool_status(status: &ToolStatus, backgrounded: bool) -> StatusMark {
86    let (icon, label, tone, spins) = match status {
87        ToolStatus::Running if backgrounded => (
88            icondata::FaHourglassHalfSolid,
89            "in the background",
90            QUIET,
91            false,
92        ),
93        ToolStatus::Running => (icondata::FaCircleNotchSolid, "running", RUNNING, true),
94        ToolStatus::Completed => (icondata::FaCircleCheckSolid, "done", DONE, false),
95        ToolStatus::Failed => (icondata::FaCircleXmarkSolid, "failed", FAILED, false),
96        ToolStatus::Cancelled => (icondata::FaBanSolid, "cancelled", QUIET, false),
97        ToolStatus::Interrupted => (icondata::FaCirclePauseSolid, "interrupted", QUIET, false),
98    };
99    StatusMark {
100        icon,
101        label,
102        tone,
103        spins,
104    }
105}

107/// A session's status in the session list.
108pub fn session_status(status: SessionStatus) -> StatusMark {
109    let (icon, label, tone, spins) = match status {
110        SessionStatus::Running => (icondata::FaCircleNotchSolid, "running", RUNNING, true),
111        SessionStatus::Idle => (icondata::FaCircleRegular, "idle", QUIET, false),
112        SessionStatus::Completed => (icondata::FaCircleCheckSolid, "completed", DONE, false),
113        SessionStatus::Failed => (icondata::FaCircleXmarkSolid, "failed", FAILED, false),
114        SessionStatus::Cancelled => (icondata::FaBanSolid, "cancelled", QUIET, false),
115        SessionStatus::Interrupted => (icondata::FaCirclePauseSolid, "interrupted", QUIET, false),
116        SessionStatus::Unknown => (icondata::FaCircleQuestionRegular, "unknown", QUIET, false),
117    };
118    StatusMark {
119        icon,
120        label,
121        tone,
122        spins,
123    }
124}

126/// The mark a session's row shows before its title, if any: a turning
127/// circle while it runs, and a mark when its last run failed or was
128/// interrupted. A finished, idle, or cancelled session has none, so its
129/// title starts at the row's edge. (A session waiting on a producer, such
130/// as an active goal, will have one too, which running outranks.)
131pub fn row_status(status: SessionStatus) -> Option<StatusMark> {
132    match status {
133        SessionStatus::Running | SessionStatus::Failed | SessionStatus::Interrupted => {
134            Some(session_status(status))
135        }
136        SessionStatus::Idle
137        | SessionStatus::Completed
138        | SessionStatus::Cancelled
139        | SessionStatus::Unknown => None,
140    }
141}

153/// The permission mode's icon, a shield of its own for each mode (a
154/// phone's bottom bar shows only the icon), and what the mode does.
155pub fn permission_mode(mode: PermissionMode) -> (Icon, &'static str) {
156    match mode {
157        PermissionMode::AutoApprove => (
158            icondata::FaShieldHalvedSolid,
159            "the classifier approves safe calls and asks you about the rest",
160        ),
161        PermissionMode::AutoApproveN => (
162            icondata::FaShieldSolid,
163            "the classifier approves safe calls and rejects the rest without asking",
164        ),
165        PermissionMode::AutoApproveY => (
166            icondata::FaShieldHeartSolid,
167            "the classifier approves safe calls and approves the rest without asking",
168        ),
169        PermissionMode::Ask => (
170            icondata::FaUserShieldSolid,
171            "every call that needs approval asks you",
172        ),
173        PermissionMode::Yolo => (
174            icondata::FaShieldVirusSolid,
175            "every call that would ask is approved",
176        ),
177    }
178}

180#[cfg(test)]
181mod tests {
182    use cookie_agent_client_state::ToolStatus;
183    use cookie_agent_protocol::{SessionStatus, ToolHint};
184
185    use super::{hint_icon, permission_mode, row_status, session_status, svg, tool_status};

187    #[test]
188    fn every_permission_mode_has_its_own_icon() {
189        use cookie_agent_protocol::PermissionMode;
190        let mut icons = [
191            PermissionMode::AutoApprove,
192            PermissionMode::AutoApproveN,
193            PermissionMode::AutoApproveY,
194            PermissionMode::Ask,
195            PermissionMode::Yolo,
196        ]
197        .map(|mode| permission_mode(mode).0.data)
198        .to_vec();
199        icons.sort_unstable();
200        icons.dedup();
201        assert_eq!(icons.len(), 5);
202    }
203
204    #[test]
205    fn every_documented_hint_has_its_own_icon_and_others_are_generic() {
206        let generic = hint_icon(Some(ToolHint::GENERIC));
207        assert_eq!(hint_icon(None), generic);
208        assert_eq!(hint_icon(Some("teleport")), generic);
209        assert_eq!(hint_icon(Some("shell")), icondata::FaTerminalSolid);
210        assert_eq!(
211            hint_icon(Some(ToolHint::FILE_WRITE)),
212            hint_icon(Some(ToolHint::FILE_EDIT))
213        );
214        let mut distinct = ToolHint::DOCUMENTED
215            .iter()
216            .filter(|hint| **hint != ToolHint::FILE_EDIT)
217            .map(|hint| hint_icon(Some(hint)).data)
218            .collect::<Vec<_>>();
219        let documented = distinct.len();
220        distinct.sort_unstable();
221        distinct.dedup();
222        assert_eq!(distinct.len(), documented, "two hints share an icon");
223    }
224
225    #[test]
226    fn statuses_name_themselves_and_only_running_ones_spin() {
227        for status in [
228            ToolStatus::Running,
229            ToolStatus::Completed,
230            ToolStatus::Failed,
231            ToolStatus::Cancelled,
232            ToolStatus::Interrupted,
233        ] {
234            let mark = tool_status(&status, false);
235            assert_eq!(mark.spins, status == ToolStatus::Running, "{status:?}");
236            assert!(!mark.label.is_empty());
237        }
238        let backgrounded = tool_status(&ToolStatus::Running, true);
239        assert!(!backgrounded.spins);
240        assert_eq!(backgrounded.label, "in the background");
241        assert_eq!(tool_status(&ToolStatus::Failed, false).label, "failed");
242
243        assert!(session_status(SessionStatus::Running).spins);
244        assert_eq!(session_status(SessionStatus::Completed).label, "completed");
245        assert_eq!(
246            session_status(SessionStatus::Failed).icon,
247            tool_status(&ToolStatus::Failed, false).icon
248        );
249        assert_eq!(session_status(SessionStatus::Unknown).label, "unknown");
250        // Rows mark only what runs and what went wrong.
251        assert!(row_status(SessionStatus::Running).is_some_and(|mark| mark.spins));
252        assert!(row_status(SessionStatus::Failed).is_some());
253        assert!(row_status(SessionStatus::Interrupted).is_some());
254        for quiet in [
255            SessionStatus::Idle,
256            SessionStatus::Completed,
257            SessionStatus::Cancelled,
258            SessionStatus::Unknown,
259        ] {
260            assert!(row_status(quiet).is_none(), "{quiet:?}");
261        }
262    }
