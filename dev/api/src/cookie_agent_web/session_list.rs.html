1//! The session list: `session.list` read on connecting and on a resync, and
2//! kept current between reads by `sessions.changed`. A change that lands
3//! while a read is on its way may postdate what that read saw, so it is
4//! applied again to the read's answer when that lands; replaying one the
5//! answer already holds is harmless, as every later change follows it.
6
7use cookie_agent_client_state::refresh::{Latest, Ticket};
8use cookie_agent_protocol::{SessionMeta, SessionOrigin, SessionsChangedNotification};
9
10/// What a change did to the session list.
11#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
12pub struct Applied {
13    /// The list differs: its readers must look again.
14    pub changed: bool,
15    /// A session moved between projects.
16    pub moved: bool,
17}
18
19/// Applies an upsert or a removal to `sessions`: an upsert replaces a
20/// session the list holds and adds a new root, and leaves a subagent it
21/// does not hold out. An upsert of a session as the list holds it already
22/// changes nothing.
23pub fn apply_change(
24    sessions: &mut Vec<SessionMeta>,
25    change: &SessionsChangedNotification,
26) -> Applied {
27    match change {
28        SessionsChangedNotification::Upserted { session } => {
29            if let Some(existing) = sessions
30                .iter_mut()
31                .find(|existing| existing.session_id == session.session_id)
32            {
33                if *existing == **session {
34                    return Applied::default();
35                }
36                let moved = existing.project != session.project;
37                *existing = (**session).clone();
38                Applied {
39                    changed: true,
40                    moved,
41                }
42            } else if matches!(session.origin, SessionOrigin::Root) {
43                sessions.push((**session).clone());
44                Applied {
45                    changed: true,
46                    moved: session.project.is_some(),
47                }
48            } else {
49                Applied::default()
50            }
51        }
52        SessionsChangedNotification::Removed { session_id } => {
53            let before = sessions.len();
54            sessions.retain(|session| session.session_id != *session_id);
55            Applied {
56                changed: sessions.len() != before,
57                moved: false,
58            }
59        }
60        _ => Applied::default(),
61    }
62}

64/// The session list's reads, and the changes that came while the newest
65/// was on its way.
66#[derive(Debug, Default)]
67pub struct ListReads {
68    reads: Latest,
69    pending: Option<Ticket>,
70    since: Vec<SessionsChangedNotification>,
71}
72
73impl ListReads {
74    /// Starts a read; an answer to an earlier one is dropped.
75    pub fn begin(&mut self) -> Ticket {
76        let read = self.reads.begin();
77        self.pending = Some(read);
78        self.since.clear();
79        read
80    }
81
82    /// Notes `change`, applied to the list shown, for the answer on its
83    /// way.
84    pub fn note(&mut self, change: &SessionsChangedNotification) {
85        if self.pending.is_some()
86            && matches!(
87                change,
88                SessionsChangedNotification::Upserted { .. }
89                    | SessionsChangedNotification::Removed { .. }
90            )
91        {
92            self.since.push(change.clone());
93        }
94    }
95
96    /// The list to show for `read`'s answer `sessions`, with the changes
97    /// since it was asked for applied; `None` when a later read started.
98    pub fn land(
99        &mut self,
100        read: Ticket,
101        mut sessions: Vec<SessionMeta>,
102    ) -> Option<Vec<SessionMeta>> {
103        if !self.reads.is_current(read) {
104            return None;
105        }
106        self.pending = None;
107        for change in std::mem::take(&mut self.since) {
108            apply_change(&mut sessions, &change);
109        }
110        Some(sessions)
111    }
112
113    /// `read` failed: nothing waits for the changes any more.
114    pub fn fail(&mut self, read: Ticket) {
115        if self.reads.is_current(read) {
116            self.pending = None;
117            self.since.clear();
118        }
119    }
120}

169    fn upserted(session: &SessionMeta) -> SessionsChangedNotification {
170        SessionsChangedNotification::Upserted {

188    /// A session another client creates while `session.list` is on its
189    /// way, and one it deletes: the answer, read before both, lands after
190    /// them and still shows the new session and not the deleted one.
191    #[test]
192    fn changes_during_a_read_apply_to_its_answer() {
193        let kept = meta("kept", false);
194        let deleted = meta("deleted", false);
195        let created = meta("created", false);
196        let mut reads = ListReads::default();
197        let mut shown = Vec::new();
198        let read = reads.begin();
199        for change in [
200            upserted(&created),
201            SessionsChangedNotification::Removed {
202                session_id: deleted.session_id,
203            },
204        ] {
205            reads.note(&change);
206            apply_change(&mut shown, &change);
207        }
208        let shown = reads
209            .land(read, vec![kept.clone(), deleted.clone()])
210            .expect("the newest read");
211        assert_eq!(titles(&shown), ["kept", "created"]);
212
213        // Nothing is on its way now: a change is not kept for later.
214        reads.note(&upserted(&meta("later", false)));
215        let read = reads.begin();
216        let shown = reads.land(read, vec![kept]).expect("the newest read");
217        assert_eq!(titles(&shown), ["kept"]);
218    }

220    /// Two reads overlap: the earlier one's answer is dropped, and the
221    /// later one's gets only the changes since it was asked for.
222    #[test]
223    fn an_older_read_is_dropped() {
224        let first_session = meta("first", false);
225        let mut reads = ListReads::default();
226        let first = reads.begin();
227        reads.note(&upserted(&first_session));
228        let second = reads.begin();
229        let renamed = meta("second", false);
230        reads.note(&upserted(&renamed));
231        assert_eq!(reads.land(first, Vec::new()), None);
232        let shown = reads
233            .land(second, vec![first_session])
234            .expect("the newest read");
235        assert_eq!(titles(&shown), ["first", "second"]);
236    }

267    /// The daemon announces a session again as it was: the list's readers
268    /// are not woken for it.
269    #[test]
270    fn an_upsert_or_removal_that_changes_nothing_says_so() {
271        let held = meta("held", false);
272        let mut sessions = vec![held.clone()];
273        assert_eq!(
274            apply_change(&mut sessions, &upserted(&held)),
275            Applied::default()
276        );
277        let gone = SessionsChangedNotification::Removed {
278            session_id: SessionId::new_v7(),
279        };
280        assert_eq!(apply_change(&mut sessions, &gone), Applied::default());
281        let removed = SessionsChangedNotification::Removed {
282            session_id: held.session_id,
283        };
284        assert!(apply_change(&mut sessions, &removed).changed);
285        assert!(sessions.is_empty());
