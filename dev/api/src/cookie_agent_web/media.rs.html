1//! Media display blocks shown inline: their artifact is read from the daemon
2//! (`artifact.read`) and shown through a `Blob` URL, kept for the page's
3//! life so a re-rendered block shows it again at once.
4
5use std::{cell::RefCell, collections::HashMap};
6
7use leptos::task::spawn_local;
8use wasm_bindgen::JsValue;
9
10use crate::workspace::Workspace;
11
12thread_local! {
13    static URLS: RefCell<HashMap<String, (String, String)>> = RefCell::new(HashMap::new());
14}
15
16/// Fills every media block under `root` that is not shown yet.
17pub fn show_within(root: &web_sys::Element, workspace: Workspace) {
18    let Ok(blocks) = root.query_selector_all("[data-media]:not([data-shown])") else {
19        return;
20    };
21    for index in 0..blocks.length() {
22        let Some(element) = blocks
23            .item(index)
24            .and_then(|node| wasm_bindgen::JsCast::dyn_into::<web_sys::Element>(node).ok())
25        else {
26            continue;
27        };
28        let Some(sha256) = element.get_attribute("data-media") else {
29            continue;
30        };
31        let _ = element.set_attribute("data-shown", "loading");
32        let alt = element.get_attribute("data-alt").unwrap_or_default();
33        if let Some((url, media_type)) = URLS.with(|urls| urls.borrow().get(&sha256).cloned()) {
34            show(&element, &url, &media_type, &alt);
35            continue;
36        }
37        spawn_local(async move {
38            match workspace.read_media(sha256.clone()).await {
39                Ok((bytes, media_type)) => match blob_url(&bytes, &media_type) {
40                    Ok(url) => {
41                        URLS.with(|urls| {
42                            urls.borrow_mut()
43                                .insert(sha256, (url.clone(), media_type.clone()));
44                        });
45                        show(&element, &url, &media_type, &alt);
46                    }
47                    Err(_) => element.set_text_content(Some("the media could not be shown")),
48                },
49                Err(error) => {
50                    element
51                        .set_text_content(Some(&format!("the media could not be read: {error}")));
52                }
53            }
54        });
55    }
56}

58fn blob_url(bytes: &[u8], media_type: &str) -> Result<String, JsValue> {
59    let array = js_sys::Uint8Array::from(bytes);
60    let parts = js_sys::Array::of1(&array);
61    let options = web_sys::BlobPropertyBag::new();
62    options.set_type(media_type);
63    let blob = web_sys::Blob::new_with_u8_array_sequence_and_options(&parts, &options)?;
64    web_sys::Url::create_object_url_with_blob(&blob)
65}

67fn show(element: &web_sys::Element, url: &str, media_type: &str, alt: &str) {
68    let Some(document) = element.owner_document() else {
69        return;
70    };
71    let (tag, attributes): (&str, &[(&str, &str)]) = if media_type.starts_with("image/") {
72        (
73            "img",
74            &[("alt", alt), ("class", "max-h-96 max-w-full rounded")],
75        )
76    } else if media_type.starts_with("video/") {
77        (
78            "video",
79            &[("controls", ""), ("class", "max-h-96 max-w-full rounded")],
80        )
81    } else if media_type.starts_with("audio/") {
82        ("audio", &[("controls", "")])
83    } else {
84        (
85            "a",
86            &[
87                ("download", ""),
88                ("class", "text-caramel underline dark:text-maple"),
89            ],
90        )
91    };
92    let Ok(shown) = document.create_element(tag) else {
93        return;
94    };
95    let source = if tag == "a" { "href" } else { "src" };
96    let _ = shown.set_attribute(source, url);
97    for (name, value) in attributes {
98        let _ = shown.set_attribute(name, value);
99    }
100    if tag == "a" {
101        shown.set_text_content(Some(if alt.is_empty() { "download" } else { alt }));
102    }
103    element.set_inner_html("");
104    let _ = element.append_child(&shown);
105    let _ = element.set_attribute("data-shown", "yes");
106}
