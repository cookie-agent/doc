26/// Replaces the page's fragment in the address bar and history without
27/// reloading; an empty `fragment` removes it.
28pub fn replace_fragment(fragment: &str) {
29    let Some(window) = window() else {
30        return;
31    };
32    let location = window.location();
33    let (Ok(path), Ok(search)) = (location.pathname(), location.search()) else {
34        return;
35    };
36    if let Ok(history) = window.history() {
37        let _ = history.replace_state_with_url(
38            &wasm_bindgen::JsValue::NULL,
39            "",
40            Some(&format!("{path}{search}{fragment}")),
41        );
42    }
43}

107/// Turns the dark palette on or off: the `dark` class on `<html>`, which
108/// Tailwind's `dark:` variant follows.
109pub fn set_dark(dark: bool) {
110    let Some(root) = window()
111        .and_then(|window| window.document())
112        .and_then(|document| document.document_element())
113    else {
114        return;
115    };
116    let classes = root.class_list();
117    let _ = if dark {
118        classes.add_1("dark")
119    } else {
120        classes.remove_1("dark")
121    };
122    set_theme_color(if dark { THEME_DARK } else { THEME_LIGHT });
123}

125/// The browser's colour around the page (`theme-color`) in the light and
126/// the dark theme: the palette's cream and night, as in `index.html` and

131/// Sets every `theme-color` of the page to `colour`, so the browser's bar,
132/// or an installed app's title bar, follows the user's theme rather than
133/// only the system's.
134fn set_theme_color(colour: &str) {
135    let Some(metas) = window()
136        .and_then(|window| window.document())
137        .and_then(|document| document.query_selector_all("meta[name=theme-color]").ok())
138    else {
139        return;
140    };
141    for index in 0..metas.length() {
142        if let Some(meta) = metas
143            .item(index)
144            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
145        {
146            let _ = meta.set_attribute("content", colour);
147        }
148    }
149}

164/// Registers the page's service worker, `sw.js` beside it, which caches
165/// nothing (it makes the page installable). Only a secure page can; the
166/// browser checks it for a new build on every load, past its HTTP cache.
167pub fn register_service_worker() {
168    let Some(window) = window().filter(web_sys::Window::is_secure_context) else {
169        return;
170    };
171    let navigator = window.navigator();
172    // A private window may have no service workers.
173    if !js_sys::Reflect::has(&navigator, &"serviceWorker".into()).unwrap_or(false) {
174        return;
175    }
176    let options = web_sys::RegistrationOptions::new();
177    options.set_update_via_cache(web_sys::ServiceWorkerUpdateViaCache::None);
178    let registered = navigator
179        .service_worker()
180        .register_with_options("sw.js", &options);
181    wasm_bindgen_futures::spawn_local(async move {
182        let _ = wasm_bindgen_futures::JsFuture::from(registered).await;
183    });
184}

232/// Opens the browser's install prompt, which it offers once: call it from
233/// the click that asks for it. Resolves to whether the user installed the
234/// page; `None` without a prompt.
235pub fn prompt_install() -> Option<impl std::future::Future<Output = bool>> {
236    let window = window()?;
237    let event = window_value("cookieInstallPrompt");
238    if !event.is_object() {
239        return None;
240    }
241    let _ = js_sys::Reflect::set(
242        &window,
243        &"cookieInstallPrompt".into(),
244        &wasm_bindgen::JsValue::NULL,
245    );
246    let prompt = js_sys::Reflect::get(&event, &"prompt".into())
247        .ok()?
248        .dyn_into::<js_sys::Function>()
249        .ok()?;
250    prompt.call0(&event).ok()?;
251    let choice = js_sys::Reflect::get(&event, &"userChoice".into())
252        .ok()
253        .and_then(|choice| choice.dyn_into::<js_sys::Promise>().ok());
254    Some(async move {
255        let Some(choice) = choice else {
256            return false;
257        };
258        wasm_bindgen_futures::JsFuture::from(choice)
259            .await
260            .ok()
261            .and_then(|choice| js_sys::Reflect::get(&choice, &"outcome".into()).ok())
262            .and_then(|outcome| outcome.as_string())
263            .is_some_and(|outcome| outcome == "accepted")
264    })
265}

267/// Calls `shown` whenever the page becomes visible again, as a phone's tab
268/// does after it was suspended, for as long as the page lives.
269pub fn on_visible(shown: impl Fn() + 'static) {
270    let Some(document) = window().and_then(|window| window.document()) else {
271        return;
272    };
273    let watched = document.clone();
274    let listener = Closure::<dyn Fn()>::new(move || {
275        if watched.visibility_state() == web_sys::VisibilityState::Visible {
276            shown();
277        }
278    });
279    let _ = document
280        .add_event_listener_with_callback("visibilitychange", listener.as_ref().unchecked_ref());
281    listener.forget();
282}

306/// A `touchmove` listener on the window that keeps the page from
307/// scrolling under a finger while it drags something; [`TouchHold::remove`]
308/// takes it away.
309pub struct TouchHold {
310    listener: Closure<dyn Fn(web_sys::Event)>,
311}
312
313/// Holds the page still under a moving finger whenever `holds()` says so.
314/// The listener is not passive, or the browser would scroll regardless;
315/// the browser decides at a move's first event, so a finger that moves
316/// before `holds()` turns true scrolls as usual.
317pub fn hold_touch_scroll(holds: impl Fn() -> bool + 'static) -> Option<TouchHold> {
318    let window = window()?;
319    let listener = Closure::<dyn Fn(web_sys::Event)>::new(move |event: web_sys::Event| {
320        if holds() && event.cancelable() {
321            event.prevent_default();
322        }
323    });
324    let options = web_sys::AddEventListenerOptions::new();
325    options.set_passive(false);
326    window
327        .add_event_listener_with_callback_and_add_event_listener_options(
328            "touchmove",
329            listener.as_ref().unchecked_ref(),
330            &options,
331        )
332        .ok()?;
333    Some(TouchHold { listener })
334}
335
336impl TouchHold {
337    pub fn remove(self) {
338        if let Some(window) = window() {
339            let _ = window.remove_event_listener_with_callback(
340                "touchmove",
341                self.listener.as_ref().unchecked_ref(),
342            );
343        }
344    }
345}

347/// Marks the page as pressed, or not: while it is, nothing on the page can
348/// be selected (`html.pressing` in the style sheet), and what a long press
349/// already selected is cleared. A phone otherwise selects the words under
350/// and around a finger held on a row, and extends that selection as the
351/// finger drags.
352pub fn set_pressing(pressing: bool) {
353    let Some(window) = window() else {
354        return;
355    };
356    if let Some(root) = window
357        .document()
358        .and_then(|document| document.document_element())
359    {
360        let classes = root.class_list();
361        let _ = if pressing {
362            classes.add_1("pressing")
363        } else {
364            classes.remove_1("pressing")
365        };
366    }
367    if let Ok(Some(selection)) = window.get_selection() {
368        let _ = selection.remove_all_ranges();
369    }
370}

407/// The `attribute` of the element `selector` matches that is at `x`, `y`
408/// in the viewport or holds what is: what a drag is over.
409pub fn attribute_at(x: f64, y: f64, selector: &str, attribute: &str) -> Option<String> {
410    // A viewport coordinate is far inside `f32`.
411    let element = window()?
412        .document()?
413        .element_from_point(x as f32, y as f32)?;
414    element.closest(selector).ok()??.get_attribute(attribute)
415}

417/// Scrolls the element `selector` matches when `y` is near its top or
418/// bottom edge, so a drag reaches what is out of view.
419pub fn scroll_near_edge(selector: &str, y: f64) {
420    const EDGE: f64 = 48.0;
421    const STEP: f64 = 12.0;
422    let Some(element) = window()
423        .and_then(|window| window.document())
424        .and_then(|document| document.query_selector(selector).ok().flatten())
425    else {
426        return;
427    };
428    let frame = element.get_bounding_client_rect();
429    if y < frame.top() + EDGE {
430        element.scroll_by_with_x_and_y(0.0, -STEP);
431    } else if y > frame.bottom() - EDGE {
432        element.scroll_by_with_x_and_y(0.0, STEP);
433    }
434}

436/// Keeps `--app-height` on `<html>` at the visual viewport's height, which a
437/// phone's on-screen keyboard shrinks while the layout viewport (and
438/// `100dvh`) may not, so the composer and an open sheet stay above the
439/// keyboard. The page's frame and sheets size themselves from it.
440pub fn follow_visual_viewport() {
441    let Some(window) = window() else {
442        return;
443    };
444    let Some(viewport) = window.visual_viewport() else {
445        return;
446    };
447    let update = {
448        let viewport = viewport.clone();
449        let window = window.clone();
450        move || {
451            let Some(root) = window
452                .document()
453                .and_then(|document| document.document_element())
454                .and_then(|root| root.dyn_into::<web_sys::HtmlElement>().ok())
455            else {
456                return;
457            };
458            let _ = root
459                .style()
460                .set_property("--app-height", &format!("{}px", viewport.height()));
461            // A browser that scrolled the page to show the focused field
462            // leaves the frame where it was; bring it back.
463            if viewport.offset_top() > 0.0 {
464                window.scroll_to_with_x_and_y(0.0, 0.0);
465            }
466        }
467    };
468    update();
469    let listener = Closure::<dyn Fn()>::new(update);
470    let _ = viewport.add_event_listener_with_callback("resize", listener.as_ref().unchecked_ref());
471    listener.forget();
472}

485/// Whether the page matches the media `query` now.
486pub fn matches(query: &str) -> bool {
487    media_query(query).is_some_and(|list| list.matches())
488}

490/// Calls `changed` with whether the page matches `query` whenever that
491/// changes, for as long as the page lives.
492pub fn on_media_change(query: &str, changed: impl Fn(bool) + 'static) {
493    let Some(list) = media_query(query) else {
494        return;
495    };
496    let watched = list.clone();
497    let listener = Closure::<dyn Fn()>::new(move || changed(watched.matches()));
498    let _ = list.add_event_listener_with_callback("change", listener.as_ref().unchecked_ref());
499    listener.forget();

502/// Adds a history entry for the page as it is, carrying `depth`: how many
503/// of the page's layers are open with it. Browser Back then returns to the
504/// entry below instead of leaving the page.
505pub fn push_history(depth: usize) {
506    if let Some(history) = window().and_then(|window| window.history().ok()) {
507        let _ = history.push_state(&wasm_bindgen::JsValue::from(depth as f64), "");
508    }
509}
510
511/// Goes `steps` entries back in the history.
512pub fn history_back(steps: usize) {
513    if let Some(history) = window().and_then(|window| window.history().ok()) {
514        let _ = history.go_with_delta(-(steps as i32));
515    }
516}

518/// Calls `arrived` with the depth of the history entry the browser moved
519/// to (0 for one the page did not add), for as long as the page lives.
520pub fn on_history_move(arrived: impl Fn(usize) + 'static) {
521    let Some(window) = window() else {
522        return;
523    };
524    let listener =
525        Closure::<dyn Fn(web_sys::PopStateEvent)>::new(move |event: web_sys::PopStateEvent| {
526            let depth = event.state().as_f64().map_or(0, |depth| depth as usize);
527            arrived(depth);
528        });
529    let _ = window.add_event_listener_with_callback("popstate", listener.as_ref().unchecked_ref());
530    listener.forget();
