1//! Installing the page as an app (spec 9), independent of the browser:
2//! which offer the page makes from what the browser says about itself.
3//!
4//! Chromium hands the page a prompt (`beforeinstallprompt`) once the page
5//! qualifies, which the page shows as **Install app**. iOS and iPadOS have
6//! no prompt: the page says how to add it from the Share menu instead. A
7//! page that is not a secure context (plain HTTP on a LAN address) cannot
8//! be installed at all, and a page already running as an app offers
9//! nothing.
10
11/// The `localStorage` key of the phone's install banner: `dismissed` once
12/// the user closed it.
13pub const BANNER_KEY: &str = "cookie.install_banner";
14
15/// What the browser says about installing the page.
16#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
17pub struct InstallContext {
18    /// The page is a secure context: HTTPS, or `localhost` and `127.0.0.1`.
19    pub secure: bool,
20    /// The page runs as an installed app: `display-mode: standalone`, or
21    /// iOS's `navigator.standalone`.
22    pub standalone: bool,
23    /// The browser said it installed the page (`appinstalled`).
24    pub installed: bool,
25    /// The browser handed the page an install prompt it has not used.
26    pub prompt: bool,
27    /// The browser runs on iOS or iPadOS.
28    pub ios: bool,
29}
30
31/// What the page offers for installing it.
32#[derive(Clone, Copy, Debug, Eq, PartialEq)]
33pub enum InstallOffer {
34    /// It runs as an app, or the browser just installed it: nothing.
35    Installed,
36    /// **Install app**, which opens the browser's prompt.
37    Prompt,
38    /// Steps for iOS: Share, then Add to Home Screen.
39    AddToHomeScreen,
40    /// Installing needs HTTPS.
41    NeedsHttps,
42    /// The browser offered no prompt (yet): its own menu may install it.
43    BrowserMenu,
44}
45
46impl InstallContext {
47    /// The offer the page makes.
48    pub const fn offer(self) -> InstallOffer {
49        if self.standalone || self.installed {
50            InstallOffer::Installed
51        } else if self.prompt {
52            InstallOffer::Prompt
53        } else if !self.secure {
54            InstallOffer::NeedsHttps
55        } else if self.ios {
56            InstallOffer::AddToHomeScreen
57        } else {
58            InstallOffer::BrowserMenu
59        }
60    }
61}
62
63impl InstallOffer {
64    /// Whether the page has an action for it outside Settings (the connect
65    /// screen's, and the phone's banner): a prompt, or iOS's steps.
66    pub const fn actionable(self) -> bool {
67        matches!(self, Self::Prompt | Self::AddToHomeScreen)
68    }
69}
70
71/// Whether a browser of `user_agent`, whose screen takes
72/// `max_touch_points` touches, runs on iOS or iPadOS. Every iOS browser is
73/// Safari underneath, whose Share menu adds the page to the home screen.
74/// iPadOS asks for desktop pages by default and names itself a Mac, which
75/// has no touch screen.
76pub fn is_ios(user_agent: &str, max_touch_points: i32) -> bool {
77    ["iPhone", "iPad", "iPod"]
78        .iter()
79        .any(|device| user_agent.contains(device))
80        || (user_agent.contains("Macintosh") && max_touch_points > 1)
81}
82
83/// Whether a phone shows the install banner: the page has an action for
84/// installing it and the user has not dismissed the banner.
85pub const fn shows_banner(offer: InstallOffer, phone: bool, dismissed: bool) -> bool {
86    phone && offer.actionable() && !dismissed
87}
88
89/// Whether the value stored under [`BANNER_KEY`] dismissed the banner.
90pub fn banner_dismissed(stored: Option<&str>) -> bool {
91    stored == Some("dismissed")
92}

194    /// The width and height of a PNG, or of a WebP of a lossy (`VP8 `) or
195    /// extended (`VP8X`) kind.
196    fn image_size(bytes: &[u8]) -> (u32, u32) {
197        let u32_be = |at: usize| u32::from_be_bytes(bytes[at..at + 4].try_into().unwrap());
198        let u24_le = |at: usize| {
199            u32::from(bytes[at]) | u32::from(bytes[at + 1]) << 8 | u32::from(bytes[at + 2]) << 16
200        };
201        if bytes.starts_with(b"\x89PNG") {
202            (u32_be(16), u32_be(20))
203        } else if bytes.starts_with(b"RIFF") && &bytes[12..16] == b"VP8 " {
204            (u24_le(26) & 0x3fff, u24_le(28) & 0x3fff)
205        } else if bytes.starts_with(b"RIFF") && &bytes[12..16] == b"VP8X" {
206            (u24_le(24) + 1, u24_le(27) + 1)
207        } else {
208            panic!("not a PNG or a WebP this test reads");
209        }
210    }

212    /// What Chromium's install criteria ask of the manifest, and its richer
213    /// install dialog of the screenshots; the files it names are the sizes
214    /// it says.
215    #[test]
216    fn the_manifest_meets_the_install_criteria() {
217        let manifest: serde_json::Value =
218            serde_json::from_slice(&asset("assets/manifest.webmanifest")).expect("JSON");
219        for field in ["name", "short_name", "description"] {
220            assert!(
221                manifest[field]
222                    .as_str()
223                    .is_some_and(|value| !value.is_empty())
224            );
225        }
226        // Relative to the manifest, so the page installs from any path.
227        for field in ["id", "start_url", "scope"] {
228            assert_eq!(manifest[field], "./", "{field}");
229        }
230        assert_eq!(manifest["display"], "standalone");
231        assert!(manifest.get("prefer_related_applications").is_none());
232        let icons = manifest["icons"].as_array().expect("icons");
233        for (size, side) in [("192x192", 192), ("512x512", 512)] {
234            for purpose in ["any", "maskable"] {
235                let icon = icons
236                    .iter()
237                    .find(|icon| icon["sizes"] == size && icon["purpose"] == purpose)
238                    .unwrap_or_else(|| panic!("a {size} {purpose} icon"));
239                assert_eq!(icon["type"], "image/png");
240                let src = icon["src"].as_str().unwrap();
241                assert_eq!(
242                    image_size(&asset(&format!("assets/{src}"))),
243                    (side, side),
244                    "{src}"
245                );
246            }
247        }
248        let screenshots = manifest["screenshots"].as_array().expect("screenshots");
249        for form_factor in ["narrow", "wide"] {
250            assert!(
251                screenshots
252                    .iter()
253                    .any(|shot| shot["form_factor"] == form_factor),
254                "a {form_factor} screenshot"
255            );
256        }
257        for shot in screenshots {
258            let src = shot["src"].as_str().unwrap();
259            let (width, height) = image_size(&asset(&format!("assets/{src}")));
260            assert_eq!(shot["sizes"], format!("{width}x{height}"), "{src}");
261            assert_eq!(shot["type"], "image/webp", "{src}");
262            assert!(
263                shot["label"]
264                    .as_str()
265                    .is_some_and(|label| !label.is_empty())
266            );
267            // Chromium shows screenshots 320 to 3840 px a side, the longer
268            // at most 2.3 times the shorter.
269            for side in [width, height] {
270                assert!((320..=3840).contains(&side), "{src}: {width}x{height}");
271            }
272            let (long, short) = (width.max(height), width.min(height));
273            assert!(f64::from(long) <= 2.3 * f64::from(short), "{src}");
274        }
275    }

277    /// The service worker caches nothing, takes over at once, keeps off the
278    /// daemon's WebSocket endpoints, shows every push, opens a clicked
279    /// notification's session, and carries the line the build stamps;
280    /// `index.html` copies it under its own name and names the manifest,
281    /// iOS's tags, and the early keeper of Chromium's install prompt.
282    #[test]
283    fn the_service_worker_is_network_only() {
284        let worker = String::from_utf8(asset("assets/sw.js")).unwrap();
285        assert!(worker.contains("self.skipWaiting()"));
286        assert!(worker.contains("self.clients.claim()"));
287        assert!(worker.contains(r#"addEventListener("fetch""#));
288        // A failed fetch becomes the network error, not an uncaught rejection.
289        assert!(worker.contains("event.respondWith(fetch(request).catch(() => Response.error()))"));
290        assert!(worker.contains(r"/\/(ws|runtime)$/"));
291        // No Cache Storage: it caches nothing.
292        assert!(!worker.contains("caches."));
293        // Every push shows a notification, whatever its payload: Safari
294        // revokes a subscription whose pushes show none (push
295        // notifications spec section 9.2).
296        assert!(worker.contains(r#"addEventListener("push""#));
297        assert_eq!(worker.matches("showNotification(").count(), 1);
298        assert!(
299            worker.contains("event.waitUntil(self.registration.showNotification(title, options))")
300        );
301        assert!(worker.contains(r#": "Cookie Agent";"#));
302        // A click opens its session in an open window, else a new one.
303        assert!(worker.contains(r#"addEventListener("notificationclick""#));
304        assert!(worker.contains(r#"open.postMessage({ type: "open-session", session })"#));
305        assert!(worker.contains(r#"url.hash = "session=" + session"#));
306        assert_eq!(worker.lines().last(), Some("// build: source"));
307        let index = String::from_utf8(asset("index.html")).unwrap();
308        for needle in [
309            r#"<link data-trunk rel="copy-file" href="assets/sw.js" />"#,
310            r#"<link rel="manifest" href="manifest.webmanifest" />"#,
311            r#"<link rel="apple-touch-icon" href="icons/apple-touch-icon.png" />"#,
312            r#"<meta name="apple-mobile-web-app-capable" content="yes" />"#,
313            r#"<meta name="apple-mobile-web-app-title" content="Cookie Agent" />"#,
314            r#"addEventListener("beforeinstallprompt""#,
315            "event.preventDefault()",
316        ] {
317            assert!(index.contains(needle), "{needle}");
318        }
319    }
