4/// A command the page runs.

21    Approvals,

34    Permissions,

189/// The command a composer text names and its argument: `/<name>` or an
190/// alias, followed by an argument only for a command that takes one.
191pub fn parse(text: &str) -> Option<(Command, String)> {
192    let text = text.trim().strip_prefix('/')?;
193    COMMANDS.iter().find_map(|spec| {
194        std::iter::once(spec.name)
195            .chain(spec.aliases.iter().copied())
196            .find_map(|name| {
197                let rest = text.strip_prefix(name)?;
198                let argument = if rest.is_empty() {
199                    ""
200                } else {
201                    rest.strip_prefix(char::is_whitespace)?.trim()
202                };
203                (argument.is_empty() || spec.takes_argument)
204                    .then(|| (spec.command, argument.to_owned()))
205            })
206    })
207}

209/// The commands whose names or aliases start with what follows `/` in
210/// `text`, for the menu while the user types one.
211pub fn completions(text: &str) -> Vec<&'static CommandSpec> {
212    let Some(prefix) = text.trim_start().strip_prefix('/') else {
213        return Vec::new();
214    };
215    COMMANDS
216        .iter()
217        .filter(|spec| {
218            spec.name.starts_with(prefix)
219                || spec.aliases.iter().any(|alias| alias.starts_with(prefix))
220        })
221        .collect()
222}

224#[cfg(test)]
225mod tests {
226    use super::{Command, completions, parse};
227
228    #[test]
229    fn names_and_aliases_parse_as_in_the_tui() {
230        let command = |text: &str| parse(text).map(|(command, _)| command);
231        assert_eq!(command("/new"), Some(Command::New));
232        assert_eq!(command(" /resume "), Some(Command::Sessions));
233        assert_eq!(command("/continue"), Some(Command::Sessions));
234        assert_eq!(command("/usage"), Some(Command::Usage));
235        assert_eq!(command("/events"), Some(Command::Events));
236        assert_eq!(command("/bg"), Some(Command::Background));
237        assert_eq!(command("/perms"), Some(Command::Permissions));
238        assert_eq!(command("/image"), Some(Command::Attach));
239        assert_eq!(command("/agent-panel hide"), Some(Command::AgentPanelHide));
240        assert_eq!(command("new"), None);
241        assert_eq!(command("/nothing"), None);
242        assert_eq!(command("/new session"), None);
243        assert_eq!(command("/newer"), None);
244        assert_eq!(
245            parse("/goal ship the release "),
246            Some((Command::Goal, "ship the release".into()))
247        );
248        assert_eq!(parse("/goal"), Some((Command::Goal, String::new())));
249        assert_eq!(
250            parse("/compact keep the API notes"),
251            Some((Command::Compact, "keep the API notes".into()))
252        );
253    }
254
255    #[test]
256    fn completions_follow_the_typed_prefix() {
257        let names = completions("/m")
258            .into_iter()
259            .map(|spec| spec.name)
260            .collect::<Vec<_>>();
261        assert_eq!(names, ["model", "mode"]);
262        assert!(completions("hello").is_empty());
263        assert_eq!(completions("/").len(), super::COMMANDS.len());
264    }
