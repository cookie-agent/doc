1//! The session's commands, as the title bar's ⋮ menu lists them. The web
2//! composer has no slash commands: text that starts with `/` is a message
3//! like any other, and every command is a menu entry with a readable name,
4//! an icon, and a line saying what it does. Commands with somewhere else to
5//! live are not here: the bottom bar picks the agent, the model, and the
6//! permission mode, the composer attaches and stops, the sessions list
7//! starts and opens sessions, and the rail opens the daemon-wide views.
8
9use icondata::Icon;
10
11/// A command of the session's ⋮ menu. Each opens a sheet, but Rename
12/// edits the title in the title bar.
13#[derive(Clone, Copy, Debug, Eq, PartialEq)]
14pub enum Command {
15    /// Rename the session, or go back to its automatic title.
16    Rename,
17    /// Compact the session's context, with an optional focus.
18    Compact,
19    /// Set the root session's goal, or pause, resume, or cancel it.
20    Goal,
21    /// Schedule a prompt into the session: the cron job form, in the
22    /// Scheduled prompts view (client protocol 1.15).
23    Schedule,
24    /// Search the skills and run one.
25    Skills,
26    /// The background calls, to stop one.
27    Background,
28    /// The session's runtimes and the daemon's machines.
29    Runtime,
30    /// The session's permission overrides.
31    Permissions,
32    /// The session's usage and cost.
33    Usage,
34    /// Choose which diagnostic rows the transcript shows.
35    Events,
36    /// The agent preset for new sessions.
37    Preset,
38}
39
40/// A menu entry: what it runs, its `data-command` id, its name, its icon,
41/// and what it does.
42pub struct MenuEntry {
43    pub command: Command,
44    pub id: &'static str,
45    pub label: &'static str,
46    pub icon: Icon,
47    pub description: &'static str,
48}

132/// Whether the menu shows `command` to a daemon that does or does not keep
133/// cron jobs (`has_crons`, client protocol 1.15).
134pub const fn offered(command: Command, has_crons: bool) -> bool {
135    !matches!(command, Command::Schedule) || has_crons
136}
137
138/// The menu entry of `command`.
139pub fn entry(command: Command) -> &'static MenuEntry {
140    MENU.iter()
141        .find(|entry| entry.command == command)
142        .expect("every command is in the menu")
143}

145#[cfg(test)]
146mod tests {
147    use super::{Command, MENU, entry, offered};
148
149    #[test]
150    fn only_a_daemon_that_keeps_cron_jobs_offers_scheduling() {
151        assert!(offered(Command::Schedule, true));
152        assert!(!offered(Command::Schedule, false));
153        assert!(offered(Command::Goal, false));
154    }
155
156    #[test]
157    fn the_menu_lists_every_command_once_by_a_readable_name() {
158        for command in [
159            Command::Rename,
160            Command::Compact,
161            Command::Goal,
162            Command::Schedule,
163            Command::Skills,
164            Command::Background,
165            Command::Runtime,
166            Command::Permissions,
167            Command::Usage,
168            Command::Events,
169            Command::Preset,
170        ] {
171            assert_eq!(
172                MENU.iter().filter(|entry| entry.command == command).count(),
173                1,
174                "{command:?}"
175            );
176            assert_eq!(entry(command).command, command);
177        }
178        assert_eq!(MENU.len(), 11);
179        let mut ids = MENU.iter().map(|entry| entry.id).collect::<Vec<_>>();
180        ids.sort_unstable();
181        ids.dedup();
182        assert_eq!(ids.len(), MENU.len(), "ids are distinct");
183        for entry in MENU {
184            // A name and a sentence, not slash syntax.
185            assert!(!entry.label.contains('/'), "{}", entry.label);
186            assert!(
187                entry.label.starts_with(char::is_uppercase),
188                "{}",
189                entry.label
190            );
191            assert!(!entry.description.contains('/'), "{}", entry.description);
192            assert!(!entry.description.is_empty());
193        }
194        assert_eq!(
195            MENU.iter()
196                .map(|entry| entry.id)
197                .take(3)
198                .collect::<Vec<_>>(),
199            ["rename", "compact", "goal"]
200        );
201    }
