1//! Assets on the page (`docs/site/specs/assets.md` section 8): each asset
2//! placeholder (`data-asset`) is filled with the asset as it is now, read
3//! with `asset.get` and `asset.read`, in the way its `data-asset-view`
4//! names:
5//!
6//! - `figure`, an `asset` block shown whole: images, SVG, video, and audio
7//!   as the media blocks show them, a Markdown document rendered by the
8//!   transcript's own renderer and cut to 480 px until *Show all*, an HTML
9//!   page in a sandboxed frame of `viewer.html` (section 8.2), which gets
10//!   the page by `postMessage` with a token and reports its height back
11//!   with it, anything else as a download card;
12//! - a reply's cards (`crate::render::attachments`): `thumb` a picture or
13//!   a video's first frame, whole when it is alone, `page` a live
14//!   thumbnail, `doc` the first lines, `audio` a player, and `file` only
15//!   the asset's state;
16//! - the side panel's `panel` (the asset whole, the page filling it) and
17//!   `source` (its text);
18//! - the Assets tab's `tile`: the thumbnail the daemon makes
19//!   (`asset.thumbnail`, `docs/site/specs/asset-thumbnails.md`), kept by
20//!   [`crate::thumbnail_store`] across page loads, or the kind's icon when
21//!   it makes none; the tab never reads a whole file to draw a card.
22//!
23//! A `doc` card of an asset reads only the start of it.
24//!
25//! `assets.changed` makes every one shown read its asset again: one whose
26//! content changed is filled anew and says when it was updated, one deleted
27//! says so.

107#[derive(Clone, Debug, Eq, PartialEq)]
108pub enum Source {
109    /// An asset, shown as it is now.
110    Asset { store: AssetStore, asset: AssetId },
111    /// A stored file, which never changes.
112    Artifact {
113        sha256: String,
114        media_type: String,
115        name: String,
116    },
117}

119fn placeholder(element: &web_sys::Element) -> Option<Placeholder> {
120    let source = if element.get_attribute("data-asset")? == "artifact" {
121        Source::Artifact {
122            sha256: element.get_attribute("data-artifact-sha")?,
123            media_type: element.get_attribute("data-asset-type").unwrap_or_default(),
124            name: element.get_attribute("data-asset-name").unwrap_or_default(),
125        }
126    } else {
127        Source::Asset {
128            store: serde_json::from_str(&element.get_attribute("data-asset-store")?).ok()?,
129            asset: element.get_attribute("data-asset")?.parse().ok()?,
130        }
131    };
132    Some(Placeholder {
133        source,
134        shown_sha: element.get_attribute("data-asset-sha").unwrap_or_default(),
135        title: element
136            .get_attribute("data-asset-title")
137            .unwrap_or_default(),
138        view: element
139            .get_attribute("data-asset-view")
140            .unwrap_or_else(|| "figure".to_owned()),
141    })
142}

144impl Placeholder {
145    /// The stored file as the card code reads an asset: its title, type,
146    /// name, and content; it is never replaced.
147    fn artifact_asset(&self) -> Option<Asset> {
148        let Source::Artifact {
149            sha256,
150            media_type,
151            name,
152        } = &self.source
153        else {
154            return None;
155        };
156        let stamp = cookie_agent_protocol::AssetStamp {
157            at: jiff::Timestamp::now(),
158            session: cookie_agent_protocol::SessionId::new_v7(),
159            agent: None,
160        };
161        Some(Asset {
162            id: AssetId::new_v7(),
163            handle: cookie_agent_protocol::AssetHandle::new("asset_00000000").ok()?,
164            store: AssetStore::Shared,
165            title: self.title.clone(),
166            media_type: media_type.clone(),
167            byte_length: 0,
168            sha256: sha256.clone(),
169            filename: Some(name.clone()).filter(|name| !name.is_empty()),
170            created: stamp.clone(),
171            updated: stamp,
172        })
173    }
174}

256/// Whether a file of `media_type` is text to show as its source: a page, a
257/// document, any `text/*`, JSON, XML, or YAML.
258pub fn is_text(media_type: &str) -> bool {
259    media_type.starts_with("text/")
260        || matches!(
261            media_type,
262            "application/json" | "application/xml" | "application/yaml" | "application/x-yaml"

264        || media_type.ends_with("+json")
265        || media_type.ends_with("+xml") && media_type != "image/svg+xml"
266}

333/// Reads the asset of `viewing`, a file of the media viewer not read yet,
334/// and shows it in the viewer's deck, and in the viewer while it still
335/// shows it.
336pub fn load_viewing(viewing: crate::media::Viewing, workspace: Workspace) {
337    let Some((store, id)) = viewing.asset.clone() else {
338        return;
339    };
340    spawn_local(async move {
341        let read = async {
342            let asset = workspace
343                .get_asset(store.clone(), id)
344                .await?
345                .ok_or_else(|| "that asset was deleted".to_owned())?;
346            content(&asset, &Source::Asset { store, asset: id }, workspace).await
347        }
348        .await;
349        let showing = workspace
350            .media_viewer
351            .with_untracked(|shown| shown.as_ref() == Some(&viewing));
352        match read {
353            Ok(content) => {
354                let filled = crate::media::Viewing {
355                    url: content.url,
356                    media_type: content.media_type,
357                    ..viewing.clone()
358                };
359                workspace.media_group.update(|group| {
360                    for shown in group.iter_mut().filter(|shown| **shown == viewing) {
361                        *shown = filled.clone();
362                    }
363                });
364                if showing {
365                    workspace.media_viewer.set(Some(filled));
366                }
367            }
368            Err(error) => {
369                workspace.status.set(error);
370                if showing {
371                    workspace.media_viewer.set(None);
372                }
373            }
374        }
375    });
376}

383/// Marks every asset card on the page whose asset has a live link among
384/// `shares` with `data-shared`, which keeps its share button shown, and
385/// unmarks the rest.
386pub fn mark_shared(shares: &[cookie_agent_protocol::Share]) {
387    let Some(document) = web_sys::window().and_then(|window| window.document()) else {
388        return;
389    };
390    let shared = shared_ids(shares);
391    let Ok(cards) = document.query_selector_all("[data-asset]") else {
392        return;
393    };
394    for index in 0..cards.length() {
395        if let Some(card) = cards
396            .item(index)
397            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
398        {
399            mark_card(&card, &shared);
400        }
401    }
402}

410    } else {

412    }

415/// Fills every asset placeholder under `root` not filled yet.
416pub fn show_within(root: &web_sys::Element, workspace: Workspace) {
417    listen();
418    let Ok(blocks) = root.query_selector_all("[data-asset]:not([data-shown])") else {
419        return;
420    };
421    for index in 0..blocks.length() {
422        let Some(element) = blocks
423            .item(index)
424            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
425        else {
426            continue;
427        };
428        let Some(shown) = placeholder(&element) else {
429            continue;
430        };
431        let _ = element.set_attribute("data-shown", "loading");
432        let shared =
433            leptos::prelude::WithUntracked::with_untracked(&workspace.share_list, |list| {
434                list.as_ref()
435                    .and_then(|list| list.as_ref().ok())
436                    .map(|list| shared_ids(&list.shares))
437                    .unwrap_or_default()
438            });
439        mark_card(&element, &shared);
440        spawn_local(fill(element, shown, workspace));
441    }
442}

444/// Brings the newest card of the asset `handle` into view and opens it as
445/// a click does, as a `#…&asset=` link asks; else its newest figure, shown
446/// full screen when it can be. It waits up to 30 s for the session's
447/// transcript to show it.
448pub fn reveal(handle: String, workspace: Workspace) {
449    spawn_local(async move {
450        let handle = handle.replace(['"', '\\'], "");
451        for _ in 0..120 {
452            gloo_timers::future::sleep(std::time::Duration::from_millis(250)).await;
453            if let Some(card) = card_of(&handle) {
454                card.scroll_into_view();
455                open_card(&card, None, workspace);
456                return;
457            }
458            let Some(figure) = last(&format!(
459                "#transcript [data-asset-handle=\"{handle}\"][data-asset-view=\"figure\"][data-shown=\"yes\"]"
460            )) else {
461                continue;
462            };
463            figure.scroll_into_view();
464            if figure
465                .query_selector("[data-action='asset-full']")
466                .ok()
467                .flatten()
468                .is_some()
469            {
470                full_screen(&figure);
471            } else if let Ok(Some(media)) = figure.query_selector(".media-open")
472                && let Some(media) = media.dyn_ref::<web_sys::HtmlElement>()
473            {
474                media.click();
475            }
476            return;
477        }
478    });
479}

498/// Runs a click on an asset's control, `target` the element whose
499/// `data-action` it is and `clicked` the element clicked:
500///
501/// - a figure's: `asset-full` shows it over the page and back,
502///   `asset-reload` runs its page again, `asset-show-all` uncuts its
503///   document;
504/// - a card's: `asset-panel` opens it in the side panel, `asset-deck` opens
505///   the media viewer at the card, to step through its gallery's pictures
506///   and videos, `asset-download`
507///   saves it, `asset-copy` copies its handle;
508/// - `asset-chip`, a handle in a reply, opens the newest card of its asset.
509pub fn act(
510    target: &web_sys::Element,
511    clicked: Option<&web_sys::Element>,
512    action: &str,
513    workspace: Workspace,
514) {
515    let card = || target.closest("[data-asset]").ok().flatten();
516    match action {
517        "asset-full" => {
518            if let Some(figure) = card() {
519                full_screen(&figure);
520            }
521        }
522        "asset-reload" => {
523            if let Some(figure) = card()
524                && let Ok(Some(frame)) = figure.query_selector("[data-asset-frame]")
525            {
526                // Loading the viewer again sends it the page again.
527                let _ = frame.set_attribute("src", "viewer.html");
528            }
529        }
530        "asset-show-all" => {
531            if let Some(figure) = card()
532                && let Ok(Some(document)) = figure.query_selector("[data-asset-document]")
533            {
534                uncut(&document);
535            }
536            target.remove();
537        }
538        "asset-panel" => {
539            if let Some(card) = card() {
540                open_card(&card, None, workspace);
541            }
542        }
543        "asset-deck" => {
544            let index = clicked
545                .unwrap_or(target)
546                .closest("[data-index]")
547                .ok()
548                .flatten()
549                .and_then(|card| card.get_attribute("data-index"))
550                .and_then(|index| index.parse().ok())
551                .unwrap_or(0);
552            if let Ok(Some(gallery)) = target.closest("[data-deck]") {
553                open_deck(&gallery, index, workspace);
554            }
555        }
556        "asset-download" => {
557            if let Some(card) = card()
558                && let Some(shown) = placeholder(&card)
559            {
560                download_source(workspace, shown.source, shown.title);
561            }
562        }
563        "attachment-allow" | "attachment-deny" => {
564            if let Some(key) = attach_key(target) {
565                workspace.decide_attachment(key, action == "attachment-allow");
566            }
567        }
568        // Saving asks where first: this session's assets or the project's.
569        "attachment-save" => {
570            if let Some(key) = target.get_attribute("data-attach-key") {
571                target.set_inner_html(&format!(
572                    "<span class=\"flex items-center gap-1\"><span role=\"button\" tabindex=\"0\" class=\"{BUTTON}\" data-action=\"attachment-save-session\" data-attach-key=\"{key}\">Save for this session</span><span role=\"button\" tabindex=\"0\" class=\"{BUTTON}\" data-action=\"attachment-save-project\" data-attach-key=\"{key}\">for the project</span></span>",
573                    key = escape(&key),
574                ));
575                let _ = target.set_attribute("data-action", "attachment-saving");
576                let _ = target.class_list().remove_1("opacity-0");
577            }
578        }
579        "attachment-save-session" | "attachment-save-project" => {
580            if let Some(key) = attach_key(target) {
581                let scope = if action == "attachment-save-project" {
582                    cookie_agent_protocol::AssetScope::Project
583                } else {
584                    cookie_agent_protocol::AssetScope::Session
585                };
586                workspace.save_attachment(key, scope);
587                if let Ok(Some(corner)) = target.closest("[data-action='attachment-saving']") {
588                    corner.remove();
589                }
590            }
591        }
592        "asset-copy" => {
593            if let Some(handle) = target.get_attribute("data-asset-handle") {
594                copy_handle(handle, workspace);
595            }
596        }
597        "asset-share" => {
598            if let Some(card) = card()
599                && let Some(Placeholder {
600                    source: Source::Asset { store, asset },
601                    title,
602                    ..
603                }) = placeholder(&card)
604            {
605                // The title the card shows now, which follows a rename.
606                let title = card
607                    .query_selector("[data-asset-title]")
608                    .ok()
609                    .flatten()
610                    .and_then(|title| title.text_content())
611                    .filter(|shown| !shown.trim().is_empty())
612                    .unwrap_or(title);
613                workspace.open_share_dialog(store, asset, title);
614            }
615        }
616        "attachment-share" => {
617            if let Some(key) = attach_key(target) {
618                workspace.share_attachment(key);
619            }
620        }
621        "asset-chip" => match target
622            .get_attribute("data-asset-handle")
623            .and_then(|handle| card_of(&handle))
624        {
625            Some(card) => open_card(&card, None, workspace),
626            None => workspace.status.set("that asset is not shown here".into()),
627        },
628        _ => {}
629    }
630}

632/// Opens the card `card` as a click on it does: a deck's at the card, a
633/// page or a document in the side panel; any other comes into view.
634fn open_card(card: &web_sys::Element, index: Option<usize>, workspace: Workspace) {

657/// Opens the deck `deck` in the media viewer at its card `index`, to step
658/// through every card filled.
659fn open_deck(deck: &web_sys::Element, index: usize, workspace: Workspace) {
660    let Ok(cards) = deck.query_selector_all("[data-asset][data-index]") else {
661        return;
662    };
663    let mut group = Vec::new();
664    let mut start = 0;
665    for at in 0..cards.length() {
666        let Some(card) = cards
667            .item(at)
668            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
669        else {
670            continue;
671        };
672        let content = card
673            .get_attribute("data-asset-current")
674            .filter(|_| card.get_attribute("data-shown").as_deref() == Some("yes"))
675            .and_then(|sha| CONTENTS.with(|contents| contents.borrow_mut().get(&sha)));
676        let Some(content) = content else {
677            continue;
678        };
679        let card_index: usize = card
680            .get_attribute("data-index")
681            .and_then(|index| index.parse().ok())
682            .unwrap_or(0);
683        group.push((
684            card_index,
685            crate::media::Viewing {
686                url: content.url,
687                media_type: content.media_type,
688                alt: card.get_attribute("data-asset-title").unwrap_or_default(),
689                handle: card.get_attribute("data-asset-handle"),
690                asset: match placeholder(&card).map(|shown| shown.source) {
691                    Some(Source::Asset { store, asset }) => Some((store, asset)),
692                    _ => None,
693                },
694            },
695        ));
696    }
697    // Drawn back to front: in the order shown again.
698    group.sort_by_key(|(card_index, _)| *card_index);
699    if let Some(at) = group
700        .iter()
701        .position(|(card_index, _)| *card_index == index)
702    {
703        start = at;
704    }
705    let group = group
706        .into_iter()
707        .map(|(_, viewing)| viewing)
708        .collect::<Vec<_>>();
709    let Some(first) = group.get(start).cloned() else {
710        workspace.status.set("the deck is still loading".into());
711        return;
712    };
713    workspace.media_group.set(group);
714    workspace.media_viewer.set(Some(first));
715}

789/// Reads every shown figure's asset again, as `assets.changed` asks.
790pub fn refresh(workspace: Workspace) {
791    let Some(document) = web_sys::window().and_then(|window| window.document()) else {
792        return;
793    };
794    let Ok(blocks) = document.query_selector_all("[data-asset][data-shown]") else {
795        return;
796    };
797    for index in 0..blocks.length() {
798        let Some(element) = blocks
799            .item(index)
800            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
801        else {
802            continue;
803        };
804        // A stored file never changes.
805        let Some(shown) = placeholder(&element) else {
806            continue;
807        };
808        let Source::Asset { store, asset } = shown.source.clone() else {
809            continue;
810        };
811        let current = element
812            .get_attribute("data-asset-current")
813            .unwrap_or_default();
814        spawn_local(async move {
815            match workspace.get_asset(store, asset).await {
816                Ok(Some(asset)) if asset.sha256 == current => {
817                    // The same content: only its title may have changed.
818                    set_head(&element, &asset, &shown);
819                }
820                Ok(_) => {
821                    let _ = element.set_attribute("data-shown", "loading");
822                    fill(element, shown, workspace).await;
823                }
824                Err(_) => {}
825            }
826        });
827    }
828}

830async fn fill(element: web_sys::Element, shown: Placeholder, workspace: Workspace) {
831    if shown.view == "tile" {
832        if let Source::Asset { store, asset } = &shown.source {
833            fill_tile(&element, &shown, store.clone(), *asset, workspace).await;
834        }
835        return;
836    }
837    let figure = shown.view == "figure";
838    let failed = |element: &web_sys::Element, error: String| {
839        body(element).set_text_content(Some(&format!("the asset could not be read: {error}")));
840        let _ = element.set_attribute("data-shown", "failed");
841    };
842    let asset = match &shown.source {
843        Source::Artifact { .. } => match shown.artifact_asset() {
844            Some(asset) => asset,
845            None => return failed(&element, "it names no file".into()),
846        },
847        Source::Asset { store, asset } => match workspace.get_asset(store.clone(), *asset).await {
848            Ok(Some(asset)) => asset,
849            Ok(None) if figure => {
850                gone(&element, &shown);
851                return;
852            }
853            Ok(None) => {
854                card_gone(&element);
855                return;
856            }
857            Err(error) => return failed(&element, error.to_string()),
858        },
859    };
860    // A file's chip shows what the asset is, not what it holds.
861    if shown.view == "file" {
862        let _ = element.set_attribute("data-asset-current", &asset.sha256);
863        card_state(&element, &asset, &shown);
864        let _ = element.set_attribute("data-shown", "yes");
865        return;
866    }
867    if shown.view == "doc"
868        && let Source::Asset { store, asset: id } = &shown.source
869    {
870        let start = match workspace
871            .read_asset_start(store.clone(), *id, EXCERPT_BYTES)
872            .await
873        {
874            Ok(Some(start)) => start,
875            Ok(None) => return card_gone(&element),
876            Err(error) => return failed(&element, error),
877        };
878        let _ = element.set_attribute("data-asset-current", &asset.sha256);
879        let target = body(&element);
880        target.set_inner_html("");
881        excerpt(&target, text_start(&start));
882        card_state(&element, &asset, &shown);
883        let _ = element.set_attribute("data-shown", "yes");
884        return;
885    }
886    let content = match content(&asset, &shown.source, workspace).await {
887        Ok(content) => content,
888        Err(error) => return failed(&element, error),
889    };
890    let _ = element.set_attribute("data-asset-current", &asset.sha256);
891    let target = body(&element);
892    target.set_inner_html("");
893    let _ = target.remove_attribute("data-shown");
894    match shown.view.as_str() {
895        "thumb" => thumb(&target, &content, &asset),
896        "page" => match &content.text {
897            Some(text) => frame(&target, text, Some(Fixed::Thumbnail)),
898            None => target.set_text_content(Some(&asset.title)),
899        },
900        "doc" => match &content.text {
901            Some(text) => excerpt(&target, text),
902            None => target.set_text_content(Some("too large to show")),
903        },
904        "audio" => crate::media::show(&target, &content.url, &content.media_type, &asset.title),
905        "panel" => panel(&target, &content, &asset),
906        "source" => {
907            let text = content
908                .text
909                .as_deref()
910                .unwrap_or("This file has no text to show.");
911            target.set_inner_html(&format!(
912                "<pre class=\"m-0 p-3 font-mono text-xs whitespace-pre-wrap wrap-anywhere\" data-asset-source>{}</pre>",
913                escape(text)
914            ));
915        }
916        _ => {
917            set_head(&element, &asset, &shown);
918            figure_body(&target, &content, &asset);
919        }
920    }
921    if !figure {
922        card_state(&element, &asset, &shown);
923    }
924    let _ = element.set_attribute("data-shown", "yes");
925}

996/// What this page or the browser knows of the content `sha256`'s
997/// thumbnail: `Some(Some(url))` its `Blob` URL, `Some(None)` that it has
998/// none, `None` nothing yet.
999async fn known_thumbnail(sha256: &str) -> Option<Option<String>> {
1000    if let Some(url) = THUMBNAILS.with(|thumbnails| thumbnails.borrow_mut().get(sha256)) {
1001        return Some(Some(url));
1002    }
1003    if NO_THUMBNAIL.with(|none| none.borrow().contains(sha256)) {
1004        return Some(None);
1005    }
1006    match crate::thumbnail_store::get(sha256).await? {
1007        crate::thumbnail_store::Kept::Thumbnail(blob) => {
1008            let url = web_sys::Url::create_object_url_with_blob(&blob).ok()?;
1009            Some(Some(remember_thumbnail(sha256, url, &blob)))
1010        }
1011        crate::thumbnail_store::Kept::None => {
1012            NO_THUMBNAIL.with(|none| none.borrow_mut().insert(sha256.to_owned()));
1013            Some(None)
1014        }
1015    }
1016}

1018/// The `Blob` URL of the daemon's `thumbnail` of the content `sha256`,
1019/// kept in this page and in the browser.
1020fn daemon_thumbnail(
1021    sha256: &str,
1022    thumbnail: &cookie_agent_protocol::AssetThumbnail,
1023) -> Option<String> {
1024    use base64::Engine as _;
1025    let bytes = base64::engine::general_purpose::STANDARD
1026        .decode(&thumbnail.data)
1027        .ok()?;
1028    let (blob, url) = blob_url(&bytes, &thumbnail.media_type).ok()?;
1029    crate::thumbnail_store::keep_thumbnail(sha256.to_owned(), blob.clone());
1030    Some(remember_thumbnail(sha256, url, &blob))
1031}

1033/// Keeps `url`, the `Blob` URL of `blob`, the content `sha256`'s
1034/// thumbnail, in this page while it fits, and gives it back.
1035fn remember_thumbnail(sha256: &str, url: String, blob: &web_sys::Blob) -> String {
1036    #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
1037    let bytes = blob.size() as u64;
1038    let left = THUMBNAILS.with(|thumbnails| {
1039        thumbnails
1040            .borrow_mut()
1041            .insert(sha256.to_owned(), url.clone(), bytes, |url| {
1042                url_on_page(url)
1043            })
1044    });
1045    for left in left {
1046        let _ = web_sys::Url::revoke_object_url(&left);
1047    }
1048    url
1049}

1327/// An HTML page in a frame of `viewer.html`, which gets the page once it
1328/// loads, with this frame's token.
1329fn frame(target: &web_sys::Element, text: &std::rc::Rc<str>, fixed: Option<Fixed>) {
1330    let Some(document) = target.owner_document() else {
1331        return;
1332    };
1333    let Ok(iframe) = document.create_element("iframe") else {
1334        return;
1335    };
1336    let token = uuid::Uuid::now_v7().simple().to_string();
1337    let start = format!("height: {FRAME_START}px");
1338    let thumbnail = thumbnail_style(target);
1339    let (class, style) = match fixed {
1340        None => (
1341            "block w-full rounded border border-line bg-white",
1342            start.as_str(),
1343        ),
1344        Some(Fixed::Thumbnail) => (
1345            "pointer-events-none block border-0 bg-white",

1347        ),
1348        Some(Fixed::Filling) => (
1349            "block min-h-0 w-full flex-1 border-0 bg-white",

1354        ("sandbox", "allow-scripts allow-popups"),
1355        ("referrerpolicy", "no-referrer"),
1356        ("src", "viewer.html"),
1357        ("title", "HTML page"),
1358        ("data-asset-frame", token.as_str()),
1359        ("class", class),
1360        ("style", style),
1361    ] {
1362        let _ = iframe.set_attribute(name, value);
1363    }
1364    if let Some(fixed) = fixed {
1365        let _ = iframe.set_attribute("data-fixed", "");
1366        if matches!(fixed, Fixed::Thumbnail) {
1367            let _ = iframe.set_attribute("tabindex", "-1");
1368            let _ = iframe.set_attribute("aria-hidden", "true");
1369            let _ = iframe.set_attribute("loading", "lazy");
1370        }
1371    }
1372    let html = std::rc::Rc::clone(text);
1373    let sent = token.clone();
1374    let frame = iframe.clone();
1375    let on_load = Closure::<dyn Fn()>::new(move || {
1376        send_page(&frame, &html, &sent);
1377    });
1378    let _ = iframe.add_event_listener_with_callback("load", on_load.as_ref().unchecked_ref());
1379    on_load.forget();
1380    FRAMES.with(|frames| frames.borrow_mut().insert(token, iframe.clone()));
1381    let _ = target.append_child(&iframe);
1382}

1465/// Sends the PDF `pdf` and pdf.js to the frame `frame`, with `token`.
1466async fn send_pdf(frame: web_sys::Element, pdf: web_sys::Blob, token: String) {
1467    let sent = async {
1468        let (library, worker) = pdf_library().await?;
1469        let window = content_window(&frame).ok_or_else(|| JsValue::from_str("no frame"))?;
1470        let dark = web_sys::window()
1471            .and_then(|window| window.document())
1472            .and_then(|document| document.document_element())
1473            .is_some_and(|root| root.class_list().contains("dark"));
1474        let message = js_sys::Object::new();
1475        for (name, value) in [
1476            ("type", JsValue::from_str("cookie-pdf")),
1477            ("token", JsValue::from_str(&token)),
1478            ("dark", JsValue::from_bool(dark)),
1479            ("pdf", pdf.into()),
1480            ("library", library.into()),
1481            ("worker", worker.into()),
1482        ] {
1483            js_sys::Reflect::set(&message, &name.into(), &value)?;
1484        }
1485        let post: js_sys::Function =
1486            js_sys::Reflect::get(&window, &"postMessage".into())?.dyn_into()?;
1487        post.call2(&window, &message, &"*".into())?;
1488        Ok::<(), JsValue>(())
1489    }
1490    .await;
1491    if sent.is_err() {
1492        frame.set_outer_html(&format!(
1493            "<span class=\"p-3 {NOTE}\">the PDF viewer could not be loaded</span>"
1494        ));
1495    }
1496}

1522/// Listens, once per page, for the height the frames' pages report. A
1523/// report counts only from the frame its token names.
1524fn listen() {
1525    if LISTENING.with(|listening| std::mem::replace(&mut *listening.borrow_mut(), true)) {
1526        return;
1527    }
1528    let Some(window) = web_sys::window() else {
1529        return;
1530    };
1531    let on_message =
1532        Closure::<dyn Fn(web_sys::MessageEvent)>::new(|event: web_sys::MessageEvent| {
1533            let data = event.data();
1534            let text = |name: &str| {
1535                js_sys::Reflect::get(&data, &name.into())
1536                    .ok()
1537                    .and_then(|value| value.as_string())
1538            };
1539            // A page's height.
1540            if text("type").as_deref() != Some("cookie-asset-height") {
1541                return;
1542            }
1543            let Some(token) = text("token") else {
1544                return;
1545            };
1546            let Some(height) = js_sys::Reflect::get(&data, &"height".into())
1547                .ok()
1548                .and_then(|value| value.as_f64())
1549                .filter(|number| number.is_finite() && *number > 0.0)
1550            else {
1551                return;
1552            };
1553            let Some(frame) = FRAMES.with(|frames| frames.borrow().get(&token).cloned()) else {
1554                return;
1555            };
1556            let source = event.source().map(JsValue::from);
1557            let from_frame = match (source, content_window(&frame)) {
1558                (Some(source), Some(window)) => js_sys::Object::is(&source, &window),
1559                _ => false,
1560            };
1561            // Each frame speaks only of its own kind.
1562            if !from_frame {
1563                return;
1564            }
1565            if frame.has_attribute("data-asset-frame") {
1566                size_frame(&frame, height);
1567            }
1568        });
1569    let _ = window.add_event_listener_with_callback("message", on_message.as_ref().unchecked_ref());
1570    on_message.forget();
1571}

1573/// The height a frame takes for content `reported` pixels tall in a window
1574/// `window_height` tall, or `None` when it stays at `current`.
1575pub fn frame_height(reported: f64, current: f64, window_height: f64) -> Option<f64> {
1576    let most = (window_height * FRAME_MAX_SHARE).max(FRAME_MIN);
1577    let height = reported.clamp(FRAME_MIN, most).round();
1578    ((height - current).abs() >= FRAME_TOLERANCE).then_some(height)
1579}

1590    let window_height = web_sys::window()
1591        .and_then(|window| window.inner_height().ok())
1592        .and_then(|height| height.as_f64())
1593        .unwrap_or(800.0);
1594    if let Some(height) = frame_height(reported, current, window_height) {
1595        let _ = element
1596            .style()
1597            .set_property("height", &format!("{height}px"));

1692/// Reads `asset`'s content whole, once per digest, as a `Blob` URL, with
1693/// its text when it is a document or a page.
1694async fn content(asset: &Asset, source: &Source, workspace: Workspace) -> Result<Content, String> {
1695    if let Some(content) = CONTENTS.with(|contents| contents.borrow_mut().get(&asset.sha256)) {
1696        return Ok(content);
1697    }
1698    let (bytes, sha256, media_type) = match source {
1699        Source::Asset { store, asset } => workspace.read_asset_bytes(store.clone(), *asset).await?,
1700        // A stored file's type is what its attachment found it to be.
1701        Source::Artifact {
1702            sha256, media_type, ..
1703        } => {
1704            let (bytes, _) = workspace.read_media(sha256.clone()).await?;
1705            (bytes, sha256.clone(), media_type.clone())
1706        }
1707    };
1708    // A page's text whatever its size; another text file's up to what a
1709    // document shows.
1710    let text = (media_type == "text/html"
1711        || (is_text(&media_type) && bytes.len() <= MAX_DOCUMENT_BYTES))
1712        .then(|| String::from_utf8(bytes.clone()).ok())
1713        .flatten()
1714        .map(std::rc::Rc::from);
1715    let (blob, url) =
1716        blob_url(&bytes, &media_type).map_err(|_| "the asset could not be shown".to_owned())?;
1717    let content = Content {
1718        url,
1719        blob,
1720        text,
1721        media_type,
1722    };
1723    let left = CONTENTS.with(|contents| {
1724        contents.borrow_mut().insert(
1725            sha256,
1726            content.clone(),
1727            (bytes.len() + content.text.as_ref().map_or(0, |text| text.len())) as u64,
1728            |content| url_on_page(&content.url),
1729        )
1730    });
1731    for left in left {
1732        let _ = web_sys::Url::revoke_object_url(&left.url);
1733    }
1734    Ok(content)
1735}
1736
1737/// A `Blob` of `bytes` and its URL.
1738fn blob_url(bytes: &[u8], media_type: &str) -> Result<(web_sys::Blob, String), JsValue> {
1739    let array = js_sys::Uint8Array::from(bytes);
1740    let parts = js_sys::Array::of1(&array);
1741    let options = web_sys::BlobPropertyBag::new();
1742    // An HTML page's blob is a download only, never a page of this origin.
1743    options.set_type(if media_type == "text/html" {
1744        "application/octet-stream"
1745    } else {
1746        media_type
1747    });
1748    let blob = web_sys::Blob::new_with_u8_array_sequence_and_options(&parts, &options)?;
1749    let url = web_sys::Url::create_object_url_with_blob(&blob)?;
1750    Ok((blob, url))
1751}

1753/// Saves `asset`'s current content as a file: read, then handed to the
1754/// browser's download.
1755pub fn download(workspace: Workspace, asset: Asset) {
1756    let source = Source::Asset {
1757        store: asset.store.clone(),
1758        asset: asset.id,
1759    };
1760    download_from(workspace, asset, source);
1761}
1762
1763/// Saves `asset`, read from `source`, as a file.
1764fn download_from(workspace: Workspace, asset: Asset, source: Source) {
1765    spawn_local(async move {
1766        let url = match CONTENTS.with(|contents| contents.borrow_mut().get(&asset.sha256)) {
1767            Some(content) => content.url,
1768            None => match content(&asset, &source, workspace).await {
1769                Ok(content) => content.url,
1770                Err(error) => {
1771                    workspace
1772                        .status
1773                        .set(format!("{} could not be read: {error}", asset.title));
1774                    return;
1775                }
1776            },
1777        };
1778        let Some(document) = web_sys::window().and_then(|window| window.document()) else {
1779            return;
1780        };
1781        let Ok(link) = document.create_element("a") else {
1782            return;
1783        };
1784        let _ = link.set_attribute("href", &url);
1785        let _ = link.set_attribute("download", &download_name(&asset));
1786        if let Some(link) = link.dyn_ref::<web_sys::HtmlElement>() {
1787            link.click();
1788        }
1789    });
1790}
