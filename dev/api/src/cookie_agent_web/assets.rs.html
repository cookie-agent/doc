1//! Assets on the page (`docs/site/specs/assets.md` section 8): each asset
2//! placeholder (`data-asset`) is filled with the asset as it is now, read
3//! with `asset.get` and `asset.read`, in the way its `data-asset-view`
4//! names:
5//!
6//! - `figure`, an `asset` block shown whole: images, SVG, video, and audio
7//!   as the media blocks show them, a Markdown document rendered by the
8//!   transcript's own renderer and cut to 480 px until *Show all*, an HTML
9//!   page in a sandboxed frame of `viewer.html` (section 8.2), which gets
10//!   the page by `postMessage` with a token and reports its height back
11//!   with it, anything else as a download card;
12//! - the shelf's cards (`crate::render::shelf`): `thumb` in a deck, `page`
13//!   a live thumbnail, `doc` the first lines, `audio` a player, and `file`
14//!   only the asset's state;
15//! - the side panel's `panel` (the asset whole, the page filling it) and
16//!   `source` (its text).
17//!
18//! `assets.changed` makes every one shown read its asset again: one whose
19//! content changed is filled anew and says when it was updated, one deleted
20//! says so.
21

62/// What a placeholder names: an asset, or a file an attachment stored in
63/// the session's artifacts; what it was when shown; and how it shows.
64#[derive(Clone)]
65struct Placeholder {
66    source: Source,
67    shown_sha: String,
68    title: String,
69    view: String,
70}

72#[derive(Clone, Debug, Eq, PartialEq)]
73pub enum Source {
74    /// An asset, shown as it is now.
75    Asset { store: AssetStore, asset: AssetId },
76    /// A stored file, which never changes.
77    Artifact {
78        sha256: String,
79        media_type: String,
80        name: String,
81    },
82}
83
84fn placeholder(element: &web_sys::Element) -> Option<Placeholder> {
85    let source = if element.get_attribute("data-asset")? == "artifact" {
86        Source::Artifact {
87            sha256: element.get_attribute("data-artifact-sha")?,
88            media_type: element.get_attribute("data-asset-type").unwrap_or_default(),
89            name: element.get_attribute("data-asset-name").unwrap_or_default(),
90        }
91    } else {
92        Source::Asset {
93            store: serde_json::from_str(&element.get_attribute("data-asset-store")?).ok()?,
94            asset: element.get_attribute("data-asset")?.parse().ok()?,
95        }
96    };
97    Some(Placeholder {
98        source,
99        shown_sha: element.get_attribute("data-asset-sha").unwrap_or_default(),
100        title: element
101            .get_attribute("data-asset-title")
102            .unwrap_or_default(),
103        view: element
104            .get_attribute("data-asset-view")
105            .unwrap_or_else(|| "figure".to_owned()),
106    })
107}

109impl Placeholder {
110    /// The stored file as the card code reads an asset: its title, type,
111    /// name, and content; it is never replaced.
112    fn artifact_asset(&self) -> Option<Asset> {
113        let Source::Artifact {
114            sha256,
115            media_type,
116            name,
117        } = &self.source
118        else {
119            return None;
120        };
121        let stamp = cookie_agent_protocol::AssetStamp {
122            at: jiff::Timestamp::now(),
123            session: cookie_agent_protocol::SessionId::new_v7(),
124            agent: None,
125        };
126        Some(Asset {
127            id: AssetId::new_v7(),
128            handle: cookie_agent_protocol::AssetHandle::new("asset_00000000").ok()?,
129            store: AssetStore::Shared,
130            title: self.title.clone(),
131            media_type: media_type.clone(),
132            byte_length: 0,
133            sha256: sha256.clone(),
134            filename: Some(name.clone()).filter(|name| !name.is_empty()),
135            created: stamp.clone(),
136            updated: stamp,
137        })
138    }
139}

141/// An asset open in the side panel: which, and what it was when the card
142/// that opened it showed it.
143#[derive(Clone, Debug, Eq, PartialEq)]
144pub struct OpenAsset {
145    pub source: Source,
146    /// The asset's handle; empty for a stored file.
147    pub handle: String,
148    pub title: String,
149    pub media_type: String,
150    pub shown_sha: String,
151}
152
153impl OpenAsset {
154    /// The asset, or stored file, the placeholder `element` shows.
155    pub fn of(element: &web_sys::Element) -> Option<Self> {
156        let shown = placeholder(element)?;
157        Some(Self {
158            source: shown.source,
159            handle: element
160                .get_attribute("data-asset-handle")
161                .unwrap_or_default(),
162            title: shown.title,
163            media_type: element.get_attribute("data-asset-type").unwrap_or_default(),
164            shown_sha: shown.shown_sha,
165        })
166    }
167
168    /// Whether it has text to show as its source: a page or a document.
169    pub fn has_source(&self) -> bool {
170        self.media_type == "text/html" || self.media_type == "text/markdown"
171    }
172
173    /// The side panel's placeholder of it: shown whole, or its `source`.
174    pub fn placeholder_html(&self, source: bool) -> String {
175        let view = if source { "source" } else { "panel" };
176        let named = match &self.source {
177            Source::Asset { store, asset } => format!(
178                "data-asset=\"{}\" data-asset-store=\"{}\" data-asset-handle=\"{}\"",
179                escape(&asset.to_string()),
180                escape(&serde_json::to_string(store).unwrap_or_default()),
181                escape(&self.handle),
182            ),
183            Source::Artifact { sha256, name, .. } => format!(
184                "data-asset=\"artifact\" data-artifact-sha=\"{}\" data-asset-name=\"{}\"",
185                escape(sha256),
186                escape(name),
187            ),
188        };
189        format!(
190            "<div class=\"flex min-h-full flex-col\" {named} data-asset-sha=\"{}\" data-asset-title=\"{}\" data-asset-type=\"{}\" data-asset-view=\"{view}\"><div class=\"flex min-h-0 flex-1 flex-col\" data-asset-body><span class=\"p-3 {NOTE}\">Loading…</span></div></div>",
191            escape(&self.shown_sha),
192            escape(&self.title),
193            escape(&self.media_type),
194        )
195    }
196}

198/// Fills every asset placeholder under `root` not filled yet.
199pub fn show_within(root: &web_sys::Element, workspace: Workspace) {
200    listen();
201    let Ok(blocks) = root.query_selector_all("[data-asset]:not([data-shown])") else {
202        return;
203    };
204    for index in 0..blocks.length() {
205        let Some(element) = blocks
206            .item(index)
207            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
208        else {
209            continue;
210        };
211        let Some(shown) = placeholder(&element) else {
212            continue;
213        };
214        let _ = element.set_attribute("data-shown", "loading");
215        spawn_local(fill(element, shown, workspace));
216    }
217}

219/// Brings the newest card of the asset `handle` into view and opens it as
220/// a click does, as a `#…&asset=` link asks; else its newest figure, shown
221/// full screen when it can be. It waits up to 30 s for the session's
222/// transcript to show it.
223pub fn reveal(handle: String, workspace: Workspace) {
224    spawn_local(async move {
225        let handle = handle.replace(['"', '\\'], "");
226        for _ in 0..120 {
227            gloo_timers::future::sleep(std::time::Duration::from_millis(250)).await;
228            if let Some(card) = card_of(&handle) {
229                card.scroll_into_view();
230                open_card(&card, None, workspace);
231                return;
232            }
233            let Some(figure) = last(&format!(
234                "#transcript [data-asset-handle=\"{handle}\"][data-asset-view=\"figure\"][data-shown=\"yes\"]"
235            )) else {
236                continue;
237            };
238            figure.scroll_into_view();
239            if figure
240                .query_selector("[data-action='asset-full']")
241                .ok()
242                .flatten()
243                .is_some()
244            {
245                full_screen(&figure);
246            } else if let Ok(Some(media)) = figure.query_selector(".media-open")
247                && let Some(media) = media.dyn_ref::<web_sys::HtmlElement>()
248            {
249                media.click();
250            }
251            return;
252        }
253    });
254}

273/// Runs a click on an asset's control, `target` the element whose
274/// `data-action` it is and `clicked` the element clicked:
275///
276/// - a figure's: `asset-full` shows it over the page and back,
277///   `asset-reload` runs its page again, `asset-show-all` uncuts its
278///   document;
279/// - a card's: `asset-panel` opens it in the side panel, `asset-deck` opens
280///   the deck in the media viewer at the card clicked, `asset-download`
281///   saves it, `asset-copy` copies its handle;
282/// - `asset-chip`, a handle in a reply, opens the newest card of its asset.
283pub fn act(
284    target: &web_sys::Element,
285    clicked: Option<&web_sys::Element>,
286    action: &str,
287    workspace: Workspace,
288) {
289    let card = || target.closest("[data-asset]").ok().flatten();
290    match action {
291        "asset-full" => {
292            if let Some(figure) = card() {
293                full_screen(&figure);
294            }
295        }
296        "asset-reload" => {
297            if let Some(figure) = card()
298                && let Ok(Some(frame)) = figure.query_selector("[data-asset-frame]")
299            {
300                // Loading the viewer again sends it the page again.
301                let _ = frame.set_attribute("src", "viewer.html");
302            }
303        }
304        "asset-show-all" => {
305            if let Some(figure) = card()
306                && let Ok(Some(document)) = figure.query_selector("[data-asset-document]")
307            {
308                uncut(&document);
309            }
310            target.remove();
311        }
312        "asset-panel" => {
313            if let Some(card) = card() {
314                open_card(&card, None, workspace);
315            }
316        }
317        "asset-deck" => {
318            let index = clicked
319                .and_then(|clicked| clicked.closest("[data-index]").ok().flatten())
320                .and_then(|card| card.get_attribute("data-index"))
321                .and_then(|index| index.parse().ok())
322                .unwrap_or(0);
323            open_deck(target, index, workspace);
324        }
325        "asset-download" => {
326            if let Some(card) = card()
327                && let Some(shown) = placeholder(&card)
328            {
329                download_source(workspace, shown.source, shown.title);
330            }
331        }
332        "attachment-allow" | "attachment-deny" => {
333            if let Some(key) = attach_key(target) {
334                workspace.decide_attachment(key, action == "attachment-allow");
335            }
336        }
337        // Saving asks where first: this session's assets or the project's.
338        "attachment-save" => {
339            if let Some(key) = target.get_attribute("data-attach-key") {
340                target.set_inner_html(&format!(
341                    "<span class=\"flex items-center gap-1\"><span role=\"button\" tabindex=\"0\" class=\"{BUTTON}\" data-action=\"attachment-save-session\" data-attach-key=\"{key}\">Save for this session</span><span role=\"button\" tabindex=\"0\" class=\"{BUTTON}\" data-action=\"attachment-save-project\" data-attach-key=\"{key}\">for the project</span></span>",
342                    key = escape(&key),
343                ));
344                let _ = target.set_attribute("data-action", "attachment-saving");
345                let _ = target.class_list().remove_1("opacity-0");
346            }
347        }
348        "attachment-save-session" | "attachment-save-project" => {
349            if let Some(key) = attach_key(target) {
350                let scope = if action == "attachment-save-project" {
351                    cookie_agent_protocol::AssetScope::Project
352                } else {
353                    cookie_agent_protocol::AssetScope::Session
354                };
355                workspace.save_attachment(key, scope);
356                if let Ok(Some(corner)) = target.closest("[data-action='attachment-saving']") {
357                    corner.remove();
358                }
359            }
360        }
361        "asset-copy" => {
362            if let Some(handle) = target.get_attribute("data-asset-handle") {
363                copy_handle(handle, workspace);
364            }
365        }
366        "asset-chip" => match target
367            .get_attribute("data-asset-handle")
368            .and_then(|handle| card_of(&handle))
369        {
370            Some(card) => open_card(&card, None, workspace),
371            None => workspace.status.set("that asset is not shown here".into()),
372        },
373        _ => {}
374    }
375}

377/// Opens the card `card` as a click on it does: a deck's at the card, a
378/// page or a document in the side panel; any other comes into view.
379fn open_card(card: &web_sys::Element, index: Option<usize>, workspace: Workspace) {
380    if card.get_attribute("data-shown").as_deref() == Some("deleted") {
381        workspace.status.set("that asset was deleted".into());
382        return;
383    }
384    match card.get_attribute("data-asset-view").as_deref() {
385        Some("thumb") => {
386            let index = index
387                .or_else(|| card.get_attribute("data-index")?.parse().ok())
388                .unwrap_or(0);
389            if let Ok(Some(deck)) = card.closest("[data-deck]") {
390                open_deck(&deck, index, workspace);
391            }
392        }
393        Some("page" | "doc") => workspace.asset_panel.set(OpenAsset::of(card)),
394        _ => {
395            let options = web_sys::ScrollIntoViewOptions::new();
396            options.set_block(web_sys::ScrollLogicalPosition::Center);
397            card.scroll_into_view_with_scroll_into_view_options(&options);
398        }
399    }
400}

402/// Opens the deck `deck` in the media viewer at its card `index`, to step
403/// through every card filled.
404fn open_deck(deck: &web_sys::Element, index: usize, workspace: Workspace) {
405    let Ok(cards) = deck.query_selector_all("[data-asset][data-index]") else {
406        return;
407    };
408    let mut group = Vec::new();
409    let mut start = 0;
410    for at in 0..cards.length() {
411        let Some(card) = cards
412            .item(at)
413            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
414        else {
415            continue;
416        };
417        let content = card
418            .get_attribute("data-asset-current")
419            .filter(|_| card.get_attribute("data-shown").as_deref() == Some("yes"))
420            .and_then(|sha| CONTENTS.with(|contents| contents.borrow().get(&sha).cloned()));
421        let Some(content) = content else {
422            continue;
423        };
424        let card_index: usize = card
425            .get_attribute("data-index")
426            .and_then(|index| index.parse().ok())
427            .unwrap_or(0);
428        group.push((
429            card_index,
430            crate::media::Viewing {
431                url: content.url,
432                media_type: content.media_type,
433                alt: card.get_attribute("data-asset-title").unwrap_or_default(),
434                handle: card.get_attribute("data-asset-handle"),
435            },
436        ));
437    }
438    // Drawn back to front: in the order shown again.
439    group.sort_by_key(|(card_index, _)| *card_index);
440    if let Some(at) = group
441        .iter()
442        .position(|(card_index, _)| *card_index == index)
443    {
444        start = at;
445    }
446    let group = group
447        .into_iter()
448        .map(|(_, viewing)| viewing)
449        .collect::<Vec<_>>();
450    let Some(first) = group.get(start).cloned() else {
451        workspace.status.set("the deck is still loading".into());
452        return;
453    };
454    workspace.media_group.set(group);
455    workspace.media_viewer.set(Some(first));
456}

501/// Saves what `source` names as a file: an asset as it is now, or a
502/// stored file.
503pub fn download_source(workspace: Workspace, source: Source, title: String) {
504    if let Source::Artifact { .. } = &source {
505        let shown = Placeholder {
506            source: source.clone(),
507            shown_sha: String::new(),
508            title,
509            view: "file".into(),
510        };
511        if let Some(asset) = shown.artifact_asset() {
512            download_from(workspace, asset, source);
513        }
514        return;
515    }
516    let Source::Asset { store, asset } = source else {
517        return;
518    };
519    spawn_local(async move {
520        match workspace.get_asset(store, asset).await {
521            Ok(Some(asset)) => download(workspace, asset),
522            Ok(None) => workspace.status.set("that asset was deleted".into()),
523            Err(error) => workspace
524                .status
525                .set(format!("the asset could not be read: {error}")),
526        }
527    });
528}

530/// Reads every shown figure's asset again, as `assets.changed` asks.
531pub fn refresh(workspace: Workspace) {
532    let Some(document) = web_sys::window().and_then(|window| window.document()) else {
533        return;
534    };
535    let Ok(blocks) = document.query_selector_all("[data-asset][data-shown]") else {
536        return;
537    };
538    for index in 0..blocks.length() {
539        let Some(element) = blocks
540            .item(index)
541            .and_then(|node| node.dyn_into::<web_sys::Element>().ok())
542        else {
543            continue;
544        };
545        // A stored file never changes.
546        let Some(shown) = placeholder(&element) else {
547            continue;
548        };
549        let Source::Asset { store, asset } = shown.source.clone() else {
550            continue;
551        };
552        let current = element
553            .get_attribute("data-asset-current")
554            .unwrap_or_default();
555        spawn_local(async move {
556            match workspace.get_asset(store, asset).await {
557                Ok(Some(asset)) if asset.sha256 == current => {
558                    // The same content: only its title may have changed.
559                    set_head(&element, &asset, &shown);
560                }
561                Ok(_) => {
562                    let _ = element.set_attribute("data-shown", "loading");
563                    fill(element, shown, workspace).await;
564                }
565                Err(_) => {}
566            }
567        });
568    }
569}

571async fn fill(element: web_sys::Element, shown: Placeholder, workspace: Workspace) {
572    let figure = shown.view == "figure";
573    let failed = |element: &web_sys::Element, error: String| {
574        body(element).set_text_content(Some(&format!("the asset could not be read: {error}")));
575        let _ = element.set_attribute("data-shown", "failed");
576    };
577    let asset = match &shown.source {
578        Source::Artifact { .. } => match shown.artifact_asset() {
579            Some(asset) => asset,
580            None => return failed(&element, "it names no file".into()),
581        },
582        Source::Asset { store, asset } => match workspace.get_asset(store.clone(), *asset).await {
583            Ok(Some(asset)) => asset,
584            Ok(None) if figure => {
585                gone(&element, &shown);
586                return;
587            }
588            Ok(None) => {
589                card_gone(&element);
590                return;
591            }
592            Err(error) => return failed(&element, error.to_string()),
593        },
594    };
595    // A file's chip shows what the asset is, not what it holds.
596    if shown.view == "file" {
597        let _ = element.set_attribute("data-asset-current", &asset.sha256);
598        card_state(&element, &asset, &shown);
599        let _ = element.set_attribute("data-shown", "yes");
600        return;
601    }
602    let content = match content(&asset, &shown.source, workspace).await {
603        Ok(content) => content,
604        Err(error) => return failed(&element, error),
605    };
606    let _ = element.set_attribute("data-asset-current", &asset.sha256);
607    let target = body(&element);
608    target.set_inner_html("");
609    let _ = target.remove_attribute("data-shown");
610    match shown.view.as_str() {
611        "thumb" => thumb(&target, &content, &asset),
612        "page" => match &content.text {
613            Some(text) => frame(&target, text, Some(Fixed::Thumbnail)),
614            None => target.set_text_content(Some(&asset.title)),
615        },
616        "doc" => match &content.text {
617            Some(text) => excerpt(&target, text),
618            None => target.set_text_content(Some("too large to show")),
619        },
620        "audio" => crate::media::show(&target, &content.url, &content.media_type, &asset.title),
621        "panel" => panel(&target, &content, &asset),
622        "source" => {
623            let text = content
624                .text
625                .as_deref()
626                .unwrap_or("This file has no text to show.");
627            target.set_inner_html(&format!(
628                "<pre class=\"m-0 p-3 font-mono text-xs whitespace-pre-wrap wrap-anywhere\" data-asset-source>{}</pre>",
629                escape(text)
630            ));
631        }
632        _ => {
633            set_head(&element, &asset, &shown);
634            figure_body(&target, &content, &asset);
635        }
636    }
637    if !figure {
638        card_state(&element, &asset, &shown);
639    }
640    let _ = element.set_attribute("data-shown", "yes");
641}

643/// A figure's body: the asset whole, as its type shows.
644fn figure_body(target: &web_sys::Element, content: &Content, asset: &Asset) {
645    let media = &content.media_type;
646    if media.starts_with("image/") || media.starts_with("video/") || media.starts_with("audio/") {
647        crate::media::show(target, &content.url, media, &asset.title);
648    } else if media == "text/markdown" {
649        match &content.text {
650            Some(text) if text.len() <= MAX_DOCUMENT_BYTES => document(target, text),
651            _ => download_card(target, asset, &content.url, Some("too large to show")),
652        }
653    } else if media == "text/html" {
654        match &content.text {
655            Some(text) => frame(target, text, None),
656            None => download_card(target, asset, &content.url, None),
657        }
658    } else {
659        download_card(target, asset, &content.url, None);
660    }
661}

888/// A frame whose size its page does not set.
889#[derive(Clone, Copy)]
890enum Fixed {
891    /// A card's thumbnail: the page laid out at 960 px wide, shown at a
892    /// quarter of its size, taking no input.
893    Thumbnail,
894    /// Filling the side panel.
895    Filling,

898/// An HTML page in a frame of `viewer.html`, which gets the page once it
899/// loads, with this frame's token.
900fn frame(target: &web_sys::Element, text: &std::rc::Rc<str>, fixed: Option<Fixed>) {
901    let Some(document) = target.owner_document() else {
902        return;
903    };
904    let Ok(iframe) = document.create_element("iframe") else {
905        return;
906    };
907    let token = uuid::Uuid::now_v7().simple().to_string();
908    let start = format!("height: {FRAME_START}px");
909    let (class, style) = match fixed {
910        None => (
911            "block w-full rounded border border-line bg-white",
912            start.as_str(),
913        ),
914        Some(Fixed::Thumbnail) => (
915            "pointer-events-none block border-0 bg-white",
916            "width: 960px; height: 528px; transform: scale(0.25); transform-origin: 0 0",
917        ),
918        Some(Fixed::Filling) => (
919            "block min-h-0 w-full flex-1 border-0 bg-white",
920            "height: 100%",
921        ),
922    };
923    for (name, value) in [
924        ("sandbox", "allow-scripts allow-popups"),
925        ("referrerpolicy", "no-referrer"),
926        ("src", "viewer.html"),
927        ("title", "HTML page"),
928        ("data-asset-frame", token.as_str()),
929        ("class", class),
930        ("style", style),
931    ] {
932        let _ = iframe.set_attribute(name, value);
933    }
934    if let Some(fixed) = fixed {
935        let _ = iframe.set_attribute("data-fixed", "");
936        if matches!(fixed, Fixed::Thumbnail) {
937            let _ = iframe.set_attribute("tabindex", "-1");
938            let _ = iframe.set_attribute("aria-hidden", "true");
939            let _ = iframe.set_attribute("loading", "lazy");
940        }
941    }
942    let html = std::rc::Rc::clone(text);
943    let sent = token.clone();
944    let frame = iframe.clone();
945    let on_load = Closure::<dyn Fn()>::new(move || {
946        send_page(&frame, &html, &sent);
947    });
948    let _ = iframe.add_event_listener_with_callback("load", on_load.as_ref().unchecked_ref());
949    on_load.forget();
950    FRAMES.with(|frames| frames.borrow_mut().insert(token, iframe.clone()));
951    let _ = target.append_child(&iframe);
952}
953
954fn send_page(frame: &web_sys::Element, html: &str, token: &str) {
955    let Some(window) = content_window(frame) else {
956        return;
957    };
958    let message = js_sys::Object::new();
959    let _ = js_sys::Reflect::set(&message, &"type".into(), &"cookie-asset-page".into());
960    let _ = js_sys::Reflect::set(&message, &"token".into(), &token.into());
961    let _ = js_sys::Reflect::set(&message, &"html".into(), &html.into());
962    let post = js_sys::Reflect::get(&window, &"postMessage".into())
963        .ok()
964        .and_then(|post| post.dyn_into::<js_sys::Function>().ok());
965    if let Some(post) = post {
966        let _ = post.call2(&window, &message, &"*".into());
967    }
968}

970/// A frame's window, as a plain object: cross-origin, so only messages
971/// reach it.
972fn content_window(frame: &web_sys::Element) -> Option<JsValue> {
973    js_sys::Reflect::get(frame, &"contentWindow".into())
974        .ok()
975        .filter(|window| !window.is_null() && !window.is_undefined())
976}

978/// Listens, once per page, for the height the frames' pages report. A
979/// report counts only from the frame its token names.
980fn listen() {
981    if LISTENING.with(|listening| std::mem::replace(&mut *listening.borrow_mut(), true)) {
982        return;
983    }
984    let Some(window) = web_sys::window() else {
985        return;
986    };
987    let on_message =
988        Closure::<dyn Fn(web_sys::MessageEvent)>::new(|event: web_sys::MessageEvent| {
989            let data = event.data();
990            let text = |name: &str| {
991                js_sys::Reflect::get(&data, &name.into())
992                    .ok()
993                    .and_then(|value| value.as_string())
994            };
995            if text("type").as_deref() != Some("cookie-asset-height") {
996                return;
997            }
998            let Some(token) = text("token") else {
999                return;
1000            };
1001            let Some(height) = js_sys::Reflect::get(&data, &"height".into())
1002                .ok()
1003                .and_then(|value| value.as_f64())
1004                .filter(|height| height.is_finite() && *height > 0.0)
1005            else {
1006                return;
1007            };
1008            let Some(frame) = FRAMES.with(|frames| frames.borrow().get(&token).cloned()) else {
1009                return;
1010            };
1011            let source = event.source().map(JsValue::from);
1012            let from_frame = match (source, content_window(&frame)) {
1013                (Some(source), Some(window)) => js_sys::Object::is(&source, &window),
1014                _ => false,
1015            };
1016            if from_frame {
1017                size_frame(&frame, height);
1018            }
1019        });
1020    let _ = window.add_event_listener_with_callback("message", on_message.as_ref().unchecked_ref());
1021    on_message.forget();
1022}

1024/// The height a frame takes for content `reported` pixels tall in a window
1025/// `window_height` tall, or `None` when it stays at `current`.
1026pub fn frame_height(reported: f64, current: f64, window_height: f64) -> Option<f64> {
1027    let most = (window_height * FRAME_MAX_SHARE).max(FRAME_MIN);
1028    let height = reported.clamp(FRAME_MIN, most).round();
1029    ((height - current).abs() >= FRAME_TOLERANCE).then_some(height)
1030}
1031
1032fn size_frame(frame: &web_sys::Element, reported: f64) {
1033    // A frame shown full screen fills it, and a fixed one keeps its size.
1034    if frame.has_attribute("data-fixed") || frame.closest("#asset-full").ok().flatten().is_some() {
1035        return;
1036    }
1037    let Some(element) = frame.dyn_ref::<web_sys::HtmlElement>() else {
1038        return;
1039    };
1040    let current = f64::from(element.offset_height());
1041    let window_height = web_sys::window()
1042        .and_then(|window| window.inner_height().ok())
1043        .and_then(|height| height.as_f64())
1044        .unwrap_or(800.0);
1045    if let Some(height) = frame_height(reported, current, window_height) {
1046        let _ = element
1047            .style()
1048            .set_property("height", &format!("{height}px"));
1049    }
1050}

1052/// Shows `figure`'s asset over the whole page: an HTML page in a frame of
1053/// its own, which runs it again, or the whole document. *Close* or Escape
1054/// ends it.
1055fn full_screen(figure: &web_sys::Element) {
1056    let Some(document) = figure.owner_document() else {
1057        return;
1058    };
1059    let Some(sha256) = figure.get_attribute("data-asset-current") else {
1060        return;
1061    };
1062    let Some(content) = CONTENTS.with(|contents| contents.borrow().get(&sha256).cloned()) else {
1063        return;
1064    };
1065    let Some(text) = content.text.clone() else {
1066        return;
1067    };
1068    close_full_screen(&document);
1069    let title = figure
1070        .query_selector("[data-asset-title]")
1071        .ok()
1072        .flatten()
1073        .and_then(|title| title.text_content())
1074        .unwrap_or_default();
1075    let Ok(overlay) = document.create_element("div") else {
1076        return;
1077    };
1078    let _ = overlay.set_attribute("id", "asset-full");
1079    let _ = overlay.set_attribute("role", "dialog");
1080    let _ = overlay.set_attribute("aria-label", &title);
1081    let _ = overlay.set_attribute(
1082        "class",
1083        "fixed inset-0 z-50 flex flex-col gap-2 bg-surface p-3 text-muted",
1084    );
1085    overlay.set_inner_html(&format!(
1086        "<div class=\"flex min-w-0 items-center gap-2\"><span class=\"min-w-0 flex-1 truncate text-sm font-bold\">{}</span><button type=\"button\" id=\"asset-full-close\" class=\"{BUTTON}\" title=\"Close (Escape)\">{}<span>Close</span></button></div><div class=\"min-h-0 flex-1 overflow-auto\" data-asset-body></div>",
1087        escape(&title),
1088        crate::icons::svg(icondata::LuX, "")
1089    ));
1090    let body = body(&overlay);
1091    if content.media_type == "text/html" {
1092        frame(&body, &text, None);
1093        if let Ok(Some(iframe)) = body.query_selector("iframe")
1094            && let Some(iframe) = iframe.dyn_ref::<web_sys::HtmlElement>()
1095        {
1096            let _ = iframe.style().set_property("height", "100%");
1097        }
1098    } else {
1099        body.set_inner_html(&format!(
1100            "<div class=\"markdown mx-auto max-w-3xl\">{}</div>",
1101            crate::render::markdown::markdown_html(&text)
1102        ));
1103        crate::highlight::highlight_within(&body);
1104    }
1105    let close = Closure::<dyn Fn()>::new(move || {
1106        if let Some(document) = web_sys::window().and_then(|window| window.document()) {
1107            close_full_screen(&document);
1108        }
1109    });
1110    if let Ok(Some(button)) = overlay.query_selector("#asset-full-close") {
1111        let _ = button.add_event_listener_with_callback("click", close.as_ref().unchecked_ref());
1112    }
1113    close.forget();
1114    let on_key = Closure::<dyn Fn(web_sys::KeyboardEvent)>::new(|event: web_sys::KeyboardEvent| {
1115        if event.key() == "Escape"
1116            && let Some(document) = web_sys::window().and_then(|window| window.document())
1117        {
1118            close_full_screen(&document);
1119        }
1120    });
1121    let _ = overlay.add_event_listener_with_callback("keydown", on_key.as_ref().unchecked_ref());
1122    on_key.forget();
1123    let _ = overlay.set_attribute("tabindex", "-1");
1124    if let Some(root) = document.body() {
1125        let _ = root.append_child(&overlay);
1126    }
1127    if let Some(overlay) = overlay.dyn_ref::<web_sys::HtmlElement>() {
1128        let _ = overlay.focus();
1129    }
1130}
1131
1132fn close_full_screen(document: &web_sys::Document) {
1133    if let Some(overlay) = document.get_element_by_id("asset-full") {
1134        if let Ok(Some(frame)) = overlay.query_selector("[data-asset-frame]")
1135            && let Some(token) = frame.get_attribute("data-asset-frame")
1136        {
1137            FRAMES.with(|frames| frames.borrow_mut().remove(&token));
1138        }
1139        overlay.remove();
1140    }
1141}

1143/// Reads `asset`'s content whole, once per digest, as a `Blob` URL, with
1144/// its text when it is a document or a page.
1145async fn content(asset: &Asset, source: &Source, workspace: Workspace) -> Result<Content, String> {
1146    if let Some(content) = CONTENTS.with(|contents| contents.borrow().get(&asset.sha256).cloned()) {
1147        return Ok(content);
1148    }
1149    let (bytes, sha256, media_type) = match source {
1150        Source::Asset { store, asset } => workspace.read_asset_bytes(store.clone(), *asset).await?,
1151        // A stored file's type is what its attachment found it to be.
1152        Source::Artifact {
1153            sha256, media_type, ..
1154        } => {
1155            let (bytes, _) = workspace.read_media(sha256.clone()).await?;
1156            (bytes, sha256.clone(), media_type.clone())
1157        }
1158    };
1159    let text = (media_type == "text/markdown" || media_type == "text/html")
1160        .then(|| String::from_utf8(bytes.clone()).ok())
1161        .flatten()
1162        .map(std::rc::Rc::from);
1163    let url =
1164        blob_url(&bytes, &media_type).map_err(|_| "the asset could not be shown".to_owned())?;
1165    let content = Content {
1166        url,
1167        text,
1168        media_type,
1169    };
1170    CONTENTS.with(|contents| contents.borrow_mut().insert(sha256, content.clone()));
1171    Ok(content)
1172}
1173
1174fn blob_url(bytes: &[u8], media_type: &str) -> Result<String, JsValue> {
1175    let array = js_sys::Uint8Array::from(bytes);
1176    let parts = js_sys::Array::of1(&array);
1177    let options = web_sys::BlobPropertyBag::new();
1178    // An HTML page's blob is a download only, never a page of this origin.
1179    options.set_type(if media_type == "text/html" {
1180        "application/octet-stream"
1181    } else {
1182        media_type
1183    });
1184    let blob = web_sys::Blob::new_with_u8_array_sequence_and_options(&parts, &options)?;
1185    web_sys::Url::create_object_url_with_blob(&blob)
1186}

1188/// Saves `asset`'s current content as a file: read, then handed to the
1189/// browser's download.
1190pub fn download(workspace: Workspace, asset: Asset) {
1191    let source = Source::Asset {
1192        store: asset.store.clone(),
1193        asset: asset.id,
1194    };
1195    download_from(workspace, asset, source);
1196}
1197
1198/// Saves `asset`, read from `source`, as a file.
1199fn download_from(workspace: Workspace, asset: Asset, source: Source) {
1200    spawn_local(async move {
1201        let url = match CONTENTS.with(|contents| contents.borrow().get(&asset.sha256).cloned()) {
1202            Some(content) => content.url,
1203            None => match content(&asset, &source, workspace).await {
1204                Ok(content) => content.url,
1205                Err(error) => {
1206                    workspace
1207                        .status
1208                        .set(format!("{} could not be read: {error}", asset.title));
1209                    return;
1210                }
1211            },
1212        };
1213        let Some(document) = web_sys::window().and_then(|window| window.document()) else {
1214            return;
1215        };
1216        let Ok(link) = document.create_element("a") else {
1217            return;
1218        };
1219        let _ = link.set_attribute("href", &url);
1220        let _ = link.set_attribute("download", &download_name(&asset));
1221        if let Some(link) = link.dyn_ref::<web_sys::HtmlElement>() {
1222            link.click();
1223        }
1224    });
1225}
