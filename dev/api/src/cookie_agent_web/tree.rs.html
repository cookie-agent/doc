1//! The subagents nested under their session in the sessions list: which
2//! rows the open session's tree shows, collapsed or expanded, the title
3//! bar's path to a viewed subagent, and the subagent a delegate call
4//! started.
5//!
6//! Only the open session's row has a tree. Collapsed (the default), it
7//! shows the subagents that run, at any depth, with their finished
8//! ancestors dimmed so each sits in its place, and the subagent the page
9//! shows; with nothing running it shows nothing. Expanded, it shows the
10//! whole hierarchy. A click on the open session's row switches between the
11//! two, and the browser remembers the choice per session.
12

19/// The `localStorage` key of whether `session_id`'s tree is expanded.
20pub fn expanded_key(session_id: SessionId) -> String {
21    format!("cookie.tree:{session_id}")
22}
23
24/// Whether the value stored under [`expanded_key`] expands the tree: only
25/// `expanded` does, so a tree starts collapsed.
26pub fn expanded_from_storage(stored: Option<&str>) -> bool {
27    stored == Some("expanded")
28}

30/// One row of the tree.
31#[derive(Clone, Debug, PartialEq)]
32pub struct TreeRow {
33    pub session_id: SessionId,
34    pub depth: usize,
35    /// The agent's name.
36    pub agent: String,
37    /// What it was asked: the session's title.
38    pub task: String,
39    /// The short handle the model refers to it by.
40    pub handle: Option<String>,
41    pub status: SessionStatus,
42    pub has_children: bool,
43    pub collapsed: bool,
44    /// Shown only to place a running subagent under it: it does not run
45    /// itself, and the tree is collapsed.
46    pub dimmed: bool,
47}
48
49/// The rows of the expanded `tree`, depth first, without the children of
50/// the sessions the user folded.
51pub fn tree_rows(tree: &SessionTree, collapsed: &HashSet<SessionId>) -> Vec<TreeRow> {
52    let mut rows = Vec::new();
53    push_rows(tree, 0, collapsed, &mut rows);
54    rows
55}
56
57fn push_rows(
58    tree: &SessionTree,
59    depth: usize,
60    collapsed: &HashSet<SessionId>,
61    rows: &mut Vec<TreeRow>,
62) {
63    let session_id = tree.session.session_id;
64    let folded = collapsed.contains(&session_id);
65    rows.push(row(tree, depth, folded, false));
66    if !folded {
67        for child in &tree.children {
68            push_rows(child, depth + 1, collapsed, rows);
69        }
70    }
71}
72
73fn row(tree: &SessionTree, depth: usize, collapsed: bool, dimmed: bool) -> TreeRow {
74    TreeRow {
75        session_id: tree.session.session_id,
76        depth,
77        agent: tree.session.creation_selection.agent.to_string(),
78        task: task(&tree.session),
79        handle: tree.session.short_id.clone(),
80        status: tree.session.status,
81        has_children: !tree.children.is_empty(),
82        collapsed,
83        dimmed,
84    }
85}

87/// The rows of the collapsed `tree`, depth first: the subagents that run
88/// and `viewed` (the session the page shows), each under its ancestors,
89/// which are dimmed unless they run (or are viewed) too. The root's row is
90/// first when anything shows; nothing shows when nothing runs.
91pub fn running_rows(tree: &SessionTree, viewed: Option<SessionId>) -> Vec<TreeRow> {
92    let mut rows = Vec::new();
93    push_running(tree, 0, viewed, &mut rows);
94    rows
95}
96
97/// Pushes `tree`'s rows when it, or a session below it, runs or is
98/// viewed; whether it pushed any.
99fn push_running(
100    tree: &SessionTree,
101    depth: usize,
102    viewed: Option<SessionId>,
103    rows: &mut Vec<TreeRow>,
104) -> bool {
105    let at = rows.len();
106    let shown_below = tree.children.iter().fold(Vec::new(), |mut below, child| {
107        push_running(child, depth + 1, viewed, &mut below);
108        below
109    });
110    // The root's own state does not count: its row is the session's.
111    let shown_itself = depth > 0
112        && (tree.session.status == SessionStatus::Running
113            || viewed == Some(tree.session.session_id));
114    if !shown_itself && shown_below.is_empty() {
115        return false;
116    }
117    rows.insert(at, row(tree, depth, false, !shown_itself && depth > 0));
118    rows.extend(shown_below);
119    true
120}

122fn task(session: &SessionMeta) -> String {

126        .map_or_else(|| "untitled".to_owned(), ToString::to_string)

129/// A subagent's name in the title bar: its agent and its task.
130pub fn subagent_label(session: &SessionMeta) -> String {
131    format!("{}: {}", session.creation_selection.agent, task(session))

134/// The sessions from `tree`'s root down to `session_id`, both included;
135/// empty when the tree does not hold it.
136pub fn path(tree: &SessionTree, session_id: SessionId) -> Vec<&SessionMeta> {
137    if tree.session.session_id == session_id {
138        return vec![&tree.session];
139    }
140    for child in &tree.children {
141        let mut below = path(child, session_id);
142        if !below.is_empty() {
143            below.insert(0, &tree.session);
144            return below;
145        }
146    }
147    Vec::new()
148}

150/// The subagents `parent` started, by the delegate call that started each.
151pub fn delegates_of(tree: &SessionTree, parent: SessionId) -> Vec<(ToolCallId, &SessionMeta)> {
152    let mut found = Vec::new();
153    collect_delegates(tree, parent, &mut found);
154    found
155}
156
157fn collect_delegates<'a>(
158    tree: &'a SessionTree,
159    parent: SessionId,
160    found: &mut Vec<(ToolCallId, &'a SessionMeta)>,
161) {
162    for child in &tree.children {
163        if let SessionOrigin::Delegated {
164            parent_session_id,
165            parent_tool_call_id,
166            ..
167        } = &child.session.origin
168            && *parent_session_id == parent
169        {
170            found.push((*parent_tool_call_id, &child.session));
171        }
172        collect_delegates(child, parent, found);
173    }
174}

176/// What a running subagent is doing, as a delegate call's row says it.
177#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
178pub enum Activity {
179    /// An agent's question waits for the user's answer.
180    WaitingForAnswer,
181    WaitingForApproval,
182    /// Running a tool call.
183    RunningTool,
184    /// Its model is answering.
185    Thinking,
186}
187
188impl Activity {
189    /// What the subagent's session state says it does: a question waiting
190    /// for the user outranks an approval, which outranks a running tool,
191    /// which outranks the run itself. `None` for a subagent that is not
192    /// running.
193    pub fn of(
194        running: bool,
195        tool_running: bool,
196        approval_waiting: bool,
197        question_waiting: bool,
198    ) -> Option<Self> {
199        if question_waiting {
200            Some(Self::WaitingForAnswer)
201        } else if approval_waiting {
202            Some(Self::WaitingForApproval)
203        } else if tool_running {
204            Some(Self::RunningTool)
205        } else if running {
206            Some(Self::Thinking)
207        } else {
208            None
209        }
210    }
211
212    pub const fn label(self) -> &'static str {
213        match self {
214            Self::WaitingForAnswer => "waiting for your answer",
215            Self::WaitingForApproval => "waiting for approval",
216            Self::RunningTool => "running a tool",
217            Self::Thinking => "thinking",
218        }
219    }
220}

222/// The metadata of `session_id` in `tree`.
223pub fn find(tree: &SessionTree, session_id: SessionId) -> Option<&SessionMeta> {
224    if tree.session.session_id == session_id {
225        return Some(&tree.session);
226    }
227    tree.children
228        .iter()
229        .find_map(|child| find(child, session_id))
230}
231
232/// Whether `session_id` is `tree`'s root or one of its descendants.
233pub fn contains(tree: &SessionTree, session_id: SessionId) -> bool {
234    tree.session.session_id == session_id
235        || tree
236            .children
237            .iter()
238            .any(|child| contains(child, session_id))
239}

291        let tree = node(

332    fn a_collapsed_tree_shows_running_subagents_under_dimmed_ancestors() {
333        // primary ─ planner (completed) ─ coder (running)
334        //         │                     └ tester (completed)
335        //         ├ reviewer (failed)
336        //         └ scout (running) ─ reader (completed)
337        let tree = node(
338            "primary",
339            SessionStatus::Idle,
340            vec![
341                node(
342                    "planner",
343                    SessionStatus::Completed,
344                    vec![
345                        node("coder", SessionStatus::Running, Vec::new()),
346                        node("tester", SessionStatus::Completed, Vec::new()),
347                    ],
348                ),
349                node("reviewer", SessionStatus::Failed, Vec::new()),
350                node(
351                    "scout",
352                    SessionStatus::Running,
353                    vec![node("reader", SessionStatus::Completed, Vec::new())],
354                ),
355            ],
356        );
357        let rows = running_rows(&tree, None);
358        assert_eq!(rows[0].depth, 0, "the root leads");
359        assert_eq!(
360            shown(&rows),
361            [
362                (1, "planner", true),
363                (2, "coder", false),
364                (1, "scout", false),
365            ]
366        );
367        // The viewed session shows too, though it finished.
368        let reviewer = tree.children[1].session.session_id;
369        assert_eq!(
370            shown(&running_rows(&tree, Some(reviewer))),
371            [
372                (1, "planner", true),
373                (2, "coder", false),
374                (1, "reviewer", false),
375                (1, "scout", false),
376            ]
377        );
378        // Expanded, everything shows and nothing is dimmed.
379        assert_eq!(
380            shown(&tree_rows(&tree, &HashSet::new())),
381            [
382                (1, "planner", false),
383                (2, "coder", false),
384                (2, "tester", false),
385                (1, "reviewer", false),
386                (1, "scout", false),
387                (2, "reader", false),
388            ]
389        );
390    }

423    /// `child` as delegated by `parent`'s call `call`.
424    fn delegated(mut child: SessionTree, parent: &SessionTree, call: &str) -> SessionTree {
425        child.session.origin = serde_json::from_value(serde_json::json!({
426            "type": "delegated",
427            "root_session_id": parent.session.session_id,
428            "parent_session_id": parent.session.session_id,
429            "parent_run_id": uuid::Uuid::now_v7(),
430            "parent_tool_call_id": call,
431            "invocation_id": uuid::Uuid::now_v7(),
432            "depth": 1,
433        }))
434        .expect("a delegated origin");
435        child
436    }

438    #[test]
439    fn the_path_leads_from_the_root_to_a_subagent() {
440        let mut tree = node("primary", SessionStatus::Idle, Vec::new());
441        let mut worker = node("worker", SessionStatus::Running, Vec::new());
442        worker.session.title =
443            Some(cookie_agent_protocol::SessionTitle::new("check tests").expect("a title"));
444        let helper = node("helper", SessionStatus::Idle, Vec::new());
445        let helper_id = helper.session.session_id;
446        worker.children.push(helper);
447        tree.children.push(worker);
448        let labels = path(&tree, helper_id)
449            .into_iter()
450            .map(|session| session.creation_selection.agent.to_string())
451            .collect::<Vec<_>>();
452        assert_eq!(labels, ["primary", "worker", "helper"]);
453        assert_eq!(
454            subagent_label(&tree.children[0].session),
455            "worker: check tests"
456        );
457        assert_eq!(path(&tree, tree.session.session_id).len(), 1);
458        assert!(path(&tree, SessionId::new_v7()).is_empty());
459    }

461    #[test]
462    fn a_delegate_call_finds_the_subagent_it_started() {
463        let (call_7, call_9) = (
464            uuid::Uuid::now_v7().to_string(),
465            uuid::Uuid::now_v7().to_string(),
466        );
467        let mut tree = node("primary", SessionStatus::Idle, Vec::new());
468        let child = delegated(
469            node("worker", SessionStatus::Running, Vec::new()),
470            &tree,
471            &call_7,
472        );
473        let grandchild = delegated(
474            node("helper", SessionStatus::Idle, Vec::new()),
475            &child,
476            &call_9,
477        );
478        let mut child = child;
479        child.children.push(grandchild);
480        let child_id = child.session.session_id;
481        tree.children.push(child);
482        let found = delegates_of(&tree, tree.session.session_id);
483        assert_eq!(found.len(), 1);
484        assert_eq!(found[0].0.to_string(), call_7);
485        assert_eq!(found[0].1.session_id, child_id);
486        assert_eq!(delegates_of(&tree, child_id)[0].0.to_string(), call_9);
487    }
