1//! Fills a transcript row's code and media as it nears the view. Opening a
2//! session draws all its rows; highlighting and fetching for every one of them
3//! would freeze the first paint and read every image of the session, and most
4//! of those rows are never scrolled to.
5
6use std::cell::RefCell;
7use std::rc::Rc;
8
9use wasm_bindgen::{JsCast as _, closure::Closure};
10use web_sys::{IntersectionObserver, IntersectionObserverEntry, IntersectionObserverInit};
11
12/// How far from the view a row starts filling: about two screens, so that it
13/// is ready before scrolling reaches it.
14const MARGIN: &str = "1600px 0px";
15
16/// Watches the rows of one scrolling transcript.
17struct NearView {
18    observer: IntersectionObserver,
19    root: web_sys::Element,
20    _callback: Closure<dyn Fn(js_sys::Array, IntersectionObserver)>,
21}
22
23thread_local! {
24    /// The transcript on the page: the page shows one at a time.
25    static CURRENT: RefCell<Option<Rc<NearView>>> = const { RefCell::new(None) };
26}
27
28/// Starts watching for `root`, the scrolling transcript, replacing any before.
29/// `fill` runs on each watched element as it comes near, once per [`watch`].
30pub fn start(root: &web_sys::Element, fill: impl Fn(&web_sys::Element) + 'static) {
31    let callback = Closure::<dyn Fn(js_sys::Array, IntersectionObserver)>::new(
32        move |entries: js_sys::Array, observer: IntersectionObserver| {
33            for entry in entries.iter() {
34                let Ok(entry) = entry.dyn_into::<IntersectionObserverEntry>() else {
35                    continue;
36                };
37                if entry.is_intersecting() {
38                    let target = entry.target();
39                    observer.unobserve(&target);
40                    fill(&target);
41                }
42            }
43        },
44    );
45    let options = IntersectionObserverInit::new();
46    options.set_root(Some(root));
47    options.set_root_margin(MARGIN);
48    let Ok(observer) =
49        IntersectionObserver::new_with_options(callback.as_ref().unchecked_ref(), &options)
50    else {
51        return;
52    };
53    let view = Rc::new(NearView {
54        observer,
55        root: root.clone(),
56        _callback: callback,
57    });
58    if let Some(before) = CURRENT.with(|current| current.borrow_mut().replace(view)) {
59        before.observer.disconnect();
60    }
61}
62
63/// Stops watching for `root`, if it is the transcript watched.
64pub fn stop(root: &web_sys::Element) {
65    let stopped = CURRENT.with(|current| {
66        let mut current = current.borrow_mut();
67        if current.as_ref().is_some_and(|view| view.root == *root) {
68            current.take()
69        } else {
70            None
71        }
72    });
73    if let Some(view) = stopped {
74        view.observer.disconnect();
75    }
76}
77
78/// Has `element` filled when it comes near, at once if it is near already.
79/// Returns false when no transcript on the page holds it, for the caller to
80/// fill it now.
81pub fn watch(element: &web_sys::Element) -> bool {
82    CURRENT.with(|current| match current.borrow().as_ref() {
83        Some(view) if view.root.contains(Some(element)) => {
84            view.observer.observe(element);
85            true
86        }
87        _ => false,
88    })
89}
