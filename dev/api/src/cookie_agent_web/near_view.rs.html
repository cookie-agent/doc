1//! Fills a transcript row's code and media as it nears the view. Opening a
2//! session draws all its rows; highlighting and fetching for every one of them
3//! would freeze the first paint and read every image of the session, and most
4//! of those rows are never scrolled to. Within a row that came near, a card
5//! that reads its file whole (a picture, a video, a player) waits again until
6//! the card itself is within a screen of the view
7//! (`docs/site/specs/attachment-handling.md` section 5.1).
8
9use std::cell::RefCell;
10use std::rc::Rc;
11
12use wasm_bindgen::{JsCast as _, closure::Closure};
13use web_sys::{IntersectionObserver, IntersectionObserverEntry, IntersectionObserverInit};
14
15/// How far from the view a row starts filling: about two screens, so that it
16/// is ready before scrolling reaches it.
17const MARGIN: &str = "1600px 0px";
18
19/// How far from the view a card starts reading its file: one screen.
20const CARD_MARGIN: &str = "100% 0px";
21
22type Callback = Closure<dyn Fn(js_sys::Array, IntersectionObserver)>;
23
24/// Watches the rows, and the cards, of one scrolling transcript.
25struct NearView {
26    observer: IntersectionObserver,
27    cards: IntersectionObserver,
28    root: web_sys::Element,
29    _callbacks: (Callback, Callback),
30}
31
32thread_local! {
33    /// The transcript on the page: the page shows one at a time.
34    static CURRENT: RefCell<Option<Rc<NearView>>> = const { RefCell::new(None) };
35}
36
37/// An observer of `root` with `margin` that runs `fill` on each element
38/// it watches as it comes near, once.
39fn observer(
40    root: &web_sys::Element,
41    margin: &str,
42    fill: impl Fn(&web_sys::Element) + 'static,
43) -> Option<(IntersectionObserver, Callback)> {
44    let callback = Callback::new(
45        move |entries: js_sys::Array, observer: IntersectionObserver| {
46            for entry in entries.iter() {
47                let Ok(entry) = entry.dyn_into::<IntersectionObserverEntry>() else {
48                    continue;
49                };
50                if entry.is_intersecting() {
51                    let target = entry.target();
52                    observer.unobserve(&target);
53                    fill(&target);
54                }
55            }
56        },
57    );
58    let options = IntersectionObserverInit::new();
59    options.set_root(Some(root));
60    options.set_root_margin(margin);
61    let observer =
62        IntersectionObserver::new_with_options(callback.as_ref().unchecked_ref(), &options).ok()?;
63    Some((observer, callback))
64}

66/// Starts watching for `root`, the scrolling transcript, replacing any before.
67/// `fill` runs on each watched row as it comes near, once per [`watch`];
68/// `fill_card` on each watched card, once per [`watch_card`].
69pub fn start(
70    root: &web_sys::Element,
71    fill: impl Fn(&web_sys::Element) + 'static,
72    fill_card: impl Fn(&web_sys::Element) + 'static,
73) {
74    let (Some((observer, rows)), Some((cards, card_callback))) = (
75        observer(root, MARGIN, fill),
76        observer(root, CARD_MARGIN, fill_card),
77    ) else {
78        return;
79    };
80    let view = Rc::new(NearView {
81        observer,
82        cards,
83        root: root.clone(),
84        _callbacks: (rows, card_callback),
85    });
86    if let Some(before) = CURRENT.with(|current| current.borrow_mut().replace(view)) {
87        before.observer.disconnect();
88        before.cards.disconnect();
89    }
90}
91
92/// Stops watching for `root`, if it is the transcript watched.
93pub fn stop(root: &web_sys::Element) {
94    let stopped = CURRENT.with(|current| {
95        let mut current = current.borrow_mut();
96        if current.as_ref().is_some_and(|view| view.root == *root) {
97            current.take()
98        } else {
99            None
100        }
101    });
102    if let Some(view) = stopped {
103        view.observer.disconnect();
104        view.cards.disconnect();
105    }
106}

108/// Has `element` filled when it comes near, at once if it is near already.
109/// Returns false when no transcript on the page holds it, for the caller to
110/// fill it now.
111pub fn watch(element: &web_sys::Element) -> bool {
112    CURRENT.with(|current| match current.borrow().as_ref() {
113        Some(view) if view.root.contains(Some(element)) => {
114            view.observer.observe(element);
115            true
116        }
117        _ => false,
118    })
119}
120
121/// [`watch`] for a card, which fills within a screen of the view.
122pub fn watch_card(element: &web_sys::Element) -> bool {
123    CURRENT.with(|current| match current.borrow().as_ref() {
124        Some(view) if view.root.contains(Some(element)) => {
125            view.cards.observe(element);
126            true
127        }
128        _ => false,
129    })
