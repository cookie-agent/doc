1//! What a key does in the composer, as in the TUI: Enter steers, Alt+Enter
2//! queues, Shift+Enter, Ctrl+Enter, and Ctrl+J insert a newline, Escape
3//! cancels, and Up in an empty composer recalls this client's newest
4//! pending or queued message. Also which buttons the composer shows, and
5//! how tall its text grows.
6
7/// One key press in the composer: the `KeyboardEvent.key` and modifiers.
8#[derive(Clone, Copy, Debug, Default)]
9pub struct KeyPress<'a> {
10    pub key: &'a str,
11    pub alt: bool,
12    pub shift: bool,
13    pub ctrl: bool,
14    pub meta: bool,
15    /// The composer holds no text.
16    pub empty: bool,
17}
18
19/// What the composer does with a key.
20#[derive(Clone, Copy, Debug, Eq, PartialEq)]
21pub enum ComposerAction {
22    /// Submit, steering a running run.
23    Steer,
24    /// Submit to the queue behind a running run.
25    Queue,
26    /// Insert a newline.
27    Newline,
28    /// Cancel the running run.
29    Cancel,
30    /// Take back this client's newest pending or queued message.
31    Recall,
32    /// The browser handles the key.
33    Type,
34}
35
36/// The composer's action for `press`.
37pub fn composer_action(press: KeyPress<'_>) -> ComposerAction {
38    match press.key {
39        "Enter" if press.alt => ComposerAction::Queue,
40        "Enter" if press.shift || press.ctrl || press.meta => ComposerAction::Newline,
41        "Enter" => ComposerAction::Steer,
42        "j" | "J" if press.ctrl && !press.alt && !press.meta => ComposerAction::Newline,
43        "Escape" => ComposerAction::Cancel,
44        "ArrowUp" if press.empty && !press.alt && !press.shift && !press.ctrl => {
45            ComposerAction::Recall
46        }
47        _ => ComposerAction::Type,
48    }
49}

51/// The buttons at the composer's bottom right, by whether the session runs
52/// (a run or a compaction) and whether there is anything to send.
53#[derive(Clone, Copy, Debug, Eq, PartialEq)]
54pub enum ComposerButtons {
55    /// Send (Enter): starts a run; disabled with nothing to send.
56    Send { enabled: bool },
57    /// Stop (Escape): cancels the run or the compaction.
58    Stop,
59    /// Steer (Enter) and Queue (Alt+Enter), side by side.
60    SteerAndQueue,
61    /// Resume (Enter) in place of Send while the session is held
62    /// (`docs/site/specs/held-sessions.md`): sends the draft, if any, with
63    /// what waits.
64    Resume,
65}
66
67/// The composer's buttons for a session that is `running`, a draft that
68/// `has_input` (text or attached files), and a session that is `held`.
69pub const fn composer_buttons(running: bool, has_input: bool, held: bool) -> ComposerButtons {
70    match (running, has_input) {
71        (false, _) if held => ComposerButtons::Resume,
72        (false, has_input) => ComposerButtons::Send { enabled: has_input },
73        (true, false) => ComposerButtons::Stop,
74        (true, true) => ComposerButtons::SteerAndQueue,
75    }
76}

78/// The most lines the composer grows to before it scrolls.
79pub const COMPOSER_MAX_LINES: u32 = 5;
80
81/// The composer's height for text whose content is `content` px tall
82/// (`scrollHeight` with the box at its smallest), with lines of `line` px
83/// and `padding` px above and below them together: one line at least,
84/// [`COMPOSER_MAX_LINES`] at most. The flag says the text scrolls inside.
85pub fn composer_height(content: f64, line: f64, padding: f64) -> (f64, bool) {
86    let smallest = line + padding;
87    let largest = line * f64::from(COMPOSER_MAX_LINES) + padding;
88    (content.clamp(smallest, largest), content > largest + 0.5)
89}

91#[cfg(test)]
92mod tests {
93    use super::{
94        COMPOSER_MAX_LINES, ComposerAction, ComposerButtons, KeyPress, composer_action,
95        composer_buttons, composer_height,
96    };
97
98    #[test]
99    fn the_buttons_follow_the_run_and_the_draft() {
100        assert_eq!(
101            composer_buttons(false, false, false),
102            ComposerButtons::Send { enabled: false }
103        );
104        assert_eq!(
105            composer_buttons(false, true, false),
106            ComposerButtons::Send { enabled: true }
107        );
108        assert_eq!(composer_buttons(true, false, false), ComposerButtons::Stop);
109        assert_eq!(
110            composer_buttons(true, true, false),
111            ComposerButtons::SteerAndQueue
112        );
113        // A held session resumes, with a draft or without; a running one
114        // still stops.
115        assert_eq!(
116            composer_buttons(false, false, true),
117            ComposerButtons::Resume
118        );
119        assert_eq!(composer_buttons(false, true, true), ComposerButtons::Resume);
120        assert_eq!(composer_buttons(true, false, true), ComposerButtons::Stop);
121    }
122
123    #[test]
124    fn the_composer_grows_from_one_line_to_five_then_scrolls() {
125        let (line, padding) = (20.0, 24.0);
126        assert_eq!(composer_height(0.0, line, padding), (44.0, false));
127        assert_eq!(composer_height(44.0, line, padding), (44.0, false));
128        assert_eq!(composer_height(84.0, line, padding), (84.0, false));
129        let five = line * f64::from(COMPOSER_MAX_LINES) + padding;
130        assert_eq!(composer_height(five, line, padding), (five, false));
131        assert_eq!(composer_height(five + 20.0, line, padding), (five, true));
132    }
133
134    fn press(key: &str) -> KeyPress<'_> {
135        KeyPress {
136            key,
137            ..KeyPress::default()
138        }
139    }
140
141    #[test]
142    fn keys_map_as_in_the_tui() {
143        assert_eq!(composer_action(press("Enter")), ComposerAction::Steer);
144        assert_eq!(
145            composer_action(KeyPress {
146                alt: true,
147                ..press("Enter")
148            }),
149            ComposerAction::Queue
150        );
151        for newline in [
152            KeyPress {
153                shift: true,
154                ..press("Enter")
155            },
156            KeyPress {
157                ctrl: true,
158                ..press("Enter")
159            },
160            KeyPress {
161                ctrl: true,
162                ..press("j")
163            },
164        ] {
165            assert_eq!(composer_action(newline), ComposerAction::Newline);
166        }
167        assert_eq!(composer_action(press("Escape")), ComposerAction::Cancel);
168        assert_eq!(
169            composer_action(KeyPress {
170                empty: true,
171                ..press("ArrowUp")
172            }),
173            ComposerAction::Recall
174        );
175        // Up in a draft moves the caret.
176        assert_eq!(composer_action(press("ArrowUp")), ComposerAction::Type);
177        assert_eq!(composer_action(press("j")), ComposerAction::Type);
178        assert_eq!(composer_action(press("a")), ComposerAction::Type);
179    }
