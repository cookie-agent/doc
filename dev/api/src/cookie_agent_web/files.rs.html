1//! The browser side of attaching files: the hidden file inputs `/attach`
2//! and the camera open, reading a `File`, and decoding and re-encoding an
3//! image through a canvas for a downscale.
4
5use wasm_bindgen::{JsCast as _, JsValue};
6use wasm_bindgen_futures::JsFuture;
7
8/// The file input `/attach` opens.
9pub const FILE_INPUT: &str = "attach-input";
10/// The file input that asks a phone for its camera.
11pub const CAMERA_INPUT: &str = "camera-input";
12
13/// Opens the file picker, or the camera on a phone.
14pub fn pick_files(camera: bool) {
15    let id = if camera { CAMERA_INPUT } else { FILE_INPUT };
16    if let Some(input) = web_sys::window()
17        .and_then(|window| window.document())
18        .and_then(|document| document.get_element_by_id(id))
19        .and_then(|element| element.dyn_into::<web_sys::HtmlElement>().ok())
20    {
21        input.click();
22    }
23}

25/// The files of a file list, in order.
26pub fn files_of(list: Option<web_sys::FileList>) -> Vec<web_sys::File> {
27    let Some(list) = list else {
28        return Vec::new();
29    };
30    (0..list.length())
31        .filter_map(|index| list.get(index))
32        .collect()
33}

35/// A file's bytes.
36pub async fn read_file(file: &web_sys::File) -> Result<Vec<u8>, String> {
37    let buffer = JsFuture::from(file.array_buffer())
38        .await
39        .map_err(describe)?;
40    Ok(js_sys::Uint8Array::new(&buffer).to_vec())
41}

43/// `bytes` of `media_type` decoded as an image.
44pub async fn decode_image(
45    bytes: &[u8],
46    media_type: &str,
47) -> Result<web_sys::HtmlImageElement, String> {
48    let array = js_sys::Uint8Array::from(bytes);
49    let options = web_sys::BlobPropertyBag::new();
50    options.set_type(media_type);
51    let blob = web_sys::Blob::new_with_u8_array_sequence_and_options(
52        &js_sys::Array::of1(&array),
53        &options,
54    )
55    .map_err(describe)?;
56    let url = web_sys::Url::create_object_url_with_blob(&blob).map_err(describe)?;
57    let image = web_sys::HtmlImageElement::new().map_err(describe)?;
58    image.set_src(&url);
59    let decoded = JsFuture::from(image.decode()).await;
60    let _ = web_sys::Url::revoke_object_url(&url);
61    decoded.map_err(describe)?;
62    Ok(image)
63}

65/// `image` drawn at `width`×`height` and encoded as `encoding`.
66pub fn encode_image(
67    image: &web_sys::HtmlImageElement,
68    (width, height): (u32, u32),
69    encoding: &str,
70) -> Result<Vec<u8>, String> {
71    let document = web_sys::window()
72        .and_then(|window| window.document())
73        .ok_or("no document")?;
74    let canvas = document
75        .create_element("canvas")
76        .map_err(describe)?
77        .dyn_into::<web_sys::HtmlCanvasElement>()
78        .map_err(|_| "no canvas")?;
79    canvas.set_width(width);
80    canvas.set_height(height);
81    let context = canvas
82        .get_context("2d")
83        .map_err(describe)?
84        .ok_or("no 2d canvas")?
85        .dyn_into::<web_sys::CanvasRenderingContext2d>()
86        .map_err(|_| "no 2d canvas")?;
87    context
88        .draw_image_with_html_image_element_and_dw_and_dh(
89            image,
90            0.0,
91            0.0,
92            f64::from(width),
93            f64::from(height),
94        )
95        .map_err(describe)?;
96    let url = canvas
97        .to_data_url_with_type_and_encoder_options(encoding, &JsValue::from_f64(0.85))
98        .map_err(describe)?;
99    let data = url
100        .split_once(',')
101        .map(|(_, data)| data)
102        .ok_or("the canvas gave no data")?;
103    use base64::Engine as _;
104    base64::engine::general_purpose::STANDARD
105        .decode(data)
106        .map_err(|error| error.to_string())
107}
108
109fn describe(error: JsValue) -> String {
110    error
111        .dyn_ref::<js_sys::Error>()
112        .map(|error| String::from(error.message()))
113        .or_else(|| error.as_string())
114        .unwrap_or_else(|| "the browser refused".into())
115}
