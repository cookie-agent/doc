1//! Opening the daemon connection from the page, with the tab's identity.
2
3use cookie_agent_protocol::{
4    Client, ClientError, ClientIdentity, ClientKind, HandshakeToken, ServerHello,
5};

7use crate::{browser, connect::IDENTITY_KEY};

9/// This tab's client identity: the one kept in `sessionStorage`, or a new
10/// one kept there. Every reconnect of the tab presents it, so its events
11/// keep one origin; another tab is another client.
12pub fn tab_identity() -> ClientIdentity {
13    if let Some(identity) = browser::load_tab(IDENTITY_KEY)
14        .and_then(|stored| serde_json::from_str::<ClientIdentity>(&stored).ok())
15    {
16        return identity;
17    }
18    let identity =
19        ClientIdentity::generate(ClientKind::new("web").expect("`web` is a valid client kind"));
20    if let Ok(stored) = serde_json::to_string(&identity) {
21        browser::store_tab(IDENTITY_KEY, &stored);
22    }
23    identity
24}

26/// Connects to the daemon at `url` and authenticates as `identity` with
27/// `token` in the handshake, as a browser must.
28pub async fn open(
29    url: &str,
30    token: &str,
31    identity: ClientIdentity,
32) -> Result<(Client, ServerHello), ClientError> {
33    let client = connect(url).await?;
34    let hello = client
35        .handshake_with_token(identity, HandshakeToken::new(token))
36        .await?;
37    Ok((client, hello))
38}

43}
