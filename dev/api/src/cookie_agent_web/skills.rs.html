1//! `/skills`: which skills a search shows, ranked as the TUI ranks them.
2
3use cookie_agent_protocol::SkillDescriptor;
4
5/// The skills a user can run that match `query`: by name (exact, prefix,
6/// then anywhere), then by description, best first.
7pub fn matching_skills<'a>(skills: &'a [SkillDescriptor], query: &str) -> Vec<&'a SkillDescriptor> {
8    let query = query.trim().to_lowercase();
9    let mut ranked = skills
10        .iter()
11        .filter(|skill| skill.precedence_winner && skill.user_invocable)
12        .filter_map(|skill| {
13            let name = skill.name.as_str();
14            let rank = if query.is_empty() || name == query {
15                Some(0)
16            } else if name.starts_with(&query) {
17                Some(1)
18            } else if name.contains(&query) {
19                Some(2)
20            } else {
21                skill
22                    .description
23                    .to_lowercase()
24                    .contains(&query)
25                    .then_some(3)
26            };
27            rank.map(|rank| (rank, skill))
28        })
29        .collect::<Vec<_>>();
30    ranked.sort_by_key(|(rank, _)| *rank);
31    ranked.into_iter().map(|(_, skill)| skill).collect()
32}

34#[cfg(test)]
35mod tests {
36    use cookie_agent_protocol::SkillDescriptor;
37
38    use super::matching_skills;
39
40    fn skill(name: &str, description: &str, user_invocable: bool) -> SkillDescriptor {
41        serde_json::from_value(serde_json::json!({
42            "name": name,
43            "description": description,
44            "when_to_use": null,
45            "location": "/skills",
46            "source": "project",
47            "precedence_winner": true,
48            "permission_effect": "allow",
49            "visible": true,
50            "user_invocable": user_invocable,
51            "argument_hint": null,
52        }))
53        .expect("skill")
54    }
55
56    #[test]
57    fn names_rank_before_descriptions_and_hidden_skills_never_show() {
58        let skills = [
59            skill("summarize", "write a review summary", true),
60            skill("review", "review the diff", true),
61            skill("preview", "render a page", true),
62            skill("reviewer", "internal", false),
63        ];
64        let names = |query| {
65            matching_skills(&skills, query)
66                .into_iter()
67                .map(|skill| skill.name.as_str())
68                .collect::<Vec<_>>()
69        };
70        assert_eq!(names("review"), ["review", "preview", "summarize"]);
71        assert_eq!(names(""), ["summarize", "review", "preview"]);
72        assert!(names("nothing").is_empty());
73    }
