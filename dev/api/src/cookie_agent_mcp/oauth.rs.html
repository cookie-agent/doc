165pub(crate) fn auth_challenge(error: &(dyn std::error::Error + 'static)) -> Option<String> {
166    use rmcp::transport::streamable_http_client::{AuthRequiredError, InsufficientScopeError};
167
168    let mut source = Some(error);
169    while let Some(current) = source {
170        if let Some(required) = current.downcast_ref::<AuthRequiredError>() {
171            return Some(required.www_authenticate_header.clone());
172        }
173        if let Some(scope) = current.downcast_ref::<InsufficientScopeError>() {
174            return Some(scope.www_authenticate_header.clone());
175        }
176        source = current.source();
177    }
178    None
179}
180
181fn oauth_requires_auth(error: &(dyn std::error::Error + 'static)) -> bool {
182    if auth_challenge(error).is_some() {
183        return true;
184    }
185    let mut source = Some(error);
186    while let Some(current) = source {
187        if let Some(auth) = current.downcast_ref::<rmcp::transport::AuthError>()
188            && matches!(
189                auth,
190                rmcp::transport::AuthError::AuthorizationRequired
191                    | rmcp::transport::AuthError::TokenExpired
192                    | rmcp::transport::AuthError::TokenRefreshFailed(_)
193                    | rmcp::transport::AuthError::TokenRefreshRejected(_)
194            )
195        {
196            return true;
197        }
198        source = current.source();
199    }
200    false
201}

203pub(crate) fn initialize_error_requires_auth(error: &rmcp::service::ClientInitializeError) -> bool {
204    error.is_authorization_required()
205        || match error {
206            rmcp::service::ClientInitializeError::TransportError { error, .. } => {
207                oauth_requires_auth(error.error.as_ref())
208            }
209            _ => false,
210        }

262    fn update(
263        &self,
264        update: impl FnOnce(&mut BTreeMap<String, PersistedOAuthCredential>),
265    ) -> Result<(), ()> {
266        #[cfg(windows)]
267        {
268            let _transaction = self
269                .inner
270                .transaction
271                .lock()
272                .unwrap_or_else(|poisoned| poisoned.into_inner());
273            let (directory, name) = oauth_store_directory_windows(&self.inner.path)?;
274            let lock = directory
275                .lock_within(OAUTH_STORE_LOCK_FILE, DEFAULT_LOCK_BUDGET)
276                .map_err(|_| ())?;
277            let mut candidate = load_oauth_store_from_lock_windows(&lock, &name)?;
278            update(&mut candidate);
279            let bytes = serde_json::to_vec(&candidate).map_err(|_| ())?;
280            lock.atomic_replace(&name, &bytes).map_err(|_| ())
281        }
282        #[cfg(unix)]
283        {
284            let _transaction = self
285                .inner
286                .transaction
287                .lock()
288                .unwrap_or_else(|poisoned| poisoned.into_inner());
289            let _file_lock = OAuthStoreLock::acquire(&self.inner.path)?;
290            let mut candidate = load_oauth_store(&self.inner.path)?;
291            update(&mut candidate);
292            persist_oauth_store(&self.inner.path, &candidate)?;
293            Ok(())
294        }
295    }
296
297    /// Lock-free read of one credential entry (D6).
298    ///
299    /// The store is only ever mutated via temp-file plus atomic replace, so a
300    /// reader always observes a complete old-or-new document; there is no torn
301    /// state to guard. On Windows the general data reader opens with
302    /// `FILE_SHARE_DELETE`, so a lock-free reader cannot block the writer's
303    /// replace.
304    fn get(&self, key: &str) -> Result<Option<PersistedOAuthCredential>, ()> {
305        #[cfg(windows)]
306        {
307            Ok(load_oauth_store_windows(&self.inner.path)?
308                .get(key)
309                .cloned())
310        }
311        #[cfg(unix)]
312        {
313            Ok(load_oauth_store(&self.inner.path)?.get(key).cloned())
314        }
315    }
316
317    /// Bounded-lock read used by the refresh-claim check (D6-correctness).
318    ///
319    /// Unlike [`Self::get`] this takes the exclusive lock so the pre-flight
320    /// compare against disk is coherent; any error (including contention) is
321    /// reported to the caller, which skips the optimization and proceeds with
322    /// the live request.
323    fn get_for_claim(&self, key: &str) -> Result<Option<PersistedOAuthCredential>, ()> {
324        #[cfg(windows)]
325        {
326            let _transaction = self
327                .inner
328                .transaction
329                .lock()
330                .unwrap_or_else(|poisoned| poisoned.into_inner());
331            let (directory, name) = oauth_store_directory_windows(&self.inner.path)?;
332            let lock = directory
333                .lock_within(OAUTH_STORE_LOCK_FILE, DEFAULT_LOCK_BUDGET)
334                .map_err(|_| ())?;
335            Ok(load_oauth_store_from_lock_windows(&lock, &name)?
336                .get(key)
337                .cloned())
338        }
339        #[cfg(unix)]
340        {
341            let _transaction = self
342                .inner
343                .transaction
344                .lock()
345                .unwrap_or_else(|poisoned| poisoned.into_inner());
346            let _file_lock = OAuthStoreLock::acquire(&self.inner.path)?;
347            Ok(load_oauth_store(&self.inner.path)?.get(key).cloned())
348        }
349    }
350}

352#[async_trait]
353impl CredentialStore for ServerCredentialStore {
354    async fn load(&self) -> Result<Option<StoredCredentials>, rmcp::transport::AuthError> {
355        let stored = self
356            .file
357            .get(&self.key)
358            .map_err(|()| oauth_store_auth_error())?;
359        let Some(stored) = stored else {
360            return Ok(None);
361        };
362        if stored.binding != self.binding {
363            self.clear().await?;
364            return Ok(None);
365        }
366        Ok(Some(stored.credentials.into_rmcp()))
367    }
368
369    async fn save(&self, credentials: StoredCredentials) -> Result<(), rmcp::transport::AuthError> {
370        let stored = PersistedOAuthCredential {
371            binding: self.binding.clone(),
372            credentials: StrictStoredCredentials::from_rmcp(credentials),
373        };
374        self.file
375            .update(|all| {
376                if let Some(existing) = all.get(&self.key)
377                    && credential_superseded(&existing.credentials, &stored.credentials)
378                {
379                    // Compare-before-write CAS (D6-correctness): the on-disk
380                    // entry is newer (or a same-timestamp rotation), so the
381                    // caller's credentials are superseded. The DISK wins; skip
382                    // the write without error.
383                    return;
384                }
385                all.insert(self.key.clone(), stored);
386            })
387            .map_err(|()| oauth_store_auth_error())
388    }
389
390    async fn clear(&self) -> Result<(), rmcp::transport::AuthError> {
391        self.file
392            .update(|all| {
393                all.remove(&self.key);
394            })
395            .map_err(|()| oauth_store_auth_error())
396    }
397}

399impl ServerCredentialStore {
400    /// Bounded-lock disk read for the refresh-claim check.
401    ///
402    /// Any error (including lock contention) yields `None`: the optimization is
403    /// skipped and the live request proceeds. This is never a failure mode.
404    fn claim_credential(&self) -> Option<PersistedOAuthCredential> {
405        self.file.get_for_claim(&self.key).ok().flatten()
406    }
407}
408
409/// Compare-before-write CAS predicate (D6-correctness).
410///
411/// Returns `true` when the freshly re-read on-disk entry supersedes the
412/// candidate being saved, in which case the write is skipped. The disk wins
413/// when it is strictly newer by `token_received_at`, or carries the same
414/// timestamp with a different refresh token. Equal-refresh-token idempotent
415/// writes and `token_response: None` administrative downgrades are allowed.
416fn credential_superseded(
417    existing: &StrictStoredCredentials,
418    candidate: &StrictStoredCredentials,
419) -> bool {
420    if candidate.token_response.is_none() {
421        return false;
422    }
423    let strictly_newer = existing.token_received_at > candidate.token_received_at;
424    let same_timestamp = existing.token_received_at == candidate.token_received_at;
425    strictly_newer || (same_timestamp && existing.refresh_token() != candidate.refresh_token())
426}

428fn refresh_token_of(response: &OAuthTokenResponse) -> Option<String> {

431        .get("refresh_token")?

433        .map(str::to_owned)
434}

436impl OAuthCredentialBinding {
437    fn from_config(config: &McpServerConfig) -> Option<Self> {
438        let resource_url = canonical_oauth_resource_url(config.url.as_deref()?);
439        let settings = config.oauth.settings();
440        let client_secret_sha256 = settings.client_secret.as_ref().map(|secret| {
441            let digest = Sha256::digest(secret.as_bytes());
442            digest
443                .iter()
444                .map(|byte| format!("{byte:02x}"))
445                .collect::<String>()
446        });
447        Some(Self {
448            resource_url,
449            configured_client_id: settings.client_id,
450            client_metadata_url: settings.client_metadata_url,
451            client_secret_sha256,
452            scopes: settings.scopes,
453        })
454    }
455}

457fn canonical_oauth_resource_url(resource_url: &str) -> String {
458    let Ok(mut url) = url::Url::parse(resource_url) else {
459        // Invalid URLs stay fully distinct instead of falling back to a broader key.
460        return resource_url.to_owned();
461    };
462    if matches!(
463        (url.scheme(), url.port()),
464        ("http", Some(80)) | ("https", Some(443))
465    ) {
466        let _ = url.set_port(None);
467    }
468    url.into()
469}

471fn oauth_credential_key(server: &str, canonical_resource_url: &str) -> String {
472    let digest = Sha256::digest(canonical_resource_url.as_bytes());
473    let hash = digest
474        .iter()
475        .map(|byte| format!("{byte:02x}"))
476        .collect::<String>();
477    format!("{server}:{hash}")
478}

506impl McpOAuthHttpClient {
507    fn new(exchange: OAuthExchangeState, store: ServerCredentialStore) -> Result<Self, ToolError> {
508        let base = || {
509            reqwest::Client::builder()
510                .timeout(Duration::from_secs(30))
511                .pool_max_idle_per_host(0)
512        };
513        let follow_redirects = base()
514            .build()
515            .map_err(|_| ToolError::execution("MCP OAuth HTTP client setup failed"))?;
516        let stop_redirects = base()
517            .redirect(reqwest::redirect::Policy::none())
518            .build()
519            .map_err(|_| ToolError::execution("MCP OAuth HTTP client setup failed"))?;
520        Ok(Self {
521            follow_redirects,
522            stop_redirects,
523            exchange,
524            store,
525        })
526    }
527
528    /// Pre-flight refresh claim (D6-correctness).
529    ///
530    /// Compares the refresh token about to be presented to the IdP against the
531    /// on-disk credential under a bounded lock. When disk has rotated ahead,
532    /// another process already refreshed: synthesize a 200 adopting the disk
533    /// credentials instead of making a doomed IdP call. Any read error or
534    /// contention skips the check so the live request proceeds.
535    fn adopted_success(&self, requested: Option<&str>) -> Option<http::Response<Vec<u8>>> {
536        let requested = requested?;
537        let disk = self.store.claim_credential()?;
538        if disk.binding != self.store.binding {
539            return None;
540        }
541        let token_response = disk.credentials.token_response.as_ref()?;
542        let disk_refresh = refresh_token_of(token_response)?;
543        if disk_refresh == requested {
544            return None;
545        }
546        let body = serde_json::to_vec(token_response).ok()?;
547        http::Response::builder()
548            .status(http::StatusCode::OK)
549            .header(http::header::CONTENT_TYPE, "application/json")
550            .body(body)
551            .ok()
552    }
553}

555/// Extracts the refresh-token grant's token from an OAuth form body.
556///
557/// Returns `Some(token)` only for `grant_type=refresh_token` requests; `None`
558/// otherwise. This sniffing is the fragile part of the refresh claim and is
559/// covered by a debug test that fails loudly if rmcp stops routing refreshes
560/// through this client.
561fn refresh_grant_token(body: &[u8]) -> Option<String> {
562    let mut grant_type = None;
563    let mut refresh_token = None;
564    for (name, value) in url::form_urlencoded::parse(body) {
565        match name.as_ref() {
566            "grant_type" => grant_type = Some(value.into_owned()),
567            "refresh_token" => refresh_token = Some(value.into_owned()),
568            _ => {}
569        }
570    }
571    if grant_type.as_deref() != Some("refresh_token") {
572        return None;
573    }
574    refresh_token
575}

577/// Whether a token-endpoint response is the `invalid_grant` rejection.
578fn response_is_invalid_grant(response: &http::Response<Vec<u8>>) -> bool {
579    if response.status() != http::StatusCode::BAD_REQUEST {
580        return false;
581    }
582    if let Ok(value) = serde_json::from_slice::<serde_json::Value>(response.body())
583        && value.get("error").and_then(serde_json::Value::as_str) == Some("invalid_grant")
584    {
585        return true;
586    }
587    response
588        .body()
589        .windows(b"invalid_grant".len())
590        .any(|window| window == b"invalid_grant")
591}

593impl OAuthExchangeState {
594    fn diagnostic(&self, error: &(dyn std::error::Error + 'static)) -> String {
595        self.response_error
596            .lock()
597            .unwrap_or_else(|p| p.into_inner())
598            .clone()
599            .unwrap_or_else(|| cookie_agent_protocol::diagnostics::error_chain(error))
600    }
601
602    fn install(&self, code: String) {
603        *self
604            .code
605            .lock()
606            .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(zeroize::Zeroizing::new(code));
607    }
608
609    fn restore_in_request(&self, request: &mut http::Request<Vec<u8>>) {
610        let parameters = url::form_urlencoded::parse(request.body())
611            .into_owned()
612            .collect::<Vec<_>>();
613        if !parameters
614            .iter()
615            .any(|(name, value)| name == "code" && value == REDACTED_AUTHORIZATION_CODE)
616        {
617            return;
618        }
619        let Some(code) = self
620            .code
621            .lock()
622            .unwrap_or_else(|poisoned| poisoned.into_inner())
623            .take()
624        else {
625            return;
626        };
627        let mut encoded = url::form_urlencoded::Serializer::new(String::new());
628        for (name, value) in parameters {
629            encoded.append_pair(
630                &name,
631                if name == "code" && value == REDACTED_AUTHORIZATION_CODE {
632                    code.as_str()
633                } else {
634                    &value
635                },
636            );
637        }
638        *request.body_mut() = encoded.finish().into_bytes();
639    }
640}

642impl OAuthHttpClient for McpOAuthHttpClient {
643    fn execute(&self, operation: rmcp::transport::OAuthHttpRequest) -> OAuthHttpClientFuture<'_> {
644        Box::pin(async move {
645            let client = match operation.redirect_policy {
646                OAuthHttpRedirectPolicy::Follow => &self.follow_redirects,
647                OAuthHttpRedirectPolicy::Stop => &self.stop_redirects,
648                _ => &self.stop_redirects,
649            };
650            let timeout = operation.timeout;
651            let mut oauth_request = operation.request;
652            self.exchange.restore_in_request(&mut oauth_request);
653            // D6-correctness pre-flight: for a refresh grant, adopt a rotation
654            // that already landed on disk instead of making a doomed IdP call.
655            let refresh_token = refresh_grant_token(oauth_request.body());
656            if let Some(response) = self.adopted_success(refresh_token.as_deref()) {
657                return Ok(response);
658            }
659            let mut request = reqwest::Request::try_from(oauth_request)
660                .map_err(|_| Box::new(OAuthHttpFailure) as rmcp::transport::OAuthHttpClientError)?;
661            *request.timeout_mut() = timeout;
662            let response = client
663                .execute(request)
664                .await
665                .map_err(|_| Box::new(OAuthHttpFailure) as rmcp::transport::OAuthHttpClientError)?;
666            let status = response.status();
667            let version = response.version();
668            let headers = response.headers().clone();
669            let mut body = Vec::new();
670            let mut stream = response.bytes_stream();
671            while let Some(chunk) = stream.next().await {
672                let chunk = chunk.map_err(|_| {
673                    Box::new(OAuthHttpFailure) as rmcp::transport::OAuthHttpClientError
674                })?;
675                if chunk.len() > OAUTH_STORE_MAX_BYTES as usize - body.len() {
676                    return Err(Box::new(OAuthHttpFailure) as rmcp::transport::OAuthHttpClientError);
677                }
678                body.extend_from_slice(&chunk);
679            }
680            // Keep the original response for OAuth error classification. Retain a bounded
681            // display copy because rmcp can discard the body on parse failures.
682            *self
683                .exchange
684                .response_error
685                .lock()
686                .unwrap_or_else(|p| p.into_inner()) = (!status.is_success()).then(|| {
687                cookie_agent_protocol::diagnostics::detail(&format!(
688                    "OAuth HTTP {status}\nResponse body:\n{}",
689                    String::from_utf8_lossy(&body)
690                ))
691                .to_string()
692            });
693            let mut response = http::Response::builder().status(status).version(version);
694            for (name, value) in &headers {
695                response = response.header(name, value);
696            }
697            let response = response
698                .body(body)
699                .map_err(|_| Box::new(OAuthHttpFailure) as rmcp::transport::OAuthHttpClientError)?;
700            // D6-correctness recovery: a refresh the IdP rejected with
701            // `invalid_grant` may have lost a cross-process race. Re-check disk
702            // once; if a sibling landed a rotation, adopt it instead of
703            // surfacing re-login.
704            if refresh_token.is_some()
705                && response_is_invalid_grant(&response)
706                && let Some(adopted) = self.adopted_success(refresh_token.as_deref())
707            {
708                return Ok(adopted);
709            }
710            Ok(response)
711        })
712    }
713}

715/// The redirects a client can catch: its own loopback callback (RFC 8252
716/// §7.3), plain HTTP to a loopback IP literal or `localhost`, or an `https`
717/// address such as the web client's own page. Neither may carry user info or
718/// a fragment; any path and query are kept. The verifier stays in the daemon,
719/// so a code sent anywhere else cannot be exchanged.
720fn client_redirect_uri(value: &str) -> Result<ClientRedirect, ToolError> {
721    url::Url::parse(value)
722        .ok()
723        .filter(|url| {
724            url.username().is_empty() && url.password().is_none() && url.fragment().is_none()
725        })
726        .and_then(|url| {
727            let kind = match url.scheme() {
728                "https" if url.host_str().is_some_and(|host| !host.is_empty()) => {
729                    RedirectKind::Https
730                }
731                "http" if is_loopback_host(url.host()) => RedirectKind::Loopback,
732                _ => return None,
733            };
734            Some(ClientRedirect { url, kind })
735        })
736        .ok_or_else(|| {
737            ToolError::execution(
738                "OAuth redirect URI must be an https URL, or http to a loopback address \
739                 (127.0.0.1, [::1], or localhost) such as http://127.0.0.1:<port>/callback, \
740                 without user info or a fragment",
741            )
742        })
743}
744
745fn is_loopback_host(host: Option<url::Host<&str>>) -> bool {
746    match host {
747        Some(url::Host::Ipv4(address)) => address.is_loopback(),
748        Some(url::Host::Ipv6(address)) => address.is_loopback(),
749        Some(url::Host::Domain(domain)) => domain.eq_ignore_ascii_case("localhost"),
750        None => false,
751    }
752}

754/// A redirect `mcp.auth.begin` accepted.
755struct ClientRedirect {
756    url: url::Url,
757    kind: RedirectKind,
758}

760#[derive(Clone, Copy, Debug, Eq, PartialEq)]
761enum RedirectKind {
762    /// Plain HTTP to the client's machine.
763    Loopback,
764    /// An `https` address, such as a web page's own.
765    Https,
766}
767
768impl RedirectKind {
769    /// The OIDC Dynamic Client Registration `application_type` its client
770    /// registers as: a native client may only use loopback redirects, a web
771    /// client only `https` ones.
772    const fn application_type(self) -> &'static str {
773        match self {
774            Self::Loopback => "native",
775            Self::Https => "web",
776        }
777    }

780/// Whether `callback` is a redirect to `redirect_uri` carrying an
781/// authorization response (a code or an error). The redirect's own query
782/// parameters, such as the web client's `oauth=callback`, must all be there.
783fn is_callback_for(callback: &url::Url, redirect_uri: &url::Url) -> bool {
784    let parameters = callback.query_pairs().collect::<Vec<_>>();
785    callback.scheme() == redirect_uri.scheme()
786        && callback.host() == redirect_uri.host()
787        && callback.port() == redirect_uri.port()
788        && callback.path() == redirect_uri.path()
789        && redirect_uri
790            .query_pairs()
791            .all(|pair| parameters.contains(&pair))
792        && callback
793            .query_pairs()
794            .any(|(key, _)| matches!(key.as_ref(), "code" | "error"))
795}
796
797fn parse_oauth_callback(url: &str) -> Result<AuthorizationCallback, String> {
798    AuthorizationCallback::from_redirect_url(url).map_err(|error| {
799        let mut message = error.to_string();
800        if let Ok(url) = url::Url::parse(url) {
801            for (key, value) in url
802                .query_pairs()
803                .filter(|(key, _)| matches!(key.as_ref(), "error" | "error_description"))
804            {
805                message.push_str(&format!("\n{key}: {value}"));
806            }
807        }
808        cookie_agent_protocol::diagnostics::detail(&message).to_string()
809    })
810}

868#[cfg(unix)]
869impl OAuthStoreLock {
870    fn acquire(store_path: &Path) -> Result<Self, ()> {
871        let parent = store_path.parent().ok_or(())?;
872        if !parent.exists() {
873            use std::os::unix::fs::DirBuilderExt as _;
874
875            let mut builder = fs::DirBuilder::new();
876            builder.recursive(true).mode(0o700);
877            builder.create(parent).map_err(|_| ())?;
878        }
879        let path = parent.join(OAUTH_STORE_LOCK_FILE);
880        let mut options = OpenOptions::new();
881        options.read(true).write(true).create(true);
882        #[cfg(unix)]
883        {
884            use std::os::unix::fs::OpenOptionsExt as _;
885            options.mode(0o600).custom_flags(libc::O_CLOEXEC);
886        }
887        let file = options.open(&path).map_err(|_| ())?;
888        if !lock_within_file(&file, DEFAULT_LOCK_BUDGET).map_err(|_| ())? {
889            return Err(());
890        }
891        Ok(Self { file })
892    }
893}

927#[cfg(unix)]
928fn persist_oauth_store(
929    path: &Path,
930    credentials: &BTreeMap<String, PersistedOAuthCredential>,
931) -> Result<(), ()> {
932    let parent = path.parent().ok_or(())?;
933    if !parent.exists() {
934        use std::os::unix::fs::DirBuilderExt as _;
935
936        let mut builder = fs::DirBuilder::new();
937        builder.recursive(true).mode(0o700);
938        builder.create(parent).map_err(|_| ())?;
939    }
940    let bytes = serde_json::to_vec(credentials).map_err(|_| ())?;
941    let temporary = parent.join(format!(".{OAUTH_STORE_FILE}.{}.tmp", uuid::Uuid::now_v7()));
942    let mut options = OpenOptions::new();
943    options.write(true).create_new(true);
944    {
945        use std::os::unix::fs::OpenOptionsExt as _;
946        options
947            .mode(0o600)
948            .custom_flags(libc::O_NOFOLLOW | libc::O_CLOEXEC);
949    }
950    let mut file = options.open(&temporary).map_err(|_| ())?;
951    {
952        use std::os::unix::fs::PermissionsExt as _;
953        file.set_permissions(fs::Permissions::from_mode(0o600))
954            .map_err(|_| ())?;
955    }
956    let result = (|| {
957        file.write_all(&bytes).map_err(|_| ())?;
958        file.sync_all().map_err(|_| ())?;
959        drop(file);
960        fs::rename(&temporary, path).map_err(|_| ())?;
961        sync_oauth_directory(parent).map_err(|_| ())?;
962        Ok(())
963    })();
964    if result.is_err() {
965        let _ = fs::remove_file(temporary);
966    }
967    result
968}

990    }

994impl McpRegistry {
995    /// Begins replacing `previous` with `loaded`: holds `previous`'s
996    /// credential lock, forgets its credential when the OAuth binding
997    /// changes, and supersedes it. Returns the lock and whether the
998    /// replacement must authenticate before it connects.
999    pub(crate) async fn replace_credentials<'a>(
1000        &self,
1001        previous: Option<&'a Arc<ServerRuntime>>,
1002        name: &str,
1003        loaded: &LoadedMcpServer,
1004    ) -> Result<(Option<tokio::sync::MutexGuard<'a, ()>>, bool), ToolError> {
1005        let Some(previous) = previous else {
1006            return Ok((None, false));
1007        };
1008        let previous_binding = OAuthCredentialBinding::from_config(&previous.loaded.config);
1009        let binding_changed =
1010            previous_binding != OAuthCredentialBinding::from_config(&loaded.config);
1011        let auth = previous.auth.lock.lock().await;
1012        if binding_changed
1013            && let Some(binding) = &previous_binding
1014            && let Some(credentials) = &self.inner.oauth_credentials
1015        {
1016            credentials.remove(name, binding)?;
1017        }
1018        previous.supersede();
1019        Ok((
1020            Some(auth),
1021            binding_changed
1022                && self.inner.oauth_credentials.is_some()
1023                && oauth_enabled_config(&loaded.config),
1024        ))
1025    }

1027    /// Holds `server`'s credential lock and forgets its credential, as when
1028    /// it is removed.
1029    pub(crate) async fn forget_credentials<'a>(
1030        &self,
1031        server: &'a Arc<ServerRuntime>,
1032    ) -> Result<tokio::sync::MutexGuard<'a, ()>, ToolError> {
1033        let auth = server.auth.lock.lock().await;
1034        if let Some(binding) = OAuthCredentialBinding::from_config(&server.loaded.config)
1035            && let Some(credentials) = &self.inner.oauth_credentials
1036        {
1037            credentials.remove(&server.name, &binding)?;
1038        }
1039        Ok(auth)
1040    }

1052    pub fn cancel_auth(&self, name: &str) -> Result<(), ToolError> {

1056            .flow

1127            config,

1179    async fn authorization_manager(
1180        &self,
1181    ) -> Result<(AuthorizationManager, OAuthExchangeState), ToolError> {
1182        let url = self.loaded.config.url.as_ref().ok_or_else(|| {
1183            ToolError::execution("OAuth is available only for remote MCP servers")
1184        })?;
1185        *self
1186            .auth
1187            .response_error
1188            .lock()
1189            .unwrap_or_else(|p| p.into_inner()) = None;
1190        let authorization_code = OAuthExchangeState {
1191            response_error: self.auth.response_error.clone(),
1192            ..Default::default()
1193        };

1195        // the refresh claim can compare the outgoing refresh token against
1196        // disk (D6-correctness).

1198        let http_client = McpOAuthHttpClient::new(authorization_code.clone(), store.clone())?;
1199        let mut manager =
1200            AuthorizationManager::new_with_oauth_http_client(url, Arc::new(http_client))
1201                .await
1202                .map_err(|error| ToolError::execution(authorization_code.diagnostic(&error)))?;
1203        manager.set_credential_store(store);
1204        Ok((manager, authorization_code))
1205    }

1207    async fn begin_auth(self: &Arc<Self>, redirect_uri: &str) -> Result<String, ToolError> {
1208        let ClientRedirect {
1209            url: redirect_uri,
1210            kind,
1211        } = client_redirect_uri(redirect_uri)?;
1212        if !self.oauth_enabled() {
1213            return Err(ToolError::execution(format!(
1214                "MCP server `{}` does not use OAuth (stdio, oauth=false, or a static Authorization header takes precedence)",
1215                self.name
1216            )));
1217        }
1218        if self.current_state() != McpServerState::NeedsAuth {
1219            return Err(ToolError::execution(format!(
1220                "MCP server `{}` does not currently require OAuth authorization",
1221                self.name
1222            )));
1223        }
1224        if let Some(url) = self.pending_authorization_url(&redirect_uri)? {
1225            return Ok(url);
1226        }
1227
1228        let store = self.oauth_store()?;
1229        store
1230            .clear()
1231            .await
1232            .map_err(|_| ToolError::execution("MCP OAuth credential reset failed"))?;
1233        let (mut manager, authorization_code) = self.authorization_manager().await?;
1234        let settings = self.loaded.config.oauth.settings();
1235        let mut request = AuthorizationRequest::new(redirect_uri.as_str())
1236            .with_client_name("Cookie Agent")
1237            .with_application_type(kind.application_type());
1238        let challenge = self
1239            .auth
1240            .challenge
1241            .lock()
1242            .unwrap_or_else(|poisoned| poisoned.into_inner())
1243            .clone();
1244        if let Some(challenge) = &challenge {
1245            request = request.with_challenge(challenge);
1246        }
1247        let metadata = manager
1248            .resolve_metadata_from_challenge(challenge.as_deref())
1249            .await
1250            .map_err(|error| ToolError::execution(authorization_code.diagnostic(&error)))?;
1251        manager.set_metadata(metadata.metadata);
1252        request = apply_oauth_settings(request, &settings);
1253        let session = AuthorizationSession::new(manager, request)
1254            .await
1255            .map_err(|(_, error)| ToolError::execution(authorization_code.diagnostic(&error)))?;
1256        let authorization_url = session.get_authorization_url().to_owned();
1257        let cancellation = CancellationToken::new();
1258        let generation = self.auth.generation.fetch_add(1, Ordering::AcqRel) + 1;
1259        {
1260            let mut flow = self
1261                .auth
1262                .flow
1263                .lock()
1264                .unwrap_or_else(|poisoned| poisoned.into_inner());
1265            if let Some(existing) = flow.as_ref() {
1266                if existing.redirect_uri == redirect_uri {
1267                    return Ok(existing.authorization_url.clone());
1268                }
1269                if existing.exchange.is_none() {
1270                    return Err(self.completing_error());
1271                }
1272                existing.cancellation.cancel();
1273            }
1274            *flow = Some(OAuthFlowState {
1275                generation,
1276                authorization_url: authorization_url.clone(),
1277                redirect_uri,
1278                cancellation: cancellation.clone(),
1279                exchange: Some((session, authorization_code)),
1280            });
1281        }
1282        self.set_auth_in_progress(true);
1283        self.set_status(
1284            McpServerState::NeedsAuth,
1285            Some("waiting for OAuth browser callback".into()),
1286        );
1287
1288        let registry = self
1289            .registry
1290            .upgrade()
1291            .ok_or_else(|| ToolError::execution("MCP registry is unavailable"))?;
1292        let server = Arc::clone(self);
1293        let task_registry = Arc::clone(&registry);
1294        let task = tokio::spawn(async move {
1295            tokio::select! {
1296                () = tokio::time::sleep(OAUTH_CALLBACK_TIMEOUT) => {}
1297                () = cancellation.cancelled() => return,
1298                () = task_registry.shutdown.cancelled() => return,
1299                () = server.superseded.cancelled() => return,
1300            }
1301            if server.expire_auth_flow(generation) {
1302                server.set_status(
1303                    McpServerState::NeedsAuth,
1304                    Some("OAuth authorization timed out; authenticate to try again".into()),
1305                );
1306            }
1307        });
1308        registry
1309            .connection_tasks
1310            .lock()
1311            .unwrap_or_else(|poisoned| poisoned.into_inner())
1312            .push(task);
1313        Ok(authorization_url)
1314    }
1315
1316    /// The URL of a flow already waiting on `redirect_uri`, so a client that
1317    /// asks again resumes it. A flow waiting on another client's callback is
1318    /// replaced once the new one is built, unless its code is already being
1319    /// exchanged.
1320    fn pending_authorization_url(
1321        &self,
1322        redirect_uri: &url::Url,
1323    ) -> Result<Option<String>, ToolError> {
1324        let flow = self
1325            .auth
1326            .flow
1327            .lock()
1328            .unwrap_or_else(|poisoned| poisoned.into_inner());
1329        match flow.as_ref() {
1330            Some(existing) if existing.redirect_uri == *redirect_uri => {
1331                Ok(Some(existing.authorization_url.clone()))
1332            }
1333            Some(existing) if existing.exchange.is_none() => Err(self.completing_error()),
1334            _ => Ok(None),
1335        }
1336    }
1337
1338    fn completing_error(&self) -> ToolError {
1339        ToolError::execution(format!(
1340            "MCP server `{}` is already completing OAuth authorization",
1341            self.name
1342        ))
1343    }
1344
1345    /// Exchanges the code in the browser's redirect, which the client caught
1346    /// on its own loopback or its own page. A URL that is not this flow's callback is
1347    /// rejected without ending the flow, so a mistaken paste can be retried;
1348    /// anything that reaches the exchange, including a denial or a state
1349    /// mismatch, ends it.
1350    async fn complete_auth(self: &Arc<Self>, callback_url: &str) -> Result<(), ToolError> {
1351        let callback = url::Url::parse(callback_url.trim()).ok();
1352        let registry = self
1353            .registry
1354            .upgrade()
1355            .filter(|registry| !registry.shutdown.is_cancelled())
1356            .ok_or_else(|| ToolError::execution("MCP registry is unavailable"))?;
1357        let (generation, session, authorization_code, cancellation) = {
1358            let mut flow = self
1359                .auth
1360                .flow
1361                .lock()
1362                .unwrap_or_else(|poisoned| poisoned.into_inner());
1363            let state = flow.as_mut().ok_or_else(|| {
1364                ToolError::execution(format!(
1365                    "MCP server `{}` has no OAuth authorization in progress",
1366                    self.name
1367                ))
1368            })?;
1369            if !callback
1370                .as_ref()
1371                .is_some_and(|callback| is_callback_for(callback, &state.redirect_uri))
1372            {
1373                return Err(ToolError::execution(format!(
1374                    "not the OAuth callback for MCP server `{}`: expected the browser's redirect to {}",
1375                    self.name, state.redirect_uri
1376                )));
1377            }
1378            let (session, authorization_code) = state
1379                .exchange
1380                .take()
1381                .ok_or_else(|| self.completing_error())?;
1382            (
1383                state.generation,
1384                session,
1385                authorization_code,
1386                state.cancellation.clone(),
1387            )
1388        };
1389        let callback = callback.expect("validated callback URL");
1390        let cancelled = || ToolError::execution("OAuth authorization was cancelled");
1391        let auth_guard = tokio::select! {
1392            lock = self.auth.lock.lock() => lock,
1393            () = cancellation.cancelled() => return Err(cancelled()),
1394            () = registry.shutdown.cancelled() => return Err(cancelled()),
1395        };
1396        let exchanged = match parse_oauth_callback(callback.as_str()) {
1397            Ok(callback) => {
1398                authorization_code.install(callback.code);
1399                tokio::select! {
1400                    result = session.handle_callback_with_issuer(
1401                        REDACTED_AUTHORIZATION_CODE,
1402                        &callback.csrf_token,
1403                        callback.issuer.as_deref(),
1404                    ) => result.map(drop).map_err(|error| authorization_code.diagnostic(&error)),
1405                    () = cancellation.cancelled() => return Err(cancelled()),
1406                    () = registry.shutdown.cancelled() => return Err(cancelled()),
1407                }
1408            }
1409            Err(error) => Err(error),
1410        };
1411        drop(auth_guard);
1412        if !self.finish_auth_flow(generation) {
1413            return Err(cancelled());
1414        }
1415        if let Err(message) = exchanged {
1416            let message = cookie_agent_protocol::diagnostics::detail(&format!(
1417                "OAuth authorization failed; authenticate to try again\n{message}"
1418            ))
1419            .to_string();
1420            self.set_status(McpServerState::NeedsAuth, Some(message.clone()));
1421            return Err(ToolError::execution(message));
1422        }
1423        *self
1424            .auth
1425            .challenge
1426            .lock()
1427            .unwrap_or_else(|poisoned| poisoned.into_inner()) = None;
1428        self.cancel_tool_refresh();
1429        if let Some(mut service) = self.service.lock().await.take() {
1430            let _ = service.close_with_timeout(Duration::from_secs(4)).await;
1431        }
1432        self.set_status(McpServerState::Disconnected, None);
1433        let server = Arc::clone(self);
1434        let task = tokio::spawn(async move {
1435            if let Err(error) = server.connect().await {
1436                server.fail(error.to_string());
1437            }
1438        });
1439        registry
1440            .connection_tasks
1441            .lock()
1442            .unwrap_or_else(|poisoned| poisoned.into_inner())
1443            .push(task);
1444        Ok(())
1445    }
1446
1447    /// Ends a flow nobody completed in time. A flow whose code is being
1448    /// exchanged is left to finish.
1449    fn expire_auth_flow(&self, generation: u64) -> bool {
1450        let mut flow = self
1451            .auth
1452            .flow
1453            .lock()
1454            .unwrap_or_else(|poisoned| poisoned.into_inner());
1455        if !flow
1456            .as_ref()
1457            .is_some_and(|flow| flow.generation == generation && flow.exchange.is_some())
1458        {
1459            return false;
1460        }
1461        *flow = None;
1462        drop(flow);
1463        self.set_auth_in_progress(false);
1464        true
1465    }
1466
1467    fn finish_auth_flow(&self, generation: u64) -> bool {
1468        let mut flow = self
1469            .auth
1470            .flow
1471            .lock()
1472            .unwrap_or_else(|poisoned| poisoned.into_inner());
1473        let finished = if flow
1474            .as_ref()
1475            .is_some_and(|flow| flow.generation == generation)
1476        {
1477            if let Some(flow) = flow.take() {
1478                flow.cancellation.cancel();
1479            }
1480            true
1481        } else {
1482            false
1483        };
1484        drop(flow);
1485        if finished {
1486            self.set_auth_in_progress(false);
1487        }
1488        finished
1489    }
1490}
