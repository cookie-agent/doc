1//! Filesystem targets resolved once at preparation and used at execution.
2//!
3//! Preparation resolves the requested path and keeps the target's parent
4//! directory open; execution works relative to that handle. A symlink swapped
5//! into the route while an approval is pending therefore cannot redirect the
6//! operation to a path other than the one that was approved. The target's
7//! content is deliberately not pinned: execution sees, and replaces, whatever
8//! the file holds at that moment.

37        fn reserve() -> Result<Self, ToolError> {
38            reserve_weight(
39                &PREPARED_WEIGHT,
40                PREPARED_WEIGHT_LIMIT,

44        }

50        }

53    pub(super) fn reserve_weight(
54        counter: &AtomicUsize,
55        limit: usize,
56        weight: usize,
57    ) -> Result<(), ToolError> {
58        // A compare-exchange loop rather than `fetch_update`, which Rust 1.99
59        // deprecates in favour of a `try_update` the 1.88 MSRV lacks.
60        let mut used = counter.load(Ordering::Acquire);
61        loop {
62            let Some(next) = used.checked_add(weight).filter(|next| *next <= limit) else {
63                return Err(ToolError::resource_limit(format!(
64                    "aggregate prepared-object weight would exceed {limit}"
65                )));
66            };
67            match counter.compare_exchange_weak(used, next, Ordering::AcqRel, Ordering::Acquire) {
68                Ok(_) => return Ok(()),
69                Err(actual) => used = actual,
70            }
71        }
72    }

217    impl PreparedAbsent {
218        fn create(&self, bytes: &[u8]) -> Result<(), ToolError> {
219            let (leaf, directories) = self
220                .missing
221                .split_last()
222                .ok_or_else(|| ToolError::execution("missing path is empty"))?;
223            let mut created = None;
224            for name in directories {
225                let parent = created.as_ref().unwrap_or(&self.parent);
226                match rustix::fs::mkdirat(parent, name, permission_mode(CREATED_DIRECTORY_MODE)) {
227                    Ok(()) | Err(rustix::io::Errno::EXIST) => {}
228                    Err(error) => return Err(super::io_error(error)),
229                }
230                // NOFOLLOW: a symlink planted here since preparation must not
231                // redirect the write.
232                let next = rustix::fs::openat(
233                    parent,
234                    name,
235                    OFlags::RDONLY | OFlags::DIRECTORY | OFlags::NOFOLLOW | OFlags::CLOEXEC,
236                    Mode::empty(),
237                )
238                .map_err(|error| match error {
239                    rustix::io::Errno::LOOP | rustix::io::Errno::NOTDIR => {
240                        ToolError::operation_changed(
241                            "a prepared missing directory was replaced by another type",
242                        )
243                    }
244                    error => super::io_error(error),
245                })?;
246                created = Some(File::from(next));
247            }
248            publish(
249                created.as_ref().unwrap_or(&self.parent),
250                leaf,
251                bytes,
252                CREATED_FILE_MODE,
253            )
254        }
255    }

257    /// Writes `bytes` to a sibling temporary and renames it over `name`: one
258    /// file sync before the rename and one directory sync after it.
259    fn publish(parent: &File, name: &OsStr, bytes: &[u8], mode: u32) -> Result<(), ToolError> {
260        let mode = match rustix::fs::statat(parent, name, AtFlags::SYMLINK_NOFOLLOW) {
261            Ok(stat) => match FileType::from_raw_mode(stat.st_mode) {
262                FileType::Directory => {
263                    return Err(ToolError::execution("write target is a directory"));
264                }
265                #[allow(clippy::unnecessary_cast)]
266                FileType::RegularFile => stat.st_mode as u32,
267                _ => mode,
268            },
269            Err(_) => mode,
270        };
271        let temporary = OsString::from(format!(
272            ".cookie-agent-stage-{}-{}",
273            std::process::id(),
274            TEMP_SEQUENCE.fetch_add(1, Ordering::Relaxed)
275        ));
276        let mut file = File::from(
277            rustix::fs::openat(
278                parent,
279                &temporary,
280                OFlags::WRONLY | OFlags::CREATE | OFlags::EXCL | OFlags::CLOEXEC,
281                permission_mode(mode),
282            )
283            .map_err(super::io_error)?,
284        );
285        let staged = file
286            .write_all(bytes)
287            .and_then(|()| file.sync_all())
288            .map_err(super::io_error)
289            .and_then(|()| {
290                rustix::fs::renameat(parent, &temporary, parent, name).map_err(super::io_error)
291            });
292        if let Err(error) = staged {
293            let _ = rustix::fs::unlinkat(parent, &temporary, AtFlags::empty());
294            return Err(error);
295        }
296        rustix::fs::fsync(parent).map_err(super::io_error)
297    }

314    pub fn prepare_target(cwd: &Path, requested: &Path) -> Result<PreparedTarget, ToolError> {
315        let absolute = normalize_absolute(if requested.is_absolute() {
316            requested.to_owned()
317        } else {
318            cwd.join(requested)
319        })?;
320        if absolute.file_name().is_none() {
321            return Err(ToolError::unsupported_security("target has no basename"));
322        }
323        let budget = BudgetReservation::reserve()?;
324        let ResolvedPath {
325            parent,
326            parent_path,
327            basename,
328            missing,
329        } = resolve_path(&absolute)?;
330        if !missing.is_empty() {
331            let mut resolved = parent_path;
332            resolved.extend(&missing);
333            return Ok(PreparedTarget::Absent(PreparedAbsent {
334                parent,
335                missing,
336                resolved,
337                display_path: absolute,
338                _budget: budget,
339            }));
340        }
341        let flags = OFlags::RDONLY | OFlags::NOFOLLOW | OFlags::CLOEXEC | OFlags::NONBLOCK;
342        match rustix::fs::openat(&parent, &basename, flags, Mode::empty()) {
343            Ok(fd) => {
344                let object = File::from(fd);
345                let metadata = object.metadata().map_err(super::io_error)?;
346                if !metadata.is_file() && !metadata.is_dir() {
347                    return Err(ToolError::unsupported_security(
348                        "prepared filesystem object is not a regular file or directory",
349                    ));
350                }
351                let mut resolved = parent_path;
352                if basename != "." {
353                    resolved.push(&basename);
354                }
355                Ok(PreparedTarget::Existing(PreparedExisting {
356                    parent,
357                    basename,
358                    object,
359                    resolved,
360                    display_path: absolute,
361                    directory: metadata.is_dir(),
362                    mode: metadata.mode(),
363                    _budget: budget,
364                }))
365            }
366            Err(rustix::io::Errno::NOENT) => {
367                let mut resolved = parent_path;
368                resolved.push(&basename);
369                Ok(PreparedTarget::Absent(PreparedAbsent {
370                    parent,
371                    missing: vec![basename],
372                    resolved,
373                    display_path: absolute,
374                    _budget: budget,
375                }))
376            }
377            Err(rustix::io::Errno::LOOP) => Err(ToolError::operation_changed(
378                "prepared target became a symlink during resolution",
379            )),
380            Err(error) => Err(super::io_error(error)),
381        }
382    }

384    fn normalize_absolute(path: PathBuf) -> Result<PathBuf, ToolError> {
385        if !path.is_absolute() {
386            return Err(ToolError::unsupported_security(
387                "filesystem path is not absolute",
388            ));
389        }
390        let mut normalized = PathBuf::from("/");
391        for component in path.components() {
392            match component {
393                Component::RootDir | Component::CurDir => {}
394                Component::Normal(name) => normalized.push(name),
395                Component::ParentDir => {
396                    if !normalized.pop() {
397                        return Err(ToolError::unsupported_security(
398                            "filesystem path traverses above its root",
399                        ));
400                    }
401                }
402                Component::Prefix(_) => {
403                    return Err(ToolError::unsupported_security(
404                        "filesystem path has an unsupported prefix",
405                    ));
406                }
407            }
408        }
409        Ok(normalized)
410    }

420    /// Walks `path` one component at a time from `/`, expanding symlinks in
421    /// place, so `..` is always evaluated from the directory actually reached.
422    fn resolve_path(path: &Path) -> Result<ResolvedPath, ToolError> {
423        let flags = OFlags::RDONLY | OFlags::DIRECTORY | OFlags::NOFOLLOW | OFlags::CLOEXEC;
424        let open_root = || {
425            rustix::fs::open("/", flags, Mode::empty())
426                .map(File::from)
427                .map_err(super::io_error)
428        };
429        let mut current = open_root()?;
430        let mut current_path = PathBuf::from("/");
431        let mut pending = path
432            .components()
433            .map(|part| part.as_os_str().to_owned())
434            .collect::<std::collections::VecDeque<_>>();
435        let mut links = 0;
436        let mut steps = 0;
437        while let Some(name) = pending.pop_front() {
438            steps += 1;
439            if steps > 4096 || pending.len() > 4096 {
440                return Err(ToolError::resource_limit(
441                    "prepared path resolution is too deep",
442                ));
443            }
444            if name == "/" {
445                current = open_root()?;
446                current_path = PathBuf::from("/");
447                continue;
448            }
449            if name == "." {
450                continue;
451            }
452            let stat = match rustix::fs::statat(&current, &name, AtFlags::SYMLINK_NOFOLLOW) {
453                Ok(stat) => stat,
454                Err(rustix::io::Errno::NOENT) => {
455                    if pending.back().is_some_and(|part| part == ".") {
456                        return Err(ToolError::execution(
457                            "symlink destination requires a directory",
458                        ));
459                    }
460                    let mut missing = vec![name];
461                    for part in pending {
462                        if part == ".." {
463                            return Err(ToolError::execution(
464                                "symlink target traverses a missing directory",
465                            ));
466                        }
467                        if part != "." {
468                            missing.push(part);
469                        }
470                    }
471                    let basename = missing.last().expect("missing leaf").clone();
472                    return Ok(ResolvedPath {
473                        parent: current,
474                        parent_path: current_path,
475                        basename,
476                        missing,
477                    });
478                }
479                Err(error) => return Err(super::io_error(error)),
480            };
481            if FileType::from_raw_mode(stat.st_mode) == FileType::Symlink {
482                links += 1;
483                if links > 40 {
484                    return Err(ToolError::unsupported_security(
485                        "too many symlinks in prepared path",
486                    ));
487                }
488                let target =
489                    rustix::fs::readlinkat(&current, &name, Vec::new()).map_err(super::io_error)?;
490                let target = OsStr::from_bytes(target.as_bytes()).to_owned();
491                if target.as_bytes().ends_with(b"/") || target.as_bytes().ends_with(b"/.") {
492                    pending.push_front(OsString::from("."));
493                }
494                for part in Path::new(&target).components().rev() {
495                    pending.push_front(part.as_os_str().to_owned());
496                }
497                continue;
498            }
499            if pending.is_empty() && name != ".." {
500                return Ok(ResolvedPath {
501                    parent: current,
502                    parent_path: current_path,
503                    basename: name,
504                    missing: Vec::new(),
505                });
506            }
507            current = File::from(
508                rustix::fs::openat(&current, &name, flags, Mode::empty())
509                    .map_err(super::io_error)?,
510            );
511            if name == ".." {
512                current_path.pop();
513            } else {
514                current_path.push(&name);
515            }
516        }
517        Ok(ResolvedPath {
518            parent: current,
519            parent_path: current_path,
520            basename: OsString::from("."),
521            missing: Vec::new(),
522        })
523    }

568    pub(crate) fn lexical_path_spelling(path: &Path) -> PathBuf {
569        if !path.is_absolute() || !path.as_os_str().as_encoded_bytes().contains(&b'~') {
570            return path.to_owned();
571        }
572        let mut result = PathBuf::new();
573        let mut components = path.components();
574        while let Some(component) = components.next() {
575            let Component::Normal(name) = component else {
576                result.push(component.as_os_str());
577                continue;
578            };
579            let candidate = result.join(name);
580            let found = validate_windows_component(name)
581                .and_then(|()| wide_path(&candidate))
582                .ok()
583                .and_then(|wide| {
584                    let mut data = WIN32_FIND_DATAW::default();
585                    let handle = unsafe { FindFirstFileW(wide.as_ptr(), &mut data) };
586                    if handle == INVALID_HANDLE_VALUE {
587                        return None;
588                    }
589                    unsafe { FindClose(handle) };
590                    Some(data)
591                });
592            let Some(data) = found else {
593                result.push(name);
594                result.extend(components.map(|part| part.as_os_str()));
595                break;
596            };
597            let length = data
598                .cFileName
599                .iter()
600                .position(|value| *value == 0)
601                .unwrap_or(data.cFileName.len());
602            result.push(OsString::from_wide(&data.cFileName[..length]));
603            // FindFirstFile reports the link's own entry name. Never inspect its
604            // descendants for spelling: permission labels must retain the alias.
605            if data.dwFileAttributes & FILE_ATTRIBUTE_REPARSE_POINT != 0 {
606                result.extend(components.map(|part| part.as_os_str()));
607                break;
608            }
609        }
610        result
611    }

832    fn resolve_path(path: &Path, allow_missing: bool) -> Result<ResolvedPath, ToolError> {
833        let path = normalize_absolute(path)?;
834        let (mut current, names) = root_and_names(&path)?;
835        let mut pending = names
836            .into_iter()
837            .map(PendingComponent::Normal)
838            .collect::<VecDeque<_>>();
839        let mut visited = 1;
840        let mut links = 0;
841        ensure_resolution_bound(visited, pending.len())?;
842        open_for_identity(&current)?;
843
844        while let Some(component) = pending.pop_front() {
845            let name = match component {
846                PendingComponent::Normal(name) => name,
847                PendingComponent::Parent => {
848                    if current.file_name().is_none() || !current.pop() {
849                        return Err(ToolError::unsupported_security(
850                            "link target traverses above its native root",
851                        ));
852                    }
853                    continue;
854                }
855                PendingComponent::RequireDirectory => {
856                    if !fs::metadata(&current).map_err(super::io_error)?.is_dir() {
857                        return Err(ToolError::execution(
858                            "symlink destination requires a directory",
859                        ));
860                    }
861                    continue;
862                }
863            };
864            let candidate = current.join(&name);
865            match fs::symlink_metadata(&candidate) {
866                Ok(_) => {
867                    visited += 1;
868                    ensure_resolution_bound(visited, pending.len())?;
869                    let object = open_for_identity(&candidate)?;
870                    if opened_file_is_reparse(&object)? {
871                        let target = read_supported_link(&candidate)?;
872                        links += 1;
873                        if links > MAX_LINK_TRAVERSALS {
874                            return Err(ToolError::unsupported_security(
875                                "filesystem path exceeds the Windows link traversal limit",
876                            ));
877                        }
878                        let (target_root, target_components) =
879                            link_target_components(&current, &target)?;
880                        ensure_resolution_bound(
881                            visited + usize::from(target_root.is_some()),
882                            pending.len().saturating_add(target_components.len()),
883                        )?;
884                        if let Some(target_root) = target_root {
885                            visited += 1;
886                            open_for_identity(&target_root)?;
887                            current = target_root;
888                        }
889                        for component in target_components.into_iter().rev() {
890                            pending.push_front(component);
891                        }
892                    } else {
893                        if !pending.is_empty()
894                            && !object.metadata().map_err(super::io_error)?.is_dir()
895                        {
896                            return Err(ToolError::execution(format!(
897                                "filesystem path component is not a directory: {}",
898                                candidate.display()
899                            )));
900                        }
901                        current = candidate;
902                    }
903                }
904                Err(error) if error.kind() == std::io::ErrorKind::NotFound && allow_missing => {
905                    ensure_resolution_bound(visited, pending.len().saturating_add(1))?;
906                    if matches!(pending.back(), Some(PendingComponent::RequireDirectory)) {
907                        return Err(ToolError::execution(
908                            "symlink destination requires a directory",
909                        ));
910                    }
911                    let canonical_parent = current.canonicalize().map_err(super::io_error)?;
912                    let first_missing = canonical_parent.join(&name);
913                    current = first_missing.clone();
914                    for component in pending {
915                        match component {
916                            PendingComponent::Normal(component) => current.push(component),
917                            PendingComponent::Parent => {
918                                return Err(ToolError::execution(
919                                    "link target traverses a missing directory",
920                                ));
921                            }
922                            PendingComponent::RequireDirectory => {}
923                        }
924                    }
925                    return Ok(ResolvedPath {
926                        execution_path: current,
927                        first_missing: Some(first_missing),
928                    });
929                }
930                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
931                    return Err(ToolError::execution(format!(
932                        "filesystem path does not exist: {}",
933                        candidate.display()
934                    )));
935                }
936                Err(error) => {
937                    return Err(super::io_error(format!(
938                        "symlink_metadata failed for {}: {error}",
939                        candidate.display()
940                    )));
941                }
942            }
943        }
944        Ok(ResolvedPath {
945            execution_path: current.canonicalize().map_err(super::io_error)?,
946            first_missing: None,
947        })
948    }

963    fn link_target_components(
964        current: &Path,
965        target: &Path,
966    ) -> Result<(Option<PathBuf>, Vec<PendingComponent>), ToolError> {
967        let rooted = target.has_root();
968        let root = if target.is_absolute() {
969            Some(native_root(target)?)
970        } else if rooted {
971            Some(native_root(current)?)
972        } else {
973            None
974        };
975        let mut components = Vec::new();
976        for component in target.components() {
977            match component {
978                Component::Normal(name) => {
979                    validate_windows_component(name)?;
980                    components.push(PendingComponent::Normal(name.to_owned()));
981                }
982                Component::ParentDir => components.push(PendingComponent::Parent),
983                Component::CurDir | Component::RootDir if rooted => {}
984                Component::CurDir => {}
985                Component::Prefix(_) if target.is_absolute() => {}
986                Component::Prefix(_) => {
987                    return Err(ToolError::unsupported_security(
988                        "drive-relative link targets are unsupported",
989                    ));
990                }
991                Component::RootDir => {
992                    return Err(ToolError::unsupported_security(
993                        "link target has an unsupported root",
994                    ));
995                }
996            }
997        }
998        if link_target_requires_directory(target) {
999            components.push(PendingComponent::RequireDirectory);
1000        }
1001        Ok((root, components))
1002    }

1004    fn link_target_requires_directory(target: &Path) -> bool {
1005        let target = target.as_os_str().encode_wide().collect::<Vec<_>>();
1006        match target.as_slice() {
1007            [.., character] if *character == u16::from(b'/') || *character == u16::from(b'\\') => {
1008                true
1009            }
1010            [.., prior, character]
1011                if *character == u16::from(b'.')
1012                    && (*prior == u16::from(b'/') || *prior == u16::from(b'\\')) =>
1013            {
1014                true
1015            }
1016            _ => false,
1017        }
1018    }

1165    fn component_equal(left: &std::ffi::OsStr, right: &std::ffi::OsStr) -> bool {
1166        let normalize = |value: &std::ffi::OsStr| {
1167            let value = value.to_string_lossy();
1168            if let Some(unc) = value.strip_prefix(r"\\?\UNC\") {
1169                format!(r"\\{unc}").to_ascii_lowercase()
1170            } else {
1171                value
1172                    .strip_prefix(r"\\?\")
1173                    .unwrap_or(&value)
1174                    .to_ascii_lowercase()
1175            }
1176        };
1177        normalize(left) == normalize(right)
1178    }

1266        let parent = target
1267            .parent()
1268            .ok_or_else(|| ToolError::unsupported_security("target has no parent"))?;

1270            .file_name()
1271            .ok_or_else(|| ToolError::unsupported_security("target has no basename"))?;

1276            TEMP_SEQUENCE.fetch_add(1, Ordering::Relaxed)

1279            .write(true)
1280            .create_new(true)

1284                    "CreateFileW failed for staging file {}: {error}",

1286                ))
1287            })?;
1288        if let Err(error) = file.write_all(bytes).and_then(|()| file.sync_all()) {
1289            let _ = fs::remove_file(&temporary);
1290            return Err(super::io_error(format!(
1291                "write or flush failed for staging file {}: {error}",
1292                temporary.display()

1294        }

1298    fn move_file(source: &Path, target: &Path) -> Result<(), ToolError> {
1299        let target_wide = wide_path(target)?;
1300        // Prefer the superseding POSIX rename: classic MoveFileExW unlinks the
1301        // target before linking the source, so a reader that opens the path by
1302        // name can find it absent mid-write, and it rejects an open destination
1303        // even when that shares DELETE. The classic call remains the fallback
1304        // for filesystems without POSIX rename semantics.
1305        if rename_over_open_target(source, &target_wide).is_ok() {
1306            return Ok(());
1307        }
1308        let source_wide = wide_path(source)?;
1309        let flags = MOVEFILE_WRITE_THROUGH | MOVEFILE_REPLACE_EXISTING;
1310        if unsafe { MoveFileExW(source_wide.as_ptr(), target_wide.as_ptr(), flags) } == 0 {
1311            Err(super::io_error(format!(
1312                "atomic rename failed from {} to {}: {}",
1313                source.display(),
1314                target.display(),
1315                std::io::Error::last_os_error()
1316            )))
1317        } else {
1318            Ok(())
1319        }
1320    }

1322    fn rename_over_open_target(source: &Path, target: &[u16]) -> std::io::Result<()> {
1323        const REPLACE_IF_EXISTS: u32 = 0x1;
1324        const POSIX_SEMANTICS: u32 = 0x2;
1325
1326        let source = fs::OpenOptions::new()
1327            .access_mode(DELETE)
1328            .share_mode(FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE)
1329            .custom_flags(
1330                FILE_FLAG_OPEN_REPARSE_POINT | FILE_FLAG_BACKUP_SEMANTICS | FILE_FLAG_WRITE_THROUGH,
1331            )
1332            .open(source)?;
1333        let filename_bytes = u32::try_from(std::mem::size_of_val(&target[..target.len() - 1]))
1334            .map_err(|error| std::io::Error::new(std::io::ErrorKind::InvalidInput, error))?;
1335        let buffer_bytes =
1336            std::mem::offset_of!(FILE_RENAME_INFO, FileName) + std::mem::size_of_val(target);
1337        let buffer_length = u32::try_from(buffer_bytes)
1338            .map_err(|error| std::io::Error::new(std::io::ErrorKind::InvalidInput, error))?;
1339        let mut buffer = vec![
1340            FILE_RENAME_INFO::default();
1341            buffer_bytes.div_ceil(std::mem::size_of::<FILE_RENAME_INFO>())
1342        ];
1343        let info = buffer.as_mut_ptr();
1344        // The header array provides FILE_RENAME_INFO alignment and enough storage
1345        // for its variable-length UTF-16 tail, including the terminator.
1346        unsafe {
1347            (*info).Anonymous = FILE_RENAME_INFO_0 {
1348                Flags: REPLACE_IF_EXISTS | POSIX_SEMANTICS,
1349            };
1350            (*info).RootDirectory = std::ptr::null_mut();
1351            (*info).FileNameLength = filename_bytes;
1352            std::ptr::copy_nonoverlapping(
1353                target.as_ptr(),
1354                std::ptr::addr_of_mut!((*info).FileName).cast::<u16>(),
1355                target.len(),
1356            );
1357            if SetFileInformationByHandle(
1358                source.as_raw_handle() as HANDLE,
1359                FileRenameInfoEx,
1360                info.cast(),
1361                buffer_length,
1362            ) == 0
1363            {
1364                return Err(std::io::Error::last_os_error());
1365            }
1366        }
1367        Ok(())
1368    }
