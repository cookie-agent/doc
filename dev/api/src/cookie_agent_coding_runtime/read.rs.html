38#[serde(deny_unknown_fields)]
39struct ReadArgs {
40    #[serde(
41        rename = "filePath",
42        deserialize_with = "crate::path_args::deserialize"
43    )]
44    /// Path to the file or directory. Relative paths resolve against the session working directory.
45    file_path: String,
46    /// The most lines, or directory entries, to return. Defaults to 2000.
47    limit: Option<usize>,
48    /// The 1-based line, or directory entry, to start at. Defaults to 1.
49    offset: Option<usize>,
50}

52/// What the model reads about `read`.
53const DESCRIPTION: &str = "Read a file or a directory. A text file comes back as numbered lines, `N: text`, counting from 1. `offset` is the 1-based line to start at (default 1) and `limit` the most lines to return (default 2000). A line longer than 4 KiB is cut and ends with `... [line truncated: N more bytes]`, and a page stops before it passes the result size limit (about 2 MiB). A page that leaves lines out ends with a note such as `[120 more lines; continue with offset=2001]`: read again from that offset. An offset past the last line is an error. A directory lists its entries, one per line with `/` after a folder's name, paged by the same 1-based `offset` and `limit`. An image, video, or other media file the model accepts is attached instead.";

80impl ToolProvider for ReadTool {
81    fn provider_id(&self) -> &'static str {
82        "builtin.read"
83    }
84
85    fn tools_for_session(&self, _: &SessionToolContext) -> Result<Vec<ToolSpec>, ToolError> {
86        Ok(vec![ToolSpec {
87            background: false,
88            background_default: false,
89            collision: Default::default(),
90            output: Default::default(),
91            concurrency: cookie_agent_runtime_api::ToolConcurrency::Parallel,
92            result_truncation: cookie_agent_runtime_api::ToolResultTruncationPolicy::OptOut,
93            name: "read".into(),
94            permission_name: Self::get_permission_name("read")?.map(Into::into),
95            description: DESCRIPTION.into(),
96            parameters: schema::<ReadArgs>(),
97        }])
98    }
99
100    fn get_permission_name(tool_name: &str) -> Result<Option<&'static str>, ToolError> {
101        match tool_name {
102            "read" => Ok(Some("read")),
103            _ => Err(ToolError::execution("read provider received another tool")),
104        }
105    }
106
107    fn get_permission_resource(
108        &self,
109        name: &str,
110        arguments: &serde_json::Value,
111    ) -> Result<(Option<&'static str>, Option<String>), ToolError> {
112        let permission_name = Self::get_permission_name(name)?;
113        let args: ReadArgs = parse_args("read", arguments.clone())?;
114        if args.file_path.is_empty() {
115            return Err(ToolError::execution("filePath must not be empty"));
116        }
117        Ok((
118            permission_name,
119            Some(crate::permission_path_label(
120                &args.file_path,
121                &self.workspace,
122            )),
123        ))
124    }
125
126    fn get_display_argument(
127        &self,
128        name: &str,
129        arguments: &serde_json::Value,
130    ) -> Result<String, ToolError> {
131        if name != "read" {
132            return Err(ToolError::execution("read provider received another tool"));
133        }
134        let args: ReadArgs = parse_args("read", arguments.clone())?;
135        if args.file_path.is_empty() {
136            return Err(ToolError::execution("filePath must not be empty"));
137        }
138        let path = crate::permission_path_label(&args.file_path, &self.workspace);
139        let mut display = crate::abbreviated_display_path(&path, &self.workspace);
140        let window = match (args.offset, args.limit) {
141            (Some(offset), Some(limit)) => Some(format!("offset={offset}, limit={limit}")),
142            (Some(offset), None) => Some(format!("offset={offset}")),
143            (None, Some(limit)) => Some(format!("limit={limit}")),
144            (None, None) => None,
145        };
146        if let Some(window) = window {
147            display.push_str(" [");
148            display.push_str(&window);
149            display.push(']');
150        }
151        Ok(display)
152    }
153
154    fn presentation(&self, call: &ToolCall) -> PreparedPresentation {
155        let Ok(primary) = self.get_display_argument(&call.name, &call.arguments) else {
156            return cookie_agent_runtime_api::tool_title_only(&call.name);
157        };
158        let Ok(args) = parse_args::<ReadArgs>("read", call.arguments.clone()) else {
159            return cookie_agent_runtime_api::tool_title_only(&call.name);
160        };
161        let shown = self.shown_path(&args).display().to_string();
162        // The title and primary argument already name the file.
163        let mut entries = Vec::new();
164        entries.extend(args.offset.map(|offset| ("offset", offset.to_string())));
165        entries.extend(args.limit.map(|limit| ("limit", limit.to_string())));
166        let presentation =
167            tool_presentation(&format!("Read {shown}"), &primary).with_hint(ToolHint::FILE_READ);
168        if entries.is_empty() {
169            presentation
170        } else {
171            presentation.with_arguments([DisplayBlock::key_value(entries)])
172        }
173    }
174
175    async fn prepare(
176        &self,
177        _ctx: ToolPreparationContext,
178        call: ToolCall,
179    ) -> Result<PreparedTool, ToolError> {
180        if call.name != "read" {
181            return Err(ToolError::execution("read provider received another tool"));
182        }
183        let args: ReadArgs = parse_args("read", call.arguments)?;
184        let window = LineWindow::new(args.offset, args.limit)?;
185        let workspace = self.workspace.clone();
186        let shown_path = self.shown_path(&args);
187        crate::blocking(move || {
188            let target =
189                fs_cap::prepare_existing(&workspace, std::path::Path::new(&args.file_path))?;
190            let normalized_arguments = serde_json::json!({
191                "filePath": target.display_path,
192                "offset": window.offset,
193                "limit": window.limit,
194            });
195            PreparedTool::declared(
196                normalized_arguments,
197                None,
198                Box::new(ReadExecutor {
199                    target,
200                    window,
201                    shown_path,
202                }),
203            )
204        })
205        .await
206    }
207}

209#[async_trait]
210impl PreparedExecutor for ReadExecutor {
211    async fn execute(
212        self: Box<Self>,
213        context: ToolExecutionContext,
214    ) -> Result<cookie_agent_runtime_api::ToolCompletion, ToolError> {
215        let result = crate::blocking(move || {
216            if context.cancellation.is_cancelled() {
217                return Err(ToolError::execution(
218                    "prepared read cancelled before execution",
219                ));
220            }
221            if self.target.directory {
222                let entries = self.target.directory_entries()?;
223                let target = self.target.display_path.display().to_string();
224                self.window
225                    .check_in_range(entries.len(), "entry", &target)?;
226                let page = directory_page(&entries, self.window).collect::<Vec<_>>();
227                let mut output =
228                    format!("<path>{target}</path>\n<type>directory</type>\n<entries>\n");
229                for (name, directory) in &page {
230                    output.push_str(name);
231                    if *directory {
232                        output.push('/');
233                    }
234                    output.push('\n');
235                }
236                output.push_str("</entries>");
237                let marker = self.window.remaining_marker(entries.len(), "entry");
238                if let Some(marker) = &marker {
239                    output.push('\n');
240                    output.push_str(marker);
241                }
242                let table = DisplayBlock::Table {
243                    columns: vec!["name".into(), "type".into()],
244                    rows: page
245                        .iter()
246                        .map(|(name, directory)| {
247                            vec![
248                                name.clone(),
249                                if *directory { "directory" } else { "file" }.into(),
250                            ]
251                        })
252                        .collect(),
253                    omitted_rows: None,
254                };
255                return Ok(ToolResult {
256                    display: vec![Open::Known(display::fit_block(table, None)?)],
257                    background: None,
258                    retained_output: None,
259                    title: crate::safe_title(format!("Read {target}")),
260                    output,
261                    metadata: serde_json::json!({
262                        "kind": "directory",
263                        "offset": self.window.offset,
264                        "limit": self.window.limit,
265                        "shown": page.len(),
266                        "total_entries": entries.len(),
267                        "next_offset": marker.is_some().then(|| self.window.end_index() + 1),
268                    }),
269                    attachments: Vec::new(),
270                    additional_messages: Vec::new(),
271                });
272            }
273            let file = self.target.open_file()?;
274            let len = file.metadata().map_err(io_error)?.len();
275            let mut reader = BufReader::with_capacity(READ_BUFFER_BYTES, file);
276            let head = reader.fill_buf().map_err(io_error)?;
277            let path = &self.target.display_path;
278            if let Some(mime) = sniff_media_type(path, head, len)? {
279                // Gate on the file size before reading the whole file.
280                let gate = gate_attachment(&context.turn_context.input_media, mime, len);
281                if let Some(error) = attachment_gate_error(
282                    gate,
283                    mime,
284                    &context.turn_context.model,
285                    &context.turn_context.adapter,
286                ) {
287                    return Err(ToolError::execution(error));
288                }
289                let mut bytes = Vec::with_capacity(usize::try_from(len).unwrap_or(0));
290                reader.read_to_end(&mut bytes).map_err(io_error)?;
291                let mime = approved_media_type(path, &bytes)?
292                    .ok_or_else(|| ToolError::execution("file changed while it was read"))?;
293                return attachment_result(&context, path, mime, gate, &bytes);
294            }
295            if std::str::from_utf8(head).is_err_and(|error| error.error_len().is_some()) {
296                return Err(not_text());
297            }
298            text_result(path, &self.shown_path, &mut reader, self.window)
299        })
300        .await;
301        result.map(cookie_agent_runtime_api::ToolCompletion::single)
302    }
303}

313fn attachment_result(
314    context: &ToolExecutionContext,
315    path: &Path,
316    mime: &str,
317    gate: AttachmentGate,
318    bytes: &[u8],
319) -> Result<ToolResult, ToolError> {
320    let attachment = context.retain_validated_attachment(
321        mime,
322        path.file_name()
323            .map(|name| name.to_string_lossy().into_owned()),
324        bytes,
325    )?;
326    let sha256 = attachment.sha256.clone();
327    let media = DisplayBlock::Media {
328        artifact: attachment.artifact_ref(),
329        alt: path
330            .file_name()
331            .map(|name| name.to_string_lossy().into_owned()),
332    };
333    let (output, attachments, additional_messages) = match gate {
334        AttachmentGate::AttachToolResult => (
335            format!("Attached {mime} ({} bytes).", bytes.len()),
336            vec![attachment],
337            Vec::new(),
338        ),
339        AttachmentGate::DeliverViaUserTurn => (
340            format!(
341                "Attached {mime} ({} bytes), delivered in the following message.",
342                bytes.len()
343            ),
344            Vec::new(),
345            vec![
346                ToolEmittedMessage::new(
347                    ToolEmittedMessageRole::User,
348                    vec![ToolEmittedContent::File(attachment)],
349                )
350                .map_err(|error| ToolError::execution(error.to_string()))?,
351            ],
352        ),
353        AttachmentGate::RejectUnsupportedModel
354        | AttachmentGate::RejectUnsupportedFamily
355        | AttachmentGate::RejectTooLarge { .. } => {
356            unreachable!("rejected attachment gates returned an error")
357        }
358    };
359    Ok(ToolResult {
360        display: vec![Open::Known(media)],
361        background: None,
362        retained_output: None,
363        title: crate::safe_title(format!("Read {}", path.display())),
364        output,
365        metadata: serde_json::json!({"kind":"attachment","mime_type":mime,"sha256":sha256}),
366        attachments,
367        additional_messages,
368    })
369}

371/// One page of numbered lines, read in a single streaming pass that also
372/// counts the file's lines. Clients see the page as an excerpt of
373/// `shown_path`.
374fn text_result(
375    path: &Path,
376    shown_path: &Path,
377    reader: &mut impl BufRead,
378    window: LineWindow,
379) -> Result<ToolResult, ToolError> {
380    let target = path.display().to_string();
381    let mut output = format!("<path>{target}</path>\n<type>file</type>\n<content>\n");
382    let page = text_tools::page_reader(reader, window, MAX_PAGE_BYTES - output.len()).map_err(
383        |error| match error {
384            PageError::Io(error) => io_error(error),
385            PageError::NotText => not_text(),
386        },
387    )?;
388    page.check_in_range(&target)?;
389    let excerpt = page.excerpt_block(
390        shown_path.display().to_string(),
391        display::language_for_path(shown_path),
392    );
393    output.push_str(&page.text);
394    output.push_str("</content>");
395    if let Some(marker) = page.continuation() {
396        output.push('\n');
397        output.push_str(&marker);
398    }
399    Ok(ToolResult {
400        display: vec![Open::Known(display::fit_block(excerpt, None)?)],
401        background: None,
402        retained_output: None,
403        title: crate::safe_title(format!("Read {target}")),
404        output,
405        metadata: serde_json::json!({
406            "kind": "text",
407            "offset": window.offset,
408            "limit": window.limit,
409            "total_lines": page.total_lines,
410            "next_offset": page.next_offset,
411        }),
412        attachments: Vec::new(),
413        additional_messages: Vec::new(),
414    })
415}

417fn directory_page<T>(entries: &[T], window: LineWindow) -> impl Iterator<Item = &T> {
418    entries.iter().skip(window.first_index()).take(window.limit)
419}
