51/// The file a write replaces, as far as its diff needs it.
52enum Replaced {
53    /// No file: the write creates one.
54    Nothing,
55    /// UTF-8 text of at most [`MAX_DIFF_BYTES`].
56    Text(String),
57    /// Anything else, by its line count.
58    Lines(u64),
59}
60
61impl Replaced {
62    /// What `target` holds now. A file that cannot be read counts no lines.
63    fn of(target: &fs_cap::PreparedTarget) -> Self {
64        match target {
65            fs_cap::PreparedTarget::Existing(existing) => {
66                Self::read(existing).unwrap_or(Self::Lines(0))
67            }
68            fs_cap::PreparedTarget::Absent(_) => Self::Nothing,
69        }
70    }
71
72    fn read(existing: &fs_cap::PreparedExisting) -> Result<Self, ToolError> {
73        let io = |error: std::io::Error| ToolError::execution(error.to_string());
74        let mut file = existing.open_file()?;
75        let mut head = Vec::new();
76        (&mut file)
77            .take(MAX_DIFF_BYTES as u64 + 1)
78            .read_to_end(&mut head)
79            .map_err(io)?;
80        if head.len() <= MAX_DIFF_BYTES {
81            return Ok(match String::from_utf8(head) {
82                Ok(text) => Self::Text(text),
83                Err(error) => Self::Lines(line_count(error.as_bytes())),
84            });
85        }
86        // Count the rest without holding it.
87        let mut newlines = memchr::memchr_iter(b'\n', &head).count() as u64;
88        let mut last = head.last().copied();
89        let mut chunk = vec![0; 64 * 1024];
90        loop {
91            let read = file.read(&mut chunk).map_err(io)?;
92            if read == 0 {
93                break;
94            }
95            newlines += memchr::memchr_iter(b'\n', &chunk[..read]).count() as u64;
96            last = Some(chunk[read - 1]);
97        }
98        Ok(Self::Lines(
99            newlines + u64::from(last.is_some_and(|byte| byte != b'\n')),
100        ))
101    }
102}
103
104/// The lines of `bytes`, a last line without a newline included.
105fn line_count(bytes: &[u8]) -> u64 {
106    memchr::memchr_iter(b'\n', bytes).count() as u64
107        + u64::from(bytes.last().is_some_and(|byte| *byte != b'\n'))
108}

110/// The diff of writing `new` over `replaced`, as of `as_of`: a line diff
111/// when both are text within [`MAX_DIFF_BYTES`], else the change and its
112/// line counts.
113fn write_diff(path: &Path, replaced: &Replaced, new: &[u8], as_of: DiffAsOf) -> DisplayBlock {
114    let change = match replaced {
115        Replaced::Nothing => DiffChange::Created,
116        Replaced::Text(_) | Replaced::Lines(_) => DiffChange::Modified,
117    };
118    let old = match replaced {
119        Replaced::Nothing => Some(""),
120        Replaced::Text(text) => Some(text.as_str()),
121        Replaced::Lines(_) => None,
122    };
123    let new_text = (new.len() <= MAX_DIFF_BYTES)
124        .then(|| std::str::from_utf8(new).ok())
125        .flatten();
126    if let (Some(old), Some(new_text)) = (old, new_text) {
127        let changes = display::line_diff_changes(old, new_text);
128        return DisplayBlock::Diff(display::diff_block(
129            path,
130            change,
131            &display::raw_lines(old),
132            &changes,
133            as_of,
134        ));
135    }
136    let removed = match replaced {
137        Replaced::Nothing => 0,
138        Replaced::Text(text) => line_count(text.as_bytes()),
139        Replaced::Lines(lines) => *lines,
140    };
141    DisplayBlock::Diff(DiffBlock {
142        path: path.display().to_string(),
143        old_path: None,
144        language: display::language_for_path(path),
145        change,
146        hunks: Vec::new(),
147        added: line_count(new),
148        removed,
149        omitted: None,
150        as_of,
151        full: None,
152    })
153}

257/// Prepares a write, with a preview of its diff against the file there now.
258fn prepare_write(tool: &WriteTool, args: WriteArgs) -> Result<PreparedTool, ToolError> {
259    let target = fs_cap::prepare_target(&tool.workspace, std::path::Path::new(&args.file_path))?;
260    if matches!(&target, fs_cap::PreparedTarget::Existing(existing) if existing.directory) {
261        return Err(ToolError::unsupported_security(
262            "write target is a directory",
263        ));
264    }
265    let normalized_arguments = serde_json::json!({
266        "filePath": target.display_path(),
267        "content": args.content,
268    });
269    let shown_path = tool.shown_path(&args);
270    let preview = display::fit_block(
271        write_diff(
272            &shown_path,
273            &Replaced::of(&target),
274            args.content.as_bytes(),
275            DiffAsOf::Prepare,
276        ),
277        None,
278    )?;
279    let presentation = tool.presentation_of(&args).with_preview([preview]);
280    // Writes to one location run one at a time, in call order.
281    let serialization_key = PreparedSerializationKey::new(target.serialization_key());
282    Ok(PreparedTool::declared(
283        normalized_arguments,
284        Some(serialization_key),
285        Box::new(WriteExecutor {
286            target,
287            bytes: args.content.into_bytes(),
288            shown_path,
289        }),
290    )?
291    .with_presentation(presentation))
292}

294#[async_trait]
295impl PreparedExecutor for WriteExecutor {
296    async fn execute(
297        self: Box<Self>,
298        context: ToolExecutionContext,
299    ) -> Result<cookie_agent_runtime_api::ToolCompletion, ToolError> {
300        if context.cancellation.is_cancelled() {
301            return Err(ToolError::execution(
302                "prepared write cancelled before commit",
303            ));
304        }
305        let artifacts = Arc::clone(&context.artifacts);
306        let session = context.session;
307        let (path, bytes, diff) = crate::blocking(move || {
308            let replaced = Replaced::of(&self.target);
309            self.target.write(&self.bytes)?;
310            let diff = display::fit_block(
311                write_diff(&self.shown_path, &replaced, &self.bytes, DiffAsOf::Execute),
312                Some(display::Retain::diff_of(
313                    artifacts.as_ref(),
314                    session,
315                    &self.shown_path,
316                )),
317            )?;
318            Ok((
319                self.target.display_path().to_owned(),
320                self.bytes.len(),
321                diff,
322            ))
323        })
324        .await?;
325        let path = path.display();
326        Ok(cookie_agent_runtime_api::ToolCompletion::single(
327            ToolResult {
328                display: vec![Open::Known(diff)],
329                background: None,
330                retained_output: None,
331                title: crate::safe_title(format!("Wrote {path}")),
332                output: format!("Wrote {bytes} bytes to {path}"),
333                metadata: serde_json::json!({ "bytes": bytes }),
334                attachments: Vec::new(),
335                additional_messages: Vec::new(),
336            },
337        ))
338    }
339}

689    fn only_diff(blocks: &[Open<DisplayBlock>]) -> &DiffBlock {

697    fn numbers(diff: &DiffBlock) -> Vec<(u64, u64, u64, u64)> {
698        diff.hunks
699            .iter()
700            .map(|hunk| {
701                (
702                    hunk.old_start,
703                    hunk.old_lines,
704                    hunk.new_start,
705                    hunk.new_lines,
706                )
707            })
708            .collect()
709    }

729        assert_eq!(

734        let result = prepared

736            .await
737            .expect("write");
738        let written = only_diff(&result.display);
739        assert_eq!(
740            (
741                written.change,
742                written.as_of,
743                written.added,
744                written.removed
745            ),
746            (DiffChange::Created, DiffAsOf::Execute, 2, 0)
747        );
748        assert_eq!(
749            written.hunks[0].lines,

805        let path = std::path::Path::new("data.bin");
806        let large = "x\n".repeat(super::MAX_DIFF_BYTES);
807        let DisplayBlock::Diff(diff) = super::write_diff(
808            path,
809            &super::Replaced::Text("a\nb\nc".into()),

811            DiffAsOf::Prepare,
812        ) else {
813            panic!("a diff");
814        };
815        assert!(diff.hunks.is_empty());
816        assert_eq!(
817            (diff.change, diff.added, diff.removed),

819        );
820        let DisplayBlock::Diff(diff) = super::write_diff(
821            path,
822            &super::Replaced::Lines(7),
823            b"text\n",
824            DiffAsOf::Execute,
825        ) else {
826            panic!("a diff");
827        };
828        assert!(diff.hunks.is_empty());

834            DiffAsOf::Execute,
835        ) else {
836            panic!("a diff");
837        };
838        assert!(diff.hunks.is_empty());
839        assert_eq!(
840            (diff.change, diff.added, diff.removed),

868    async fn parallel_same_file_writes_apply_in_call_order() {
869        use std::fs;
870
871        let root = tempfile::tempdir().expect("root");
872        fs::write(root.path().join("value.txt"), "old").expect("fixture");
873        let prepared = WriteTool::new(&crate::test_workspace(root.path()))
874            .prepare_parallel(
875                context(root.path()),
876                vec![
877                    write_call("value.txt", "one"),
878                    write_call("value.txt", "two"),
879                ],
880            )
881            .await;
882        let mut prepared = prepared.into_iter();
883        let first = prepared.next().expect("first").expect("first write");
884        let second = prepared.next().expect("second").expect("second write");
885        let tool = WriteTool::new(&crate::test_workspace(root.path()));
886        let first = crate::identified(&tool, "write", root.path(), first);
887        let second = crate::identified(&tool, "write", root.path(), second);
888        assert!(prepared.next().is_none());
889        assert_ne!(
890            OperationFingerprint::from_prepared_operation(first.operation()),
891            OperationFingerprint::from_prepared_operation(second.operation())
892        );
893
894        first
895            .execute_for_test(execution_context(root.path()))
896            .await
897            .expect("first write");
898        second
899            .execute_for_test(execution_context(root.path()))
900            .await
901            .expect("second write");
902        // The last write in call order wins deterministically.
903        assert_eq!(
904            fs::read_to_string(root.path().join("value.txt")).unwrap(),
905            "two"
906        );
907    }
