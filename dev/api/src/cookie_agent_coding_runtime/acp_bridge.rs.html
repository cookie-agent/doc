1//! The local sockets an ACP agent's MCP client reaches cookie's tools
2//! through (`docs/site/specs/acp-agents.md` section 9).
3//!
4//! A runtime opens one [`BridgeListener`] per ACP session: a Unix socket in
5//! a directory only the runtime's user may enter, or a named pipe on
6//! Windows that refuses remote clients. The agent runs
7//! `cookie runtime mcp-bridge --socket <path>` as a stdio MCP server, and
8//! [`run_bridge`] copies its standard input to the socket and the socket to
9//! its standard output, one MCP message per line. The listener is removed
10//! when it drops.
11
12use std::{io, path::Path};
13
14use tokio::io::{AsyncRead, AsyncWrite};
15
16/// One connection to a bridge socket.
17pub trait BridgeStream: AsyncRead + AsyncWrite + Send + Unpin {}
18
19impl<T: AsyncRead + AsyncWrite + Send + Unpin> BridgeStream for T {}
20
21/// A bridge socket the runtime listens on.
22pub struct BridgeListener {
23    path: String,
24    #[cfg(unix)]
25    listener: tokio::net::UnixListener,
26    /// The private directory the socket lives in, removed on drop.
27    #[cfg(unix)]
28    directory: std::path::PathBuf,
29    /// The pipe instance the next client connects to.
30    #[cfg(windows)]
31    next: tokio::sync::Mutex<tokio::net::windows::named_pipe::NamedPipeServer>,
32}
33
34impl BridgeListener {
35    /// Creates a socket with an unguessable name, reachable only by this
36    /// user.
37    pub fn bind() -> io::Result<Self> {
38        let mut bytes = [0_u8; 16];
39        getrandom::getrandom(&mut bytes).map_err(io::Error::other)?;
40        let name = bytes
41            .iter()
42            .map(|byte| format!("{byte:02x}"))
43            .collect::<String>();
44        #[cfg(unix)]
45        {
46            use std::os::unix::fs::DirBuilderExt as _;
47            let directory = std::env::temp_dir().join(format!("cookie-acp-{}", &name[..16]));
48            std::fs::DirBuilder::new().mode(0o700).create(&directory)?;
49            let socket = directory.join("mcp.sock");
50            let listener = match tokio::net::UnixListener::bind(&socket) {
51                Ok(listener) => listener,
52                Err(error) => {
53                    let _ = std::fs::remove_dir_all(&directory);
54                    return Err(error);
55                }
56            };
57            Ok(Self {
58                path: socket.to_string_lossy().into_owned(),
59                listener,
60                directory,
61            })
62        }
63        #[cfg(windows)]
64        {
65            let path = format!(r"\\.\pipe\cookie-acp-{name}");
66            let first = tokio::net::windows::named_pipe::ServerOptions::new()
67                .first_pipe_instance(true)
68                .reject_remote_clients(true)
69                .create(&path)?;
70            Ok(Self {
71                path,
72                next: tokio::sync::Mutex::new(first),
73            })
74        }
75    }
76
77    /// The path [`connect`] takes.
78    #[must_use]
79    pub fn path(&self) -> &str {
80        &self.path
81    }
82
83    /// The next client that connects.
84    pub async fn accept(&self) -> io::Result<Box<dyn BridgeStream>> {
85        #[cfg(unix)]
86        {
87            let (stream, _) = self.listener.accept().await?;
88            Ok(Box::new(stream))
89        }
90        #[cfg(windows)]
91        {
92            let mut next = self.next.lock().await;
93            next.connect().await?;
94            let following = tokio::net::windows::named_pipe::ServerOptions::new()
95                .reject_remote_clients(true)
96                .create(&self.path)?;
97            let connected = std::mem::replace(&mut *next, following);
98            Ok(Box::new(connected))
99        }
100    }
101}
102
103impl Drop for BridgeListener {
104    fn drop(&mut self) {
105        #[cfg(unix)]
106        {
107            let _ = std::fs::remove_dir_all(&self.directory);
108        }
109    }
110}
111
112/// Connects to the bridge socket at `path`.
113pub async fn connect(path: &Path) -> io::Result<Box<dyn BridgeStream>> {
114    #[cfg(unix)]
115    {
116        Ok(Box::new(tokio::net::UnixStream::connect(path).await?))
117    }
118    #[cfg(windows)]
119    {
120        Ok(Box::new(
121            tokio::net::windows::named_pipe::ClientOptions::new().open(path)?,
122        ))
123    }
124}
125
126/// `cookie runtime mcp-bridge --socket <path>`: copies standard input to
127/// the bridge socket and the socket to standard output until either ends.
128pub async fn run_bridge(path: &Path) -> io::Result<()> {
129    let stream = connect(path).await?;
130    let (mut socket_read, mut socket_write) = tokio::io::split(stream);
131    let mut stdin = tokio::io::stdin();
132    let mut stdout = tokio::io::stdout();
133    let upstream = async {
134        tokio::io::copy(&mut stdin, &mut socket_write).await?;
135        tokio::io::AsyncWriteExt::shutdown(&mut socket_write).await
136    };
137    let downstream = async {
138        tokio::io::copy(&mut socket_read, &mut stdout).await?;
139        tokio::io::AsyncWriteExt::flush(&mut stdout).await
140    };
141    tokio::select! {
142        result = upstream => result,
143        result = downstream => result,
144    }
145}
