1//! Restarting the daemon (`docs/site/specs/engine-configuration.md`
2//! section 11): `cookie daemon restart`, and how `cookie daemon` relaunches
3//! itself once a client asked it to.
4//!
5//! The daemon remembers how it was started — its executable, arguments,
6//! environment, and directory — before it changes any of them. When a
7//! scheduled restart is due it shuts down as on a stop signal and then
8//! relaunches that command:
9//!
10//! - on Unix with an in-place `exec`, which keeps the process ID, so a
11//!   service manager sees no change. Every descriptor the daemon opened is
12//!   close-on-exec (Rust opens them so, and [`relaunch`] marks any other
13//!   before the `exec`), so the new image starts with only standard input,
14//!   output, and error, and binds the port the old one released;
15//! - on Windows by starting the same command line and exiting. The new
16//!   process finds [`RESTARTED_ENV`] and retries binding its port for up to
17//!   [`RESTART_BIND_WAIT`] while the old one exits.
18//!
19//! The executable is the path the daemon resolved when it started, so a
20//! restart after an upgrade runs the new build.
21

33/// Set in the environment of the process a restart starts, which removes it
34/// at once: the daemon then retries binding its port while the process it
35/// replaces lets go of it.
36pub(crate) const RESTARTED_ENV: &str = "COOKIE_DAEMON_RESTARTED";
37
38/// How long a restarted daemon retries binding a port still in use.
39pub(crate) const RESTART_BIND_WAIT: Duration = Duration::from_secs(30);
40
41/// How long `cookie daemon restart` waits for the daemon to come back once
42/// it closed the connection.
43const RECONNECT_WAIT: Duration = Duration::from_secs(60);
44
45#[derive(Debug, PartialEq, Eq, Subcommand)]
46pub(crate) enum DaemonCommand {
47    /// Restart the daemon once no turn is running, or at once with `--now`.
48    /// It relaunches its own executable with the same arguments and
49    /// environment, applying the configuration that needs a restart and a
50    /// binary upgraded on disk.
51    Restart {
52        /// Restart at once, interrupting running turns.
53        #[arg(long)]
54        now: bool,
55        /// The daemon's client URL; `[server] url` in this machine's
56        /// `config.toml`, or the local daemon, when omitted.
57        #[arg(long)]
58        url: Option<String>,
59        /// The daemon's bearer token; `[server] token` in this machine's
60        /// `config.toml` when omitted.
61        #[arg(
62            long,
63            env = "COOKIE_DAEMON_TOKEN",
64            hide_env_values = true,
65            allow_hyphen_values = true
66        )]
67        token: Option<String>,
68    },
69}

71/// How this process was started, to start it again.
72#[derive(Debug)]
73pub(crate) struct Relaunch {
74    executable: PathBuf,
75    arguments: Vec<OsString>,
76    environment: Vec<(OsString, OsString)>,
77    directory: PathBuf,
78}
79
80impl Relaunch {
81    /// This process's command, before anything changes it; `None` when the
82    /// executable or the directory cannot be found, and the daemon then
83    /// refuses to restart. [`RESTARTED_ENV`] is left out of the environment.
84    pub(crate) fn capture() -> Option<Self> {
85        let executable = std::env::current_exe().ok()?;
86        let directory = std::env::current_dir().ok()?;
87        Some(Self {
88            executable,
89            arguments: std::env::args_os().collect(),
90            environment: std::env::vars_os()
91                .filter(|(name, _)| name != RESTARTED_ENV)
92                .collect(),
93            directory,
94        })
95    }
96
97    fn command(&self) -> std::process::Command {
98        let mut command = std::process::Command::new(&self.executable);
99        command
100            .args(self.arguments.iter().skip(1))
101            .env_clear()
102            .envs(self.environment.iter().map(|(name, value)| (name, value)))
103            .env(RESTARTED_ENV, "1")
104            .current_dir(&self.directory);
105        command
106    }
107}

109/// Starts the daemon again as `relaunch` says, replacing this process. Only
110/// returns, with the reason, when that fails.
111#[cfg(unix)]
112pub(crate) fn relaunch(relaunch: &Relaunch) -> anyhow::Error {
113    use std::os::unix::process::CommandExt as _;
114    flush_standard_streams();
115    mark_descriptors_close_on_exec();
116    let mut command = relaunch.command();
117    if let Some(name) = relaunch.arguments.first() {
118        command.arg0(name);
119    }
120    let error = command.exec();
121    anyhow::Error::new(error).context(format!("relaunch {}", relaunch.executable.display()))
122}
123
124/// Starts the daemon again as `relaunch` says and exits; only returns, with
125/// the reason, when starting it fails.
126#[cfg(windows)]
127pub(crate) fn relaunch(relaunch: &Relaunch) -> anyhow::Error {
128    flush_standard_streams();
129    match relaunch.command().spawn() {
130        Ok(_) => std::process::exit(0),
131        Err(error) => {
132            anyhow::Error::new(error).context(format!("relaunch {}", relaunch.executable.display()))
133        }
134    }
135}
136
137fn flush_standard_streams() {
138    use std::io::Write as _;
139    let _ = std::io::stdout().flush();
140    let _ = std::io::stderr().flush();
141}

143/// Marks every descriptor above standard error close-on-exec, so nothing
144/// the daemon held open, such as a socket a library opened without the
145/// flag, survives into the relaunched image.
146#[cfg(unix)]
147fn mark_descriptors_close_on_exec() {
148    let listing = if cfg!(target_os = "linux") {
149        "/proc/self/fd"
150    } else {
151        "/dev/fd"
152    };
153    let Ok(entries) = std::fs::read_dir(listing) else {
154        return;
155    };
156    let descriptors = entries
157        .filter_map(|entry| {
158            entry
159                .ok()?
160                .file_name()
161                .to_str()?
162                .parse::<libc::c_int>()
163                .ok()
164        })
165        .filter(|descriptor| *descriptor > libc::STDERR_FILENO)
166        .collect::<Vec<_>>();
167    for descriptor in descriptors {
168        // SAFETY: `F_GETFD` and `F_SETFD` read and set one descriptor's flags
169        // and touch no memory; a descriptor closed meanwhile (the listing's
170        // own) fails harmlessly.
171        unsafe {
172            let flags = libc::fcntl(descriptor, libc::F_GETFD);
173            if flags >= 0 && flags & libc::FD_CLOEXEC == 0 {
174                libc::fcntl(descriptor, libc::F_SETFD, flags | libc::FD_CLOEXEC);
175            }
176        }
177    }
178}

180/// Binds with `bind` until it succeeds, an error other than the address
181/// being in use, or `wait` passes.
182pub(crate) async fn bind_retrying<T, F, Fut>(
183    wait: Duration,
184    mut bind: F,
185) -> Result<T, cookie_agent_server::ServerError>
186where
187    F: FnMut() -> Fut,
188    Fut: std::future::Future<Output = Result<T, cookie_agent_server::ServerError>>,
189{
190    let deadline = Instant::now() + wait;
191    loop {
192        match bind().await {
193            Err(cookie_agent_server::ServerError::Listen(error))
194                if error.kind() == std::io::ErrorKind::AddrInUse && Instant::now() < deadline =>
195            {
196                tokio::time::sleep(Duration::from_millis(100)).await;
197            }
198            result => return result,
199        }
200    }
201}

203/// What the daemon answered, as a line.
204pub(crate) fn scheduled_line(when: RestartWhen, result: &DaemonRestartResult) -> String {
205    let waiting = waiting_for(result.active_runs, result.background_calls);
206    match (when, waiting.is_empty()) {
207        (_, true) => "The daemon is restarting.".to_owned(),
208        (RestartWhen::Now, false) => {
209            format!("The daemon is restarting now, interrupting {waiting}.")
210        }
211        (_, false) => format!(
212            "The daemon restarts once {waiting} finish{}. Turns started meanwhile are waited for too.",
213            if result.active_runs + result.background_calls == 1 {
214                "es"
215            } else {
216                ""
217            }
218        ),
219    }
220}

222/// `cookie daemon restart`: asks the daemon to restart, then waits until it
223/// is back.
224pub(crate) async fn run_restart(url: &str, token: &str, now: bool) -> anyhow::Result<()> {
225    let client = Client::connect_websocket_with_token(url, token).await?;
226    let mut deliveries = client
227        .subscribe_deliveries()
228        .context("the connection's deliveries")?;
229    let hello = client
230        .handshake_as(crate::client_identity("cli"))
231        .await
232        .context("handshake with daemon")?;
233    let when = if now {
234        RestartWhen::Now
235    } else {
236        RestartWhen::Idle
237    };
238    let result = client
239        .restart_daemon(when)
240        .await
241        .context("restart the daemon")?;
242    println!("{}", scheduled_line(when, &result));
243    let mut waiting_for_now = (result.active_runs, result.background_calls);
244    if waiting_for_now != (0, 0) && when == RestartWhen::Idle {
245        println!(
246            "Waiting; Ctrl-C stops waiting, and the restart stays scheduled \
247             (`cookie daemon restart --now` restarts at once)."
248        );
249    }
250    while let Some(delivery) = deliveries.recv().await {
251        match delivery {
252            ClientDelivery::DaemonRestarting(restarting) => {
253                let now = (restarting.active_runs, restarting.background_calls);
254                if now != waiting_for_now && now != (0, 0) {
255                    waiting_for_now = now;
256                    println!("Waiting for {}.", waiting_for(now.0, now.1));
257                }
258            }
259            ClientDelivery::Disconnected { .. } => break,
260            _ => {}
261        }
262    }
263    drop(client);
264    let deadline = Instant::now() + RECONNECT_WAIT;
265    loop {
266        tokio::time::sleep(Duration::from_millis(250)).await;
267        match reconnect(url, token).await {
268            Ok(engine_version) => {
269                if engine_version == hello.engine_version {
270                    println!("The daemon is back.");
271                } else {
272                    println!(
273                        "The daemon is back, now version {engine_version} (was {}).",
274                        hello.engine_version
275                    );
276                }
277                return Ok(());
278            }
279            Err(error) if Instant::now() >= deadline => {
280                return Err(error.context(format!(
281                    "the daemon did not come back within {} seconds",
282                    RECONNECT_WAIT.as_secs()
283                )));
284            }
285            Err(_) => {}
286        }
287    }
288}

290/// Connects again; the daemon's version.
291async fn reconnect(url: &str, token: &str) -> anyhow::Result<String> {
292    let client = Client::connect_websocket_with_token(url, token).await?;
293    let hello = client.handshake_as(crate::client_identity("cli")).await?;
294    client.shutdown();
295    Ok(hello.engine_version)
296}

306    #[test]
307    fn the_answer_says_what_the_restart_waits_for() {
308        let result = |active_runs| DaemonRestartResult {
309            scheduled: true,
310            active_runs,
311            background_calls: 0,
312        };
313        assert_eq!(
314            scheduled_line(RestartWhen::Idle, &result(0)),
315            "The daemon is restarting."
316        );
317        assert_eq!(
318            scheduled_line(RestartWhen::Idle, &result(1)),
319            "The daemon restarts once 1 running turn finishes. Turns started meanwhile are waited for too."
320        );
321        assert_eq!(
322            scheduled_line(RestartWhen::Idle, &result(3)),
323            "The daemon restarts once 3 running turns finish. Turns started meanwhile are waited for too."
324        );
325        assert_eq!(
326            scheduled_line(RestartWhen::Now, &result(2)),
327            "The daemon is restarting now, interrupting 2 running turns."
328        );
329        let background = DaemonRestartResult {
330            scheduled: true,
331            active_runs: 1,
332            background_calls: 2,
333        };
334        assert_eq!(
335            scheduled_line(RestartWhen::Idle, &background),
336            "The daemon restarts once 1 running turn, 2 background calls finish. Turns started \
337             meanwhile are waited for too."
338        );
339        assert_eq!(
340            scheduled_line(

342                &DaemonRestartResult {
343                    scheduled: true,
344                    active_runs: 0,
345                    background_calls: 1,
346                }
347            ),
348            "The daemon restarts once 1 background call finishes. Turns started meanwhile are \
349             waited for too."

351    }

354        tokio::net::TcpListener::bind(("127.0.0.1", port))

359    /// A restarted daemon waits for its port to free; a daemon started
360    /// anew fails at once on a port in use.
361    #[tokio::test]
362    async fn binding_retries_a_port_in_use_only_while_waiting() {
363        let held = std::net::TcpListener::bind("127.0.0.1:0").expect("hold a port");
364        let port = held.local_addr().expect("address").port();
365        let refused = bind_retrying(Duration::ZERO, || bind(port)).await;
366        assert!(
367            matches!(&refused, Err(cookie_agent_server::ServerError::Listen(error))
368                if error.kind() == std::io::ErrorKind::AddrInUse),
369            "{refused:?}"
370        );
371        let release = std::thread::spawn(move || {
372            std::thread::sleep(Duration::from_millis(300));
373            drop(held);
374        });
375        bind_retrying(Duration::from_secs(10), || bind(port))
376            .await
377            .expect("bound once the port freed");
378        release.join().expect("released");
379    }
