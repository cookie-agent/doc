1//! `cookie share`: list, make, and revoke a daemon's share links with the
2//! `share.*` methods, so it works against a remote daemon
3//! (`docs/site/specs/share-links.md` section 7.1).

17#[derive(Debug, PartialEq, Eq, Subcommand)]
18pub(crate) enum ShareCommand {
19    /// List the live links: ID, title, expiry, views, and address.
20    List,
21    /// Make a link to an asset and print its address. An asset that
22    /// already has a live link prints that one, unless `--new`.
23    Create {
24        /// The asset's handle, such as `asset_3f9a1c2e`, or its UUID.
25        asset: String,
26        /// Look for the handle among this session tree's own assets.
27        #[arg(long, value_name = "SESSION_ID", conflicts_with = "project")]
28        session: Option<SessionId>,
29        /// Look for the handle among this project's assets.
30        #[arg(long, value_name = "PROJECT_ID")]
31        project: Option<ProjectId>,
32        /// How long the link works: `1d`, `7d`, `30d`, or `never`.
33        #[arg(long, default_value = "7d", value_parser = parse_lifetime)]
34        expires: ShareLifetime,
35        /// Make a new link even when the asset has one.
36        #[arg(long)]
37        new: bool,
38    },
39    /// Make a link that lets whoever opens it use a root session as a
40    /// guest, and print its address. Its guests use the session's tools,
41    /// machines, and approvals, on your provider account.
42    Session {
43        /// The root session.
44        session: SessionId,
45        /// Who the link is for; messages sent through it show this.
46        #[arg(long)]
47        label: Option<String>,
48        /// Its guests can read the conversation and not act in it.
49        #[arg(long)]
50        watch_only: bool,
51        /// How long the link works: `1d`, `7d`, `30d`, or `never`.
52        #[arg(long, default_value = "7d", value_parser = parse_lifetime)]
53        expires: ShareLifetime,
54    },
55    /// Revoke a link, or every link with `--all`; it stops working at
56    /// once.
57    Revoke {
58        #[arg(value_parser = parse_share_id, required_unless_present = "all")]
59        id: Option<ShareId>,
60        #[arg(long, conflicts_with = "id")]
61        all: bool,
62    },
63}

65fn parse_lifetime(value: &str) -> Result<ShareLifetime, String> {
66    ShareLifetime::from_code(value).ok_or_else(|| "expected 1d, 7d, 30d, or never".to_owned())
67}
68
69fn parse_share_id(value: &str) -> Result<ShareId, String> {
70    ShareId::new(value).map_err(|error| error.to_string())
71}

73pub(crate) async fn run_share(url: &str, token: &str, command: ShareCommand) -> anyhow::Result<()> {
74    let client = Client::connect_websocket_with_token(url, token).await?;
75    super::handshake_ready(&client, super::client_identity("cli")).await?;
76    let page = cookie_agent_client_state::shares::page_base(url).unwrap_or_default();
77    run_share_with(&client, &page, command, &mut super::memory::StdioMemoryIo).await
78}

80/// When a link stops working.
81fn expiry_line(expires_at: Option<Timestamp>, zone: &TimeZone) -> String {
82    expires_at.map_or_else(
83        || "never expires".to_owned(),
84        |at| {
85            format!(
86                "expires {}",
87                at.to_zoned(zone.clone()).strftime("%Y-%m-%d %H:%M")
88            )
89        },
90    )
91}

93/// One line of a conversation share: its ID, session, label, what it
94/// allows, expiry, guests, and address.
95fn session_share_line(share: &SessionShare, page: &str, zone: &TimeZone) -> String {
96    format!(
97        "{}  {:?}  for {:?}  {}  {}  {} connected  {}",
98        share.id,
99        share.title.as_deref().unwrap_or("session"),
100        share.label,
101        if share.can_send {
102            "can send"
103        } else {
104            "watch only"
105        },
106        expiry_line(share.expires_at, zone),
107        share.connected,
108        cookie_agent_client_state::shares::link_address(share.url.as_deref(), &share.path, page),
109    )
110}
111
112/// One line of a link: its ID, title, expiry, views, and address.
113fn share_line(share: &Share, page: &str, zone: &TimeZone) -> String {
114    let expiry = expiry_line(share.expires_at, zone);
115    format!(
116        "{}  {:?}  {expiry}  {} view{}  {}",
117        share.id,
118        share.title,
119        share.views,
120        if share.views == 1 { "" } else { "s" },
121        share_address(share, page)
122    )
123}

125/// The asset `reference` names, by handle or UUID, among the assets of
126/// `store` or, without one, of every store.
127async fn find_asset(
128    client: &Client,
129    reference: &str,
130    store: Option<&AssetStore>,
131) -> anyhow::Result<Asset> {
132    let assets = match store {
133        Some(store) => {
134            client
135                .list_assets(store.clone())
136                .await
137                .context("assets.list failed")?
138                .assets
139        }
140        None => {
141            client
142                .search_assets(AssetsSearchParams::default())
143                .await
144                .context("assets.search failed")?
145                .assets
146        }
147    };
148    let matches = assets
149        .into_iter()
150        .filter(|asset| asset.handle.as_str() == reference || asset.id.to_string() == reference)
151        .collect::<Vec<_>>();
152    match matches.as_slice() {
153        [asset] => Ok(asset.clone()),
154        [] => anyhow::bail!("there is no asset {reference:?}"),
155        several => anyhow::bail!(
156            "{reference:?} names {} assets; pick one with --session or --project:\n{}",
157            several.len(),
158            several
159                .iter()
160                .map(|asset| format!("  {} in {:?}", asset.summary(), asset.store))
161                .collect::<Vec<_>>()
162                .join("\n")
163        ),
164    }
165}

167pub(crate) async fn run_share_with<I: MemoryIo>(
168    client: &Client,
169    page: &str,
170    command: ShareCommand,
171    io: &mut I,
172) -> anyhow::Result<()> {
173    anyhow::ensure!(
174        daemon_has_shares(client.daemon_version()),
175        "this daemon has no share links: it needs client protocol 1.86 or later"
176    );
177    let zone = TimeZone::system();
178    match command {
179        ShareCommand::List => {
180            let listed = client
181                .list_shares(SharesListParams::default())
182                .await
183                .context("shares.list failed")?;
184            if !listed.enabled {
185                io.write_line("Share links are off ([share] enabled = false): no link works.")?;
186            }
187            if listed.shares.is_empty() && listed.sessions.is_empty() {
188                return io.write_line("No share links.");
189            }
190            for share in &listed.sessions {
191                io.write_line(&session_share_line(share, page, &zone))?;
192            }
193            for share in &listed.shares {
194                io.write_line(&share_line(share, page, &zone))?;
195            }
196            Ok(())
197        }
198        ShareCommand::Create {
199            asset,
200            session,
201            project,
202            expires,
203            new,
204        } => {
205            let store = match (session, project) {
206                (Some(session), _) => Some(AssetStore::Session { session }),
207                (None, Some(project)) => Some(AssetStore::Project { project }),
208                (None, None) => None,
209            };
210            let asset = find_asset(client, asset.trim(), store.as_ref()).await?;
211            let made = client
212                .create_share(ShareCreateParams {
213                    store: asset.store.clone(),
214                    asset: asset.id,
215                    expiry: expires.expiry(Timestamp::now()),
216                    reuse: !new,
217                })
218                .await
219                .map_err(anyhow::Error::new)?;
220            if !made.created {
221                io.write_line(&format!(
222                    "{} already has a live link (pass --new for another):",
223                    asset.handle
224                ))?;
225            }
226            io.write_line(&share_line(&made.share, page, &zone))
227        }
228        ShareCommand::Session {
229            session,
230            label,
231            watch_only,
232            expires,
233        } => {
234            let made = client
235                .create_session_share(SessionShareCreateParams {
236                    session_id: session,
237                    label,
238                    can_send: !watch_only,
239                    expiry: expires.expiry(Timestamp::now()),
240                })
241                .await
242                .map_err(anyhow::Error::new)?;
243            io.write_line(&session_share_line(&made.share, page, &zone))
244        }
245        ShareCommand::Revoke { id: Some(id), .. } => {
246            client
247                .revoke_share(id.clone())
248                .await
249                .map_err(anyhow::Error::new)?;
250            io.write_line(&format!("Revoked {id}."))
251        }
252        ShareCommand::Revoke { id: None, .. } => {
253            let revoked = client
254                .revoke_all_shares()
255                .await
256                .map_err(anyhow::Error::new)?
257                .revoked;
258            io.write_line(&format!(
259                "Revoked {revoked} link{}.",
260                if revoked == 1 { "" } else { "s" }
261            ))
262        }
263    }
264}
