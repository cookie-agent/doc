16/// The daemon's tokens: `[server] token`, which clients present, and
17/// `[server] runtime_token`, which dial-in runtimes may present instead.
18pub(super) struct DaemonTokens {
19    pub(super) token: Zeroizing<String>,
20    pub(super) runtime_token: Zeroizing<String>,
21    /// The token was created by this start, not read from the configuration.
22    pub(super) token_created: bool,
23    /// The runtime token was created by this start.
24    pub(super) runtime_token_created: bool,
25}
26
27/// The daemon's tokens from its configuration, with a fresh one written
28/// there for each that is missing, so they stay the same across restarts.
29pub(super) fn daemon_tokens(
30    server: &cookie_agent_config::ServerConfig,
31    config_path: Option<&Path>,
32) -> anyhow::Result<DaemonTokens> {
33    let fresh = |configured: &Option<String>| -> anyhow::Result<Option<Zeroizing<String>>> {
34        match configured {
35            Some(_) => Ok(None),
36            None => generate_token()
37                .map(Some)
38                .context("generate daemon authentication token"),
39        }
40    };
41    let new_token = fresh(&server.token)?;
42    let new_runtime_token = fresh(&server.runtime_token)?;
43    if new_token.is_some() || new_runtime_token.is_some() {
44        let path = config_path.context("no user config.toml to keep the daemon tokens in")?;
45        cookie_agent_config::write_server_tokens(
46            path,
47            new_token.as_deref().map(String::as_str),
48            new_runtime_token.as_deref().map(String::as_str),
49        )
50        .with_context(|| format!("keep the daemon tokens in {}", path.display()))?;
51    }
52    let token_created = new_token.is_some();
53    let runtime_token_created = new_runtime_token.is_some();
54    Ok(DaemonTokens {
55        token: new_token
56            .or_else(|| server.token.clone().map(Zeroizing::new))
57            .context("a daemon token")?,
58        runtime_token: new_runtime_token
59            .or_else(|| server.runtime_token.clone().map(Zeroizing::new))
60            .context("a runtime token")?,
61        token_created,
62        runtime_token_created,
63    })
64}

66/// What the daemon prints after its ready line about its web page: where
67/// it is.
68pub(super) fn web_page_line(client_url: &str) -> Option<String> {
69    cookie_agent_server::page_url(client_url)
70        .map(|page| format!("web page at {page} (`cookie web` prints it with the token)"))
71}

73/// `cookie daemon`: listens on `port` or `listen`, over TLS with `tls`
74/// (certificate, key), then composes the daemon over `workspace` and serves
75/// it until a stop signal or a restart, which `launch` then carries out.
76/// Clients connect while it composes and see what it is doing
77/// (`docs/site/specs/daemon-startup.md`).
78pub(super) async fn serve(
79    workspace: &Path,
80    port: Option<u16>,
81    listen: Option<std::net::SocketAddr>,
82    tls: Option<(PathBuf, PathBuf)>,
83    launch: &mut Launch,
84) -> anyhow::Result<()> {
85    let tls = match tls {
86        Some((cert, key)) => {
87            Some(cookie_agent_server::load_tls(&cert, &key).context("load TLS identity")?)
88        }
89        None => None,
90    };
91    let configuration =
92        cookie_agent_config::load().context("load user configuration, agents, and skills")?;
93    let tokens = daemon_tokens(
94        &configuration.runtime.server,
95        configuration.config_paths.user.as_deref(),
96    )?;
97    let address = listen.unwrap_or_else(|| {
98        std::net::SocketAddr::from((
99            [127, 0, 0, 1],
100            port.unwrap_or(configuration.runtime.server.port),
101        ))
102    });
103    let exit = run_daemon(
104        workspace,
105        configuration,
106        DaemonListen {
107            address,
108            tls,
109            tokens,
110        },
111        launch,
112    )
113    .await?;
114    launch.relaunch_now = matches!(exit, DaemonExit::Restart);
115    Ok(())
116}

118/// Where and how `cookie daemon` listens.
119pub(super) struct DaemonListen {
120    address: std::net::SocketAddr,
121    tls: Option<cookie_agent_server::TlsConfig>,
122    tokens: DaemonTokens,
123}

125/// Why the daemon stopped.
126pub(super) enum DaemonExit {
127    /// A stop signal.
128    Stopped,
129    /// A client's `daemon.restart`: relaunch.
130    Restart,
131}

133/// How long a daemon whose startup failed waits for the clients connected
134/// meanwhile to hear why before it exits.
135const FAILED_START_NOTICE: Duration = Duration::from_secs(5);
136
137pub(super) async fn run_daemon(
138    workspace: &Path,
139    configuration: cookie_agent_config::LoadedConfiguration,
140    options: DaemonListen,
141    launch: &Launch,
142) -> anyhow::Result<DaemonExit> {
143    let DaemonListen {
144        address,
145        tls,
146        tokens:
147            DaemonTokens {
148                token,
149                runtime_token,
150                token_created,
151                runtime_token_created,
152            },
153    } = options;
154    let scheme = if tls.is_some() { "wss" } else { "ws" };
155    // A restarted daemon waits for the process it replaces to let go of the
156    // port.
157    let bind_wait = if launch.restarted {
158        daemon_restart::RESTART_BIND_WAIT
159    } else {
160        Duration::ZERO
161    };
162    let listener = cookie_agent_server::DaemonListener::new(
163        cookie_agent_engine::status::StatusBoard::new(cookie_agent_protocol::DaemonPhase::Starting),
164    )
165    .with_page_url(configuration.runtime.server.page_url.clone())
166    .with_runtime_token(runtime_token.clone())
167    .with_share_url(
168        configuration
169            .runtime
170            .share
171            .url
172            .clone()
173            .or_else(|| configuration.runtime.server.page_url.clone()),
174    );
175    let share_listen = configuration.runtime.share.listen;
176    let mut signals =
177        cookie_agent_protocol::StopSignals::new().context("listen for stop signals")?;
178    let running = daemon_restart::bind_retrying(bind_wait, || {
179        listener.serve_on(address, token.clone(), tls.clone())
180    })
181    .await
182    .map_err(|error| anyhow::Error::new(error).context("start WebSocket daemon"))?;
183    let url = format!("{scheme}://{}/ws", running.address());
184    // Launchers wait for this line: the daemon takes connections from here
185    // on, and tells them it is starting until it is ready.
186    println!("{}", ready_line(&url));
187    println!(
188        "cookie daemon listening on {url} (client protocol {}, process {})",
189        cookie_agent_protocol::CLIENT_PROTOCOL_VERSION,
190        std::process::id()
191    );
192    if let Some(line) = web_page_line(&url) {
193        println!("{line}");
194    }
195    // The share-only listener (`[share] listen`): links answer as gone
196    // until the daemon is ready.
197    let shares = match share_listen {
198        Some(address) => match listener.serve_shares_on(address).await {
199            Ok(shares) => {
200                println!("share links served alone at http://{}/s/", shares.address());
201                Some(shares)
202            }
203            Err(error) => {
204                listener.shutdown();
205                running.wait().await;
206                return Err(anyhow::Error::new(error)
207                    .context(format!("listen for share links on {address}")));
208            }
209        },
210        None => None,
211    };
212    let compose = async {
213        #[cfg(debug_assertions)]
214        hold_start_for_test(listener.board()).await;
215        super::compose::compose_for_daemon(workspace, configuration, &listener).await
216    };
217    let composed = tokio::select! {
218        composed = compose => composed,
219        () = signals.recv() => {
220            println!("cookie daemon stopped while starting");
221            listener.shutdown();
222            running.wait().await;
223            if let Some(shares) = shares {
224                shares.wait().await;
225            }
226            return Ok(DaemonExit::Stopped);
227        }
228    };
229    let mut runtime = match composed {
230        Ok(runtime) => runtime,
231        Err(error) => {
232            listener
233                .fail(&format!("{error:#}"), FAILED_START_NOTICE)
234                .await;
235            running.wait().await;
236            if let Some(shares) = shares {
237                shares.wait().await;
238            }
239            return Err(error);
240        }
241    };
242    runtime.engine.note_served_server_tokens(
243        token_created.then_some(token.as_str()),
244        runtime_token_created.then_some(runtime_token.as_str()),
245    );
246    drop(token);
247    drop(runtime_token);
248    // Only a daemon that knows how it was started can start itself again.
249    let restart = CancellationToken::new();
250    if launch.relaunch.is_some() {
251        let restart = restart.clone();
252        runtime
253            .engine
254            .set_restart_hook(Box::new(move || restart.cancel()));
255    }
256    listener.hand_over(std::sync::Arc::clone(&runtime.server));
257    println!("cookie daemon ready");
258    // Phones with push notifications hear the daemon is up again.
259    runtime.server.announce_started();
260    let exit = tokio::select! {
261        () = signals.recv() => DaemonExit::Stopped,
262        () = restart.cancelled() => DaemonExit::Restart,
263    };
264    exit_on_next_stop(signals);
265    match exit {
266        DaemonExit::Stopped => {
267            runtime.server.shutdown();
268            runtime.stop_catalog_refresh().await;
269            running.wait().await;
270            if let Some(shares) = shares {
271                shares.wait().await;
272            }
273            runtime.engine.shutdown().await;
274        }
275        DaemonExit::Restart => {
276            println!("cookie daemon restarting");
277            // Before anything stops: phones with push notifications hear it,
278            // but the one whose page asked.
279            runtime.server.announce_restarting().await;
280            // The engine stops first, while clients are still connected:
281            // they see the interrupted turns end, then the connection close
282            // saying the daemon is restarting.
283            runtime.stop_catalog_refresh().await;
284            runtime.engine.shutdown().await;
285            runtime.server.shutdown_for_restart();
286            // Returns once every client connection sent that close frame
287            // (bounded), so the relaunch does not cut it off.
288            running.wait().await;
289            if let Some(shares) = shares {
290                shares.wait().await;
291            }
292        }
293    }
294    Ok(exit)
295}

297/// Holds the start in debug builds, as a blocking task, so a test connects
298/// to a daemon that is still starting: for `COOKIE_AGENT_TEST_STARTUP_HOLD_MS`,
299/// and while the file `COOKIE_AGENT_TEST_STARTUP_HOLD_FILE` names exists,
300/// which a test creates before it restarts the daemon.
301#[cfg(debug_assertions)]
302async fn hold_start_for_test(board: &cookie_agent_engine::status::StatusBoard) {
303    let hold = std::env::var("COOKIE_AGENT_TEST_STARTUP_HOLD_MS")
304        .ok()
305        .and_then(|millis| millis.parse().ok())
306        .map(Duration::from_millis);
307    let file = std::env::var_os("COOKIE_AGENT_TEST_STARTUP_HOLD_FILE")
308        .map(PathBuf::from)
309        .filter(|file| file.exists());
310    if hold.is_none() && file.is_none() {
311        return;
312    }
313    let _task = board.start("Holding the start for a test", true);
314    if let Some(hold) = hold {
315        tokio::time::sleep(hold).await;
316    }
317    if let Some(file) = file {
318        while file.exists() {
319            tokio::time::sleep(Duration::from_millis(100)).await;
320        }
321    }
322}

324/// Exits at once on the next stop signal, for a graceful stop that hangs.
325fn exit_on_next_stop(mut signals: cookie_agent_protocol::StopSignals) {
326    tokio::spawn(async move {
327        signals.recv().await;
328        eprintln!("cookie: stopping at once");
329        std::process::exit(1);
330    });
331}
332
333/// Cancels `stop` on the first stop signal; see [`exit_on_next_stop`] for
334/// the second.
335pub(super) fn stop_on_signal(stop: CancellationToken) -> anyhow::Result<()> {
336    let mut signals =
337        cookie_agent_protocol::StopSignals::new().context("listen for stop signals")?;
338    tokio::spawn(async move {
339        signals.recv().await;
340        stop.cancel();
341        exit_on_next_stop(signals);
342    });
343    Ok(())
