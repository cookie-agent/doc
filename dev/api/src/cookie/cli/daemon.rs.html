16/// The daemon's token: `[server] token` from its configuration, or a fresh
17/// one written there when there is none, so it stays the same across
18/// restarts.
19pub(super) fn daemon_token(
20    configured: Option<String>,
21    config_path: Option<&Path>,
22) -> anyhow::Result<Zeroizing<String>> {
23    if let Some(token) = configured {
24        return Ok(Zeroizing::new(token));
25    }
26    let token = generate_token().context("generate daemon authentication token")?;
27    let path = config_path.context("no user config.toml to keep the daemon token in")?;
28    cookie_agent_config::write_server_token(path, &token)
29        .with_context(|| format!("keep the daemon token in {}", path.display()))?;
30    Ok(token)
31}

33/// What the daemon prints after its ready line about its web page: where
34/// it is.
35pub(super) fn web_page_line(client_url: &str) -> Option<String> {
36    cookie_agent_server::page_url(client_url)
37        .map(|page| format!("web page at {page} (`cookie web` prints it with the token)"))
38}

40/// `cookie daemon`: listens on `port` or `listen`, over TLS with `tls`
41/// (certificate, key), then composes the daemon over `workspace` and serves
42/// it until a stop signal or a restart, which `launch` then carries out.
43/// Clients connect while it composes and see what it is doing
44/// (`docs/site/specs/daemon-startup.md`).
45pub(super) async fn serve(
46    workspace: &Path,
47    port: Option<u16>,
48    listen: Option<std::net::SocketAddr>,
49    tls: Option<(PathBuf, PathBuf)>,
50    launch: &mut Launch,
51) -> anyhow::Result<()> {
52    let tls = match tls {
53        Some((cert, key)) => {
54            Some(cookie_agent_server::load_tls(&cert, &key).context("load TLS identity")?)
55        }
56        None => None,
57    };
58    let configuration =
59        cookie_agent_config::load().context("load user configuration, agents, and skills")?;
60    let configured_token = configuration.runtime.server.token.clone();
61    let token = daemon_token(
62        configured_token.clone(),
63        configuration.config_paths.user.as_deref(),
64    )?;
65    let address = listen.unwrap_or_else(|| {
66        std::net::SocketAddr::from((
67            [127, 0, 0, 1],
68            port.unwrap_or(configuration.runtime.server.port),
69        ))
70    });
71    let exit = run_daemon(
72        workspace,
73        configuration,
74        DaemonListen {
75            address,
76            tls,
77            token,
78            token_configured: configured_token.is_some(),
79        },
80        launch,
81    )
82    .await?;
83    launch.relaunch_now = matches!(exit, DaemonExit::Restart);
84    Ok(())
85}

87/// Where and how `cookie daemon` listens.
88pub(super) struct DaemonListen {
89    address: std::net::SocketAddr,
90    tls: Option<cookie_agent_server::TlsConfig>,
91    token: Zeroizing<String>,
92    /// The token comes from the configuration, not this start.
93    token_configured: bool,
94}

96/// Why the daemon stopped.
97pub(super) enum DaemonExit {
98    /// A stop signal.
99    Stopped,
100    /// A client's `daemon.restart`: relaunch.
101    Restart,
102}

104/// How long a daemon whose startup failed waits for the clients connected
105/// meanwhile to hear why before it exits.
106const FAILED_START_NOTICE: Duration = Duration::from_secs(5);
107
108pub(super) async fn run_daemon(
109    workspace: &Path,
110    configuration: cookie_agent_config::LoadedConfiguration,
111    options: DaemonListen,
112    launch: &Launch,
113) -> anyhow::Result<DaemonExit> {
114    let DaemonListen {
115        address,
116        tls,
117        token,
118        token_configured,
119    } = options;
120    let scheme = if tls.is_some() { "wss" } else { "ws" };
121    // A restarted daemon waits for the process it replaces to let go of the
122    // port.
123    let bind_wait = if launch.restarted {
124        daemon_restart::RESTART_BIND_WAIT
125    } else {
126        Duration::ZERO
127    };
128    let listener = cookie_agent_server::DaemonListener::new(
129        cookie_agent_engine::status::StatusBoard::new(cookie_agent_protocol::DaemonPhase::Starting),
130    );
131    let mut signals =
132        cookie_agent_protocol::StopSignals::new().context("listen for stop signals")?;
133    let running = daemon_restart::bind_retrying(bind_wait, || {
134        listener.serve_on(address, token.clone(), tls.clone())
135    })
136    .await
137    .map_err(|error| anyhow::Error::new(error).context("start WebSocket daemon"))?;
138    let url = format!("{scheme}://{}/ws", running.address());
139    // Launchers wait for this line: the daemon takes connections from here
140    // on, and tells them it is starting until it is ready.
141    println!("{}", ready_line(&url));
142    println!(
143        "cookie daemon listening on {url} (client protocol {}, process {})",
144        cookie_agent_protocol::CLIENT_PROTOCOL_VERSION,
145        std::process::id()
146    );
147    if let Some(line) = web_page_line(&url) {
148        println!("{line}");
149    }
150    let compose = async {
151        #[cfg(debug_assertions)]
152        hold_start_for_test(listener.board()).await;
153        super::compose::compose_for_daemon(workspace, configuration, &listener).await
154    };
155    let composed = tokio::select! {
156        composed = compose => composed,
157        () = signals.recv() => {
158            println!("cookie daemon stopped while starting");
159            listener.shutdown();
160            running.wait().await;
161            return Ok(DaemonExit::Stopped);
162        }
163    };
164    let mut runtime = match composed {
165        Ok(runtime) => runtime,
166        Err(error) => {
167            listener
168                .fail(&format!("{error:#}"), FAILED_START_NOTICE)
169                .await;
170            running.wait().await;
171            return Err(error);
172        }
173    };
174    if !token_configured {
175        runtime.engine.note_served_server_token(token.as_str());
176    }
177    drop(token);
178    // Only a daemon that knows how it was started can start itself again.
179    let restart = CancellationToken::new();
180    if launch.relaunch.is_some() {
181        let restart = restart.clone();
182        runtime
183            .engine
184            .set_restart_hook(Box::new(move || restart.cancel()));
185    }
186    listener.hand_over(std::sync::Arc::clone(&runtime.server));
187    println!("cookie daemon ready");
188    // Phones with push notifications hear the daemon is up again.
189    runtime.server.announce_started();
190    let exit = tokio::select! {
191        () = signals.recv() => DaemonExit::Stopped,
192        () = restart.cancelled() => DaemonExit::Restart,
193    };
194    exit_on_next_stop(signals);
195    match exit {
196        DaemonExit::Stopped => {
197            runtime.server.shutdown();
198            runtime.stop_catalog_refresh().await;
199            running.wait().await;
200            runtime.engine.shutdown().await;
201        }
202        DaemonExit::Restart => {
203            println!("cookie daemon restarting");
204            // Before anything stops: phones with push notifications hear it,
205            // but the one whose page asked.
206            runtime.server.announce_restarting().await;
207            // The engine stops first, while clients are still connected:
208            // they see the interrupted turns end, then the connection close
209            // saying the daemon is restarting.
210            runtime.stop_catalog_refresh().await;
211            runtime.engine.shutdown().await;
212            runtime.server.shutdown_for_restart();
213            // Returns once every client connection sent that close frame
214            // (bounded), so the relaunch does not cut it off.
215            running.wait().await;
216        }
217    }
218    Ok(exit)
219}

221/// Holds the start in debug builds, as a blocking task, so a test connects
222/// to a daemon that is still starting: for `COOKIE_AGENT_TEST_STARTUP_HOLD_MS`,
223/// and while the file `COOKIE_AGENT_TEST_STARTUP_HOLD_FILE` names exists,
224/// which a test creates before it restarts the daemon.
225#[cfg(debug_assertions)]
226async fn hold_start_for_test(board: &cookie_agent_engine::status::StatusBoard) {
227    let hold = std::env::var("COOKIE_AGENT_TEST_STARTUP_HOLD_MS")
228        .ok()
229        .and_then(|millis| millis.parse().ok())
230        .map(Duration::from_millis);
231    let file = std::env::var_os("COOKIE_AGENT_TEST_STARTUP_HOLD_FILE")
232        .map(PathBuf::from)
233        .filter(|file| file.exists());
234    if hold.is_none() && file.is_none() {
235        return;
236    }
237    let _task = board.start("Holding the start for a test", true);
238    if let Some(hold) = hold {
239        tokio::time::sleep(hold).await;
240    }
241    if let Some(file) = file {
242        while file.exists() {
243            tokio::time::sleep(Duration::from_millis(100)).await;
244        }
245    }
246}

248/// Exits at once on the next stop signal, for a graceful stop that hangs.
249fn exit_on_next_stop(mut signals: cookie_agent_protocol::StopSignals) {
250    tokio::spawn(async move {
251        signals.recv().await;
252        eprintln!("cookie: stopping at once");
253        std::process::exit(1);
254    });
255}
256
257/// Cancels `stop` on the first stop signal; see [`exit_on_next_stop`] for
258/// the second.
259pub(super) fn stop_on_signal(stop: CancellationToken) -> anyhow::Result<()> {
260    let mut signals =
261        cookie_agent_protocol::StopSignals::new().context("listen for stop signals")?;
262    tokio::spawn(async move {
263        signals.recv().await;
264        stop.cancel();
265        exit_on_next_stop(signals);
266    });
267    Ok(())
