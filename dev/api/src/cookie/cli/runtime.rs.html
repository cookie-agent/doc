13#[derive(Debug, PartialEq, Eq, Subcommand)]
14pub(super) enum RuntimeCommand {
15    /// Serve this machine's runtime: the `coding` capability (`read`,
16    /// `write`, `edit`, `bash`, `webfetch`) for every workspace sessions
17    /// attach.
18    #[command(group(clap::ArgGroup::new("transport").required(true).args(["stdio", "daemon"])))]
19    Serve {
20        /// Speak the runtime protocol over standard input and output, one
21        /// JSON-RPC message per line, as a template starts it.
22        #[arg(long)]
23        stdio: bool,
24        /// Dial in to a daemon and stay connected, reconnecting after drops:
25        /// the daemon at this client URL (`ws://` or `wss://` ending in
26        /// `/ws`), or, without one, `[server] url` in this machine's
27        /// `config.toml`, else the local daemon.
28        #[arg(long, value_name = "URL", num_args = 0..=1)]
29        daemon: Option<Option<String>>,
30        /// The runtime ID to dial in as; this machine's host name when
31        /// omitted.
32        #[arg(long, value_name = "RUNTIME_ID", conflicts_with = "stdio")]
33        id: Option<String>,
34        /// The daemon's bearer token, for `--daemon`; `[server] token` in
35        /// this machine's `config.toml` when omitted.
36        #[arg(
37            long,
38            env = "COOKIE_DAEMON_TOKEN",
39            hide_env_values = true,
40            allow_hyphen_values = true
41        )]
42        token: Option<String>,
43        /// Limit `coding` to workspaces under this directory; repeat for
44        /// several. A leading `~` is expanded. Any workspace when omitted.
45        #[arg(long = "root", value_name = "DIR")]
46        roots: Vec<PathBuf>,
47    },
48    /// Connect an ACP agent's MCP client to cookie's tools: copy standard
49    /// input to the runtime's bridge socket and the socket to standard
50    /// output. The runtime hands this command to the agents it hosts; it
51    /// is not meant to be run by hand.
52    McpBridge {
53        /// The bridge socket the runtime opened for the agent's session.
54        #[arg(long, value_name = "PATH")]
55        socket: PathBuf,
56    },
57}

59/// `cookie runtime serve --stdio`: serves one coding runtime for the
60/// workspace over this process's stdio until the daemon sends
61/// `runtime/shutdown` or closes standard input. Standard output carries only
62/// protocol messages; errors go to standard error.
63pub(super) async fn run_runtime(command: RuntimeCommand) -> anyhow::Result<()> {
64    let (stdio, daemon, id, token, roots) = match command {
65        RuntimeCommand::Serve {
66            stdio,
67            daemon,
68            id,
69            token,
70            roots,
71        } => (stdio, daemon, id, token, roots),
72        RuntimeCommand::McpBridge { socket } => {
73            return cookie_agent_coding_runtime::acp_bridge::run_bridge(&socket)
74                .await
75                .with_context(|| format!("MCP bridge at {}", socket.display()));
76        }
77    };
78    let roots = roots
79        .iter()
80        .map(|root| {
81            let expanded = cookie_agent_coding_runtime::server::expand_home(root);
82            let root = cookie_agent_protocol::paths::canonicalize(&expanded)
83                .with_context(|| format!("root {} is unavailable", expanded.display()))?;
84            anyhow::ensure!(root.is_dir(), "root {} is not a directory", root.display());
85            Ok(root)
86        })
87        .collect::<anyhow::Result<Vec<_>>>()?;
88    if !stdio && let Some(daemon) = daemon {
89        let DaemonTarget { url: daemon, token } = daemon_target(daemon, token)?;
90        let id = id.unwrap_or_else(|| host_runtime_id("", &host_name()));
91        dial_in_id(&id)?;
92        cookie_agent_protocol_transport::runtime_dial_in_url(&daemon, &id)?;
93        // A runtime that dials in hosts the MCP servers and ACP agents the
94        // daemon's `[runtimes.<id>]` gives it, never this machine's own.
95        warn_local_runtime_settings(&id);
96        let stop = CancellationToken::new();
97        stop_on_signal(stop.clone())?;
98        return dial_in(
99            &daemon,
100            &token,
101            &id,
102            cookie_agent_coding_runtime::server::ServeOptions {
103                roots,
104                stop,
105                ..Default::default()
106            },
107        )
108        .await;
109    }
110    let stop = CancellationToken::new();
111    stop_on_signal(stop.clone())?;
112    cookie_agent_coding_runtime::server::serve_stdio(
113        cookie_agent_coding_runtime::server::ServeOptions {
114            roots,
115            stop,
116            ..Default::default()
117        },
118    )
119    .await;
120    // The blocking standard-input reader may still wait for a line that
121    // never comes; exit without waiting for it.
122    std::process::exit(0)
123}

125/// A runtime ID for this machine: `<prefix>_<host>`, or the host name alone
126/// without a prefix, with the host name lowercased, every character outside
127/// `[a-z0-9_]` replaced by `_`, repeats and edges trimmed, at most 32 bytes.
128/// A bare host name that does not start with a letter gets `host_` in front;
129/// no host name at all gives the prefix, or `host`.
130pub(super) fn host_runtime_id(prefix: &str, host: &str) -> String {
131    let mut name = String::new();
132    for character in host.chars() {
133        let character = character.to_ascii_lowercase();
134        let character = if character.is_ascii_lowercase() || character.is_ascii_digit() {
135            character
136        } else {
137            '_'
138        };
139        if character == '_' && (name.is_empty() || name.ends_with('_')) {
140            continue;
141        }
142        name.push(character);
143    }
144    let name = name.trim_end_matches('_');
145    let mut id = match (prefix, name) {
146        ("", "") => "host".to_owned(),
147        (prefix, "") => prefix.to_owned(),
148        ("", name) if name.starts_with(|character: char| character.is_ascii_lowercase()) => {
149            name.to_owned()
150        }
151        ("", name) => format!("host_{name}"),
152        (prefix, name) => format!("{prefix}_{name}"),
153    };
154    id.truncate(32);
155    id.truncate(id.trim_end_matches('_').len());
156    id
157}

159/// Says once that this machine's `[runtimes.local]` MCP servers and ACP
160/// agents, which only its own daemon's built-in runtime hosts, are not the
161/// ones runtime `id` dials in with
162/// (`docs/site/specs/runtime-management.md` section 3.2).
163fn warn_local_runtime_settings(id: &str) {
164    let Ok(config) = cookie_agent_config::load() else {
165        return;
166    };
167    let local = config
168        .runtimes
169        .iter()
170        .find(|(runtime, _)| runtime.as_str() == cookie_agent_protocol::RuntimeId::LOCAL)
171        .map(|(_, template)| template);
172    if local.is_some_and(|local| !local.mcp_servers.is_empty() || !local.acp_agents.is_empty()) {
173        eprintln!(
174            "cookie runtime: this machine's [runtimes.local] MCP servers and ACP agents are its \
175             own daemon's; runtime `{id}` hosts the ones [runtimes.{id}] gives it in the \
176             configuration of the daemon it dials in to"
177        );
178    }
179}

181/// Checks `id` as a runtime ID a runtime may dial in under: well formed
182/// and not a built-in runtime's, which the daemon refuses.
183fn dial_in_id(id: &str) -> anyhow::Result<()> {
184    let runtime_id = cookie_agent_protocol::RuntimeId::new(id).map_err(|_| {
185        anyhow::anyhow!("invalid runtime ID `{id}`: it must match [a-z][a-z0-9_]{{0,31}}")
186    })?;
187    anyhow::ensure!(
188        !runtime_id.is_reserved(),
189        "runtime ID `{id}` is reserved for the daemon's built-in runtime; choose another with --id"
190    );
191    Ok(())
192}

194/// Dials in to the daemon at `daemon` as runtime `id` and serves it,
195/// connecting again after every drop or refusal for now with a growing
196/// wait, until stopped. A connection that has carried nothing, not even a
197/// pong to its keepalive pings, for `DIAL_IN_DEAD_AFTER` counts as a drop,
198/// so one that died without a close is noticed, and so does a connect that
199/// the daemon has not answered within `CONNECT_TIMEOUT` (a peer that takes
200/// the TCP connection and never upgrades it). A refusal for good, such as
201/// a configured or reserved ID or a wrong token, ends it with the daemon's
202/// reason.
203pub(super) async fn dial_in(
204    daemon: &str,
205    token: &str,
206    id: &str,
207    options: cookie_agent_coding_runtime::server::ServeOptions,
208) -> anyhow::Result<()> {
209    const LONGEST_WAIT: Duration = Duration::from_secs(30);
210    let stop = options.stop.clone();
211    let mut waits =
212        cookie_agent_identity::Backoff::new(Duration::from_secs(1), LONGEST_WAIT).delays();
213    let mut wait = waits.next_delay();
214    loop {
215        let connected = tokio::select! {
216            () = stop.cancelled() => return Ok(()),
217            connected = cookie_agent_protocol_transport::WebSocketTransport::connect_runtime(daemon, token, id) => connected,
218        };
219        match connected {
220            Ok(transport) => {
221                eprintln!("cookie runtime: connected to {daemon} as `{id}`");
222                let started = std::time::Instant::now();
223                let reason = cookie_agent_coding_runtime::server::serve(
224                    Box::new(transport),
225                    options.clone(),
226                )
227                .await;
228                if stop.is_cancelled() {
229                    return Ok(());
230                }
231                eprintln!("cookie runtime: disconnected from {daemon}: {reason}");
232                if started.elapsed() > LONGEST_WAIT {
233                    waits.reset();
234                    wait = waits.next_delay();
235                }
236            }
237            Err(error) if error.is_permanent() => {
238                anyhow::bail!("cannot dial in to {daemon} as `{id}`: {error}");
239            }
240            Err(error) => eprintln!(
241                "cookie runtime: cannot dial in to {daemon} as `{id}`: {error}; retrying in {} s",
242                wait.as_secs()
243            ),
244        }
245        tokio::select! {
246            () = tokio::time::sleep(wait) => {}
247            () = stop.cancelled() => return Ok(()),
248        }
249        wait = waits.next_delay();
250    }
