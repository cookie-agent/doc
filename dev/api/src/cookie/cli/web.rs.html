1//! `cookie web`: the daemon's web page with its token.
2
3use anyhow::Context as _;
4use zeroize::Zeroizing;
5
6use super::daemon_target;
7
8/// `cookie web`: prints the daemon's web page with its token, and opens it
9/// in the system browser with `open`.
10pub(super) fn run_web(
11    url: Option<String>,
12    token: Option<String>,
13    open: bool,
14) -> anyhow::Result<()> {
15    let target = daemon_target(url, token)?;
16    let link = web_link(&target.url, &target.token)?;
17    println!("{}", link.as_str());
18    if open {
19        open_in_browser(&link)?;
20    }
21    Ok(())
22}
23
24/// The daemon's web page with `token` in its fragment, for the daemon whose
25/// client endpoint is `url`: `ws`→`http`, `wss`→`https`, the endpoint's
26/// `ws` dropped. The fragment never reaches a server or its logs.
27pub(super) fn web_link(url: &str, token: &str) -> anyhow::Result<Zeroizing<String>> {
28    let page = cookie_agent_server::page_url(url)
29        .with_context(|| format!("{url} is not a daemon client URL ending in /ws"))?;
30    Ok(Zeroizing::new(format!("{page}#token={token}")))
31}
32
33/// Opens `link` in the system browser.
34pub(super) fn open_in_browser(link: &str) -> anyhow::Result<()> {
35    let mut command = if cfg!(target_os = "macos") {
36        std::process::Command::new("open")
37    } else if cfg!(windows) {
38        let mut command = std::process::Command::new("cmd");
39        command.args(["/C", "start", ""]);
40        command
41    } else {
42        std::process::Command::new("xdg-open")
43    };
44    let status = command
45        .arg(link)
46        .stdin(std::process::Stdio::null())
47        .stdout(std::process::Stdio::null())
48        .stderr(std::process::Stdio::null())
49        .status()
50        .context("open the system browser")?;
51    anyhow::ensure!(
52        status.success(),
53        "the system browser did not open ({status})"
54    );
55    Ok(())
56}
