1use std::{env, time::Duration};
2
3use clap::Parser;
4
5/// The environment variable a daemon token may come from, which clap reads
6/// into the `--token` arguments.
7const DAEMON_TOKEN_ENV: &str = "COOKIE_DAEMON_TOKEN";
8
9/// How this process was started, for `cookie daemon` to restart itself.
10struct Launch {
11    /// The command to start again; `None` when it could not be found, and
12    /// the daemon then refuses `daemon.restart`.
13    relaunch: Option<daemon_restart::Relaunch>,
14    /// This process was started by a restart, so the daemon waits for its
15    /// port to free.
16    restarted: bool,
17    /// The daemon stopped to restart: relaunch once the runtime is gone.
18    relaunch_now: bool,
19}
20
21/// The most async runtime worker threads: the daemon waits on models,
22/// tools, and clients far more than it computes, and every worker, with the
23/// malloc arena it ends up with, costs memory on a many-core machine while
24/// more of them only add scheduling overhead. `TOKIO_WORKER_THREADS`
25/// overrides it.
26const MAX_WORKER_THREADS: usize = 8;
27
28/// The most glibc malloc arenas, where glibc would make up to eight per
29/// core: each holds on to what its threads freed. `MALLOC_ARENA_MAX`
30/// overrides it.
31#[cfg(all(target_os = "linux", target_env = "gnu"))]
32const MALLOC_ARENA_MAX: libc::c_int = 2;
33
34/// Caps glibc's malloc arenas, unless `MALLOC_ARENA_MAX` already did. Runs
35/// before any thread exists, which an arena limit needs to apply to all of
36/// them.
37#[cfg(all(target_os = "linux", target_env = "gnu"))]
38fn limit_malloc_arenas() {
39    if env::var_os("MALLOC_ARENA_MAX").is_none() {
40        // SAFETY: `mallopt` only sets an allocator parameter; no other
41        // thread exists yet to allocate concurrently.
42        unsafe { libc::mallopt(libc::M_ARENA_MAX, MALLOC_ARENA_MAX) };
43    }
44}
45
46fn main() {
47    #[cfg(all(target_os = "linux", target_env = "gnu"))]
48    limit_malloc_arenas();
49    let cli = cli::Cli::parse();
50    // Before anything changes the environment: how to start this command
51    // again.
52    let mut launch = Launch {
53        relaunch: daemon_restart::Relaunch::capture(),
54        restarted: env::var_os(daemon_restart::RESTARTED_ENV).is_some(),
55        relaunch_now: false,
56    };
57    // Likewise the path ACP agents' MCP bridges start from, before an
58    // update can replace the binary under it.
59    cookie_agent_coding_runtime::server::pin_executable();
60    #[cfg(windows)]
61    cli::update::remove_replaced_binaries();
62    // SAFETY: as for the token below, no other thread exists yet.
63    unsafe { env::remove_var(daemon_restart::RESTARTED_ENV) };
64    // Clap has read the token into `cli`. Drop the variable so no process
65    // cookie starts (the `bash` tool, MCP servers, plugins, runtime
66    // templates, editors) inherits the daemon's credential.
67    //
68    // SAFETY: `remove_var` races only with other threads using the
69    // environment. This runs on the main thread before the Tokio runtime or
70    // anything else that starts a thread exists; argument parsing starts
71    // none.
72    unsafe { env::remove_var(DAEMON_TOKEN_ENV) };
73    let mut builder = tokio::runtime::Builder::new_multi_thread();
74    if env::var_os("TOKIO_WORKER_THREADS").is_none() {
75        let cores = std::thread::available_parallelism().map_or(1, std::num::NonZero::get);
76        builder.worker_threads(cores.min(MAX_WORKER_THREADS));
77    }
78    let runtime = match builder.enable_all().build() {
79        Ok(runtime) => runtime,
80        Err(error) => {
81            eprintln!("cookie: cannot start the async runtime: {error}");
82            std::process::exit(1);
83        }
84    };
85    let error_details = cli.error_details;
86    let result = runtime.block_on(cli::dispatch(cli, &mut launch));
87    if result.is_ok()
88        && launch.relaunch_now
89        && let Some(relaunch) = &launch.relaunch
90    {
91        // Every task and socket goes with the runtime before the new image
92        // starts.
93        runtime.shutdown_timeout(Duration::from_secs(5));
94        let error = daemon_restart::relaunch(relaunch);
95        eprintln!(
96            "cookie: {}",
97            cookie_agent_protocol::diagnostics::detail(&format!("{error:#}"))
98        );
99        std::process::exit(1);
100    }
101    if let Err(error) = result {
102        eprintln!(
103            "cookie: {}",
104            cookie_agent_protocol::diagnostics::detail(&format!("{error:#}"))
105        );
106        // A daemon error shows its message and hint above; its technical
107        // detail only on request.
108        if let Some(detail) = error
109            .chain()
110            .find_map(|cause| cause.downcast_ref::<cookie_agent_protocol_transport::ClientError>())
111            .and_then(cookie_agent_protocol_transport::ClientError::detail)
112        {
113            if error_details {
114                eprintln!(
115                    "details: {}",
116                    cookie_agent_protocol::diagnostics::detail(&detail)
117                );
118            } else {
119                eprintln!("(run again with --error-details for the technical detail)");
120            }
121        }
122        std::process::exit(1);
123    }
124}
