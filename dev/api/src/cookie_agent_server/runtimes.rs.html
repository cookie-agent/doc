1//! The `runtimes.list` and `runtimes.reconnect` views of the engine's
2//! runtime registry, and the `machines.list` view of its machines.
3
4use cookie_agent_engine::{Engine, RuntimeInstanceStatus, RuntimeInstanceSummary};
5use cookie_agent_protocol::{
6    AttachmentName, CapabilityName, RuntimeId, RuntimeInstanceInfo, RuntimeListing,
7    RuntimesListResult,

10/// What one session may learn of `listing`: the runtimes of `attachments`
11/// other than `global`, each saying what it is. What a runtime is doing and why
12/// it last failed describe the daemon beyond the session, and a connection's
13/// error can hold hosts and usernames, so they stay out.
14pub(crate) fn session_runtimes_list(
15    listing: RuntimesListResult,
16    attachments: &[cookie_agent_protocol::RuntimeAttachment],
17) -> RuntimesListResult {
18    RuntimesListResult {
19        runtimes: listing
20            .runtimes
21            .into_iter()
22            .filter(|runtime| {
23                runtime.runtime_id.as_str() != RuntimeId::GLOBAL
24                    && attachments
25                        .iter()
26                        .any(|attachment| attachment.runtime_id == runtime.runtime_id)
27            })
28            .map(|runtime| RuntimeListing {
29                capabilities: Vec::new(),
30                message: None,
31                active_runs: 0,
32                background_calls: 0,
33                ..runtime
34            })
35            .collect(),
36        instances: Vec::new(),
37    }
38}

40pub(crate) fn runtimes_list(engine: &Engine) -> RuntimesListResult {
41    let mut activity = engine.runtime_activity();
42    RuntimesListResult {
43        runtimes: engine
44            .runtime_templates()
45            .into_iter()
46            .filter_map(|template| {
47                // A runtime this process has not seen connected shows what
48                // the registry remembers of it.
49                let remembered = (!template.listing.connected)
50                    .then(|| engine.remembered_runtime(&template.runtime_id))
51                    .flatten();
52                let (platform, home) = match remembered {
53                    Some(remembered) => (
54                        template.listing.platform.or(remembered.platform),
55                        template.listing.home.or(remembered.home),
56                    ),
57                    None => (template.listing.platform, template.listing.home),
58                };
59                let busy = activity.remove(&template.runtime_id).unwrap_or_default();
60                Some(RuntimeListing {
61                    runtime_id: RuntimeId::new(template.runtime_id.as_str()).ok()?,
62                    lifetime: template.lifetime,
63                    connected: template.listing.connected,
64                    capabilities: template.listing.capabilities,
65                    platform,
66                    machine: engine.runtime_machine(&template.runtime_id),
67                    home,
68                    message: template.listing.message,
69                    active_runs: busy.active_runs,
70                    background_calls: busy.background_calls,
71                })
72            })
73            .collect(),
74        instances: engine
75            .runtime_instances()
76            .into_iter()
77            .filter_map(instance_info)
78            .collect(),
79    }
80}

82/// An instance as clients see it; `None` for an invalid runtime ID.
83pub(crate) fn instance_info(instance: RuntimeInstanceSummary) -> Option<RuntimeInstanceInfo> {
84    Some(RuntimeInstanceInfo {
85        runtime_id: RuntimeId::new(instance.runtime_id).ok()?,
86        capability: CapabilityName::new(instance.capability).ok()?,
87        attachment_name: instance
88            .attachment_name
89            .and_then(|name| AttachmentName::new(name).ok()),
90        kind: instance.kind,
91        workspace: instance.workspace,
92        platform: instance.platform,
93        status: match instance.status {
94            RuntimeInstanceStatus::Starting => {
95                cookie_agent_protocol::RuntimeInstanceStatus::Starting
96            }
97            RuntimeInstanceStatus::Ready => cookie_agent_protocol::RuntimeInstanceStatus::Ready,
98            RuntimeInstanceStatus::Reconnecting => {
99                cookie_agent_protocol::RuntimeInstanceStatus::Reconnecting
100            }
101            RuntimeInstanceStatus::Failed => cookie_agent_protocol::RuntimeInstanceStatus::Failed,
102            RuntimeInstanceStatus::Stopped => cookie_agent_protocol::RuntimeInstanceStatus::Stopped,
103        },
104        message: instance.message,
105    })
106}

108impl crate::Server {
109    /// Runtime `runtime_id` as `runtimes.list` reports it, for the answer of
110    /// `runtimes.connect` and `runtimes.disconnect`.
111    pub(crate) fn runtime_listing(&self, runtime_id: &RuntimeId) -> Option<RuntimeListing> {
112        runtimes_list(&self.engine)
113            .runtimes
114            .into_iter()
115            .find(|runtime| runtime.runtime_id == *runtime_id)
116    }
117
118    /// `machines.list`, and the answer of every `machines.*` change.
119    pub(crate) fn machines_list(&self) -> cookie_agent_protocol::MachinesListResult {
120        cookie_agent_protocol::MachinesListResult {
121            machines: self.engine.machines(),
122        }
123    }
124}

126#[cfg(test)]
127mod session_runtimes_tests {
128    use cookie_agent_protocol::{
129        AttachmentName, CapabilityName, MachineName, RuntimeAttachment, RuntimeCapabilityInfo,
130        RuntimeKind, RuntimeLifetime,
131    };
132
133    use super::*;
134
135    fn listed(id: &str) -> RuntimeListing {
136        RuntimeListing {
137            runtime_id: RuntimeId::new(id).unwrap(),
138            lifetime: RuntimeLifetime::Spawned,
139            connected: true,
140            capabilities: vec![RuntimeCapabilityInfo {
141                name: CapabilityName::coding(),
142                kind: RuntimeKind::new("coding").unwrap(),
143                takes_workspace: true,
144                description: String::new(),
145            }],
146            platform: Some("linux".into()),
147            machine: Some(MachineName::new(format!("{id}-box")).unwrap()),
148            home: Some("/home/someone".into()),
149            message: Some("ssh someone@host: refused".into()),
150            active_runs: 2,
151            background_calls: 1,
152        }
153    }
154
155    fn attached(id: &str) -> RuntimeAttachment {
156        RuntimeAttachment {
157            name: AttachmentName::new(format!("{id}_work")).unwrap(),
158            runtime_id: RuntimeId::new(id).unwrap(),
159            capability: CapabilityName::coding(),
160            kind: RuntimeKind::new("coding").unwrap(),
161            workspace: None,
162        }
163    }
164
165    #[test]
166    fn a_session_learns_its_own_runtimes_and_what_they_are_alone() {
167        let listing = RuntimesListResult {
168            runtimes: vec![listed(RuntimeId::GLOBAL), listed("mine"), listed("other")],
169            instances: Vec::new(),
170        };
171        let result =
172            session_runtimes_list(listing, &[attached(RuntimeId::GLOBAL), attached("mine")]);
173        assert_eq!(result.runtimes.len(), 1);
174        let mine = &result.runtimes[0];
175        assert_eq!(mine.runtime_id.as_str(), "mine");
176        assert_eq!(mine.machine.as_ref().unwrap().as_str(), "mine-box");
177        assert_eq!(mine.home.as_deref(), Some("/home/someone"));
178        assert_eq!(mine.platform.as_deref(), Some("linux"));
179        assert!(mine.connected);
180        assert!(mine.capabilities.is_empty());
181        assert_eq!(mine.message, None);
182        assert_eq!((mine.active_runs, mine.background_calls), (0, 0));
183        assert!(result.instances.is_empty());
184    }
