1//! The `runtimes.list` and `runtimes.reconnect` views of the engine's
2//! runtime registry, and the `machines.list` view of its machines.
3
4use cookie_agent_engine::{Engine, RuntimeInstanceStatus, RuntimeInstanceSummary};
5use cookie_agent_protocol::{
6    AttachmentName, CapabilityName, RuntimeId, RuntimeInstanceInfo, RuntimeListing,
7    RuntimesListResult,
8};
9
10pub(crate) fn runtimes_list(engine: &Engine) -> RuntimesListResult {
11    let mut activity = engine.runtime_activity();
12    RuntimesListResult {
13        runtimes: engine
14            .runtime_templates()
15            .into_iter()
16            .filter_map(|template| {
17                // A runtime this process has not seen connected shows what
18                // the registry remembers of it.
19                let remembered = (!template.listing.connected)
20                    .then(|| engine.remembered_runtime(&template.runtime_id))
21                    .flatten();
22                let (platform, home) = match remembered {
23                    Some(remembered) => (
24                        template.listing.platform.or(remembered.platform),
25                        template.listing.home.or(remembered.home),
26                    ),
27                    None => (template.listing.platform, template.listing.home),
28                };
29                let busy = activity.remove(&template.runtime_id).unwrap_or_default();
30                Some(RuntimeListing {
31                    runtime_id: RuntimeId::new(template.runtime_id.as_str()).ok()?,
32                    lifetime: template.lifetime,
33                    connected: template.listing.connected,
34                    capabilities: template.listing.capabilities,
35                    platform,
36                    machine: engine.runtime_machine(&template.runtime_id),
37                    home,
38                    message: template.listing.message,
39                    active_runs: busy.active_runs,
40                    background_calls: busy.background_calls,
41                })
42            })
43            .collect(),
44        instances: engine
45            .runtime_instances()
46            .into_iter()
47            .filter_map(instance_info)
48            .collect(),
49    }
50}

52/// An instance as clients see it; `None` for an invalid runtime ID.
53pub(crate) fn instance_info(instance: RuntimeInstanceSummary) -> Option<RuntimeInstanceInfo> {
54    Some(RuntimeInstanceInfo {
55        runtime_id: RuntimeId::new(instance.runtime_id).ok()?,
56        capability: CapabilityName::new(instance.capability).ok()?,
57        attachment_name: instance
58            .attachment_name
59            .and_then(|name| AttachmentName::new(name).ok()),
60        kind: instance.kind,
61        workspace: instance.workspace,
62        platform: instance.platform,
63        status: match instance.status {
64            RuntimeInstanceStatus::Starting => {
65                cookie_agent_protocol::RuntimeInstanceStatus::Starting
66            }
67            RuntimeInstanceStatus::Ready => cookie_agent_protocol::RuntimeInstanceStatus::Ready,
68            RuntimeInstanceStatus::Reconnecting => {
69                cookie_agent_protocol::RuntimeInstanceStatus::Reconnecting
70            }
71            RuntimeInstanceStatus::Failed => cookie_agent_protocol::RuntimeInstanceStatus::Failed,
72            RuntimeInstanceStatus::Stopped => cookie_agent_protocol::RuntimeInstanceStatus::Stopped,
73        },
74        message: instance.message,
75    })
76}

78impl crate::Server {
79    /// Runtime `runtime_id` as `runtimes.list` reports it, for the answer of
80    /// `runtimes.connect` and `runtimes.disconnect`.
81    pub(crate) fn runtime_listing(&self, runtime_id: &RuntimeId) -> Option<RuntimeListing> {
82        runtimes_list(&self.engine)
83            .runtimes
84            .into_iter()
85            .find(|runtime| runtime.runtime_id == *runtime_id)
86    }
87
88    /// `machines.list`, and the answer of every `machines.*` change.
89    pub(crate) fn machines_list(&self) -> cookie_agent_protocol::MachinesListResult {
90        cookie_agent_protocol::MachinesListResult {
91            machines: self.engine.machines(),
92        }
93    }
