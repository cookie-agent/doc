32/// How long needs-you signals of one tree gather into one push.
33const NEEDS_YOU_WINDOW: Duration = Duration::from_secs(1);
34/// An approval with less time left than this is not pushed.
35const MIN_APPROVAL_LEFT: Duration = Duration::from_secs(5);
36/// How long a push service keeps a message for an offline device.
37const DEFAULT_TTL_SECONDS: u64 = 86_400;
38/// Retries of a send the push service asked to slow down or failed.
39const RETRIES: u32 = 3;
40/// How long one send keeps retrying.
41const RETRY_BUDGET: Duration = Duration::from_secs(60);
42/// The wait before retry `n` (from 1) without a `Retry-After`: 2^n s.
43const RETRY_BACKOFF: cookie_agent_identity::Backoff =
44    cookie_agent_identity::Backoff::new(Duration::from_secs(1), RETRY_BUDGET);
45/// The VAPID token's lifetime; it is reused until an hour before it ends.
46const TOKEN_LIFETIME_SECONDS: i64 = 12 * 3600;
47const TOKEN_REFRESH_MARGIN_SECONDS: i64 = 3600;

49/// Why a notifications request failed.
50#[derive(Debug, thiserror::Error)]
51pub(crate) enum PushError {
52    #[error("there is no push subscription {0}")]
53    NotFound(String),
54    #[error("{field}: {reason}")]
55    Invalid { field: &'static str, reason: String },
56    #[error("{0}")]
57    Unavailable(String),
58}

60/// What one send came to.
61#[derive(Debug, PartialEq)]
62enum Delivery {
63    Sent,
64    Gone,
65    Failed(String),
66}

83/// The daemon's push notifications.
84pub(crate) struct PushService {
85    engine: Engine,
86    store: PushStore,
87    subscriptions: Mutex<Subscriptions>,
88    /// Serializes the changes to the subscriptions and their writes, so the
89    /// file and [`Self::subscriptions`] change in the same order while
90    /// readers of the list never wait for a write.
91    subscription_writes: Mutex<()>,
92    key: Mutex<Option<Arc<VapidKey>>>,
93    tokens: Mutex<HashMap<(String, String), (String, i64)>>,
94    /// The session each connection shows the user, by connection id.
95    presence: Mutex<HashMap<u64, SessionId>>,
96    /// Bumped whenever the subscriptions change.
97    changes: watch::Sender<u64>,
98    /// Needs-you signals gathering per root, sent when their window ends.
99    gathering: Mutex<HashMap<SessionId, Vec<NeedsYouSignal>>>,
100    /// The subscription each connection registered, by connection id.
101    registered: Mutex<HashMap<u64, String>>,
102    /// The subscription of the device that asked for the restart under
103    /// way: it is not told the daemon is restarting.
104    restart_requester: Mutex<Option<String>>,
105    http: reqwest::Client,
106    /// Tests send to a local `http://` push service.
107    allow_http: bool,
108}

110fn lock<T>(mutex: &Mutex<T>) -> MutexGuard<'_, T> {
111    mutex
112        .lock()
113        .unwrap_or_else(std::sync::PoisonError::into_inner)
114}

116/// Runs `work`, which reads or writes files, off the async worker when on
117/// a multi-threaded runtime.
118fn off_worker<T>(work: impl FnOnce() -> T) -> T {
119    if tokio::runtime::Handle::try_current()
120        .is_ok_and(|runtime| runtime.runtime_flavor() == tokio::runtime::RuntimeFlavor::MultiThread)
121    {
122        tokio::task::block_in_place(work)
123    } else {
124        work()
125    }
126}

165    /// Follows the engine's push signals until `shutdown`.
166    pub(crate) fn start(self: &Arc<Self>, shutdown: CancellationToken) {
167        let mut signals = self.engine.subscribe_push_signals();
168        let service = Arc::clone(self);
169        tokio::spawn(async move {
170            loop {
171                let signal = tokio::select! {
172                    () = shutdown.cancelled() => return,
173                    signal = signals.recv() => signal,
174                };
175                match signal {
176                    Ok(signal) => service.signal(signal),
177                    Err(broadcast::error::RecvError::Lagged(missed)) => {
178                        eprintln!("push notifications: {missed} signals dropped");
179                    }
180                    Err(broadcast::error::RecvError::Closed) => return,
181                }
182            }
183        });
184    }

186    fn signal(self: &Arc<Self>, signal: PushSignal) {
187        // A signal nobody present on the tree saw marks its root unread,
188        // whatever `[notifications]` says of pushing it (unread sessions
189        // spec section 2).
190        let root = match &signal {
191            PushSignal::Settled(settled) => settled.root,
192            PushSignal::NeedsYou(needs) => needs.root,
193        };
194        if !self.watched(root) {
195            self.engine.mark_unread(root);
196        }
197        let config = self.config();
198        if !config.enabled {
199            return;
200        }
201        match signal {
202            PushSignal::Settled(settled) => {
203                if settled.cron && !config.cron_sessions {
204                    return;
205                }
206                if self.watched(settled.root) {
207                    return;
208                }
209                let message = Message {
210                    urgency: if settled.failed() { "high" } else { "normal" },
211                    topic: Some(payload::topic(settled.root)),
212                    payload: Payload::settled(&settled, config.preview).into_bytes(),
213                    ttl: DEFAULT_TTL_SECONDS,
214                    deadline: None,
215                };
216                self.send_all(message);
217            }
218            PushSignal::NeedsYou(needs) => {
219                if !config.approvals {
220                    return;
221                }
222                let root = needs.root;
223                let first = {
224                    let mut gathering = lock(&self.gathering);
225                    let items = gathering.entry(root).or_default();
226                    items.push(needs);
227                    items.len() == 1
228                };
229                if first {
230                    let service = Arc::clone(self);
231                    tokio::spawn(async move {
232                        tokio::time::sleep(NEEDS_YOU_WINDOW).await;
233                        service.flush_needs_you(root);
234                    });
235                }
236            }
237        }
238    }
239
240    /// Sends what gathered for `root`: the items with time left, as one
241    /// push.
242    fn flush_needs_you(self: &Arc<Self>, root: SessionId) {
243        let items = lock(&self.gathering).remove(&root).unwrap_or_default();
244        let config = self.config();
245        if !config.enabled || !config.approvals || self.watched(root) {
246            return;
247        }
248        let now = Timestamp::now();
249        let items = items
250            .into_iter()
251            .filter(|item| {
252                item.expires_at().is_none_or(|expires| {
253                    expires.duration_since(now).unsigned_abs() >= MIN_APPROVAL_LEFT && expires > now
254                })
255            })
256            .collect::<Vec<_>>();
257        let Some(first) = items.first() else {
258            return;
259        };
260        let deadline = items.iter().filter_map(NeedsYouSignal::expires_at).min();
261        let ttl = deadline.map_or(DEFAULT_TTL_SECONDS, |deadline| {
262            u64::try_from(deadline.duration_since(now).as_secs())
263                .unwrap_or(1)
264                .max(1)
265        });
266        let message = Message {
267            payload: Payload::needs_you(first, items.len(), config.preview).into_bytes(),
268            ttl,
269            urgency: "high",
270            topic: Some(payload::topic(root)),
271            deadline,
272        };
273        self.send_all(message);
274    }

276    /// Whether a connection shows the user a session of `root`'s tree.
277    fn watched(&self, root: SessionId) -> bool {
278        let shown = lock(&self.presence).values().copied().collect::<Vec<_>>();
279        shown
280            .into_iter()
281            .any(|session| self.engine.root_session_of(session) == root)
282    }

284    /// Records what connection `connection` shows the user; `None` when the
285    /// user is not there.
286    pub(crate) fn set_presence(&self, connection: u64, session: Option<SessionId>) -> bool {
287        let mut presence = lock(&self.presence);
288        let first = !presence.contains_key(&connection);
289        match session {
290            Some(session) => {
291                presence.insert(connection, session);
292            }
293            None => {
294                presence.remove(&connection);
295            }
296        }
297        drop(presence);
298        // The user is there: the tree is read.
299        if let Some(session) = session {
300            self.engine.mark_tree_read(session);
301        }
302        first && session.is_some()
303    }

327    /// The daemon goes down to restart: every device but the one that
328    /// asked is told, and the daemon that comes back finds the mark
329    /// ([`Self::announce_started`]). Waits at most `wait` for the sends.
330    pub(crate) async fn announce_restarting(&self, wait: Duration) {
331        if let Err(error) = self.store.mark_restarting() {
332            eprintln!("push notifications: could not mark the restart: {error}");
333        }
334        let requester = lock(&self.restart_requester).take();
335        if !self.config().enabled {
336            return;
337        }
338        let message = daemon_message(DaemonEvent::Restarting, 600);
339        self.send_and_wait(&message, requester.as_deref(), wait)
340            .await;
341    }

343    /// The persistent daemon is listening: every device is told it
344    /// restarted, or started when the daemon before it did not go down to
345    /// restart (a service manager's restart, a boot). Waits at most `wait`.
346    pub(crate) async fn announce_started(&self, wait: Duration) {
347        let restarted = self.store.take_restarting();
348        if !self.config().enabled {
349            return;
350        }
351        let event = if restarted {
352            DaemonEvent::Restarted
353        } else {
354            DaemonEvent::Started
355        };
356        let message = daemon_message(event, 3600);
357        self.send_and_wait(&message, None, wait).await;
358    }

360    /// Sends `message` to every subscription but `except`, together, and
361    /// waits at most `wait` for them.
362    async fn send_and_wait(&self, message: &Message, except: Option<&str>, wait: Duration) {
363        let Ok(subscriptions) = self.read_subscriptions(|list| Ok(list.to_vec())) else {
364            return;
365        };
366        let sends = subscriptions
367            .into_iter()
368            .filter(|subscription| Some(subscription.id.as_str()) != except)
369            .map(|subscription| async move {
370                let delivery = self.deliver(&subscription, message).await;
371                self.record(&subscription.id, &delivery);
372            });
373        let _ = tokio::time::timeout(wait, futures_util::future::join_all(sends)).await;
374    }

385    /// The daemon's VAPID key, generated and stored on first use. A new key
386    /// drops every subscription: they were bound to the one before.
387    fn key(&self) -> Result<Arc<VapidKey>, PushError> {
388        if let Some(key) = lock(&self.key).as_ref() {
389            return Ok(Arc::clone(key));
390        }
391        // First use reads, and may write, files: off the worker, so the
392        // threads that wait for the key do not hold one up either.
393        off_worker(|| self.load_key())
394    }
395
396    fn load_key(&self) -> Result<Arc<VapidKey>, PushError> {
397        let mut slot = lock(&self.key);
398        if let Some(key) = slot.as_ref() {
399            return Ok(Arc::clone(key));
400        }
401        let unavailable = |error: std::io::Error| PushError::Unavailable(error.to_string());
402        let key = match self.store.read_key().map_err(unavailable)? {
403            Some(pkcs8) => VapidKey::from_pkcs8(&pkcs8)
404                .map_err(|error| PushError::Unavailable(error.to_string()))?,
405            None => {
406                let (key, pkcs8) = VapidKey::generate()
407                    .map_err(|error| PushError::Unavailable(error.to_string()))?;
408                self.store.write_key(&pkcs8).map_err(unavailable)?;
409                self.change_subscriptions(|list| {
410                    let dropped = !list.is_empty();
411                    list.clear();
412                    Ok(((), dropped))
413                })?;
414                key
415            }
416        };
417        let key = Arc::new(key);
418        *slot = Some(Arc::clone(&key));
419        Ok(key)
420    }

422    /// What `read` makes of the subscriptions as they are now.
423    fn read_subscriptions<T>(
424        &self,
425        read: impl FnOnce(&[StoredSubscription]) -> Result<T, PushError>,
426    ) -> Result<T, PushError> {
427        let subscriptions = lock(&self.subscriptions);
428        let list = subscriptions
429            .list
430            .as_ref()
431            .map_err(|error| PushError::Unavailable(error.clone()))?;
432        read(list)
433    }

435    /// Applies `change` to the subscriptions, and writes them when it says
436    /// to. The write happens off the worker and without the subscriptions'
437    /// lock: readers keep the list as it was until the file holds the new
438    /// one.
439    fn change_subscriptions<T>(
440        &self,
441        change: impl FnOnce(&mut Vec<StoredSubscription>) -> Result<(T, bool), PushError>,
442    ) -> Result<T, PushError> {
443        let _writes = lock(&self.subscription_writes);
444        let mut next = lock(&self.subscriptions)
445            .list
446            .as_ref()
447            .map_err(|error| PushError::Unavailable(error.clone()))?
448            .clone();
449        let (value, write) = change(&mut next)?;
450        if write {
451            off_worker(|| self.store.write_subscriptions(&next))
452                .map_err(|error| PushError::Unavailable(error.to_string()))?;
453            lock(&self.subscriptions).list = Ok(next);
454            self.changed();
455        }
456        Ok(value)
457    }

470    pub(crate) fn subscribe(
471        &self,
472        params: cookie_agent_protocol::NotificationsSubscribeParams,
473    ) -> Result<PushSubscription, PushError> {
474        self.key()?;
475        let endpoint = params.endpoint.trim().to_owned();
476        let url = reqwest::Url::parse(&endpoint).map_err(|error| PushError::Invalid {
477            field: "endpoint",
478            reason: error.to_string(),
479        })?;
480        if url.scheme() != "https" && !(self.allow_http && url.scheme() == "http") {
481            return Err(PushError::Invalid {
482                field: "endpoint",
483                reason: "must be an https:// URL".into(),
484            });
485        }
486        if endpoint.len() > PUSH_ENDPOINT_MAX_BYTES || url.host_str().is_none() {
487            return Err(PushError::Invalid {
488                field: "endpoint",
489                reason: format!(
490                    "must be a URL with a host, at most {PUSH_ENDPOINT_MAX_BYTES} bytes"
491                ),
492            });
493        }
494        let p256dh = decode_key(&params.p256dh, "p256dh")
495            .ok()
496            .filter(|key| key.len() == 65 && key[0] == 4)
497            .ok_or_else(|| PushError::Invalid {
498                field: "p256dh",
499                reason: "must be an uncompressed P-256 point in base64url".into(),
500            })?;
501        let auth = decode_key(&params.auth, "auth")
502            .ok()
503            .filter(|secret| secret.len() == 16)
504            .ok_or_else(|| PushError::Invalid {
505                field: "auth",
506                reason: "must be a 16-byte secret in base64url".into(),
507            })?;
508        let label = params.label.trim().to_owned();
509        let characters = label.chars().count();
510        if characters == 0 || characters > PUSH_LABEL_MAX_CHARS {
511            return Err(PushError::Invalid {
512                field: "label",
513                reason: format!("must be 1 to {PUSH_LABEL_MAX_CHARS} characters"),
514            });
515        }
516        use base64::Engine as _;
517        let encode = |bytes: &[u8]| base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(bytes);
518        self.change_subscriptions(|list| {
519            if let Some(existing) = list.iter_mut().find(|stored| stored.endpoint == endpoint) {
520                existing.p256dh = encode(&p256dh);
521                existing.auth = encode(&auth);
522                existing.label = label;
523                return Ok((visible(existing), true));
524            }
525            let stored = StoredSubscription {
526                id: new_id(list),
527                endpoint,
528                p256dh: encode(&p256dh),
529                auth: encode(&auth),
530                label,
531                created_at: Timestamp::now(),
532                last_sent_at: None,
533                last_error: None,
534            };
535            let shown = visible(&stored);
536            list.push(stored);
537            Ok((shown, true))
538        })
539    }

541    pub(crate) fn unsubscribe(&self, id: &str) -> Result<(), PushError> {
542        self.change_subscriptions(|list| {
543            let before = list.len();
544            list.retain(|stored| stored.id != id);
545            if list.len() == before {
546                return Err(PushError::NotFound(id.to_owned()));
547            }
548            Ok(((), true))
549        })
550    }

552    pub(crate) async fn test(&self, id: &str) -> Result<NotificationsTestResult, PushError> {
553        let subscription = self.read_subscriptions(|list| {
554            list.iter()
555                .find(|stored| stored.id == id)
556                .cloned()
557                .ok_or_else(|| PushError::NotFound(id.to_owned()))
558        })?;
559        let message = Message {
560            payload: Payload::test().into_bytes(),
561            ttl: 60,
562            urgency: "normal",
563            topic: None,
564            deadline: None,
565        };
566        let delivery = self.deliver(&subscription, &message).await;
567        self.record(&subscription.id, &delivery);
568        Ok(match delivery {
569            Delivery::Sent => NotificationsTestResult {
570                status: NotificationsTestStatus::Sent,
571                detail: None,
572            },
573            Delivery::Gone => NotificationsTestResult {
574                status: NotificationsTestStatus::Gone,
575                detail: None,

577            Delivery::Failed(detail) => NotificationsTestResult {
578                status: NotificationsTestStatus::Failed,
579                detail: Some(detail),

584    /// Sends `message` to every subscription, each on its own task.
585    fn send_all(self: &Arc<Self>, message: Message) {
586        let Ok(subscriptions) = self.read_subscriptions(|list| Ok(list.to_vec())) else {
587            return;
588        };
589        let message = Arc::new(message);
590        for subscription in subscriptions {
591            let service = Arc::clone(self);
592            let message = Arc::clone(&message);
593            tokio::spawn(async move {
594                let delivery = service.deliver(&subscription, &message).await;
595                service.record(&subscription.id, &delivery);
596            });
597        }
598    }

600    /// Keeps what a send came to: when it was sent, why it failed, or, for
601    /// a subscription that is gone, nothing.
602    fn record(&self, id: &str, delivery: &Delivery) {
603        let result = self.change_subscriptions(|list| {
604            let Some(index) = list.iter().position(|stored| stored.id == id) else {
605                return Ok(((), false));
606            };
607            match delivery {
608                Delivery::Sent => {
609                    list[index].last_sent_at = Some(Timestamp::now());
610                    list[index].last_error = None;
611                }
612                Delivery::Gone => {
613                    list.remove(index);
614                }
615                Delivery::Failed(error) => list[index].last_error = Some(error.clone()),
616            }
617            Ok(((), true))
618        });
619        if let Err(error) = result {
620            eprintln!("push notifications: could not record a send: {error}");
621        }
622    }

624    /// The `Authorization` value for the push service at `audience`.
625    fn authorization(&self, audience: &str) -> Result<String, String> {
626        let key = self.key().map_err(|error| error.to_string())?;
627        let subject = self
628            .config()
629            .contact
630            .unwrap_or_else(|| env!("CARGO_PKG_REPOSITORY").to_owned());
631        let now = Timestamp::now().as_second();
632        let cache_key = (audience.to_owned(), subject.clone());
633        let mut tokens = lock(&self.tokens);
634        let token = match tokens.get(&cache_key) {
635            Some((token, expires)) if *expires - TOKEN_REFRESH_MARGIN_SECONDS > now => {
636                token.clone()
637            }
638            _ => {
639                let expires = now + TOKEN_LIFETIME_SECONDS;
640                let token = key
641                    .token(audience, expires, &subject)
642                    .map_err(|error| error.to_string())?;
643                tokens.insert(cache_key, (token.clone(), expires));
644                token
645            }
646        };
647        Ok(format!("vapid t={token}, k={}", key.public_key()))
648    }
649
650    /// Sends `message` to one subscription, retrying what the push service
651    /// asked to retry.
652    async fn deliver(&self, subscription: &StoredSubscription, message: &Message) -> Delivery {
653        let prepared = (|| {
654            let url =
655                reqwest::Url::parse(&subscription.endpoint).map_err(|error| error.to_string())?;
656            let audience = url.origin().ascii_serialization();
657            let authorization = self.authorization(&audience)?;
658            let p256dh =
659                decode_key(&subscription.p256dh, "p256dh").map_err(|error| error.to_string())?;
660            let auth = decode_key(&subscription.auth, "auth").map_err(|error| error.to_string())?;
661            let body = crypto::encrypt(&p256dh, &auth, &message.payload)
662                .map_err(|error| error.to_string())?;
663            Ok::<_, String>((url, authorization, body))
664        })();
665        let (url, authorization, body) = match prepared {
666            Ok(prepared) => prepared,
667            Err(error) => return Delivery::Failed(error),
668        };
669        let started = tokio::time::Instant::now();
670        let mut attempt = 0;
671        loop {
672            let mut request = self
673                .http
674                .post(url.clone())
675                .header("Authorization", &authorization)
676                .header("Content-Encoding", "aes128gcm")
677                .header("Content-Type", "application/octet-stream")
678                .header("TTL", message.ttl.to_string())
679                .header("Urgency", message.urgency)
680                .body(body.clone());
681            if let Some(topic) = &message.topic {
682                request = request.header("Topic", topic);
683            }
684            let (retry_after, failure) = match request.send().await {
685                Ok(response) => {
686                    let status = response.status();
687                    if status.is_success() {
688                        return Delivery::Sent;
689                    }
690                    if status.as_u16() == 404 || status.as_u16() == 410 {
691                        return Delivery::Gone;
692                    }
693                    let retry_after = response
694                        .headers()
695                        .get("Retry-After")
696                        .and_then(|value| value.to_str().ok())
697                        .and_then(|value| value.trim().parse::<u64>().ok())
698                        .map(Duration::from_secs);
699                    let retryable = status.as_u16() == 429 || status.is_server_error();
700                    let text = error_text(response).await;
701                    let detail =
702                        format!("{status}: {}", text.chars().take(200).collect::<String>());
703                    if !retryable {
704                        return Delivery::Failed(detail);
705                    }
706                    (retry_after, detail)
707                }
708                Err(error) => {
709                    // A network error is retried like a push service that
710                    // is down.
711                    (None, error.to_string())
712                }
713            };
714            attempt += 1;
715            let wait = retry_after.unwrap_or_else(|| RETRY_BACKOFF.delay(attempt));
716            let past_deadline = message.deadline.is_some_and(|deadline| {
717                Timestamp::now()
718                    .checked_add(wait)
719                    .is_ok_and(|then| then >= deadline)
720            });
721            if attempt > RETRIES || started.elapsed() + wait > RETRY_BUDGET || past_deadline {
722                return Delivery::Failed(failure);
723            }
724            tokio::time::sleep(wait).await;
725        }
726    }
727}

755/// Eight lowercase letters and digits no subscription in `list` has.
756fn new_id(list: &[StoredSubscription]) -> String {
757    const ALPHABET: &[u8] = b"abcdefghijklmnopqrstuvwxyz0123456789";
758    loop {
759        let mut random = [0u8; 8];
760        let _ = getrandom::getrandom(&mut random);
761        let id = random
762            .iter()
763            .map(|byte| ALPHABET[usize::from(*byte) % ALPHABET.len()] as char)
764            .collect::<String>();
765        if !list.iter().any(|stored| stored.id == id) {
766            return id;
767        }
768    }
769}

771/// The most of a push service's error body the daemon reads.
772const ERROR_BODY_LIMIT: usize = 4 * 1024;
773
774/// The start of an error answer's body, for its detail: at most
775/// [`ERROR_BODY_LIMIT`] bytes, so a push service that answers with a huge or
776/// endless body cannot hold the delivery or its memory.
777async fn error_text(mut response: reqwest::Response) -> String {
778    let mut body = Vec::new();
779    while body.len() < ERROR_BODY_LIMIT {
780        match response.chunk().await {
781            Ok(Some(chunk)) => body.extend_from_slice(&chunk),
782            Ok(None) | Err(_) => break,
783        }
784    }
785    body.truncate(ERROR_BODY_LIMIT);
786    String::from_utf8_lossy(&body).into_owned()
787}
