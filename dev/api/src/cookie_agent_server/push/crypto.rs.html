1//! Web Push message encryption (RFC 8291, `aes128gcm` of RFC 8188) and the
2//! daemon's VAPID identity (RFC 8292), on `aws-lc-rs`, the crypto library
3//! rustls already brings (push notifications spec section 6).
4
5use aws_lc_rs::{
6    aead::{AES_128_GCM, Aad, LessSafeKey, Nonce, UnboundKey},
7    agreement::{self, ECDH_P256, PrivateKey, UnparsedPublicKey},
8    hkdf::{HKDF_SHA256, KeyType, Salt},
9    rand::{SecureRandom, SystemRandom},
10    signature::{ECDSA_P256_SHA256_FIXED_SIGNING, EcdsaKeyPair, KeyPair},
11};
12use base64::{Engine as _, engine::general_purpose::URL_SAFE_NO_PAD};

14/// The record size the header announces. One record carries the whole
15/// message, which push services cap at 4096 octets.
16const RECORD_SIZE: u32 = 4096;
17
18/// The plaintext a message can carry: a record of 4096 octets less the
19/// 86-octet header, the 16-octet tag, and the padding delimiter.
20pub(crate) const MAX_PLAINTEXT: usize = RECORD_SIZE as usize - 86 - 16 - 1;
21
22/// What went wrong encrypting or signing.
23#[derive(Debug, thiserror::Error)]
24pub(crate) enum PushCryptoError {
25    #[error("the subscription's {0} key is not valid")]
26    InvalidKey(&'static str),
27    #[error("the message is longer than one push record")]
28    TooLong,
29    #[error("the VAPID key is not a P-256 PKCS#8 key")]
30    InvalidVapidKey,
31    #[error("a cryptographic operation failed")]
32    Failed,
33}
34
35/// The length an HKDF expansion produces.
36struct Length(usize);
37
38impl KeyType for Length {
39    fn len(&self) -> usize {
40        self.0
41    }
42}
43
44/// `HKDF-Expand(prk, info, length)` of `HKDF-Extract(salt, ikm)`.
45fn hkdf(
46    salt: &[u8],
47    ikm: &[u8],
48    info: &[&[u8]],
49    length: usize,
50) -> Result<Vec<u8>, PushCryptoError> {
51    let prk = Salt::new(HKDF_SHA256, salt).extract(ikm);
52    let okm = prk
53        .expand(info, Length(length))
54        .map_err(|_| PushCryptoError::Failed)?;
55    let mut out = vec![0; length];
56    okm.fill(&mut out).map_err(|_| PushCryptoError::Failed)?;
57    Ok(out)
58}

60/// Decodes a base64url key a browser handed over, padded or not.
61pub(crate) fn decode_key(value: &str, which: &'static str) -> Result<Vec<u8>, PushCryptoError> {
62    URL_SAFE_NO_PAD
63        .decode(value.trim_end_matches('='))
64        .map_err(|_| PushCryptoError::InvalidKey(which))
65}

67/// Encrypts `plaintext` for the subscription with `ua_public` (its
68/// `p256dh`, an uncompressed P-256 point) and `auth_secret` (its `auth`):
69/// one `aes128gcm` record, header included, ready to POST.
70pub(crate) fn encrypt(
71    ua_public: &[u8],
72    auth_secret: &[u8],
73    plaintext: &[u8],
74) -> Result<Vec<u8>, PushCryptoError> {
75    let private = PrivateKey::generate(&ECDH_P256).map_err(|_| PushCryptoError::Failed)?;
76    let mut salt = [0; 16];
77    SystemRandom::new()
78        .fill(&mut salt)
79        .map_err(|_| PushCryptoError::Failed)?;
80    encrypt_with(&private, &salt, ua_public, auth_secret, plaintext)
81}
82
83/// [`encrypt`] with a given application server key and salt.
84fn encrypt_with(
85    as_private: &PrivateKey,
86    salt: &[u8; 16],
87    ua_public: &[u8],
88    auth_secret: &[u8],
89    plaintext: &[u8],
90) -> Result<Vec<u8>, PushCryptoError> {
91    if ua_public.len() != 65 || ua_public[0] != 4 {
92        return Err(PushCryptoError::InvalidKey("p256dh"));
93    }
94    if auth_secret.len() != 16 {
95        return Err(PushCryptoError::InvalidKey("auth"));
96    }
97    if plaintext.len() > MAX_PLAINTEXT {
98        return Err(PushCryptoError::TooLong);
99    }
100    let as_public = as_private
101        .compute_public_key()
102        .map_err(|_| PushCryptoError::Failed)?;
103    let as_public = as_public.as_ref();
104    let ecdh_secret = agreement::agree(
105        as_private,
106        UnparsedPublicKey::new(&ECDH_P256, ua_public),
107        PushCryptoError::InvalidKey("p256dh"),
108        |secret| Ok(secret.to_vec()),
109    )?;
110    let ikm = hkdf(
111        auth_secret,
112        &ecdh_secret,
113        &[b"WebPush: info\0", ua_public, as_public],
114        32,
115    )?;
116    let cek = hkdf(salt, &ikm, &[b"Content-Encoding: aes128gcm\0"], 16)?;
117    let nonce = hkdf(salt, &ikm, &[b"Content-Encoding: nonce\0"], 12)?;
118    let key =
119        LessSafeKey::new(UnboundKey::new(&AES_128_GCM, &cek).map_err(|_| PushCryptoError::Failed)?);
120    let nonce = Nonce::try_assume_unique_for_key(&nonce).map_err(|_| PushCryptoError::Failed)?;
121    // The last (and only) record ends with the delimiter 0x02, no padding.
122    let mut record = Vec::with_capacity(plaintext.len() + 17);
123    record.extend_from_slice(plaintext);
124    record.push(2);
125    key.seal_in_place_append_tag(nonce, Aad::empty(), &mut record)
126        .map_err(|_| PushCryptoError::Failed)?;
127    let mut body = Vec::with_capacity(86 + record.len());
128    body.extend_from_slice(salt);
129    body.extend_from_slice(&RECORD_SIZE.to_be_bytes());
130    body.push(u8::try_from(as_public.len()).map_err(|_| PushCryptoError::Failed)?);
131    body.extend_from_slice(as_public);
132    body.extend_from_slice(&record);
133    Ok(body)
134}

136/// The daemon's VAPID key pair: a P-256 signing key kept as PKCS#8.
137pub(crate) struct VapidKey {
138    pair: EcdsaKeyPair,
139}
140
141impl VapidKey {
142    /// A new key and its PKCS#8 document, to store.
143    pub(crate) fn generate() -> Result<(Self, Vec<u8>), PushCryptoError> {
144        let document =
145            EcdsaKeyPair::generate_pkcs8(&ECDSA_P256_SHA256_FIXED_SIGNING, &SystemRandom::new())
146                .map_err(|_| PushCryptoError::Failed)?;
147        let pkcs8 = document.as_ref().to_vec();
148        Ok((Self::from_pkcs8(&pkcs8)?, pkcs8))
149    }
150
151    pub(crate) fn from_pkcs8(pkcs8: &[u8]) -> Result<Self, PushCryptoError> {
152        EcdsaKeyPair::from_pkcs8(&ECDSA_P256_SHA256_FIXED_SIGNING, pkcs8)
153            .map(|pair| Self { pair })
154            .map_err(|_| PushCryptoError::InvalidVapidKey)
155    }
156
157    /// The public key, an uncompressed point in base64url: what browsers
158    /// subscribe with (`applicationServerKey`) and the `k` of the header.
159    pub(crate) fn public_key(&self) -> String {
160        URL_SAFE_NO_PAD.encode(self.pair.public_key().as_ref())
161    }
162
163    /// A VAPID JWT (ES256) for the push service at `audience` (an origin),
164    /// expiring at `expires` (Unix seconds), naming `subject`.
165    pub(crate) fn token(
166        &self,
167        audience: &str,
168        expires: i64,
169        subject: &str,
170    ) -> Result<String, PushCryptoError> {
171        let header = URL_SAFE_NO_PAD.encode(br#"{"typ":"JWT","alg":"ES256"}"#);
172        let claims = serde_json::json!({ "aud": audience, "exp": expires, "sub": subject });
173        let claims = URL_SAFE_NO_PAD.encode(claims.to_string());
174        let signed = format!("{header}.{claims}");
175        let signature = self
176            .pair
177            .sign(&SystemRandom::new(), signed.as_bytes())
178            .map_err(|_| PushCryptoError::Failed)?;
179        Ok(format!(
180            "{signed}.{}",
181            URL_SAFE_NO_PAD.encode(signature.as_ref())
182        ))
183    }
184}

230    #[test]
231    fn a_fresh_message_has_the_header_and_fits_a_record() {
232        let receiver = PrivateKey::generate(&ECDH_P256).expect("receiver");
233        let ua_public = receiver.compute_public_key().expect("public");
234        let body = encrypt(ua_public.as_ref(), &[7; 16], &[b'x'; MAX_PLAINTEXT]).expect("fits");
235        assert_eq!(body.len(), RECORD_SIZE as usize);
236        assert_eq!(&body[16..20], &RECORD_SIZE.to_be_bytes());
237        assert_eq!(body[20], 65);
238        assert!(matches!(
239            encrypt(ua_public.as_ref(), &[7; 16], &[b'x'; MAX_PLAINTEXT + 1]),
240            Err(PushCryptoError::TooLong)
241        ));
242        assert!(matches!(
243            encrypt(&ua_public.as_ref()[1..], &[7; 16], b"x"),
244            Err(PushCryptoError::InvalidKey("p256dh"))
245        ));
246        assert!(matches!(
247            encrypt(ua_public.as_ref(), &[7; 15], b"x"),
248            Err(PushCryptoError::InvalidKey("auth"))
249        ));
250    }

252    /// The token verifies with the public key the daemon hands browsers,
253    /// and carries `aud`, `exp`, and `sub` (RFC 8292 section 2).
254    #[test]
255    fn the_vapid_token_verifies_with_the_public_key() {
256        let (key, pkcs8) = VapidKey::generate().expect("key");
257        let reloaded = VapidKey::from_pkcs8(&pkcs8).expect("reloads");
258        assert_eq!(reloaded.public_key(), key.public_key());
259        let token = key
260            .token(
261                "https://web.push.apple.com",
262                1_700_000_000,
263                "https://example.com",
264            )
265            .expect("token");
266        let mut parts = token.split('.');
267        let (header, claims, signature) = (
268            parts.next().expect("header"),
269            parts.next().expect("claims"),
270            parts.next().expect("signature"),
271        );
272        assert!(parts.next().is_none());
273        let public = b64(&key.public_key());
274        assert_eq!(public.len(), 65);
275        VerifyingKey::new(&ECDSA_P256_SHA256_FIXED, &public)
276            .verify(format!("{header}.{claims}").as_bytes(), &b64(signature))
277            .expect("verifies");
278        let claims: serde_json::Value = serde_json::from_slice(&b64(claims)).expect("json");
279        assert_eq!(
280            claims,
281            serde_json::json!({
282                "aud": "https://web.push.apple.com",
283                "exp": 1_700_000_000,
284                "sub": "https://example.com",
285            })
286        );
287        assert!(VapidKey::from_pkcs8(b"not a key").is_err());
288    }
