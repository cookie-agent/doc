1//! `<data_root>/push/`: the daemon's VAPID key (`vapid.pk8`) and the
2//! browsers it pushes to (`subscriptions.json`), both owner-only files
3//! written through a flushed temporary file renamed over them (push
4//! notifications spec section 6.4).
5
6use std::{
7    fs, io,
8    path::{Path, PathBuf},
9};
10
11use jiff::Timestamp;
12use serde::{Deserialize, Serialize};
13
14pub(crate) const PUSH_DIR: &str = "push";
15const KEY_FILE: &str = "vapid.pk8";
16const SUBSCRIPTIONS_FILE: &str = "subscriptions.json";
17/// Left by a daemon going down to restart, for the one that comes back.
18const RESTARTING_FILE: &str = "restarting";
19
20/// A browser the daemon pushes to, with what it needs to reach it.
21#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
22#[serde(deny_unknown_fields)]
23pub(crate) struct StoredSubscription {
24    pub(crate) id: String,
25    pub(crate) endpoint: String,
26    pub(crate) p256dh: String,
27    pub(crate) auth: String,
28    pub(crate) label: String,
29    pub(crate) created_at: Timestamp,
30    #[serde(default, skip_serializing_if = "Option::is_none")]
31    pub(crate) last_sent_at: Option<Timestamp>,
32    #[serde(default, skip_serializing_if = "Option::is_none")]
33    pub(crate) last_error: Option<String>,
34}
35
36#[derive(Debug, Default, Deserialize, Serialize)]
37#[serde(deny_unknown_fields)]
38struct SubscriptionsFile {
39    subscriptions: Vec<StoredSubscription>,
40}
41
42/// The `push` folder under the data root.
43pub(crate) struct PushStore {
44    directory: PathBuf,
45}
46
47impl PushStore {
48    pub(crate) fn new(data_root: &Path) -> Self {
49        Self {
50            directory: data_root.join(PUSH_DIR),
51        }
52    }
53
54    /// The stored PKCS#8 key, `None` when there is none yet.
55    pub(crate) fn read_key(&self) -> io::Result<Option<Vec<u8>>> {
56        match fs::read(self.directory.join(KEY_FILE)) {
57            Ok(bytes) => Ok(Some(bytes)),
58            Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(None),
59            Err(error) => Err(error),
60        }
61    }
62
63    pub(crate) fn write_key(&self, pkcs8: &[u8]) -> io::Result<()> {
64        self.write(KEY_FILE, pkcs8)
65    }
66
67    /// Marks that the daemon goes down to restart.
68    pub(crate) fn mark_restarting(&self) -> io::Result<()> {
69        self.write(
70            RESTARTING_FILE,
71            jiff::Timestamp::now().to_string().as_bytes(),
72        )
73    }
74
75    /// Whether the daemon before this one went down to restart, forgetting
76    /// it.
77    pub(crate) fn take_restarting(&self) -> bool {
78        fs::remove_file(self.directory.join(RESTARTING_FILE)).is_ok()
79    }
80
81    /// The stored subscriptions; none when the file does not exist. A file
82    /// that does not parse, or has a member this build does not know, is
83    /// an error.
84    pub(crate) fn read_subscriptions(&self) -> io::Result<Vec<StoredSubscription>> {
85        let path = self.directory.join(SUBSCRIPTIONS_FILE);
86        match fs::read(&path) {
87            Ok(bytes) => serde_json::from_slice::<SubscriptionsFile>(&bytes)
88                .map(|file| file.subscriptions)
89                .map_err(|error| {
90                    io::Error::new(
91                        io::ErrorKind::InvalidData,
92                        format!("{}: {error}", path.display()),
93                    )
94                }),
95            Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(Vec::new()),
96            Err(error) => Err(error),
97        }
98    }
99
100    pub(crate) fn write_subscriptions(
101        &self,
102        subscriptions: &[StoredSubscription],
103    ) -> io::Result<()> {
104        let file = SubscriptionsFile {
105            subscriptions: subscriptions.to_vec(),
106        };
107        let mut bytes = serde_json::to_vec_pretty(&file).expect("the subscriptions serialize");
108        bytes.push(b'\n');
109        self.write(SUBSCRIPTIONS_FILE, &bytes)
110    }
111
112    /// Writes `name` through a flushed temporary file, owner-only on Unix,
113    /// renamed over it.
114    fn write(&self, name: &str, bytes: &[u8]) -> io::Result<()> {
115        fs::create_dir_all(&self.directory)?;
116        let mut random = [0; 8];
117        getrandom::getrandom(&mut random).map_err(io::Error::other)?;
118        let temporary = self.directory.join(format!(
119            ".{name}.{}.tmp",
120            random
121                .iter()
122                .map(|byte| format!("{byte:02x}"))
123                .collect::<String>()
124        ));
125        let result = (|| {
126            let mut options = fs::OpenOptions::new();
127            options.write(true).create_new(true);
128            #[cfg(unix)]
129            std::os::unix::fs::OpenOptionsExt::mode(&mut options, 0o600);
130            let mut output = options.open(&temporary)?;
131            io::Write::write_all(&mut output, bytes)?;
132            output.sync_all()?;
133            drop(output);
134            fs::rename(&temporary, self.directory.join(name))?;
135            #[cfg(unix)]
136            fs::File::open(&self.directory)?.sync_all()?;
137            Ok(())
138        })();
139        if result.is_err() {
140            let _ = fs::remove_file(&temporary);
141        }
142        result
143    }
144}

146#[cfg(test)]
147mod tests {
148    use super::*;
149
150    fn subscription(id: &str) -> StoredSubscription {
151        StoredSubscription {
152            id: id.into(),
153            endpoint: format!("https://push.example/{id}"),
154            p256dh: "key".into(),
155            auth: "auth".into(),
156            label: "Phone".into(),
157            created_at: Timestamp::UNIX_EPOCH,
158            last_sent_at: None,
159            last_error: Some("410".into()),
160        }
161    }
162
163    #[test]
164    fn subscriptions_and_the_key_round_trip_owner_only() {
165        let root = tempfile::tempdir().expect("tempdir");
166        let store = PushStore::new(root.path());
167        assert_eq!(store.read_subscriptions().expect("none yet"), Vec::new());
168        assert_eq!(store.read_key().expect("none yet"), None);
169        store
170            .write_subscriptions(&[subscription("a"), subscription("b")])
171            .expect("writes");
172        store.write_key(b"pkcs8").expect("writes");
173        assert_eq!(
174            store.read_subscriptions().expect("reads"),
175            vec![subscription("a"), subscription("b")]
176        );
177        assert_eq!(
178            store.read_key().expect("reads").as_deref(),
179            Some(&b"pkcs8"[..])
180        );
181        #[cfg(unix)]
182        for name in [KEY_FILE, SUBSCRIPTIONS_FILE] {
183            use std::os::unix::fs::PermissionsExt;
184            let mode = fs::metadata(root.path().join(PUSH_DIR).join(name))
185                .expect("metadata")
186                .permissions()
187                .mode();
188            assert_eq!(mode & 0o777, 0o600, "{name}");
189        }
190    }
191
192    #[test]
193    fn a_subscriptions_file_with_an_unknown_member_is_an_error() {
194        let root = tempfile::tempdir().expect("tempdir");
195        fs::create_dir_all(root.path().join(PUSH_DIR)).expect("dir");
196        fs::write(
197            root.path().join(PUSH_DIR).join(SUBSCRIPTIONS_FILE),
198            r#"{"subscriptions":[],"extra":1}"#,
199        )
200        .expect("write");
201        assert!(PushStore::new(root.path()).read_subscriptions().is_err());
202    }
