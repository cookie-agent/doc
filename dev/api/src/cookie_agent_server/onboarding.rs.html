2//! wire's run request for the engine. There is no starter kit since client
3//! protocol 1.56.0: `onboarding.preview` refuses, and `onboarding.run`
4//! ignores `install_kit` and `source` and writes no files.

16pub(crate) fn requirement_to_wire(
17    requirement: OnboardingRequirement,
18) -> cookie_agent_protocol::OnboardingRequirement {
19    match requirement {
20        OnboardingRequirement::Provider => cookie_agent_protocol::OnboardingRequirement::Provider,
21        OnboardingRequirement::DefaultModel => {
22            cookie_agent_protocol::OnboardingRequirement::DefaultModel
23        }
24        OnboardingRequirement::DefaultAgent => {
25            cookie_agent_protocol::OnboardingRequirement::DefaultAgent
26        }
27    }
28}

30pub(crate) fn status_to_wire(state: OnboardingState) -> OnboardingStatus {
31    let model = |field: &cookie_agent_engine::OnboardingField| {
32        field
33            .first
34            .clone()
35            .map(|(model, variant)| OnboardingModel { model, variant })
36    };
37    let list = |field: &cookie_agent_engine::OnboardingField| {
38        field
39            .configured
40            .entries()
41            .iter()
42            .map(entry_to_wire)
43            .collect::<Vec<_>>()
44    };
45    let paths = cookie_agent_protocol::OnboardingPaths {
46        models: cookie_agent_protocol::OnboardingModelsPath {
47            complete: state.models_complete(),
48            missing: state
49                .models_missing
50                .iter()
51                .copied()
52                .map(requirement_to_wire)
53                .collect(),
54        },
55        acp: cookie_agent_protocol::OnboardingAcpPath {
56            complete: state.acp_complete(),
57            agents: state
58                .acp_agents
59                .iter()
60                .map(|agent| cookie_agent_protocol::OnboardingAcpAgent {
61                    id: agent.id.clone(),
62                    runtime: agent.runtime.clone(),
63                    command: agent.command.clone(),
64                    env_keys: agent.env_keys.clone(),
65                    description: agent.description.clone(),
66                    enabled: agent.enabled,
67                    preset: agent.preset.clone(),
68                    runtimes: agent
69                        .runtimes
70                        .iter()
71                        .map(|(runtime_id, available)| {
72                            cookie_agent_protocol::OnboardingAcpRuntime {
73                                runtime_id: runtime_id.clone(),
74                                available: *available,
75                            }
76                        })
77                        .collect(),
78                })
79                .collect(),
80        },
81    };
82    OnboardingStatus {
83        ready: state.ready(),
84        paths: Some(paths),
85        acp_presets: state.presets.acp.clone(),
86        mcp_presets: state.presets.mcp.clone(),
87        mcp_servers: state
88            .mcp_servers
89            .iter()
90            .map(
91                |(name, runtime)| cookie_agent_protocol::OnboardingMcpServer {
92                    name: name.clone(),
93                    runtime: runtime.clone(),
94                },
95            )
96            .collect(),
97        default_agent: state.default_agent.clone(),
98        missing: state.missing.into_iter().map(requirement_to_wire).collect(),
99        providers_connected: state.providers_connected,
100        default_model: state.default_model.as_ref().and_then(model),
101        small_model: state.small_model.as_ref().and_then(model),
102        utility_model: state.utility_model.as_ref().and_then(model),
103        default_model_list: state.default_model.as_ref().map(list),
104        small_model_list: state.small_model.as_ref().map(list),
105        utility_model_list: state.utility_model.as_ref().map(list),
106    }
107}

109const fn field_to_wire(field: ModelField) -> OnboardingModelField {
110    match field {
111        ModelField::Default => OnboardingModelField::DefaultModel,
112        ModelField::Small => OnboardingModelField::SmallModel,
113        ModelField::Utility => OnboardingModelField::UtilityModel,
114    }
115}

117fn entry_to_wire(entry: &ConfiguredModel) -> OnboardingModelEntry {
118    let (model, field) = match &entry.model {
119        ConfiguredModelRef::Model(model) => (Some(model.clone()), None),
120        ConfiguredModelRef::Field(field) => (None, Some(field_to_wire(*field))),
121    };
122    OnboardingModelEntry {
123        model,
124        field,
125        variant: entry.variant.clone(),
126    }
127}

129fn model_value(model: OnboardingModelInput) -> serde_json::Value {
130    match model {
131        OnboardingModelInput::Key(key) => serde_json::Value::String(key.to_string()),
132        OnboardingModelInput::Table(table) => {
133            serde_json::to_value(table).expect("a model table serializes")
134        }
135    }
136}

138/// `entries` as the configuration takes them: an entry is a model key or a
139/// field's variable, a table when it has a variant, and a list of one is
140/// the entry alone.
141fn list_value(name: &str, entries: Vec<OnboardingModelEntry>) -> Result<serde_json::Value, String> {
142    if entries.is_empty() || entries.len() > cookie_agent_protocol::MAX_ONBOARDING_MODEL_ENTRIES {
143        return Err(format!(
144            "`{name}_list` holds 1 to {} entries",
145            cookie_agent_protocol::MAX_ONBOARDING_MODEL_ENTRIES
146        ));
147    }
148    let mut values = entries
149        .into_iter()
150        .map(|entry| {
151            let model = match (entry.model, entry.field) {
152                (Some(model), None) => model.to_string(),
153                (None, Some(field)) => {
154                    let field = match field {
155                        OnboardingModelField::DefaultModel => ModelField::Default,
156                        OnboardingModelField::SmallModel => ModelField::Small,
157                        OnboardingModelField::UtilityModel => ModelField::Utility,
158                        OnboardingModelField::Unknown => {
159                            return Err(format!("`{name}_list` names an unknown field"));
160                        }
161                    };
162                    field.expression().to_owned()
163                }
164                _ => {
165                    return Err(format!(
166                        "each entry of `{name}_list` has exactly one of `model` and `field`"
167                    ));
168                }
169            };
170            Ok(match entry.variant {
171                None => serde_json::Value::String(model),
172                Some(variant) => serde_json::json!({ "model": model, "variant": variant }),
173            })
174        })
175        .collect::<Result<Vec<_>, _>>()?;
176    Ok(if values.len() == 1 {
177        values.remove(0)
178    } else {
179        serde_json::Value::Array(values)
180    })
181}

183/// The engine's request for `params`; a message for `invalid_params` when a
184/// field is given both as one model and as a list, or a list is malformed.
185pub(crate) fn request_from_wire(params: OnboardingRunParams) -> Result<OnboardingRequest, String> {
186    let field = |name: &str,
187                 single: Option<OnboardingModelInput>,
188                 list: Option<Vec<OnboardingModelEntry>>| {
189        match (single, list) {
190            (Some(_), Some(_)) => Err(format!("give `{name}` or `{name}_list`, not both")),
191            (Some(single), None) => Ok(Some(model_value(single))),
192            (None, Some(list)) => list_value(name, list).map(Some),
193            (None, None) => Ok(None),
194        }
195    };
196    Ok(OnboardingRequest {
197        default_model: field(
198            "default_model",
199            params.default_model,
200            params.default_model_list,
201        )?,
202        small_model: field("small_model", params.small_model, params.small_model_list)?,
203        utility_model: field(
204            "utility_model",
205            params.utility_model,
206            params.utility_model_list,
207        )?,
208        acp_agents: params
209            .acp_agents
210            .unwrap_or_default()
211            .into_iter()
212            .map(|agent| OnboardingAcpRequest {
213                id: agent.id,
214                runtime: agent.runtime,
215                command: agent.command,
216                env: agent.env,
217                description: agent.description,
218                model: agent.model,
219                effort: agent.effort,
220                cache_ttl: agent.cache_ttl,
221            })
222            .collect(),
223        mcp_servers: params
224            .mcp_servers
225            .unwrap_or_default()
226            .into_iter()
227            .map(|server| OnboardingMcpRequest {
228                name: server.name,
229                runtime: server.runtime,
230                server: server.server,
231            })
232            .collect(),
233        default_agent: params.default_agent,
234    })
235}

237pub(crate) fn outcome_to_wire(outcome: OnboardingOutcome) -> OnboardingRunResult {
238    OnboardingRunResult {
239        written: Vec::new(),
240        skipped: Vec::new(),
241        written_new: Vec::new(),
242        settings: outcome.settings,
243        status: status_to_wire(outcome.state),
