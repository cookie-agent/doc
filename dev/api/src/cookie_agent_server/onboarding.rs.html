1//! Onboarding as clients see it: the engine's state on the wire, and the
2//! wire's run request for the engine. There is no starter kit since client
3//! protocol 1.56.0: `onboarding.preview` refuses, and `onboarding.run`
4//! ignores `install_kit` and `source` and writes no files.
5
6use cookie_agent_engine::{
7    OnboardingOutcome, OnboardingRequest, OnboardingRequirement, OnboardingState,
8};
9use cookie_agent_protocol::{
10    OnboardingModel, OnboardingModelInput, OnboardingRunParams, OnboardingRunResult,
11    OnboardingStatus,
12};
13
14pub(crate) fn requirement_to_wire(
15    requirement: OnboardingRequirement,
16) -> cookie_agent_protocol::OnboardingRequirement {
17    match requirement {
18        OnboardingRequirement::Provider => cookie_agent_protocol::OnboardingRequirement::Provider,
19        OnboardingRequirement::DefaultModel => {
20            cookie_agent_protocol::OnboardingRequirement::DefaultModel
21        }
22        OnboardingRequirement::DefaultAgent => {
23            cookie_agent_protocol::OnboardingRequirement::DefaultAgent
24        }
25    }
26}
27
28pub(crate) fn status_to_wire(state: OnboardingState) -> OnboardingStatus {
29    let model = |model: cookie_agent_config::ConfiguredModel| OnboardingModel {
30        model: model.model,
31        variant: model.variant,
32    };
33    OnboardingStatus {
34        ready: state.ready(),
35        missing: state.missing.into_iter().map(requirement_to_wire).collect(),
36        providers_connected: state.providers_connected,
37        default_model: state.default_model.map(model),
38        small_model: state.small_model.map(model),
39        utility_model: state.utility_model.map(model),
40    }
41}
42
43fn model_value(model: OnboardingModelInput) -> serde_json::Value {
44    match model {
45        OnboardingModelInput::Key(key) => serde_json::Value::String(key.to_string()),
46        OnboardingModelInput::Table(table) => {
47            serde_json::to_value(table).expect("a model table serializes")
48        }
49    }
50}
51
52pub(crate) fn request_from_wire(params: OnboardingRunParams) -> OnboardingRequest {
53    OnboardingRequest {
54        default_model: params.default_model.map(model_value),
55        small_model: params.small_model.map(model_value),
56        utility_model: params.utility_model.map(model_value),
57    }
58}
59
60pub(crate) fn outcome_to_wire(outcome: OnboardingOutcome) -> OnboardingRunResult {
61    OnboardingRunResult {
62        written: Vec::new(),
63        skipped: Vec::new(),
64        written_new: Vec::new(),
65        settings: outcome.settings,
66        status: status_to_wire(outcome.state),
67    }
68}
