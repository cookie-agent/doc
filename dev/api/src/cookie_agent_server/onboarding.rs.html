1//! Onboarding as clients see it: the engine's state on the wire, and the
2//! wire's run request for the engine. There is no starter kit since client
3//! protocol 1.56.0: `onboarding.preview` refuses, and `onboarding.run`
4//! ignores `install_kit` and `source` and writes no files.

15pub(crate) fn requirement_to_wire(
16    requirement: OnboardingRequirement,
17) -> cookie_agent_protocol::OnboardingRequirement {
18    match requirement {
19        OnboardingRequirement::Provider => cookie_agent_protocol::OnboardingRequirement::Provider,
20        OnboardingRequirement::DefaultModel => {
21            cookie_agent_protocol::OnboardingRequirement::DefaultModel
22        }
23        OnboardingRequirement::DefaultAgent => {
24            cookie_agent_protocol::OnboardingRequirement::DefaultAgent
25        }
26    }
27}

29pub(crate) fn status_to_wire(state: OnboardingState) -> OnboardingStatus {
30    let model = |field: &cookie_agent_engine::OnboardingField| {
31        field
32            .first
33            .clone()
34            .map(|(model, variant)| OnboardingModel { model, variant })
35    };
36    let list = |field: &cookie_agent_engine::OnboardingField| {
37        field
38            .configured
39            .entries()
40            .iter()
41            .map(entry_to_wire)
42            .collect::<Vec<_>>()
43    };
44    OnboardingStatus {
45        ready: state.ready(),
46        missing: state.missing.into_iter().map(requirement_to_wire).collect(),
47        providers_connected: state.providers_connected,
48        default_model: state.default_model.as_ref().and_then(model),
49        small_model: state.small_model.as_ref().and_then(model),
50        utility_model: state.utility_model.as_ref().and_then(model),
51        default_model_list: state.default_model.as_ref().map(list),
52        small_model_list: state.small_model.as_ref().map(list),
53        utility_model_list: state.utility_model.as_ref().map(list),
54    }
55}

57const fn field_to_wire(field: ModelField) -> OnboardingModelField {
58    match field {
59        ModelField::Default => OnboardingModelField::DefaultModel,
60        ModelField::Small => OnboardingModelField::SmallModel,
61        ModelField::Utility => OnboardingModelField::UtilityModel,
62    }
63}

65fn entry_to_wire(entry: &ConfiguredModel) -> OnboardingModelEntry {
66    let (model, field) = match &entry.model {
67        ConfiguredModelRef::Model(model) => (Some(model.clone()), None),
68        ConfiguredModelRef::Field(field) => (None, Some(field_to_wire(*field))),
69    };
70    OnboardingModelEntry {
71        model,
72        field,
73        variant: entry.variant.clone(),
74    }
75}

77fn model_value(model: OnboardingModelInput) -> serde_json::Value {
78    match model {
79        OnboardingModelInput::Key(key) => serde_json::Value::String(key.to_string()),
80        OnboardingModelInput::Table(table) => {
81            serde_json::to_value(table).expect("a model table serializes")
82        }
83    }
84}

86/// `entries` as the configuration takes them: an entry is a model key or a
87/// field's variable, a table when it has a variant, and a list of one is
88/// the entry alone.
89fn list_value(name: &str, entries: Vec<OnboardingModelEntry>) -> Result<serde_json::Value, String> {
90    if entries.is_empty() || entries.len() > cookie_agent_protocol::MAX_ONBOARDING_MODEL_ENTRIES {
91        return Err(format!(
92            "`{name}_list` holds 1 to {} entries",
93            cookie_agent_protocol::MAX_ONBOARDING_MODEL_ENTRIES
94        ));
95    }
96    let mut values = entries
97        .into_iter()
98        .map(|entry| {
99            let model = match (entry.model, entry.field) {
100                (Some(model), None) => model.to_string(),
101                (None, Some(field)) => {
102                    let field = match field {
103                        OnboardingModelField::DefaultModel => ModelField::Default,
104                        OnboardingModelField::SmallModel => ModelField::Small,
105                        OnboardingModelField::UtilityModel => ModelField::Utility,
106                        OnboardingModelField::Unknown => {
107                            return Err(format!("`{name}_list` names an unknown field"));
108                        }
109                    };
110                    field.expression().to_owned()
111                }
112                _ => {
113                    return Err(format!(
114                        "each entry of `{name}_list` has exactly one of `model` and `field`"
115                    ));
116                }
117            };
118            Ok(match entry.variant {
119                None => serde_json::Value::String(model),
120                Some(variant) => serde_json::json!({ "model": model, "variant": variant }),
121            })
122        })
123        .collect::<Result<Vec<_>, _>>()?;
124    Ok(if values.len() == 1 {
125        values.remove(0)
126    } else {
127        serde_json::Value::Array(values)
128    })
129}

131/// The engine's request for `params`; a message for `invalid_params` when a
132/// field is given both as one model and as a list, or a list is malformed.
133pub(crate) fn request_from_wire(params: OnboardingRunParams) -> Result<OnboardingRequest, String> {
134    let field = |name: &str,
135                 single: Option<OnboardingModelInput>,
136                 list: Option<Vec<OnboardingModelEntry>>| {
137        match (single, list) {
138            (Some(_), Some(_)) => Err(format!("give `{name}` or `{name}_list`, not both")),
139            (Some(single), None) => Ok(Some(model_value(single))),
140            (None, Some(list)) => list_value(name, list).map(Some),
141            (None, None) => Ok(None),
142        }
143    };
144    Ok(OnboardingRequest {
145        default_model: field(
146            "default_model",
147            params.default_model,
148            params.default_model_list,
149        )?,
150        small_model: field("small_model", params.small_model, params.small_model_list)?,
151        utility_model: field(
152            "utility_model",
153            params.utility_model,
154            params.utility_model_list,
155        )?,
156    })
157}

159pub(crate) fn outcome_to_wire(outcome: OnboardingOutcome) -> OnboardingRunResult {
160    OnboardingRunResult {
161        written: Vec::new(),
162        skipped: Vec::new(),
163        written_new: Vec::new(),
164        settings: outcome.settings,
165        status: status_to_wire(outcome.state),
166    }
