15pub(crate) fn requirement_to_wire(
16    requirement: OnboardingRequirement,
17) -> cookie_agent_protocol::OnboardingRequirement {
18    match requirement {
19        OnboardingRequirement::Provider => cookie_agent_protocol::OnboardingRequirement::Provider,
20        OnboardingRequirement::DefaultModel => {
21            cookie_agent_protocol::OnboardingRequirement::DefaultModel
22        }
23        OnboardingRequirement::SmallModel => {
24            cookie_agent_protocol::OnboardingRequirement::SmallModel
25        }
26    }
27}

29pub(crate) fn status_to_wire(state: OnboardingState) -> OnboardingStatus {
30    let model = |model: cookie_agent_config::ConfiguredModel| OnboardingModel {
31        model: model.model,
32        variant: model.variant,
33    };
34    OnboardingStatus {
35        ready: state.ready(),
36        missing: state.missing.into_iter().map(requirement_to_wire).collect(),
37        providers_connected: state.providers_connected,
38        default_model: state.default_model.map(model),
39        small_model: state.small_model.map(model),
40    }
41}

43pub(crate) fn source_from_wire(
44    source: Option<OnboardingKitSource>,
45) -> Result<KitSource, EngineError> {
46    match source.unwrap_or_default() {
47        OnboardingKitSource::Release => Ok(KitSource::Release),
48        OnboardingKitSource::Archive { data } => {
49            let bytes = base64::engine::general_purpose::STANDARD
50                .decode(data.as_bytes())
51                .map_err(|_| {
52                    EngineError::OnboardingKit(cookie_agent_engine::KitError::Archive(
53                        "the archive is not standard base64".into(),
54                    ))
55                })?;
56            if bytes.len() > MAX_ONBOARDING_ARCHIVE_BYTES {
57                return Err(EngineError::OnboardingKit(
58                    cookie_agent_engine::KitError::TooLarge,
59                ));
60            }
61            Ok(KitSource::Archive(bytes))
62        }
63    }
64}

66pub(crate) fn preview_to_wire(preview: KitPreview) -> OnboardingPreviewResult {
67    OnboardingPreviewResult {
68        kit_version: preview.manifest.kit_version,
69        cookie_version: preview.manifest.cookie_version,
70        origin: preview.origin,
71        files: preview
72            .files
73            .into_iter()
74            .map(|file| OnboardingKitFile {
75                path: file.path.as_str().to_owned(),
76                action: match file.action {
77                    KitFileAction::Write => OnboardingFileAction::Write,
78                    KitFileAction::Skip => OnboardingFileAction::Skip,
79                    KitFileAction::WriteNew => OnboardingFileAction::WriteNew,
80                },
81                diff: file.diff,
82            })
83            .collect(),
84    }
85}

87fn model_value(model: OnboardingModelInput) -> serde_json::Value {
88    match model {
89        OnboardingModelInput::Key(key) => serde_json::Value::String(key.to_string()),
90        OnboardingModelInput::Table(table) => {
91            serde_json::to_value(table).expect("a model table serializes")
92        }
93    }
94}

96pub(crate) fn request_from_wire(
97    params: OnboardingRunParams,
98) -> Result<OnboardingRequest, EngineError> {
99    Ok(OnboardingRequest {
100        install_kit: params.install_kit,
101        source: if params.install_kit {
102            source_from_wire(params.source)?
103        } else {
104            KitSource::Release
105        },
106        default_model: params.default_model.map(model_value),
107        small_model: params.small_model.map(model_value),
108    })
109}

111pub(crate) fn outcome_to_wire(outcome: OnboardingOutcome) -> OnboardingRunResult {

115        written_new: outcome.written_new,
116        settings: outcome.settings,
