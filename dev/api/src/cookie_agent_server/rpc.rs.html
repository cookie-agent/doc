55    pub(crate) fn provider_connect(request: &ProviderConnectParams, error: EngineError) -> Self {
56        let code = match &error {
57            EngineError::ModelManager(error) => match error {
58                ModelManagerError::UnknownProvider => &errors::UNKNOWN_PROVIDER,
59                ModelManagerError::UnsupportedProvider => &errors::UNSUPPORTED_PROVIDER,
60                ModelManagerError::RemovedWithoutRetainedRecipeMatch => {
61                    &errors::REMOVED_WITHOUT_RETAINED_RECIPE_MATCH
62                }
63                ModelManagerError::CustomProviderNotStoreBacked => {
64                    &errors::CUSTOM_PROVIDER_NOT_STORE_BACKED
65                }
66                ModelManagerError::InvalidSetup => &errors::INVALID_SETUP_FIELD,
67                ModelManagerError::MissingSetup => &errors::MISSING_SETUP_FIELD,
68                ModelManagerError::MissingCredentials => &errors::MISSING_CREDENTIAL,
69                ModelManagerError::UnsupportedAuthMethod => &errors::UNSUPPORTED_AUTH_METHOD,
70                ModelManagerError::InvalidCredentials => &errors::INVALID_CREDENTIAL,
71                ModelManagerError::ProviderStore(ProviderStoreError::CatalogRevisionConflict) => {
72                    &errors::CATALOG_REVISION_CONFLICT
73                }
74                ModelManagerError::ProviderStore(ProviderStoreError::IdempotencyConflict) => {
75                    &errors::IDEMPOTENCY_CONFLICT
76                }
77                ModelManagerError::ProviderStore(ProviderStoreError::Contention)
78                | ModelManagerError::Contention => &errors::LOCK_CONTENTION,
79                ModelManagerError::ProviderStore(_) => &errors::PROVIDER_STORE_WRITE_FAILED,
80                _ => &errors::RUNTIME_COMPILE_FAILED,
81            },
82            _ => &errors::RUNTIME_COMPILE_FAILED,
83        };
84        let mut builder = code
85            .error()
86            .field("provider_id", &request.provider_id)
87            .detail(error.user_message());
88        if std::ptr::eq(code, &errors::IDEMPOTENCY_CONFLICT) {
89            builder = builder.field("client_connect_id", &request.client_connect_id);
90        }
91        Self::of(builder)
92    }

94    pub(crate) fn provider_disconnect(
95        request: &ProviderDisconnectParams,
96        error: EngineError,
97    ) -> Self {
98        let code = match &error {
99            EngineError::ModelManager(error) => match error {
100                ModelManagerError::CustomProviderNotStoreBacked => {
101                    &errors::CUSTOM_PROVIDER_NOT_STORE_BACKED
102                }
103                ModelManagerError::ProviderStore(ProviderStoreError::RuntimeRevisionConflict) => {
104                    &errors::RUNTIME_REVISION_CONFLICT
105                }
106                ModelManagerError::ProviderStore(
107                    ProviderStoreError::ProviderStateRevisionConflict,
108                ) => &errors::PROVIDER_STATE_REVISION_CONFLICT,
109                ModelManagerError::ProviderStore(ProviderStoreError::StaleConnectionGeneration) => {
110                    &errors::STALE_PROVIDER_CONNECTION_GENERATION
111                }
112                ModelManagerError::ProviderStore(ProviderStoreError::IdempotencyConflict) => {
113                    &errors::IDEMPOTENCY_CONFLICT
114                }
115                ModelManagerError::ProviderStore(ProviderStoreError::Contention)
116                | ModelManagerError::Contention => &errors::LOCK_CONTENTION,
117                ModelManagerError::ProviderStore(_) => &errors::PROVIDER_STORE_WRITE_FAILED,
118                _ => &errors::RUNTIME_COMPILE_FAILED,
119            },
120            _ => &errors::RUNTIME_COMPILE_FAILED,
121        };
122        let mut builder = code
123            .error()
124            .field("provider_id", &request.provider_id)
125            .detail(error.user_message());
126        if std::ptr::eq(code, &errors::IDEMPOTENCY_CONFLICT) {
127            builder = builder.field("client_request_id", &request.client_request_id);
128        }
129        Self::of(builder)
130    }

132    /// A refused `config.apply`: a conflict, an invalid result, or a change
133    /// that cannot be made, with why as the detail; anything else is an
134    /// ordinary engine fault.
135    pub(crate) fn config_apply(error: EngineError) -> Self {
136        let (reason, detail) = match error {
137            EngineError::ConfigConflict(message) => (ConfigApplyErrorCode::Conflict, message),
138            EngineError::ConfigChangeRefused(message) => {
139                (ConfigApplyErrorCode::InvalidChange, message)
140            }
141            EngineError::Config(error) => (
142                ConfigApplyErrorCode::InvalidConfiguration,
143                error.to_string(),
144            ),
145            error => return engine_fault(error),
146        };
147        Self::of(
148            errors::CONFIG_APPLY_REFUSED
149                .error()
150                .field("reason", reason)
151                .detail(detail),
152        )
153    }

155    pub(crate) fn rename(request: &SessionRenameParams, error: EngineError) -> Self {
156        let builder = match &error {
157            EngineError::RenameConflict => errors::IDEMPOTENCY_CONFLICT
158                .error()
159                .field("client_rename_id", &request.client_rename_id),
160            EngineError::Session(SessionError::Missing(_)) | EngineError::MissingActor(_) => {
161                errors::SESSION_NOT_FOUND.error()
162            }
163            _ => return engine_fault(error),
164        };
165        Self::of(builder.field("session_id", request.session_id))
166    }

168    pub(crate) fn approval(request: &ApprovalRespondParams, error: EngineError) -> Self {
169        match error {
170            EngineError::ApprovalResponse(failure) => Self::approval_failure(request, &failure),
171            EngineError::ApprovalNotPending { .. } => Self::approval_code(
172                request,
173                ApprovalRespondErrorCode::ApprovalNotPending,
174                None,
175                None,
176            ),
177            EngineError::Session(SessionError::Missing(_)) | EngineError::MissingActor(_) => {
178                Self::approval_code(
179                    request,
180                    ApprovalRespondErrorCode::ApprovalNotFound,
181                    None,
182                    None,
183                )
184            }
185            error => engine_fault(error),
186        }
187    }
188
189    fn approval_code(
190        request: &ApprovalRespondParams,
191        code: ApprovalRespondErrorCode,
192        expected_revision: Option<u64>,
193        found_revision: Option<u64>,
194    ) -> Self {
195        let mut builder = code
196            .error_code()
197            .error()
198            .field("approval_id", request.approval_id);
199        match code {
200            ApprovalRespondErrorCode::ApprovalRevisionConflict => {
201                builder = builder
202                    .field("expected_revision", expected_revision)
203                    .field("found_revision", found_revision);
204            }
205            ApprovalRespondErrorCode::IdempotencyConflict => {
206                builder = builder
207                    .field("session_id", request.session_id)
208                    .field("client_response_id", &request.client_response_id);
209            }
210            _ => {}
211        }
212        Self::of(builder)
213    }
214
215    fn approval_failure(request: &ApprovalRespondParams, failure: &ApprovalRespondFailure) -> Self {
216        let (expected_revision, found_revision) =
217            if failure.code == ApprovalRespondErrorCode::ApprovalRevisionConflict {
218                (failure.current_revision, Some(request.request_revision))
219            } else {
220                (None, None)
221            };
222        let client_response_id = ClientResponseId::new(failure.client_response_id.clone())
223            .unwrap_or_else(|_| request.client_response_id.clone());
224        let request = ApprovalRespondParams {
225            session_id: failure.session_id,
226            approval_id: failure.approval_id,
227            client_response_id,
228            ..request.clone()
229        };
230        Self::approval_code(&request, failure.code, expected_revision, found_revision)
231    }

233    pub(crate) fn session_tree_usage(error: EngineError) -> Self {
234        match error {
235            EngineError::Session(SessionError::Missing(session_id)) => Self::of(
236                errors::TREE_USAGE_SESSION_MISSING
237                    .error()
238                    .field("session_id", session_id),
239            ),
240            EngineError::DelegationEvents(DelegationEventError::Corrupt(invocation_id)) => {
241                Self::of(
242                    errors::TREE_USAGE_DELEGATION_CORRUPT
243                        .error()
244                        .field("invocation_id", invocation_id),
245                )
246            }
247            error => engine_fault(error),
248        }
249    }

293/// The registry error that answers `error`: a specific code where the
294/// engine's error names one, else `engine_error`. The engine's own
295/// description is the detail.
296pub(crate) fn engine_fault(error: EngineError) -> RpcFault {
297    let detail = error.user_message();
298    match event_log_error(&error) {
299        Some(EventLogError::RequiresNewerEngine {
300            event,
301            history_format,
302            ..
303        }) => {
304            return RpcFault::of(
305                errors::SESSION_REQUIRES_NEWER_ENGINE
306                    .error()
307                    .field("event", event)
308                    .field("history_format", history_format)
309                    .detail(detail),
310            );
311        }
312        Some(EventLogError::SequenceBroken { seq, .. }) => {
313            return RpcFault::of(
314                errors::SEQUENCE_BROKEN
315                    .error()
316                    .field("seq", seq)
317                    .detail(detail),
318            );
319        }
320        _ => {}
321    }
322    let builder = match &error {
323        EngineError::Media(error) => media_error(error),
324        EngineError::PluginHookFailed(failure) => errors::PLUGIN_HOOK_FAILED
325            .error()
326            .field("plugin", &failure.plugin)
327            .field("hook", failure.hook.name()),
328        EngineError::NoRunnableModel => errors::NO_RUNNABLE_MODEL.error(),
329        EngineError::OnboardingIncomplete(missing) => errors::ONBOARDING_INCOMPLETE.error().field(
330            "missing",
331            missing
332                .iter()
333                .copied()
334                .map(crate::onboarding::requirement_to_wire)
335                .collect::<Vec<_>>(),
336        ),
337        EngineError::OnboardingKit(_) => errors::ONBOARDING_KIT_UNAVAILABLE.error(),
338        EngineError::ProviderStoreReloadFailed => errors::PROVIDER_STORE_RELOAD_FAILED.error(),
339        EngineError::RuntimeCompileFailed => errors::RUNTIME_COMPILE_FAILED.error(),
340        EngineError::SessionOwnedByAnotherProcess(session_id) => errors::SESSION_OWNED_ELSEWHERE
341            .error()
342            .field("session_id", session_id),
343        EngineError::Session(SessionError::Missing(session_id))
344        | EngineError::MissingActor(session_id) => errors::SESSION_NOT_FOUND
345            .error()
346            .field("session_id", session_id),
347        EngineError::SessionRunning(session_id) => errors::SESSION_RUNNING
348            .error()
349            .field("session_id", session_id),
350        EngineError::MissingRun(run_id) => errors::RUN_NOT_FOUND.error().field("run_id", run_id),
351        EngineError::Restarting => errors::DAEMON_RESTARTING.error(),
352        EngineError::RestartUnsupported => errors::RESTART_UNSUPPORTED.error(),
353        EngineError::StaleRevert {
354            session_id,
355            current_last_seq,
356        } => errors::STALE_REVERT
357            .error()
358            .field("session_id", session_id)
359            .field("current_last_seq", current_last_seq),
360        EngineError::NotARootSession {
361            session_id,
362            root_session_id,
363        } => errors::NOT_A_ROOT_SESSION
364            .error()
365            .field("session_id", session_id)
366            .field("root_session_id", root_session_id),
367        EngineError::SessionTreeRunning { root, busy } => errors::SESSION_TREE_RUNNING
368            .error()
369            .field("session_id", root)
370            .field("running_session_id", busy),
371        EngineError::StaleDelete {
372            session_id,
373            current_last_seq,
374        } => errors::SESSION_CHANGED
375            .error()
376            .field("session_id", session_id)
377            .field("current_last_seq", current_last_seq),
378        EngineError::ForkIdempotencyConflict
379        | EngineError::RunIdempotencyConflict
380        | EngineError::RenameConflict => errors::IDEMPOTENCY_CONFLICT.error(),
381        EngineError::Goal(reason) => errors::GOAL_REJECTED
382            .error()
383            .field("reason", reason_text(reason)),
384        EngineError::Producer(_) => errors::PRODUCER_REJECTED.error(),
385        EngineError::Plugin(reason)
386        | EngineError::InputHandled(reason)
387        | EngineError::ModelSelectionBlocked(reason)
388        | EngineError::SessionOperationBlocked(reason)
389        | EngineError::CompactionCancelled(reason) => errors::PLUGIN_REJECTED
390            .error()
391            .field("reason", reason_text(reason)),
392        EngineError::AcpUnsupported(reason) => errors::ACP_AGENT_UNSUPPORTED
393            .error()
394            .field("reason", reason_text(reason)),
395        EngineError::Runtime(reason) => errors::RUNTIME_REJECTED
396            .error()
397            .field("reason", reason_text(reason)),
398        EngineError::Machine(error) => machine_error(error),
399        EngineError::MemoryRequest(error) => memory_error(error),
400        EngineError::Project(error) => project_error(error),
401        EngineError::Cron(error) => cron_error(error),
402        EngineError::Mcp(reason) => errors::MCP_REJECTED
403            .error()
404            .field("reason", reason_text(reason)),
405        EngineError::ProviderSignIn {
406            provider_id,
407            reason,
408        } => errors::PROVIDER_SIGN_IN_REJECTED
409            .error()
410            .field("provider_id", provider_id)
411            .field("reason", reason_text(reason)),
412        EngineError::UnreadableModelBinding(model) => errors::UNREADABLE_MODEL_BINDING
413            .error()
414            .field("model", model),
415        EngineError::IneligibleAgent(agent) | EngineError::DisabledAgent(agent) => {
416            errors::AGENT_UNAVAILABLE.error().field("agent", agent)
417        }
418        EngineError::UnknownAgentModel { agent, .. }
419        | EngineError::UnavailableAgentModel { agent, .. } => {
420            errors::AGENT_UNAVAILABLE.error().field("agent", agent)
421        }
422        EngineError::UnknownAgentPreset(_) => errors::AGENT_UNAVAILABLE.error(),
423        EngineError::ApprovalNotPending { approval_id, .. } => errors::APPROVAL_NOT_PENDING
424            .error()
425            .field("approval_id", approval_id),
426        EngineError::ElicitationNotPending(elicitation_id) => errors::ELICITATION_NOT_PENDING
427            .error()
428            .field("elicitation_id", elicitation_id),
429        EngineError::ElicitationAnswerInvalid(elicitation_id, reason) => {
430            errors::ELICITATION_ANSWER_INVALID
431                .error()
432                .field("elicitation_id", elicitation_id)
433                .field("reason", reason)
434        }
435        _ => errors::ENGINE_FAILED.error(),
436    };
437    RpcFault::of(builder.detail(detail))
438}

440/// The registry error of a refused upload, read, or input media.
441fn media_error(error: &MediaError) -> ErrorBuilder {
442    match error {
443        MediaError::UploadNotFound => errors::UPLOAD_NOT_FOUND.error(),
444        MediaError::OffsetMismatch {
445            expected_offset, ..
446        } => errors::UPLOAD_OFFSET_MISMATCH
447            .error()
448            .field("expected_offset", expected_offset),
449        MediaError::SizeMismatch {
450            byte_length,
451            received,
452        } => errors::UPLOAD_SIZE_MISMATCH
453            .error()
454            .field("byte_length", byte_length)
455            .field("received", received),
456        MediaError::TooManyUploads { max_uploads } => errors::TOO_MANY_UPLOADS
457            .error()
458            .field("max_uploads", max_uploads),
459        MediaError::DigestMismatch { .. } => errors::DIGEST_MISMATCH.error(),
460        MediaError::TypeUnsupported { media_type, .. } => errors::MEDIA_TYPE_UNSUPPORTED
461            .error()
462            .field("media_type", media_type),
463        MediaError::TooLarge {
464            kind,
465            byte_length,
466            max_bytes,
467            width,
468            height,
469            max_dimension,
470            max_pixels,
471        } => errors::MEDIA_TOO_LARGE
472            .error()
473            .field("kind", kind)
474            .field("byte_length", byte_length)
475            .field("max_bytes", max_bytes)
476            .field("width", width)
477            .field("height", height)
478            .field("max_dimension", max_dimension)
479            .field("max_pixels", max_pixels),
480        MediaError::Unsupported { kind, model } => errors::MEDIA_UNSUPPORTED
481            .error()
482            .field("kind", kind)
483            .field("model", model),
484        MediaError::LimitExceeded {
485            kind,
486            count,
487            max_count,
488        } => errors::MEDIA_LIMIT_EXCEEDED
489            .error()
490            .field("kind", kind)
491            .field("count", count)
492            .field("max_count", max_count),
493        MediaError::ArtifactNotFound { .. } => errors::ARTIFACT_NOT_FOUND.error(),
494        MediaError::InvalidRequest(_) => errors::BAD_PARAMS.error(),
495    }
496}

498/// The registry entry `error` is answered with.
499#[cfg(test)]
500fn code_of(fault: RpcFault) -> &'static errors::ErrorCode {
501    let code = fault
502        .0
503        .data
504        .as_ref()
505        .and_then(|data| data["code"].as_str())
506        .expect("registry data");
507    errors::lookup(code).expect("a registered code")
508}

525        MemoryRequestError::RevisionMismatch { url, revision } => errors::MEMORY_REVISION_MISMATCH
526            .error()
527            .field("url", url)
528            .field("revision", revision),

536            .field("reason", reason_text(reason)),
537        MemoryRequestError::Busy(_) => errors::MEMORY_BUSY.error(),

611            .error()
612            .field("runtime", runtime.as_str()),
613        MachineError::Store(_) => errors::MACHINE_STORE_FAILED.error(),
614    }
