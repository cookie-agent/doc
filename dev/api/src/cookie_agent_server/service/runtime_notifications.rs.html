7    pub(super) fn start_runtime_notifications(&self, context: ServerContext) {
8        let mut receiver = self.engine.subscribe_runtime_changes();
9        let startup = self.engine.current_runtime().result.snapshot.clone();
10        let engine = self.engine.clone();
11        tokio::spawn(async move {
12            let shutdown = context.shutdown();
13            let startup = cookie_agent_protocol::RuntimeChangedNotification {
14                previous_revision: None,
15                snapshot: startup,
16                reasons: vec![cookie_agent_protocol::RuntimeChangeReason::Startup],
17            };
18            if context
19                .notify(cookie_agent_protocol::RUNTIME_CHANGED_METHOD, &startup)
20                .await
21                .is_err()
22            {
23                return;
24            }
25            loop {
26                let changed = tokio::select! {
27                    _ = shutdown.cancelled() => return,
28                    changed = receiver.recv() => match changed {
29                        Ok(changed) => changed,
30                        // The skipped changes broke this connection's revision
31                        // chain: hand it the current runtime to start over from.
32                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => {
33                            cookie_agent_protocol::RuntimeChangedNotification {
34                                previous_revision: None,
35                                snapshot: engine.current_runtime().result.snapshot.clone(),
36                                reasons: vec![cookie_agent_protocol::RuntimeChangeReason::Resync],
37                            }
38                        }
39                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
40                    },
41                };
42                if context
43                    .notify(cookie_agent_protocol::RUNTIME_CHANGED_METHOD, &changed)
44                    .await
45                    .is_err()
46                {
47                    return;
48                }
49            }
50        });
51    }

82    /// Signals MCP server changes to this connection; bursts coalesce.
83    /// A connection failure is sent on its own, naming the subscribed
84    /// sessions that use the server.
85    pub(super) fn start_mcp_change_notifications(&self, context: ServerContext) {
86        let mut receiver = self.engine.subscribe_mcp_changes();
87        receiver.mark_unchanged();
88        let mut failures = self.engine.subscribe_mcp_failures();
89        let engine = self.engine.clone();
90        tokio::spawn(async move {
91            let shutdown = context.shutdown();
92            loop {
93                let notification = tokio::select! {
94                    _ = shutdown.cancelled() => return,
95                    changed = receiver.changed() => {
96                        if changed.is_err() {
97                            return;
98                        }
99                        cookie_agent_protocol::McpChangedNotification::default()
100                    }
101                    failure = failures.recv() => match failure {
102                        Ok(failure) => cookie_agent_protocol::McpChangedNotification {
103                            failures: mcp_failure(&engine, &context.subscribed_sessions(), failure)
104                                .into_iter()
105                                .collect(),
106                        },
107                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => continue,
108                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
109                    },
110                };
111                if context
112                    .notify(cookie_agent_protocol::MCP_CHANGED_METHOD, &notification)
113                    .await
114                    .is_err()
115                {
116                    return;
117                }
118            }
119        });
120    }

252    /// Sends the current background calls of a subscribed session whenever
253    /// they change. A connection that falls behind gets every subscribed
254    /// session's list again.
255    pub(super) fn start_background_notifications(&self, context: ServerContext) {
256        let mut receiver = self.engine.subscribe_background_changes();
257        let engine = self.engine.clone();
258        tokio::spawn(async move {
259            let shutdown = context.shutdown();
260            loop {
261                let sessions = tokio::select! {
262                    _ = shutdown.cancelled() => return,
263                    changed = receiver.recv() => match changed {
264                        Ok(session) => context
265                            .subscribed_sessions()
266                            .into_iter()
267                            .filter(|subscribed| *subscribed == session)
268                            .collect::<Vec<_>>(),
269                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => {
270                            context.subscribed_sessions()
271                        }
272                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
273                    },
274                };
275                for session_id in sessions {
276                    let notification = cookie_agent_protocol::BackgroundChangedNotification {
277                        session_id,
278                        calls: engine.background_calls(session_id),
279                    };
280                    if context
281                        .notify(
282                            cookie_agent_protocol::BACKGROUND_CHANGED_METHOD,
283                            &notification,
284                        )
285                        .await
286                        .is_err()
287                    {
288                        return;
289                    }
290                }
291            }
292        });
293    }

295    /// Tells a connection that just subscribed to `session` about the
296    /// failures that already affect it: failures reported before the
297    /// subscription named no session of this connection.
298    pub(super) fn replay_connection_failures(&self, context: &ServerContext, session: SessionId) {
299        let sessions = [session];
300        let mcp = self
301            .engine
302            .mcp_failures()
303            .into_iter()
304            .filter_map(|failure| mcp_failure(&self.engine, &sessions, failure))
305            .collect::<Vec<_>>();
306        let runtimes = self
307            .engine
308            .runtime_instances()
309            .into_iter()
310            .filter(|instance| {
311                instance.status == cookie_agent_engine::RuntimeInstanceStatus::Failed
312            })
313            .filter_map(|instance| {
314                runtime_failure(
315                    &self.engine,
316                    &sessions,
317                    cookie_agent_engine::RuntimeFailure {
318                        runtime_id: instance.runtime_id,
319                        capability: instance.capability,
320                        workspace: instance.workspace,
321                        reason: instance.message.unwrap_or_default(),
322                    },
323                )
324            })
325            .collect::<Vec<_>>();
326        if mcp.is_empty() && runtimes.is_empty() {
327            return;
328        }
329        let context = context.clone();
330        tokio::spawn(async move {
331            if !mcp.is_empty() {
332                let notification = cookie_agent_protocol::McpChangedNotification { failures: mcp };
333                if context
334                    .notify(cookie_agent_protocol::MCP_CHANGED_METHOD, &notification)
335                    .await
336                    .is_err()
337                {
338                    return;
339                }
340            }
341            if !runtimes.is_empty() {
342                let notification =
343                    cookie_agent_protocol::RuntimesChangedNotification { failures: runtimes };
344                let _ = context
345                    .notify(
346                        cookie_agent_protocol::RUNTIMES_CHANGED_METHOD,
347                        &notification,
348                    )
349                    .await;
350            }
351        });
352    }

354    pub(crate) fn start_engine_event_notifications(&self, context: ServerContext) {
355        let mut receiver = self.engine.subscribe_engine_events();
356        tokio::spawn(async move {
357            let shutdown = context.shutdown();
358            loop {
359                let event = tokio::select! {
360                    _ = shutdown.cancelled() => return,
361                    event = receiver.recv() => match event {
362                        Ok(event) => event,
363                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => continue,
364                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
365                    },
366                };
367                let cookie_agent_engine::EngineEvent::PluginEvent {
368                    session_id,
369                    plugin,
370                    name,
371                    payload,
372                } = event;
373                if !context.is_session_subscribed(session_id) {
374                    continue;
375                }
376                let params = cookie_agent_protocol::ExtensionBusEventParams {
377                    session_id,
378                    context_id: None,
379                    plugin,
380                    name,
381                    payload,
382                };
383                if context.notify("events.plugin", &params).await.is_err() {
384                    return;
385                }
386            }
387        });
388    }

391/// The sessions among `sessions` attached to a runtime that `matches`.
392fn affected_sessions(
393    engine: &Engine,
394    sessions: &[SessionId],
395    matches: impl Fn(&cookie_agent_protocol::RuntimeAttachment) -> bool,
396) -> Vec<SessionId> {
397    let mut affected = sessions
398        .iter()
399        .copied()
400        .filter(|session| {
401            engine
402                .session_runtimes(*session)
403                .is_ok_and(|attachments| attachments.iter().any(&matches))
404        })
405        .collect::<Vec<_>>();
406    affected.sort_unstable();
407    affected
408}

410/// `failure` as a client sees it, when one of `sessions` uses the server:
411/// it is attached to the instance that hosts it. Top-level MCP servers
412/// belong to the global runtime.
413fn mcp_failure(
414    engine: &Engine,
415    sessions: &[SessionId],
416    failure: cookie_agent_engine::McpServerFailure,
417) -> Option<cookie_agent_protocol::McpServerFailure> {
418    let runtime_id = RuntimeId::new(failure.runtime_id.as_str()).ok()?;
419    let session_ids = affected_sessions(engine, sessions, |attachment| {
420        attachment.runtime_id == runtime_id && attachment.workspace == failure.workspace
421    });
422    (!session_ids.is_empty()).then(|| cookie_agent_protocol::McpServerFailure {
423        runtime_id: runtime_id.clone(),
424        workspace: failure.workspace,
425        server: failure.server,
426        reason: failure.reason,
427        session_ids,
428    })
429}

431/// `failure` as a client sees it, when one of `sessions` is attached to the
432/// instance.
433fn runtime_failure(
434    engine: &Engine,
435    sessions: &[SessionId],
436    failure: cookie_agent_engine::RuntimeFailure,
437) -> Option<cookie_agent_protocol::RuntimeFailure> {
438    let runtime_id = RuntimeId::new(failure.runtime_id.as_str()).ok()?;
439    let capability = cookie_agent_protocol::CapabilityName::new(failure.capability).ok()?;
440    let session_ids = affected_sessions(engine, sessions, |attachment| {
441        attachment.runtime_id == runtime_id
442            && attachment.capability == capability
443            && attachment.workspace == failure.workspace
444    });
445    (!session_ids.is_empty()).then(|| cookie_agent_protocol::RuntimeFailure {
446        runtime_id: runtime_id.clone(),
447        capability: capability.clone(),
448        workspace: failure.workspace,
449        reason: failure.reason,
450        session_ids,
451    })
