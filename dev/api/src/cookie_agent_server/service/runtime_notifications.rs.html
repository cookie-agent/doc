7    pub(super) fn start_runtime_notifications(&self, context: ServerContext) {
8        let mut receiver = self.engine.subscribe_runtime_changes();
9        let startup = self.engine.current_runtime().result.snapshot.clone();
10        let engine = self.engine.clone();
11        tokio::spawn(async move {
12            let shutdown = context.shutdown();
13            let startup = cookie_agent_protocol::RuntimeChangedNotification {
14                previous_revision: None,
15                snapshot: startup,
16                reasons: vec![cookie_agent_protocol::RuntimeChangeReason::Startup],
17            };
18            if context
19                .notify(cookie_agent_protocol::RUNTIME_CHANGED_METHOD, &startup)
20                .await
21                .is_err()
22            {
23                return;
24            }
25            loop {
26                let changed = tokio::select! {
27                    _ = shutdown.cancelled() => return,
28                    changed = receiver.recv() => match changed {
29                        Ok(changed) => changed,
30                        // The skipped changes broke this connection's revision
31                        // chain: hand it the current runtime to start over from.
32                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => {
33                            cookie_agent_protocol::RuntimeChangedNotification {
34                                previous_revision: None,
35                                snapshot: engine.current_runtime().result.snapshot.clone(),
36                                reasons: vec![cookie_agent_protocol::RuntimeChangeReason::Resync],
37                            }
38                        }
39                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
40                    },
41                };
42                if context
43                    .notify(cookie_agent_protocol::RUNTIME_CHANGED_METHOD, &changed)
44                    .await
45                    .is_err()
46                {
47                    return;
48                }
49            }
50        });
51    }

82    /// Signals MCP server changes to this connection; bursts coalesce.
83    /// A connection failure is sent on its own, naming the subscribed
84    /// sessions that use the server.
85    pub(super) fn start_mcp_change_notifications(&self, context: ServerContext) {
86        let mut receiver = self.engine.subscribe_mcp_changes();
87        receiver.mark_unchanged();
88        let mut failures = self.engine.subscribe_mcp_failures();
89        let engine = self.engine.clone();
90        tokio::spawn(async move {
91            let shutdown = context.shutdown();
92            loop {
93                let notification = tokio::select! {
94                    _ = shutdown.cancelled() => return,
95                    changed = receiver.changed() => {
96                        if changed.is_err() {
97                            return;
98                        }
99                        cookie_agent_protocol::McpChangedNotification::default()
100                    }
101                    failure = failures.recv() => match failure {
102                        Ok(failure) => cookie_agent_protocol::McpChangedNotification {
103                            failures: mcp_failure(&engine, &context.subscribed_sessions(), failure)
104                                .into_iter()
105                                .collect(),
106                        },
107                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => continue,
108                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
109                    },
110                };
111                if context
112                    .notify(cookie_agent_protocol::MCP_CHANGED_METHOD, &notification)
113                    .await
114                    .is_err()
115                {
116                    return;
117                }
118            }
119        });
120    }

169    /// Tells a connection that just subscribed to `session` about the
170    /// failures that already affect it: failures reported before the
171    /// subscription named no session of this connection.
172    pub(super) fn replay_connection_failures(&self, context: &ServerContext, session: SessionId) {
173        let sessions = [session];
174        let mcp = self
175            .engine
176            .mcp_failures()
177            .into_iter()
178            .filter_map(|failure| mcp_failure(&self.engine, &sessions, failure))
179            .collect::<Vec<_>>();
180        let runtimes = self
181            .engine
182            .runtime_instances()
183            .into_iter()
184            .filter(|instance| {
185                instance.status == cookie_agent_engine::RuntimeInstanceStatus::Failed
186            })
187            .filter_map(|instance| {
188                runtime_failure(
189                    &self.engine,
190                    &sessions,
191                    cookie_agent_engine::RuntimeFailure {
192                        runtime_id: instance.runtime_id,
193                        workspace: instance.workspace,
194                        reason: instance.message.unwrap_or_default(),
195                    },
196                )
197            })
198            .collect::<Vec<_>>();
199        if mcp.is_empty() && runtimes.is_empty() {
200            return;
201        }
202        let context = context.clone();
203        tokio::spawn(async move {
204            if !mcp.is_empty() {
205                let notification = cookie_agent_protocol::McpChangedNotification { failures: mcp };
206                if context
207                    .notify(cookie_agent_protocol::MCP_CHANGED_METHOD, &notification)
208                    .await
209                    .is_err()
210                {
211                    return;
212                }
213            }
214            if !runtimes.is_empty() {
215                let notification =
216                    cookie_agent_protocol::RuntimesChangedNotification { failures: runtimes };
217                let _ = context
218                    .notify(
219                        cookie_agent_protocol::RUNTIMES_CHANGED_METHOD,
220                        &notification,
221                    )
222                    .await;
223            }
224        });
225    }

227    pub(crate) fn start_engine_event_notifications(&self, context: ServerContext) {
228        let mut receiver = self.engine.subscribe_engine_events();
229        tokio::spawn(async move {
230            let shutdown = context.shutdown();
231            loop {
232                let event = tokio::select! {
233                    _ = shutdown.cancelled() => return,
234                    event = receiver.recv() => match event {
235                        Ok(event) => event,
236                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => continue,
237                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
238                    },
239                };
240                let cookie_agent_engine::EngineEvent::PluginEvent {
241                    session_id,
242                    plugin,
243                    name,
244                    payload,
245                } = event;
246                if !context.is_session_subscribed(session_id) {
247                    continue;
248                }
249                let params = cookie_agent_protocol::ExtensionBusEventParams {
250                    session_id,
251                    context_id: None,
252                    plugin,
253                    name,
254                    payload,
255                };
256                if context.notify("events.plugin", &params).await.is_err() {
257                    return;
258                }
259            }
260        });
261    }

264/// The sessions among `sessions` attached to a runtime that `matches`.
265fn affected_sessions(
266    engine: &Engine,
267    sessions: &[SessionId],
268    matches: impl Fn(&cookie_agent_protocol::RuntimeAttachment) -> bool,
269) -> Vec<SessionId> {
270    let mut affected = sessions
271        .iter()
272        .copied()
273        .filter(|session| {
274            engine
275                .session_runtimes(*session)
276                .is_ok_and(|attachments| attachments.iter().any(&matches))
277        })
278        .collect::<Vec<_>>();
279    affected.sort_unstable();
280    affected
281}

283/// `failure` as a client sees it, when one of `sessions` uses the server:
284/// it is attached to the instance that hosts it. Top-level MCP servers
285/// belong to the global runtime.
286fn mcp_failure(
287    engine: &Engine,
288    sessions: &[SessionId],
289    failure: cookie_agent_engine::McpServerFailure,
290) -> Option<cookie_agent_protocol::McpServerFailure> {
291    let runtime_id = RuntimeId::new(failure.runtime_id.as_str()).ok()?;
292    let session_ids = affected_sessions(engine, sessions, |attachment| {
293        attachment.runtime_id == runtime_id && attachment.workspace == failure.workspace
294    });
295    (!session_ids.is_empty()).then(|| cookie_agent_protocol::McpServerFailure {
296        runtime_id: runtime_id.clone(),
297        workspace: failure.workspace,
298        server: failure.server,
299        reason: failure.reason,
300        session_ids,
301    })
302}
