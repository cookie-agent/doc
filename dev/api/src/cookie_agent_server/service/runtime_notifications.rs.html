6    pub(super) fn start_runtime_notifications(&self, context: ServerContext) {
7        let mut receiver = self.engine.subscribe_runtime_changes();
8        let startup = self.engine.current_runtime().result.snapshot.clone();
9        let engine = self.engine.clone();
10        tokio::spawn(async move {
11            let shutdown = context.shutdown();
12            let startup = cookie_agent_protocol::RuntimeChangedNotification {
13                previous_revision: None,
14                snapshot: startup,
15                reasons: vec![cookie_agent_protocol::RuntimeChangeReason::Startup],
16            };
17            if context
18                .notify(cookie_agent_protocol::RUNTIME_CHANGED_METHOD, &startup)
19                .await
20                .is_err()
21            {
22                return;
23            }
24            loop {
25                let changed = tokio::select! {
26                    _ = shutdown.cancelled() => return,
27                    changed = receiver.recv() => match changed {
28                        Ok(changed) => changed,
29                        // The skipped changes broke this connection's revision
30                        // chain: hand it the current runtime to start over from.
31                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => {
32                            cookie_agent_protocol::RuntimeChangedNotification {
33                                previous_revision: None,
34                                snapshot: engine.current_runtime().result.snapshot.clone(),
35                                reasons: vec![cookie_agent_protocol::RuntimeChangeReason::Resync],
36                            }
37                        }
38                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
39                    },
40                };
41                if context
42                    .notify(cookie_agent_protocol::RUNTIME_CHANGED_METHOD, &changed)
43                    .await
44                    .is_err()
45                {
46                    return;
47                }
48            }
49        });
50    }

52    /// Forwards every session index change to this connection. A connection
53    /// too slow to keep up is told to resync instead of silently missing
54    /// changes.
55    pub(super) fn start_session_change_notifications(&self, context: ServerContext) {
56        let mut receiver = self.engine.subscribe_session_changes();
57        tokio::spawn(async move {
58            let shutdown = context.shutdown();
59            loop {
60                let change = tokio::select! {
61                    _ = shutdown.cancelled() => return,
62                    change = receiver.recv() => match change {
63                        Ok(change) => change,
64                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => {
65                            cookie_agent_protocol::SessionsChangedNotification::Resync
66                        }
67                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
68                    },
69                };
70                if context
71                    .notify(cookie_agent_protocol::SESSIONS_CHANGED_METHOD, &change)
72                    .await
73                    .is_err()
74                {
75                    return;
76                }
77            }
78        });
79    }

81    /// Signals MCP server changes to this connection; bursts coalesce.
82    pub(super) fn start_mcp_change_notifications(&self, context: ServerContext) {
83        let mut receiver = self.engine.subscribe_mcp_changes();
84        receiver.mark_unchanged();
85        tokio::spawn(async move {
86            let shutdown = context.shutdown();
87            loop {
88                tokio::select! {
89                    _ = shutdown.cancelled() => return,
90                    changed = receiver.changed() => if changed.is_err() {
91                        return;
92                    },
93                }
94                if context
95                    .notify(
96                        cookie_agent_protocol::MCP_CHANGED_METHOD,
97                        &cookie_agent_protocol::McpChangedNotification {},
98                    )
99                    .await
100                    .is_err()
101                {
102                    return;
103                }
104            }
105        });
106    }

108    pub(crate) fn start_engine_event_notifications(&self, context: ServerContext) {
109        let mut receiver = self.engine.subscribe_engine_events();
110        tokio::spawn(async move {
111            let shutdown = context.shutdown();
112            loop {
113                let event = tokio::select! {
114                    _ = shutdown.cancelled() => return,
115                    event = receiver.recv() => match event {
116                        Ok(event) => event,
117                        Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => continue,
118                        Err(tokio::sync::broadcast::error::RecvError::Closed) => return,
119                    },
120                };
121                let cookie_agent_engine::EngineEvent::PluginEvent {
122                    session_id,
123                    plugin,
124                    name,
125                    payload,
126                } = event;
127                if !context.is_session_subscribed(session_id) {
128                    continue;
129                }
130                let params = cookie_agent_protocol::ExtensionBusEventParams {
131                    session_id,
132                    context_id: None,
133                    plugin,
134                    name,
135                    payload,
136                };
137                if context.notify("events.plugin", &params).await.is_err() {
138                    return;
139                }
140            }
141        });
142    }
