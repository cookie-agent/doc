1//! Share links on the daemon's listeners (`docs/site/specs/share-links.md`
2//! section 5): `/s/<id>` is a link's landing page, `/s/<id>/raw/<name>` its
3//! asset's bytes inline, and `/s/<id>/download/<name>` the same as a
4//! download. No token, ticket, or origin is checked: the ID is the only
5//! key. Every failure answers the same 404 page, so nobody learns whether
6//! a link ever existed.
7//!
8//! A conversation share's `/s/<id>` is the guest page
9//! (`docs/site/specs/conversation-shares.md` section 6): the web client's own
10//! page, whose files are at `/s/_app/`, in a sandbox that gives it an opaque
11//! origin, so it cannot touch what the daemon's page keeps on this origin.
12

24const PREFIX: &str = "/s/";
25/// The landing page's font, below [`PREFIX`]: no ID is this short.
26pub(super) const FONT_PATH: &str = "_font/inter.woff2";
27/// Where the web bundle keeps the font.
28const BUNDLED_FONT: &str = "fonts/InterVariable.woff2";

30/// The policy of shared HTML: the viewer's, whose `sandbox` gives the page
31/// an opaque origin even when opened at the top level.
32const HTML_POLICY: &str = cookie_agent_web_assets::VIEWER_POLICY;
33/// The policy of shared SVG: sandboxed, and no scripts.
34const SVG_POLICY: &str = "sandbox; default-src 'none'; style-src 'unsafe-inline'; \
35     img-src data:; font-src data:; frame-ancestors 'self'";
36/// The policy of every other inline type but PDF, which the browser's
37/// viewer cannot show sandboxed.
38const PLAIN_POLICY: &str = "sandbox; default-src 'none'; img-src 'self'; media-src 'self'; \
39     style-src 'unsafe-inline'; frame-ancestors 'self'";
40const PDF_POLICY: &str = "frame-ancestors 'self'";

42/// The directory below `/s/` the guest page's files are in: no ID is this
43/// short.
44const APP_PREFIX: &str = "_app/";
45/// The sandbox of the guest page. Links it opens are ordinary tabs.
46const GUEST_SANDBOX: &str = "sandbox allow-scripts allow-forms allow-modals allow-downloads \
47     allow-popups allow-popups-to-escape-sandbox";

49/// What a path below `/s/` asks for.
50#[derive(Debug, Eq, PartialEq)]
51enum Wanted {
52    Landing(ShareId),
53    /// `/s/<id>/`: the landing page without its trailing slash.
54    Slash(ShareId),
55    Raw(ShareId),
56    Download(ShareId),
57    /// A guest's content, below `f/` (`docs/site/specs/folder-artifacts.md`
58    /// section 6.5).
59    Content(ShareId),
60    /// A guest's web forward, below `p/` (`docs/site/specs/forwards.md`
61    /// section 6.1).
62    GuestForward(ShareId),
63    /// Any other path of a link, which only a link to a web forward serves.
64    Other(ShareId),
65    Font,
66    /// A file of the guest page.
67    App(String),
68    /// A guest's connection.
69    Socket(ShareId),
70}
71
72fn wanted(path: &str) -> Option<Wanted> {
73    let below = path.strip_prefix(PREFIX)?;
74    if below == FONT_PATH {
75        return Some(Wanted::Font);
76    }
77    if let Some(file) = below.strip_prefix(APP_PREFIX) {
78        return Some(Wanted::App(file.to_owned()));
79    }
80    let (id, rest) = match below.find('/') {
81        Some(slash) => (&below[..slash], &below[slash..]),
82        None => (below, ""),
83    };
84    let id = ShareId::new(id).ok()?;
85    match rest {
86        "" => Some(Wanted::Landing(id)),
87        "/" => Some(Wanted::Slash(id)),
88        rest => {
89            let (kind, _name) = match rest[1..].find('/') {
90                Some(slash) => (&rest[1..=slash], &rest[slash + 2..]),
91                None => (&rest[1..], ""),
92            };
93            match kind {
94                "raw" => Some(Wanted::Raw(id)),
95                "download" => Some(Wanted::Download(id)),
96                "f" => Some(Wanted::Content(id)),
97                "p" => Some(Wanted::GuestForward(id)),
98                "ws" if rest == "/ws" => Some(Wanted::Socket(id)),
99                _ => Some(Wanted::Other(id)),
100            }
101        }
102    }
103}

105/// Serves a path below `/s/`.
106pub(crate) async fn share_pages(State(state): State<WebSocketState>, request: Request) -> Response {
107    let method = request.method().clone();
108    // A web forward's app takes every method; everything else reads.
109    let reads = method == Method::GET || method == Method::HEAD;
110    let Some(wanted) = wanted(request.uri().path()) else {
111        return gone();
112    };
113    if matches!(wanted, Wanted::Font | Wanted::App(_)) && !reads {
114        return gone();
115    }
116    if wanted == Wanted::Font {
117        return font();
118    }
119    if let Wanted::App(file) = &wanted {
120        return app_file(file, request.headers());
121    }
122    if let Wanted::Socket(id) = wanted {
123        return super::guest::guest_upgrade(state, id, request).await;
124    }
125    let Some(server) = state.server.get() else {
126        return gone();
127    };
128    let engine = &server.engine;
129    let id = match &wanted {
130        Wanted::Landing(id)
131        | Wanted::Slash(id)
132        | Wanted::Raw(id)
133        | Wanted::Download(id)
134        | Wanted::Content(id)
135        | Wanted::GuestForward(id)
136        | Wanted::Other(id) => id.clone(),
137        Wanted::Font | Wanted::App(_) | Wanted::Socket(_) => unreachable!("answered above"),
138    };
139    let open = match engine.open_share(id.clone()).await {
140        Ok(Some(OpenedLink::Asset(open))) => *open,
141        Ok(Some(OpenedLink::Forward(handle))) => {
142            return forward_link(engine, &id, &handle, request).await;
143        }
144        Ok(Some(OpenedLink::Session(link))) if !reads => {
145            return match wanted {
146                Wanted::GuestForward(_) => {
147                    let below = after(&request, &id, "p").to_owned();
148                    let outside = share_outside(engine, &request);
149                    let base = format!("{}/s/{id}/p", outside_root(&outside));
150                    super::forward_pages::serve(
151                        engine,
152                        &below,
153                        &base,
154                        &outside,
155                        Some(link.session),
156                        request,
157                    )
158                    .await
159                }
160                _ => gone(),
161            };
162        }
163        _ if !reads => return gone(),
164        Ok(Some(OpenedLink::Session(link))) => {
165            return match wanted {
166                Wanted::Slash(id) => redirect(&format!("../{id}")),
167                Wanted::Landing(_) => guest_page(engine, &link),
168                Wanted::GuestForward(_) => {
169                    let below = after(&request, &id, "p").to_owned();
170                    let outside = share_outside(engine, &request);
171                    let base = format!("{}/s/{id}/p", outside_root(&outside));
172                    super::forward_pages::serve(
173                        engine,
174                        &below,
175                        &base,
176                        &outside,
177                        Some(link.session),
178                        request,
179                    )
180                    .await
181                }
182                Wanted::Content(_) => {
183                    let below = after(&request, &id, "f").to_owned();
184                    guest_content(
185                        engine,
186                        &link,
187                        request.uri().query(),
188                        request.headers(),
189                        &below,
190                    )
191                    .await
192                }
193                _ => gone(),
194            };
195        }
196        _ => return gone(),
197    };
198    // A folder's files are below `raw/`, and its zip is its download.
199    if let Some(folder) = open.asset.folder.clone()
200        && matches!(wanted, Wanted::Raw(_) | Wanted::Download(_))
201    {
202        let download = matches!(wanted, Wanted::Download(_));
203        let rest = if download {
204            String::new()
205        } else {
206            match super::content_pages::decoded(after(&request, &id, "raw")) {
207                Some(rest) if !rest.is_empty() => rest,
208                Some(_) => folder.main.clone(),
209                None => return gone(),
210            }
211        };
212        let store = cookie_agent_engine::ContentStore::Assets(open.asset.store.clone());
213        let wanted = super::content_pages::Wanted {
214            store: store.label(),
215            sha256: open.asset.sha256.clone(),
216            key: String::new(),
217            zip: download,
218            rest,
219        };
220        let name = open.share.title.replace(['/', '\\'], "_");
221        return super::content_pages::serve(
222            engine,
223            store,
224            &wanted,
225            false,
226            Some(name),
227            request.headers(),
228            true,
229        )
230        .await;
231    }
232    match wanted {
233        Wanted::Slash(id) => redirect(&format!("../{id}")),
234        Wanted::Landing(_) => {
235            let kind = share_landing::Kind::of(&open.asset.media_type);
236            let content = if kind.reads_content() {
237                match engine.read_shared_asset(&open.asset).await {
238                    Ok(Some(bytes)) => Some(bytes),
239                    _ => return gone(),
240                }
241            } else {
242                None
243            };
244            if method == Method::GET {
245                engine.count_share_view(&id);
246            }
247            let manifest = match open.asset.folder {
248                Some(_) => match engine
249                    .content_entry(
250                        cookie_agent_engine::ContentStore::Assets(open.asset.store.clone()),
251                        open.asset.sha256.clone(),
252                    )
253                    .await
254                {
255                    Ok(Some(cookie_agent_engine::ContentEntry::Folder(manifest))) => Some(manifest),
256                    _ => return gone(),
257                },
258                None => None,
259            };
260            let page = share_landing::page_of(
261                &open,
262                kind,
263                content.as_deref(),
264                manifest.as_ref(),
265                jiff::Timestamp::now(),
266            );
267            let mut headers = common_headers();
268            headers.insert(
269                header::CONTENT_TYPE,
270                HeaderValue::from_static("text/html; charset=utf-8"),
271            );
272            headers.insert(
273                header::CONTENT_SECURITY_POLICY,
274                HeaderValue::from_static(share_landing::POLICY),
275            );
276            (StatusCode::OK, headers, page).into_response()
277        }
278        Wanted::Raw(_) | Wanted::Download(_) => {
279            let total = open.asset.byte_length;
280            let part = wanted_part(request.headers(), total);
281            let (offset, length) = part.extent(total);
282            let Ok(Some(bytes)) = engine
283                .read_shared_asset_part(&open.asset, offset, length)
284                .await
285            else {
286                return gone();
287            };
288            let download = matches!(wanted, Wanted::Download(_))
289                || !share_landing::Kind::of(&open.asset.media_type).inline();
290            bytes_response(
291                &open.asset.media_type,
292                &share_landing::file_name(&open.asset),
293                download,
294                part,
295                total,
296                bytes,
297            )
298        }
299        Wanted::Content(_) | Wanted::GuestForward(_) | Wanted::Other(_) => gone(),
300        Wanted::Font | Wanted::App(_) | Wanted::Socket(_) => unreachable!("answered above"),
301    }
302}

319/// A link to a web forward: its app at the link's root,
320/// `/s/<id>/<path>`, for anyone with the link while it and the forward
321/// live.
322async fn forward_link(engine: &Engine, id: &ShareId, handle: &str, request: Request) -> Response {
323    let Some(forward) = engine.web_forward(handle) else {
324        return gone();
325    };
326    let path = request.uri().path().to_owned();
327    let rest = path
328        .strip_prefix(PREFIX)
329        .and_then(|below| below.strip_prefix(id.as_str()))
330        .unwrap_or_default()
331        .to_owned();
332    if rest.is_empty() {
333        return redirect(&format!("{id}/"));
334    }
335    if rest == "/" && request.method() == Method::GET {
336        engine.count_share_view(id);
337    }
338    let outside = share_outside(engine, &request);
339    let prefix = format!("{}/s/{id}", outside_root(&outside));
340    super::forward_pages::serve_at(engine, &forward, &rest, &prefix, &outside, request).await
341}

343/// The path after `/s/<id>/<kind>/` of `request`, as it was sent.
344fn after<'a>(request: &'a Request, id: &ShareId, kind: &str) -> &'a str {
345    let prefix = format!("{PREFIX}{id}/{kind}/");
346    request
347        .uri()
348        .path()
349        .strip_prefix(prefix.as_str())
350        .unwrap_or_default()
351}

353/// A guest's content: a file or a folder of the shared session's tree, or
354/// of an asset store its transcript can show (`docs/site/specs/
355/// folder-artifacts.md` section 6.5). The share is the key, so it ends with
356/// the share.
357async fn guest_content(
358    engine: &Engine,
359    link: &cookie_agent_engine::GuestLink,
360    query: Option<&str>,
361    headers: &HeaderMap,
362    below: &str,
363) -> Response {
364    use cookie_agent_engine::ContentStore;
365    use cookie_agent_protocol::AssetStore;
366    let Some(wanted) = super::content_pages::wanted_shared(below) else {
367        return gone();
368    };
369    let Some(store) = ContentStore::parse(&wanted.store) else {
370        return gone();
371    };
372    let shows = match &store {
373        ContentStore::Artifacts(root) => *root == link.session,
374        ContentStore::Assets(AssetStore::Session { session }) => *session == link.session,
375        // Named only by content nobody can guess, as `asset.read` names
376        // them by IDs nobody can guess.
377        ContentStore::Assets(AssetStore::Project { .. } | AssetStore::Shared) => true,
378    };
379    if !shows {
380        return gone();
381    }
382    let download = query.is_some_and(|query| query.split('&').any(|pair| pair == "download"));
383    let name = query.and_then(|query| {
384        query
385            .split('&')
386            .find_map(|pair| pair.strip_prefix("name="))
387            .and_then(super::content_pages::decoded)
388    });
389    super::content_pages::serve(engine, store, &wanted, download, name, headers, true).await
390}

392/// A file of the guest page. Its page has an opaque origin, so every file it
393/// loads is another origin's and answers to any.
394fn app_file(file: &str, headers: &HeaderMap) -> Response {
395    let policy = guest_file_policy(file);
396    // `index.html` is the page itself, which is only served in its sandbox.
397    let served = (file != "index.html")
398        .then(|| super::page::file(file, headers, &policy))
399        .flatten();
400    let Some(mut response) = served else {
401        return gone();
402    };
403    response.headers_mut().insert(
404        header::ACCESS_CONTROL_ALLOW_ORIGIN,
405        HeaderValue::from_static("*"),
406    );
407    response
408}

410/// The policy a file of the guest page is served with: the page's, or a
411/// viewer's own (`viewer.html` for HTML assets, `pdf-viewer.html` for
412/// PDFs) without its `frame-ancestors`. The guest page frames the viewers,
413/// and its origin is opaque, which no source in `frame-ancestors` matches,
414/// `*` included: with one, every page and PDF a guest opens is refused. A
415/// viewer is sandboxed and holds nothing until its page hands it over.
416fn guest_file_policy(file: &str) -> String {
417    let policy = cookie_agent_web_assets::policy_of(file);
418    if file != cookie_agent_web_assets::VIEWER_PATH
419        && file != cookie_agent_web_assets::PDF_VIEWER_PATH
420    {
421        return policy.to_owned();
422    }
423    policy
424        .split(';')
425        .map(str::trim)
426        .filter(|directive| !directive.starts_with("frame-ancestors"))
427        .collect::<Vec<_>>()
428        .join("; ")
429}

431/// The guest page: the web client's page, reading its files from
432/// [`APP_PREFIX`], in its sandbox. The first thing a link preview sees of it
433/// is the session's title.
434fn guest_page(engine: &Engine, link: &GuestLink) -> Response {
435    let Some(index) = cookie_agent_web_assets::get("index.html") else {
436        return gone();
437    };
438    let title = engine
439        .session_meta(link.session)
440        .ok()
441        .and_then(|meta| meta.title)
442        .map_or_else(
443            || "A shared conversation".to_owned(),
444            |title| title.as_str().to_owned(),
445        );
446    let page = guest_page_html(&String::from_utf8_lossy(index.bytes), &title);
447    let mut headers = common_headers();
448    headers.insert(
449        header::CONTENT_TYPE,
450        HeaderValue::from_static("text/html; charset=utf-8"),
451    );
452    if let Ok(policy) = HeaderValue::from_str(&guest_policy()) {
453        headers.insert(header::CONTENT_SECURITY_POLICY, policy);
454    }
455    headers.insert(
456        header::ACCESS_CONTROL_ALLOW_ORIGIN,
457        HeaderValue::from_static("*"),
458    );
459    (StatusCode::OK, headers, page).into_response()
460}

462/// The page's own policy with the sandbox, and a `<base>` for its files.
463fn guest_policy() -> String {
464    let policy = cookie_agent_web_assets::content_security_policy()
465        .replace("base-uri 'none'", "base-uri 'self'");
466    format!("{GUEST_SANDBOX}; {policy}")
467}

469/// `index` as the guest page: its relative URLs read below [`APP_PREFIX`],
470/// its stylesheet loads across origins as its integrity check needs, and it
471/// carries `title` for link previews.
472fn guest_page_html(index: &str, title: &str) -> String {
473    let title = share_landing::escape(title);
474    let head = format!(
475        "<head><base href=\"{APP_PREFIX}\" />\
476         <meta property=\"og:title\" content=\"{title}\" />\
477         <meta property=\"og:type\" content=\"website\" />\
478         <meta property=\"og:description\" content=\"A conversation shared from cookie agent\" />"
479    );
480    index
481        .replacen("<head>", &head, 1)
482        .replace(
483            "<link rel=\"stylesheet\"",
484            "<link crossorigin=\"anonymous\" rel=\"stylesheet\"",
485        )
486        .replacen(
487            "<title>Cookie Agent</title>",
488            &format!("<title>{title}</title>"),
489            1,
490        )
491}

498/// The headers every answer carries.
499fn common_headers() -> HeaderMap {
500    let mut headers = HeaderMap::new();
501    headers.insert(header::CACHE_CONTROL, HeaderValue::from_static("no-store"));
502    headers.insert(
503        header::REFERRER_POLICY,
504        HeaderValue::from_static("no-referrer"),
505    );
506    headers.insert(
507        header::HeaderName::from_static("x-robots-tag"),
508        HeaderValue::from_static("noindex, nofollow"),
509    );
510    headers.insert(
511        header::X_CONTENT_TYPE_OPTIONS,
512        HeaderValue::from_static("nosniff"),
513    );
514    headers
515}

517/// The one answer to every link that does not work. Any page may read it: a
518/// guest page, whose origin is opaque, asks after its own link.
519fn gone() -> Response {
520    let mut headers = common_headers();
521    headers.insert(
522        header::ACCESS_CONTROL_ALLOW_ORIGIN,
523        HeaderValue::from_static("*"),
524    );
525    headers.insert(
526        header::CONTENT_TYPE,
527        HeaderValue::from_static("text/html; charset=utf-8"),
528    );
529    headers.insert(
530        header::CONTENT_SECURITY_POLICY,
531        HeaderValue::from_static(share_landing::POLICY),
532    );
533    (StatusCode::NOT_FOUND, headers, share_landing::GONE_PAGE).into_response()
534}

536fn redirect(location: &str) -> Response {
537    let mut headers = common_headers();
538    match HeaderValue::from_str(location) {
539        Ok(location) => {
540            headers.insert(header::LOCATION, location);
541            (StatusCode::PERMANENT_REDIRECT, headers).into_response()
542        }
543        Err(_) => gone(),
544    }
545}

547fn font() -> Response {
548    let Some(font) = cookie_agent_web_assets::get(BUNDLED_FONT) else {
549        return StatusCode::NOT_FOUND.into_response();
550    };
551    let mut headers = HeaderMap::new();
552    headers.insert(header::CONTENT_TYPE, HeaderValue::from_static("font/woff2"));
553    headers.insert(
554        header::CACHE_CONTROL,
555        HeaderValue::from_static("public, max-age=86400"),
556    );
557    headers.insert(
558        header::X_CONTENT_TYPE_OPTIONS,
559        HeaderValue::from_static("nosniff"),
560    );
561    (StatusCode::OK, headers, font.bytes).into_response()
562}

564/// The policy a shared asset's bytes are served with inline.
565fn inline_policy(media_type: &str) -> &'static str {
566    match media_type {
567        "text/html" => HTML_POLICY,
568        "image/svg+xml" => SVG_POLICY,
569        "application/pdf" => PDF_POLICY,
570        _ => PLAIN_POLICY,

574/// The `Content-Type` a shared asset's bytes are served with: text as
575/// UTF-8 plain text, so a browser never runs what it is not meant to.
576fn served_type(media_type: &str) -> String {
577    match media_type {
578        "text/html" => "text/html; charset=utf-8".to_owned(),
579        "image/svg+xml" => media_type.to_owned(),
580        _ if share_landing::Kind::of(media_type) == share_landing::Kind::Text
581            || share_landing::Kind::of(media_type) == share_landing::Kind::Markdown =>
582        {
583            "text/plain; charset=utf-8".to_owned()
584        }
585        _ => media_type.to_owned(),
586    }
587}

589/// `Content-Disposition` for `name`, with an ASCII fallback and the UTF-8
590/// name (RFC 6266).
591pub(super) fn disposition(kind: &str, name: &str) -> String {
592    let ascii = name
593        .chars()
594        .map(|character| {
595            if character.is_ascii_graphic() && character != '"' && character != '\\' {
596                character
597            } else if character == ' ' {
598                ' '
599            } else {
600                '_'
601            }
602        })
603        .collect::<String>();
604    format!(
605        "{kind}; filename=\"{ascii}\"; filename*=UTF-8''{}",
606        share_landing::percent_encode(name)
607    )
608}

629/// [`bytes_response`] served as `content_type` when inline, its policy
630/// chosen by `media_type`.
631pub(super) fn bytes_response_typed(
632    content_type: &str,
633    media_type: &str,
634    name: &str,
635    download: bool,
636    part: Part,
637    total: u64,
638    bytes: Vec<u8>,
639) -> Response {
640    let mut headers = common_headers();
641    let content_type = if download {
642        "application/octet-stream".to_owned()
643    } else {
644        content_type.to_owned()
645    };
646    if let Ok(value) = HeaderValue::from_str(&content_type) {
647        headers.insert(header::CONTENT_TYPE, value);
648    }
649    if let Ok(value) = HeaderValue::from_str(&disposition(
650        if download { "attachment" } else { "inline" },
651        name,
652    )) {
653        headers.insert(header::CONTENT_DISPOSITION, value);
654    }
655    if !download {
656        headers.insert(
657            header::CONTENT_SECURITY_POLICY,
658            HeaderValue::from_static(inline_policy(media_type)),
659        );
660    }
661    headers.insert(header::ACCEPT_RANGES, HeaderValue::from_static("bytes"));
662    match part {
663        Part::Whole => (StatusCode::OK, headers, Body::from(bytes)).into_response(),
664        Part::Range(start, end) => {
665            if let Ok(value) = HeaderValue::from_str(&format!("bytes {start}-{end}/{total}")) {
666                headers.insert(header::CONTENT_RANGE, value);
667            }
668            (StatusCode::PARTIAL_CONTENT, headers, Body::from(bytes)).into_response()
669        }
670        Part::Unsatisfiable => {
671            if let Ok(value) = HeaderValue::from_str(&format!("bytes */{total}")) {
672                headers.insert(header::CONTENT_RANGE, value);
673            }
674            (StatusCode::RANGE_NOT_SATISFIABLE, headers).into_response()
675        }
676    }
677}

679/// The most bytes one range answer reads. A player asks for `bytes=0-` and
680/// seeks away; it asks again from where the answer ended, as `Content-Range`
681/// tells it.
682const MAX_RANGE_BYTES: u64 = 8 * 1024 * 1024;
683
684/// What of an asset a request wants.
685#[derive(Clone, Copy, Debug, Eq, PartialEq)]
686pub(super) enum Part {
687    Whole,
688    /// Inclusive of both ends.
689    Range(u64, u64),
690    Unsatisfiable,
691}
692
693impl Part {
694    /// Where to read this part of `total` bytes, and how much.
695    pub(super) fn extent(self, total: u64) -> (u64, u64) {
696        match self {
697            Self::Whole => (0, total),
698            Self::Range(start, end) => (start, end - start + 1),
699            Self::Unsatisfiable => (0, 0),
700        }
701    }
702}
703
704pub(super) fn wanted_part(request: &HeaderMap, total: u64) -> Part {
705    request
706        .get(header::RANGE)
707        .and_then(|value| value.to_str().ok())
708        .map_or(Part::Whole, |value| match byte_range(value, total) {
709            Some((start, end)) => {
710                Part::Range(start, end.min(start.saturating_add(MAX_RANGE_BYTES - 1)))
711            }
712            None => Part::Unsatisfiable,
713        })
714}
715
716/// The one range `bytes=a-b`, `bytes=a-`, or `bytes=-n` asks for of
717/// `total` bytes, inclusive; `None` when it cannot be satisfied. Several
718/// ranges are answered with the first.
719fn byte_range(value: &str, total: u64) -> Option<(u64, u64)> {
720    let spec = value.trim().strip_prefix("bytes=")?;
721    let first = spec.split(',').next()?.trim();
722    let (start, end) = first.split_once('-')?;
723    if total == 0 {
724        return None;
725    }
726    let last = total - 1;
727    let (start, end) = match (start.trim(), end.trim()) {
728        ("", suffix) => {
729            let length = suffix.parse::<u64>().ok().filter(|length| *length > 0)?;
730            (total.saturating_sub(length), last)
731        }
732        (start, "") => (start.parse().ok()?, last),
733        (start, end) => (start.parse().ok()?, end.parse::<u64>().ok()?.min(last)),
734    };
735    (start <= end && start <= last).then_some((start, end))
736}
