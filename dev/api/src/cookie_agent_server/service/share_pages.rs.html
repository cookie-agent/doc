1//! Share links on the daemon's listeners (`docs/site/specs/share-links.md`
2//! section 5): `/s/<id>` is a link's landing page, `/s/<id>/raw/<name>` its
3//! asset's bytes inline, and `/s/<id>/download/<name>` the same as a
4//! download. No token, ticket, or origin is checked: the ID is the only
5//! key. Every failure answers the same 404 page, so nobody learns whether
6//! a link ever existed.
7

18const PREFIX: &str = "/s/";
19/// The landing page's font, below [`PREFIX`]: no ID is this short.
20pub(super) const FONT_PATH: &str = "_font/inter.woff2";
21/// Where the web bundle keeps the font.
22const BUNDLED_FONT: &str = "fonts/InterVariable.woff2";
23
24/// The policy of shared HTML: the viewer's, whose `sandbox` gives the page
25/// an opaque origin even when opened at the top level.
26const HTML_POLICY: &str = cookie_agent_web_assets::VIEWER_POLICY;
27/// The policy of shared SVG: sandboxed, and no scripts.
28const SVG_POLICY: &str = "sandbox; default-src 'none'; style-src 'unsafe-inline'; \
29     img-src data:; font-src data:; frame-ancestors 'self'";
30/// The policy of every other inline type but PDF, which the browser's
31/// viewer cannot show sandboxed.
32const PLAIN_POLICY: &str = "sandbox; default-src 'none'; img-src 'self'; media-src 'self'; \
33     style-src 'unsafe-inline'; frame-ancestors 'self'";
34const PDF_POLICY: &str = "frame-ancestors 'self'";

36/// What a path below `/s/` asks for.
37#[derive(Debug, Eq, PartialEq)]
38enum Wanted {
39    Landing(ShareId),
40    /// `/s/<id>/`: the landing page without its trailing slash.
41    Slash(ShareId),
42    Raw(ShareId),
43    Download(ShareId),
44    Font,
45}
46
47fn wanted(path: &str) -> Option<Wanted> {
48    let below = path.strip_prefix(PREFIX)?;
49    if below == FONT_PATH {
50        return Some(Wanted::Font);
51    }
52    let (id, rest) = match below.find('/') {
53        Some(slash) => (&below[..slash], &below[slash..]),
54        None => (below, ""),
55    };
56    let id = ShareId::new(id).ok()?;
57    match rest {
58        "" => Some(Wanted::Landing(id)),
59        "/" => Some(Wanted::Slash(id)),
60        rest => {
61            let (kind, _name) = match rest[1..].find('/') {
62                Some(slash) => (&rest[1..=slash], &rest[slash + 2..]),
63                None => (&rest[1..], ""),
64            };
65            match kind {
66                "raw" => Some(Wanted::Raw(id)),
67                "download" => Some(Wanted::Download(id)),
68                _ => None,
69            }
70        }
71    }
72}

74/// Serves a path below `/s/`.
75pub(crate) async fn share_pages(State(state): State<WebSocketState>, request: Request) -> Response {
76    let method = request.method().clone();
77    if method != Method::GET && method != Method::HEAD {
78        return gone();
79    }
80    let Some(wanted) = wanted(request.uri().path()) else {
81        return gone();
82    };
83    if wanted == Wanted::Font {
84        return font();
85    }
86    let Some(server) = state.server.get() else {
87        return gone();
88    };
89    let engine = &server.engine;
90    let id = match &wanted {
91        Wanted::Landing(id) | Wanted::Slash(id) | Wanted::Raw(id) | Wanted::Download(id) => {
92            id.clone()
93        }
94        Wanted::Font => unreachable!("answered above"),
95    };
96    let Ok(Some(open)) = engine.open_share(id.clone()).await else {
97        return gone();
98    };
99    match wanted {
100        Wanted::Slash(id) => redirect(&format!("../{id}")),
101        Wanted::Landing(_) => {
102            let kind = share_landing::Kind::of(&open.asset.media_type);
103            let content = if kind.reads_content() {
104                match engine.read_shared_asset(&open.asset).await {
105                    Ok(Some(bytes)) => Some(bytes),
106                    _ => return gone(),
107                }
108            } else {
109                None
110            };
111            if method == Method::GET {
112                engine.count_share_view(&id);
113            }
114            let page = share_landing::page(&open, kind, content.as_deref(), jiff::Timestamp::now());
115            let mut headers = common_headers();
116            headers.insert(
117                header::CONTENT_TYPE,
118                HeaderValue::from_static("text/html; charset=utf-8"),
119            );
120            headers.insert(
121                header::CONTENT_SECURITY_POLICY,
122                HeaderValue::from_static(share_landing::POLICY),
123            );
124            (StatusCode::OK, headers, page).into_response()
125        }
126        Wanted::Raw(_) | Wanted::Download(_) => {
127            let Ok(Some(bytes)) = engine.read_shared_asset(&open.asset).await else {
128                return gone();
129            };
130            let download = matches!(wanted, Wanted::Download(_))
131                || !share_landing::Kind::of(&open.asset.media_type).inline();
132            bytes_response(
133                request.headers(),
134                &open.asset.media_type,
135                &share_landing::file_name(&open.asset),
136                download,
137                bytes,
138            )
139        }
140        Wanted::Font => unreachable!("answered above"),
141    }
142}

144/// Serves only `/s/…`; anything else on the share-only listener is 404.

149/// The headers every answer carries.
150fn common_headers() -> HeaderMap {
151    let mut headers = HeaderMap::new();
152    headers.insert(header::CACHE_CONTROL, HeaderValue::from_static("no-store"));
153    headers.insert(
154        header::REFERRER_POLICY,
155        HeaderValue::from_static("no-referrer"),
156    );
157    headers.insert(
158        header::HeaderName::from_static("x-robots-tag"),
159        HeaderValue::from_static("noindex, nofollow"),
160    );
161    headers.insert(
162        header::X_CONTENT_TYPE_OPTIONS,
163        HeaderValue::from_static("nosniff"),
164    );
165    headers
166}
167
168/// The one answer to every link that does not work.
169fn gone() -> Response {
170    let mut headers = common_headers();
171    headers.insert(
172        header::CONTENT_TYPE,
173        HeaderValue::from_static("text/html; charset=utf-8"),
174    );
175    headers.insert(
176        header::CONTENT_SECURITY_POLICY,
177        HeaderValue::from_static(share_landing::POLICY),
178    );
179    (StatusCode::NOT_FOUND, headers, share_landing::GONE_PAGE).into_response()
180}
181
182fn redirect(location: &str) -> Response {
183    let mut headers = common_headers();
184    match HeaderValue::from_str(location) {
185        Ok(location) => {
186            headers.insert(header::LOCATION, location);
187            (StatusCode::PERMANENT_REDIRECT, headers).into_response()
188        }
189        Err(_) => gone(),
190    }
191}

193fn font() -> Response {
194    let Some(font) = cookie_agent_web_assets::get(BUNDLED_FONT) else {
195        return StatusCode::NOT_FOUND.into_response();
196    };
197    let mut headers = HeaderMap::new();
198    headers.insert(header::CONTENT_TYPE, HeaderValue::from_static("font/woff2"));
199    headers.insert(
200        header::CACHE_CONTROL,
201        HeaderValue::from_static("public, max-age=86400"),
202    );
203    headers.insert(
204        header::X_CONTENT_TYPE_OPTIONS,
205        HeaderValue::from_static("nosniff"),
206    );
207    (StatusCode::OK, headers, font.bytes).into_response()
208}

210/// The policy a shared asset's bytes are served with inline.
211fn inline_policy(media_type: &str) -> &'static str {
212    match media_type {
213        "text/html" => HTML_POLICY,
214        "image/svg+xml" => SVG_POLICY,
215        "application/pdf" => PDF_POLICY,
216        _ => PLAIN_POLICY,
217    }
218}

220/// The `Content-Type` a shared asset's bytes are served with: text as
221/// UTF-8 plain text, so a browser never runs what it is not meant to.
222fn served_type(media_type: &str) -> String {
223    match media_type {
224        "text/html" => "text/html; charset=utf-8".to_owned(),
225        "image/svg+xml" => media_type.to_owned(),
226        _ if share_landing::Kind::of(media_type) == share_landing::Kind::Text
227            || share_landing::Kind::of(media_type) == share_landing::Kind::Markdown =>
228        {
229            "text/plain; charset=utf-8".to_owned()
230        }
231        _ => media_type.to_owned(),
232    }
233}

235/// `Content-Disposition` for `name`, with an ASCII fallback and the UTF-8
236/// name (RFC 6266).
237fn disposition(kind: &str, name: &str) -> String {
238    let ascii = name
239        .chars()
240        .map(|character| {
241            if character.is_ascii_graphic() && character != '"' && character != '\\' {
242                character
243            } else if character == ' ' {
244                ' '
245            } else {
246                '_'
247            }
248        })
249        .collect::<String>();
250    format!(
251        "{kind}; filename=\"{ascii}\"; filename*=UTF-8''{}",
252        share_landing::percent_encode(name)
253    )
254}

256fn bytes_response(
257    request: &HeaderMap,
258    media_type: &str,
259    name: &str,
260    download: bool,
261    bytes: Vec<u8>,
262) -> Response {
263    let mut headers = common_headers();
264    let content_type = if download {
265        "application/octet-stream".to_owned()
266    } else {
267        served_type(media_type)
268    };
269    if let Ok(value) = HeaderValue::from_str(&content_type) {
270        headers.insert(header::CONTENT_TYPE, value);
271    }
272    if let Ok(value) = HeaderValue::from_str(&disposition(
273        if download { "attachment" } else { "inline" },
274        name,
275    )) {
276        headers.insert(header::CONTENT_DISPOSITION, value);
277    }
278    if !download {
279        headers.insert(
280            header::CONTENT_SECURITY_POLICY,
281            HeaderValue::from_static(inline_policy(media_type)),
282        );
283    }
284    headers.insert(header::ACCEPT_RANGES, HeaderValue::from_static("bytes"));
285    let total = bytes.len() as u64;
286    match request
287        .get(header::RANGE)
288        .and_then(|value| value.to_str().ok())
289        .map(|value| byte_range(value, total))
290    {
291        None => (StatusCode::OK, headers, Body::from(bytes)).into_response(),
292        Some(Some((start, end))) => {
293            if let Ok(value) = HeaderValue::from_str(&format!("bytes {start}-{end}/{total}")) {
294                headers.insert(header::CONTENT_RANGE, value);
295            }
296            let part = bytes[start as usize..=end as usize].to_vec();
297            (StatusCode::PARTIAL_CONTENT, headers, Body::from(part)).into_response()
298        }
299        Some(None) => {
300            if let Ok(value) = HeaderValue::from_str(&format!("bytes */{total}")) {
301                headers.insert(header::CONTENT_RANGE, value);
302            }
303            (StatusCode::RANGE_NOT_SATISFIABLE, headers).into_response()
304        }
305    }
306}

308/// The one range `bytes=a-b`, `bytes=a-`, or `bytes=-n` asks for of
309/// `total` bytes, inclusive; `None` when it cannot be satisfied. Several
310/// ranges are answered with the first.
311fn byte_range(value: &str, total: u64) -> Option<(u64, u64)> {
312    let spec = value.trim().strip_prefix("bytes=")?;
313    let first = spec.split(',').next()?.trim();
314    let (start, end) = first.split_once('-')?;
315    if total == 0 {
316        return None;
317    }
318    let last = total - 1;
319    let (start, end) = match (start.trim(), end.trim()) {
320        ("", suffix) => {
321            let length = suffix.parse::<u64>().ok().filter(|length| *length > 0)?;
322            (total.saturating_sub(length), last)
323        }
324        (start, "") => (start.parse().ok()?, last),
325        (start, end) => (start.parse().ok()?, end.parse::<u64>().ok()?.min(last)),
326    };
327    (start <= end && start <= last).then_some((start, end))
328}

339    #[test]
340    fn paths_name_a_link_and_what_of_it() {
341        let id = || cookie_agent_protocol::ShareId::new(ID).unwrap();
342        assert_eq!(wanted(&format!("/s/{ID}")), Some(Wanted::Landing(id())));
343        assert_eq!(wanted(&format!("/s/{ID}/")), Some(Wanted::Slash(id())));
344        assert_eq!(
345            wanted(&format!("/s/{ID}/raw/loss.html")),
346            Some(Wanted::Raw(id()))
347        );
348        assert_eq!(wanted(&format!("/s/{ID}/raw")), Some(Wanted::Raw(id())));
349        assert_eq!(
350            wanted(&format!("/s/{ID}/download/a/b.png")),
351            Some(Wanted::Download(id()))
352        );
353        assert_eq!(wanted("/s/_font/inter.woff2"), Some(Wanted::Font));
354        for path in [
355            "/s/",
356            "/s/short",
357            &format!("/s/{ID}/other/x"),
358            &format!("/s/{ID}x"),
359            "/plugins/x",
360        ] {
361            assert_eq!(wanted(path), None, "{path}");
362        }
363    }

365    #[test]
366    fn a_range_is_one_satisfiable_span() {
367        assert_eq!(byte_range("bytes=0-99", 1000), Some((0, 99)));
368        assert_eq!(byte_range("bytes=900-", 1000), Some((900, 999)));
369        assert_eq!(byte_range("bytes=-100", 1000), Some((900, 999)));
370        assert_eq!(byte_range("bytes=-2000", 1000), Some((0, 999)));
371        assert_eq!(byte_range("bytes=990-2000", 1000), Some((990, 999)));
372        assert_eq!(byte_range("bytes=0-0,5-9", 1000), Some((0, 0)));
373        assert_eq!(byte_range("bytes=1000-", 1000), None);
374        assert_eq!(byte_range("bytes=5-4", 1000), None);
375        assert_eq!(byte_range("bytes=0-", 0), None);
376        assert_eq!(byte_range("items=0-1", 1000), None);
377    }
