1//! Share links on the daemon's listeners (`docs/site/specs/share-links.md`
2//! section 5): `/s/<id>` is a link's landing page, `/s/<id>/raw/<name>` its
3//! asset's bytes inline, and `/s/<id>/download/<name>` the same as a
4//! download. No token, ticket, or origin is checked: the ID is the only
5//! key. Every failure answers the same 404 page, so nobody learns whether
6//! a link ever existed.
7//!
8//! A conversation share's `/s/<id>` is the guest page
9//! (`docs/site/specs/conversation-shares.md` section 6): the web client's own
10//! page, whose files are at `/s/_app/`, in a sandbox that gives it an opaque
11//! origin, so it cannot touch what the daemon's page keeps on this origin.
12

24const PREFIX: &str = "/s/";
25/// The landing page's font, below [`PREFIX`]: no ID is this short.
26pub(super) const FONT_PATH: &str = "_font/inter.woff2";
27/// Where the web bundle keeps the font.
28const BUNDLED_FONT: &str = "fonts/InterVariable.woff2";

30/// The policy of shared HTML: the viewer's, whose `sandbox` gives the page
31/// an opaque origin even when opened at the top level.
32const HTML_POLICY: &str = cookie_agent_web_assets::VIEWER_POLICY;
33/// The policy of shared SVG: sandboxed, and no scripts.
34const SVG_POLICY: &str = "sandbox; default-src 'none'; style-src 'unsafe-inline'; \
35     img-src data:; font-src data:; frame-ancestors 'self'";
36/// The policy of every other inline type but PDF, which the browser's
37/// viewer cannot show sandboxed.
38const PLAIN_POLICY: &str = "sandbox; default-src 'none'; img-src 'self'; media-src 'self'; \
39     style-src 'unsafe-inline'; frame-ancestors 'self'";
40const PDF_POLICY: &str = "frame-ancestors 'self'";

42/// The directory below `/s/` the guest page's files are in: no ID is this
43/// short.
44const APP_PREFIX: &str = "_app/";
45/// The sandbox of the guest page. Links it opens are ordinary tabs.
46const GUEST_SANDBOX: &str = "sandbox allow-scripts allow-forms allow-modals allow-downloads \
47     allow-popups allow-popups-to-escape-sandbox";

49/// What a path below `/s/` asks for.
50#[derive(Debug, Eq, PartialEq)]
51enum Wanted {
52    Landing(ShareId),
53    /// `/s/<id>/`: the landing page without its trailing slash.
54    Slash(ShareId),
55    Raw(ShareId),
56    Download(ShareId),
57    Font,
58    /// A file of the guest page.
59    App(String),
60    /// A guest's connection.
61    Socket(ShareId),
62}
63
64fn wanted(path: &str) -> Option<Wanted> {
65    let below = path.strip_prefix(PREFIX)?;
66    if below == FONT_PATH {
67        return Some(Wanted::Font);
68    }
69    if let Some(file) = below.strip_prefix(APP_PREFIX) {
70        return Some(Wanted::App(file.to_owned()));
71    }
72    let (id, rest) = match below.find('/') {
73        Some(slash) => (&below[..slash], &below[slash..]),
74        None => (below, ""),
75    };
76    let id = ShareId::new(id).ok()?;
77    match rest {
78        "" => Some(Wanted::Landing(id)),
79        "/" => Some(Wanted::Slash(id)),
80        rest => {
81            let (kind, _name) = match rest[1..].find('/') {
82                Some(slash) => (&rest[1..=slash], &rest[slash + 2..]),
83                None => (&rest[1..], ""),
84            };
85            match kind {
86                "raw" => Some(Wanted::Raw(id)),
87                "download" => Some(Wanted::Download(id)),
88                "ws" if rest == "/ws" => Some(Wanted::Socket(id)),
89                _ => None,
90            }
91        }
92    }
93}

95/// Serves a path below `/s/`.
96pub(crate) async fn share_pages(State(state): State<WebSocketState>, request: Request) -> Response {
97    let method = request.method().clone();
98    if method != Method::GET && method != Method::HEAD {
99        return gone();
100    }
101    let Some(wanted) = wanted(request.uri().path()) else {
102        return gone();
103    };
104    if wanted == Wanted::Font {
105        return font();
106    }
107    if let Wanted::App(file) = &wanted {
108        return app_file(file, request.headers());
109    }
110    if let Wanted::Socket(id) = wanted {
111        return super::guest::guest_upgrade(state, id, request).await;
112    }
113    let Some(server) = state.server.get() else {
114        return gone();
115    };
116    let engine = &server.engine;
117    let id = match &wanted {
118        Wanted::Landing(id) | Wanted::Slash(id) | Wanted::Raw(id) | Wanted::Download(id) => {
119            id.clone()
120        }
121        Wanted::Font | Wanted::App(_) | Wanted::Socket(_) => unreachable!("answered above"),
122    };
123    let open = match engine.open_share(id.clone()).await {
124        Ok(Some(OpenedLink::Asset(open))) => *open,
125        Ok(Some(OpenedLink::Session(link))) => {
126            return match wanted {
127                Wanted::Slash(id) => redirect(&format!("../{id}")),
128                Wanted::Landing(_) => guest_page(engine, &link),
129                _ => gone(),
130            };
131        }
132        _ => return gone(),
133    };
134    match wanted {
135        Wanted::Slash(id) => redirect(&format!("../{id}")),
136        Wanted::Landing(_) => {
137            let kind = share_landing::Kind::of(&open.asset.media_type);
138            let content = if kind.reads_content() {
139                match engine.read_shared_asset(&open.asset).await {
140                    Ok(Some(bytes)) => Some(bytes),
141                    _ => return gone(),
142                }
143            } else {
144                None
145            };
146            if method == Method::GET {
147                engine.count_share_view(&id);
148            }
149            let page = share_landing::page(&open, kind, content.as_deref(), jiff::Timestamp::now());
150            let mut headers = common_headers();
151            headers.insert(
152                header::CONTENT_TYPE,
153                HeaderValue::from_static("text/html; charset=utf-8"),
154            );
155            headers.insert(
156                header::CONTENT_SECURITY_POLICY,
157                HeaderValue::from_static(share_landing::POLICY),
158            );
159            (StatusCode::OK, headers, page).into_response()
160        }
161        Wanted::Raw(_) | Wanted::Download(_) => {
162            let Ok(Some(bytes)) = engine.read_shared_asset(&open.asset).await else {
163                return gone();
164            };
165            let download = matches!(wanted, Wanted::Download(_))
166                || !share_landing::Kind::of(&open.asset.media_type).inline();
167            bytes_response(
168                request.headers(),
169                &open.asset.media_type,
170                &share_landing::file_name(&open.asset),
171                download,
172                bytes,
173            )
174        }
175        Wanted::Font | Wanted::App(_) | Wanted::Socket(_) => unreachable!("answered above"),
176    }
177}

179/// A file of the guest page. Its page has an opaque origin, so every file it
180/// loads is another origin's and answers to any.
181fn app_file(file: &str, headers: &HeaderMap) -> Response {
182    // A frame of `viewer.html` is framed by the guest page, whose opaque
183    // origin is never `'self'`; nothing is in it until the page hands it over.
184    let framed = cookie_agent_web_assets::VIEWER_POLICY
185        .replace("frame-ancestors 'self'", "frame-ancestors *");
186    let policy = if file == cookie_agent_web_assets::VIEWER_PATH {
187        framed.as_str()
188    } else {
189        cookie_agent_web_assets::content_security_policy()
190    };
191    // `index.html` is the page itself, which is only served in its sandbox.
192    let served = (file != "index.html")
193        .then(|| super::page::file(file, headers, policy))
194        .flatten();
195    let Some(mut response) = served else {
196        return gone();
197    };
198    response.headers_mut().insert(
199        header::ACCESS_CONTROL_ALLOW_ORIGIN,
200        HeaderValue::from_static("*"),
201    );
202    response
203}

205/// The guest page: the web client's page, reading its files from
206/// [`APP_PREFIX`], in its sandbox. The first thing a link preview sees of it
207/// is the session's title.
208fn guest_page(engine: &Engine, link: &GuestLink) -> Response {
209    let Some(index) = cookie_agent_web_assets::get("index.html") else {
210        return gone();
211    };
212    let title = engine
213        .session_meta(link.session)
214        .ok()
215        .and_then(|meta| meta.title)
216        .map_or_else(
217            || "A shared conversation".to_owned(),
218            |title| title.as_str().to_owned(),
219        );
220    let page = guest_page_html(&String::from_utf8_lossy(index.bytes), &title);
221    let mut headers = common_headers();
222    headers.insert(
223        header::CONTENT_TYPE,
224        HeaderValue::from_static("text/html; charset=utf-8"),
225    );
226    if let Ok(policy) = HeaderValue::from_str(&guest_policy()) {
227        headers.insert(header::CONTENT_SECURITY_POLICY, policy);
228    }
229    headers.insert(
230        header::ACCESS_CONTROL_ALLOW_ORIGIN,
231        HeaderValue::from_static("*"),
232    );
233    (StatusCode::OK, headers, page).into_response()
234}

236/// The page's own policy with the sandbox, and a `<base>` for its files.
237fn guest_policy() -> String {
238    let policy = cookie_agent_web_assets::content_security_policy()
239        .replace("base-uri 'none'", "base-uri 'self'");
240    format!("{GUEST_SANDBOX}; {policy}")
241}

243/// `index` as the guest page: its relative URLs read below [`APP_PREFIX`],
244/// its stylesheet loads across origins as its integrity check needs, and it
245/// carries `title` for link previews.
246fn guest_page_html(index: &str, title: &str) -> String {
247    let title = share_landing::escape(title);
248    let head = format!(
249        "<head><base href=\"{APP_PREFIX}\" />\
250         <meta property=\"og:title\" content=\"{title}\" />\
251         <meta property=\"og:type\" content=\"website\" />\
252         <meta property=\"og:description\" content=\"A conversation shared from cookie agent\" />"
253    );
254    index
255        .replacen("<head>", &head, 1)
256        .replace(
257            "<link rel=\"stylesheet\"",
258            "<link crossorigin=\"anonymous\" rel=\"stylesheet\"",
259        )
260        .replacen(
261            "<title>Cookie Agent</title>",
262            &format!("<title>{title}</title>"),
263            1,
264        )
265}

272/// The headers every answer carries.
273fn common_headers() -> HeaderMap {
274    let mut headers = HeaderMap::new();
275    headers.insert(header::CACHE_CONTROL, HeaderValue::from_static("no-store"));
276    headers.insert(
277        header::REFERRER_POLICY,
278        HeaderValue::from_static("no-referrer"),
279    );
280    headers.insert(
281        header::HeaderName::from_static("x-robots-tag"),
282        HeaderValue::from_static("noindex, nofollow"),
283    );
284    headers.insert(
285        header::X_CONTENT_TYPE_OPTIONS,
286        HeaderValue::from_static("nosniff"),
287    );
288    headers
289}

291/// The one answer to every link that does not work. Any page may read it: a
292/// guest page, whose origin is opaque, asks after its own link.
293fn gone() -> Response {
294    let mut headers = common_headers();
295    headers.insert(
296        header::ACCESS_CONTROL_ALLOW_ORIGIN,
297        HeaderValue::from_static("*"),
298    );
299    headers.insert(
300        header::CONTENT_TYPE,
301        HeaderValue::from_static("text/html; charset=utf-8"),
302    );
303    headers.insert(
304        header::CONTENT_SECURITY_POLICY,
305        HeaderValue::from_static(share_landing::POLICY),
306    );
307    (StatusCode::NOT_FOUND, headers, share_landing::GONE_PAGE).into_response()
308}

310fn redirect(location: &str) -> Response {
311    let mut headers = common_headers();
312    match HeaderValue::from_str(location) {
313        Ok(location) => {
314            headers.insert(header::LOCATION, location);
315            (StatusCode::PERMANENT_REDIRECT, headers).into_response()
316        }
317        Err(_) => gone(),
318    }
319}

321fn font() -> Response {
322    let Some(font) = cookie_agent_web_assets::get(BUNDLED_FONT) else {
323        return StatusCode::NOT_FOUND.into_response();
324    };
325    let mut headers = HeaderMap::new();
326    headers.insert(header::CONTENT_TYPE, HeaderValue::from_static("font/woff2"));
327    headers.insert(
328        header::CACHE_CONTROL,
329        HeaderValue::from_static("public, max-age=86400"),
330    );
331    headers.insert(
332        header::X_CONTENT_TYPE_OPTIONS,
333        HeaderValue::from_static("nosniff"),
334    );
335    (StatusCode::OK, headers, font.bytes).into_response()
336}

338/// The policy a shared asset's bytes are served with inline.
339fn inline_policy(media_type: &str) -> &'static str {
340    match media_type {
341        "text/html" => HTML_POLICY,
342        "image/svg+xml" => SVG_POLICY,
343        "application/pdf" => PDF_POLICY,
344        _ => PLAIN_POLICY,
345    }
346}

348/// The `Content-Type` a shared asset's bytes are served with: text as
349/// UTF-8 plain text, so a browser never runs what it is not meant to.
350fn served_type(media_type: &str) -> String {
351    match media_type {
352        "text/html" => "text/html; charset=utf-8".to_owned(),
353        "image/svg+xml" => media_type.to_owned(),
354        _ if share_landing::Kind::of(media_type) == share_landing::Kind::Text
355            || share_landing::Kind::of(media_type) == share_landing::Kind::Markdown =>
356        {
357            "text/plain; charset=utf-8".to_owned()
358        }
359        _ => media_type.to_owned(),
360    }
361}

363/// `Content-Disposition` for `name`, with an ASCII fallback and the UTF-8
364/// name (RFC 6266).
365fn disposition(kind: &str, name: &str) -> String {
366    let ascii = name
367        .chars()
368        .map(|character| {
369            if character.is_ascii_graphic() && character != '"' && character != '\\' {
370                character
371            } else if character == ' ' {
372                ' '
373            } else {
374                '_'
375            }
376        })
377        .collect::<String>();
378    format!(
379        "{kind}; filename=\"{ascii}\"; filename*=UTF-8''{}",
380        share_landing::percent_encode(name)
381    )
382}

384fn bytes_response(
385    request: &HeaderMap,
386    media_type: &str,
387    name: &str,
388    download: bool,
389    bytes: Vec<u8>,
390) -> Response {
391    let mut headers = common_headers();
392    let content_type = if download {
393        "application/octet-stream".to_owned()
394    } else {
395        served_type(media_type)
396    };
397    if let Ok(value) = HeaderValue::from_str(&content_type) {
398        headers.insert(header::CONTENT_TYPE, value);
399    }
400    if let Ok(value) = HeaderValue::from_str(&disposition(
401        if download { "attachment" } else { "inline" },
402        name,
403    )) {
404        headers.insert(header::CONTENT_DISPOSITION, value);
405    }
406    if !download {
407        headers.insert(
408            header::CONTENT_SECURITY_POLICY,
409            HeaderValue::from_static(inline_policy(media_type)),
410        );
411    }
412    headers.insert(header::ACCEPT_RANGES, HeaderValue::from_static("bytes"));
413    let total = bytes.len() as u64;
414    match request
415        .get(header::RANGE)
416        .and_then(|value| value.to_str().ok())
417        .map(|value| byte_range(value, total))
418    {
419        None => (StatusCode::OK, headers, Body::from(bytes)).into_response(),
420        Some(Some((start, end))) => {
421            if let Ok(value) = HeaderValue::from_str(&format!("bytes {start}-{end}/{total}")) {
422                headers.insert(header::CONTENT_RANGE, value);
423            }
424            let part = bytes[start as usize..=end as usize].to_vec();
425            (StatusCode::PARTIAL_CONTENT, headers, Body::from(part)).into_response()
426        }
427        Some(None) => {
428            if let Ok(value) = HeaderValue::from_str(&format!("bytes */{total}")) {
429                headers.insert(header::CONTENT_RANGE, value);
430            }
431            (StatusCode::RANGE_NOT_SATISFIABLE, headers).into_response()
432        }
433    }
434}

436/// The one range `bytes=a-b`, `bytes=a-`, or `bytes=-n` asks for of
437/// `total` bytes, inclusive; `None` when it cannot be satisfied. Several
438/// ranges are answered with the first.
439fn byte_range(value: &str, total: u64) -> Option<(u64, u64)> {
440    let spec = value.trim().strip_prefix("bytes=")?;
441    let first = spec.split(',').next()?.trim();
442    let (start, end) = first.split_once('-')?;
443    if total == 0 {
444        return None;
445    }
446    let last = total - 1;
447    let (start, end) = match (start.trim(), end.trim()) {
448        ("", suffix) => {
449            let length = suffix.parse::<u64>().ok().filter(|length| *length > 0)?;
450            (total.saturating_sub(length), last)
451        }
452        (start, "") => (start.parse().ok()?, last),
453        (start, end) => (start.parse().ok()?, end.parse::<u64>().ok()?.min(last)),
454    };
455    (start <= end && start <= last).then_some((start, end))
456}
