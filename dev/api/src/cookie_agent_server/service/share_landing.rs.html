25/// How the page previews an asset.
26#[derive(Clone, Copy, Debug, Eq, PartialEq)]
27pub(super) enum Kind {
28    Html,
29    Image,
30    Video,
31    Audio,
32    Pdf,
33    Markdown,
34    Text,
35    File,
36}
37
38impl Kind {
39    pub(super) fn of(media_type: &str) -> Self {
40        let essence = media_type
41            .split(';')
42            .next()
43            .unwrap_or_default()
44            .trim()
45            .to_ascii_lowercase();
46        match essence.as_str() {
47            "text/html" => Self::Html,
48            "text/markdown" => Self::Markdown,
49            "application/pdf" => Self::Pdf,
50            "application/json"
51            | "application/xml"
52            | "application/x-yaml"
53            | "application/yaml"
54            | "application/toml"
55            | "application/javascript" => Self::Text,
56            other if other.starts_with("image/") => Self::Image,
57            other if other.starts_with("video/") => Self::Video,
58            other if other.starts_with("audio/") => Self::Audio,
59            other if other.starts_with("text/") => Self::Text,
60            _ => Self::File,
61        }
62    }
63
64    /// Whether the page shows the content itself, read by the daemon.
65    pub(super) const fn reads_content(self) -> bool {
66        matches!(self, Self::Markdown | Self::Text)
67    }
68
69    /// Whether `raw` shows the bytes in the browser rather than downloading
70    /// them.
71    pub(super) const fn inline(self) -> bool {
72        !matches!(self, Self::File)
73    }
74
75    /// Whether the page offers *Open full page*.
76    const fn opens(self) -> bool {
77        matches!(self, Self::Html | Self::Image | Self::Video | Self::Pdf)
78    }
79
80    fn label(self, media_type: &str) -> String {
81        match self {
82            Self::Html => "HTML page".to_owned(),
83            Self::Markdown => "Markdown".to_owned(),
84            Self::Pdf => "PDF".to_owned(),
85            Self::Image if media_type == "image/svg+xml" => "SVG image".to_owned(),
86            Self::Image | Self::Video | Self::Audio => {
87                let (_, subtype) = media_type.split_once('/').unwrap_or(("", media_type));
88                let noun = match self {
89                    Self::Image => "image",
90                    Self::Video => "video",
91                    _ => "audio",
92                };
93                format!("{} {noun}", subtype.to_ascii_uppercase())
94            }
95            Self::Text | Self::File => media_type.to_owned(),
96        }
97    }
98}

100/// `&`, `<`, `>`, `"`, and `'` escaped for HTML text and attributes.
101pub(super) fn escape(text: &str) -> String {
102    let mut escaped = String::with_capacity(text.len());
103    for character in text.chars() {
104        match character {
105            '&' => escaped.push_str("&amp;"),
106            '<' => escaped.push_str("&lt;"),
107            '>' => escaped.push_str("&gt;"),
108            '"' => escaped.push_str("&quot;"),
109            '\'' => escaped.push_str("&#39;"),
110            other => escaped.push(other),
111        }
112    }
113    escaped
114}

116/// `text` percent-encoded as one URL path segment: everything but
117/// unreserved characters.
118pub(super) fn percent_encode(text: &str) -> String {
119    text.bytes().fold(String::new(), |mut encoded, byte| {
120        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
121            encoded.push(byte as char);
122        } else {
123            let _ = write!(encoded, "%{byte:02X}");
124        }
125        encoded
126    })
127}

152/// The name the asset is downloaded as: the file it was written from, or
153/// its title slugged with its type's extension.
154pub(crate) fn file_name(asset: &Asset) -> String {
155    if let Some(name) = asset.filename.as_deref().filter(|name| !name.is_empty()) {
156        return name.to_owned();
157    }
158    let mut slug = String::new();
159    for character in asset.title.chars() {
160        if character.is_alphanumeric() {
161            slug.extend(character.to_lowercase());
162        } else if !slug.ends_with('-') && !slug.is_empty() {
163            slug.push('-');
164        }
165    }
166    let slug = slug.trim_end_matches('-');
167    let slug = if slug.is_empty() { "asset" } else { slug };
168    format!("{slug}.{}", extension(&asset.media_type))
169}

171/// "in 7 days", "in 5 hours", or "in a few minutes", each rounded to the
172/// nearest, so a 7-day link opened a minute after it was made says 7 days;
173/// from a day and a half on it counts days, from 23½ hours it is a day.
174fn time_left(now: jiff::Timestamp, at: jiff::Timestamp) -> String {
175    const HOUR: i64 = 3600;
176    let seconds = at.as_second() - now.as_second();
177    let hours = (seconds + HOUR / 2) / HOUR;
178    if seconds >= 36 * HOUR {
179        format!("in {} days", (seconds + 12 * HOUR) / (24 * HOUR))
180    } else if seconds >= 23 * HOUR + HOUR / 2 {
181        "in 1 day".to_owned()
182    } else if hours >= 2 {
183        format!("in {hours} hours")
184    } else if hours == 1 {
185        "in 1 hour".to_owned()
186    } else {
187        "in a few minutes".to_owned()
188    }
189}

191/// Markdown rendered to HTML, with raw HTML in the source escaped as text.
192fn markdown(source: &str) -> String {
193    use pulldown_cmark::{Event, Options, Parser, html};
194
195    let options = Options::ENABLE_TABLES
196        | Options::ENABLE_STRIKETHROUGH
197        | Options::ENABLE_TASKLISTS
198        | Options::ENABLE_FOOTNOTES;
199    let events = Parser::new_ext(source, options).map(|event| match event {
200        Event::Html(text) | Event::InlineHtml(text) => Event::Text(text),
201        other => other,
202    });
203    let mut rendered = String::new();
204    html::push_html(&mut rendered, events);
205    rendered
206}

208/// At most [`TEXT_PREVIEW_BYTES`] of `bytes` as text, cut at a character,
209/// and whether it was cut; `None` when it is not UTF-8.
210fn text_preview(bytes: &[u8]) -> Option<(&str, bool)> {
211    let cut = bytes.len() > TEXT_PREVIEW_BYTES;
212    let mut end = bytes.len().min(TEXT_PREVIEW_BYTES);
213    loop {
214        match std::str::from_utf8(&bytes[..end]) {
215            Ok(text) => return Some((text, cut)),
216            // A character split by the cut: step back to its start.
217            Err(error) if cut && error.error_len().is_none() => end = error.valid_up_to(),
218            Err(_) => return None,
219        }
220    }
221}

323/// The landing page of `open` at `now`; `content` is the asset's bytes
324/// when its kind [reads content](Kind::reads_content).
325pub(super) fn page(
326    open: &OpenShare,
327    kind: Kind,
328    content: Option<&[u8]>,
329    now: jiff::Timestamp,
330) -> String {
331    let share = &open.share;
332    let asset = &open.asset;
333    let name = file_name(asset);
334    let encoded = percent_encode(&name);
335    let raw = format!("{}/raw/{encoded}", share.id);
336    let download = format!("{}/download/{encoded}", share.id);
337    let title = escape(&share.title);
338
339    let mut meta = vec![
340        kind.label(&asset.media_type),
341        human_bytes(asset.byte_length),
342    ];
343    if name != share.title {
344        meta.push(name.clone());
345    }
346    let meta = meta
347        .iter()
348        .map(|part| format!("<span>{}</span>", escape(part)))
349        .collect::<String>();
350
351    let open_button = if kind.opens() {
352        format!(
353            r#"<a class="btn" href="{raw}" target="_blank" rel="noopener noreferrer">{OPEN_ICON}Open full page</a>"#
354        )
355    } else {
356        String::new()
357    };
358    let download_button =
359        format!(r#"<a class="btn primary" href="{download}" download>{DOWNLOAD_ICON}Download</a>"#);
360
361    let preview = match kind {
362        Kind::Html => format!(
363            r#"<div class="bar"><span>Interactive page · runs sandboxed</span><a href="{}" rel="noreferrer">Reload</a></div><iframe class="frame" src="{raw}" sandbox="allow-scripts allow-popups" referrerpolicy="no-referrer" title="{title}"></iframe>"#,
364            share.id
365        ),
366        Kind::Image => format!(
367            r#"<div class="img"><a href="{raw}" target="_blank" rel="noopener noreferrer"><img src="{raw}" alt="{title}"></a></div>"#
368        ),
369        Kind::Video => {
370            format!(r#"<video class="media" src="{raw}" controls preload="metadata"></video>"#)
371        }
372        Kind::Audio => format!(
373            r#"<div class="audio"><audio src="{raw}" controls preload="metadata"></audio></div>"#
374        ),
375        Kind::Pdf => format!(r#"<iframe class="frame" src="{raw}" title="{title}"></iframe>"#),
376        Kind::Markdown => match content.and_then(text_preview) {
377            Some((text, cut)) => format!(
378                r#"<article class="doc">{}</article>{}"#,
379                markdown(text),
380                cut_note(cut)
381            ),
382            None => file_card(asset, &name, &download),
383        },
384        Kind::Text => match content.and_then(text_preview) {
385            Some((text, cut)) => format!(
386                r#"<pre class="text">{}</pre>{}"#,
387                escape(text),
388                cut_note(cut)
389            ),
390            None => file_card(asset, &name, &download),
391        },
392        Kind::File => file_card(asset, &name, &download),
393    };
394
395    let expiry = share.expires_at.map_or_else(String::new, |at| {
396        let date = at
397            .to_zoned(jiff::tz::TimeZone::UTC)
398            .strftime("%b %-d, %Y")
399            .to_string();
400        let exact = at
401            .to_zoned(jiff::tz::TimeZone::UTC)
402            .strftime("%b %-d, %Y, %H:%M UTC")
403            .to_string();
404        format!(
405            r#"<span class="exp" title="Expires {exact}">Link expires {} · {date}</span>"#,
406            time_left(now, at)
407        )
408    });
409
410    let og_image = match (kind, &share.url) {
411        (Kind::Image, Some(url)) => format!(
412            r#"<meta property="og:image" content="{}">"#,
413            escape(&format!("{url}/raw/{encoded}"))
414        ),
415        _ => String::new(),
416    };
417
418    format!(
419        r#"<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1"><meta name="robots" content="noindex, nofollow"><meta name="referrer" content="no-referrer"><title>{title}</title><meta property="og:title" content="{title}"><meta property="og:type" content="website">{og_image}<style>{style}</style></head><body><main class="page"><div class="brand">{MARK}cookie agent</div><header class="head"><div><h1>{title}</h1><div class="meta">{meta}</div></div><div class="actions">{open_button}{download_button}</div></header><section class="card">{preview}</section><footer class="foot"><span>Shared from cookie agent</span>{expiry}</footer></main></body></html>"#,
420        style = style(),
421    )
422}

435fn file_card(asset: &Asset, name: &str, download: &str) -> String {
436    let extension = name
437        .rsplit_once('.')
438        .map(|(_, extension)| extension)
439        .filter(|extension| !extension.is_empty() && extension.len() <= 5)
440        .unwrap_or("FILE")
441        .to_ascii_uppercase();
442    format!(

444        escape(&extension),
