32#[derive(Clone)]
33pub struct Server {

45/// A daemon's listener, which serves clients from before the daemon has
46/// its [`Server`]: while it starts, clients connect and see its status
47/// (`docs/site/specs/daemon-startup.md`), and once [`Self::hand_over`]
48/// gives it the server, the same connections are served by it.
49#[derive(Clone)]
50pub struct DaemonListener {
51    pub(crate) shared: Arc<Shared>,
52    /// Set once the daemon is ready to serve. The server does not hold it,
53    /// so the two do not keep each other alive.
54    pub(crate) server: ServerSlot,
55}
56
57/// Where a listener finds the server it serves.
58pub(crate) type ServerSlot = Arc<std::sync::OnceLock<Arc<Server>>>;
59
60/// What a daemon's listener and its server share.
61pub(crate) struct Shared {
62    pub(crate) board: StatusBoard,
63    pub(crate) shutdown: CancellationToken,
64    /// Set when the daemon shuts down to restart: client connections then
65    /// close with code 1012 and the reason `restarting`.
66    pub(crate) restarting: Arc<AtomicBool>,
67    /// Every WebSocket client connection holds a receiver of this channel
68    /// until it has closed, so a shutdown can wait for each to send its
69    /// close frame ([`RunningServer::wait`]).
70    pub(crate) connections: Arc<tokio::sync::watch::Sender<()>>,
71    /// The daemon's `[server] page_url`, which handshakes report while the
72    /// daemon starts; once ready, the server reads its configuration.
73    pub(crate) page_url: std::sync::RwLock<Option<String>>,
74    /// The page ticket of the token the daemon's listener serves with,
75    /// which handshakes report once the daemon is ready.
76    pub(crate) page_ticket: std::sync::RwLock<Option<String>>,
77    /// The daemon's `[server] runtime_token`, which `/runtime` takes as
78    /// well as the client token, and `/ws` never
79    /// (`docs/site/specs/runtime-token.md`).
80    pub(crate) runtime_token: std::sync::OnceLock<Zeroizing<String>>,
81    /// The base of the daemon's share links, `[public] url` or `[server]
82    /// page_url`, which handshakes report while the daemon starts.
83    pub(crate) share_url: std::sync::RwLock<Option<String>>,
84}

86impl DaemonListener {
87    /// A listener reporting `board`, the status the daemon's engine opens
88    /// with.
89    #[must_use]
90    pub fn new(board: StatusBoard) -> Self {
91        Self {
92            server: Arc::new(std::sync::OnceLock::new()),
93            shared: Arc::new(Shared {
94                board,
95                shutdown: CancellationToken::new(),
96                restarting: Arc::new(AtomicBool::new(false)),
97                connections: Arc::new(tokio::sync::watch::Sender::new(())),
98                page_url: std::sync::RwLock::new(None),
99                page_ticket: std::sync::RwLock::new(None),
100                runtime_token: std::sync::OnceLock::new(),
101                share_url: std::sync::RwLock::new(None),
102            }),
103        }
104    }
105
106    /// Reports `page_url`, the daemon's `[server] page_url`, in the
107    /// handshakes of clients that connect while it starts.
108    #[must_use]
109    pub fn with_page_url(self, page_url: Option<String>) -> Self {
110        *self
111            .shared
112            .page_url
113            .write()
114            .unwrap_or_else(|poisoned| poisoned.into_inner()) = page_url;
115        self
116    }
117
118    /// Lets dial-in runtimes present `runtime_token` on `/runtime`, as
119    /// well as the client token.
120    #[must_use]
121    pub fn with_runtime_token(self, runtime_token: Zeroizing<String>) -> Self {
122        let _ = self.shared.runtime_token.set(runtime_token);
123        self
124    }
125
126    /// Reports `share_url`, the base of the daemon's share links, in the
127    /// handshakes of clients that connect while it starts.
128    #[must_use]
129    pub fn with_share_url(self, share_url: Option<String>) -> Self {
130        *self
131            .shared
132            .share_url
133            .write()
134            .unwrap_or_else(|poisoned| poisoned.into_inner()) = share_url;
135        self
136    }
137
138    /// The daemon's status.
139    #[must_use]
140    pub fn board(&self) -> &StatusBoard {
141        &self.shared.board
142    }
143
144    /// The server for `engine` that this listener hands over to: it shares
145    /// the listener's shutdown.
146    #[must_use]
147    pub fn server(&self, engine: Engine) -> Server {
148        Server::with_shared(engine, Arc::clone(&self.shared))
149    }
150
151    /// Serves `server` from now on, and says the daemon is ready.
152    pub fn hand_over(&self, server: Arc<Server>) {
153        let _ = self.server.set(server);
154        self.shared.board.ready();
155    }
156
157    /// Says startup failed with `reason`, then waits until every client
158    /// connected meanwhile has been told and has closed, for at most
159    /// `wait`.
160    pub async fn fail(&self, reason: &str, wait: Duration) {
161        self.shared.board.fail(
162            cookie_agent_protocol::errors::DAEMON_START_FAILED
163                .error()
164                .detail(reason)
165                .build(),
166        );
167        let _ = tokio::time::timeout(wait, self.shared.connections.closed()).await;
168        self.shared.shutdown.cancel();
169    }
170
171    /// Stops listening.
172    pub fn shutdown(&self) {
173        self.shared.shutdown.cancel();
174    }
175}

177impl Server {
178    /// A server for `engine`. Within a Tokio runtime it starts sending the
179    /// engine's push notifications at once, until [`Self::shutdown`].
180    #[must_use]
181    pub fn new(engine: Engine) -> Self {
182        let board = engine.status_board().clone();
183        let listener = DaemonListener::new(board);
184        Self::with_shared(engine, listener.shared)
185    }
186
187    fn with_shared(engine: Engine, shared: Arc<Shared>) -> Self {
188        let push = crate::push::PushService::new(engine.clone());
189        if tokio::runtime::Handle::try_current().is_ok() {
190            push.start(shared.shutdown.clone());
191        }
192        Self {
193            engine,
194            shutdown: shared.shutdown.clone(),
195            restarting: Arc::clone(&shared.restarting),
196            shared,
197            push,
198        }
199    }
200
201    /// Stops serving: the daemon is `stopping`.
202    pub fn shutdown(&self) {
203        self.engine.status_board().stopping(false);
204        self.shutdown.cancel();
205    }
206
207    /// Tells the devices with push notifications that the daemon goes down
208    /// to restart, but the one that asked; waits a few seconds at most.
209    pub async fn announce_restarting(&self) {
210        self.push
211            .announce_restarting(std::time::Duration::from_secs(5))
212            .await;
213    }
214
215    /// Tells the devices with push notifications that the persistent daemon
216    /// is up: restarted, or started. Sends in the background.
217    pub fn announce_started(&self) {
218        let push = Arc::clone(&self.push);
219        tokio::spawn(async move {
220            push.announce_started(std::time::Duration::from_secs(30))
221                .await;
222        });
223    }
224
225    /// Shuts down to restart: as [`Self::shutdown`], but client connections
226    /// close saying the daemon is restarting
227    /// ([`cookie_agent_protocol::RESTART_CLOSE_CODE`]).
228    pub fn shutdown_for_restart(&self) {
229        self.engine.status_board().stopping(true);
230        self.restarting.store(true, Ordering::Release);
231        self.shutdown.cancel();
232    }
233
234    /// Serves one client connection that authenticates as `authentication`
235    /// says.
236    pub async fn serve_stream<S>(
237        self: Arc<Self>,
238        stream: S,
239        authentication: Authentication,
240    ) -> Result<(), TransportError>
241    where
242        S: MessageStream,
243    {
244        let connection_shutdown = self.shutdown.child_token();
245        cookie_agent_protocol_transport::serve(self, stream, connection_shutdown, authentication)
246            .await
247    }
248
249    /// Serves one in-process connection on a spawned task and returns the
250    /// protocol client attached to it. The session ends with the server's
251    /// shutdown or when the client is dropped.
252    pub fn connect_in_process(self: Arc<Self>) -> Client {
253        let (client, service) = in_process_pair(128);
254        tokio::spawn(async move {
255            let _ = self
256                .serve_stream(service, Authentication::Authenticated)
257                .await;
258        });
259        Client::connect_stream(client)
260    }
261}

263/// How long [`RunningServer::wait`] waits for client connections to close
264/// once the listener stopped. Each gets the protocol's shutdown grace (one
265/// second) to send what it queued and its close frame; a client slower than
266/// this loses its close frame rather than holding up a stop or a restart.
267const CONNECTIONS_CLOSE_WAIT: Duration = Duration::from_secs(2);
268
269pub struct RunningServer {
270    pub(super) address: SocketAddr,
271    pub(super) task: JoinHandle<()>,
272    pub(super) connections: Arc<tokio::sync::watch::Sender<()>>,
273}
274
275impl RunningServer {
276    #[must_use]
277    pub const fn address(&self) -> SocketAddr {
278        self.address
279    }
280
281    /// Waits for the listener to stop after a shutdown, then, for up to
282    /// `CONNECTIONS_CLOSE_WAIT`, for every client connection to close:
283    /// each has then written its close frame, which says why it closed (the
284    /// daemon restarting, say), and the client answered it or its grace ran
285    /// out. Without this, a daemon exiting or re-executing right after
286    /// would cut connections before their close frames left.
287    pub async fn wait(self) {
288        let _ = self.task.await;
289        let _ = tokio::time::timeout(CONNECTIONS_CLOSE_WAIT, self.connections.closed()).await;
290    }
291}
