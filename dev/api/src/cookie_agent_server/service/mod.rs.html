25/// Protocol service composed with one coherent engine runtime.
26#[derive(Clone)]
27pub struct Server {
28    pub(crate) engine: Engine,
29    pub(crate) shutdown: CancellationToken,
30    /// Set when the daemon shuts down to restart: client connections then
31    /// close with code 1012 and the reason `restarting`.
32    pub(crate) restarting: Arc<AtomicBool>,
33    /// Every WebSocket client connection holds a receiver of this channel
34    /// until it has closed, so a shutdown can wait for each to send its
35    /// close frame ([`RunningServer::wait`]).
36    pub(crate) connections: Arc<tokio::sync::watch::Sender<()>>,
37    /// Web Push notifications to the user's browsers.
38    pub(crate) push: Arc<crate::push::PushService>,
39}
40
41impl Server {
42    /// A server for `engine`. Within a Tokio runtime it starts sending the
43    /// engine's push notifications at once, until [`Self::shutdown`].
44    #[must_use]
45    pub fn new(engine: Engine) -> Self {
46        let shutdown = CancellationToken::new();
47        let push = crate::push::PushService::new(engine.clone());
48        if tokio::runtime::Handle::try_current().is_ok() {
49            push.start(shutdown.clone());
50        }
51        Self {
52            engine,
53            shutdown,
54            restarting: Arc::new(AtomicBool::new(false)),
55            connections: Arc::new(tokio::sync::watch::Sender::new(())),
56            push,
57        }
58    }
59
60    pub fn shutdown(&self) {
61        self.shutdown.cancel();
62    }
63
64    /// Tells the devices with push notifications that the daemon goes down
65    /// to restart, but the one that asked; waits a few seconds at most.
66    pub async fn announce_restarting(&self) {
67        self.push
68            .announce_restarting(std::time::Duration::from_secs(5))
69            .await;
70    }
71
72    /// Tells the devices with push notifications that the persistent daemon
73    /// is up: restarted, or started. Sends in the background.
74    pub fn announce_started(&self) {
75        let push = Arc::clone(&self.push);
76        tokio::spawn(async move {
77            push.announce_started(std::time::Duration::from_secs(30))
78                .await;
79        });
80    }
81
82    /// Shuts down to restart: as [`Self::shutdown`], but client connections
83    /// close saying the daemon is restarting
84    /// ([`cookie_agent_protocol::RESTART_CLOSE_CODE`]).
85    pub fn shutdown_for_restart(&self) {
86        self.restarting.store(true, Ordering::Release);
87        self.shutdown.cancel();
88    }
89
90    /// Serves one client connection that authenticates as `authentication`
91    /// says.
92    pub async fn serve_stream<S>(
93        self: Arc<Self>,
94        stream: S,
95        authentication: Authentication,
96    ) -> Result<(), TransportError>
97    where
98        S: MessageStream,
99    {
100        let connection_shutdown = self.shutdown.child_token();
101        cookie_agent_protocol_transport::serve(self, stream, connection_shutdown, authentication)
102            .await
103    }
104
105    /// Serves one in-process connection on a spawned task and returns the
106    /// protocol client attached to it. The session ends with the server's
107    /// shutdown or when the client is dropped.
108    pub fn connect_in_process(self: Arc<Self>) -> Client {
109        let (client, service) = in_process_pair(128);
110        tokio::spawn(async move {
111            let _ = self
112                .serve_stream(service, Authentication::Authenticated)
113                .await;
114        });
115        Client::connect_stream(client)
116    }
117}

119/// How long [`RunningServer::wait`] waits for client connections to close
120/// once the listener stopped. Each gets the protocol's shutdown grace (one
121/// second) to send what it queued and its close frame; a client slower than
122/// this loses its close frame rather than holding up a stop or a restart.
123const CONNECTIONS_CLOSE_WAIT: Duration = Duration::from_secs(2);
124
125pub struct RunningServer {
126    pub(super) address: SocketAddr,
127    pub(super) task: JoinHandle<()>,
128    pub(super) connections: Arc<tokio::sync::watch::Sender<()>>,
129}
130
131impl RunningServer {
132    #[must_use]
133    pub const fn address(&self) -> SocketAddr {
134        self.address
135    }
136
137    /// Waits for the listener to stop after a shutdown, then, for up to
138    /// `CONNECTIONS_CLOSE_WAIT`, for every client connection to close:
139    /// each has then written its close frame, which says why it closed (the
140    /// daemon restarting, say), and the client answered it or its grace ran
141    /// out. Without this, a daemon exiting or re-executing right after
142    /// would cut connections before their close frames left.
143    pub async fn wait(self) {
144        let _ = self.task.await;
145        let _ = tokio::time::timeout(CONNECTIONS_CLOSE_WAIT, self.connections.closed()).await;
146    }
147}
